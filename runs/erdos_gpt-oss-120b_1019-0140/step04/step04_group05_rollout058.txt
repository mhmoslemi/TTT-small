def compute_gradient_matrix(h):
    n = h.size
    G = np.empty((2*n-1, n), dtype=float)
    # k from -(n-1) to (n-1)
    for idx, k in enumerate(range(-(n-1), n)):
        # forward shift: h_{j+k}
        if k >= 0:
            forward = np.concatenate([h[k:], np.zeros(k, dtype=float)])
        else:
            forward = np.concatenate([np.zeros(-k, dtype=float), h[:n+k]])

    return G

def compute_gradient_dx(h, dx):
    n = h.shape[0]
    M = 2*n -1
    Gdx = np.empty((M, n), dtype=float)
    # shift k from -(n-1) to n-1
    for idx, k in enumerate(range(-(n-1), n)):
        if k > 0:
            forward = np.concatenate([h[k:], np.zeros(k, dtype=float)])
            backward = np.concatenate([np.zeros(k, dtype=float), h[:n-k]])
        elif k < 0:
            p = -k
            forward = np.concatenate([np.zeros(p, dtype=float), h[:n-p]])
            backward = np.concatenate([h[p:], np.zeros(p, dtype=float)])
        else:  # k==0
            forward = h
            backward = h
        Gdx[idx] = (1.0 - forward - backward) * dx
    return Gdx

def simulated_annealing(h0, budget_s, rng):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < budget_s:
        # choose i,j randomly
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        # temperature schedule: linear decay
        elapsed = time.time() - start
        temp = 0.02 * (1.0 - elapsed / budget_s)
        if temp < 1e-6:
            temp = 1e-6
        delta = max_delta * rng.random() * (temp / 0.02)  # scale with temp
        if delta <= 0.0:
            continue
        # apply move
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # acceptance
            if rng.random() < np.exp(-(new_score - cur_score) / temp):
                cur_score = new_score
            else:
                # revert
                h[i] += delta
                h[j] -= delta
    return best_h, best_score

def lp_optimize_within_time(h0, time_limit, rng):
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    cur_c5 = compute_c5(h)
    start = time.time()
    # limit number of LP iterations to avoid endless loops
    max_iters = 30
    for it in range(max_iters):
        if time.time() - start >= time_limit:
            break
        # compute gradient matrix
        Gdx = compute_gradient_dx(h, dx)
        # compute corr vector
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        # build LP
        M = Gdx.shape[0]
        # A_ub
        A_ub = np.hstack([Gdx, -np.ones((M, 1), dtype=float)])
        b_ub = -corr
        # objective
        c = np.zeros(n + 1, dtype=float)
        c[-1] = 1.0
        # bounds for delta
        bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(None, None)]
        # equality sum(delta) = 0
        A_eq = np.zeros((1, n + 1), dtype=float)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0], dtype=float)
        # solve
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if not res.success:
            break
        delta = res.x[:n]
        new_h = h + delta
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5 - 1e-9:
            h = new_h
            cur_c5 = new_c5
        else:
            break
    return h, cur_c5

def compute_gradient_dx(h, dx):
    n = h.shape[0]
    M = 2 * n - 1
    Gdx = np.empty((M, n), dtype=float)
    # shifts from -(n-1) to (n-1)
    for idx, k in enumerate(range(-(n-1), n)):
        if k > 0:
            # forward: zeros at end
            forward = np.concatenate((h[k:], np.zeros(k, dtype=float)))
            # backward: zeros at start
            backward = np.concatenate((np.zeros(k, dtype=float), h[:n-k]))
        elif k < 0:
            p = -k
            forward = np.concatenate((np.zeros(p, dtype=float), h[:n-p]))
            backward = np.concatenate((h[p:], np.zeros(p, dtype=float)))
        else:  # k == 0
            forward = h
            backward = h
        Gdx[idx] = (1.0 - forward - backward) * dx
    return Gdx

import time
import numpy as np
from scipy.optimize import linprog

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradient_dx(h, dx):
    """Gradient matrix Gdx where Gdx[k,i] = dx*(1 - h[i+k] - h[i-k]) with zero padding."""
    n = h.shape[0]
    M = 2 * n - 1
    Gdx = np.empty((M, n), dtype=float)
    for idx, k in enumerate(range(-(n - 1), n)):
        if k > 0:
            forward = np.concatenate((h[k:], np.zeros(k, dtype=float)))
            backward = np.concatenate((np.zeros(k, dtype=float), h[:n - k]))
        elif k < 0:
            p = -k
            forward = np.concatenate((np.zeros(p, dtype=float), h[:n - p]))
            backward = np.concatenate((h[p:], np.zeros(p, dtype=float)))
        else:
            forward = h
            backward = h
        Gdx[idx] = (1.0 - forward - backward) * dx
    return Gdx


def lp_optimize_within_time(h0, time_limit, rng):
    """Linear-programming based improvement of h0 within time_limit seconds."""
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    cur_c5 = compute_c5(h)
    start = time.time()
    max_iters = 30
    for _ in range(max_iters):
        if time.time() - start >= time_limit:
            break
        Gdx = compute_gradient_dx(h, dx)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = Gdx.shape[0]

        A_ub = np.hstack([Gdx, -np.ones((M, 1), dtype=float)])
        b_ub = -corr
        c = np.zeros(n + 1, dtype=float)
        c[-1] = 1.0

        bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(None, None)]
        A_eq = np.zeros((1, n + 1), dtype=float)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0], dtype=float)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        delta = res.x[:n]
        new_h = h + delta
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5 - 1e-9:
            h = new_h
            cur_c5 = new_c5
        else:
            break
    return h, cur_c5


def simulated_annealing(h0, time_limit, rng):
    """Simple SA preserving sum and box constraints."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < time_limit:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        elapsed = time.time() - start
        temp = 0.02 * (1.0 - elapsed / time_limit)
        if temp < 1e-6:
            temp = 1e-6
        delta = max_delta * rng.random() * (temp / 0.02)
        if delta <= 0.0:
            continue
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / temp):
                cur_score = new_score
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_target = 512

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_balanced_binary(n_target, rng).astype(float)

    if base.size != n_target:
        base = upsample(base, n_target)

    base = project_onto_box_sum(base, n_target / 2.0)
    best_h = base.copy()
    best_c5 = compute_c5(best_h)

    # Random-restart loop with LP refinement
    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining <= 1.0:
            break
        t_restart = min(20.0, remaining * 0.5)

        if rng.random() < 0.2:
            h0 = best_h.copy()
        else:
            if rng.random() < 0.5:
                h0 = random_balanced_binary(n_target, rng).astype(float)
            else:
                h0 = rng.random(n_target)
                h0 = project_onto_box_sum(h0, n_target / 2.0)

        h_opt, c5_opt = lp_optimize_within_time(h0, t_restart, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if best_c5 <= 0.3808:
            break

    # Final SA fine-tuning
    remaining = budget_s - (time.time() - start)
    if remaining > 0.5:
        h_sa, c5_sa = simulated_annealing(best_h, remaining, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    best_h = project_onto_box_sum(best_h, n_target / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_target
