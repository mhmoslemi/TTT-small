def compute_grad_weighted_corr(h, w):
    n = h.size
    # w length L = 2n-1, indexing k = idx - (n-1)
    grad = np.zeros_like(h)
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # positions i=0..n-k-1: add weight * (1 - h[i+k])
            grad[:n - k] += weight * (1.0 - h[k:])
            # positions i=k..n-1: subtract weight * h[i-k]
            grad[k:] -= weight * h[:n - k]
        else:
            m = -k
            # i=m..n-1: add weight * (1 - h[i-m])
            grad[m:] += weight * (1.0 - h[:n - m])
            # i=0..n-m-1: subtract weight * h[i+m]
            grad[:n - m] -= weight * h[m:]
    return grad

def compute_grad_weighted_corr(h, w):
    n = h.size
    # determine FFT length
    L = 2 * n - 1
    M = 1 << ((L + n - 2).bit_length())  # next power of two >= L + n - 1
    # compute FFTs
    W_fft = np.fft.fft(w, M)
    H_fft = np.fft.fft(h, M)
    Hc_fft = np.fft.fft(1.0 - h, M)
    conv_h = np.fft.ifft(W_fft * H_fft).real
    conv_hc = np.fft.ifft(W_fft * Hc_fft).real
    # gradient is conv_hc[:n] - conv_h[:n]
    grad = conv_hc[:n] - conv_h[:n]
    return grad

def gradient_descent_smooth(h0, n_iter, step0, beta, rng, time_left, alpha_start=5.0, alpha_max=50.0):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0.0:
            break
        # compute correlation (scaled)
        corr_raw = compute_corr_full_fft(h) * (2.0 / n)
        max_corr = np.max(corr_raw)
        # compute smoothing weight
        alpha = alpha_start + (alpha_max - alpha_start) * (it / n_iter)
        w = np.exp(alpha * (corr_raw - max_corr))
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient weighted
        grad = compute_grad_weighted_corr(h, w)
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional perturbation
        if it % 3000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr_raw, max_c, tol=1e-12):
    n = h.size
    idxes = np.where(corr_raw >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def compute_grad_weighted_corr(h, w):
    n = h.size
    L = 2 * n - 1
    M = 1 << ((L + n - 2).bit_length())
    W_fft = np.fft.fft(w, M)
    H_fft = np.fft.fft(h, M)
    Hc_fft = np.fft.fft(1.0 - h, M)
    conv_h = np.fft.ifft(W_fft * H_fft).real
    conv_hc = np.fft.ifft(W_fft * Hc_fft).real
    grad = conv_hc[:n] - conv_h[:n]
    return grad

def gradient_descent_max(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    corr_raw = compute_corr_full_fft(h)
    best_score = np.max(corr_raw) * (2.0 / n)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:

        corr_raw = compute_corr_full_fft(h)
        max_c = np.max(corr_raw)
        grad = compute_grad_max_shifts(h, corr_raw, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)

        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def gradient_descent_smooth(h0, n_iter, step0, beta, rng,
                            time_left, alpha_start=5.0, alpha_max=50.0):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    corr_raw = compute_corr_full_fft(h) * (2.0 / n)
    best_score = np.max(corr_raw)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr_raw = compute_corr_full_fft(h) * (2.0 / n)
        max_c = np.max(corr_raw)
        alpha = alpha_start + (alpha_max - alpha_start) * (it / n_iter)
        w = np.exp(alpha * (corr_raw - max_c))
        w_sum = np.sum(w)
        if w_sum == 0.0:

        else:
            w /= w_sum
        grad = compute_grad_weighted_corr(h, w)

        h_new = h - eta * momentum
        if it % 3000 == 0:

        h_new = project_onto_box_sum(h_new, s)

        score = np.max(corr_new)
