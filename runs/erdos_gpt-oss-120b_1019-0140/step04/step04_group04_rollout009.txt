def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

        return np.clip(v - lam, lo, hi).sum()

        corr = compute_corr_fft(h)
        grad = compute_gradient_max_shifts(h, corr)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    best_h = None
    best_score = np.inf

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)
    base_score = compute_c5_fft(base)
    if base_score < best_score:
        best_h, best_score = base.copy(), base_score

    scales = [64, 128, 256, 512]
    target_iters = {64: 40000, 128: 60000, 256: 100000, 512: 200000}
    target_c5 = 0.3808

    for n in scales:
        if time_left() <= 2:
            break
        if best_h is not None and best_h.size != n:
            base = upsample(best_h, n)
        elif base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        max_possible = int(time_left() * 500)
        n_iter = min(target_iters.get(n, 50000), max_possible)
        if n_iter <= 0:
            continue
        step0 = 0.5 if n <= 256 else 0.3
        beta = 0.95
        h_opt, sc_opt = gd_momentum(base, n_iter=n_iter, step0=step0,
                                    beta=beta, rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_h, best_score = h_opt.copy(), sc_opt
        if best_score <= target_c5:
            break

    if best_h is None:
        best_h = random_binary_balanced(64, rng)
        best_score = compute_c5_fft(best_h)

    # local fine‑tuning
    n_final = best_h.size
    while time_left() > 2:
        i, j = rng.choice(n_final, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.uniform(0.0, 0.5)
        if delta <= 0:
            continue
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5_fft(h_new)
        if sc_new < best_score:
            best_h, best_score = h_new, sc_new
        if rng.random() < 0.1:
            pert = rng.normal(scale=0.02, size=n_final)
            h_pert = best_h + pert
            h_pert = project_onto_box_sum(h_pert, n_final / 2.0)
            sc_pert = compute_c5_fft(h_pert)
            if sc_pert < best_score:
                best_h, best_score = h_pert, sc_pert
        if best_score <= target_c5:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
