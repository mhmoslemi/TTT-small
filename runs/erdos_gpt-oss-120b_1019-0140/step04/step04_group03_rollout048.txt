def compute_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)[:2*n-1]
    return corr

def compute_c5(h):
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.shape[0])

def compute_gradients_max_shifts(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    # Find shifts that achieve max
    # use tolerance: maybe 1e-12 * max_corr?
    tol = 1e-12
    shift_idxs = np.where(corr >= max_corr - tol)[0]
    shifts = shift_idxs - (n - 1)
    grad = np.zeros_like(h)
    # compute contributions for each shift
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            grad[:sz] += (1.0 - h[shift:])
            grad[shift:] += -h[:sz]
        else:
            s = -shift
            sz = n - s
            grad[s:] += (1.0 - h[:sz])
            grad[:s] += -h[s:]
    # average over shifts
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def adam_optimize(h0, max_iters, step0, beta1, beta2, eps, start_time, max_time):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters+1):
        if time.time() - start_time >= max_time:
            break
        grad = compute_gradients_max_shifts(h)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step schedule: decreasing step0 / sqrt(t) maybe multiply
        eta = step0 / np.sqrt(t)
        h_new = h - eta * m_hat / (np.sqrt(v_hat) + eps)
        # project onto box and sum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h, cur_score, time_remaining, rng):
    n = h.size
    total_steps = int(min(200000, max(5000, time_remaining * 200.0))) # steps per second ratio
    # schedule: linear cooling from T0 to T_final
    T0 = 0.02
    T_final = 1e-6
    best_h = h.copy()
    best_score = cur_score
    for step in range(total_steps):
        # compute remaining time maybe not needed
        T = T0 * ((total_steps - step) / total_steps) + T_final
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # ensure we can move from i to j (i has >0, j <1)
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        # optionally scale delta by temperature? Might help.
        delta *= min(1.0, T / T0) # smaller moves at low T?
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # ensure bounds enclose solution
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if np.abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)[:2 * n - 1]
    return corr

def compute_c5(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_gradients_max_shifts(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    # tolerance to capture all max shifts
    tol = 1e-12
    idxes = np.where(corr >= max_corr - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            grad[:sz] += (1.0 - h[shift:])
            grad[shift:] += -h[:sz]
        else:
            s = -shift
            sz = n - s
            grad[s:] += (1.0 - h[:sz])
            grad[:s] += -h[s:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def adam_optimize(h0, max_iters, step0, beta1, beta2, eps, start_time, max_time):
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters + 1):
        if time.time() - start_time >= max_time:
            break
        grad = compute_gradients_max_shifts(h)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        eta = step0 / np.sqrt(t)
        h_new = h - eta * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h, cur_score, start_time, max_time, rng):
    n = h.shape[0]
    time_left = max_time - (time.time() - start_time)
    if time_left <= 0:
        return h, cur_score
    total_steps = int(min(200000, max(5000, time_left * 400.0)))
    T0 = 0.02
    T_final = 1e-6
    best_h = h.copy()
    best_score = cur_score
    for step in range(total_steps):
        if time.time() - start_time >= max_time:
            break
        T = T0 * ((total_steps - step) / total_steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # final resolution
    n_final = 1024
    scales = [128, 256, 512, n_final]

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(scales[0], rng)

    best_h = None
    best_score = np.inf

    for idx, n in enumerate(scales):
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # number of restarts per scale (fewer for large n)
        restarts = 3 if n <= 512 else 2

        for r in range(restarts):
            if time.time() - start >= budget_s - 2.0:
                break
            h0 = base.copy()
            if r > 0:
                h0 += rng.normal(scale=0.05, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, n / 2.0)

            # allocate time for Adam (roughly half of remaining)
            time_left = budget_s - (time.time() - start)
            max_time_adam = min(time_left * 0.5, 300.0)
            # iterative budget: aim for ~2000 updates per second
            max_iters = int(min(500000, max(50000, max_time_adam * 2000)))
            h_opt, sc_opt = adam_optimize(
                h0, max_iters, step0=0.5,
                beta1=0.9, beta2=0.999, eps=1e-8,
                start_time=start, max_time=budget_s
            )
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            # follow with simulated annealing if time permits
            if budget_s - (time.time() - start) > 5.0:
                h_opt, sc_opt = simulated_annealing(
                    h_opt, sc_opt, start, budget_s, rng
                )
                if sc_opt < best_score:
                    best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt  # use latest as base for next restart

        # prepare for next finer scale
        if idx + 1 < len(scales):
            base = best_h if best_h is not None else base
            base = upsample(base, scales[idx + 1])

    # final refinement on the fullest resolution
    if best_h is not None and (budget_s - (time.time() - start)) > 5.0:
        time_left = budget_s - (time.time() - start)
        max_iters = int(min(500000, max(50000, time_left * 2000)))
        best_h, best_score = adam_optimize(
            best_h, max_iters, step0=0.3,
            beta1=0.9, beta2=0.999, eps=1e-8,
            start_time=start, max_time=budget_s
        )
        if budget_s - (time.time() - start) > 5.0:
            best_h, best_score = simulated_annealing(
                best_h, best_score, start, budget_s, rng
            )

    if best_h is None:
        n = n_final
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5(best_h)

    # final projection and evaluation to guarantee feasibility
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
