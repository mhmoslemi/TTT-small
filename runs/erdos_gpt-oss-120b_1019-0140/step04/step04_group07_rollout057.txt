def gradient_descent_momentum(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = compute_correlation(h)
        max_corr = corr.max()
        # compute shifts where corr is within small epsilon
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n-1)
        # gradient average over these shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                grad[:n - shift] += (1.0 - h[shift:]) - h[:n - shift]  # Actually we need derivative of h_i*(1 - h_{i+shift}) w.r.t. h_i and h_{i+shift}
                # Let's derive gradient of the max correlation term for each shift.
                # The objective is max_k sum_i h_i*(1 - h_{i+k}). For a particular shift s, the contribution is sum_i h_i*(1 - h_{i+s}). Gradient w.r.t. h_i is (1 - h_{i+s}) for i in range, because derivative of h_i term. Gradient w.r.t. h_{i+s} is -h_i (since derivative of (1 - h_{i+s}) is -1 times h_i). But we treat each variable's gradient as sum of contributions across all i that appear as either i or i+s in terms for shift s. So we need to accumulate both contributions.
                # So for shift >=0:
                # For i in [0, n - shift):
                #   grad[i] += (1 - h[i+shift])
                #   grad[i+shift] += -h[i]
                # For shift<0 similarly.
            else:
                s = -shift
                # For i in [s, n):
                #   grad[i] += (1 - h[i-s])
                #   grad[i-s] += -h[i]
        # At the end, divide by number of shifts (if >0) to get average
        if len(shifts) > 0:
            grad /= len(shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
        else:
            step *= 0.5
            if step < 1e-8:
                step = step0 * 0.1
        # optionally jitter every some iterations
        if it % 2000 == 0 and it>0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_h, best_score = h.copy(), cur_score
    return best_h, best_score

We need to compute gradient contributions correctly. Let's derive precisely.

Objective: f(h) = max_s S_s(h) where S_s(h) = sum_i h_i * (1 - h_{i+s}) for shift s (positive, integer). For s >= 0: i ranges 0..n-s-1; for s<0: shift negative, i ranges -s..n-1 (makes sure i+s is in range). Also we take the max over s. Gradient of f is subgradient: For any s achieving max (within epsilon), gradient is sum over contributions from S_s. So we can average over s.

Now the derivative of S_s w.r.t h_j:

- For each term i where j = i: term h_i*(1 - h_{i+s}) derivative wrt h_i is (1 - h_{i+s}).
- For each term i where j = i+s: term h_i*(1 - h_{i+s}) derivative wrt h_{i+s} is -h_i.
- For other indices, zero.

Thus gradient computation as described.

Thus for shift >= 0:
- For i = 0..n-s-1:
    grad[i] += (1 - h[i+s])
    grad[i+s] += -h[i]

def compute_full_corr_fft(h):
    n = h.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)
    conv = np.fft.ifft(H * G).real[:2*n - 1]
    return conv

def simulated_annealing(h0, max_steps, rng, time_left):
    h = h0.copy()
    n = len(h)
    s = n/2.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    # schedule temperature: start T0 = initial_temp, end T_end = final_temp
    T0 = 0.01
    T_end = 0.0005
    for step_idx in range(max_steps):
        if time_left() <= 0:
            break
        t = step_idx / max_steps
        T = T0 * (1 - t) + T_end * t
        # propose move: random i, j distinct
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        # choose delta proportion; maybe random factor up to 0.5
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / T):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
    return best_h, best_score

def local_pair_moves(h0, rng, time_left):
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # try moving from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc = compute_c5(h_new)
            if sc < best_score:
                best_h, best_score = h_new, sc
                continue
        # try moving from j to i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            sc2 = compute_c5(h_new2)
            if sc2 < best_score:
                best_h, best_score = h_new2, sc2
    return best_h, best_score

def improve_via_max_shift(h0, rng, time_left):
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = compute_full_corr_fft(best_h)
        max_corr = corr.max()
        # get worst shift(s)
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        shift = (shift_idxs[rng.integers(len(shift_idxs))]) - (n - 1)
        if shift >= 0:
            i_range = slice(0, n - shift)
            contrib = best_h[i_range] * (1.0 - best_h[i_range + shift])
            i_local = np.argmax(contrib)
            i = i_local
            j = i + shift
        else:
            s = -shift
            i_range = slice(s, n)
            contrib = best_h[i_range] * (1.0 - best_h[i_range - s])
            i_local = np.argmax(contrib)
            i = i_local + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc = compute_c5(h_new)
        if sc < best_score:
            best_h, best_score = h_new, sc
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):

        fm = f(lam_mid)

        else:

            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc = compute_c5(h_new)
            if sc < best_score:
                best_h, best_score = h_new, sc
                continue
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            sc2 = compute_c5(h_new2)
            if sc2 < best_score:
                best_h, best_score = h_new2, sc2
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]
    for n in scales:
        if time_left() <= 5:
            break
        if base.size != n:
            base = upsample(base, n)
            base = project_onto_box_sum(base, n / 2.0)

        max_iters = int(min(60000, max(20000, time_left() * (20000.0 / n))))
        h_gd, sc_gd = gradient_descent_momentum(base, max_iters, step0=1.0,
                                                beta=0.9, rng=rng,
                                                time_left=time_left)
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd
        base = h_gd

        if time_left() > 2:
            steps_sa = int(min(20000, max(5000, time_left() * (8000.0 / n))))
            h_sa, sc_sa = simulated_annealing(base, steps_sa, rng, time_left)
            if sc_sa < best_score:
                best_h, best_score = h_sa.copy(), sc_sa
            base = h_sa

        if time_left() > 2:
            h_ms, sc_ms = improve_via_max_shift(base, rng, time_left)
            if sc_ms < best_score:
                best_h, best_score = h_ms.copy(), sc_ms
            base = h_ms

        if time_left() > 2:
            h_pair, sc_pair = local_pair_moves(base, rng, time_left)
            if sc_pair < best_score:
                best_h, best_score = h_pair.copy(), sc_pair
            base = h_pair

        if time_left() > 5:
            for _ in range(2):
                if time_left() < 2:
                    break
                h_rand = random_balanced_binary(n, rng)
                max_iters_rand = int(min(12000, max(3000, time_left() * (4000.0 / n))))
                h_rand_ref, sc_rand = gradient_descent_momentum(h_rand,
                                                                max_iters_rand,
                                                                step0=0.5,
                                                                beta=0.9,
                                                                rng=rng,
                                                                time_left=time_left)
                if sc_rand < best_score:
                    best_h, best_score = h_rand_ref.copy(), sc_rand
                if time_left() > 2:
                    h_rand_pair, sc_rand_pair = local_pair_moves(h_rand_ref, rng, time_left)
                    if sc_rand_pair < best_score:
                        best_h, best_score = h_rand_pair.copy(), sc_rand_pair

    if time_left() > 5:
        final_iters = int(min(100000, max(30000, time_left() * 30)))
        best_h, best_score = gradient_descent_momentum(best_h, final_iters,
                                                       step0=0.3, beta=0.9,
                                                       rng=rng, time_left=time_left)

    if time_left() > 2:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
