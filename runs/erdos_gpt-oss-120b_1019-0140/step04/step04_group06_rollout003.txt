import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    # max correlation ignoring ends? Use max of entire array.
    return np.max(corr) * (2.0 / n)

def binary_from_continuous(arr):
    n = arr.shape[0]
    k = n // 2
    idx = np.argpartition(-arr, k-1)[:k]  # top k
    h = np.zeros(n, dtype=np.int8)
    h[idx] = 1
    return h

def repair_binary(v, target):
    # v is int array with ones and zeros possibly mismatched count
    cur = int(v.sum())
    diff = target - cur
    # diff >0 need to add ones: choose zeros randomly
    if diff > 0:
        zeros = np.where(v == 0)[0]
        if zeros.size > 0:
            # random choose diff indices
            add = np.random.choice(zeros, size=diff, replace=False)
            v[add] = 1
    elif diff < 0:
        ones = np.where(v == 1)[0]
        if ones.size > 0:
            rem = np.random.choice(ones, size=-diff, replace=False)
            v[rem] = 0
    # no return needed (in-place)
    return v

def crossover(p1, p2, rng):
    mask = rng.integers(0, 2, size=p1.shape[0], dtype=bool)
    child = np.where(mask, p1, p2).astype(np.int8)
    return child

def mutate(v, rng, mutation_rate):
    # swap bits with some probability per individual
    # Perform at most a few swaps if chosen
    if rng.random() < mutation_rate:
        # pick a 1 and a 0
        ones = np.where(v == 1)[0]
        zeros = np.where(v == 0)[0]
        if ones.size > 0 and zeros.size > 0:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            v[i], v[j] = 0, 1
    return v

def tournament_select(pop, scores, tournament_k, rng):
    idxs = rng.choice(len(pop), size=tournament_k, replace=False)
    best = idxs[0]
    best_score = scores[best]
    for i in idxs[1:]:
        s = scores[i]
        if s < best_score:
            best = i
            best_score = s
    return pop[best]

def ga_optimize(n, init_h, rng, time_left):
    target = n // 2
    # population
    popsize = 80
    # initialize population
    pop = []
    # start with init_h if binary and correct size, else convert
    if init_h is not None and init_h.shape[0] == n:
        # convert to binary if needed
        if not np.array_equal(init_h, init_h.astype(bool)):
            init_h_bin = binary_from_continuous(init_h)
        else:
            init_h_bin = init_h.astype(np.int8)
        init_h_bin = repair_binary(init_h_bin, target)
        pop.append(init_h_bin)
    while len(pop) < popsize:
        # random binary balanced
        v = np.zeros(n, dtype=np.int8)
        idx = rng.choice(n, size=target, replace=False)
        v[idx] = 1
        pop.append(v)
    pop = np.array(pop, dtype=np.int8)

    # cache dict
    fitness_cache = {}
    # compute initial fitness
    scores = np.empty(popsize, dtype=float)
    for i in range(popsize):
        key = bytes(np.packbits(pop[i]))
        if key in fitness_cache:
            scores[i] = fitness_cache[key]
        else:
            sc = compute_c5(pop[i])
            fitness_cache[key] = sc
            scores[i] = sc
    # best
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    generation = 0
    # parameters
    tournament_k = 5
    crossover_prob = 0.9
    mutation_rate = 0.3

    while time_left() > 0.5:
        generation += 1
        # elite keep
        new_pop = [best_h]  # elite
        # fill rest
        while len(new_pop) < popsize:
            # selection
            p1 = tournament_select(pop, scores, tournament_k, rng)
            p2 = tournament_select(pop, scores, tournament_k, rng)
            # crossover
            if rng.random() < crossover_prob:
                child = crossover(p1, p2, rng)
            else:
                child = p1.copy()
            # repair to correct sum
            child = repair_binary(child, target)
            # mutation
            child = mutate(child, rng, mutation_rate)
            child = repair_binary(child, target)  # ensure after mutation
            new_pop.append(child)
        pop = np.array(new_pop, dtype=np.int8)
        # evaluate
        scores = np.empty(popsize, dtype=float)
        for i in range(popsize):
            key = bytes(np.packbits(pop[i]))
            if key in fitness_cache:
                scores[i] = fitness_cache[key]
            else:
                sc = compute_c5(pop[i])
                fitness_cache[key] = sc
                scores[i] = sc
        # update best
        cur_best_idx = int(np.argmin(scores))
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_h = pop[cur_best_idx].copy()
            # optional: reduce mutation_rate adaptively
        # maybe early stop if score <= target
        if best_score <= 0.3808:
            break
    return best_h, best_score

    t_alloc1 = budget_s * 0.25

        best_h2, best_score2 = ga_optimize(n2, best_h_n2, rng, time_left2)
    else:
        best_h2, best_score2 = best_h_n2, compute_c5(best_h_n2)

    # final local search (hill climbing)
    # simple random swaps improvement loop
    best_h = best_h2
    best_score = best_score2
    target = best_h.size // 2
    # compute worst shift list (shifts achieving max complement)
    # For local improvement we can just try random swaps a lot until time runs out.
    while time_left() > 1.0:
        # propose random swap
        i = rng.integers(0, best_h.size)
        if best_h[i] == 1:
            # choose a zero j
            zeros = np.where(best_h == 0)[0]
            if zeros.size == 0:
                break
            j = rng.choice(zeros)
        else:
            ones = np.where(best_h == 1)[0]
            if ones.size == 0:
                break
            j = rng.choice(ones)
        # swap
        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
            # optionally break if threshold met
            if best_score <= 0.3808:
                break
    # Ensure final vector is binary and sum constraint
    # Might be non-binary due to earlier? but all binary.
    # Return results
    return best_h.astype(float), best_score, best_h.size

import time
import numpy as np

def compute_c5(h):
    """Return the overlap bound C5 for binary (or float) vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def binary_from_continuous(arr):
    """Select the top half entries of arr to form a binary vector."""
    n = arr.shape[0]
    k = n // 2
    idx = np.argpartition(-arr, k - 1)[:k]
    b = np.zeros(n, dtype=np.int8)
    b[idx] = 1
    return b

def repair_binary(v, target, rng):
    """Adjust v (int8 0/1) so that sum(v) == target by random swaps."""
    cur = int(v.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(v == 0)[0]
        if zeros.size:
            add = rng.choice(zeros, size=diff, replace=False)
            v[add] = 1
    elif diff < 0:
        ones = np.where(v == 1)[0]
        if ones.size:
            rem = rng.choice(ones, size=-diff, replace=False)
            v[rem] = 0
    return v

def crossover(p1, p2, rng):
    """Uniform crossover between two parents."""
    mask = rng.integers(0, 2, size=p1.shape[0], dtype=bool)
    child = np.where(mask, p1, p2).astype(np.int8)
    return child

def mutate(v, rng, rate):
    """Swap a random 1 with a random 0 with probability rate."""
    if rng.random() < rate:
        ones = np.where(v == 1)[0]
        zeros = np.where(v == 0)[0]
        if ones.size and zeros.size:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            v[i], v[j] = 0, 1
    return v

def tournament_select(pop, scores, k, rng):
    """Select one individual via k-way tournament."""
    idxs = rng.choice(len(pop), size=k, replace=False)
    best = idxs[0]
    best_score = scores[best]
    for i in idxs[1:]:
        s = scores[i]
        if s < best_score:
            best = i
            best_score = s
    return pop[best]

def ga_optimize(n, init_h, rng, time_left):
    """Genetic algorithm for binary vectors of length n."""
    target = n // 2
    popsize = 80
    # Build initial population
    pop = []
    if init_h is not None and init_h.shape[0] == n:
        if not np.array_equal(init_h, init_h.astype(bool)):
            init_bin = binary_from_continuous(init_h)
        else:
            init_bin = init_h.astype(np.int8)
        init_bin = repair_binary(init_bin, target, rng)
        pop.append(init_bin)
    while len(pop) < popsize:
        v = np.zeros(n, dtype=np.int8)
        v[rng.choice(n, size=target, replace=False)] = 1
        pop.append(v)
    pop = np.array(pop, dtype=np.int8)

    # Fitness cache
    fitness_cache = {}
    scores = np.empty(popsize, dtype=float)
    for i in range(popsize):
        key = bytes(np.packbits(pop[i]))
        if key in fitness_cache:
            scores[i] = fitness_cache[key]
        else:
            sc = compute_c5(pop[i])
            fitness_cache[key] = sc
            scores[i] = sc

    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    tournament_k = 5
    crossover_prob = 0.9
    mutation_rate = 0.3

    while time_left() > 0.5:
        new_pop = [best_h]  # elitism
        while len(new_pop) < popsize:
            p1 = tournament_select(pop, scores, tournament_k, rng)
            p2 = tournament_select(pop, scores, tournament_k, rng)
            child = crossover(p1, p2, rng) if rng.random() < crossover_prob else p1.copy()
            child = repair_binary(child, target, rng)
            child = mutate(child, rng, mutation_rate)
            child = repair_binary(child, target, rng)
            new_pop.append(child)
        pop = np.array(new_pop, dtype=np.int8)

        for i in range(popsize):
            key = bytes(np.packbits(pop[i]))
            if key in fitness_cache:
                scores[i] = fitness_cache[key]
            else:
                sc = compute_c5(pop[i])
                fitness_cache[key] = sc
                scores[i] = sc

        cur_best_idx = int(np.argmin(scores))
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_h = pop[cur_best_idx].copy()
            if best_score <= 0.3808:
                break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # Acquire initial vector if provided
    init_h = None
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=float)

    # Phase 1: optimize at n=128
    n1 = 128
    if init_h is not None and init_h.shape[0] != n1:
        x_old = np.linspace(0.0, 1.0, init_h.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n1, endpoint=False)
        init_h_n1 = np.interp(x_new, x_old, init_h)
    else:
        init_h_n1 = init_h

    t_alloc1 = budget_s * 0.25
    deadline1 = start + t_alloc1
    time_left1 = lambda: max(0.0, deadline1 - time.time())
    best_h1, best_score1 = ga_optimize(n1, init_h_n1, rng, time_left1)

    # Phase 2: upscale to n=256 and refine
    n2 = 256
    best_h_n2 = np.repeat(best_h1, 2)
    if best_h_n2.shape[0] != n2:
        best_h_n2 = best_h_n2[:n2]

    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        time_left2 = lambda: max(0.0, remaining - (time.time() - start))
        best_h2, best_score2 = ga_optimize(n2, best_h_n2, rng, time_left2)
    else:
        best_h2 = best_h_n2
        best_score2 = compute_c5(best_h2)

    # Local hill‑climbing improvement
    best_h = best_h2
    best_score = best_score2
    while time_left() > 1.0:
        i = rng.integers(0, best_h.size)
        if best_h[i] == 1:
            zeros = np.where(best_h == 0)[0]
            if zeros.size == 0:
                break
            j = rng.choice(zeros)
        else:
            ones = np.where(best_h == 1)[0]
            if ones.size == 0:
                break
            j = rng.choice(ones)
        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
            if best_score <= 0.3808:
                break

    # Return as float array per specification
    return best_h.astype(float), float(best_score), int(best_h.size)
