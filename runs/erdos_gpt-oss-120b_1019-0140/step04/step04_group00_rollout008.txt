def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def binary_swap_anneal(h0, rng, time_left, max_iters=200000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h = h.copy()
    best_s = cur
    T, T_min, decay = 0.2, 1e-5, 0.9999  # adjust decay for faster cooling
    it = 0
    n = h.size
    while it < max_iters and time_left() > 0.0:
        # Choose random indices with h close to 1 and 0
        # Instead of exact equality, allow near values for continuous case
        # We'll pick random i with h[i] > 0.5 and j with h[j] <= 0.5
        # But easier: sample random index i,j and propose swapping values.
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        # propose swap only if it changes values significantly (to encourage change)
        if hi == hj:
            continue
        # perform swap
        h[i], h[j] = hj, hi
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s = new
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = hi, hj
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    # initial guess: h0 and its c5
    t0 = compute_c5_fast(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]  # t >= 0; can be any 
    # Actually t needs to be >= 0, but also we want minimize t, so lower bound 0.
    # Let's bound t as (0.0, 1.0) maybe but could be larger. Upper bound can be 2? Not needed. Use (0.0, None)
    # SciPy requires finite bounds? It can accept None for no bound.
    # We'll set (0.0, None).
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # expand bounds if necessary
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_corr_fft(h):
    n = h.size
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    return conv[:2 * n - 1]


def compute_c5_fast(h):
    n = h.size
    corr = compute_corr_fft(h)
    return corr.max() * (2.0 / n)


def gradient_of_shift(h, shift):
    n = h.size
    grad = np.ones(n, dtype=float)
    if shift >= 0:
        m = n - shift
        grad[:m] -= h[shift:]
        grad[shift:] -= h[:m]
    else:
        s = -shift
        m = n - s
        grad[s:] -= h[:m]
        grad[:m] -= h[s:]
    return grad


def compute_gradient_max_shift(h):
    n = h.size
    corr = compute_corr_fft(h)
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h, best_s = h.copy(), compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s


def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def resample(arr, new_n):
    old_n = arr.size
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def binary_swap_anneal(h0, rng, time_left, max_iters=200000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.9999
    n = h.size
    it = 0
    while it < max_iters and time_left() > 0.0:
        i, j = rng.integers(n, size=2)
        if i == j or h[i] == h[j]:
            it += 1
            continue
        hi, hj = h[i], h[j]
        h[i], h[j] = hj, hi
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = hi, hj
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s


def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    t0 = compute_c5_fast(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    except Exception:
        h_opt = h0
    return h_opt, compute_c5_fast(h_opt)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target_val = 0.3808
    n0 = 256
    target_sum0 = n0 / 2.0

    # --- initial candidates ---
    cand = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n0) if init.size != n0 else init.copy()
        cand.append(project_onto_box_sum(init, target_sum0))
    cand.append(thue_morse_binary(n0))
    for _ in range(3):
        cand.append(random_binary_balanced(n0, rng))
    for _ in range(2):
        v = rng.random(n0)
        cand.append(project_onto_box_sum(v, target_sum0))

    best_h = None
    best_s = np.inf
    best_n = n0

    # --- coarse optimization ---
    for h0 in cand:
        if time_left() < 5.0:
            break
        h_opt, s_opt = gradient_descent_momentum(h0, time_left,
                                                 step0=0.8, beta=0.9,
                                                 max_iter=200000)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= target_val:
            return best_h, best_s, best_n

    # --- refinement at higher resolution ---
    if time_left() > 30.0:
        n1 = 512
        target_sum1 = n1 / 2.0
        h_up = resample(best_h, n1) if best_h is not None else rng.random(n1)
        h_up = project_onto_box_sum(h_up, target_sum1)
        h_opt, s_opt = gradient_descent_momentum(h_up, time_left,
                                                 step0=0.8, beta=0.9,
                                                 max_iter=300000)
        if s_opt < best_s:
            best_s, best_h, best_n = s_opt, h_opt.copy(), n1
        if best_s <= target_val:
            return best_h, best_s, best_n
        # short SLSQP polish
        if time_left() > 5.0:
            try:
                h_pol, s_pol = slsqp_optimize_candidate(best_h, max_iter=300)
                if s_pol < best_s:
                    best_s, best_h = s_pol, h_pol.copy()
                    best_n = n1
            except Exception:
                pass
        # optional anneal
        if time_left() > 10.0:
            h_ann, s_ann = binary_swap_anneal(best_h, rng, time_left,
                                             max_iters=200000)
            if s_ann < best_s:
                best_s, best_h, best_n = s_ann, h_ann.copy(), n1

    # final projection and evaluation
    if best_h is None:
        best_n = n0
        best_h = np.full(best_n, 0.5)
    best_h = project_onto_box_sum(best_h, best_n / 2.0)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s, best_n
