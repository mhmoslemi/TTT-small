def simulated_annealing_binary(x0, rng, time_left):
    x = x0.copy()
    n = x.shape[0]
    half = n // 2
    # Pre-compute indices of ones and zeros
    ones = np.flatnonzero(x == 1)
    zeros = np.flatnonzero(x == 0)
    best_x = x.copy()
    best_score = compute_c5(x)
    cur_score = best_score
    # Cooling schedule: initial temperature based on score range
    # Let's set T0 = 0.01 maybe.
    T = 0.02
    # Determine total iterations based on time left estimate: assume each iteration ~0.002 sec.
    # We'll loop until time runs out.
    while time_left() > 0.2:
        # pick random swap
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        x[i] = 0
        x[j] = 1
        # recompute score
        new_score = compute_c5(x)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # accept with prob
            prob = np.exp(-delta / T) if T > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            cur_score = new_score
            # update ones/zeros lists accordingly
            ones[i_idx] = j
            zeros[j_idx] = i
            # maybe also swap references? Actually we need to keep sorted? Not needed for random.
            if cur_score < best_score:
                best_score = cur_score
                best_x = x.copy()
        else:
            # revert swap
            x[i] = 1
            x[j] = 0
        # cooling
        T *= 0.9999  # slow cooling
        # occasionally reset temperature if T gets too low
        if T < 1e-6:
            T = 0.01
    return best_x, best_score

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n: return np.full_like(v, lo)
    if s >= hi * n: return np.full_like(v, hi)
    lam_low, lam_high = np.min(v) - hi, np.max(v) - lo
    f = lambda lam: np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s: lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s: lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12: break
        if fm > s: lam_low = lam_mid
        else: lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n: return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            w = np.zeros(n, dtype=h.dtype)
            w[:n - shift] = h[shift:]
            grad[:n - shift] += 1.0 - h[shift:] - w[:n - shift]
            w2 = np.zeros(n, dtype=h.dtype)
            w2[shift:] = h[:n - shift]
            grad[shift:] += 1.0 - h[:n - shift] - w2[shift:]
        else:
            s = -shift
            w = np.zeros(n, dtype=h.dtype)
            w[s:] = h[:n - s]
            grad[s:] += 1.0 - h[:n - s] - w[s:]
            w2 = np.zeros(n, dtype=h.dtype)
            w2[:n - s] = h[s:]
            grad[:n - s] += 1.0 - h[s:] - w2[:n - s]
    if len(shifts) > 0: grad /= len(shifts)
    return grad

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for _ in range(max_iters):
        if time_left() <= 0: break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = project_onto_box_sum(h - step * momentum, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h, cur_score = h_new, new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
        else:
            step *= 0.5
            if step < 1e-6: step = step0 * 0.1
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h, best_score = h.copy(), compute_c5(h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, 2, replace=False)
        if best_h[i] > 0 and best_h[j] < 1:
            delta = min(best_h[i], 1.0 - best_h[j]) * rng.random() * 0.5
            if delta > 0:
                h_new = best_h.copy()
                h_new[i] -= delta; h_new[j] += delta
                s = compute_c5(h_new)
                if s < best_score:
                    best_h, best_score = h_new, s
        if best_h[j] > 0 and best_h[i] < 1:
            delta = min(best_h[j], 1.0 - best_h[i]) * rng.random() * 0.5
            if delta > 0:
                h_new = best_h.copy()
                h_new[j] -= delta; h_new[i] += delta
                s = compute_c5(h_new)
                if s < best_score:
                    best_h, best_score = h_new, s
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    best_h, best_score = h.copy(), compute_c5(h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0: break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0: continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta; h_new[j] += delta
        s = compute_c5(h_new)
        if s < best_score:
            best_h, best_score = h_new, s
    return best_h, best_score

def to_binary_from_continuous(h):
    n = h.shape[0]; k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    b = np.zeros_like(h, dtype=float)
    b[idx] = 1.0
    return b

def simulated_annealing_binary(x0, rng, time_left):
    x = x0.copy()
    n = x.shape[0]
    best_x, best_score = x.copy(), compute_c5(x)
    cur_score = best_score
    T = 0.02
    while time_left() > 0.5:
        i, j = rng.integers(n, size=2)
        if x[i] == x[j]: continue
        x[i], x[j] = 0.0, 1.0
        new_score = compute_c5(x)
        delta = new_score - cur_score
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_x = cur_score, x.copy()
        else:
            x[i], x[j] = 1.0, 0.0
        T *= 0.9999
        if T < 1e-6: T = 0.02
    return best_x, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    final_n = 512
    if base.shape[0] != final_n:
        base = upsample(base, final_n)
    base = project_onto_box_sum(base, final_n / 2.0)

    best_h, best_score = base.copy(), compute_c5(base)

    # intensive gradient refinement
    if time_left() > 10:
        iters = int(min(150000, max(30000, time_left() * 25)))
        best_h, best_score = gradient_descent_adaptive(best_h, iters,
                                                       step0=0.5, beta=0.9,
                                                       rng=rng, time_left=time_left)
    # target max‑shift improvement
    if time_left() > 5:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    # local pair moves
    if time_left() > 5:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    # binary conversion and simulated annealing
    bin_h = to_binary_from_continuous(best_h)
    if time_left() > 5:
        bin_h, bin_score = simulated_annealing_binary(bin_h, rng, time_left)
        if bin_score < best_score:
            best_h, best_score = bin_h.astype(float), bin_score

    # final fractional refinement
    if time_left() > 5:
        iters = int(min(80000, max(20000, time_left() * 20)))
        best_h, best_score = gradient_descent_adaptive(best_h, iters,
                                                       step0=0.3, beta=0.9,
                                                       rng=rng, time_left=time_left)

    best_h = project_onto_box_sum(best_h, final_n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, final_n
