def apply_swap(corr, b, i, j):
    n = b.size
    L = 2*n - 1
    center = n - 1
    diff = np.zeros(L, dtype=float)
    diff[i:i+n] += 2 * b
    diff[j:j+n] -= 2 * b
    new_corr = corr + diff
    delta = j - i
    new_corr[center] -= 2
    new_corr[center + delta] += 1
    new_corr[center - delta] += 1
    return new_corr

But caution: diff[i:i+n] and diff[j:j+n] overlap if i and j are close; adding then subtracting the same b segment may produce net values, which is correct.

def anneal(b_initial, time_allowed, rng):
    n = len(b_initial)
    b = b_initial.copy()
    corr = np.convolve(b, 1 - b, mode='full')
    cur_max = corr.max()
    best_b = b.copy()
    best_max = cur_max
    T = 20.0
    cooling = 0.9995
    start = time.time()
    it = 0
    while True:
        # time check per 2048 iters
        if it % 2048 == 0:
            if time.time() - start >= time_allowed:
                break
        i = int(rng.integers(0, n))
        if b[i] != 1:
            continue
        j = int(rng.integers(0, n))
        if b[j] != 0:
            continue
        # compute new corr
        new_corr = apply_swap(corr, b, i, j)
        new_max = new_corr.max()
        delta = new_max - cur_max
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            b[i] = 0
            b[j] = 1
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_b = b.copy()
        T = max(T * cooling, 1e-8)
        it += 1
    return best_b, best_max

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # choose n = 512
    n = 512
    # upsample initial if available, else random
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        # upsample to n
        if base.shape[0] != n:
            # upsample with linear interpolation
            m = base.shape[0]
            x_old = np.linspace(0.0, 1.0, m, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            base_fine = np.interp(x_new, x_old, base)
        else:
            base_fine = base.copy()
        # binarize to exactly n/2 ones
        order = np.argsort(base_fine)
        b0 = np.zeros(n, dtype=float)
        b0[order[-n//2:]] = 1.0
    else:
        # random balanced binary
        perm = rng.permutation(n)
        b0 = np.zeros(n, dtype=float)
        b0[perm[:n//2]] = 1.0

    # allocate time for initial annealing
    init_time = min(400.0, time_left() * 0.5)  # up to half budget
    best_b, best_max = anneal(b0, init_time, rng)
    best_scaled = best_max * (2.0 / n)

    # If good enough, finish early
    TARGET = 0.3808
    if best_scaled <= TARGET:
        return best_b, best_scaled, n

    # Remaining time for restarts
    while time_left() > 20.0:
        # random restart
        perm = rng.permutation(n)
        b_start = np.zeros(n, dtype=float)
        b_start[perm[:n//2]] = 1.0
        # allocate small time slice
        slice_time = min(30.0, time_left() * 0.3)
        b_opt, max_opt = anneal(b_start, slice_time, rng)
        if max_opt < best_max:
            best_b, best_max = b_opt, max_opt
            best_scaled = best_max * (2.0 / n)
            if best_scaled <= TARGET:
                break

    # Final projection just in case (should already be binary and balanced)
    # Return results
    return best_b, best_scaled, n

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    TARGET = 0.3808

    best_b_global = None
    best_scaled_global = np.inf
    # Try sizes
    for n in (256, 512):
        if time_left() < 5.0:
            break
        # Prepare initial binary
        if 'initial_h_values' in globals():
            base = np.asarray(initial_h_values, dtype=float)
            if base.shape[0] != n:
                # upsample using linear interpolation
                m = base.shape[0]
                x_old = np.linspace(0.0, 1.0, m, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                base_fine = np.interp(x_new, x_old, base)
            else:
                base_fine = base.copy()
            order = np.argsort(base_fine)
            b0 = np.zeros(n, dtype=float)
            b0[order[-n//2:]] = 1.0
        else:
            # random balanced binary
            perm = rng.permutation(n)
            b0 = np.zeros(n, dtype=float)
            b0[perm[:n//2]] = 1.0

        # Time allocation: try initial SA 30% of remaining time or 60 seconds, whichever smaller
        remaining = time_left()
        init_time = min(60.0, remaining * 0.3)
        # run annealing
        best_b, best_max = anneal(b0, init_time, rng, TARGET, n)
        best_scaled = best_max * (2.0 / n)
        if best_scaled < best_scaled_global:
            best_scaled_global = best_scaled
            best_b_global = best_b.copy()
        if best_scaled <= TARGET:
            return best_b_global, best_scaled_global, n

        # Additional restarts
        while time_left() > 5.0:
            # random restart
            perm = rng.permutation(n)

def anneal(b_init, max_seconds, rng, target_scaled, n):
    import time, numpy as np
    b = b_init.copy()
    # compute initial correlation
    corr = np.correlate(b, 1 - b, mode='full').astype(float)
    cur_max = corr.max()
    best_b = b.copy()
    best_max = cur_max
    target_max_unscaled = target_scaled * n / 2.0
    T = 20.0
    cooling = 0.9997
    start = time.time()
    l = np.arange(-(n-1), n, dtype=int)
    half = n // 2
    while True:
        if time.time() - start >= max_seconds:
            break
        # optionally break if best_max already <= target_max_unscaled
        if best_max <= target_max_unscaled:
            break
        # random swap
        i = int(rng.integers(0, n))
        if b[i] != 1.0:
            continue
        j = int(rng.integers(0, n))
        if b[j] != 0.0:
            continue
        # compute diff efficiently
        # compute term arrays
        idx_i_minus_l = i - l
        idx_j_minus_l = j - l
        idx_i_plus_l = i + l
        idx_j_plus_l = j + l

        term_a = np.where((0 <= idx_i_minus_l) & (idx_i_minus_l < n), b[idx_i_minus_l], 0.0)
        term_b = np.where((0 <= idx_j_minus_l) & (idx_j_minus_l < n), b[idx_j_minus_l], 0.0)
        term_c = np.where((0 <= idx_i_plus_l) & (idx_i_plus_l < n), b[idx_i_plus_l], 0.0)
        term_d = np.where((0 <= idx_j_plus_l) & (idx_j_plus_l < n), b[idx_j_plus_l], 0.0)

        diff = -term_b + term_a + term_c - term_d

        center = n - 1
        delta = j - i
        diff[center] -= 2.0
        diff[center + delta] += 1.0
        diff[center - delta] += 1.0

        new_corr = corr + diff
        new_max = new_corr.max()
        delta_max = new_max - cur_max
        if delta_max <= 0.0 or rng.random() < np.exp(-delta_max / T):
            # accept
            b[i] = 0.0
            b[j] = 1.0
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_b = b.copy()
        T = max(T * cooling, 1e-8)

    return best_b, best_max

import time
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def anneal(b_init, max_seconds, rng, target_scaled, n):
    """Simulated annealing on a balanced binary vector."""
    b = b_init.copy()
    v = 1.0 - b
    corr = np.correlate(b, v, mode='full').astype(float)
    cur_max = corr.max()
    best_b = b.copy()
    best_max = cur_max
    target_max_unscaled = target_scaled * n / 2.0
    L = 2 * n - 1
    center = n - 1
    l_arr = np.arange(-(n - 1), n, dtype=int)
    b_pad = np.pad(b, (n, n), mode='constant')
    v_pad = np.pad(v, (n, n), mode='constant')
    start = time.time()
    T = 20.0
    cooling = 0.9995
    while True:
        if time.time() - start >= max_seconds:
            break
        if best_max <= target_max_unscaled:
            break
        i = int(rng.integers(0, n))
        if b[i] != 1.0:
            continue
        j = int(rng.integers(0, n))
        if b[j] != 0.0:
            continue
        delta = j - i
        # term A uses b, term B uses v
        term_A = b_pad[i + n - l_arr] - b_pad[j + n - l_arr]

        diff[center + delta] += 1.0
        diff[center - delta] += 1.0
        new_corr = corr + diff

        i = int(rng.integers(0, n))
        if b[i] != 1.0:
            continue
        j = int(rng.integers(0, n))
        if b[j] != 0.0:
            continue
        delta = j - i
        term_A = b_pad[i + n - l_arr] - b_pad[j + n - l_arr]
        term_B = v_pad[j + n + l_arr] - v_pad[i + n + l_arr]
        diff = term_A + term_B
        diff[center] -= 2.0
        diff[center + delta] += 1.0
        diff[center - delta] += 1.0
        new_corr = corr + diff
        new_max = new_corr.max()
        delta_max = new_max - cur_max
        if delta_max <= 0.0 or rng.random() < np.exp(-delta_max / T):
            b[i] = 0.0
            b[j] = 1.0
            v[i] = 1.0
            v[j] = 0.0
            b_pad[i + n] = 0.0
            b_pad[j + n] = 1.0
            v_pad[i + n] = 1.0
            v_pad[j + n] = 0.0
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_b = b.copy()
        T = max(T * cooling, 1e-9)
    return best_b, best_max

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    TARGET = 0.3808
    best_h_global = None
    best_score_global = np.inf
    for n in (256, 512):
        if time_left() < 5.0:
            break
        if 'initial_h_values' in globals():
            base = np.asarray(initial_h_values, dtype=float)
            base_fine = upsample(base, n) if base.shape[0] != n else base.copy()
            order = np.argsort(base_fine)
            b0 = np.zeros(n, dtype=float)
            b0[order[-n // 2:]] = 1.0
        else:
            perm = rng.permutation(n)
            b0 = np.zeros(n, dtype=float)
            b0[perm[:n // 2]] = 1.0
        init_sec = min(60.0, time_left() * 0.3)
        b_opt, max_opt = anneal(b0, init_sec, rng, TARGET, n)
        score = max_opt * (2.0 / n)
        if score < best_score_global:
            best_score_global = score
            best_h_global = b_opt.copy()
        if best_score_global <= TARGET:
            return best_h_global, best_score_global, n
        while time_left() > 5.0:
            perm = rng.permutation(n)
            b_start = np.zeros(n, dtype=float)
            b_start[perm[:n // 2]] = 1.0
            slice_sec = min(30.0, time_left() * 0.3)
            b_opt2, max_opt2 = anneal(b_start, slice_sec, rng, TARGET, n)
            score2 = max_opt2 * (2.0 / n)
            if score2 < best_score_global:
                best_score_global = score2
                best_h_global = b_opt2.copy()
                if best_score_global <= TARGET:
                    return best_h_global, best_score_global, n
    if best_h_global is None:
        n = 256
        thue = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                           dtype=float, count=n)
        best_h_global = thue
        best_score_global = np.max(np.correlate(best_h_global,
                                                1 - best_h_global,
                                                mode='full')) * (2.0 / n)
    return best_h_global, best_score_global, best_h_global.size
