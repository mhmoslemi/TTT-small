def compute_c5_fast(h):
    n = h.shape[0]
    # Compute conv of h and (1-h) with correlation via FFT
    # correlation = conv(h_rev, 1-h) where h_rev = reverse of h
    # But we can compute directly: using FFT of reversed h and FFT of (1-h)
    # Use at least size N being power of two >= 2*n-1.
    N = 1 << ((2*n-1).bit_length())
    # using real FFT
    fft_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n-1]
    # compute max times dx = 2.0 / n
    return corr.max() * (2.0 / n)

def repair_binary(ind, target, rng):
    cur = int(ind.sum())
    if cur == target:
        return ind
    diff = cur - target
    if diff > 0:
        # too many ones: turn diff ones to zeros
        ones = np.where(ind == 1)[0]
        # randomly pick diff ones to zero
        to_zero = rng.choice(ones, size=diff, replace=False)
        ind[to_zero] = 0
    else:
        # too few ones: turn -diff zeros to ones
        zeros = np.where(ind == 0)[0]
        to_one = rng.choice(zeros, size=-diff, replace=False)
        ind[to_one] = 1
    return ind

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n/2
    h = project_onto_box_sum(h0, target)
    best_h, best_score = h.copy(), compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_score:
            best_score, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

binary_swap_anneal:

def binary_swap_anneal(h, rng, time_left, max_iters=400000):
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    while it < max_iters and time_left() > 0.01:
        # positions of ones and zeros
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.size
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n, dtype=float)
        term2 = np.zeros(n, dtype=float)
        term1[:n_s] = h[shift:]
        term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n, dtype=float)
        term2 = np.zeros(n, dtype=float)
        term1[s:] = h[:n_s]
        term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.where(corr >= mx - 1e-12)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h, best_s = h.copy(), compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    arr = np.zeros(n, dtype=float)
    arr[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return arr

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_binary(v, k):
    idx = np.argpartition(v, -k)[-k:]
    out = np.zeros_like(v, dtype=float)
    out[idx] = 1.0
    return out

def binary_swap_anneal(h, rng, time_left, max_iters=400000):
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def tournament_select(fitness, rng, tour=3):
    idxs = rng.choice(fitness.shape[0], size=tour, replace=False)
    best = idxs[0]
    best_val = fitness[best]
    for i in idxs[1:]:
        if fitness[i] < best_val:
            best = i
            best_val = fitness[i]
    return best

def crossover(p1, p2, rng):
    n = p1.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i > j:
        i, j = j, i
    child = p1.copy()
    child[i:j] = p2[i:j]
    return child

def mutate_swap(ind, rng):
    ones = np.where(ind == 1)[0]
    zeros = np.where(ind == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return ind
    i = rng.choice(ones)
    j = rng.choice(zeros)
    ind[i], ind[j] = 0.0, 1.0
    return ind

def repair_binary(ind, target, rng):
    cur = int(ind.sum())
    if cur == target:
        return ind
    diff = cur - target
    if diff > 0:
        ones = np.where(ind == 1)[0]
        to_zero = rng.choice(ones, size=diff, replace=False)
        ind[to_zero] = 0.0
    else:
        zeros = np.where(ind == 0)[0]
        to_one = rng.choice(zeros, size=-diff, replace=False)
        ind[to_one] = 1.0
    return ind
