def compute_corr_and_c5(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())

    comp_rev_fft = np.fft.rfft(np.flip(1 - h), L)

    c5 = max_corr * (2.0 / n)
    return corr, c5

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen intervals if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_corr_and_c5(h):
    """Return full linear correlation of h with 1-h and the C5 bound."""
    n = h.shape[0]
    L_fft = 1 << ((2 * n - 1).bit_length())  # next power of two
    h_fft = np.fft.rfft(h, L_fft)
    comp_rev_fft = np.fft.rfft(np.flip(1.0 - h), L_fft)
    conv = np.fft.irfft(h_fft * comp_rev_fft, L_fft)
    corr = conv[: 2 * n - 1]
    max_corr = np.max(corr)
    c5 = max_corr * (2.0 / n)
    return corr, c5


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum on the C5 objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    corr, cur_score = compute_corr_and_c5(h)
    best_h = h.copy()
    best_score = cur_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break

        # Identify worst shifts (within tolerance)
        max_corr = np.max(corr)
        # use both atol and rtol to be robust
        tol = max(1e-12, max_corr * 1e-12)
        shifts = np.where(np.abs(corr - max_corr) <= tol)[0] - (n - 1)

        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)

        corr_new, new_score = compute_corr_and_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            corr = corr_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            corr, cur_score = compute_corr_and_c5(h)
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score

    return best_h, best_score


def simulated_annealing(h0, init_temp, final_temp, max_iters, rng, time_left):
    """Simple annealing: move mass between two positions with Metropolis acceptance."""
    h = h0.copy()
    _, cur_score = compute_corr_and_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.shape[0]

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        # linear temperature schedule
        T = init_temp + (final_temp - init_temp) * (it / max_iters)

        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        # scale delta with temperature (larger moves early)
        delta = max_delta * rng.random() * (0.5 + 0.5 * (T / init_temp))
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        _, new_score = compute_corr_and_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # Initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    _, best_score = compute_corr_and_c5(best_h)

    # Multi‑scale schedule
    scales = [64, 128, 256, 512, 1024]
    for n in scales:
        if time_left() < 5.0:

        if base.shape[0] != n:

        base = project_onto_box_sum(base, n / 2.0)

        # Gradient descent phase
        max_iters_gd = int(min(20000, max(2000, time_left() * (15000.0 / n))))
        h_gd, sc_gd = gradient_descent(base, max_iters_gd, step0=0.5, beta=0.9,
                                        rng=rng, time_left=time_left)
        if sc_gd < best_score:

        base = h_gd

        # Simulated annealing phase
        max_iters_ann = int(min(30000, max(2000, time_left() * 12.0)))
        h_ann, sc_ann = simulated_annealing(base, init_temp=0.02, final_temp=0.0001,
                                            max_iters=max_iters_ann, rng=rng,
                                            time_left=time_left)
        if sc_ann < best_score:

                    max_iters_ann2 = int(min(10000, max(500, time_left() * 5.0)))
                    h_rand_ann, sc_rand_ann = simulated_annealing(h_rand_gd,
                                                                  init_temp=0.01,
                                                                  final_temp=0.0001,
                                                                  max_iters=max_iters_ann2,
                                                                  rng=rng,
                                                                  time_left=time_left)
