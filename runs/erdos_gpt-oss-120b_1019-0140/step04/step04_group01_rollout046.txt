        min_idx = np.argmin(collisions)
        offset = min_idx - (n - 1)  # integer shift
        # compute contributions
        idx_arr = np.arange(n)
        idx1 = idx_arr + offset
        idx2 = idx_arr - offset
        valid1 = (idx1 >= 0) & (idx1 < n)

        zeros = np.where(best_h == 0)[0]

        if zeros.size == 0 or ones.size == 0:
            break

        zero_contrib = contrib[zeros]
        one_contrib = contrib[ones]
        best_zero_idx = zeros[np.argmax(zero_contrib)]

def compute_collisions_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    corr = np.fft.irfft(H * np.conj(H), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def binary_greedy_local_opt(h_bin, time_left_func):
    n = h_bin.size
    h = h_bin.copy()
    best_score = compute_c5_fft(h)
    while time_left_func() > 0.2:
        collisions = compute_collisions_fft(h)  # collisions scaled by dx
        # find shift with minimal collisions
        min_idx = np.argmin(collisions)
        offset = min_idx - (n - 1)
        # compute contributions
        idx_arr = np.arange(n)
        idx1 = idx_arr + offset
        idx2 = idx_arr - offset
        # valid masks
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        contrib = np.zeros(n, dtype=int)  # contributions integer for binary
        contrib[valid1] = h[idx1[valid1]]
        contrib[valid2] += h[idx2[valid2]]
        # get zero and one indices
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break

        one_contrib = contrib[ones]

        best_one = ones[np.argmin(one_contrib)]

            new_score = compute_c5_fft(h)
            if new_score < best_score - 1e-12:
                best_score = new_score
            # continue loop
        else:
            # no improving swap
            break
    return h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array of h and (1-h), scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_collisions_fft(h):
    """Correlation of h with itself, scaled by dx (i.e., collisions)."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    corr = np.fft.irfft(H * np.conj(H), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func,
                                        step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func,
                      max_iters=400000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=1500):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_scaled_fft(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

def binary_greedy_local_opt(h_bin, time_left_func):
    """Greedy hill‑climb on binary vectors to increase minimal collisions."""
    n = h_bin.size
    h = h_bin.copy()
    best_score = compute_c5_fft(h)
    while time_left_func() > 0.2:
        coll = compute_collisions_fft(h)
        min_idx = np.argmin(coll)
        offset = min_idx - (n - 1)
        idx = np.arange(n)
        idx1 = idx + offset
        idx2 = idx - offset
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        contrib = np.zeros(n, dtype=int)
        contrib[valid1] = h[idx1[valid1]].astype(int)
        contrib[valid2] += h[idx2[valid2]].astype(int)
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        zero_contrib = contrib[zeros]
        one_contrib = contrib[ones]
        best_zero = zeros[np.argmax(zero_contrib)]
        best_one = ones[np.argmin(one_contrib)]
        if contrib[best_zero] > contrib[best_one]:
            h[best_one] = 0
            h[best_zero] = 1
            new_score = compute_c5_fft(h)
            if new_score < best_score - 1e-12:
                best_score = new_score
        else:
            break
    return h, best_score
