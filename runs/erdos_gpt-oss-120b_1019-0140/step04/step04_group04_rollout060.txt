def compute_c5(h):
    n = h.size
    # Use FFT to compute linear convolution of h and (1-h) reversed?
    # correlation h*(1-h) at shift k (where k from -(n-1) to n-1) is sum_i h[i] * (1-h[i+k]) where out-of-range indices are omitted.
    # For full linear correlation, we can compute convolution of h with reversed (1-h):
    # c = np.convolve(h, (1-h)[::-1], mode='full')
    # But np.convolve is O(n^2). We'll use FFT:
    x = h
    y = (1 - h)[::-1]
    nfft = 1 << ((2*n - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:2*n-1]
    # c corresponds to shifts from -(n-1) to (n-1)
    # compute max and multiply by dx: dx = 2/n
    return np.max(c) * (2.0 / n)
```

def compute_c5(h):
    n = h.shape[0]
    # Compute correlation via FFT
    y = 1.0 - h
    # y reversed for correlation
    y_rev = y[::-1]
    n_fft = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    Y = np.fft.rfft(y_rev, n=n_fft)
    corr = np.fft.irfft(H * Y, n=n_fft)[:2*n-1]
    return np.max(corr) * (2.0 / n)
```

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0: break
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        # get all shifts achieving max (within tolerance)
        shifts = np.where(np.abs(corr - max_corr) < 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasionally add noise
        if it % 500 == 0:

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    n = h.shape[0]
    y_rev = (1.0 - h)[::-1]
    n_fft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    Y = np.fft.rfft(y_rev, n=n_fft)
    corr = np.fft.irfft(H * Y, n=n_fft)[: 2 * n - 1]
    return corr

def compute_c5(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize(v):
    n = v.size
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    b = np.zeros_like(v)
    b[idx] = 1.0
    return b

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        shifts = np.where(np.abs(corr - max_corr) < 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score_new = compute_c5(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_fft(h) * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # base construction (n=64)
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    target_n = 512
    if base.shape[0] != target_n:
        base = upsample(base, target_n)
    base = project_onto_box_sum(base, target_n / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # ---- random binary seeds with quick local search ----
    rand_phase_end = start + min(60.0, max(0.0, time_left() - 200.0))
    while time.time() < rand_phase_end and time_left() > 30.0:
        h_rand = random_binary_balanced(target_n, rng)
        score_rand = compute_c5(h_rand)
        if score_rand < best_score:
            best_h, best_score = h_rand.copy(), score_rand
        quick_end = time.time() + 5.0
        cur_h = h_rand.copy()
        cur_score = score_rand
        while time.time() < quick_end and time_left() > 0:
            i = rng.integers(target_n)
            j = rng.integers(target_n - 1)
            if j >= i:
                j += 1
            if cur_h[i] == cur_h[j]:
                continue
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            new_score = compute_c5(cur_h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_h, best_score = cur_h.copy(), new_score
            else:
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # ---- binarize and long annealing on binary vectors ----
    binary_h = binarize(best_h)
    binary_score = compute_c5(binary_h)
    if binary_score < best_score:
        best_h, best_score = binary_h.copy(), binary_score

    remaining = time_left()
    if remaining > 20.0:
        anneal_time = min(200.0, max(0.0, remaining - 100.0))
        anneal_start = time.time()
        anneal_end = anneal_start + anneal_time
        cur_h = binary_h.copy()
        cur_score = binary_score
        n = target_n
        T0 = 0.001
        T_end = 1e-6
        while time.time() < anneal_end and time_left() > 0:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if cur_h[i] == cur_h[j]:
                continue
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            new_score = compute_c5(cur_h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_h, best_score = cur_h.copy(), new_score
            else:
                progress = (time.time() - anneal_start) / anneal_time
                T = T0 * (T_end / T0) ** progress
                if rng.random() < np.exp(-(new_score - cur_score) / T):
                    cur_score = new_score
                    if new_score < best_score:
                        best_h, best_score = cur_h.copy(), new_score
                else:
                    cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # ---- gradient descent refinement ----
    remaining = time_left()
    if remaining > 20.0:
        best_h, best_score = gradient_descent_momentum_custom(
            best_h,
            n_iter=int(1e6),
            step0=0.2,
            beta=0.95,
            rng=rng,
            time_left=time_left)

    # ---- final binary rounding and hill climbing ----
    remaining = time_left()
    if remaining > 5.0:
        binary_h2 = binarize(best_h)
        binary_score2 = compute_c5(binary_h2)
        if binary_score2 < best_score:
            best_h, best_score = binary_h2, binary_score2
        cur_h = best_h.copy()
        cur_score = best_score
        n = cur_h.size
        while time_left() > 0:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if cur_h[i] == cur_h[j]:
                continue
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            new_score = compute_c5(cur_h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_h, best_score = cur_h.copy(), new_score
            else:
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # ---- optional SLSQP polishing if time permits ----
    remaining = time_left()
    if remaining > 20.0:
        try:
            h_opt, t_opt = slsqp_optimize_candidate(best_h, max_iter=500)
            if t_opt < best_score:
                best_h, best_score = h_opt, t_opt
        except Exception:

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
