def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2*n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2*n-1]
    return corr * (2.0 / n)

def compute_subgradient(h):
    n = h.size
    corr = compute_corr_scaled_fft(h)
    max_corr = np.max(corr)
    # threshold to include all max approx:
    eps = 1e-12
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    # compute gradient similar to previous compute_gradients_for_shifts but we can compute directly
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    # average over number of shifts
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def adam_optimize(h0, time_left_func, max_iter=500000, lr0=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    h = np.copy(h0)
    n = h.shape[0]
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    t = 0
    while t < max_iter and time_left_func() > 0.0:
        t += 1
        grad = compute_subgradient(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)

            best_h = h_new.copy()
            # maybe adjust lr0 based on improvement

    return best_h, best_score

    n_target = None
    # Determine base resolution:
    if 'initial_h_values' in globals():

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # expand bounds if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    """Full correlation (h * (1-h)) scaled by dx = 2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    """Maximum overlap integral C5 for discretized h."""
    return np.max(compute_corr_scaled_fft(h))

    return grad

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

def upsample(arr, new_n):
    """Linear interpolation to a larger number of points."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808

    # Base resolution: use provided initial_h_values if present
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        h_init = rng.random(512).astype(float)
    n0 = h_init.size
    h_init = project_onto_box_sum(h_init, n0 / 2.0)

    best_h = h_init
    best_score = compute_c5_fft(best_h)

    # Primary continuous optimization at base resolution
    if time_left() > 5.0:
        h_opt, sc_opt = adam_optimize(best_h, time_left,
                                      max_iter=300000, lr0=0.5)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Random continuous restarts
    while time_left() > 8.0:
        h_rand = project_onto_box_sum(rng.random(n0), n0 / 2.0)
        h_opt, sc_opt = adam_optimize(h_rand, time_left,
                                      max_iter=60000, lr0=0.5)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size
        if time_left() < 5.0:
            break

    # SLSQP refinement at current resolution
    if time_left() > 5.0:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=1000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Upsample to higher resolution and refine
    if time_left() > 5.0:
        n_high = n0 * 2
        h_up = upsample(best_h, n_high)
        h_up = project_onto_box_sum(h_up, n_high / 2.0)
        h_opt, sc_opt = adam_optimize(h_up, time_left,
                                      max_iter=200000, lr0=0.5)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Final SLSQP pass if time permits
    if time_left() > 2.0:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=500)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Ensure feasibility before returning
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
