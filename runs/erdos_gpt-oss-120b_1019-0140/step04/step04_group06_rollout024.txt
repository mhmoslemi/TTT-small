    best_h = h.copy()

    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:

        corr = compute_corr_full(h)
        maxc = corr.max()
        # find top k
        k = 3  # maybe 3 or 5
        # get top indices
        if k < len(corr):
            idxs = np.argpartition(-corr, k-1)[:k]
        else:
            idxs = np.arange(len(corr))
        # compute shifts
        shifts = (idxs - (h.size - 1)).tolist()
        # compute gradient
        grad = np.zeros_like(h)

                term2[shift:] = h[:h.size-shift]

                term2 = np.zeros_like(h)
                term1[sft:] = h[:h.size-sft]
                term2[:h.size-sft] = h[sft:]

def compute_corr_full_fft(h):
    n = h.size
    L = 2*n
    H = np.fft.fft(h, L)
    G = np.fft.fft((1 - h)[::-1], L)
    corr = np.fft.ifft(H * G).real[:2*n-1]
    return corr

def simulated_anneal_sub(h0, compute_c5_func, rng, time_left):
    h = h0.copy()
    cur_score = compute_c5_func(h)
    best_h, best_score = h.copy(), cur_score
    n = h.size
    start = time.time()
    while time_left() > 0.1:
        # cooling schedule based on elapsed fraction
        elapsed = time.time() - start
        fraction = elapsed / (time_left() + elapsed)  # approximate progress
        # T linearly from T0 to T1
        T0, T1 = 0.01, 1e-5
        T = T0 * ((T1 / T0) ** fraction)
        # random move: select two indices i from ones, j from zeros
        # Determine indices sets quickly? we can compute binary mask: (h>0.5) approximates ones.
        # But h may be fractional. We'll treat as continuous values; propose small delta shift.
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # move delta from i to j (preserving sum)
        delta = min(h[i], 1 - h[j]) * rng.random() * 0.5  # up to 0.5 fraction
        if delta < 1e-8:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5_func(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def grad_descent_subgrad(h0, n_iter, step0, beta, rng, time_left, top_k=3):
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = compute_corr_full_fft(h)
        # top shifts
        shifts = get_top_shifts(corr, h.size, top_k)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_anneal_sub(h0, rng, time_left):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    n = h.size
    start = time.time()
    while time_left() > 0.1:
        elapsed = time.time() - start
        total_remaining = time_left()
        # progress fraction: elapsed / (elapsed + total_remaining)
        frac = elapsed / (elapsed + total_remaining) if (elapsed + total_remaining) > 0 else 0.0
        T0, T1 = 0.01, 1e-5
        T = T0 * ((T1 / T0) ** frac)
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = min(h[i], 1 - h[j]) * rng.random() * 0.5
        if delta < 1e-9:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            h, cur_score = h_new, sc_new
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()

        if rng.random() < 0.05:
            # compute gradient for max shift
            corr = compute_corr_full_fft(h)
            shifts = get_max_shifts(corr, n)
            grad = compute_gradients_for_shifts(h, shifts)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft((1 - h)[::-1], L)
    corr = np.fft.ifft(H * G).real[:2*n-1]
    return corr

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Adjust bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]

        return arr.copy()

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1 - h)[::-1], L)
    corr = np.fft.irfft(H * G, L)[:2*n-1]
    return corr

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        g = np.ones_like(h)
        if shift >= 0:
            s = shift
            if s < n:
                g[:n-s] -= h[s:]
                g[s:] -= h[:n-s]
        else:
            s = -shift
            if s < n:
                g[s:] -= h[:n-s]
                g[:n-s] -= h[s:]
        grad += g
    grad /= len(shifts)
    return grad

def grad_descent_subgrad(h0, n_iter, step0, beta, rng, start, budget_s, top_k=3):

    best_h = h.copy()

        shifts = get_top_shifts(corr, h.size, top_k)

def simulated_anneal_sub(h0, rng, start, budget_s):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    # allocate max iterations based on remaining budget; approx 200 per sec
    max_iters = int((budget_s - (time.time() - start)) * 200) + 1
    for _ in range(max_iters):
        if budget_s - (time.time() - start) <= 0:
            break
        elapsed = time.time() - start
        remain = budget_s - elapsed
        frac = elapsed / (elapsed + remain) if (elapsed + remain) > 0 else 0.0
        T0, T1 = 0.01, 1e-5
        T = T0 * ((T1 / T0) ** frac)
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = min(h[i], 1 - h[j]) * rng.random() * 0.5
        if delta < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            h, cur_score = h_new, sc_new
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        if rng.random() < 0.05:
            corr = compute_corr_full_fft(h)
            shifts = get_max_shifts(corr, n)
            grad = compute_gradients_for_shifts(h, shifts)
            h_tmp = h - 0.02 * grad
            h_tmp = project_onto_box_sum(h_tmp, n / 2.0)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < cur_score:
                h, cur_score = h_tmp, sc_tmp
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
    return best_h, best_score

        T0, T1 = 0.01, 1e-5
        T = T0 * ((T1 / T0) ** frac)
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = min(h[i], 1 - h[j]) * rng.random() * 0.5
        if delta < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            h, cur_score = h_new, sc_new
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        if rng.random() < 0.05:
            corr = compute_corr_full_fft(h)
            shifts = get_max_shifts(corr, n)
            grad = compute_gradients_for_shifts(h, shifts)
            h_tmp = h - 0.02 * grad
            h_tmp = project_onto_box_sum(h_tmp, n / 2.0)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < cur_score:
                h, cur_score = h_tmp, sc_tmp
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.zeros(64, dtype=float)
        base[:32] = 1.0
    base = project_onto_box_sum(base, base.size / 2.0)
    scales = [64, 128, 256, 512, 1024]
    best_h = None
    best_score = np.inf
    for idx, n in enumerate(scales):
        if budget_s - (time.time() - start) <= 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        remaining = budget_s - (time.time() - start)
        if remaining > 10:
            max_iters = int(min(30000, max(5000, remaining * (2000.0 / n))))
            h_gd, sc_gd = grad_descent_subgrad(base, max_iters, 0.5, 0.8,
                                               rng, start, budget_s, top_k=3)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()
            base = h_gd
        remaining = budget_s - (time.time() - start)
        if remaining > 10:
            h_ann, sc_ann = simulated_anneal_sub(base, rng, start, budget_s)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()
            base = h_ann
        remaining = budget_s - (time.time() - start)
        if remaining > 5:
            h_tmp = base.copy()
            for _ in range(5):
                i, j = rng.integers(n, size=2)
                delta = min(h_tmp[i], 1 - h_tmp[j]) * rng.random() * 0.2
                if delta > 1e-12:
                    h_tmp[i] -= delta
                    h_tmp[j] += delta
            h_tmp = project_onto_box_sum(h_tmp, n / 2.0)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < best_score:
                best_score, best_h = sc_tmp, h_tmp.copy()
                base = h_tmp
    final_n = best_h.shape[0] if best_h is not None else scales[-1]
    while budget_s - (time.time() - start) > 5:
        h_rand = random_binary_balanced(final_n, rng)
        remaining = budget_s - (time.time() - start)
        iters = int(min(15000, max(3000, remaining * 3)))
        h_dd, sc_dd = grad_descent_subgrad(h_rand, iters, 0.4, 0.9,
                                           rng, start, budget_s, top_k=3)
        if sc_dd < best_score:
            best_score, best_h = sc_dd, h_dd.copy()
        for _ in range(10):
            i, j = rng.integers(final_n, size=2)
            delta = min(best_h[i], 1 - best_h[j]) * rng.random() * 0.3
            if delta > 1e-12:
                h_new = best_h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                sc_new = compute_c5(h_new)
                if sc_new < best_score:
                    best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
