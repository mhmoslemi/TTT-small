def compute_gradient_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros(n, dtype=float)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            grad[:sz] += (1 - h[shift:])  # derivative from first term
            grad[shift:] += -h[:sz]  # derivative from second term
        else:
            s = -shift
            sz = n - s
            grad[s:] += (1 - h[:sz])
            grad[:s] += -h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    # returns projection onto [lo,hi] and sum constraint target_sum
    # Using water filling: find lambda s.t. sum(np.clip(v - lambda, lo, hi)) = target_sum
    n = v.size
    # early bounds
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        s = w.sum()
        if abs(s - target_sum) < 1e-12:
            return w
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def gradient_descent_proj(h0, max_iters, step0, beta, time_limit, rng):

    start = time.time()
    h = h0.copy()
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time.time() - start >= time_limit:
            break
        # compute correlation and max
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient for these shifts
        grad = compute_gradient_for_shifts(h, shifts)
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        # update
        h_new = h - eta * momentum
        # occasional noise
        if (it % 1000) == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, target)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        s = w.sum()
        if abs(s - target_sum) < 1e-12:
            return w
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradient_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            grad[:sz] += (1.0 - h[shift:])
            grad[shift:] += -h[:sz]
        else:
            s = -shift
            sz = n - s
            grad[s:] += (1.0 - h[:sz])
            grad[:s] += -h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_proj(h0, max_iters, step0, beta, time_limit, rng):
    start = time.time()
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time.time() - start >= time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradient_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h, time_limit, init_T, final_T, rng):
    start = time.time()
    n = h.size
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        fraction = elapsed / time_limit
        T = init_T * (1.0 - fraction) + final_T * fraction
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        scale = 0.5 + 0.5 * (T / init_T)  # between 0.5 and 1.0
        delta = max_delta * rng.random() * scale
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = compute_c5(cur_h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_score = np.inf
    scales = [64, 128, 256, 512]
    # base vector for the first scale
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    for idx, n in enumerate(scales):
        # time allocation proportional to n
        remaining = budget_s - (time.time() - start)
        if remaining <= 1.0:
            break
        sum_remaining_scales = sum(scales[idx:])
        scale_time = remaining * (n / sum_remaining_scales)
        # prepare base at correct resolution
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        # generate a few candidates
        candidates = [base]
        if (n & (n - 1)) == 0:  # power of two
            tm = thue_morse(n)
            tm = project_onto_box_sum(tm, n / 2.0)
            candidates.append(tm)
        for _ in range(3):
            candidates.append(random_binary_balanced(n, rng))
        for cand in candidates:
            sc = compute_c5(cand)
            if sc < best_score:
                best_score, best_h = sc, cand.copy()
        # allocate half of scale_time to gradient descent, half to annealing
        gd_time = scale_time * 0.5
        ann_time = scale_time - gd_time
        # gradient descent
        max_iters = int(gd_time * 2000)  # rough estimate
        if max_iters > 0:
            h_gd, sc_gd = gradient_descent_proj(best_h.copy(), max_iters,
                                                step0=0.4, beta=0.9,
                                                time_limit=gd_time, rng=rng)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()
        # simulated annealing
        if ann_time > 0.5:
            h_ann, sc_ann = simulated_annealing(best_h.copy(),
                                                time_limit=ann_time,
                                                init_T=0.02, final_T=1e-6,
                                                rng=rng)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()
        # optional short gradient refinement after annealing
        remaining_scale = scale_time - (time.time() - start)
        if remaining_scale > 0.5:
            extra_iters = int(remaining_scale * 2000)
            if extra_iters > 0:
                h_ref, sc_ref = gradient_descent_proj(best_h.copy(),
                                                      extra_iters,
                                                      step0=0.2, beta=0.9,
                                                      time_limit=remaining_scale,
                                                      rng=rng)
                if sc_ref < best_score:
                    best_score, best_h = sc_ref, h_ref.copy()
        # set base for next scale
        base = best_h.copy()
    # final refinement with any leftover time
    leftover = budget_s - (time.time() - start)
    if leftover > 5.0:
        final_iters = int(leftover * 2000)
        if final_iters > 0:
            best_h, best_score = gradient_descent_proj(best_h.copy(),
                                                       final_iters,
                                                       step0=0.2, beta=0.9,
                                                       time_limit=leftover,
                                                       rng=rng)
    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
