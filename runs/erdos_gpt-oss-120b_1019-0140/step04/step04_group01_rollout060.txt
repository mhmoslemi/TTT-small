    if f(lam_low) < s:
        lam_low = np.min(v) - 10*hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10*lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1[i] = h[i+shift] for i < n-shift
            # term2[i+shift] = h[i] for i < n-shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    # returns binary values 0/1
    # Use vectorized method: parity of number of 1 bits.
    # Could use np.unpackbits but n may not be power of 2.
    # Simpler: compute using bit counts
    i = np.arange(n, dtype=np.uint64)
    # compute popcount via builtin np.binary_repr? but not efficient. Could use bitwise operations.
    # Use vectorized bit count using numpy's builtin popcount? Not available.
    # We'll compute using Python's bin count function via fromiter as before.
    return np.fromiter(((bin(v).count('1') & 1) for v in range(n)), dtype=float, count=n)

def adam_optimize(h0, time_left, max_iter=200000, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    t = 1
    while t <= max_iter and time_left() > 0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # find all shifts achieving max
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        t += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=2000):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_scaled_fft(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n/2.0)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

    if best_h is not None and time_left() > 5.0:

        n = best_h.size

        idx = np.argpartition(-best_h, k)[:k]

            n = h.size

            best_h_loc = h.copy()
            best_score_loc = current_score
            T = T0
            it = 0
            while it < max_iters and time_left() > 0.01:
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i] = 0.0
                h[j] = 1.0
                new_score = compute_c5_fft(h)
                delta = new_score - current_score
                accept = False
                if delta < 0:
                    accept = True
                else:
                    if T > 0 and rng.random() < np.exp(-delta/T):
                        accept = True
                if accept:

                        best_score_loc = new_score
                        best_h_loc = h.copy()
                else:
                    h[i] = 1.0
                    h[j] = 0.0
                T = max(Tmin, T * decay)
                it += 1
            return best_h_loc, best_score_loc
        h_bin_opt, sc_bin_opt = binary_swap_anneal(h_bin, rng, time_left)
        update_best(h_bin_opt)
        if best_score <= target:
            return best_h, best_score, n_target

    # final fallback: return best found or fallback to simple construction
    if best_h is None:
        # fallback simple balanced continuous: all 0.5
        h_fallback = np.full(n_target, 0.5)
        h_fallback = project_onto_box_sum(h_fallback, n_target/2.0)
        best_h = h_fallback
        best_score = compute_c5_fft(best_h)
    return best_h, best_score, n_target

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo≤x_i≤hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += 1.0 - term1 - term2
    if shifts:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def adam_optimize(h0, start, budget_s, max_iter=200000, lr=0.1,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    t = 1
    while t <= max_iter:
        if time.time() - start > budget_s:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        t += 1
    return best_h, best_score

def slsqp_obj(x, n):
    return x[n]

def slsqp_eq_sum(x, n):
    return np.sum(x[:n]) - n / 2.0

def slsqp_ineq_fun(x, n):
    h = x[:n]
    t = x[n]
    corr = compute_corr_scaled_fft(h)
    return t - corr

def slsqp_optimize_candidate(h0, max_iter=2000):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': slsqp_eq_sum, 'args': (n,)},
        {'type': 'ineq', 'fun': slsqp_ineq_fun, 'args': (n,)}
    ]
    try:
        res = opt.minimize(slsqp_obj, x0, args=(n,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

def binary_swap_anneal(h0, rng, start, budget_s,
                      max_iters=200000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters:
        if time.time() - start > budget_s:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    target = 0.3808
    best_h = None
    best_score = np.inf
    n_target = 1024  # resolution

    def update_best(h):
        nonlocal best_h, best_score
        sc = compute_c5_fft(h)
        if sc < best_score:
            best_score, best_h = sc, h.copy()

    # initial candidates
    cand_list = []

    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.size != n_target:
            init_arr = upsample(init_arr, n_target)
        init_arr = project_onto_box_sum(init_arr, n_target / 2.0)
        cand_list.append(init_arr)

    cand_list.append(project_onto_box_sum(thue_morse(n_target), n_target / 2.0))

    for _ in range(4):
        v = rng.random(n_target)
        v = project_onto_box_sum(v, n_target / 2.0)
        cand_list.append(v)

    # short Adam runs on each candidate
    for h0 in cand_list:
        if time.time() - start > budget_s - 5.0:
            break
        h_opt, _ = adam_optimize(h0, start, budget_s, max_iter=5000, lr=0.2)
        update_best(h_opt)
        if best_score <= target:
            return best_h, best_score, n_target

    # longer Adam on best solution
    if best_h is not None and time.time() - start < budget_s - 5.0:
        h_opt, _ = adam_optimize(best_h, start, budget_s, max_iter=100000, lr=0.2)
        update_best(h_opt)
        if best_score <= target:
            return best_h, best_score, n_target

    # SLSQP refinement
    if best_h is not None and time.time() - start < budget_s - 5.0:
        h_opt, _ = slsqp_optimize_candidate(best_h, max_iter=4000)
        update_best(h_opt)
        if best_score <= target:
            return best_h, best_score, n_target

    # binary annealing (fallback)
    if best_h is not None and time.time() - start < budget_s - 5.0:
        n = best_h.size
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        h_bin_opt, _ = binary_swap_anneal(h_bin, rng, start, budget_s,
                                          max_iters=50000)
        update_best(h_bin_opt)
        if best_score <= target:
            return best_h, best_score, n_target

    # final fallback if nothing found
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_target, 0.5), n_target / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, n_target
