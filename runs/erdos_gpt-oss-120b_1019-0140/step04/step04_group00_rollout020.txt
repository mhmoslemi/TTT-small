import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n: return np.full_like(v, lo)
    if s >= hi * n: return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    # expand bounds if needed
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    ft_h = np.fft.rfft(h_rev, n=N)
    ft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(ft_h * ft_one, n=N)[:2 * n - 1]
    return conv.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.empty(n); term1[:n_s] = h[shift:]; term1[n_s:] = 0.0
        term2 = np.empty(n); term2[shift:] = h[:n_s]; term2[:shift] = 0.0
    else:
        s = -shift
        n_s = n - s
        term1 = np.empty(n); term1[s:] = h[:n_s]; term1[:s] = 0.0
        term2 = np.empty(n); term2[:n_s] = h[s:]; term2[n_s:] = 0.0
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.size
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    ft_h = np.fft.rfft(h_rev, n=N)
    ft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(ft_h * ft_one, n=N)[:2 * n - 1]
    mx = conv.max()
    idxes = np.where(conv == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent(h0, max_iter, time_left, step0=0.5, beta=0.9):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def to_binary(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h

def binary_swap_anneal(h0, rng, time_left, T0=0.3, T_min=1e-6, decay=0.9999):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = T0
    n = h.size
    while time_left() > 0.5 and T > T_min:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T *= decay
    return best_h, best_s

    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))
    n = 512
    target_sum = n / 2.0
    best_h = None
    best_s = np.inf
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        candidates.append(project_onto_box_sum(init, target_sum))
    candidates.append(thue_morse_binary(n))
    for _ in range(2):
        h = np.zeros(n)
        h[rng.choice(n, size=n//2, replace=False)] = 1.0
        candidates.append(h)
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))
    # Gradient descent from candidates
    for c in candidates:
        if time_left() < 5.0:
            break
        h_opt, s_opt = gradient_descent(c, max_iter=20000, time_left=time_left, step0=0.5, beta=0.9)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= 0.3808:
            return best_h, best_s, n
    # Binary annealing on best
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = project_onto_box_sum(best_h, target_sum)
        best_s = compute_c5_fast(best_h)
    h_bin = to_binary(best_h)
    s_bin = compute_c5_fast(h_bin)
    if s_bin < best_s:
        best_s, best_h = s_bin, h_bin.copy()
    if best_s <= 0.3808:
        return best_h, best_s, n
    h_ann, s_ann = binary_swap_anneal(h_bin, rng, time_left)
    if s_ann < best_s:
        best_s, best_h = s_ann, h_ann.copy()
    if best_s <= 0.3808:
        return best_h, best_s, n
    # Final refinement
    h_ref, s_ref = gradient_descent(best_h, max_iter=50000, time_left=time_left, step0=0.5, beta=0.9)
    if s_ref < best_s:
        best_s, best_h = s_ref, h_ref.copy()
    best_h = project_onto_box_sum(best_h, target_sum)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s, n

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    ft_h = np.fft.rfft(h_rev, n=N)
    ft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(ft_h * ft_one, n=N)[:2 * n - 1]
    return conv.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.empty(n); term1[:n_s] = h[shift:]; term1[n_s:] = 0.0
        term2 = np.empty(n); term2[shift:] = h[:n_s]; term2[:shift] = 0.0
    else:
        s = -shift
        n_s = n - s
        term1 = np.empty(n); term1[s:] = h[:n_s]; term1[:s] = 0.0
        term2 = np.empty(n); term2[:n_s] = h[s:]; term2[n_s:] = 0.0
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.size
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    ft_h = np.fft.rfft(h_rev, n=N)
    ft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(ft_h * ft_one, n=N)[:2 * n - 1]
    mx = conv.max()
    idxes = np.where(conv == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent(h0, max_iter, time_left, step0=0.5, beta=0.9):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def to_binary(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h

def binary_swap_anneal(h0, rng, time_left, T0=0.3, T_min=1e-6, decay=0.9999):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = T0
    while time_left() > 0.5 and T > T_min:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T *= decay
    return best_h, best_s

    n = h0.size
    x0 = np.concatenate([h0, [compute_c5_fast(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x): return np.sum(x[:n]) - n / 2.0
    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5_fast(h_opt)
    except Exception:
        return h0, compute_c5_fast(h0)
