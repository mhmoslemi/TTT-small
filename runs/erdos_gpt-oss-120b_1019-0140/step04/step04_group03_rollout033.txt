import time

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=30):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_balanced(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0, max_iter=30)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(
    h0, n_iter, step0, beta, rng, max_time, proj_iter=30
):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:

        h_new = project_onto_box_sum(h_new, target_sum, max_iter=proj_iter)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
