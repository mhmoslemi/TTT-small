def compute_gradient(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            # term1 = h shifted left by s (with zero fill)
            # term1[i] = h[i+s] for i in [0, n-s-1]; else 0
            # term2 are h shifted right by s
            term1 = np.zeros_like(h)
            term1[:n - s] = h[s:]
            term2 = np.zeros_like(h)
            term2[s:] = h[:n - s]
        else:
            s2 = -s
            term1 = np.zeros_like(h)
            term1[s2:] = h[:n - s2]
            term2 = np.zeros_like(h)
            term2[:n - s2] = h[s2:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    a = np.array([0], dtype=float)

def compute_c5_and_shifts(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    shifts = np.where(corr == maxc)[0] - (n - 1)
    return maxc * (2.0 / n), shifts.tolist()

    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    v_clipped = np.clip(v, lo, hi)
    total = v_clipped.sum()
    if abs(total - s) < tol:
        return v_clipped
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # binary search for lambda
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < tol:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_and_shifts(h):
    """Return (C5 value, list of shifts achieving the max)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    shifts = np.where(corr == maxc)[0] - (n - 1)
    return maxc * (2.0 / n), shifts.tolist()

def compute_gradient(h, shifts):
    """Gradient of the max overlap w.r.t. h (averaged over given shifts)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - s] = h[s:]
            term2[s:] = h[:n - s]
        else:
            s2 = -s
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s2:] = h[:n - s2]
            term2[:n - s2] = h[s2:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    """Linear interpolation upsampling."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Generate the first n values of the Thue–Morse sequence (0/1)."""
    a = np.array([0.0])
    while a.size < n:
        a = np.concatenate([a, 1.0 - a])
    return a[:n]

def random_binary_balanced(n, rng):
    """Random 0/1 vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # ---- initial vector ----------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    target_n = 512
    if base.shape[0] != target_n:
        base = upsample(base, target_n)
    base = project_onto_box_sum(base, target_n / 2.0)

    # ---- main continuous optimisation --------------------------------------
    n = target_n
    sum_target = n / 2.0
    h = base.copy()
    best_h = h.copy()
    best_score, _ = compute_c5_and_shifts(best_h)
    momentum = np.zeros_like(h)
    beta = 0.9
    step0 = 2.0
    it = 0
    no_improve = 0
    restart_thresh = 10000

    while time_left() > 5.0:
        it += 1
        # worst shift(s) and associated gradient
        _, shifts = compute_c5_and_shifts(h)
        grad = compute_gradient(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, sum_target)

        # directed mass transfer based on gradient signs
        if grad.max() > 0 and grad.min() < 0:
            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))
            delta = min(h_new[i_max], 1.0 - h_new[i_min]) * 0.1
            if delta > 0:
                h_new[i_max] -= delta
                h_new[i_min] += delta

        cur_score, _ = compute_c5_and_shifts(h_new)

        if cur_score < best_score:
            best_score = cur_score
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h = h_new

        # occasional jitter to escape shallow minima
        if it % 2000 == 0:
            h = h + rng.normal(scale=0.02, size=n)
            h = project_onto_box_sum(h, sum_target)

        # restart if stagnated
        if no_improve >= restart_thresh and time_left() > 30.0:
            h = random_binary_balanced(n, rng)
            momentum = np.zeros_like(h)
            no_improve = 0

        if best_score <= 0.3808:
            break

    # ---- upscale and fine‑tune if time permits -----------------------------
    if time_left() > 30.0:
        n2 = n * 2
        h_up = upsample(best_h, n2)
        h_up = project_onto_box_sum(h_up, n2 / 2.0)
        sum_target2 = n2 / 2.0
        momentum = np.zeros_like(h_up)
        best_h2 = h_up.copy()
        best_score2, _ = compute_c5_and_shifts(best_h2)
        beta2 = 0.9
        step0_2 = 0.8
        it2 = 0
        no_improve2 = 0
        restart_thresh2 = 10000

        while time_left() > 5.0:
            it2 += 1
            _, shifts2 = compute_c5_and_shifts(h_up)
            grad2 = compute_gradient(h_up, shifts2)
            momentum = beta2 * momentum + (1.0 - beta2) * grad2
            step2 = step0_2 / math.sqrt(it2)
            h_new2 = h_up - step2 * momentum
            h_new2 = project_onto_box_sum(h_new2, sum_target2)

            # directed mass transfer
            if grad2.max() > 0 and grad2.min() < 0:
                i_max = int(np.argmax(grad2))
                i_min = int(np.argmin(grad2))
                delta = min(h_new2[i_max], 1.0 - h_new2[i_min]) * 0.1
                if delta > 0:
                    h_new2[i_max] -= delta
                    h_new2[i_min] += delta

            cur_score2, _ = compute_c5_and_shifts(h_new2)

            if cur_score2 < best_score2:
                best_score2 = cur_score2
                best_h2 = h_new2.copy()
                no_improve2 = 0
            else:
                no_improve2 += 1

            h_up = h_new2

            if it2 % 2000 == 0:
                h_up = h_up + rng.normal(scale=0.01, size=n2)
                h_up = project_onto_box_sum(h_up, sum_target2)

            if no_improve2 >= restart_thresh2 and time_left() > 20.0:
                h_up = random_binary_balanced(n2, rng)
                momentum = np.zeros_like(h_up)
                no_improve2 = 0

            if best_score2 <= 0.3808:
                break

        best_h = best_h2
        best_score = best_score2

    # final evaluation (ensure exact projection)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score, _ = compute_c5_and_shifts(best_h)
    return best_h, final_score, best_h.size
