def compute_corr(h):
    n = h.shape[0]
    g = 1.0 - h
    size = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(g[::-1], n=size)
    conv = np.fft.irfft(H * G, n=size)
    # Take first 2*n-1 entries.
    corr = conv[:2*n-1]
    return corr

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):

    if f_low < target_sum:

        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)

def compute_corr(h, dx, size):
    g_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(g_rev, n=size)
    conv = np.fft.irfft(H * G, n=size)[:2 * h.shape[0] - 1]
    return conv * dx

def compute_c5(h, dx, size):
    return np.max(compute_corr(h, dx, size))

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # choose resolution
    n = 1024 if budget_s >= 800 else 512
    dx = 2.0 / n
    size = 1 << ((2 * n - 1).bit_length())
    target_sum = n / 2.0

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(n)
    if base.shape[0] != n:
        base = upsample(base, n)
    base = project_onto_box_sum(base, target_sum)

    h = base.copy()
    best_h = h.copy()
    best_score = compute_c5(h, dx, size)

    it = 0
    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining <= 0.2:
            break

        # correlation & worst shift
        corr = compute_corr(h, dx, size)
        max_corr = np.max(corr)
        shift = int(np.argmax(corr) - (n - 1))

        if max_corr < best_score:
            best_score = max_corr
            best_h = h.copy()

        # overlapping ranges
        if shift >= 0:
            i_start, i_end = 0, n - shift
            j_start, j_end = shift, n
        else:
            i_start, i_end = -shift, n
            j_start, j_end = 0, n + shift

        contrib = h[i_start:i_end] * (1.0 - h[j_start:j_end])

        if contrib.size == 0 or np.all(contrib <= 0):
            # random feasible move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while i == j:
                j = rng.integers(0, n)
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
        else:
            idx_local = int(np.argmax(contrib))
            i = i_start + idx_local
            j = j_start + idx_local
            x = h[i]
            y = h[j]
            max_delta = min(x, 1.0 - y)
            if max_delta <= 0:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while i == j:
                    j = rng.integers(0, n)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random()
            else:
                delta_opt = ((1.0 - y) + x) * 0.5
                delta = min(delta_opt, max_delta)

        # time‑dependent scaling (avoid too tiny steps at end)
        scale_factor = remaining / budget_s
        if scale_factor < 0.1:
            scale_factor = 0.1
        delta *= scale_factor
        if delta < 1e-12:
            delta = max_delta * 0.01

        h[i] -= delta
        h[j] += delta
        # safety clamp
        if h[i] < 0.0:
            h[i] = 0.0
        elif h[i] > 1.0:
            h[i] = 1.0
        if h[j] < 0.0:
            h[j] = 0.0
        elif h[j] > 1.0:
            h[j] = 1.0

        it += 1
        if it % 2000 == 0:
            h = project_onto_box_sum(h, target_sum)

    # optional simulated‑annealing refinement if still above target
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    TARGET = 0.3808
    if remaining > 5.0 and best_score > TARGET:
        cur_h = best_h.copy()
        cur_score = best_score
        total_steps = int(min(50000, max(5000, remaining * 800.0)))
        T0 = 0.02
        T_final = 1e-6
        for step in range(total_steps):
            if time.time() - start >= budget_s:
                break
            T = T0 * ((total_steps - step) / total_steps) + T_final
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while i == j:
                j = rng.integers(0, n)
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
            h_new = cur_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            new_score = compute_c5(h_new, dx, size)
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                cur_h = h_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = cur_h.copy()

    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h, dx, size)
    return best_h, best_score, n
