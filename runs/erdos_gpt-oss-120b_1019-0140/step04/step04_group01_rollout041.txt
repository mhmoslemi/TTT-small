import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx."""
    return np.max(compute_corr_scaled_fft(h))

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            if shift < n:
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            if s < n:
                term1[s:] = h[:n - s]
                term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=400000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=1500):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_scaled_fft(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

        best_score, best_h = sc_ref, h_ref.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # ---------- Phase 2: medium resolution (n=256) ----------
    n_mid = 256
    h_mid = upsample(best_h, n_mid)
    h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
    sc_mid = compute_c5_fft(h_mid)
    if sc_mid < best_score:
        best_score, best_h = sc_mid, h_mid.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    if time_left() > 2.0:
        h_mid_opt, sc_mid_opt = slsqp_optimize_candidate(h_mid, max_iter=1500)
        if sc_mid_opt < best_score:
            best_score, best_h = sc_mid_opt, h_mid_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    if time_left() > 2.0:
        h_mid_ref, sc_mid_ref = gradient_descent_momentum_with_time(best_h, time_left,
                                                                    step0=0.5, beta=0.9, max_iter=100000)
        if sc_mid_ref < best_score:
            best_score, best_h = sc_mid_ref, h_mid_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 3: large resolution (n=512) ----------
    n_large = 512
    h_large = upsample(best_h, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    sc_large = compute_c5_fft(h_large)
    if sc_large < best_score:
        best_score, best_h = sc_large, h_large.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    if time_left() > 2.0:
        h_large_opt, sc_large_opt = slsqp_optimize_candidate(h_large, max_iter=800)
        if sc_large_opt < best_score:
            best_score, best_h = sc_large_opt, h_large_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 4: very large resolution (n=1024) ----------
    n_final = 1024
    h_final = upsample(best_h, n_final)
    h_final = project_onto_box_sum(h_final, n_final / 2.0)
    sc_final = compute_c5_fft(h_final)
    if sc_final < best_score:
        best_score, best_h = sc_final, h_final.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # Optional continuous refinements at n=1024
    if time_left() > 5.0:
        h_opt_cont, sc_opt_cont = slsqp_optimize_candidate(h_final, max_iter=800)
        if sc_opt_cont < best_score:
            best_score, best_h = sc_opt_cont, h_opt_cont.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    if time_left() > 8.0:
        h_cont_ref, sc_cont_ref = gradient_descent_momentum_with_time(best_h, time_left,
                                                                     step0=0.5, beta=0.9, max_iter=80000)
        if sc_cont_ref < best_score:
            best_score, best_h = sc_cont_ref, h_cont_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Random continuous restarts (n=1024)
    while time_left() > 10.0:
        v = rng.random(n_final)
        v = project_onto_box_sum(v, n_final / 2.0)
        h_tmp, sc_tmp = gradient_descent_momentum_with_time(v, time_left,
                                                             step0=0.5, beta=0.9, max_iter=20000)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 5: binary conversion ----------
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5_fft(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # ---------- Phase 6: binary annealing ----------
    if time_left() > 0.5:
        h_bin_ref, sc_bin_ref = binary_swap_anneal(best_h, rng, time_left,
                                                  max_iters=10_000_000,
                                                  T0=0.2, decay=0.99995, Tmin=1e-7)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 7: greedy local improvement ----------
    if time_left() > 0.1:
        h_cur = best_h.copy()
        cur_score = best_score
        n = h_cur.size
        attempts = 0
        while time_left() > 0.0 and attempts < 200000:
            attempts += 1
            ones = np.where(h_cur > 0.5)[0]
            zeros = np.where(h_cur <= 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_cur[i], h_cur[j] = 0.0, 1.0
            new_score = compute_c5_fft(h_cur)
            if new_score <= cur_score:
                cur_score = new_score
                # recompute sets for next iteration
                # (cost is acceptable given limited attempts)
                if new_score < best_score:
                    best_score, best_h = new_score, h_cur.copy()
                    if best_score <= target:
                        return best_h, best_score, best_h.size
            else:
                h_cur[i], h_cur[j] = 1.0, 0.0

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
