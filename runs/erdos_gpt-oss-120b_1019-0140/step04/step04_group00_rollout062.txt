def compute_obj_and_grad(h, alpha):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    raw_corr = np.fft.irfft(fft_rev * fft_one, n=N)[:2 * n - 1]
    max_corr = raw_corr.max()
    # compute unnormalized weights
    exp_term = np.exp(alpha * (raw_corr - max_corr))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    # compute FFTs of h and w and w_rev
    fft_h = np.fft.rfft(h, n=N)
    fft_w = np.fft.rfft(w, n=N)
    conv_back = np.fft.irfft(fft_w * fft_h, n=N)[:2 * n - 1]
    w_rev = w[::-1]
    fft_w_rev = np.fft.rfft(w_rev, n=N)
    conv_fwd = np.fft.irfft(fft_w_rev * fft_h, n=N)[:2 * n - 1]
    start = n - 1
    B = conv_back[start:start + n]
    F = conv_fwd[start:start + n]
    grad = 1.0 - (F + B)
    smooth_obj = max_corr + np.log(w_sum) / alpha
    C5 = smooth_obj * (2.0 / n)
    return C5, grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    """Project vector v onto the box [lo,hi] while fixing sum(v)=s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # expand bounds if needed
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    """Exact C5 for discrete h (dx = 2/n)."""
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def compute_obj_and_grad(h, alpha):
    """Smooth max C5 and its subgradient for given alpha."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    raw_corr = np.fft.irfft(fft_rev * fft_one, n=N)[:2 * n - 1]
    max_corr = raw_corr.max()
    # weights for smooth max
    exp_term = np.exp(alpha * (raw_corr - max_corr))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    # FFT of h (used for both convolutions)
    fft_h = np.fft.rfft(h, n=N)
    # convolution with w (backward)
    fft_w = np.fft.rfft(w, n=N)
    conv_back = np.fft.irfft(fft_w * fft_h, n=N)[:2 * n - 1]
    # convolution with reversed w (forward)
    w_rev = w[::-1]
    fft_w_rev = np.fft.rfft(w_rev, n=N)
    conv_fwd = np.fft.irfft(fft_w_rev * fft_h, n=N)[:2 * n - 1]
    start = n - 1
    B = conv_back[start:start + n]
    F = conv_fwd[start:start + n]
    grad = 1.0 - (F + B)          # subgradient of smooth max (raw scale)
    smooth_obj = max_corr + np.log(w_sum) / alpha
    C5 = smooth_obj * (2.0 / n)   # scaled to integral
    return C5, grad

def smooth_grad_descent(h0, target_sum, time_left,
                       max_iter=20000, step0=0.5, beta=0.9,
                       alpha_start=0.5, alpha_factor=1.05,
                       alpha_max=50.0, alpha_update_interval=500):
    """Projected momentum descent on smooth max C5."""
    n = h0.size
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    alpha = alpha_start
    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        _, grad = compute_obj_and_grad(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target_sum)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        if it % alpha_update_interval == 0:
            alpha = min(alpha * alpha_factor, alpha_max)
    # final projection to ensure feasibility
    best_h = project_onto_box_sum(best_h, target_sum)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s

def resample(arr, new_n):
    """Linear interpolation resampling of a 1‑D array."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_h(h, target_sum):
    """Return a binary vector with exactly target_sum ones (largest entries)."""
    k = int(round(target_sum))
    idx = np.argpartition(-h, k)[:k]
    hb = np.zeros_like(h)
    hb[idx] = 1.0
    return hb

def binary_swap_anneal(h, rng, time_left, max_iters=200000):
    """Classic binary swap simulated annealing (preserves number of ones)."""
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    n = h.size
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def continuous_swap_anneal(h, rng, time_left, max_iters=200000, step_scale=0.5):
    """Simulated annealing on real-valued h preserving sum and box constraints."""
    n = h.size
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99995
    it = 0
    while it < max_iters and time_left() > 0.01:
        i, j = rng.choice(n, size=2, replace=False)
        # feasible delta range
        delta_max = min(1.0 - h[i], h[j])
        delta_min = -min(h[i], 1.0 - h[j])
        if delta_max <= delta_min:
            it += 1
            continue
        delta = step_scale * (rng.random() * (delta_max - delta_min) + delta_min)
        if abs(delta) < 1e-12:
            it += 1
            continue
        h[i] += delta
        h[j] -= delta
        new = compute_c5_fast(h)
        deltaC = new - cur
        if deltaC < 0 or rng.random() < np.exp(-deltaC / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s
