def compute_softmax_grad(h, alpha):
    n = h.size
    # correlation (no scaling)
    # compute corr_sum = sum_i h_i * (1 - h_{i+k}) (no dx)
    M = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2*n-1]  # not scaled
    # softmax weights
    # stabilize: subtract max
    maxc = np.max(alpha * corr)
    exp_term = np.exp(alpha * corr - maxc)
    w = exp_term / np.sum(exp_term)
    # compute convolution terms
    # w_rev = reversed w
    w_rev = w[::-1]
    # define convolution function using FFT
    # compute A = conv(w_rev, 1 - h)
    # compute B = conv(w, h)
    # conv length = len(w) + n - 1 = (2n-1) + n - 1 = 3n-2
    L = len(w) + n - 1
    M2 = 1 << (L.bit_length())
    # pad arrays
    A_fft = np.fft.rfft(w_rev, n=M2)
    B_fft = np.fft.rfft(w, n=M2)
    C_fft = np.fft.rfft(1.0 - h, n=M2)
    D_fft = np.fft.rfft(h, n=M2)

    B_full = np.fft.irfft(B_fft * D_fft, n=M2)[:L]

    start = n - 1

    B = B_full[start:end]

    # compute score: max coro * dx
    score = np.max(corr) * (2.0 / n)  # scaling to integral
    return grad, score

    # Determine M2 as next power of 2 >= 3n-2 (since that's max conv length)
    L = 3*n - 2
    M2 = 1 << ((L).bit_length())
    # FFT of h and (1-h) at length M2
    H = np.fft.rfft(h, n=M2)
    G = np.fft.rfft(1.0 - h, n=M2)
    # correlation (full) length 2n-1; compute via inverse FFT:

    # compute softmax weights

    # compute w_rev

def binary_swap_anneal(h0, rng, time_left_func, max_iters=200000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        # find a 1 and a 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

import time

# --- projection onto box and sum ---
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):

    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1] * (2.0 / n)
    return np.max(corr)

    maxc = np.max(alpha * corr_full)
    exp_term = np.exp(alpha * corr_full - maxc)
    w = exp_term / np.sum(exp_term)               # shape (2n-1,)
    w_rev = w[::-1]
    # FFTs of the weight vectors at the same length M
    w_fft = np.fft.rfft(w, n=M)
    w_rev_fft = np.fft.rfft(w_rev, n=M)
    # Convolutions to obtain A and B (full length 3n-2)
    A_full = np.fft.irfft(w_rev_fft * G, n=M)[:L]
    B_full = np.fft.irfft(w_fft * H, n=M)[:L]
    # slice the central n entries (indices n-1 .. 2n-2)
    start = n - 1
    end = start + n
    A = A_full[start:end]
    B = B_full[start:end]
    grad = A - B
    score = np.max(corr_full) * (2.0 / n)
    return grad, score

# --- projected gradient descent using softmax gradient ---
def projected_gradient_descent_softmax(h0, time_left_func, max_iter=20000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    eta0 = 0.5
    alpha = 1.0
    for it in range(1, max_iter + 1):
        if time_left_func() <= 0.0:
            break
        grad, cur_score = compute_softmax_grad(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fft(h_new)
        if new_score < best_score:
            best_score, best_h = new_score, h_new.copy()
        h = h_new
        alpha = min(alpha * 1.005, 200.0)   # slowly sharpen the max
    return best_h, best_score

# --- binary swap annealing ---
def binary_swap_anneal(h0, rng, time_left_func, max_iters=200000,
                      T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = delta < 0.0 or (T > 0.0 and rng.random() < np.exp(-delta / T))
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

# --- simple hill climbing for binary vectors ---
def binary_hill_climb(h0, rng, time_left_func, max_iters=20000):
    h = h0.copy()
    best_score = compute_c5_fft(h)
    n = h.size
    for _ in range(max_iters):
        if time_left_func() <= 0.0:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        if new_score < best_score:
            best_score = new_score
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, best_score

# --- resampling helper (linear interpolation on [0,2)) ---
def resample_h(arr, new_n):
    old_n = arr.size
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

# --- main entry point ---
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    best_h = None
    best_score = np.inf

    # initial vector (may be provided by parent)
    init_arr = globals().get('initial_h_values')
    if init_arr is None:
        # create a balanced binary vector of size 512
        n0 = 512
        init_arr = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, n0 // 2, replace=False)
        init_arr[idx] = 1.0
    else:
        init_arr = np.asarray(init_arr, dtype=float)

    # sequence of resolutions to explore
    for n in (256, 512, 1024):
        if time_left() < 2.0:
            break
        h0 = resample_h(init_arr, n)
        h0 = project_onto_box_sum(h0, n / 2.0)

        # continuous refinement
        max_iter_cont = {256: 5000, 512: 20000, 1024: 40000}[n]
        h_cont, sc_cont = projected_gradient_descent_softmax(h0, time_left, max_iter=max_iter_cont)
        if sc_cont < best_score:
            best_score, best_h = sc_cont, h_cont.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

        # binary rounding (top half)
        k = n // 2
        idx_top = np.argpartition(-h_cont, k)[:k]
        h_bin = np.zeros_like(h_cont)
        h_bin[idx_top] = 1.0
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

        # annealing starting from the best binary candidate we have so far
        start_bin = best_h if best_h is not None and best_h.size == n else h_bin
        h_ann, sc_ann = binary_swap_anneal(start_bin, rng, time_left,
                                           max_iters=200000, T0=0.2)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

        # post‑anneal hill climbing
        h_hc, sc_hc = binary_hill_climb(best_h, rng, time_left, max_iters=20000)
        if sc_hc < best_score:
            best_score, best_h = sc_hc, h_hc.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # final random binary restarts (if time permits)
    n_large = 1024
    while time_left() > 5.0:
        h0 = np.zeros(n_large, dtype=float)
        idx = rng.choice(n_large, n_large // 2, replace=False)
        h0[idx] = 1.0
        h_ann, sc_ann = binary_swap_anneal(h0, rng, time_left,
                                           max_iters=200000, T0=0.2)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()
            if best_score <= target:
                break

    # ensure feasibility of the returned vector
    if best_h is None:
        # fallback deterministic binary vector
        n = 512
        best_h = np.zeros(n, dtype=float)
        best_h[:n // 2] = 1.0
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
