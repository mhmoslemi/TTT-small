def simulated_annealing(h0, time_limit, rng):
    n = len(h0)
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    # target total iterations; but we stop based on time
    # Define linear schedule based on elapsed fraction
    init_temp = 0.05
    final_temp = 0.001
    # We'll compute total time limit and use that

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        temp = init_temp * (final_temp / init_temp) ** t_frac

        # pick i,j
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        # delta as fraction of max_delta scaled by temp relative to init_temp perhaps
        scale = temp / init_temp
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            continue
        # apply move
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def gradient_correlation(h, shift):
    """Gradient of the correlation c[shift] with respect to h."""
    n = h.shape[0]
    grad = np.zeros(n, dtype=float)
    if shift == 0:
        grad = 1.0 - 2.0 * h
        return grad
    if shift > 0:
        k = shift
        # +1 for left part
        grad[:n - k] += 1.0
        # -h[p+k] for left part
        grad[:n - k] -= h[k:]
        # -h[p-k] for right part
        grad[k:] -= h[:n - k]
    else:
        k = -shift
        # +1 for right part
        grad[k:] += 1.0
        # -h[p-k] for right part
        grad[k:] -= h[:n - k]
        # -h[p+k] for left part
        grad[:n - k] -= h[k:]
    return grad


def simulated_annealing(h0, time_limit, rng):
    """Simple SA with exponential temperature schedule."""
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.05
    final_temp = 0.001

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        t_frac = elapsed / time_limit
        temp = init_temp * (final_temp / init_temp) ** t_frac
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        scale = temp / init_temp
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            continue
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target_c5 = 0.3808

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # --- initialization ---
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_balanced_binary(64, rng)
    n_target = 512
    base = upsample(base, n_target)
    base = project_onto_box_sum(base, n_target / 2.0)

    # best solution tracking
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # gradient descent on the worst shift
    curr_h = base.copy()
    last_improve = time.time()
    restart_interval = 30.0  # seconds

    while time_left() > 5.0:
        # evaluate current worst correlation
        corr = np.correlate(curr_h, 1.0 - curr_h, mode='full')
        idx_max = int(np.argmax(corr))
        cur_score = corr[idx_max] * (2.0 / n_target)

        if cur_score < best_score:
            best_score = cur_score
            best_h = curr_h.copy()
            last_improve = time.time()
            if best_score <= target_c5:
                break

        shift = idx_max - (n_target - 1)
        grad = gradient_correlation(curr_h, shift)

        # step size decays from 0.5 to 0.001
        tr = time_left()
        step = 0.001 + 0.5 * (tr / budget_s) ** 0.5

        # gradient descent step and projection
        curr_h = project_onto_box_sum(curr_h - step * grad, n_target / 2.0)

        # occasional random restart if stalled
        if time.time() - last_improve > restart_interval and time_left() > restart_interval:
            curr_h = random_balanced_binary(n_target, rng)
            curr_h = project_onto_box_sum(curr_h, n_target / 2.0)
            last_improve = time.time()

    # final refinement with simulated annealing if time permits
    remaining = time_left()
    if remaining > 0.5:
        final_h, final_score = simulated_annealing(best_h, remaining, rng)
        if final_score < best_score:
            best_h = final_h
            best_score = final_score

    best_h = project_onto_box_sum(best_h, n_target / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_target
