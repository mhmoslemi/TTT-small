def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))

    # base: get initial
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.random.randint(0,2,size=64).astype(float)  # but we should have deterministic but time.

    # maybe start with n = 256
    target_n = 256
    base = upsample(base, target_n)
    base = project_onto_box_sum(base, target_n/2.0)

    h = base.copy()
    best_h = h.copy()
    best_score = compute_c5(h)

    # parameters
    alpha = 0.3  # initial step size
    decay = 0.9995  # per iteration
    min_alpha = 1e-6

    while time_left() > 0.5:
        max_corr, lag = compute_corr_and_maxlag(h)  # corr not multiplied by dx
        # compute gradient
        L = lag - (target_n - 1)    # shift
        # vectorized gradient
        grad = np.zeros_like(h)
        # first term
        idx_i = np.arange(target_n)
        idx_j = idx_i + L
        mask = (idx_j >= 0) & (idx_j < target_n)
        grad[mask] += (2.0/target_n)*(1.0 - h[idx_j[mask]])
        # second term
        idx_j2 = idx_i - L
        mask2 = (idx_j2 >= 0) & (idx_j2 < target_n)
        grad[mask2] -= (2.0/target_n)*(h[idx_j2[mask2]])
        # Do gradient step
        h_new = h - alpha * grad
        # projection
        h_new = project_onto_box_sum(h_new, target_n/2.0)
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        # Accept if improvement, else maybe reject
        if new_score <= compute_c5(h) or rng.random() < np.exp(-(new_score - compute_c5(h)) / max(alpha, 1e-12)):

        # decay

def compute_corr_and_maxlag(h):
    n = h.shape[0]
    m = 2 * n
    # Pad
    ha = np.concatenate([h, np.zeros(n)])
    hb = np.concatenate([1.0 - h, np.zeros(n)])
    # real FFT
    Fa = np.fft.rfft(ha)
    Fb = np.fft.rfft(hb)
    # correlation via conj of Fb
    corr = np.fft.irfft(Fa * np.conj(Fb), n=m)
    # take first 2n-1 values
    corr = corr[:2 * n - 1]
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    return max_val, max_idx

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_corr_and_maxlag(h):
    """Linear (full) correlation of h with 1-h using FFT.
    Returns (max_value, index_of_max) where index corresponds to lag = idx-(n-1)."""
    n = h.shape[0]
    m = 2 * n
    ha = np.concatenate([h, np.zeros(n)])
    hb = np.concatenate([1.0 - h, np.zeros(n)])
    Fa = np.fft.rfft(ha)
    Fb = np.fft.rfft(hb)
    corr = np.fft.irfft(Fa * np.conj(Fb), n=m)
    corr = corr[:2 * n - 1]               # valid lags -(n-1)..+(n-1)
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    max_corr, _ = compute_corr_and_maxlag(h)
    return max_corr * (2.0 / h.shape[0])


def gradient_descent_step(h, alpha):
    """One projected gradient descent step on the active lag."""
    n = h.shape[0]
    _, max_idx = compute_corr_and_maxlag(h)
    L = max_idx - (n - 1)                 # lag value
    grad = np.zeros_like(h)

    idx = np.arange(n)
    idx_plus = idx + L
    mask_plus = (idx_plus >= 0) & (idx_plus < n)
    grad[mask_plus] += (2.0 / n) * (1.0 - h[idx_plus[mask_plus]])

    idx_minus = idx - L
    mask_minus = (idx_minus >= 0) & (idx_minus < n)
    grad[mask_minus] -= (2.0 / n) * h[idx_minus[mask_minus]]

    h_new = h - alpha * grad
    h_new = project_onto_box_sum(h_new, n / 2.0)
    new_score = compute_c5(h_new)
    return h_new, new_score


def simulated_annealing(h0, time_limit, rng):
    """Simple SA that preserves sum and box constraints."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.02

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        temp = init_temp * (1.0 - frac)
        # propose move
        if rng.random() < 0.5:
            # delta mass transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            delta = max_delta * rng.random() * (temp / init_temp)
            if delta <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # swap two entries (preserves sum and bounds)
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]

        new_score = compute_c5(h_new)
        accept = False
        if new_score < cur_score:
            accept = True
        elif temp > 0.0:
            prob = np.exp(-(new_score - cur_score) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()

        # occasional random balanced binary restart
        if rng.random() < 0.001:
            h_rand = np.zeros(n, dtype=float)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_rand[idx] = 1.0
            sc_rand = compute_c5(h_rand)
            if sc_rand < best_score:
                best_score = sc_rand
                best_h = h_rand.copy()
                h = h_rand.copy()
                cur_score = sc_rand

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.zeros(64, dtype=float)
        base[:32] = 1.0

    # Stage 1: gradient descent on medium grid
    n1 = 256
    base = upsample(base, n1)
    base = project_onto_box_sum(base, n1 / 2.0)
    h = base.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    alpha = 0.5
    decay = 0.9995
    min_alpha = 1e-4

    # Run until 30% of the time budget remains
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= budget_s * 0.3 or remaining <= 1.0:
            break

        h_new, new_score = gradient_descent_step(h, alpha)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # small Metropolis acceptance to avoid stagnation
            if alpha > 0.0:
                prob = np.exp(-(new_score - cur_score) / max(alpha, 1e-12))
                if rng.random() < prob:
                    accept = True

        if accept:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()

        # decay step size
        alpha = max(min_alpha, alpha * decay)

        # occasional random jitter
        if rng.random() < 0.005:
            h = project_onto_box_sum(
                h + 0.05 * (rng.random(n1) - 0.5), n1 / 2.0
            )
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()

    # Stage 2: upsample and finalize with simulated annealing
    n2 = 512
    best_h = upsample(best_h, n2)
    best_h = project_onto_box_sum(best_h, n2 / 2.0)

    remaining = max(0.0, budget_s - (time.time() - start_time))
    if remaining > 5.0:
        cand_h, cand_score = simulated_annealing(best_h, remaining, rng)
        if cand_score < best_score:
            best_h = cand_h
            best_score = cand_score

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
