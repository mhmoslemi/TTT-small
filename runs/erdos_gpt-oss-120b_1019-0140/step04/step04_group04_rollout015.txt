def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find all shifts that achieve max (within small tolerance)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        # step schedule: step = step0 / np.sqrt(it)
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random Gaussian noise injection
        if it % 500 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=1000):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        # returns an array of constraints: t - corr >= 0
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    # Optimize t as the objective: minimize t
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    # Project back to feasible sum (some numerical slack)
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

Now upsample function:

def upsample(arr, new_n):
    n = arr.shape[0]
    if n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # find lambda such that sum(clipped(v - lam)) = s
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure the interval brackets the solution
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    n = arr.shape[0]
    if n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def beatty_balanced(n, alpha):
    x = (np.arange(n) * alpha) % 1.0
    idx = np.argsort(x)
    h = np.zeros(n, dtype=float)
    h[idx[:n // 2]] = 1.0
    return h

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
