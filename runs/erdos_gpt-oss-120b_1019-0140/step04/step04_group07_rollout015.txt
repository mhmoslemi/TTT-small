def weighted_gradient(h, weights):
    n = h.size
    grad = np.zeros_like(h)
    # weights length 2*n-1, idx shift = idx - (n-1)
    # Will iterate idx from 0..2n-2
    w = weights
    # Pre compute 1 - h and h for speed
    one_minus_h = 1.0 - h
    for idx in range(2*n-1):
        w_i = w[idx]
        if w_i == 0:
            continue
        shift = idx - (n-1)
        if shift >= 0:
            s = shift
            # vector1: i in [0, n-s)
            grad[:n-s] += w_i * one_minus_h[s:]
            # vector2: i in [s, n)
            grad[s:] += w_i * (-h[:n-s])
        else:
            t = -shift
            # vector1: i in [t, n)
            grad[t:] += w_i * one_minus_h[:n-t]
            # vector2: i in [0, n-t)
            grad[:n-t] += w_i * (-h[t:])
    return grad
```

def gradient_descent_weighted(h0, time_left, max_iters=None):
    h = h0.copy()
    n = h.size
    s_target = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = 0.8
    beta = 0.9
    it = 0
    while time_left() > 0:
        if max_iters is not None and it >= max_iters:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        min_corr = corr.min()
        score = max_corr * (2.0 / n)
        # compute weights
        temp = max(1e-8, (max_corr - min_corr) * 0.2)  # adjust factor
        w = np.exp((corr - max_corr) / temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum
        grad = weighted_gradient(h, w)
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s_target)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(1.0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = 0.8 * 0.1
        # occasional jitter
        if it % 5000 == 0 and time_left() > 0:
            jitter = np.random.standard_normal(n) * 0.01
            h = project_onto_box_sum(h + jitter, s_target)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # generate Thue-Morse binary of size 64
        # Use function thue_morse
        def thue_morse(n):
            return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
        base = thue_morse(64)
    # ensure sum constraint
    n0 = base.shape[0]
    base = project_onto_box_sum(base, n0/2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if best_h.size != n:
            # upsample to target n
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n/2.0)
        # gradient descent with weighted gradient
        if time_left() > 10:
            # allocate max_iters proportionally to time left and size
            max_iters = int(min(40000, max(5000, time_left() * (8000.0 / n))))
            gd_h, gd_score = gradient_descent_weighted(base, time_left, max_iters=max_iters)
            if gd_score < best_score:
                best_h, best_score = gd_h.copy(), gd_score
            base = gd_h
        # annealing moves
        if time_left() > 5:
            sa_h, sa_score = simulated_annealing_moves(base, time_left, rng)
            if sa_score < best_score:
                best_h, best_score = sa_h.copy(), sa_score
            base = sa_h

        # random binary restarts maybe for fine scales:
        if time_left() > 10 and n <= 512:
            for _ in range(2):
                if time_left() < 5:
                    break
                h_rand = random_balanced_binary(n, rng)
                # refine with gradient descent
                gd_h2, gd_score2 = gradient_descent_weighted(h_rand, time_left, max_iters=int(min(8000, max(2000, time_left()* (3000.0/n)))))
                if gd_score2 < best_score:
                    best_h, best_score = gd_h2.copy(), gd_score2
                # local annealing
                sa_h2, sa_score2 = simulated_annealing_moves(gd_h2, time_left, rng)
                if sa_score2 < best_score:

        best_h, best_score = final_h, final_score

    # ensure projection and final score
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

def weighted_gradient(h, weights):
    n = h.size
    grad = np.zeros_like(h)
    one_minus_h = 1.0 - h
    w = weights

        final_h, final_score = gradient_descent_weighted(best_h, time_left, rng,
                                                        max_iters=int(min(80000, max(20000, time_left() * 30))))
        best_h, best_score = final_h, final_score

    # Try binary rounding
    bin_h = make_balanced_binary(best_h)
    bin_score = compute_c5(bin_h)
    if bin_score < best_score:
        best_h, best_score = bin_h, bin_score
        if time_left() > 5:
            best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
