import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    # Euclidean projection onto box and sum constraint
    n=v.size
    if s <= lo*n:
        return np.full_like(v, lo)
    if s >= hi*n:
        return np.full_like(v, hi)
    # find lambda by bisection
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds
    if f(lam_low) < s:
        lam_low = np.min(v) - 10*hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10*lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high)*0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n=h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)*(2.0/n)

def upsample(arr, new_n):
    m=arr.shape[0]
    if m==new_n:
        return arr.copy()
    x_old = np.linspace(0.0,1.0,m,endpoint=False)
    x_new = np.linspace(0.0,1.0,new_n,endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n=h.size
    grad=np.zeros_like(h)
    for shift in shifts:
        if shift>=0:
            # term1 = h shifted left, term2 = h shifted right
            term1 = np.empty(n, dtype=h.dtype)
            term1[:n-shift]=h[shift:]
            term1[n-shift:]=0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:]=h[:n-shift]
            term2[:shift]=0.0
        else:
            s=-shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:]=h[:n-s]
            term1[:s]=0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:n-s]=h[s:]
            term2[n-s:]=0.0
        grad += (1.0 - term1 - term2)
    if shifts.size>0:
        grad/=len(shifts)
    return grad

    n=h.size

                i = rng.integers(n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto the box [lo,hi]^n with sum=v."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral C₅ for discrete h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from size m to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue‑Morse sequence (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts.size:
        grad /= len(shifts)
    return grad
