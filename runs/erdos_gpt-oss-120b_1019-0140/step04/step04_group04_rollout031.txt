def simulated_annealing(h_start, best_score, max_steps, T0, rng, time_left):
    n = h_start.size
    best_h = h_start.copy()
    best_sc = best_score
    cur_h = h_start.copy()
    cur_sc = best_sc
    # bias the temperature: linearly decreasing or exponential
    step = 0
    while step < max_steps and time_left() > 0:
        # compute temperature
        # Use linear cooling with steps vs max_steps
        T = T0 * (1 - step / max_steps)
        # Random pair
        i, j = rng.choice(n, size=2, replace=False)
        # possible delta
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step += 1
            continue
        delta = rng.random() * max_delta
        # propose new
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # no need to project; sum preserved and bounds satisfied
        # compute score
        new_sc = compute_c5(new_h)
        # accept
        if new_sc < cur_sc or rng.random() < np.exp(-(new_sc - cur_sc) / max(T, 1e-12)):
            cur_h = new_h
            cur_sc = new_sc
        # update best if better
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_h = cur_h.copy()
            # early break if target reached
            if best_sc <= 0.3808:
                break
        step += 1
    return best_h, best_sc

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lo_lam = np.min(v) - hi
    hi_lam = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lo_lam) < s:
        lo_lam = np.min(v) - 10 * hi
    if f(hi_lam) > s:
        hi_lam = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam = (lo_lam + hi_lam) * 0.5
        fm = f(lam)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lo_lam = lam
        else:
            hi_lam = lam
    return np.clip(v - lam, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_adam(h0, max_iters, lr0, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1_pow = beta1
    beta2_pow = beta2
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        tol = max_corr * 1e-12 + 1e-12
        sh = (np.where(np.abs(corr - max_corr) <= tol)[0] - (n - 1)).tolist()
        grad = compute_gradients_for_shifts(h, sh)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        lr = lr0 / np.sqrt(it)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        beta1_pow *= beta1
        beta2_pow *= beta2
    return best_h, best_score

def simulated_annealing(h_start, cur_score, max_steps, T0, rng, time_left):
    n = h_start.size
    best_h = h_start.copy()
    best_sc = cur_score
    cur_h = h_start.copy()
    cur_sc = cur_score
    step = 0
    while step < max_steps and time_left() > 0:
        T = T0 * (1 - step / max_steps)
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step += 1
            continue
        delta = rng.random() * max_delta
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_sc = compute_c5(new_h)
        if new_sc < cur_sc or rng.random() < np.exp(-(new_sc - cur_sc) / max(T, 1e-12)):
            cur_h = new_h
            cur_sc = new_sc
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_h = cur_h.copy()
            if best_sc <= 0.3808:
                break
        step += 1
    return best_h, best_sc
