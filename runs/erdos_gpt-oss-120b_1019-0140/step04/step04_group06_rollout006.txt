def grad_descent(h0, n_iter, step0, beta, rng, time_left):
   h = h0.copy()
   n = h.size
   target_sum = n / 2.0
   best_h = h.copy()
   best_score = compute_c5_fft(h)
   momentum = np.zeros_like(h)
   for it in range(1, n_iter+1):
       if time_left() <= 0: break
       shifts = max_shifts(h)
       grad = compute_gradients_for_shifts(h, shifts)
       momentum = beta * momentum + (1-beta) * grad
       eta = step0 / math.sqrt(it)
       h_new = h - eta * momentum
       # add small random noise occasionally
       if it % 2000 == 0:
           h_new += rng.normal(scale=0.01, size=n)
       h_new = project_onto_box_sum(h_new, target_sum)
       sc = compute_c5_fft(h_new)
       if sc < best_score:
           best_score, best_h = sc, h_new.copy()
       h = h_new
   return best_h, best_score
```

def simulated_anneal(h0, score0, rng, time_left):
    h = h0.copy()
    best_h, best_score = h.copy(), score0
    cur_score = score0
    start = time.time()
    T0, T1 = 1e-2, 1e-6
    while time_left() > 0:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed + 1e-12)
        T = T0 * (T1 / T0) ** frac
        i, j = rng.choice(h.size, size=2, replace=False)
        # propose a small transfer
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0:
            continue
        delta = delta_max * rng.uniform(0.1, 0.5)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5_fft(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            h, cur_score = h_new, sc_new
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
        # occasional gradient nudge
        if rng.random() < 0.05:
            shifts = max_shifts(h)
            grad = compute_gradients_for_shifts(h, shifts)
            step = 0.02
            h_tmp = h - step * grad
            h_tmp = project_onto_box_sum(h_tmp, h.size / 2.0)
            sc_tmp = compute_c5_fft(h_tmp)
            if sc_tmp < cur_score:
                h, cur_score = h_tmp, sc_tmp
                if cur_score < best_score:
                    best_score, best_h = cur_score, h.copy()
    return best_h, best_score
```

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_corr_fft(h):
    n = h.shape[0]
    m = 2 * n
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    conv = np.fft.irfft(H * np.conj(G), m)
    return conv[:2 * n - 1]

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return corr.max() * (2.0 / n)

def max_shifts(h):
    corr = compute_corr_fft(h)
    maxc = corr.max()
    idx = np.where(np.isclose(corr, maxc, atol=1e-12))[0]
    n = h.size
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_anneal(h0, score0, rng, time_left):
    h = h0.copy()
    best_h, best_score = h.copy(), score0
    cur_score = score0
    start = time.time()
    T0, T1 = 1e-2, 1e-6
    n = h.size
    while time_left() > 0:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed + 1e-12)
        T = T0 * (T1 / T0) ** frac
        i, j = rng.choice(n, size=2, replace=False)
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0:
            continue
        delta = delta_max * rng.uniform(0.1, 0.5)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5_fft(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            h, cur_score = h_new, sc_new
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
        if rng.random() < 0.05:
            shifts = max_shifts(h)
            grad = compute_gradients_for_shifts(h, shifts)
            step = 0.02
            h_tmp = h - step * grad
            h_tmp = project_onto_box_sum(h_tmp, n / 2.0)
            sc_tmp = compute_c5_fft(h_tmp)
            if sc_tmp < cur_score:
                h, cur_score = h_tmp, sc_tmp
                if cur_score < best_score:
                    best_score, best_h = cur_score, h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    scales = [64, 128, 256, 512, 1024]
    best_h = None
    best_score = float('inf')

    for n in scales:
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Gradient descent phase
        if time_left() > 5.0:
            max_iters = int(200000 / (n / 64))
            iter_budget = int(min(max_iters, max(20000, time_left() * 0.5)))
            h_gd, sc_gd = grad_descent(base, iter_budget, step0=0.5,
                                       beta=0.9, rng=rng, time_left=time_left)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()
            base = h_gd

        # Simulated annealing phase
        if time_left() > 5.0:
            h_ann, sc_ann = simulated_anneal(base, compute_c5_fft(base), rng, time_left)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()
            base = h_ann

    # Final extensive search at the finest scale
    n_final = scales[-1]
    while time_left() > 10.0:
        # Random balanced start
        h_rand = rng.random(n_final)
        h_rand = project_onto_box_sum(h_rand, n_final / 2.0)
        # Short gradient descent
        iters = int(min(50000, max(5000, time_left() * 3)))
        h_opt, sc_opt = grad_descent(h_rand, iters, step0=0.4, beta=0.9,
                                     rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        # Random pair tweak
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.1, 0.3)

            h_new[j] += delta
            sc_new = compute_c5_fft(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
