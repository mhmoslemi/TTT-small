def optimize_at_scale(h0, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    momentum = np.zeros_like(h)
    beta = 0.9
    step = 0.8
    alpha = 4.0
    iter = 0
    max_iters = int(min(50000, max(5000, time_left() * (15000.0 / n))))
    while iter < max_iters and time_left() > 0:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        w = np.exp(alpha * (corr - max_corr))
        w_rev = w[::-1]
        conv = np.convolve(w_rev, h, mode='full')
        grad = -2.0 * conv[n-1:n-1+n]
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        # acceptance
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            # maybe increase step
            step = min(step * 1.02, 1.5)
        else:
            # reject: keep old h, reduce step
            step *= 0.5
            if step < 1e-4:
                step = 1e-4

            best_h, best_score = h_new.copy(), new_score
        # adapt alpha gradually
        if iter % 4000 == 0 and iter > 0:
            alpha = min(alpha * 1.1, 30.0)
        iter += 1
    return best_h, best_score

def simulated_annealing_refine(h0, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T0 = 0.01 * cur_score  # initial temperature scaling
    # schedule: T = T0 * (time_left/total_time)
    total = time_left()
    start_t = time.time()
    while time_left() > 0:
        # propose random move
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        # if max_delta small, skip
        if max_delta <= 0:
            continue
        # delta get from uniform scaled by temperature factor?
        # Use delta proportion of max_delta, e.g., max_delta * rng.random() * 0.5
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        # acceptance probability
        # compute remaining fraction of total time
        elapsed = time.time() - start_t
        frac = elapsed / total if total > 0 else 1.0
        T = T0 * max(0.0, 1.0 - frac)  # linear cooling
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
        # occasionally random perturbation
        if rng.random() < 0.001:
            # jitter all
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from current size to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def optimize_at_scale(h0, rng, time_left):
    """Momentum gradient descent using soft‑max weighted gradient."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    momentum = np.zeros_like(h)
    beta = 0.9
    step = 0.8
    alpha = 4.0  # soft‑max sharpness

    # iteration budget guided by remaining time
    max_iters = int(min(30000, max(2000, time_left() * 30)))
    it = 0
    while it < max_iters and time_left() > 0:
        # correlation and soft‑max weights
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        w = np.exp(alpha * (corr - max_corr))          # weights ≤1
        w_rev = w[::-1]

        # weighted gradient via convolution
        conv = np.convolve(w_rev, h, mode='full')
        grad = -2.0 * conv[n - 1:n - 1 + n]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # trial step
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        new_score = compute_c5(h_new)

        # acceptance
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step * 1.02, 1.5)          # modest step increase
        else:
            step *= 0.5
            if step < 1e-4:
                step = 1e-4

        # keep global best within this scale
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score

        # adapt parameters occasionally
        if it % 4000 == 0 and it > 0:
            alpha = min(alpha * 1.1, 30.0)

        it += 1
    return best_h, best_score


def simulated_annealing_refine(h0, rng, time_left):
    """Random mass transfers with Metropolis acceptance."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    total = max(time_left(), 1e-6)
    start = time.time()
    T0 = 0.01 * cur_score + 1e-6  # initial temperature

    while time_left() > 0:
        # propose transfer i -> j
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)

        elapsed = time.time() - start
        frac = elapsed / total
        T = T0 * max(0.0, 1.0 - frac)   # linear cooling

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score

        # occasional jitter to escape plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, target_sum)
            cur_score = compute_c5(h)
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]

    for n in scales:
        if time_left() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
            base = project_onto_box_sum(base, n / 2.0)

        # gradient descent on current resolution
        base, sc = optimize_at_scale(base, rng, time_left)
        if sc < best_score:
            best_h, best_score = base.copy(), sc

    # final annealing refinement on the best resolution
    if time_left() > 0:
        best_h, best_score = simulated_annealing_refine(best_h, rng, time_left)

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
