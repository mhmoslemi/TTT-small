def binary_simulated_annealing(h0, time_limit, rng):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # sample initial delta for T0
    deltas = []
    for _ in range(20):
        # pick random 1 index
        idx1 = rng.choice(np.where(h == 1)[0])
        idx0 = rng.choice(np.where(h == 0)[0])
        h[idx1], h[idx0] = h[idx0], h[idx1]
        new_score = compute_c5(h)
        deltas.append(abs(new_score - cur_score))
        # revert
        h[idx1], h[idx0] = h[idx0], h[idx1]
    if deltas:
        T0 = np.mean(deltas) * 5.0  # factor
    else:
        T0 = cur_score * 0.1
    cool_rate = 0.99995
    iter = 0
    while True:
        if time.time() - start >= time_limit:
            break
        idx1 = rng.choice(np.where(h == 1)[0])
        idx0 = rng.choice(np.where(h == 0)[0])
        # swap
        h[idx1], h[idx0] = h[idx0], h[idx1]
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            T = T0 * (cool_rate ** iter)
            if T > 0:
                accept_prob = np.exp(-(new_score - cur_score) / T)
                if rng.random() < accept_prob:
                    cur_score = new_score
                else:
                    # revert swap
                    h[idx1], h[idx0] = h[idx0], h[idx1]
            else:
                # revert
                h[idx1], h[idx0] = h[idx0], h[idx1]
        iter += 1
        # occasional reheating: if iter % 500000 == 0: maybe reset T0?
        if iter % 500000 == 0:
            T0 = T0 * 0.5  # decrease?
    return best_h, best_score

def binary_simulated_annealing(h0, time_limit, rng):
    import time, numpy as np
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    # initial temperature guess
    # sample a few random swaps to estimate typical delta
    deltas = []
    for _ in range(20):
        # random swap
        i = rng.integers(n)
        while h[i] != 1:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 0:
            j = rng.integers(n)
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        deltas.append(abs(new_score - cur_score))
        # revert
        h[i], h[j] = h[j], h[i]
    if deltas:
        T = max(1e-6, np.mean(deltas) * 5.0)
    else:
        T = max(1e-6, cur_score * 0.1)
    cool_rate = 0.99997
    it = 0
    target = 0.3808
    while True:
        if time.time() - start >= time_limit:
            break
        # pick random i with 1
        i = rng.integers(n)
        while h[i] != 1:
            i = rng.integers(n)
        # pick random j with 0
        j = rng.integers(n)
        while h[j] != 0:
            j = rng.integers(n)
        # swap
        h[i], h[j] = h[j], h[i]

        # cool
        T *= cool_rate
        it += 1
        # optional restart if T too low (e.g., T < 1e-8)
        # but we can just continue

    return best_h, best_score

import time
import numpy as np

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def binary_simulated_annealing(h0, time_limit, rng):
    """SA on binary vectors with exactly half ones."""
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    # estimate a temperature scale
    deltas = []
    for _ in range(20):
        # pick a random 1 and a random 0
        i = rng.integers(n)
        while h[i] != 1:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 0:
            j = rng.integers(n)
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        deltas.append(abs(new_score - cur_score))
        h[i], h[j] = h[j], h[i]  # revert
    T = max(1e-8, (np.mean(deltas) if deltas else cur_score) * 5.0)
    cool_rate = 0.99997
    target = 0.3808

    while True:
        if time.time() - start >= time_limit:
            break
        # pick a random 1 and a random 0
        i = rng.integers(n)
        while h[i] != 1:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 0:
            j = rng.integers(n)
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        T *= cool_rate
        if T < 1e-12:
            T = 1e-12
    return best_h, best_score
