def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):

    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search on lambda such that sum(clamp(v - lam, lo, hi)) = s
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure interval brackets the root
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_balanced_real(n, rng):
    h = rng.random(n)
    return project_onto_box_sum(h, n / 2.0)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    noise_period = 400
    noise_scale = 0.008
    patience = max(2000, n_iter // 10)
    last_improve = 0
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % noise_period == 0:
            h_new += rng.normal(scale=noise_scale, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
            last_improve = it
        if it - last_improve > patience:
            break
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # hierarchical scaling
    scales = [128, 256, 512, 1024]
    for n in scales:
        if time_left() < 5:
            break
        base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        sc = compute_c5(base)
        if sc < best_score:
            best_h, best_score = base.copy(), sc

        # allocate iterations proportional to remaining time
        max_iters = int(min(20000, time_left() * 10))
        if max_iters > 0:
            step0 = 0.5 if n <= 512 else 0.4
            h_opt, sc_opt = gradient_descent_momentum_custom(
                base, n_iter=max_iters, step0=step0, beta=0.95,
                rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
            base = h_opt

        # occasional SLSQP refinement at moderate sizes
        if n <= 512 and time_left() > 2:
            h_slsqp, sc_slsqp = slsqp_optimize_candidate(base, max_iter=400)
            if sc_slsqp < best_score:
                best_h, best_score = h_slsqp.copy(), sc_slsqp
            base = h_slsqp

    # final refinement at best resolution
    n_final = best_h.size
    if time_left() > 5:
        max_iters = int(min(80000, time_left() * 30))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=max_iters, step0=0.2, beta=0.95,
            rng=rng, time_left=time_left)

    if time_left() > 2:
        h_slsqp, sc_slsqp = slsqp_optimize_candidate(best_h, max_iter=800)
        if sc_slsqp < best_score:
            best_h, best_score = h_slsqp.copy(), sc_slsqp

    # random restarts and perturbations
    target = 0.3808
    while time_left() > 2:
        h_rand = random_binary_balanced(n_final, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_h, best_score = h_rand.copy(), sc_rand
            # brief polish
            max_iters = int(min(15000, time_left() * 8))
            if max_iters > 0:
                h_opt, sc_opt = gradient_descent_momentum_custom(
                    best_h, n_iter=max_iters, step0=0.3, beta=0.95,
                    rng=rng, time_left=time_left)
                if sc_opt < best_score:
                    best_h, best_score = h_opt.copy(), sc_opt

        if rng.random() < 0.5 and time_left() > 4:
            pert = rng.normal(scale=0.02, size=n_final)
            h_pert = best_h + pert
            h_pert = project_onto_box_sum(h_pert, n_final / 2.0)
            sc_pert = compute_c5(h_pert)
            if sc_pert < best_score:
                best_h, best_score = h_pert.copy(), sc_pert
                max_iters = int(min(15000, time_left() * 8))
                if max_iters > 0:
                    h_opt, sc_opt = gradient_descent_momentum_custom(
                        best_h, n_iter=max_iters, step0=0.3, beta=0.95,
                        rng=rng, time_left=time_left)
                    if sc_opt < best_score:
                        best_h, best_score = h_opt.copy(), sc_opt

        if best_score <= target:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
