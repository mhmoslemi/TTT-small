def smooth_gradient_descent(h0, n_iter, step0, beta, rng, max_time, tau0=0.2, tau_final=1e-4, top_k=200):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    tau = tau0
    for it in range(1, n_iter+1):
        if time.time() - start >= max_time:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # compute weights
        # to avoid overflow, subtract max_corr
        w = np.exp((corr - max_corr) / tau)
        w_sum = w.sum()
        w /= w_sum
        # select top_k indices
        if top_k < len(w):
            idxs = np.argpartition(-w, top_k-1)[:top_k]
        else:
            idxs = np.arange(len(w))
        shifts = idxs - (n - 1)
        weights = w[idxs]
        # compute weighted gradient
        grad = np.zeros_like(h)
        # loops
        for shift, weight in zip(shifts, weights):
            if shift >= 0:
                sz = n - shift
                term1 = np.empty(n, dtype=h.dtype)
                term1[:sz] = h[shift:]
                term1[sz:] = 0.0
                term2 = np.empty(n, dtype=h.dtype)
                term2[shift:] = h[:sz]
                term2[:shift] = 0.0
            else:
                s = -shift
                sz = n - s
                term1 = np.empty(n, dtype=h.dtype)
                term1[s:] = h[:sz]
                term1[:s] = 0.0
                term2 = np.empty(n, dtype=h.dtype)
                term2[:sz] = h[s:]
                term2[sz:] = 0.0
            grad += weight * (1.0 - term1 - term2)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # update tau gradually
        # exponential decay
        tau = max(tau * 0.99, tau_final)
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def smooth_gradient_descent(h0, n_iter, step0, beta, rng, max_time,
                           tau0=0.2, tau_final=1e-4, top_k_factor=0.08):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    tau = tau0
    top_k = max(20, int(min(200, top_k_factor * n)))
    term1 = np.empty(n, dtype=h.dtype)
    term2 = np.empty(n, dtype=h.dtype)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        w = np.exp((corr - max_corr) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        if top_k < w.size:
            idxs = np.argpartition(-w, top_k - 1)[:top_k]
        else:
            idxs = np.arange(w.size)
        shifts = idxs - (n - 1)
        weights = w[idxs]
        grad = np.zeros_like(h)
        for shift, weight in zip(shifts, weights):
            term1.fill(0.0)
            term2.fill(0.0)
            if shift >= 0:
                sz = n - shift
                term1[:sz] = h[shift:]
                term2[shift:] = h[:sz]
            else:
                s = -shift
                sz = n - s
                term1[s:] = h[:sz]
                term2[:sz] = h[s:]
            grad += weight * (1.0 - term1 - term2)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        tau = max(tau * 0.99, tau_final)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_final = 1024
    scales = [64, 128, 256, 512, n_final]
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    best_h = None
    best_score = np.inf

    for idx, n in enumerate(scales):
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # random binary sampling
        sample_limit = int(min(500, max(5, 500 * 64 / n)))
        for _ in range(sample_limit):
            if time.time() - start > budget_s - 1.0:
                break
            h_rand = random_binary_balanced(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score, best_h = sc, h_rand.copy()

        # smooth gradient descent
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            max_iters = int(min(5000, max(500, remaining * (2000.0 / max(64, n)))))
            h_opt, sc_opt = smooth_gradient_descent(base, n_iter=max_iters,
                                                    step0=1.0, beta=0.9,
                                                    rng=rng,
                                                    max_time=remaining)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt

        # hard gradient descent (full shifts)
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            max_iters = int(min(20000, max(5000, remaining * (5000.0 / max(64, n)))))
            h_opt, sc_opt = gradient_descent_momentum_custom(base,
                                                             n_iter=max_iters,
                                                             step0=0.4,
                                                             beta=0.9,
                                                             rng=rng,
                                                             max_time=remaining)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt

        # simulated annealing refinement
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0 and best_h is not None and best_h.shape[0] == n:
            h_cur = best_h.copy()
            cur_score = best_score
            total_steps = int(min(20000, max(5000, remaining * 500.0)))
            T0 = 0.01
            T_final = 1e-6
            for step in range(total_steps):
                if time.time() - start > budget_s:
                    break
                T = T0 * ((total_steps - step) / total_steps) + T_final
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(h_cur[i], 1.0 - h_cur[j])
                if max_move <= 0.0:
                    continue
                delta = max_move * rng.random()
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_score = compute_c5(h_new)
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                    h_cur = h_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score, best_h = cur_score, h_cur.copy()
            base = best_h

        # prepare for next scale
        if idx < len(scales) - 1:
            base = best_h.copy() if best_h is not None else base
        if time.time() - start > budget_s - 2.0:
            break

    # final refinement on the best found
    if best_h is not None and budget_s - (time.time() - start) > 5.0:
        n = best_h.size
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(200000, max(50000, remaining * (50000.0 / max(64, n)))))
        best_h, best_score = gradient_descent_momentum_custom(best_h,
                                                             n_iter=max_iters,
                                                             step0=0.2,
                                                             beta=0.9,
                                                             rng=rng,
                                                             max_time=remaining)
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    if best_h is None:
        n = n_final
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
