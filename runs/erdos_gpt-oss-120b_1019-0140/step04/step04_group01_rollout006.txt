Pseudo:

def refine_solution(h, time_left):
    # gradient descent
    h1, sc1 = gradient_descent_momentum_with_time(h, time_left, step0=0.8, beta=0.9, max_iter=200000)
    # binary threshold
    n = h1.size
    k = n // 2
    idx = np.argpartition(-h1, k)[:k]
    h_bin = np.zeros_like(h1)
    h_bin[idx] = 1.0
    sc_bin = compute_c5_fft(h_bin)
    # if binary better, use binary else keep h1
    best_h = h_bin if sc_bin < sc1 else h1
    best_score = min(sc_bin, sc1)
    # binary annealing
    if time_left() > 0.5:
        h_bin_ref, sc_bin_ref = binary_swap_anneal(best_h, rng, time_left, max_iters=200000, T0=0.2, decay=0.99995, Tmin=1e-8)
        if sc_bin_ref < best_score:
            best_h, best_score = h_bin_ref, sc_bin_ref
    # final gradient descent
    if time_left() > 0.5:
        h_final, sc_final = gradient_descent_momentum_with_time(best_h, time_left, step0=0.8, beta=0.9, max_iter=200000)
        if sc_final < best_score:
            best_h, best_score = h_final, sc_final
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if lam_high - lam_low < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    return np.max(compute_corr_scaled_fft(h))

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)

def gradient_descent_momentum_with_time(h0, time_left_func,
                                        step0=0.8, beta=0.9,
                                        max_iter=200000):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        grad = compute_gradients_for_shifts(h, corr, epsilon=1e-6)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func,
                      max_iters=400000, T0=0.2,
                      decay=0.99995, Tmin=1e-8):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_fft(h)
    best_h, best_score = h.copy(), cur_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = delta < 0 or (T > 0.0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

    h_c, sc_c = gradient_descent_momentum_with_time(
        h, time_left, step0=0.8, beta=0.9, max_iter=200000)
    # binary conversion

    k = n // 2

        if sc_fin < best_sc:
            best_h, best_sc = h_fin, sc_fin
    return best_h, best_sc
