def try_reduce_worst_shift(h, rng):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    if shift >= 0:
        region = np.arange(0, n - shift)
    else:
        region = np.arange(-shift, n)
    # compute candidate ones and zeros
    i_candidates = region[(h[region] == 1) & (h[region + shift] == 0)]
    j_candidates = region[(h[region] == 0) & (h[region + shift] == 1)]
    if i_candidates.size > 0 and j_candidates.size > 0:
        i = rng.choice(i_candidates)
        j = rng.choice(j_candidates)
        # swap
        h[i], h[j] = h[j], h[i]  # actually both booleans but values 0/1
        return True, shift, corr[idx]
    return False, shift, corr[idx]
```

import time
import numpy as np

def compute_c5(h):
    """Return C5 for a given array h (values in [0,1])."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    seq = np.zeros(n, dtype=int)
    for i in range(n):
        seq[i] = bin(i).count('1') & 1
    return seq

def greedy_improve(h, rng, start_time, budget_s):
    """Iteratively reduce the worst shift by swapping a bad 1 with a good 0."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    # stop when very little time remains
    while budget_s - (time.time() - start_time) > 0.15:
        corr = np.correlate(h, 1 - h, mode='full')
        # indices sorted by decreasing overlap
        idxs = np.argsort(-corr)
        improved = False
        for idx in idxs:
            shift = idx - (n - 1)
            if shift >= 0:
                region = np.arange(0, n - shift)
            else:
                region = np.arange(-shift, n)
            # candidates for a beneficial swap
            region_h = h[region]
            region_h_shift = h[region + shift]
            ones_bad = region[(region_h == 1) & (region_h_shift == 0)]
            zeros_good = region[(region_h == 0) & (region_h_shift == 1)]
            if ones_bad.size and zeros_good.size:
                i = rng.choice(ones_bad)
                j = rng.choice(zeros_good)
                h[i], h[j] = 0, 1
                improved = True
                break
        if not improved:
            break  # no improving swap for any shift
        cur_score = compute_c5(h)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
            if best_score <= 0.3808:
                break
    return best_h, best_score

def simulated_annealing(h, rng, start_time, budget_s, target_score=0.3808):
    """Simple SA on binary vectors, keeping sum constraint."""
    n = h.shape[0]
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    # initial temperature scaled to current excess over target
    T = max(0.01, cur_score - target_score)
    while budget_s - (time.time() - start_time) > 0.15:
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0, 1
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = np.exp(-delta / T) if T > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:

            if new_score < best_score:

                if best_score <= target_score:
                    break
        else:
            cur_h[i], cur_h[j] = 1, 0  # revert

    return best_h, best_score
