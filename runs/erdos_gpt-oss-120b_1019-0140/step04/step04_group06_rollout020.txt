def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def anneal_binary(h0, rng, time_left_func, start_time, budget_s):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    # Temperature schedule: T0, T1
    T0 = 0.1  # maybe
    T1 = 1e-6
    while time_left_func() > 0:
        # compute fraction based on elapsed
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        T = T0 * (T1 / T0) ** frac
        # propose swap
        # select i where h[i]==1 (ones)
        ones = np.where(h > 0.5)[0]  # binary
        zeros = np.where(h < 0.5)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        # Occasionally do random restart or small perturb
        if rng.random() < 0.01:
            # random perturb: randomly flip a few bits
            k = max(1, n // 100)  # 1% of bits
            # select k ones to turn zero and k zeros to turn one
            if len(ones) >= k and len(zeros) >= k:
                i_sel = rng.choice(ones, size=k, replace=False)
                j_sel = rng.choice(zeros, size=k, replace=False)
                h[i_sel] = 0.0
                h[j_sel] = 1.0
                cur_score = compute_c5(h)
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
        # Break if best_score <= target
        if best_score <= 0.3808:
            break
    return best_h, best_score

def anneal_binary(h0, rng, start_time, budget_s):
    import time, math
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    T0 = 0.2  # initial temperature
    T1 = 1e-6

    # While time left
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left <= 0:
            break
        # temperature schedule
        frac = elapsed / budget_s
        T = T0 * (T1 / T0) ** frac

        # obtain indices of ones and zeros
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)

        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                if best_score <= 0.3808:
                    break

        # occasional large perturbation (e.g., 0.5% chance)
        if rng.random() < 0.005:
            # flip k random bits from 1 to 0 and same number of 0 to 1

                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()

import time, math, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_projection(arr):
    n = arr.shape[0]
    target = n // 2
    # get indices of top target values
    idx = np.argpartition(arr, -target)[-target:]
    out = np.zeros_like(arr, dtype=float)
    out[idx] = 1.0
    return out

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def anneal_binary(h0, rng, start_time, budget_s, max_time, target_score=0.3808):
    # Simulated annealing on binary vector with exact half ones.
    h = h0.copy()
    n = h.shape[0]
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T0 = 0.2
    T1 = 1e-6
    while True:
        now = time.time()
        if now - start_time >= max_time:
            break
        if now >= start_time + budget_s:
            break
        if best_score <= target_score:
            break
        # temperature schedule based on overall progress
        frac = (now - start_time) / budget_s
        T = T0 * (T1 / T0) ** frac
        # indices of ones and zeros
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        # occasional shake
        if rng.random() < 0.005:
            k = max(1, n // 200)  # about 0.5% of entries
            if ones.size >= k and zeros.size >= k:
                i_sel = rng.choice(ones, size=k, replace=False)
                j_sel = rng.choice(zeros, size=k, replace=False)
                h[i_sel] = 0.0
                h[j_sel] = 1.0
                cur_score = compute_c5(h)
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
    return best_h, best_score

                break

            if ones.size >= k and zeros.size >= k:
                i_sel = rng.choice(ones, size=k, replace=False)
                j_sel = rng.choice(zeros, size=k, replace=False)
                h[i_sel] = 0.0
                h[j_sel] = 1.0
                cur_score = compute_c5(h)
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    n_seq = [256, 512, 1024]
    best_h = None
    best_score = float('inf')
    best_n = None
    for n in n_seq:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 10:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = binary_projection(base)
        max_time = max(5.0, remaining * 0.25)
        h_opt, score_opt = anneal_binary(base, rng, start, budget_s, max_time)
        if score_opt < best_score:
            best_score, best_h, best_n = score_opt, h_opt, n
        base = h_opt
    if best_h is None:
        best_h = binary_projection(base)
        best_score = compute_c5(best_h)
        best_n = best_h.size
    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 5:
            break
        h_rand = random_binary_balanced(best_n, rng)
        max_time = max(5.0, remaining * 0.5)
        h_opt, score_opt = anneal_binary(h_rand, rng, start, budget_s, max_time)
        if score_opt < best_score:
            best_score, best_h = score_opt, h_opt
            if best_score <= 0.3808:
                break
    best_h = binary_projection(best_h)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
