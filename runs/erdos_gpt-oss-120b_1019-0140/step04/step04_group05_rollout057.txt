    n = h.shape[0]
    total = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    # schedule of temperature linear in iter.
    for it in range(max_iters):
        if time.time() - start >= time_limit:
            break

def gradient_softmax(h, T):
    import numpy as np
    n = h.shape[0]
    # compute f_k = correlation of h with 1 - h * dx scaling omitted (since scaling factor doesn't affect gradient)
    # We'll compute unscaled correlation because scaling irrelevant for gradient direction.
    f = np.correlate(h, 1.0 - h, mode='full')
    # softmax with temperature T
    # If T too small can cause overflow; shift by max
    mx = np.max(f)
    exp = np.exp((f - mx) / T)
    w = exp / np.sum(exp)
    grad = np.zeros_like(h)
    # loop over shifts
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            # forward part
            grad[:n - k] -= wk * h[k:]
            # backward part
            grad[k:] -= wk * h[:n - k]
        else:
            kk = -k
            grad[kk:] -= wk * h[:n - kk]
            grad[:n - kk] -= wk * h[kk:]
    return grad

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def gradient_softmax(h, T):
    """Gradient of softmax-approximated max over f_k = ⟨h,1‑h⟩ shift."""
    n = h.shape[0]
    f = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(f)
    exp = np.exp((f - mx) / T)
    w = exp / np.sum(exp)

    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        k = idx - offset
        if wk == 0.0:
            continue
        if k >= 0:
            # forward contribution
            grad[:n - k] -= wk * h[k:]
            # backward contribution
            grad[k:] -= wk * h[:n - k]
        else:
            kk = -k
            grad[kk:] -= wk * h[:n - kk]
            grad[:n - kk] -= wk * h[kk:]
    return grad


def projected_gradient_descent(h0, time_limit, rng,
                              init_alpha=0.3,
                              max_iters=2000,
                              T_start=0.2,
                              T_end=0.001):

    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    total = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(max_iters):
        if time.time() - start >= time_limit:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)
        grad = gradient_softmax(h, T)

            new_score = compute_c5(h_new)
            if new_score < best_score:
                h = h_new
                best_h = h_new.copy()
                best_score = new_score
                improved = True
                break
        if not improved:
            # could keep h unchanged
            pass
    return best_h, best_score


def optimize_scale(start_h, time_limit, rng):
    """Run many short gradient descents within allocated time."""
    end_time = time.time() + time_limit
    best_h = start_h.copy()
    best_score = compute_c5(best_h)
    n = best_h.shape[0]
    total = n / 2.0

    while time.time() < end_time - 0.1:
        remaining = end_time - time.time()
        run_time = min(1.0, remaining)
        if rng.random() < 0.3:
            h0 = random_balanced_binary(n, rng).astype(float)
        else:
            h0 = best_h.copy()
            noise = rng.normal(scale=0.1, size=n)
            h0 = np.clip(h0 + noise, 0.0, 1.0)
            h0 = project_onto_box_sum(h0, total)

        h_opt, sc_opt = projected_gradient_descent(h0, run_time, rng)
        if sc_opt < best_score:
            best_h = h_opt.copy()
            best_score = sc_opt
            if best_score <= 0.3808:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]
    fractions = {64: 0.15, 128: 0.20, 256: 0.30, 512: 0.25}

    for n in scales:
        alloc = fractions[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        if best_h.shape[0] != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)

        h_opt, sc_opt = optimize_scale(base, alloc, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        if best_score <= 0.3808:
            break

    # final refinement with remaining budget
    remaining = time_left()
    if remaining > 0.5:
        h_opt, sc_opt = projected_gradient_descent(best_h, remaining, rng)
        if sc_opt < best_score:
            best_h = h_opt
            best_score = sc_opt

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
