def compute_subgrad(h):
    n = h.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    # compute correlation via FFT
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n-1] * (2.0 / n)
    max_val = corr.max()
    # tolerance for argmax
    tol = 1e-12
    idxes = np.where(corr >= max_val - tol)[0]
    # compute gradient: average over these shifts
    grad = np.zeros_like(h)
    for idx in idxes:
        shift = idx - (n - 1)
        # compute gradient for this shift:
        s = shift
        if s >= 0:
            # first term: i from 0 to n-1-s
            i_range = np.arange(n - s)
            grad[i_range] += 1.0 - h[i_range + s]
            # second term: j from s to n-1
            j_range = np.arange(s, n)
            grad[j_range] += -h[j_range - s]
        else:
            d = -s
            # first term: i from d to n-1
            i_range = np.arange(d, n)
            grad[i_range] += 1.0 - h[i_range - d]
            # second term: j from 0 to n-1-d
            j_range = np.arange(0, n - d)
            grad[j_range] += -h[j_range + d]
    grad /= len(idxes)
    return grad, max_val

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    # handle trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure bounds are proper
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1] * (2.0 / n)
    return corr.max()

def gradient_descent_momentum(h0, target_sum, time_left, step0=0.5, beta=0.9, max_iter=50000):
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad, c5 = compute_subgrad(h)
        # optional: could use c5 for early termination or adjusting step
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target_sum)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c5

def compute_subgrad(h):
    n = h.shape[0]
    # FFT for correlation
    N = 1 << ((2 * n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1] * (2.0 / n)
    max_val = corr.max()
    # tolerance for argmax
    tol = 1e-12
    idxes = np.where(corr >= max_val - tol)[0]
    grad = np.zeros_like(h)
    for idx in idxes:
        shift = idx - (n - 1)
        if shift >= 0:
            s = shift
            # first term
            i_range = np.arange(n - s)
            grad[i_range] += 1.0 - h[i_range + s]
            # second term
            j_range = np.arange(s, n)
            grad[j_range] += -h[j_range - s]
        else:
            d = -shift
            # first term
            i_range = np.arange(d, n)
            grad[i_range] += 1.0 - h[i_range - d]
            # second term
            j_range = np.arange(0, n - d)
            grad[j_range] += -h[j_range + d]
    grad /= len(idxes)
    return grad, max_val
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1] * (2.0 / n)
    return corr.max()

def compute_subgrad(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1] * (2.0 / n)
    max_val = corr.max()
    tol = 1e-12
    idxes = np.where(corr >= max_val - tol)[0]
    grad = np.zeros_like(h)
    for idx in idxes:
        shift = idx - (n - 1)
        if shift >= 0:
            s = shift
            i_range = np.arange(n - s)
            grad[i_range] += 1.0 - h[i_range + s]
            j_range = np.arange(s, n)
            grad[j_range] += -h[j_range - s]
        else:
            d = -shift
            i_range = np.arange(d, n)
            grad[i_range] += 1.0 - h[i_range - d]
            j_range = np.arange(0, n - d)
            grad[j_range] += -h[j_range + d]
    grad /= len(idxes)
    return grad, max_val

def gradient_descent_momentum(h0, target_sum, time_left, step0=0.5, beta=0.9, max_iter=50000):
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad, cur_c5 = compute_subgrad(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target_sum)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new.copy()
        # occasional binary projection check
        if it % 500 == 0:
            bin_h = binary_proj(h_new, target_sum)
            c5_bin = compute_c5_fast(bin_h)
            if c5_bin < best_c5:
                best_c5, best_h = c5_bin, bin_h.copy()
        h = h_new
        it += 1
    return best_h, best_c5

def binary_proj(h, target_sum):
    n = h.shape[0]
    k = int(round(target_sum))
    idx = np.argpartition(-h, k)[:k]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
