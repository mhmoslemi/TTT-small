    n = h.size
    L = 2*n - 1
    best_h = h
    best_corr = corr
    best_max = np.max(corr)
    # Try loop
    it = 0
    while it < max_iter and time_left() > 0.0:
        it += 1
        # Identify worst shift(s)
        k_idx = np.argmax(best_corr)
        k = k_idx - (n - 1)
        # Compute gradient for each i for this shift
        # Compute term1: for each i where i + k ∈ [0,n-1] -> 1 - h[i+k]
        term1 = np.zeros(n)
        if k >= 0:
            term1[:n - k] = 1.0 - best_h[k:]
        else:
            m = -k
            term1[m:] = 1.0 - best_h[:n - m]
        # Compute term2: for each i where i - k ∈ [0,n-1] -> h[i - k]
        term2 = np.zeros(n)
        if k >= 0:
            term2[k:] = best_h[:n - k]
        else:
            m = -k
            term2[:n - m] = best_h[m:]

        if np.all(g <= 0) or np.all(g >= 0):

            break

        j_min = np.argmin(g)  # most negative
        # Determine delta

        if delta <= 0:

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand search interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as floats."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_full_fft(h):
    """Full cross‑correlation of h and (1‑h) (unscaled)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Sub‑gradient of max‑correlation term (averaged over maximal shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts):
        grad /= len(shifts)
    return grad


def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def mass_transfer_improvement(h, corr, max_iter, time_left, target=None):
    """Greedy mass‑transfer moves aimed at reducing the maximal correlation."""
    n = h.shape[0]
    L = 2 * n - 1
    shifts_arr = np.arange(-(n - 1), n, dtype=int)
    best_corr = corr.copy()
    best_max = np.max(best_corr)

    for _ in range(max_iter):
        if time_left() <= 0.0:
            break
        k_idx = np.argmax(best_corr)
        k = k_idx - (n - 1)

        # gradient of the worst shift w.r.t each position
        g = np.zeros(n)
        if k >= 0:
            g[:n - k] = 1.0 - h[k:]
            g[k:] -= h[:n - k]
        else:
            m = -k
            g[m:] = 1.0 - h[:n - m]
            g[:n - m] -= h[m:]

        pos = g > 0
        neg = g < 0
        if not np.any(pos) or not np.any(neg):
            break
        i_max = np.argmax(g)          # best donor (positive gradient)
        j_min = np.argmin(g)          # best receiver (negative gradient)
        delta = min(h[i_max], 1.0 - h[j_min])
        if delta <= 0:
            break
        delta *= 0.5  # move only half the possible amount

        # gradient vectors for the two indices (full shift range)
        pos_i = i_max + shifts_arr
        neg_i = i_max - shifts_arr
        mask_pos_i = (pos_i >= 0) & (pos_i < n)
        mask_neg_i = (neg_i >= 0) & (neg_i < n)
        g_i = np.zeros(L)
        g_i[mask_pos_i] = 1.0 - h[pos_i[mask_pos_i]]
        g_i[mask_neg_i] -= h[neg_i[mask_neg_i]]

        pos_j = j_min + shifts_arr
        neg_j = j_min - shifts_arr
        mask_pos_j = (pos_j >= 0) & (pos_j < n)
        mask_neg_j = (neg_j >= 0) & (neg_j < n)
        g_j = np.zeros(L)
        g_j[mask_pos_j] = 1.0 - h[pos_j[mask_pos_j]]
        g_j[mask_neg_j] -= h[neg_j[mask_neg_j]]

        # try the move
        new_corr = best_corr + delta * (g_j - g_i)
        new_max = np.max(new_corr)
        if new_max < best_max - 1e-12:
            h[i_max] -= delta
            h[j_min] += delta
            best_corr = new_corr
            best_max = new_max
            if target is not None and best_max * (2.0 / n) <= target:
                break
            continue

        # try a smaller step
        delta2 = delta * 0.5
        new_corr2 = best_corr + delta2 * (g_j - g_i)
        new_max2 = np.max(new_corr2)
        if new_max2 < best_max - 1e-12:
            h[i_max] -= delta2
            h[j_min] += delta2
            best_corr = new_corr2
            best_max = new_max2
            if target is not None and best_max * (2.0 / n) <= target:
                break
            continue
        break
    return h, best_corr, best_max


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # initialise base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128)

    scales = [128, 256, 512, 1024]
    best_h = None
    best_score = np.inf
    TARGET = 0.3808

    for n in scales:
        if time_left() < 5.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # multiple gradient‑descent restarts
        restarts = 7
        for r in range(restarts):
            if time_left() < 2.0:
                break
            h0 = base + rng.normal(scale=0.05, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)
            h_opt, sc_opt = gradient_descent(
                h0, n_iter=120000, step0=0.7, beta=0.9,
                rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_score <= TARGET:
            break

        # greedy mass‑transfer refinement
        if best_h is not None and best_h.size == n:
            corr = compute_corr_full_fft(best_h)
            best_h, corr, _ = mass_transfer_improvement(
                best_h, corr, max_iter=5000, time_left=time_left, target=TARGET)
            best_h = project_onto_box_sum(best_h, n / 2.0)
            best_score = compute_c5_fft(best_h)

        # prepare for finer scale
        base = best_h.copy() if best_h is not None else base

        if best_score <= TARGET:
            break

    # final intensive refinement at current resolution
    if best_h is not None and time_left() > 3.0:
        n = best_h.size
        h_final, sc_final = gradient_descent(
            best_h, n_iter=250000, step0=0.5, beta=0.9,
            rng=rng, time_left=time_left)
        if sc_final < best_score:
            best_h, best_score = h_final, sc_final
        corr = compute_corr_full_fft(best_h)
        best_h, corr, _ = mass_transfer_improvement(
            best_h, corr, max_iter=5000, time_left=time_left, target=TARGET)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5_fft(best_h)

    # random binary exploration as last‑ditch effort
    while time_left() > 2.0 and best_h is not None:
        n = best_h.size
        h_bin = random_binary_balanced(n, rng)
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_h, best_score = h_bin, sc_bin
            corr = compute_corr_full_fft(best_h)
            best_h, corr, _ = mass_transfer_improvement(
                best_h, corr, max_iter=2000, time_left=time_left, target=TARGET)
            best_h = project_onto_box_sum(best_h, n / 2.0)
            best_score = compute_c5_fft(best_h)

    # fallback if nothing found
    if best_h is None:
        n = 512
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, best_h.size
