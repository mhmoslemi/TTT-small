        seq[i] = state & 1
        fb = 0
        for t in taps:
            fb ^= (state >> (t-1)) & 1
        state = ((state << 1) | fb) & mask

    target_sum = n // 2
    if ones > target_sum:
        # turn some ones to zeros

        need = ones - target_sum
        seq[one_idx[:need]] = 0
    elif ones < target_sum:
        zero_idx = np.where(seq == 0)[0]
        need = target_sum - ones
        seq[zero_idx[:need]] = 1
    # Trim or pad

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        ns = n - shift
        term1 = np.zeros(n); term1[:ns] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:ns]
    else:
        s = -shift
        ns = n - s
        term1 = np.zeros(n); term1[s:] = h[:ns]
        term2 = np.zeros(n); term2[:ns] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h, best_s = h.copy(), compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def binary_swap_anneal(h0, rng, time_left, max_iters=400000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i, j = rng.choice(ones), rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def generate_m_sequence(n, rng):
    # Degree-10 LFSR with primitive polynomial x^10 + x^3 + 1
    m = 10
    taps = [10, 3]
    state = 1
    mask = (1 << m) - 1
    period = (1 << m) - 1
    seq = np.empty(period, dtype=float)
    for i in range(period):
        seq[i] = state & 1
        fb = 0
        for t in taps:
            fb ^= (state >> (t - 1)) & 1
        state = ((state << 1) | fb) & mask
    target_sum = n // 2
    ones = int(seq.sum())
    if ones > target_sum:
        one_idx = np.where(seq == 1)[0]
        seq[one_idx[:ones - target_sum]] = 0
    elif ones < target_sum:
        zero_idx = np.where(seq == 0)[0]
        seq[zero_idx[:target_sum - ones]] = 1
    if period >= n:
        return seq[:n].copy()
    out = np.zeros(n, dtype=float)
    out[:period] = seq
    cur = int(out.sum())
    if cur < target_sum:
        zero_idx = np.where(out == 0)[0]
        out[zero_idx[:target_sum - cur]] = 1
    elif cur > target_sum:
        one_idx = np.where(out == 1)[0]
        out[one_idx[:cur - target_sum]] = 0
    return out

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    n = 512
    target_sum = n // 2
    best_h = None
    best_c5 = np.inf
    binary_candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init_res = resample(init, n) if init.size != n else init.copy()
        idx = np.argpartition(-init_res, target_sum)[:target_sum]
        bin_init = np.zeros(n, dtype=float)
        bin_init[idx] = 1.0
        binary_candidates.append(bin_init)
    binary_candidates.append(thue_morse_binary(n))
    for _ in range(5):
        binary_candidates.append(random_binary_balanced(n, rng))
    binary_candidates.append(generate_m_sequence(n, rng))
    # Evaluate and anneal binary candidates
    for h_bin in binary_candidates:
        if time_left() < 5.0:
            break
        c5 = compute_c5_fast(h_bin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_bin.copy()
        h_ann, c5_ann = binary_swap_anneal(h_bin, rng, time_left, max_iters=200000)
        if c5_ann < best_c5:
            best_c5, best_h = c5_ann, h_ann.copy()
        if best_c5 <= 0.3808:
            return best_h, best_c5, n
    # Gradient-descent refinements
    if best_h is not None and time_left() > 5.0:
        for _ in range(3):
            if time_left() < 5.0:
                break
            pert = best_h + rng.normal(scale=0.1, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = project_onto_box_sum(pert, target_sum)
            h_gd, c5_gd = gradient_descent_momentum(pert, time_left, step0=0.5, beta=0.9, max_iter=80000)
            if c5_gd < best_c5:
                best_c5, best_h = c5_gd, h_gd.copy()
            if best_c5 <= 0.3808:
                return best_h, best_c5, n
        while time_left() > 10.0:
            v = rng.random(n)
            v = project_onto_box_sum(v, target_sum)
            h_gd, c5_gd = gradient_descent_momentum(v, time_left, step0=0.5, beta=0.9, max_iter=50000)
            if c5_gd < best_c5:
                best_c5, best_h = c5_gd, h_gd.copy()
            if best_c5 <= 0.3808:
                break
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), target_sum)
    else:
        best_h = project_onto_box_sum(best_h, target_sum)
    best_c5 = compute_c5_fast(best_h)
    return best_h, best_c5, n
