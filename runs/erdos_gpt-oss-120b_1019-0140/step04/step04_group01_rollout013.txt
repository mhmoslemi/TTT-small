def binary_greedy_optimize(h_bin, time_left_func, max_attempts=50000):
    n = h_bin.size
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin <= 0.5)[0]
    best_h = h_bin.copy()
    best_score = compute_c5_fft(best_h)
    attempts = 0
    while attempts < max_attempts and time_left_func() > 0.0:
        if ones.size == 0 or zeros.size == 0: break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        best_h[i] = 0.0
        best_h[j] = 1.0
        new_score = compute_c5_fft(best_h)
        if new_score < best_score:
            best_score = new_score
            # update ones and zeros sets
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h <= 0.5)[0]
        else:
            # revert
            best_h[i] = 1.0
            best_h[j] = 0.0
        attempts += 1
    return best_h, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

            return best_h, best_score, best_h.size
        h_bin_opt, sc_bin_opt = binary_greedy_optimize(best_h, rng, time_left,
                                                       max_attempts=30000)
        if sc_bin_opt < best_score:
            best_h, best_score = h_bin_opt, sc_bin_opt
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 4: binary annealing
    if time_left() > 5.0:
        h_ann, sc_ann = binary_swap_anneal(best_h, rng, time_left,
                                          max_iters=300000, T0=0.2,
                                          decay=0.99995, Tmin=1e-5)
        if sc_ann < best_score:
            best_h, best_score = h_ann, sc_ann
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 5: upscale to larger resolution and refine
    if time_left() > 20.0:
        n_large = 1024
        h_large = upsample(best_h, n_large)
        h_large = project_onto_box_sum(h_large, n_large / 2.0)
        sc_large = compute_c5_fft(h_large)
        if sc_large < best_score:
            best_h, best_score = h_large, sc_large
        if best_score <= target:
            return best_h, best_score, best_h.size
        max_iter_large = 80000
        h_large_opt, sc_large_opt = gradient_descent_momentum_with_time(best_h, time_left,
                                                                        step0=1.0, beta=0.9,
                                                                        max_iter=max_iter_large)
        if sc_large_opt < best_score:
            best_h, best_score = h_large_opt, sc_large_opt
        if best_score <= target:
            return best_h, best_score, best_h.size

        # binary refinement at large size
        if time_left() > 5.0:
            h_bin_large = make_binary_from_continuous(best_h)
            sc_bin_large = compute_c5_fft(h_bin_large)
            if sc_bin_large < best_score:
                best_h, best_score = h_bin_large, sc_bin_large
            if best_score <= target:
                return best_h, best_score, best_h.size
            h_bin_large_opt, sc_bin_large_opt = binary_swap_anneal(best_h, rng, time_left,
                                                                  max_iters=200000,
                                                                  T0=0.2, decay=0.99995, Tmin=1e-5)
            if sc_bin_large_opt < best_score:
                best_h, best_score = h_bin_large_opt, sc_bin_large_opt

    # final projection and return
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
