def smooth_c5_and_grad(h, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    # to avoid overflow
    exp_term = np.exp(alpha * (corr - max_corr))
    w = exp_term / exp_term.sum()
    # compute weighted sum of h shifted
    n = h.size
    weighted_h = np.zeros_like(h)
    for ti, weight in enumerate(w):
        shift = ti - (n - 1)
        if shift >= 0:
            # h[i - shift] => for i from shift..n-1, idx = i - shift in [0..n-1-shift]
            weighted_h[shift:] += weight * h[:n - shift]
        else:
            # shift negative => i - shift = i + |shift|
            s = -shift
            weighted_h[:n - s] += weight * h[s:]
    # compute grad
    grad = (2.0 / n) * (1 - 2 * weighted_h)  # multiply by dx = 2/n
    # objective approx max = (2/n) * (max_corr + logsumexp(alpha*(corr-max_corr))/alpha)
    # Actually original max c5 = (2/n) * max_corr.
    # Smooth approximation:
    smooth_c5 = (2.0 / n) * (max_corr + np.log(np.sum(exp_term)) / alpha)
    return smooth_c5, grad

def smooth_c5_and_grad(h, alpha):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    # stability
    exp_term = np.exp(alpha * (corr - max_corr))
    w = exp_term / exp_term.sum()
    weighted_h = np.zeros_like(h)
    # loops
    for ti, weight in enumerate(w):
        shift = ti - (n - 1)
        if shift >= 0:
            # i from shift..n-1 have index i-shift >=0
            weighted_h[shift:] += weight * h[:n - shift]
        else:
            s = -shift
            weighted_h[:n - s] += weight * h[s:]
    # gradient
    grad = dx * (1 - 2 * weighted_h)
    # smooth approximation value
    smooth_val = dx * (max_corr + np.log(np.sum(exp_term)) / alpha)
    return smooth_val, grad

        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def smooth_c5_and_grad(h, alpha):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    exp_term = np.exp(alpha * (corr - max_corr))
    w = exp_term / exp_term.sum()
    weighted_h = np.zeros_like(h)
    for ti, weight in enumerate(w):
        shift = ti - (n - 1)
        if shift >= 0:
            weighted_h[shift:] += weight * h[:n - shift]
        else:
            s = -shift
            weighted_h[:n - s] += weight * h[s:]
    grad = dx * (1.0 - 2.0 * weighted_h)
    smooth_val = dx * (max_corr + np.log(exp_term.sum()) / alpha)
    return smooth_val, grad

def smooth_grad_descent(h0, time_left, max_iter=300):
    h = project_onto_box_sum(h0, h0.size / 2.0)
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    alpha = 5.0
    step0 = 1.0
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        _, grad = smooth_c5_and_grad(h, alpha)
        step = step0
        while step > 1e-6:
            h_new = h - step * grad
            h_new = project_onto_box_sum(h_new, n / 2.0)
            new_score = compute_c5(h_new)
            if new_score < best_score - 1e-9:
                break
            step *= 0.5
        else:
            step0 *= 0.7
            alpha = min(200.0, alpha * 1.1)
            continue
        h = h_new
        best_h, best_score = h_new.copy(), new_score
        step0 = min(2.0, step0 * 1.02)
        alpha = min(200.0, alpha * 1.05)
    return best_h, best_score

def refine_max_grad_descent(h0, cur_best, time_left, max_iter=2000):
    h = h0.copy()
    best_h = h.copy()
    best_score = cur_best
    n = h.size
    s = n / 2.0
    momentum = np.zeros_like(h)
    it = 0
    while it < max_iter and time_left() > 0:
        it += 1
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = 0.9 * momentum + 0.1 * grad
        eta = 0.5 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    n_target = 256
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    if base.shape[0] != n_target:
        base = upsample(base, n_target)
    base = project_onto_box_sum(base, n_target / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    while time_left() > 8.0:
        if rng.random() < 0.5:
            h_start = best_h.copy()
            h_start += rng.normal(scale=0.02, size=n_target)
        else:
            h_start = random_binary_balanced(n_target, rng)
        if rng.random() < 0.5:
            p = rng.uniform(0.1, 0.4)
            h_start = (1 - p) * best_h + p * h_start
        h_start = project_onto_box_sum(h_start, n_target / 2.0)

        max_iter_smooth = int(min(300, max(50, time_left() * 2)))
        h_smooth, score_smooth = smooth_grad_descent(h_start, time_left, max_iter=max_iter_smooth)

        if time_left() > 5.0:
            max_refine = int(min(2000, max(200, time_left() * 10)))
            h_ref, score_ref = refine_max_grad_descent(h_smooth, score_smooth, time_left, max_iter=max_refine)
        else:
            h_ref, score_ref = h_smooth, score_smooth

        if score_ref < best_score:
            best_h, best_score = h_ref, score_ref

        if best_score <= 0.3808:
            break

    if time_left() > 25.0:
        n_high = 512
        h_up = upsample(best_h, n_high)
        h_up = project_onto_box_sum(h_up, n_high / 2.0)
        max_iter_smooth = int(min(400, max(80, time_left() * 2)))
        h_up_s, score_up_s = smooth_grad_descent(h_up, time_left, max_iter=max_iter_smooth)
        if time_left() > 5.0:
            max_refine = int(min(2500, max(300, time_left() * 10)))
            h_up_ref, score_up_ref = refine_max_grad_descent(h_up_s, score_up_s, time_left, max_iter=max_refine)
        else:
            h_up_ref, score_up_ref = h_up_s, score_up_s
        if score_up_ref < best_score:
            best_h, best_score = h_up_ref, score_up_ref
            n_target = n_high

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
