def simulated_annealing(h0, time_limit, rng):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    
    start = time.time()
    # total steps approximated by some large number; we compute dynamic schedule
    # We'll loop while time left > small epsilon
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = max(T0 * (Tf/T0)**frac, Tf) # exponential cooling
        # random move
        i = rng.integers(n)
        # choose direction (transfer from i to j or j to i)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # decide direction: move from i to j if h[i] > 0 and h[j] < 1 else maybe reverse
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 0:
            continue
        move_scale = min(1.0, T / T0 * 2.0)
        delta = max_delta * rng.random() * move_scale
        h[src] -= delta
        h[dst] += delta
        new_score = compute_c5(h)
        dscore = new_score - cur_score
        if dscore < 0 or rng.random() < np.exp(-dscore / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[src] += delta
            h[dst] -= delta
    return best_h, best_score
```

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    # base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)   # we need thue_morse function
    base = project_onto_box_sum(base, base.shape[0] / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)
    # hierarchical levels
    levels = [64, 128, 256, 512]
    fractions = {64: 0.12, 128: 0.15, 256: 0.25, 512: 0.38}
    for n in levels:
        if time_left() <= 0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        # ensure feasibility
        base = project_onto_box_sum(base, n/2.0)
        # allocate time
        alloc = min(fractions[n] * budget_s, time_left())
        if alloc <= 0.0:
            continue
        # run SA
        cand_h, cand_score = simulated_annealing(base, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        # set base to cand_h to continue upsampling from best found
        base = cand_h
        # break early if bound meets target
        if best_score <= 0.3808:
            break
    # final refinement
    while time_left() > 0.5:
        n = best_h.size
        # random balanced binary perturbation
        cand = random_balanced_binary(n, rng)
        cand = project_onto_box_sum(cand, n/2.0)
        sc = compute_c5(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()
        # small mass transfer move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.3
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
        # optionally run short SA if any time left > 10 s else break
        if time_left() > 10.0:
            # run brief SA with remaining time portion
            cand_h2, cand_score2 = simulated_annealing(best_h, min(5.0, time_left()), rng)
            if cand_score2 < best_score:
                best_score = cand_score2
                best_h = cand_h2.copy()
    # final projection to ensure sum exactly
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    # define points in [0,1) for periodic? Use linspace with endpoint=False.
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    arr_new = np.interp(x_new, x_old, arr)
    return arr_new
```

def active_shift(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    # compute index of max (0-indexed). For full mode length 2*n-1.
    idx = np.argmax(corr)
    # compute lag k: correlation index i corresponds to lag = i - (n-1)
    k = idx - (n - 1)
    return k, corr[idx]
```

def gradient_descent_opt(h0, time_limit, rng):
    h = h0.copy()
    n = h.size
    start = time.time()
    # parameters
    step0 = 0.2
    step_min = 1e-4
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = step0 * (1 - frac) + step_min * frac
        k, _ = active_shift(h)
        # compute gradient
        grad = np.zeros_like(h)
        # first term
        if k >= 0:
            grad[:n-k] += 1.0 - h[k:]
            grad[k:] -= h[:n-k]
        else:
            kk = -k
            grad[kk:] += 1.0 - h[:n-kk]
            grad[:n-kk] -= h[kk:]
        # scale by dx
        grad *= (2.0 / n)
        # take step
        h_new = h - step * grad
        # project
        h_new = project_onto_box_sum(h_new, n/2.0)
        new_score = compute_c5(h_new)
        # accept if improvement
        if new_score < compute_c5(h):
            h = h_new
    return h, compute_c5(h)

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_c5_full(h):
    """Return C5 and the index of the maximizing shift in the full correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    return corr[idx] * (2.0 / n), idx


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def simulated_annealing(h0, time_limit, rng):
    """SA preserving sum and box constraints."""
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.04
    Tf = 1e-5
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = max(T0 * (Tf / T0) ** frac, Tf)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # decide direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 0.0:
            continue
        scale = min(1.0, (T / T0) * 2.0)
        delta = max_delta * rng.random() * scale
        h[src] -= delta
        h[dst] += delta
        new_score = compute_c5(h)
        d = new_score - cur_score
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[src] += delta
            h[dst] -= delta
    return best_h, best_score


def gradient_descent_opt(h0, time_limit, rng):
    """Projected gradient descent on the active shift."""
    h = h0.copy()
    n = h.shape[0]
    start = time.time()
    step0 = 0.5
    step_min = 1e-4
    while time.time() - start < time_limit:
        cur_score, cur_idx = compute_c5_full(h)
        k = cur_idx - (n - 1)
        grad = np.zeros_like(h)
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[kk:] += 1.0 - h[:n - kk]
            grad[:n - kk] -= h[kk:]
        grad *= (2.0 / n)
        elapsed = time.time() - start
        frac = elapsed / time_limit

        if new_score < cur_score:
