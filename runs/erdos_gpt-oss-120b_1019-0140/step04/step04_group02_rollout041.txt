import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h), unscaled."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def grad_from_weights(h, w):
    """Gradient of weighted sum Σ w_l C[l] w.r.t. h."""
    n = h.shape[0]
    L = w.size  # = 2n‑1
    h_pad = np.concatenate([h, np.zeros(n - 1)], dtype=float)
    w_fft = np.fft.fft(w, L)
    h_fft = np.fft.fft(h_pad, L)
    conv1 = np.fft.ifft(w_fft * h_fft).real
    w_rev = w[::-1]
    conv2 = np.fft.ifft(np.fft.fft(w_rev, L) * h_fft).real
    grad = 1.0 - (conv1[n - 1:n - 1 + n] + conv2[n - 1:n - 1 + n])
    return grad

    start_time = time.time()

    # base construction

    best_h, best_score = adam_optimize(best_h, coarse_sched,

    # fine grid (512 points)
    n_fine = 512
    if best_h.shape[0] != n_fine:
        best_h = upsample(best_h, n_fine)
    best_h = project_onto_box_sum(best_h, n_fine / 2.0)

    fine_sched = [(10.0, 4000), (20.0, 4000), (40.0, 4000),
                  (80.0, 4000), (120.0, 4000)]
    best_h, best_score = adam_optimize(best_h, fine_sched,
        lr=0.02, beta1=0.9, beta2=0.999, epsilon=1e-8,
        start_time=start_time, budget_s=budget_s,
        rng=rng, best_h=best_h, best_score=best_score)

    # final projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)

    # optional last‑minute binary refinements if time permits
    TARGET = 0.3808
    while budget_s - (time.time() - start_time) > 5.0:
        n = best_h.size
        # top‑n/2 rounding
        idx = np.argsort(-best_h)
        h_bin = np.zeros_like(best_h)
        h_bin[idx[:n // 2]] = 1.0
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
            if best_score <= TARGET:
                break
        # random balanced binary
        h_rand = np.zeros(n, dtype=float)
        h_rand[rng.choice(n, size=n // 2, replace=False)] = 1.0
        sc_rand = compute_c5_fft(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
            if best_score <= TARGET:
                break
        # small mass transfer between two positions
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.02, 0.1)
        if delta > 1e-12:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5_fft(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                if best_score <= TARGET:
                    break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
