def compute_c5(h):
    n = h.shape[0]
    b = 1.0 - h
    rb = b[::-1]
    N = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(rb, N)
    conv = np.fft.irfft(fa * fb, N)
    # conv length N, we need first 2*n-1 entries of convolution
    corr = conv[:2*n-1]
    # adjust scaling: each term sum_i h_i * (1 - h_j) for index combos. Since we used convolution w/out dx factor.
    # Original compute_c5: corr = np.correlate(h, 1-h, mode='full') * dx where dx = 2/n.
    # So after computing conv which is same as np.correlate (full) because we used direct convolution of a and reversed(b).
    # So multiply by dx = 2.0/n.
    return np.max(corr) * (2.0 / n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_corr_full_fft(h):
    """Full correlation vector of h and (1-h) (scaled by dx)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    ha = np.fft.rfft(h, N)
    hb = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(ha * hb, N)[:2 * n - 1]
    corr = conv * (2.0 / n)
    np.maximum(corr, 0.0, out=corr)
    return corr


def compute_c5_fft(h):
    return np.max(compute_corr_full_fft(h))


def compute_c5_np(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

    h_rev = h[::-1]
    hf_rev = np.fft.rfft(h_rev, N)
    conv2 = np.fft.irfft(wf * hf_rev, N)

    offset = n - 1
    idx = np.arange(n)
    conv1_slice = conv1[offset + idx]               # sum_s w_s * h[i - s]
    conv2_slice = conv2[(2 * n - 2) - idx]          # sum_s w_s * h[i + s]

    S = np.sum(corr ** p)
    f_p = S ** (1.0 / p) if S > 0.0 else 0.0
    scale = (f_p ** (1.0 - p)) * (2.0 / n) if f_p > 0.0 else 0.0
    grad = scale * (sum_w - (conv1_slice + conv2_slice))
    return grad


def gradient_descent_smooth_custom(h0, n_iter, step0, p, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        grad = gradient_smooth(h, p)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample_nearest(arr, new_n):
    factor = new_n // arr.shape[0]
    if factor * arr.shape[0] == new_n:
        return np.repeat(arr, factor)
    # fallback to linear interpolation
    x_old = np.linspace(0.0, 1.0, arr.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # initial construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    # hierarchical refinement
    for target_n in (128, 256, 512, 1024):
        if time_left() <= 5.0:
            break
        h = upsample_nearest(best_h, target_n)
        h = project_onto_box_sum(h, target_n / 2.0)

        # smooth gradient descent
        max_iters = int(min(60000, time_left() * 30))
        if max_iters > 0:
            step0 = 0.5 if target_n <= 256 else 0.3
            p = 12 if target_n <= 256 else 10
            h_opt, sc_opt = gradient_descent_smooth_custom(
                h, n_iter=max_iters, step0=step0, p=p, beta=0.95,
                rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt

        # short greedy hill‑climb to polish
        hill_iters = int(min(20000, time_left() * 20))
        if hill_iters > 0:
            h2 = best_h.copy()
            score2 = best_score
            n = h2.size
            for _ in range(hill_iters):
                if time_left() <= 0.0:
                    break
                i, j = rng.integers(0, n, size=2, replace=False)
                cap_i = h2[i]
                cap_j = 1.0 - h2[j]
                if cap_i <= 0.0 or cap_j <= 0.0:
                    continue
                delta = min(cap_i, cap_j) * rng.uniform(0.05, 0.5)
                h_new = h2.copy()
                h_new[i] -= delta
                h_new[j] += delta
                sc_new = compute_c5_fft(h_new)
                if sc_new < score2:
                    h2, score2 = h_new, sc_new
            if score2 < best_score:
                best_h, best_score = h2.copy(), score2

    # final intensive refinement at current best resolution
    if time_left() > 5.0:
        final_iters = int(min(80000, time_left() * 30))
        if final_iters > 0:
            best_h, best_score = gradient_descent_smooth_custom(
                best_h, n_iter=final_iters, step0=0.2, p=10,
                beta=0.95, rng=rng, time_left=time_left)

    # random restarts with short smooth descent
    while time_left() > 30.0:
        h_rand = random_binary_balanced(best_h.size, rng)
        iters = int(min(20000, time_left() * 20))
        if iters <= 0:
            break
        h_opt, sc_opt = gradient_descent_smooth_custom(
            h_rand, n_iter=iters, step0=0.4, p=10,
            beta=0.95, rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_h, best_score = h_opt.copy(), sc_opt

    # final projection and exact scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_np(best_h)
    return best_h, best_score, best_h.size
