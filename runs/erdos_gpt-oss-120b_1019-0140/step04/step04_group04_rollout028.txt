def simulated_annealing(h0, n_iter, rng, time_left):
    h = h0.copy()
    n = h.size
    s_target = n/2
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    # choose initial temperature based on typical improvement magnitude; maybe set T0 = 0.01 or compute from initial variance of scores?
    T0 = 0.01  # might need to adjust.
    for t in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # temperature schedule linear down to near 0
        T = T0 * (1 - t / n_iter)
        # propose move: choose i < j, move delta
        i, j = rng.choice(n, size=2, replace=False)
        # compute max delta
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            continue
        # scale factor based on iteration: larger at early iterations
        delta = max_delta * rng.random() * (0.5 + 0.5 * (1 - t/n_iter))
        # propose
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # ensure sum unchanged automatically, but maintain bounds
        new_score = compute_c5(h_new)
        # accept if better or with probability
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score)/max(T, 1e-12)):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, start, budget_s):
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if budget_s - (time.time() - start) <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt  # imported for compatibility; not used directly

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    f_lo = np.clip(v - lam_lo, lo, hi).sum()
    if f_lo < s:
        lam_lo = np.min(v) - 10 * hi
    f_hi = np.clip(v - lam_hi, lo, hi).sum()
    if f_hi > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        f_mid = np.clip(v - lam_mid, lo, hi).sum()
        if abs(f_mid - s) < 1e-12:
            break
        if f_mid > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, start, budget_s):
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if budget_s - (time.time() - start) <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h0, n_iter, rng, start, budget_s):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T0 = max(1e-4, cur_score * 0.02)
    for t in range(1, n_iter + 1):
        if budget_s - (time.time() - start) <= 0:
            break
        T = T0 * (1.0 - t / n_iter)
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        scale = 0.5 + 0.5 * (1.0 - t / n_iter)
        delta = max_delta * rng.random() * scale
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score
        if rng.random() < 0.001:
            pert = rng.normal(scale=0.01, size=n)
            h_pert = h + pert
            h_pert = project_onto_box_sum(h_pert, n / 2.0)
            pert_score = compute_c5(h_pert)
            if pert_score < cur_score:
                h = h_pert
                cur_score = pert_score
                if cur_score < best_score:
                    best_h = h.copy()
                    best_score = cur_score
    return best_h, best_score
