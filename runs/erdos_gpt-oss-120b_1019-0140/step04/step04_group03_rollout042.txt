def compute_c5_fast(h):
    n = h.shape[0]
    f = h - 0.5
    # compute full linear correlation of f with f
    M = 2*n - 1
    # next power of two for FFT
    L = 1 << (M-1).bit_length()
    fft_f = np.fft.rfft(f, n=L)
    # correlation: ifft(fft_f * np.conj(fft_f)). real part
    corr = np.fft.irfft(fft_f * np.conj(fft_f), n=L)[:M]
    # corr is sum_i f[i] * f[i + k] for k from -(n-1) to (n-1)
    # Convert to min over all k (including zero shift? Yes, include k=0), but C5 uses max over k of 0.5 - (2/n)*corr[k_offset].
    # So C5 = max_k (0.5 - (2/n)*corr[k])
    # Equivalent to 0.5 - (2/n)*min_k corr[k].
    min_corr = np.min(corr)
    C5 = 0.5 - (2.0 / n) * min_corr
    return C5

def compute_c5_fast(h):
    # Compute correlation of h with 1-h using FFT for linear convolution.
    n = h.shape[0]
    # Compute convolution length = 2*n-1
    M = 2*n - 1
    L = 1 << ((M - 1).bit_length())
    # Zero pad both arrays to length L
    a = np.fft.rfft(h, n=L)
    b = np.fft.rfft(1.0 - h, n=L)
    # Linear convolution is ifft(a * b).real
    conv = np.fft.irfft(a * b, n=L)[:M]
    return np.max(conv) * (2.0 / n)

def simulated_annealing(h, time_limit, rng):
    n = h.size
    cur_h = h.copy()
    cur_score = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.02
    T_end = 1e-6
    total_steps = int(1e6)  # huge, but will break when time runs out.
    # But we need to compute in loop until time used
    step = 0
    while time.time() - start < time_limit and step < total_steps:
        # schedule temperature
        T = T0 * (1 - step / total_steps) + T_end * (step / total_steps)
        # propose move:
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5_fast(new_h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_h, best_score = cur_h.copy(), cur_score
        step += 1
    return best_h, best_score

def gradient_descent_momentum(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time.time() - start >= max_time:
            break
        conv = compute_corr_fast(h)  # length 2n-1
        max_corr = conv.max()
        # find shift indices where conv equals max_corr within tolerance
        shifts = np.where(conv >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradient_worst_shift(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # optional injection of noise occasionally
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project onto box sum
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate

        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_corr_fast(h):
    n = h.shape[0]
    M = 2 * n - 1
    L = 1 << ((M - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    V = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(V), n=L)[:M]
    return corr

def compute_c5_fast(h):
    n = h.shape[0]
    corr = compute_corr_fast(h)
    return np.max(corr) * (2.0 / n)

def compute_gradient_worst_shift(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            grad[:sz] += 1.0 - h[shift:]
            grad[shift:] += 1.0 - h[:sz]
        else:
            s = -shift
            sz = n - s
            grad[s:] += 1.0 - h[:sz]
            grad[:sz] += 1.0 - h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = compute_corr_fast(h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradient_worst_shift(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5_fast(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h0, max_time, rng):
    start = time.time()
    cur_h = h0.copy()
    cur_score = compute_c5_fast(cur_h)
    best_h, best_score = cur_h.copy(), cur_score
    n = cur_h.size
    T0, T_end = 0.02, 1e-6
    total_steps = 200000
    for step in range(total_steps):
        if time.time() - start >= max_time:
            break
        T = T0 * (1 - step / total_steps) + T_end * (step / total_steps)
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5_fast(new_h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_h, cur_score = new_h, new_score
            if cur_score < best_score:
                best_h, best_score = cur_h.copy(), cur_score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_final = 1024
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = rng.random(64)
    base = upsample(base, n_final)
    base = project_onto_box_sum(base, n_final / 2.0)
    best_h, best_score = base.copy(), compute_c5_fast(base)

    # Gradient descent phase (≈60% of remaining time)
    elapsed = time.time() - start
    if budget_s - elapsed > 10:
        gd_time = (budget_s - elapsed) * 0.6
        max_iters = int(min(300000, max(50000, gd_time * 30000)))
        h_gd, sc_gd = gradient_descent_momentum(
            base, n_iter=max_iters, step0=0.2, beta=0.9,
            rng=rng, max_time=gd_time)
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd

    # Simulated annealing phase (remaining time)
    elapsed = time.time() - start
    if budget_s - elapsed > 5:
        sa_time = budget_s - elapsed
        h_sa, sc_sa = simulated_annealing(best_h, max_time=sa_time, rng=rng)
        if sc_sa < best_score:
            best_h, best_score = h_sa.copy(), sc_sa

    # Final polishing with a short gradient descent if time permits
    elapsed = time.time() - start
    if budget_s - elapsed > 5:
        gd_time = budget_s - elapsed
        h_fin, sc_fin = gradient_descent_momentum(
            best_h, n_iter=50000, step0=0.1, beta=0.9,
            rng=rng, max_time=gd_time)
        if sc_fin < best_score:
            best_h, best_score = h_fin.copy(), sc_fin

    # Ensure feasibility and recompute final score
    best_h = project_onto_box_sum(best_h, n_final / 2.0)
    best_score = compute_c5_fast(best_h)
    return best_h, best_score, n_final
