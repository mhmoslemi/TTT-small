    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0: break

        autocorr = np.correlate(h, h, mode='full')
        S = autocorr  # length 2n-1
        # compute weights
        max_S = np.max(S)
        w = np.exp(-alpha * (S - max_S))  # subtract max for stability
        w_rev = w[::-1]
        # pad h
        h_pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
        # convolutions
        # Use FFT conv
        conv_len = len(w) + len(h_pad) - 1
        L = 1 << (conv_len - 1).bit_length()
        fft_w = np.fft.fft(w, L)

        fft_h_pad = np.fft.fft(h_pad, L)
        term1 = np.fft.ifft(fft_w * fft_h_pad).real[(n-1):(n-1)+n]
        term2 = np.fft.ifft(fft_w_rev * fft_h_pad).real[(n-1):(n-1)+n]
        grad = term1 + term2
        # Normalize gradient maybe
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        # step schedule: step = step0 / sqrt(it)
        step = step0 / np.sqrt(it)
        h_new = h + step * grad
        # project
        h_new = project_onto_box_sum(h_new, s)

        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    # base h
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != base_n:
            base = upsample(base, base_n)
    else:

    base = project_onto_box_sum(base, base_n / 2.0)
    best_h, best_score = base.copy(), compute_c5(base)

    # Phase 1: soft-min gradient ascent with increasing alpha
    for alpha in [5.0, 20.0, 80.0]:
        if time_left() <= 0:
            break
        # number of iterations depending on remaining time

        max_iters = int(min(5000, max(500, time_left() * 5)))  # heuristic
        h_opt, sc_opt = softmin_gradient_ascent(base, alpha, step0=0.5, max_iters=max_iters, time_left=time_left)

        base = h_opt

    # Phase 2: Random binary restarts with gradient descent refinements (like previous)
    while time_left() > 8:
        n = best_h.size
        # random binary restart
        h_rand = random_binary_balanced(n, rng)
        h_rand = project_onto_box_sum(h_rand, n / 2.0)
        # gradient descent using only max_shifts like earlier? We could reuse previous grad descent but replaced maybe by smaller steps.

        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        # local pairwise tweak
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.1, 0.3)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

                        max(800, time_left() * 3)))
        h_fine, sc_fine = softmin_gradient_ascent(best_h, 120.0, 0.5,
                                                  iters, time_left)
        if sc_fine < best_score:
            best_score, best_h = sc_fine, h_fine.copy()

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
