import time
import numpy as np

def random_binary_balanced(n, rng):
    """Return a binary vector of length n with exactly n//2 ones."""
    h = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1
    return h

def upsample_binary(arr, new_n):
    """Linear up‑sample a binary array and keep exactly half the entries as ones."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.astype(np.int8).copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    cont = np.interp(x_new, x_old, arr.astype(float))
    # pick the largest new_n//2 values
    idx = np.argpartition(cont, -new_n // 2)[-new_n // 2 :]
    h = np.zeros(new_n, dtype=np.int8)
    h[idx] = 1
    return h

def compute_Sks(h):
    """S[k] = sum_i h[i] * h[i+k] for k=0..N-1 (non‑circular)."""
    N = h.shape[0]
    corr = np.correlate(h, h, mode="full")
    return corr[N - 1 :].astype(np.int64)

def compute_D(h):
    """D[k] = sum_i h[i] * (1 - h[i+k]) for k=0..N-1."""
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    return corr[N - 1 :].astype(np.int64)

def local_search(N, rng, time_left, target_c5):
    """Hill‑climb on max(D) = C5 * N/2."""
    # initial vector (random balanced)
    h = random_binary_balanced(N, rng)

    S = compute_Sks(h)          # length N, S[0] = N/2
    D = compute_D(h)            # length N, D[0] = 0
    cur_max = int(D.max())
    target_max = int(np.floor(target_c5 * N / 2.0 + 1e-9))

    ones = np.flatnonzero(h).astype(np.int64)
    zeros = np.flatnonzero(1 - h).astype(np.int64)

    best_h = h.copy()
    best_max = cur_max
    no_improve = 0

    while time_left() > 0.5:
        improved = False
        # when far from target we try few candidates, close to target we explore more
        attempts = 3 if cur_max > target_max else 10
        for _ in range(attempts):
            # pick a random 1 and a random 0
            i0 = int(ones[rng.integers(ones.size)])   # will become 0
            i1 = int(zeros[rng.integers(zeros.size)]) # will become 1

            delta_S = np.empty(N, dtype=np.int64)
            delta_S[0] = 0
            new_max = cur_max
            new_D = np.empty(N, dtype=np.int64)

            # evaluate the effect of the swap
            for k in range(N):
                if k == 0:
                    delta_Sk = 0
                else:
                    d = 0
                    if i0 + k < N:
                        d -= h[i0 + k]
                    if i0 - k >= 0:
                        d -= h[i0 - k]
                    if i1 + k < N and i1 + k != i0:
                        d += h[i1 + k]
                    if i1 - k >= 0 and i1 - k != i0:
                        d += h[i1 - k]
                    delta_Sk = d
                delta_S[k] = delta_Sk
                delta_Ak = (i1 + k < N) - (i0 + k < N)  # bool to int
                new_val = D[k] + delta_Ak - delta_Sk
                new_D[k] = new_val
                if new_val > cur_max:
                    new_max = new_val
                    break  # no need to continue, this swap cannot improve
            if new_max <= cur_max:
                # accept the swap
                h[i0] = 0
                h[i1] = 1
                # update index containers in O(1)
                pos0 = np.where(ones == i0)[0][0]
                pos1 = np.where(zeros == i1)[0][0]
                ones[pos0] = i1
                zeros[pos1] = i0
                # update S, D, cur_max
                S += delta_S
                D = new_D

                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                break

            if no_improve >= 20000:

                S = compute_Sks(h)
                D = compute_D(h)

                ones = np.flatnonzero(h).astype(np.int64)

                no_improve = 0
        else:
            no_improve = 0

            break
