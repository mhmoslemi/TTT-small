def compute_worst_shift_gradient(h):
    n = h.shape[0]
    conv = np.correlate(h, h, mode='full')
    min_conv = conv.min()
    # tolerance
    idxes = np.where(np.abs(conv - min_conv) < 1e-12)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            # forward shift
            sz = n - k
            # grad[:sz] -= h[k:]
            grad[:sz] -= h[k:]
            grad[k:] -= h[:sz]
        else:
            s = -k
            sz = n - s
            grad[s:] -= h[:sz]
            grad[:sz] -= h[s:]
    # average over shifts
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """
    Projects vector v onto the intersection of box [lo,hi]^n and hyperplane sum = target_sum.
    Uses bisection on lambda for the formula w_i = clip(v_i - lambda, lo, hi).
    """
    n = v.size
    # Early bounds
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # Determine initial lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # Ensure lam_low gives sum >= target, lam_high gives sum <= target, else expand
    def sum_at(lam):
        return np.clip(v - lam, lo, hi).sum()
    sum_low = sum_at(lam_low)
    sum_high = sum_at(lam_high)
    # Adjust bounds if necessary
    # If sum_low < target, lam_low too high, need to decrease lam_low
    while sum_low < target_sum:
        lam_low -= (hi - lo) * 2  # arbitrary large step
        sum_low = sum_at(lam_low)
    # If sum_high > target, lam_high too low, need to increase lam_high
    while sum_high > target_sum:
        lam_high += (hi - lo) * 2
        sum_high = sum_at(lam_high)
    # Bisection
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = sum_at(lam_mid)
        if abs(s - target_sum) < 1e-12:
            break
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_worst_shift_gradient(h):
    n = h.shape[0]
    conv = np.correlate(h, h, mode='full')
    min_conv = conv.min()
    idxes = np.where(np.abs(conv - min_conv) < 1e-12)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            sz = n - k
            grad[:sz] -= h[k:]
            grad[k:] -= h[:sz]
        else:
            s = -k
            sz = n - s
            grad[s:] -= h[:sz]
            grad[:sz] -= h[s:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto { lo<=x_i<=hi, sum x_i = target_sum } via bisection on λ."""

        lam_high += (hi - lo) * 2
        s_high = sum_at(lam_high)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:

    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    for it in range(1, max_iters + 1):
        if time.time() - start > time_limit:
            break
        grad = compute_worst_shift_gradient(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fast(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def simulated_annealing(h, cur_score, time_budget, rng):
    """Simple SA that transfers mass between coordinates and occasional swaps."""
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_score = cur_score

    remaining = max(0.0, time_budget - (time.time() - start))
    if remaining <= 0.0:
        return best_h, best_score

    total_steps = int(min(80000, max(5000, remaining * 20000)))
    T0 = 0.02
    T_final = 1e-7

    for step in range(total_steps):
        if time.time() - start >= time_budget:
            break
        T = T0 * ((total_steps - step) / total_steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)

        if rng.random() < 0.15:
            new_h = h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
        else:
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 0.0:
                continue
            delta = max_move * rng.random()
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_score = compute_c5_fast(new_h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    n0 = base.shape[0]
    base = project_onto_box_sum(base, n0 * 0.5)

    # scales: powers of two up to 1024
    max_n = 1024
    scales = []
    n = n0
    while n <= max_n:
        scales.append(n)
        n = n * 2

    best_h = None
    best_score = np.inf

    for idx, n in enumerate(scales):
        if budget_s - (time.time() - start) < 5.0:
            break

        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n * 0.5)

        # random binary sampling
        sample_limit = 200 if n >= 256 else 300
        for _ in range(sample_limit):
            if budget_s - (time.time() - start) < 1.0:
                break
            h_rand = random_binary_balanced(n, rng)
            sc = compute_c5_fast(h_rand)
            if sc < best_score:
                best_score = sc
                best_h = h_rand.copy()

        # gradient descent restarts
        restarts = 4 if n <= 256 else 3
        for r in range(restarts):
            if budget_s - (time.time() - start) < 5.0:
                break
            h0 = base.copy()
            if r > 0:
                h0 += rng.normal(scale=0.04, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, n * 0.5)

            remaining = budget_s - (time.time() - start)
            max_iters = int(min(50000, max(5000, remaining * 6000)))
            h_opt, sc_opt = gradient_descent_momentum(
                h0, max_iters, step0=1.0, beta=0.9, rng=rng, time_limit=remaining)

            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()
            base = h_opt

        # simulated annealing refinement at this scale
        if best_h is not None and best_h.size == n:
            remaining = budget_s - (time.time() - start)
            if remaining > 5.0:
                h_ref, sc_ref = simulated_annealing(
                    best_h.copy(), best_score, remaining, rng)
                if sc_ref < best_score:
                    best_score = sc_ref
                    best_h = h_ref.copy()
                base = best_h

        # prepare for next scale
        if idx < len(scales) - 1:
            base = best_h if best_h is not None else base

    # final fine‑tuning if time permits
    if best_h is not None:
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            n = best_h.shape[0]
            max_iters = int(min(200000, max(50000, remaining * 8000)))
            best_h, best_score = gradient_descent_momentum(
                best_h, max_iters, step0=0.3, beta=0.9, rng=rng,
                time_limit=remaining)
            best_h = project_onto_box_sum(best_h, best_h.size * 0.5)
            best_score = compute_c5_fast(best_h)
    else:
        n = max_n
        best_h = project_onto_box_sum(thue_morse(n), n * 0.5)
        best_score = compute_c5_fast(best_h)

    return best_h, best_score, best_h.size
