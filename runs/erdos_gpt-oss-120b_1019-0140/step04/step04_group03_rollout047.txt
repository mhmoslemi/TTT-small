def compute_c5(h):
    n = h.shape[0]
    H = np.fft.rfft(h, n*2)
    G = np.fft.rfft(1 - h, n*2)
    corr = np.fft.irfft(H * np.conj(G), n*2)
    # The linear convolution values are from index 0 to 2*n-2
    max_corr = np.max(corr[:2*n-1])
    return max_corr * (2.0 / n)

def gradient_descent_subgrad(h0, max_iters, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    tol = 1e-12  # maybe adjust
    for it in range(1, max_iters+1):
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        # compute correlation full
        # Use FFT to compute corr array
        # Use 2n length
        H = np.fft.rfft(h, n*2)
        G = np.fft.rfft(1 - h, n*2)
        corr = np.fft.irfft(H * np.conj(G), n*2)
        # extract first 2n-1 entries
        corr = corr[:2*n-1]
        max_corr = np.max(corr)
        # Determine shifts where close
        # tolerance relative maybe 1e-9
        # Use absolute tolerance
        shift_idxs = np.where(corr >= max_corr - 1e-9)[0]
        shifts = shift_idxs - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        for s in shifts:
            if s >= 0:
                left = np.empty_like(h)
                left[:n-s] = h[s:]
                left[n-s:] = 0.0
                right = np.empty_like(h)
                right[s:] = h[:n-s]
                right[:s] = 0.0
            else:
                sneg = -s
                left = np.empty_like(h)
                left[sneg:] = h[:n-sneg]
                left[:sneg] = 0.0
                right = np.empty_like(h)
                right[:n-sneg] = h[sneg:]
                right[n-sneg:] = 0.0
            grad += (1.0 - left) - right
        if len(shifts) > 0:
            grad /= len(shifts)
        # update momentum
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # Possibly add small noise every some iterations
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def evolutionary_opt(initial_h, initial_score, n, total_time, rng):
    start = time.time()
    # set population size maybe 30
    pop_size = 30
    # initialize population
    pop = []
    # first entry is initial_h
    pop.append((initial_score, initial_h.copy()))
    # fill rest
    while len(pop) < pop_size:
        h = random_balanced(n, rng)
        sc = compute_c5(h)
        pop.append((sc, h))
    # sort ascending by score
    pop.sort(key=lambda x: x[0])
    best_h = pop[0][1].copy()
    best_score = pop[0][0]
    # parameters for mutation scale decay
    init_scale = 0.1  # start with large modifications
    final_scale = 0.001
    # iteration count
    it = 0
    while True:
        elapsed = time.time() - start
        remaining = total_time - elapsed
        if remaining <= 0.2:  # need small time to exit
            break
        # dynamic scale based on remaining fraction
        frac = remaining / total_time
        scale = final_scale + (init_scale - final_scale) * frac
        # select parent from top half

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=200):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5(h):
    n = h.shape[0]
    H = np.fft.rfft(h, n * 2)
    G = np.fft.rfft(1.0 - h, n * 2)
    corr = np.fft.irfft(H * np.conj(G), n * 2)
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * (2.0 / n)

def gradient_descent_subgrad(h0, max_iters, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time.time() - start >= max_time:
            break
        H = np.fft.rfft(h, n * 2)
        G = np.fft.rfft(1.0 - h, n * 2)
        corr = np.fft.irfft(H * np.conj(G), n * 2)[: 2 * n - 1]
        max_corr = np.max(corr)
        shift_idxs = np.where(corr >= max_corr - 1e-9)[0]
        shifts = shift_idxs - (n - 1)
        grad = np.zeros_like(h)
        for s in shifts:
            if s >= 0:
                left = np.empty_like(h)
                left[: n - s] = h[s:]
                left[n - s :] = 0.0
                right = np.empty_like(h)
                right[s:] = h[: n - s]
                right[:s] = 0.0
            else:
                sneg = -s
                left = np.empty_like(h)
                left[sneg:] = h[: n - sneg]
                left[:sneg] = 0.0
                right = np.empty_like(h)
                right[: n - sneg] = h[sneg:]
                right[n - sneg :] = 0.0
            grad += (1.0 - left) - right
        if len(shifts):
            grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
