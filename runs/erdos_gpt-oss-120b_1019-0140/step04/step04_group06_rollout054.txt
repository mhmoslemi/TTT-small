import time
import numpy as np
from scipy.special import logsumexp

def next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def convolve_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    L = next_pow2(len(a) + len(b) - 1)
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Fa * Fb, n=L)
    return conv[:len(a) + len(b) - 1]

def correlate_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    L = next_pow2(len(a) + len(b) - 1)
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(Fa * np.conj(Fb), n=L)
    return corr[:len(a) + len(b) - 1]

def project_onto_box_sum(v: np.ndarray, s: float, lo: float = 0.0,
                         hi: float = 1.0, max_iter: int = 60) -> np.ndarray:
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n: int, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    C = correlate_full(h, 1.0 - h)
    return np.max(C) * (2.0 / n)

def grad_descent_smooth(h0: np.ndarray, max_iter: int, eta0: float,
                        beta: float, alpha_start: float, alpha_end: float,
                        rng: np.random.Generator, time_left) -> tuple:
    h = project_onto_box_sum(h0, h0.size / 2.0)
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    L_conv = next_pow2(3 * n - 2)

    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break

        # correlation for weighting
        C = correlate_full(h, 1.0 - h)          # length 2n-1
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)
        f = logsumexp(alpha * C) / alpha
        w = np.exp(alpha * (C - f))            # sums to 1

        # FFT transforms (shared length L_conv)
        Fh = np.fft.rfft(h, n=L_conv)
        Fw = np.fft.rfft(w, n=L_conv)
        conv_full = np.fft.irfft(Fw * Fh, n=L_conv)[:3 * n - 2]
        conv1 = conv_full[n - 1:n - 1 + n]      # sum_s w_s * h[t - s]

        w_rev = w[::-1]
        Fw_rev = np.fft.rfft(w_rev, n=L_conv)
        conv_full_rev = np.fft.irfft(Fw_rev * Fh, n=L_conv)[:3 * n - 2]
        conv2 = conv_full_rev[n - 1:n - 1 + n] # sum_s w_s * h[t + s]

        grad = 1.0 - conv1 - conv2               # smooth subgradient

        # momentum and step
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(it)
        h_new = h - eta * momentum

        if (it % 500) == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_onto_box_sum(h_new, target_sum)

        c5_new = compute_c5(h_new)
        if c5_new < best_score:
            best_score = c5_new
            best_h = h_new.copy()

        h = h_new

    return best_h, best_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    best_h, best_score = None, float('inf')
    scales = [128, 256, 512]

    for n in scales:
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        remaining = time_left()
        if remaining < 5.0:
            break
        max_iters = int(min(25000, max(3000, remaining * 15)))  # heuristic

        h_opt, c5_opt = grad_descent_smooth(
            base, max_iters, eta0=0.6, beta=0.9,
            alpha_start=1.0, alpha_end=30.0,
            rng=rng, time_left=time_left
        )
        if c5_opt < best_score:
            best_score, best_h = c5_opt, h_opt.copy()
        base = h_opt

    if best_h is None:
        best_h = project_onto_box_sum(random_binary_balanced(128, rng), 64.0)
        best_score = compute_c5(best_h)

    n_final = best_h.size
    while time_left() > 8.0:
        # random restart
        h_rand = random_binary_balanced(n_final, rng)
        h_rand = project_onto_box_sum(h_rand, n_final / 2.0)
        max_iters = int(min(8000, max(1500, time_left() * 10)))
        h_rand_opt, c5_rand_opt = grad_descent_smooth(
            h_rand, max_iters, eta0=0.5, beta=0.9,
            alpha_start=1.0, alpha_end=30.0,
            rng=rng, time_left=time_left
        )
        if c5_rand_opt < best_score:
            best_score, best_h = c5_rand_opt, h_rand_opt.copy()

        # tiny deterministic tweak
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.1, 0.3)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_onto_box_sum(h_new, n_final / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < best_score:
                best_score, best_h = c5_new, h_new.copy()

        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
