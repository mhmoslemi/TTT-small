def gradient_descent_proj(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random noise to escape shallow minima
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    n = h.size
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)

        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.size
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_proj(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_local_search(h0, time_left, rng):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5_fft(cur_h)
    ones_idx = np.flatnonzero(cur_h > 0.5)
    zeros_idx = np.flatnonzero(cur_h < 0.5)
    trials_per_iter = 80
    while time_left() > 0.5:
        best_trial_h = None
        best_trial_score = cur_score
        for _ in range(trials_per_iter):
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            cand = cur_h.copy()
            cand[i] = 0.0
            cand[j] = 1.0
            sc = compute_c5_fft(cand)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial_h = cand
        if best_trial_h is not None:
            cur_h = best_trial_h
            cur_score = best_trial_score
            ones_idx = np.flatnonzero(cur_h > 0.5)
            zeros_idx = np.flatnonzero(cur_h < 0.5)
        else:
            break
    return cur_h, cur_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    rng = np.random.default_rng(seed)

    # Initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    sizes = [256, 512]
    best_h = None
    best_score = np.inf
    best_n = None

    for n in sizes:
        if time_left() < 1.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        #--- Gradient descent restarts ---
        for _ in range(5):
            if time_left() < 0.5:
                break
            h0 = base + rng.normal(scale=0.05, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)
            h_opt, sc_opt = gradient_descent_proj(
                h0, max_iter=200000, step0=0.5, beta=0.9,
                rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_score, best_h, best_n = sc_opt, h_opt.copy(), n
                if best_score <= 0.3808:
                    break
        if best_score <= 0.3808:
            break

        #--- Binary rounding of best continuous ---
        if best_h is not None and best_h.size == n:
            idx = np.argsort(-best_h)
            h_bin = np.zeros(n, dtype=float)
            h_bin[idx[:n // 2]] = 1.0
        else:
            h_bin = random_binary_balanced(n, rng)

        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h, best_n = sc_bin, h_bin.copy(), n

        #--- Random binary search ---
        for _ in range(2000):
            if time_left() < 0.2:
                break
            hb = random_binary_balanced(n, rng)
            sc = compute_c5_fft(hb)
            if sc < best_score:
                best_score, best_h, best_n = sc, hb.copy(), n

        #--- Binary local search ---
        if best_h is not None and best_h.size == n:
            h_ls, sc_ls = binary_local_search(best_h.copy(), time_left, rng)
            if sc_ls < best_score:
                best_score, best_h, best_n = sc_ls, h_ls.copy(), n

        # Prepare for next resolution
        base = best_h.copy() if best_h is not None else base

    # Final projection and evaluation
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_n / 2.0)
        best_score = compute_c5_fft(best_h)
    else:
        best_n = 256
        best_h = project_onto_box_sum(thue_morse(best_n), best_n / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, best_n
