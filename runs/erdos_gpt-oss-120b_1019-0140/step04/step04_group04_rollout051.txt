def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure brackets
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):

    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climb_binary(initial_binary, deadline, rng):
    cur = initial_binary.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_score = compute_c5(cur)
        if new_score <= cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    best_h = None
    best_score = np.inf

    scales = [64, 128, 256, 512]
    for n in scales:
        if time_left() < 6:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        base_score = compute_c5(base)
        if base_score < best_score:
            best_h, best_score = base.copy(), base_score

        # binary local search
        binary_budget = min(30.0, time_left() * 0.4)
        if binary_budget > 5:
            deadline = time.time() + binary_budget
            bin0 = binary_from_continuous(base)
            h_bin, sc_bin = hill_climb_binary(bin0, deadline, rng)
            if sc_bin < best_score:
                best_h, best_score = h_bin.astype(float), sc_bin
            # additional random restarts
            while time.time() + 3.0 < deadline:
                rand_bin = random_binary_balanced(n, rng)
                sc_rand = compute_c5(rand_bin)
                if sc_rand < best_score:
                    best_h, best_score = rand_bin.astype(float), sc_rand
                sub_dead = time.time() + 3.0
                h_bin2, sc_bin2 = hill_climb_binary(rand_bin, sub_dead, rng)
                if sc_bin2 < best_score:
                    best_h, best_score = h_bin2.astype(float), sc_bin2

        # gradient refinement
        if time_left() > 6:
            n_iter = int(min(50000, time_left() * 1500))
            if n_iter > 0:
                step0 = 0.4 if n <= 256 else 0.2
                best_h, best_score = gradient_descent_momentum_custom(
                    best_h, n_iter=n_iter, step0=step0, beta=0.95,
                    rng=rng, time_left=time_left)

        base = best_h.copy()

    # optional upsample to 1024 if time permits
    if time_left() > 30 and best_h.size < 1024:
        n = 1024
        base = upsample(best_h, n)
        base = project_onto_box_sum(base, n / 2.0)
        base_score = compute_c5(base)
        if base_score < best_score:
            best_h, best_score = base.copy(), base_score
        binary_budget = min(30.0, time_left() * 0.4)
        if binary_budget > 5:
            deadline = time.time() + binary_budget
            bin0 = binary_from_continuous(best_h)
            h_bin, sc_bin = hill_climb_binary(bin0, deadline, rng)
            if sc_bin < best_score:
                best_h, best_score = h_bin.astype(float), sc_bin
        if time_left() > 6:
            n_iter = int(min(50000, time_left() * 1500))
            if n_iter > 0:
                best_h, best_score = gradient_descent_momentum_custom(
                    best_h, n_iter=n_iter, step0=0.15, beta=0.95,
                    rng=rng, time_left=time_left)

    # final fine‑tuning
    if time_left() > 6:
        n_iter = int(min(80000, time_left() * 2000))
        if n_iter > 0:
            best_h, best_score = gradient_descent_momentum_custom(
                best_h, n_iter=n_iter, step0=0.1, beta=0.95,
                rng=rng, time_left=time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
