import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure the interval contains solution
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time_left() > 0.1:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def greedy_improve(h, max_iters, rng, time_left):
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    for _ in range(max_iters):
        if time_left() <= 0:
            break
        A = np.correlate(cur_h, cur_h, mode='full')
        A = A[n-1:]  # shifts 0..n-1
        s_min = int(A.argmin())
        if s_min == 0:
            break
        # shift contributions
        h_plus = np.concatenate([cur_h[s_min:], np.zeros(s_min, dtype=cur_h.dtype)])
        h_minus = np.concatenate([np.zeros(s_min, dtype=cur_h.dtype), cur_h[:n-s_min]])
        v = h_plus + h_minus
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        v_ones = v[ones]
        v_zeros = v[zeros]
        i = ones[int(v_ones.argmin())]
        j = zeros[int(v_zeros.argmax())]
        delta = v[j] - v[i]
        if delta <= 0:
            break
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        cur_score = compute_c5(cur_h)
        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h.copy()
    return best_h, best_score

def simulated_annealing(h, max_iters, rng, time_left):
    n = h.shape[0]
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start_temp = 0.01
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        T = start_temp * (0.995 ** it)
        # select random swap
        idx1 = rng.choice(np.where(cur_h > 0.5)[0])
        idx2 = rng.choice(np.where(cur_h < 0.5)[0])
        cur_h[idx1] = 0.0
        cur_h[idx2] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[idx1] = 1.0
            cur_h[idx2] = 0.0
        if it % 5000 == 0 and rng.random() < 0.2:
            # occasional random jitter
            a = rng.choice(n, size=2, replace=False)
            cur_h[a[0]] = 1.0 - cur_h[a[0]]
            cur_h[a[1]] = 1.0 - cur_h[a[1]]
            cur_score = compute_c5(cur_h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
    return best_h, best_score

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time_left() > 0.1:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def greedy_improve(h, max_iters, rng, time_left):
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    for _ in range(max_iters):
        if time_left() <= 0:
            break
        A = np.correlate(cur_h, cur_h, mode='full')[n - 1:]
        s_min = int(A.argmin())
        if s_min == 0:
            break
        h_plus = np.concatenate([cur_h[s_min:], np.zeros(s_min, dtype=cur_h.dtype)])
        h_minus = np.concatenate([np.zeros(s_min, dtype=cur_h.dtype), cur_h[:n - s_min]])
        v = h_plus + h_minus
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[int(v[ones].argmin())]
        j = zeros[int(v[zeros].argmax())]
        delta = v[j] - v[i]
        if delta <= 0:
            break
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        cur_score = compute_c5(cur_h)
        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h.copy()
    return best_h, best_score

def simulated_annealing(h, max_iters, rng, time_left):
    n = h.shape[0]
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start_temp = 0.01
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        T = start_temp * (0.995 ** it)
        idx1 = rng.choice(np.where(cur_h > 0.5)[0])
        idx2 = rng.choice(np.where(cur_h < 0.5)[0])
        cur_h[idx1] = 0.0
        cur_h[idx2] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[idx1] = 1.0
            cur_h[idx2] = 0.0
        if it % 5000 == 0 and rng.random() < 0.2:
            a = rng.choice(n, size=2, replace=False)
            cur_h[a[0]], cur_h[a[1]] = cur_h[a[1]], cur_h[a[0]]
            cur_score = compute_c5(cur_h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
    return best_h, best_score
