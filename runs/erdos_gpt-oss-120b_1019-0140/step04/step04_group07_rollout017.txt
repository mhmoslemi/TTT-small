def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # Get worst shifts
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term w.r.t. h (averaged over worst shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[: n - s] += 1.0 - h[s:]
            grad[s:] += 1.0 - h[: n - s]
        else:
            t = -s
            grad[t:] += 1.0 - h[: n - t]
            grad[: n - t] += 1.0 - h[t:]
    if len(shifts):
        grad /= len(shifts)
    return grad


def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
    return best_h, best_score


def round_to_binary(v):
    """Return binary vector with exactly n/2 ones by taking largest entries."""
    n = v.shape[0]
    s = n // 2
    idx = np.argpartition(v, -s)[-s:]
    b = np.zeros_like(v, dtype=float)
    b[idx] = 1.0
    return b


def binary_sa(h_bin, rng, time_budget):
    """Simulated‑annealing style swap optimisation on binary vectors."""
    start = time.time()
    def tl():
        return max(0.0, time_budget - (time.time() - start))

    cur_score = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_score = cur_score
    n = h_bin.size
    s = n // 2

    initial_temp = 0.01
    final_temp = 1e-6

    while tl() > 0:
        # pick a random 1 and a random 0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h_new = h_bin.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - cur_score

        elapsed = time.time() - start
        frac = elapsed / time_budget
        temp = initial_temp * (final_temp / initial_temp) ** frac
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            h_bin = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_bin.copy()
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # base vector
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]
    prev = base

    for n in scales:
        if prev.shape[0] != n:
            prev = upsample(prev, n)
        prev = project_onto_box_sum(prev, n / 2.0)

        if time_left() > 2:
            max_iters = int(min(20000, max(2000, time_left() * (8000.0 / n))))
            h_gd, sc_gd = gradient_descent(prev, max_iters, step0=0.6,
                                           beta=0.9, rng=rng, time_left=time_left)
            if sc_gd < best_score:
                best_h, best_score = h_gd.copy(), sc_gd
            prev = h_gd

        if time_left() > 2:
            jitter = rng.normal(scale=0.02, size=n)
            prev = project_onto_box_sum(prev + jitter, n / 2.0)
            cur = compute_c5(prev)
            if cur < best_score:
                best_h, best_score = prev.copy(), cur

    # Ensure we have a 512‑point vector for binary refinement
    if best_h.shape[0] != 512:
        best_h = upsample(best_h, 512)
        best_h = project_onto_box_sum(best_h, 512 / 2.0)

    binary = round_to_binary(best_h)

    # Binary simulated annealing on 512 points
    if time_left() > 5:
        sa_time = time_left() * 0.4
        binary, bin_score = binary_sa(binary, rng, sa_time)
        if bin_score < best_score:
            best_h, best_score = binary.copy(), bin_score

    # Upsample to 1024 and repeat binary annealing if time permits
    if time_left() > 10:
        up = upsample(best_h, 1024)
        up = project_onto_box_sum(up, 1024 / 2.0)
        binary_1024 = round_to_binary(up)
        sa_time2 = time_left() * 0.5
        binary_1024, bin_score2 = binary_sa(binary_1024, rng, sa_time2)
        if bin_score2 < best_score:
            best_h, best_score = binary_1024.copy(), bin_score2

    best_h = project_onto_box_sum(best_h, best_h.shape[0] / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.shape[0]
