def compute_gradient(h, active_shifts):
    n = h.size
    grad = np.zeros_like(h)
    for k in active_shifts:
        if k >= 0:
            # first term: for i in [0, n - k - 1], add (1 - h[i+k])
            grad[:n - k] += 1.0 - h[k:]
            # second term: for i in [k, n-1], subtract h[i - k]
            grad[k:] -= h[:n - k]
        else:
            s = -k
            # first term: for i in [s, n-1], add (1 - h[i - s])
            grad[s:] += 1.0 - h[:n - s]
            # second term: for i in [0, n-s-1], subtract h[i + s]
            grad[:n - s] -= h[s:]
    if len(active_shifts) > 0:
        grad /= len(active_shifts)
    return grad

def gradient_descent(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        # compute active shifts and gradient
        active_shifts = compute_active_shifts(h)
        grad = compute_gradient(h, active_shifts)
        # momentum update
        momentum = beta * momentum + grad
        # step size schedule: step = step0 / sqrt(it) maybe.
        step = step0 / np.sqrt(it)
        # backtracking line search (max 5 tries)
        for _ in range(5):
            h_new = h - step * momentum
            h_new = project_onto_box_sum(h_new, target_sum)
            new_score = compute_c5(h_new)
            if new_score <= best_score or new_score < compute_c5(h):
                # accept
                h = h_new
                if new_score < best_score:
                    best_h = h_new.copy()

        if it % 500 == 0:

            pert_score = compute_c5(h_pert)
            if pert_score < best_score:
                h = h_pert
                best_h = h_pert.copy()
                best_score = pert_score
    return best_h, best_score

def slsqp_refine(h0, time_left, max_iter):
    n = h0.size
    target_sum = n/2.0
    # Solves min t subject to constraints as before.
    # Use SciPy's minimize with SLSQP.
    def objective(x):
        return x[-1]
    # initial guess: h0 and current C5
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, np.inf)]
    # equality sum
    def eq_sum(x):
        return np.sum(x[:n]) - target_sum
    # inequality: for each shift k, ensure t >= correlation
    # Since constraints potentially many (2n-1) we must define function that returns array of constraints (>=0)
    # Actually for SLSQP, we define ineq constraints as lambda x: x[-1] - corr_k
    # We can define a function that returns vector of constraints for all shifts.
    # For efficiency we may only pass a few shifts (active ones), but we need ensure max holds for all.
    # Could use lambda x: x[-1] - np.correlate(x[:n], 1 - x[:n], mode='full') * (2.0 / n)
    # But that returns array; we can use 'type':'ineq', 'fun': lambda x: something.
    # SciPy SLSQP supports vector-valued constraints.
    def ineq_fun(x):
        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return x[-1] - corr
    cons = [{'type':'eq','fun':eq_sum},
            {'type':'ineq','fun':ineq_fun}]
    # Set options: maxiter = max_iter, ftol small.
    res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter,
                                                  'ftol':1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, target_sum)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Project vector v onto [lo,hi] box with exact sum s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda such that sum(clip(v - lam, lo, hi)) = s
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # widen interval if needed
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def slsqp_refine(h0, time_left, max_iter):
    """SLSQP refinement of (h, t) minimizing t subject to constraints."""
    n = h0.size
    target_sum = n / 2.0
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, np.inf)]

    def eq_sum(x):
        return np.sum(x[:n]) - target_sum

    def ineq_fun(x):
        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return x[-1] - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    # choose a safe iteration cap
    iters = int(min(max_iter, max(50, time_left() * 5)))
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': iters, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, target_sum)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:

    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
