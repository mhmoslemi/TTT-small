def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    best_h = None
    best_score = np.inf
    # If initial present
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        score_init = compute_c5_fft(h_init)
        best_h, best_score = h_init, score_init
    # define base for scaling
    base = best_h if best_h is not None else thue_morse(128)
    scales = [128, 256, 512]
    TARGET = 0.3808
    for n in scales:
        if time_left() <= 1.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n/2.0)
        # restarts
        restarts = 6
        for r in range(restarts):
            if time_left() <= 0.5:
                break
            h0 = base + rng.normal(scale=0.05, size=n)
            h0 = project_onto_box_sum(h0, n/2.0)
            h_opt, sc_opt = gradient_descent(h0, n_iter=150000, step0=0.6, beta=0.9, rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_score <= TARGET:
            break
        # binary hill climb
        if best_h is not None and time_left() > 2.0:
            nbest = best_h.size
            for _ in range(2000):
                i, j = rng.choice(nbest, size=2, replace=False)
                if best_h[i] == best_h[j]:
                    continue
                # swap values
                h_new = best_h.copy()
                h_new[i], h_new[j] = best_h[j], best_h[i]
                sc_new = compute_c5_fft(h_new)
                if sc_new < best_score:
                    best_score, best_h = sc_new, h_new.copy()
                    if best_score <= TARGET:
                        break
        # prepare base for next scale
        base = best_h if best_h is not None else base
    # final refinement
    if best_h is not None and time_left() > 5.0:
        h_ref, sc_ref = gradient_descent(best_h, n_iter=300000, step0=0.3, beta=0.9, rng=rng, time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
    # final check against initial
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        score_init = compute_c5_fft(h_init)
        if score_init < best_score:
            best_score, best_h = score_init, h_init.copy()
    # final projection & compute
    if best_h is None:
        n_ret = 256
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret/2.0)
        best_score = compute_c5_fft(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size/2.0)
        best_score = compute_c5_fft(best_h)
    return best_h, float(best_score), best_h.size

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Sub‑gradient of the max‑correlation term (averaged over max shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
