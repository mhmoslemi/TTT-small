def compute_c5_and_shifts_fft(h):
    n = h.shape[0]
    N = 2*n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H*G, N)
    # corr shape N; indices 0..N-1 correspond to lags -(n-1) .. +(n-1)
    # mapping: lag = i - (n-1)
    max_val = np.max(corr)
    max_corr = max_val * (2.0 / n)
    # find all shift indices:
    eps = 1e-12 * max_val  # tolerance maybe relative
    mask = np.abs(corr - max_val) <= eps
    shift_idxs = np.where(mask)[0] - (n - 1)
    return max_corr, shift_idxs

def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = len(h)
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        C, shifts = compute_c5_and_shifts_fft(h)
        # compute gradient
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[:n - shift] = 1.0 - h[shift:]
                term2[shift:] = h[:n - shift]
                # gradient += (1 - h_{i+shift}) - h_{i-shift}
                # Actually term1 is (1 - h_{i+shift}), term2 is h_{i-shift}
                grad += (term1 - term2)
            else:
                sft = -shift
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[sft:] = 1.0 - h[:n - sft]
                term2[:n - sft] = h[sft:]
                grad += (term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # optional jitter occasionally to escape
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def upsample(arr, new_n):
    n = arr.shape[0]
    if n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        C, shifts = compute_c5_and_shifts_fft(h)
        # compute gradient
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                sft = -shift
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[sft:] = h[:n-sft]
                term2[:n-sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise injection
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def slsqp_refine(h0, max_iter):
    n = h0.size
    def compute_corr(h):
        # returns vector of correlation values (full) scaled
        N = 2*n
        H = np.fft.fft(h, N)
        G = np.fft.fft(1.0 - h, N)
        corr = np.fft.ifft(H * G).real[:2*n-1] * (2.0/n)
        return corr
    def obj(x):
        # x includes h and scalar t
        h = x[:n]
        t = x[-1]
        return t

    # We'll define inequality: corr_i - t <= 0 => t - corr_i >= 0.

        return t - corr

def slsqp_refine(h0, start, budget_s, max_iter=300):
    n = h0.size
    def obj_fun(x):
        return x[-1]
    def eq_fun(x, n):
        return np.sum(x[:n]) - n/2.0
    def ineq_fun(x, n):
        h = x[:n]
        t = x[-1]
        N = 2*n
        H = np.fft.fft(h, N)
        G = np.fft.fft(1.0 - h, N)
        corr = np.fft.ifft(H * G).real[:2*n-1] * (2.0/n)
        return t - corr
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type':'eq', 'fun': eq_fun, 'args': (n,)},
            {'type':'ineq', 'fun': ineq_fun, 'args': (n,)}]
    # ensure time left
    elapsed = time.time() - start
    if budget_s - elapsed < 1:
        return h0, compute_c5(h0)
    res = opt.minimize(obj_fun, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter,
                                                  'ftol': 1e-12,
                                                  'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    n = arr.shape[0]
    if n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5_and_shifts_fft(h):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(H * G).real[:2 * n - 1]
    max_val = np.max(corr)
    max_corr = max_val * (2.0 / n)
    eps = 1e-12 * max_val
    shift_idxs = np.where(np.abs(corr - max_val) <= eps)[0] - (n - 1)
    return max_corr, shift_idxs

def compute_c5(h):
    return compute_c5_and_shifts_fft(h)[0]

def gradient_descent(h0, max_iters, step0, beta, rng, start, budget_s):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time.time() - start >= budget_s:
            break
        _, shifts = compute_c5_and_shifts_fft(h)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[sft:] = h[:n - sft]
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def evolve_population(n, pop_size, start, budget_s, rng):
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.size != n:
            base = upsample(base, n)
    else:
        base = random_binary_balanced(n, rng)
    base = project_onto_box_sum(base, n / 2.0)
    pop = [(base, compute_c5(base))]
    while len(pop) < pop_size:
        if rng.random() < 0.5:
            vec = base + rng.normal(scale=0.1, size=n)
        else:
            vec = random_binary_balanced(n, rng).astype(float)
        vec = project_onto_box_sum(vec, n / 2.0)
        pop.append((vec, compute_c5(vec)))
    pop.sort(key=lambda x: x[1])
    elite_frac = 0.2
    elite_size = max(1, int(pop_size * elite_frac))
    generation = 0
    while True:
        elapsed = time.time() - start
        if budget_s - elapsed <= 5:
            break
        generation += 1
        mut_scale = 0.1 * ((budget_s - elapsed) / budget_s) ** 0.5
        offspring = []
        for _ in range(pop_size):
            i1 = rng.integers(elite_size)
            i2 = rng.integers(elite_size)
            p1 = pop[i1][0]
            p2 = pop[i2][0]
            alpha = rng.random()
            child = alpha * p1 + (1.0 - alpha) * p2
            child += rng.normal(scale=mut_scale, size=n)
            child = project_onto_box_sum(child, n / 2.0)
            offspring.append((child, compute_c5(child)))
        combined = pop + offspring
        combined.sort(key=lambda x: x[1])
        pop = combined[:pop_size]
        if generation % 15 == 0:
            for _ in range(max(1, pop_size // 10)):
                rnd = random_binary_balanced(n, rng)
                rnd = project_onto_box_sum(rnd, n / 2.0)
                sc = compute_c5(rnd)
                if sc < pop[-1][1]:
                    pop[-1] = (rnd, sc)
                    pop.sort(key=lambda x: x[1])
    return pop[0][0], pop[0][1]

def slsqp_refine(h0, start, budget_s, max_iter=500):

    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'eq', 'fun': eq_sum_fun, 'args': (n,)},
            {'type': 'ineq', 'fun': ineq_fun, 'args': (n,)}]
    if time.time() - start >= budget_s:
        return h0, compute_c5(h0)
    res = opt.minimize(objective_fun, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)
