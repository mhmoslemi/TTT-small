import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,2)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n as floats."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_corr_full_fft(h):
    """Linear cross‑correlation of h with (1‑h) via zero‑padded FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    """Maximum overlap integral C5 for discretized h."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Sub‑gradient of max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2500 == 0:  # occasional perturbation
            h_new += rng.normal(scale=0.015, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climb(initial_h, init_score, rng, time_left, budget_s):
    """Pure hill‑climbing with random swaps and mass transfers."""
    best_h = initial_h.copy()
    best_score = init_score
    n = best_h.size
    it = 0
    while time_left() > 0.02:
        it += 1
        # occasional balanced binary jump
        if it % 4000 == 0:
            h_bin = np.zeros(n, dtype=float)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_bin[idx] = 1.0
            sc_bin = compute_c5_fft(h_bin)
            if sc_bin < best_score:
                best_h, best_score = h_bin, sc_bin
            continue

        if rng.random() < 0.55:  # mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            delta_max = min(best_h[i], 1.0 - best_h[j])
            if delta_max <= 1e-12:
                continue
            # bias toward smaller moves using square of uniform
            delta = delta_max * (rng.random() ** 2)
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:  # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            if best_h[i] == best_h[j]:
                continue
            h_new = best_h.copy()
            h_new[i], h_new[j] = best_h[j], best_h[i]

        sc_new = compute_c5_fft(h_new)
        if sc_new < best_score:
            best_h, best_score = h_new, sc_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    # Base construction
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    # Choose target resolution (256 is a good trade‑off)
    target_n = 256
    if base.shape[0] != target_n:
        base = upsample(base, target_n)
    base = project_onto_box_sum(base, target_n / 2.0)

    # Gradient descent phase
    gd_iters = 200000
    if time_left() > 5.0:
        h_gd, sc_gd = gradient_descent(
            base, n_iter=gd_iters, step0=0.6, beta=0.9,
            rng=rng, time_left=time_left)
    else:
        h_gd, sc_gd = base.copy(), compute_c5_fft(base)

    # Hill‑climbing refinement
    if time_left() > 2.0:
        h_opt, sc_opt = hill_climb(h_gd, sc_gd, rng, time_left, budget_s)
    else:
        h_opt, sc_opt = h_gd, sc_gd

    # Final safety projection
    h_opt = project_onto_box_sum(h_opt, h_opt.size / 2.0)
    final_score = compute_c5_fft(h_opt)
    n_ret = h_opt.size
    return h_opt, final_score, n_ret
