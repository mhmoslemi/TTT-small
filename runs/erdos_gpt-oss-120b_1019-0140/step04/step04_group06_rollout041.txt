def compute_c5_fft(h):
    n = h.shape[0]
    # compute correlation via FFT (linear) => need zero pad to length 2n.
    # Use np.fft.fft of length 2*n.
    h_fft = np.fft.fft(h, n*2)
    one_minus_h = 1.0 - h
    o_fft = np.fft.fft(one_minus_h, n*2)
    corr = np.fft.ifft(h_fft * np.conj(o_fft)).real
    # correlation with lag [-n+1..n-1] corresponds to corr[:2*n-1]
    corr = corr[:2*n-1]
    # due to rounding errors, maybe negative small values -> clip
    return np.max(corr) * (2.0 / n)

But note that the product inside ifft gives circular correlation, not linear. By zero padding to length 2*n, the circular correlation becomes linear for indices <2n-1. So that's fine.

def compute_corr_fft(h):
    n = h.shape[0]
    # Use double length for linear correlation
    N = 2 * n
    hf = np.fft.fft(h, N)
    of = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(hf * np.conj(of)).real[:2*n-1]
    # rounding errors: set small negative to zero
    # Not necessary for max.
    return corr

def get_bad_indices(h, s):
    n = h.shape[0]
    if s >= 0:
        # i from 0 to n-s-1
        if s >= n:  # shift larger than length, no overlap
            return np.empty(0, dtype=int)
        mask = (h[:n - s] == 1) & (h[s:] == 0)
        return np.nonzero(mask)[0]
    else:
        t = -s
        if t >= n:
            return np.empty(0, dtype=int)
        mask = (h[t:] == 1) & (h[:n - t] == 0)
        return np.nonzero(mask)[0] + t

def compute_corr_fft(h):
    n = h.shape[0]
    N = 2 * n
    hf = np.fft.fft(h, N)
    of = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(hf * np.conj(of)).real[:2 * n - 1]

def compute_c5_fft(h):

def get_bad_indices(h, s):
    n = h.shape[0]
    if s >= 0:
        if s >= n:
            return np.empty(0, dtype=int)
        mask = (h[:n - s] == 1) & (h[s:] == 0)

def sa_optimize(h0, rng, end_time, target):
    h = h0.copy()
    n = h.shape[0]
    cur_corr = compute_corr_fft(h)
    cur_score = cur_corr.max() * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    total_alloc = max(0.0, end_time - time.time())
    if total_alloc <= 0:
        return best_h, best_score
    T0 = 0.05
    T1 = 1e-5
    start_local = time.time()
    while True:
        remaining = end_time - time.time()
        if remaining <= 0:
            break
        if best_score <= target:
            break
        max_idx = int(np.argmax(cur_corr))
        s = max_idx - (n - 1)
        bad_i = get_bad_indices(h, s)
        if bad_i.size == 0:
            # random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        else:
            i = int(rng.choice(bad_i))
            j = i + s
        # Perform swap
        h[i], h[j] = 0.0, 1.0
        new_corr = compute_corr_fft(h)
        new_score = new_corr.max() * (2.0 / n)
        delta = new_score - cur_score
        elapsed = time.time() - start_local
        frac = elapsed / total_alloc if total_alloc > 0 else 1.0
        # temperature schedule
        T = T0 * (T1 / T0) ** frac
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            cur_corr = new_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

    return best_h, best_score, best_h.size
