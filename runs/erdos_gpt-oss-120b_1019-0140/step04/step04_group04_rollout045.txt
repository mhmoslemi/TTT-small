def adam_optimize(h0, n_iter, step0, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # compute shifts with max correlation
        shifts = get_max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        # adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - (step0 / (np.sqrt(v_hat) + eps)) * m_hat
        # optional noise every some iterations
        if t % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=h.size)
        h_new = project_onto_box_sum(h_new, h.size/2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

    else:

    best_h = None
    best_score = np.inf

    # resolution schedule
    resolutions = [64, 128, 256, 512, 1024]
    for n in resolutions:
        if time_left() < 5:
            break
        # upsample to target resolution if needed
        if base.shape[0] != n:
            base = upsample(base, n)
        # project onto constraints
        base = project_onto_box_sum(base, n/2.0)
        # evaluate
        score0 = compute_c5(base)
        if score0 < best_score:
            best_score = score0
            best_h = base.copy()

        # for larger n, reduce iterations
        max_iters = int(min(30000, time_left() * (5000.0 / n)))  # scale inversely

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_lo + lam_hi) * 0.5
        proj = np.clip(v - lam, lo, hi)
        cur = proj.sum()
        if cur > s:
            lam_lo = lam
        else:
            lam_hi = lam
        if abs(cur - s) < 1e-12:
            break
    return np.clip(v - lam, lo, hi)

def compute_c5(h):
    n = h.size
    a = np.fft.rfft(h, 2 * n)
    b = np.fft.rfft((1.0 - h)[::-1], 2 * n)
    corr = np.fft.irfft(a * b, 2 * n)
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def get_max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    tol = 1e-12 * max(1.0, mx)
    idx = np.where(np.abs(corr - mx) <= tol)[0]
    return idx - (n - 1)

def adam_optimize(h0, n_iter, step0, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_sc = compute_c5(h)
    for t in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        shifts = get_max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - (step0 / (np.sqrt(v_hat) + eps)) * m_hat
        if t % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=h.size)
        h_new = project_onto_box_sum(h_new, h.size / 2.0)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_sc

def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    def obj(x):
        return x[-1]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12,
                                'disp': False})
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    best_h = None
    best_sc = np.inf

    # hierarchical schedule
    resolutions = [64, 128, 256, 512, 1024]
    target_iters = {64: 20000, 128: 12000, 256: 6000, 512: 3000, 1024: 1500}
    for n in resolutions:
        if time_left() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        sc0 = compute_c5(base)
        if sc0 < best_sc:
            best_sc, best_h = sc0, base.copy()

        # local Adam refinement
        max_it = min(target_iters[n], int(time_left() * 5))
        if max_it > 0:
            step0 = 0.5 if n <= 128 else 0.3 if n <= 256 else 0.2
            h_ref, sc_ref = adam_optimize(base, n_iter=max_it,
                                          step0=step0, beta1=0.9, beta2=0.999,
                                          eps=1e-8, rng=rng, time_left=time_left)
            if sc_ref < best_sc:
                best_sc, best_h = sc_ref, h_ref.copy()
            base = h_ref

        # occasional SLSQP for medium sizes
        if n <= 256 and time_left() > 5:
            h_sls, sc_sls = slsqp_optimize_candidate(base, max_iter=200)
            if sc_sls < best_sc:
                best_sc, best_h = sc_sls, h_sls.copy()
            base = h_sls
