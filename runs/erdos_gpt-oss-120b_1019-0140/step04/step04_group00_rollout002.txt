def compute_c5_and_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2*n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n-1]
    max_val = corr.max()
    shift = corr.argmax() - (n - 1)  # index shift
    return max_val * (2.0 / n), shift

import time
import numpy as np

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=50):
    """Project v onto the box [lo,hi] while forcing sum(v)=target."""
    n = v.size
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    # find lambda such that sum(clip(v - lam, lo, hi)) = target
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # expand bounds if needed
    while f(lam_low) < target:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > target:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target) < 1e-12:
            break
        if fm > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def gradient_of_shift(h, shift):
    """Subgradient of the overlap for a given integer shift."""
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros_like(h)
        term1[:n_s] = h[shift:]
        term2 = np.zeros_like(h)
        term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros_like(h)
        term1[s:] = h[:n_s]
        term2 = np.zeros_like(h)
        term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_c5_and_shift(h):
    """Return (C5 value, shift achieving the maximum) for discrete h."""
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    max_idx = corr.argmax()
    max_val = corr[max_idx]
    shift = max_idx - (n - 1)
    return max_val * (2.0 / n), shift

        best_h = adam_optimize(best_h, 200000, lr0=0.2)

    # final projection and evaluation
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=np.float64)
    best_h = project_onto_box_sum(best_h, target_sum)
    best_val, _ = compute_c5_and_shift(best_h)

    return best_h, best_val, n
