def initialize_h(n, rng):

            h = h[:n]

        if cur > target:
            idx_one = np.where(h == 1)[0]
            to_flip = rng.choice(idx_one, size=cur - target, replace=False)
            h[to_flip] = 0
        elif cur < target:
            idx_zero = np.where(h == 0)[0]
            to_flip = rng.choice(idx_zero, size=target - cur, replace=False)
            h[to_flip] = 1
        return h
    else:
        h = np.zeros(n, dtype=np.uint8)
        ones_idx = rng.choice(n, size=n//2, replace=False)
        h[ones_idx] = 1
        return h

def compute_overlap(h):
    # compute self-correlation
    corr_full = np.correlate(h, h, mode='full')
    n = h.size
    ov = np.empty(n, dtype=np.int16)
    ov[0] = corr_full[n-1]
    ov[1:] = corr_full[n:]
    return ov

def adam_projected(h0, max_iters, lr0, rng, max_time):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    for t in range(1, max_iters+1):
        if time.time() - start >= max_time:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # get shifts achieving max_corr (within tolerance)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # occasional random perturbation
        if t % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # early stop if score <= target
        if best_score <= 0.3808:

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.zeros_like(h)
            term1[:sz] = h[shift:]
            term2 = np.zeros_like(h)
            term2[shift:] = h[:sz]
        else:
            s = -shift
            sz = n - s
            term1 = np.zeros_like(h)
            term1[s:] = h[:sz]
            term2 = np.zeros_like(h)
            term2[:sz] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def adam_projected(h0, max_iters, lr0, rng, max_time):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    for t in range(1, max_iters + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        if t % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        if best_score <= 0.3808:
            break
    return best_h, best_score

def thue_morse_seq(N):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(N)),
                       dtype=np.uint8, count=N)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    # initial construction
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init_bin = (init > 0.5).astype(np.uint8)
        repeat = n // init_bin.size
        h0 = np.repeat(init_bin, repeat)
        if h0.size != n:
            h0 = np.pad(h0, (0, n - h0.size), constant_values=0)
    else:
        h0 = thue_morse_seq(n)
    # enforce sum constraint
    target_sum = n / 2.0
    cur_sum = float(h0.sum())
    if cur_sum > target_sum:
        idx_one = np.where(h0 == 1)[0]
        to_flip = rng.choice(idx_one, size=int(cur_sum - target_sum), replace=False)
        h0[to_flip] = 0
    elif cur_sum < target_sum:
        idx_zero = np.where(h0 == 0)[0]
        to_flip = rng.choice(idx_zero, size=int(target_sum - cur_sum), replace=False)
        h0[to_flip] = 1
    h0 = h0.astype(float)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # main optimization loop with restarts
    while True:
        now = time.time()
        remaining = budget_s - (now - start)
        if remaining <= 2.0:
            break
        max_iters = int(remaining * 800)
        if max_iters < 1000:
            max_iters = int(remaining * 2000)
        # perturb current best and start a new Adam run
        h_start = best_h + rng.normal(scale=0.02, size=n)
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = project_onto_box_sum(h_start, target_sum)
        h_opt, sc_opt = adam_projected(h_start, max_iters, lr0=0.5,
                                       rng=rng, max_time=remaining - 0.5)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)
    return best_h.astype(float), float(best_score), n
