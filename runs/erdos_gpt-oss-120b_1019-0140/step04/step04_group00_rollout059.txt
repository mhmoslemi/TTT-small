def compute_autocorr_f(h):
    f = h - 0.5
    n = len(h)
    N = 1 << ((2*n-1).bit_length())
    # Actually need conv length 2n-1 -> N >= 2n-1
    fft_f = np.fft.rfft(f, n=N)
    fft_rev = np.fft.rfft(f[::-1], n=N)
    conv = np.fft.irfft(fft_f * fft_rev, n=N)
    autocorr = conv[:2*n-1]  # raw sum (not scaled)
    return autocorr

def compute_gradient_smooth_max(h, alpha):
    n = len(h)
    f = h - 0.5
    N_corr = 1 << ((2*n-1).bit_length())
    fft_f = np.fft.rfft(f, n=N_corr)
    fft_rev = np.fft.rfft(f[::-1], n=N_corr)
    conv_f = np.fft.irfft(fft_f * fft_rev, n=N_corr)
    autocorr = conv_f[:2*n-1]  # sum_i f_i f_{i+s}
    # compute weights
    # To avoid overflow, subtract max autocorr to stabilize
    max_autocorr = autocorr.max()
    w = np.exp(-alpha * (autocorr - max_autocorr))
    w_sum = w.sum()
    w = w / w_sum  # softmax weights

    # compute convolution w * f_{j+s} and w * f_{j-s}
    # note w_rev is w reversed
    w_rev = w[::-1]

    # conv of w_rev with f gives sum_s w_s * f_{j+s}
    N_grad = 1 << ((2*n-1 + n - 1).bit_length())
    # length needed: len(w_rev) + n - 1 = (2n-1) + n - 1 = 3n - 2
    N_grad = 1 << ((3*n - 2).bit_length())
    fft_w_rev = np.fft.rfft(w_rev, n=N_grad)
    fft_f2 = np.fft.rfft(f, n=N_grad)
    conv_plus = np.fft.irfft(fft_w_rev * fft_f2, n=N_grad)
    # conv of w with f_rev gives sum_s w_s * f_{j-s}
    fft_w = np.fft.rfft(w, n=N_grad)
    fft_f_rev2 = np.fft.rfft(f[::-1], n=N_grad)
    conv_minus = np.fft.irfft(fft_w * fft_f_rev2, n=N_grad)

    # Both conv arrays have length N_grad, we need slice for j index.
    # conv_plus at index j + (n-1) gives sum_s w_s * f_{j+s}
    # conv_minus at index j + (n-1) gives sum_s w_s * f_{j-s}
    offset = n - 1
    grad_f = - (conv_plus[offset:offset+n] + conv_minus[offset:offset+n])
    # gradient wrt h = gradient wrt f
    return grad_f

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.size
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def compute_gradient_smooth_max(h, alpha):
    n = h.shape[0]
    f = h - 0.5
    m = 2 * n - 1
    N1 = 1 << (m.bit_length())
    fft_f = np.fft.rfft(f, n=N1)
    fft_rev = np.fft.rfft(f[::-1], n=N1)
    conv = np.fft.irfft(fft_f * fft_rev, n=N1)
    autocorr = conv[:m]                     # sum_i f_i * f_{i+s}
    max_ac = autocorr.max()
    exp_term = np.exp(-alpha * (autocorr - max_ac))
    w = exp_term / exp_term.sum()
    w_rev = w[::-1]
    N2 = 1 << ((3 * n - 2).bit_length())
    fft_w_rev = np.fft.rfft(w_rev, n=N2)
    fft_f2 = np.fft.rfft(f, n=N2)
    conv_plus = np.fft.irfft(fft_w_rev * fft_f2, n=N2)
    fft_w = np.fft.rfft(w, n=N2)
    fft_f_rev = np.fft.rfft(f[::-1], n=N2)
    conv_minus = np.fft.irfft(fft_w * fft_f_rev, n=N2)
    offset = n - 1
    grad_f = - (conv_plus[offset:offset + n] + conv_minus[offset:offset + n])
    return grad_f

def gradient_descent_smooth(h0, time_left, max_iter=200000, step0=0.5, beta=0.9):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    alpha0, alpha_max = 2.0, 10.0
    eval_every = max(1, max_iter // 2000)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        alpha = alpha0 + (alpha_max - alpha0) * (it - 1) / max_iter
        grad = compute_gradient_smooth_max(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        step = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - step * mom, target)
        if it % eval_every == 0:
            s_new = compute_c5_fast(h_new)
            if s_new < best_s:
                best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    # final check
    s_final = compute_c5_fast(h)
    if s_final < best_s:
        best_s, best_h = s_final, h.copy()
    return best_h, best_s

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_swap_anneal(h0, rng, time_left, max_iters=200000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i, j = rng.choice(ones), rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    n = 512
    target_sum = n / 2.0
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        candidates.append(project_onto_box_sum(init, target_sum))
    candidates.append(thue_morse_binary(n))
    candidates.append(random_binary_balanced(n, rng))
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))
    best_h, best_s = None, np.inf
    for h0 in candidates:
        if time_left() < 5.0:
            break
        h_opt, s_opt = gradient_descent_smooth(h0, time_left, max_iter=200000)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= target:
            return best_h, best_s, n
    if best_h is not None:
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        s_bin = compute_c5_fast(h_bin)
        if s_bin < best_s:
            best_s, best_h = s_bin, h_bin.copy()
        if best_s <= target:
            return best_h, best_s, n
    if best_h is not None and time_left() > 1.0:
        h_ann, s_ann = binary_swap_anneal(best_h, rng, time_left, max_iters=200000)
        if s_ann < best_s:
            best_s, best_h = s_ann, h_ann.copy()
        if best_s <= target:
            return best_h, best_s, n
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), target_sum)
        best_s = compute_c5_fast(best_h)
    else:
        best_h = project_onto_box_sum(best_h, target_sum)
        best_s = compute_c5_fast(best_h)
    return best_h, best_s, n
