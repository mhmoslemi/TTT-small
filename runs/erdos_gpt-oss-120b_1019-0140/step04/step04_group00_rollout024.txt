def adam_optimize(h0, time_left, step0=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=100000):

    m = np.zeros_like(h)
    v = np.zeros_like(h)

def gradient_descent(h0, time_left, step0=0.5, beta=0.9, max_iter=100000):
   target_sum = h0.size / 2.0
   h = project_onto_box_sum(h0, target_sum)
   best_h, best_s = h.copy(), compute_c5_fast(h)
   mom = np.zeros_like(h)
   it = 1
   while it <= max_iter and time_left() > 0:
       grad = compute_gradient_max_shift(h)
       mom = beta * mom + (1 - beta) * grad
       eta = step0 / np.sqrt(it)
       h_new = project_onto_box_sum(h - eta * mom, target_sum)
       s_new = compute_c5_fast(h_new)
       if s_new < best_s:
           best_s, best_h = s_new, h_new.copy()
       h = h_new
       it += 1
   return best_h, best_s

def adam_optimize(h0, time_left, step0=0.2, beta1=0.9, beta2=0.999,
                  eps=1e-8, max_iter=50000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h, best_s = h.copy(), compute_c5_fast(h)
    t = 0
    while t < max_iter and time_left() > 0.0:
        t += 1
        grad = compute_gradient_max_shift(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)

        h = h_new

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

    h = np.zeros(n, dtype=float)
