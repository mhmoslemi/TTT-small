def binary_anneal(h0, rng, time_left, max_iters):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h = h.copy()
    best_c = cur
    T0 = 0.1
    T_min = 1e-6
    decay = 0.9999
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.01:
        # pick two indices: a 1 and a 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_val = compute_c5_fast(h)
        delta = new_val - cur
        # acceptance
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new_val
            if new_val < best_c:
                best_c = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_c
```

def fractional_anneal(h0, rng, time_left, max_iters):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h = h.copy()
    best_c = cur
    T0 = 0.1
    T_min = 1e-6
    decay = 0.9999
    T = T0
    eps_start = 0.05  # max delta
    eps_decay = 0.9999
    eps = eps_start
    for it in range(max_iters):
        if time_left() < 0.01:
            break
        # pick two distinct indices
        i, j = rng.choice(len(h), size=2, replace=False)
        # Ensure we can move from i to j: h[i] > 0, h[j] < 1
        # Determine permissible delta
        max_delta = min(eps, h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        h[i] -= delta
        h[j] += delta
        new = compute_c5_fast(h)
        delta_c = new - cur
        if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
            cur = new
            if new < best_c:
                best_c = new
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        T = max(T_min, T * decay)
        eps = max(1e-6, eps * eps_decay)
    return best_h, best_c

    T_min = 1e-6

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lower = np.min(v) - hi
    upper = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Expand bounds if necessary
    while f(lower) < s:
        lower -= (hi - lo) * 10
    while f(upper) > s:
        upper += (hi - lo) * 10
    for _ in range(max_iter):
        mid = (lower + upper) * 0.5
        fm = f(mid)
        if fm == s or (upper - lower) < 1e-12:
            break
        if fm > s:
            lower = mid
        else:
            upper = mid
    return np.clip(v - mid, lo, hi)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n)
        term1[:n_s] = h[shift:]
        term2 = np.zeros(n)
        term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n)
        term1[s:] = h[:n_s]
        term2 = np.zeros(n)
        term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_h_rev = np.fft.rfft(h[::-1], n=N)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_h_rev * fft_one_minus_h, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad
```

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        c_new = compute_c5_fast(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c

    eps = 0.05
    eps_decay = 0.9999
    for it in range(max_iters):
        if time_left() < 0.01:
            break
        # pick two indices
        i, j = rng.choice(n, size=2, replace=False)
        # max delta we can move from i to j without violating bounds
        max_delta = min(eps, h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        h[i] -= delta
        h[j] += delta
        new = compute_c5_fast(h)

            cur = new
            if new < best_c:
                best_c = new
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        T = max(T_min, T * decay)
        eps = max(1e-6, eps * eps_decay)
    return best_h, best_c

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lower = np.min(v) - hi
    upper = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    while f(lower) < s:
        lower -= (hi - lo) * 10
    while f(upper) > s:
        upper += (hi - lo) * 10
    for _ in range(max_iter):
        mid = (lower + upper) * 0.5
        fm = f(mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lower = mid
        else:
            upper = mid
    return np.clip(v - mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(h, n=N)
    B = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(A * B, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n)
        term1[:n_s] = h[shift:]
        term2 = np.zeros(n)
        term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n)
        term1[s:] = h[:n_s]
        term2 = np.zeros(n)
        term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    rev = np.fft.rfft(h[::-1], n=N)
    one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(rev * one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idx = np.where(corr == mx)[0]
    shifts = idx - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        c_new = compute_c5_fast(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_anneal_incremental(h0, rng, time_left, max_iters=200000):
    n = h0.size
    target = n // 2
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, h, mode='full')
    cur_c = dx * (target - corr.min())
    best_h = h.copy()
    best_c = cur_c
    T = 0.2
    T_min = 1e-6
    decay = 0.9999
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        corr_old = corr.copy()
        hi, hj = h[i], h[j]  # hi=1, hj=0
        # remove i
        corr[n - 1 - i : n - 1] -= h[:i]
        corr[n - 1 : n - 1 + (n - i)] -= h[i:]
        h[i] = 0
        # add j
        corr[n - 1 - j : n - 1] += h[:j]
        corr[n - 1 : n - 1 + (n - j)] += h[j:]
        h[j] = 1
        new_c = dx * (target - corr.min())
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            corr = corr_old
            h[i], h[j] = hi, hj
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_c

def fractional_anneal(h0, rng, time_left, max_iters=50000):
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_c = h.copy(), cur
    T = 0.2
    T_min = 1e-6
    decay = 0.9995
    eps = 0.05
    eps_decay = 0.9999
    for it in range(max_iters):
        if time_left() < 0.01:
            break
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(eps, h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        h[i] -= delta
        h[j] += delta
        new = compute_c5_fast(h)
        if new < cur or rng.random() < np.exp(-(new - cur) / T):
            cur = new
            if new < best_c:
                best_c, best_h = new, h.copy()
        else:
            h[i] += delta
            h[j] -= delta
        T = max(T_min, T * decay)
        eps = max(1e-6, eps * eps_decay)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))

    n = 512
    target_sum = n // 2
    target_bound = 0.3808

    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            init = resample(init, n)
        candidates.append(project_onto_box_sum(init, target_sum))
    candidates.append(thue_morse_binary(n))
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))

    best_h = None
    best_c = np.inf

    # binary annealing on each binary candidate
    for cand in candidates:
        if time_left() < 2.0:
            break
        if np.all(np.logical_or(cand == 0, cand == 1)):
            max_iters = int(min(400000, time_left() / 0.001))
            h_opt, c_opt = binary_anneal_incremental(cand, rng, time_left, max_iters=max_iters)
        else:
            cand = project_onto_box_sum(cand, target_sum)
            max_iters = int(min(80000, time_left() / 0.01))
            h_opt, c_opt = fractional_anneal(cand, rng, time_left, max_iters=max_iters)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        if best_c <= target_bound:
            return best_h, best_c, n

    # further binary refinement
    if best_h is not None and np.all(np.logical_or(best_h == 0, best_h == 1)):
        max_iters = int(min(800000, time_left() / 0.001))
        h_opt, c_opt = binary_anneal_incremental(best_h, rng, time_left, max_iters=max_iters)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        if best_c <= target_bound:
            return best_h, best_c, n

    # fractional refinement
    if best_h is not None:
        h_float = best_h.astype(float)
        h_float = project_onto_box_sum(h_float, target_sum)
        max_iters = int(min(150000, time_left() / 0.01))
        h_opt, c_opt = fractional_anneal(h_float, rng, time_left, max_iters=max_iters)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        if best_c <= target_bound:
            return best_h, best_c, n

    # gradient descent momentum as final polish
    if best_h is not None:
        try:
            h_opt, c_opt = gradient_descent_momentum(best_h, time_left, max_iter=200000)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
        except Exception:
            pass

    # final projection and exact evaluation
    best_h = project_onto_box_sum(best_h, target_sum)
    best_c = compute_c5_fast(best_h)
    return best_h, best_c, n
