def gradient_descent_h(h, time_allow, rng):
    start = time.time()
    n = h.shape[0]
    total_sum = n/2
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    no_improve_iter = 0
    max_no_improve = 200
    step_sizes = [0.5,0.2,0.1,0.05,0.02,0.01,0.005,0.002,0.001]
    while time.time() - start < time_allow:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        # find top lags
        # we want lags with maximum overlap (i.e., max corr)
        # We can take top_k = 2
        top_k = 2
        idxs = np.argpartition(-corr, top_k-1)[:top_k]
        # compute direction
        d = np.zeros_like(h)
        for idx in idxs:
            lag = idx - (n - 1)
            if lag >= 0:
                # shift right
                if lag < n:
                    d[:n-lag] += h[lag:]
                    d[lag:] += h[:n-lag]
            else:
                L = -lag
                # shift left (same as positive L)
                if L < n:
                    d[:n-L] += h[L:]
                    d[L:] += h[:n-L]
        # Normalize direction maybe to unit length? Not needed.
        # Try steps
        improved = False
        for step in step_sizes:
            h_candidate = project_onto_box_sum(h + step * d, total_sum)
            cand_score = compute_c5(h_candidate)
            if cand_score < cur_score - 1e-12:
                h = h_candidate
                cur_score = cand_score
                improved = True
                # update best if needed
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
                break
        if not improved:
            no_improve_iter += 1
        else:
            no_improve_iter = 0
        if no_improve_iter >= max_no_improve:
            # random restart
            h = random_balanced_binary(n, rng).astype(float)
            h = project_onto_box_sum(h, total_sum)
            cur_score = compute_c5(h)
            no_improve_iter = 0
    return best_h, best_score

def gradient_descent_h(h, time_allow, rng):
    start = time.time()
    n = h.shape[0]
    total = n / 2.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    no_improve = 0
    max_no_improve = 200
    step_sizes = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    while time.time() - start < time_allow:
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        # find top lags
        top_k = 2
        # get indices of top_k largest entries
        if top_k < len(corr):
            idxs = np.argpartition(-corr, top_k - 1)[:top_k]
        else:
            idxs = np.arange(len(corr))
        # compute direction
        d = np.zeros_like(h)
        for idx in idxs:
            lag = idx - (n - 1)
            if lag >= 0:
                if lag < n:
                    d[:n - lag] += h[lag:]
                    d[lag:] += h[:n - lag]
            else:
                L = -lag
                if L < n:
                    d[:n - L] += h[L:]
                    d[L:] += h[:n - L]
        improved = False
        for step in step_sizes:

            cand_score = compute_c5(h_candidate)

                cur_score = cand_score
                improved = True

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

            if alloc > remaining:
                alloc = remaining
        if alloc <= 0.0:
            continue

        cand_h, cand_score = gradient_descent_h(base, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        if best_score <= target:
            break

    # final local refinements using tiny mass transfers
    while time_left() > 0.6 and best_score > target:
        n = best_h.size
        # small deterministic transfer
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0.0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_h = h_new
                best_score = sc_new
                continue
        # occasional random binary restart
        if rng.random() < 0.1:
            h_rand = random_balanced_binary(n, rng).astype(float)
            sc_rand = compute_c5(h_rand)
            if sc_rand < best_score:
                best_h = h_rand
                best_score = sc_rand

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
