def compute_softmax_gradient(h, alpha):
    n = h.shape[0]
    M = 2 * n
    # correlation of h and 1-h
    corr = compute_corr_scaled_fft(h)  # length 2n-1
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    if w_sum == 0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum
    # Compute forward term
    p_rev = p[::-1]
    H = np.fft.rfft(h, n=M)
    P_rev = np.fft.rfft(p_rev, n=M)
    conv_f = np.fft.irfft(H * P_rev, n=M)
    forward = conv_f[n-1 : n-1 + n]
    # Compute backward term
    P = np.fft.rfft(p, n=M)
    conv_b = np.fft.irfft(H * P, n=M)
    backward = conv_b[n-1 : n-1 + n]
    grad = 1.0 - forward - backward
    return grad
```

def softmax_gradient_descent(h0, time_left_func, max_iter=200000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    alpha = 1.0
    alpha_max = 50.0
    alpha_inc = (alpha_max - alpha) / max_iter  # linear increase
    while it <= max_iter and time_left_func() > 0.0:
        grad = compute_softmax_gradient(h, alpha)
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # update alpha
        alpha = min(alpha_max, alpha + alpha_inc)
        it += 1
    return best_h, best_score
```

    n = h.shape[0]
    M = 2 * n
    # correlation
    corr = compute_corr_scaled_fft(h)  # length 2n-1
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    if w_sum == 0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum
    p_rev = p[::-1]
    H = np.fft.rfft(h, n=M)
    P_rev = np.fft.rfft(p_rev, n=M)
    conv_f = np.fft.irfft(H * P_rev, n=M)
    forward = conv_f[n-1 : n-1 + n]
    P = np.fft.rfft(p, n=M)
    conv_b = np.fft.irfft(H * P, n=M)
    backward = conv_b[n-1 : n-1 + n]
    grad = 1.0 - forward - backward
    return grad

        h = h_new
        alpha = min(alpha_max, alpha + alpha_inc)
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, start_time, budget_s, max_iters=400000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size

    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_remaining(start_time, budget_s) > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

import time
import numpy as np

def time_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_softmax_gradient(h, alpha):
    n = h.shape[0]
    M = 2 * n
    corr = compute_corr_scaled_fft(h)           # length 2n-1
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    if w_sum == 0.0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum
    p_rev = p[::-1]
    H = np.fft.rfft(h, n=M)
    P_rev = np.fft.rfft(p_rev, n=M)
    conv_f = np.fft.irfft(H * P_rev, n=M)
    forward = conv_f[n - 1:n - 1 + n]
    P = np.fft.rfft(p, n=M)
    conv_b = np.fft.irfft(H * P, n=M)
    backward = conv_b[n - 1:n - 1 + n]
    grad = 1.0 - forward - backward
    return grad

def softmax_gradient_descent(h0, start_time, budget_s,
                             max_iter=20000, step0=0.5, beta=0.9,
                             alpha0=1.0, alpha_max=50.0):

    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    alpha = alpha0
    if max_iter > 0:
        alpha_inc = (alpha_max - alpha0) / max_iter
    else:
        alpha_inc = 0.0
    while it <= max_iter and time_remaining(start_time, budget_s) > 0.0:
        grad = compute_softmax_gradient(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        alpha = min(alpha_max, alpha + alpha_inc)
        it += 1
    return best_h, best_score

    while it < max_iters and time_remaining(start_time, budget_s) > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target = 0.3808
    best_h = None
    best_score = np.inf

    # Phase 1: small dimension
    n_small = 64
    candidates = []
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.size != n_small:
            init_arr = downsample(init_arr, n_small)
        candidates.append(project_onto_box_sum(init_arr, n_small / 2.0))
    candidates.append(thue_morse(n_small))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(3):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))
    for cand in candidates:
        if time_remaining(start_time, budget_s) < 5.0:
            break
        h_opt, sc_opt = softmax_gradient_descent(
            cand, start_time, budget_s,
            max_iter=3000, step0=0.5, beta=0.9,
            alpha0=1.0, alpha_max=30.0)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 2: medium resolution
    n_mid = 256
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score = compute_c5_fft(best_h)
    h_mid = upsample(best_h, n_mid)
    h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
    if time_remaining(start_time, budget_s) > 5.0:
        h_mid_opt, sc_mid_opt = softmax_gradient_descent(
            h_mid, start_time, budget_s,
            max_iter=8000, step0=0.5, beta=0.9,
            alpha0=5.0, alpha_max=50.0)
        if sc_mid_opt < best_score:
            best_score, best_h = sc_mid_opt, h_mid_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 3: large resolution
    n_large = 512
    h_large = upsample(best_h, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    if time_remaining(start_time, budget_s) > 5.0:
        h_large_opt, sc_large_opt = softmax_gradient_descent(
            h_large, start_time, budget_s,
            max_iter=20000, step0=0.5, beta=0.9,
            alpha0=10.0, alpha_max=80.0)
        if sc_large_opt < best_score:
            best_score, best_h = sc_large_opt, h_large_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 4: binary conversion
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5_fft(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # Phase 5: binary annealing
    if time_remaining(start_time, budget_s) > 2.0:
        time_left = time_remaining(start_time, budget_s)
        max_iters = int(min(400000, time_left * 20000))
        h_bin_ref, sc_bin_ref = binary_swap_anneal(
            best_h, rng, start_time, budget_s,
            max_iters=max_iters, T0=0.2, decay=0.99995, Tmin=1e-5)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 6: final continuous refinement
    if time_remaining(start_time, budget_s) > 1.0:
        time_left = time_remaining(start_time, budget_s)
        max_iters = int(min(10000, time_left * 5000))
        h_ref2, sc_ref2 = softmax_gradient_descent(
            best_h, start_time, budget_s,
            max_iter=max_iters, step0=0.5, beta=0.9,
            alpha0=20.0, alpha_max=100.0)
        if sc_ref2 < best_score:
            best_score, best_h = sc_ref2, h_ref2.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 7: random continuous restarts
    while time_remaining(start_time, budget_s) >= 5.0:
        v = rng.random(n_large)
        v = project_onto_box_sum(v, n_large / 2.0)
        time_left = time_remaining(start_time, budget_s)
        max_iters = int(min(5000, time_left * 1000))
        h_tmp, sc_tmp = softmax_gradient_descent(
            v, start_time, budget_s,
            max_iter=max_iters, step0=0.5, beta=0.9,
            alpha0=1.0, alpha_max=30.0)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
            if best_score <= target:
                break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
