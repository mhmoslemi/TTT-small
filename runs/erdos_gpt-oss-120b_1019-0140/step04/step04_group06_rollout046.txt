def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # initialization
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)

    # Scales
    scales = [64, 128, 256, 512, 1024]
    best_h = None
    best_score = float('inf')
    for s_idx, n in enumerate(scales):
        if time_left() <= 0: break
        # upsample to size n if needed
        if base.shape[0] != n:
            base = upsample(base, n)
        # ensure constraints
        base = project_onto_box_sum(base, n/2)

        # allocate budget for this scale: proportion. Maybe diminishing proportion for early scales.
        # For simplicity, compute remaining time and allocate some fraction:
        # For last scale allocate most
        if n < scales[-1]:
            # allocate small fraction: e.g., 5% each of remaining time divided equally among early scales
            alloc = 0.07 * (budget_s - (time.time() - start))
        else:
            alloc = time_left()  # all remaining for final

        # gradient descent
        iters = int(min(50000, max(5000, alloc * (2000.0 / n))))  # similar to earlier

        if iters > 0 and time_left() > 0:
            # adjust step size based on n
            step0 = 0.6 if n <= 256 else 0.4
            gd_h, gd_score = grad_descent(base, n_iter=iters, step0=step0,
                                          beta=0.9, rng=rng,
                                          time_left=lambda: time_left())
            if gd_score < best_score:
                best_score, best_h = gd_score, gd_h.copy()
            base = gd_h

        # simulated annealing
        if time_left() > 0:
            sa_h, sa_score = simulated_anneal(base, compute_c5(base), rng,
                                              time_left=lambda: time_left())
            if sa_score < best_score:
                best_score, best_h = sa_score, sa_h.copy()
            base = sa_h

        # prepare for next scale
        # will be upsample in next iteration

    # Final intensive local search
    n = best_h.size if best_h is not None else scales[-1]
    # ensure best_h defined
    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(scales[-1]), scales[-1]/2)
        best_score = compute_c5(best_h)

    # local search
    while time_left() > 1.0:
        # try random pairwise move
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(best_h[i], 1 - best_h[j]) * rng.uniform(0.05, 0.3)
        if delta > 0:
            cand = best_h.copy()
            cand[i] -= delta
            cand[j] += delta
            # no need to project because delta ensures bounds, sum preserved
            cand_score = compute_c5(cand)
            if cand_score < best_score:

        # occasionally do small gradient step without projection if already feasible
        if rng.random() < 0.02:
            grads = compute_gradients_for_shifts(best_h, max_shifts(best_h))
            step = 0.01 * rng.random()

            cand = project_onto_box_sum(cand, n/2)
            cand_score = compute_c5(cand)
            if cand_score < best_score:
                best_score, best_h = cand_score, cand
        # occasional random binary restart refinement
        if rng.random() < 0.01 and time_left() > 5:

            rand_h = project_onto_box_sum(rand_h, n/2)

                                              beta=0.9, rng=rng,
                                              time_left=lambda: time_left())
            if rand_score < best_score:
                best_score, best_h = rand_score, rand_h

    best_h = project_onto_box_sum(best_h, best_h.size/2)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time, math, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent_fast(h0, step0, beta, rng, time_left, check_every=50):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time_left() > 0:
        it += 1
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        if it % check_every == 0:
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_anneal(h0, score0, rng, time_left):
    h = h0.copy()
    best_h, best_score = h.copy(), score0
    cur_score = score0
    start = time.time()
    T0, T1 = 5e-3, 1e-5
    while time_left() > 0:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed)
        T = T0 * (T1 / T0) ** frac
        i, j = rng.choice(h.size, size=2, replace=False)
        delta = min(h[i], 1.0 - h[j]) * rng.uniform(0.1, 0.5)
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            h, cur_score = h_new, sc_new
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
        if rng.random() < 0.05:
            grad = compute_gradients_for_shifts(h, max_shifts(h))
            step = 0.02
            h_tmp = h - step * grad
            h_tmp = project_onto_box_sum(h_tmp, h.size / 2.0)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < cur_score:
                h, cur_score = h_tmp, sc_tmp
                if cur_score < best_score:
                    best_score, best_h = cur_score, h.copy()
    return best_h, best_score
