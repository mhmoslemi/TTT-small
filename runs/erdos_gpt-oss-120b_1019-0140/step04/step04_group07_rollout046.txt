    n = h.shape[0]
    N = 2*n - 1
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)

def de_optimize(base, n, max_gen, pop_size, F, CR, rng, time_left):
    # base: starting vector (size n) used to seed population as one member
    # returns best_h, best_score

    s = n/2.0

    # Initialize population as random feasible vectors
    pop = np.empty((pop_size, n), dtype=float)
    # Use base for first individual
    pop[0] = base.copy()
    for i in range(1, pop_size):
        # random uniform in [0,1]
        v = rng.random(n)
        v = project_onto_box_sum(v, s)
        pop[i] = v

    # Compute fitness
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(pop[i])

    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_score = fitness[best_idx]

    # Evolution loop
    gen = 0
    while gen < max_gen and time_left() > 0.1:
        gen += 1
        for i in range(pop_size):
            # choose three distinct indices from population, not i
            # vectorized? just select with rng.choice
            idxs = rng.choice(pop_size, size=3, replace=False)
            while i in idxs:
                idxs = rng.choice(pop_size, size=3, replace=False)
            a, b, c = idxs
            mutant = pop[a] + F * (pop[b] - pop[c])
            # Clip and project
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, s)

            # Crossover
            cross_mask = rng.random(n) < CR
            # Ensure at least one dimension from mutant
            if not np.any(cross_mask):
                cross_mask[rng.integers(n)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_onto_box_sum(trial, s)

            trial_f = compute_c5(trial)
            if trial_f <= fitness[i]:
                pop[i] = trial
                fitness[i] = trial_f
                if trial_f < best_score:
                    best_score = trial_f
                    best_h = trial.copy()
        # Optionally adjust parameters? Keep constant.

    return best_h, best_score

def gradient_descent_lite(h0, iters, step, rng, time_left):
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for k in range(iters):
        if time_left() <= 0:

        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (h.size - 1)
        grad = np.zeros_like(h)

def de_optimize(base, n, max_gen, pop_size, F, CR, rng, time_left):

    pop = np.empty((pop_size, n), dtype=float)
    pop[0] = base.copy()

        v = project_onto_box_sum(v, s)
        pop[i] = v
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(pop[i])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_score = fitness[best_idx]
    gen = 0
    while gen < max_gen and time_left() > 0.1:
        gen += 1
        for i in range(pop_size):
            # select three distinct indices not i
            idxs = rng.choice(pop_size-1, size=3, replace=False)
            idxs = np.where(idxs >= i, idxs+1, idxs)
            a, b, c = idxs
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, s)
            # Crossover
            cross = rng.random(n) < CR

                    best_h = trial.copy()
    return best_h, best_score

def gradient_descent_lite(h0, max_iters, step, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = np.zeros_like(h)
        # compute gradient contributions
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros(n, dtype=h.dtype)
                term2 = np.zeros(n, dtype=h.dtype)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros(n, dtype=h.dtype)
                term2 = np.zeros(n, dtype=h.dtype)
                term1[sft:] = h[:n - sft]
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        h = h - step * grad
        h = project_onto_box_sum(h, s)
        cur = compute_c5(h)
        if cur < best_score:
            best_score = cur
            best_h = h.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    N = 2 * n - 1
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(H * np.conj(G)).real
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):

            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
