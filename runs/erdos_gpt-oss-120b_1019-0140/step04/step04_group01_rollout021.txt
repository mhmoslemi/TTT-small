def binary_swap_anneal(h0, rng, time_left_func, max_iters=300000, T0=1.0, decay=0.99996, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > Tmin and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

                           bounds=bounds, constraints=cons,

    except Exception:

    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - s] = h[s:]
            term2[s:] = h[:n - s]
        else:
            s2 = -s
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s2:] = h[:n - s2]
            term2[:n - s2] = h[s2:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_scaled_fft(h)
        maxc = np.max(corr)
        idx = np.where(corr >= maxc - 1e-12)[0]
        shifts = idx - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fft(h)
        delta = new - cur
        accept = False
        if delta < 0:
            accept = True
        elif T > Tmin and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            cur = new
            if new < best:
                best, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_scaled_fft(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5_fft(h_opt)

def interp_resize(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 512
        h0 = random_binary_balanced(n, rng)
    best_h, best_score = h0.copy(), compute_c5_fft(h0)
    if best_score <= target:
        return best_h, best_score, n
    # continuous refinement
    if time_left() > 5.0:
        h_cont, sc_cont = gradient_descent_momentum(best_h, time_left, step0=0.6, beta=0.9, max_iter=150000)
        if sc_cont < best_score:
            best_h, best_score = h_cont, sc_cont
            if best_score <= target:
                return best_h, best_score, n
    # binary conversion
    idx = np.argpartition(-best_h, n // 2)[:n // 2]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    h_bin = project_onto_box_sum(h_bin, n / 2.0)
    sc_bin = compute_c5_fft(h_bin)
    if sc_bin < best_score:
        best_h, best_score = h_bin, sc_bin
        if best_score <= target:
            return best_h, best_score, n
    # random swap hill climbing
    while time_left() > 8.0:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        batch = min(5000, ones.size * zeros.size)
        p_samp = rng.choice(ones, size=batch, replace=True)
        q_samp = rng.choice(zeros, size=batch, replace=True)
        local_best_sc = best_score
        local_best_swap = None
        for p, q in zip(p_samp, q_samp):
            h_tmp = best_h.copy()
            h_tmp[p], h_tmp[q] = 0.0, 1.0
            sc = compute_c5_fft(h_tmp)
            if sc < local_best_sc - 1e-12:
                local_best_sc, local_best_swap = sc, (p, q)
        if local_best_swap:
            best_h[local_best_swap[0]], best_h[local_best_swap[1]] = 0.0, 1.0
            best_score = local_best_sc
            if best_score <= target:
                return best_h, best_score, n
        else:
            break
    # binary simulated annealing
    if time_left() > 5.0:
        h_ann, sc_ann = binary_swap_anneal(best_h, rng, time_left,
                                          max_iters=300000, T0=1.0,
                                          decay=0.99996, Tmin=1e-6)
        if sc_ann < best_score:
            best_h, best_score = h_ann, sc_ann
            if best_score <= target:
                return best_h, best_score, n
    # continuous gradient again (may exploit new binary shape)
    if time_left() > 4.0:
        h_cont2, sc_cont2 = gradient_descent_momentum(best_h, time_left,
                                                      step0=0.5, beta=0.9, max_iter=100000)
        if sc_cont2 < best_score:
            best_h, best_score = h_cont2, sc_cont2
            if best_score <= target:
                return best_h, best_score, n
    # SLSQP at reduced resolution
    if time_left() > 5.0:
        n_mid = 256
        h_mid = interp_resize(best_h, n_mid)
        h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
        h_opt_mid, sc_opt_mid = slsqp_optimize_candidate(h_mid, max_iter=800)
        h_up = interp_resize(h_opt_mid, n)
        h_up = project_onto_box_sum(h_up, n / 2.0)
        sc_up = compute_c5_fft(h_up)
        if sc_up < best_score:
            best_h, best_score = h_up, sc_up
            if best_score <= target:
                return best_h, best_score, n
    # final projection
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, n
