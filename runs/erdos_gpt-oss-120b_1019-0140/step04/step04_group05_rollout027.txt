def gradient_descent_opt(h0, time_limit, rng):
    h = h0.copy()
    n = h.size
    total = n/2
    start = time.time()
    # set initial step size
    step = 0.1  # maybe adjust
    alpha = 1.0  # smoothing
    # schedule parameters
    # We'll run until time runs out
    iter = 0
    while time.time() - start < time_limit:
        # compute corr and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        # compute weights
        # Use alpha up to maybe 30
        if alpha < 30:
            alpha = min(30, alpha * 1.02)  # slowly increase
        # compute weights as softmax
        # To avoid overflow, subtract max
        maxc = np.max(corr)
        exp_term = np.exp(alpha * (corr - maxc))
        w = exp_term / np.sum(exp_term)
        # compute gradient
        grad = np.zeros_like(h)
        # Loop over shifts
        # To speed up, we can precompute slices; but loops fine.
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            d = idx - (n - 1)
            if d >= 0:
                nd = n - d
                if nd > 0:
                    grad[:nd] += weight * (1 - h[d:]) - weight * h[:nd]
            else:
                p = -d
                nd = n - p
                if nd > 0:
                    grad[p:] += weight * (1 - h[:nd])
                    grad[:nd] -= weight * h[p:]
        # gradient descent step with projection
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, total)
        # adapt step size: maybe backtrack if score not improved
        cur_score = compute_c5(h)
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h = h_new
            # optionally increase step size slightly
            step = min(step * 1.05, 0.5)
        else:
            step = step * 0.5
        iter += 1
        # optional break if minimal step
        if step < 1e-6:
            break
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.shape[0]/2.0)

    scales = [64, 128, 256, 512]
    fractions = {64: 0.1, 128: 0.2, 256: 0.3, 512: 0.4}
    best_h = base.copy()
    best_score = compute_c5(best_h)

    for n in scales:
        if time_left() <= 0.0:
            break
        # upsample if needed
        if base.shape[0] != n:
            base = upsample(base, n)
        # ensure feasibility
        base = project_onto_box_sum(base, n/2.0)

        alloc = fractions[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        # gradient descent for part of allocation
        gd_time = alloc * 0.8
        sa_time = alloc - gd_time

        cand_h, cand_score = gradient_descent_opt(base, gd_time, rng)
        if cand_score < best_score:
            best_h = cand_h.copy()
            best_score = cand_score

        # optional SA refinement
        if sa_time > 0.1:
            cand_h2, cand_score2 = simulated_annealing(cand_h, sa_time, rng)
            if cand_score2 < best_score:
                best_h = cand_h2.copy()
                best_score = cand_score2

        base = cand_h  # continue from gradient descent solution

    # final refinement with remaining time
    while time_left() > 0.5:
        # alternate gradient descent and SA on best_h
        # allocate small chunk

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def simulated_annealing(h0, time_limit, rng):
    n = h0.size
    h = h0.copy()
    total = n / 2.0
    cur = compute_c5(h)
    best = h.copy()
    best_sc = cur
    start = time.time()
    max_iters = int(1e7)
    init_temp = 0.008

    for it in range(max_iters):
        if time.time() - start >= time_limit:
            break
        temp = init_temp * (1.0 - it / max_iters)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * (temp / init_temp) * 2.0
        if delta <= 0.0:
            continue
        h[i] -= delta
        h[j] += delta
        new_sc = compute_c5(h)
        if new_sc < cur:
            cur = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best = h.copy()
        else:
            if temp > 0.0 and rng.random() < np.exp(-(new_sc - cur) / temp):
                cur = new_sc
            else:
                h[i] += delta
                h[j] -= delta
    best = project_onto_box_sum(best, total)
    return best, compute_c5(best)


def gradient_descent_opt(h0, time_limit, rng):
    n = h0.size
    total = n / 2.0
    h = h0.copy()
    step = 0.1
    alpha = 1.0
    start = time.time()
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        cur_sc = maxc * (2.0 / n)
        if alpha < 30.0:
            alpha = min(30.0, alpha * 1.015)
        exp_term = np.exp(alpha * (corr - maxc))
        w = exp_term / np.sum(exp_term)
        grad = np.zeros_like(h)
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            d = idx - (n - 1)
            if d >= 0:
                nd = n - d
                if nd:
                    grad[:nd] += weight * (1 - h[d:])
                    grad[d:] -= weight * h[:nd]
            else:
                p = -d
                nd = n - p
                if nd:
                    grad[p:] += weight * (1 - h[:nd])
                    grad[:nd] -= weight * h[p:]
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, total)
        new_sc = compute_c5(h_new)
        if new_sc < cur_sc:
            h = h_new
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.5
        if step < 1e-7:
            break
    return h, compute_c5(h)
