def solve_milp(n, rng):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n // 2, t >= 0]
    n_factor = n / 2.0
    for k in range(-(n-1), n):
        if k >= 0:
            i = slice(0, n - k)
            j = slice(k, n)
        else:
            i = slice(-k, n)
            j = slice(0, n + k)
        corr = cp.sum(cp.multiply(h[i], 1 - h[j]))
        constraints.append(corr <= t * n_factor)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK_MI, verbose=False)
    except cp.error.SolverError:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return None, None
    if h.value is None or t.value is None:
        return None, None
    h_bin = np.rint(h.value).astype(int)
    # adjust sum if needed
    sum_diff = int(n // 2 - h_bin.sum())
    if sum_diff > 0:
        zeros = np.where(h_bin == 0)[0]

def binary_swap_anneal(h0, rng, time_left_func, max_iters=400000, T0=0.2, decay=0.99995, Tmin=1e-5):

    n = h.size

    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0

        accept = False

                best_score = new_score
                best_h = h.copy()
        else:

            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap C5 for step function h using FFT."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def solve_milp(n, rng):
    """Solve binary MILP for overlap minimization with n points."""
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n // 2, t >= 0]
    n_factor = n / 2.0
    for k in range(-(n - 1), n):
        if k >= 0:
            i = slice(0, n - k)
            j = slice(k, n)
        else:
            s = -k
            i = slice(s, n)
            j = slice(0, n - s)
        corr = cp.sum(cp.multiply(h[i], 1 - h[j]))
        constraints.append(corr <= t * n_factor)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK_MI, verbose=False)
    except cp.error.SolverError:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return None, None
    if h.value is None or t.value is None:
        return None, None
    h_bin = np.rint(h.value).astype(int)
    sum_diff = int(n // 2 - h_bin.sum())
    if sum_diff > 0:
        zeros = np.where(h_bin == 0)[0]
        if zeros.size >= sum_diff:
            idx = rng.choice(zeros, size=sum_diff, replace=False)
            h_bin[idx] = 1

        if ones.size >= -sum_diff:
            idx = rng.choice(ones, size=-sum_diff, replace=False)
            h_bin[idx] = 0
    return h_bin.astype(float), float(t.value)
