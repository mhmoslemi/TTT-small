    grad_momentum = np.zeros_like(h)
    iteration = 0

        iteration += 1

        # step size schedule: step0 = 0.5 maybe, decreasing like step0 / sqrt(iteration)

def compute_c5_fft(h):
    n = h.shape[0]
    # correlation via FFT
    # convolution of h and reversed(1-h)
    rev = (1 - h)[::-1]
    m = 2 * n - 1

    R = np.fft.rfft(rev, n=nfft)

    corr = conv[::-1]
    return corr.max() * (2.0 / n), corr

def subgrad_max_corr_fft(h):
    n = h.shape[0]
    # compute correlation via FFT
    rev = (1 - h)[::-1]
    m = 2 * n - 1
    nfft = 1 << (m - 1).bit_length()
    H = np.fft.rfft(h, n=nfft)
    R = np.fft.rfft(rev, n=nfft)
    conv = np.fft.irfft(H * R, n=nfft)[:m]
    corr = conv[::-1]
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    grad = np.zeros_like(h)
    for ind in idx:
        shift = ind - (n - 1)
        if shift >= 0:

            tgt = slice(0, n - s)

        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < target_sum:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    v = np.zeros(n, dtype=float)
    v[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return v

def subgrad_max_corr_fft(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    R = np.fft.rfft(rev, n=nfft)
    conv = np.fft.irfft(H * R, n=nfft)[:m]
    corr = conv[::-1]
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    grad = np.zeros_like(h)
    for ind in idx:
        shift = ind - (n - 1)
        if shift >= 0:
            src = slice(0, n - shift)
            tgt = slice(shift, n)
            grad[src] += 1 - h[tgt]
            grad[tgt] -= h[src]
        else:
            s = -shift
            src = slice(s, n)
            tgt = slice(0, n - s)
            grad[src] += 1 - h[tgt]
            grad[tgt] -= h[src]
    grad = grad / len(idx)
    return grad, maxc

def gradient_descent_until_time(h, target_sum, time_limit, start_time, rng):
    n = h.shape[0]
    best_h = h.copy()
    _, maxc0 = subgrad_max_corr_fft(h)
    best_score = maxc0 * (2.0 / n)
    grad_mom = np.zeros_like(h)
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        it += 1
        grad, _ = subgrad_max_corr_fft(h)
        grad_mom = 0.9 * grad_mom + 0.1 * grad
        eta = 0.6 / math.sqrt(it)
        h_new = h - eta * grad_mom
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        _, maxc_new = subgrad_max_corr_fft(h_new)
        sc_new = maxc_new * (2.0 / n)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def compute_c5_fft(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    R = np.fft.rfft(rev, n=nfft)
    conv = np.fft.irfft(H * R, n=nfft)[:m]
    corr = conv[::-1]
    maxc = corr.max()
    return maxc * (2.0 / n), corr

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    scales = [128, 256, 512, 768]
    best_h = None
    best_score = float('inf')

    for n in scales:
        if base.shape[0] != n:
            base = upsample(base, n)
        target = n / 2.0
        base = project_onto_box_sum(base, target)

        # quick evaluation
        cur_score, _ = compute_c5_fft(base)
        if cur_score < best_score:
            best_score, best_h = cur_score, base.copy()

        # allocate time for this scale
        remaining = budget_s - (time.time() - start)
        if remaining <= 8:
            break
        scale_time = min(remaining * 0.2, 30.0)
        scale_start = time.time()
        opt_h, opt_score = gradient_descent_until_time(
            base, target, scale_time, scale_start, rng)
        if opt_score < best_score:
            best_score, best_h = opt_score, opt_h.copy()
        base = opt_h  # upscale from this refined solution

    # intensive search at final resolution
    final_n = best_h.shape[0] if best_h is not None else scales[-1]
    target = final_n / 2.0
    while True:
        remaining = budget_s - (time.time() - start)
        if remaining <= 8.0:
            break
        # random restart
        h_rand = random_binary_balanced(final_n, rng)
        restart_time = min(remaining * 0.15, 20.0)
        restart_start = time.time()
        h_opt, sc_opt = gradient_descent_until_time(
            h_rand, target, restart_time, restart_start, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= 0.3808:
                break

        # small random tweak of current best
        for _ in range(5):
            if (budget_s - (time.time() - start)) <= 0:
                break
            i, j = rng.choice(final_n, size=2, replace=False)
            if best_h[i] > 0 and best_h[j] < 1:
                delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.1, 0.3)
                h_new = best_h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                sc_new, _ = compute_c5_fft(h_new)
                if sc_new < best_score:
                    best_score, best_h = sc_new, h_new.copy()
                    if best_score <= 0.3808:
                        break

    # final projection and exact evaluation
    best_h = project_onto_box_sum(best_h, best_h.shape[0] / 2.0)
    if 'evaluate_erdos_solution' in globals():
        final_score = evaluate_erdos_solution(best_h)
    else:
        final_score, _ = compute_c5_fft(best_h)

    return best_h, final_score, best_h.shape[0]
