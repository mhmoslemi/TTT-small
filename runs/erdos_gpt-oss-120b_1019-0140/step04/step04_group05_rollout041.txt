    grad = np.empty_like(h)
    # compute (1 - h_{i+lag}) term:
    if lag >= 0:

    n = h0.size
    total = n/2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score
    start = time.time()
    it = 0
    step0 = 0.5
    # maybe compute initial grad and lag:
    while time.time() - start < time_limit:
        # compute correlation
        n2 = 2*n
        c = np.fft.ifft(np.fft.fft(h, n2) * np.fft.fft(1.0 - h, n2)).real
        idx = int(np.argmax(c))
        lag = idx - (n-1)
        c5 = c[idx] * (2.0/n)
        # update best
        if c5 < best_score:
            best_score = c5
            best_h = h.copy()
        # compute gradient for lag
        grad = gradient_for_lag(h, lag)
        # compute step size
        step = step0 / np.sqrt(it + 1.0)
        # take gradient step
        h_new = h - step * grad
        # project
        h_new = project_onto_box_sum(h_new, total)
        # compute new score
        c_new = np.fft.ifft(np.fft.fft(h_new, n2) * np.fft.fft(1.0 - h_new, n2)).real
        c5_new = c_new[idx] * (2.0/n)  # note: idx may not be worst now; compute full max
        c5_new = np.max(c_new) * (2.0/n)
        # acceptance
        if c5_new <= cur_score or rng.random() < np.exp(-(c5_new - cur_score) / max(step, 1e-8)):
            h = h_new
            cur_score = c5_new
        # occasionally random perturb

def gradient_for_lag(h, lag):
    n = h.shape[0]
    grad = np.empty(n, dtype=h.dtype)
    if lag >= 0:

            grad[:n-lag] = 1.0 - h[lag:]

            # second term subtract
            grad[lag:] -= h[:n-lag]

def gradient_descent(h0, time_limit, rng):
    n = h0.shape[0]
    total = n / 2.0
    h = h0.copy()
    cur_score, cur_lag = compute_c5_and_lag(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    it = 0
    # initial step size - can be chosen based on n
    step0 = 0.5
    while time.time() - start < time_limit:
        # compute gradient for current worst lag
        grad = gradient_for_lag(h, cur_lag)
        # step size schedule
        step = step0 / np.sqrt(it + 1.0)
        # propose new h
        h_new = h - step * grad
        # project onto constraints
        h_new = project_onto_box_sum(h_new, total)
        # evaluate new
        new_score, new_lag = compute_c5_and_lag(h_new)
        # accept if improvement
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            cur_lag = new_lag

def upsample(arr, new_n):
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx  via FFT."""
    n = h.shape[0]
    n2 = 2 * n
    g_rev = (1.0 - h)[::-1]
    conv = np.fft.ifft(np.fft.fft(h, n2) * np.fft.fft(g_rev, n2)).real
    return np.max(conv[:2 * n - 1]) * (2.0 / n)


def compute_c5_and_lag(h):
    """Return (C5 value, worst lag) using FFT."""
    n = h.shape[0]
    n2 = 2 * n
    g_rev = (1.0 - h)[::-1]
    conv = np.fft.ifft(np.fft.fft(h, n2) * np.fft.fft(g_rev, n2)).real
    conv = conv[:2 * n - 1]
    idx = int(np.argmax(conv))
    lag = (n - 1) - idx
    score = conv[idx] * (2.0 / n)
    return score, lag


def gradient_for_lag(h, lag):
    """Gradient of the worst‑lag term w.r.t. h."""
    n = h.shape[0]
    grad = np.empty_like(h)
    if lag >= 0:
        if lag < n:
            grad[:n - lag] = 1.0 - h[lag:]
            grad[n - lag:] = 1.0
            grad[lag:] -= h[:n - lag]
        else:
            grad[:] = 1.0
    else:
        L = -lag
        grad[:] = 1.0
        if L < n:
            grad[L:] = 1.0 - h[:n - L]
            grad[:n - L] -= h[L:]
    return grad


def gradient_descent(h0, time_limit, rng):
    """Projected gradient descent on the worst‑lag objective."""
    n = h0.shape[0]
    total = n / 2.0
    h = h0.copy()
    cur_score, cur_lag = compute_c5_and_lag(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    it = 0
    step0 = 0.5
    while time.time() - start < time_limit:
        grad = gradient_for_lag(h, cur_lag)
        step = step0 / np.sqrt(it + 1.0)
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, total)
        new_score, new_lag = compute_c5_and_lag(h_new)
        if new_score < cur_score:
            h, cur_score, cur_lag = h_new, new_score, new_lag
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
        # occasional small random move to escape plateaus
        if it % 200 == 0:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta > 0.0:
                delta = max_delta * rng.random() * 0.2
                h_tmp = h.copy()
                h_tmp[i] -= delta
                h_tmp[j] += delta
                h_tmp = project_onto_box_sum(h_tmp, total)
                sc_tmp, lag_tmp = compute_c5_and_lag(h_tmp)
                if sc_tmp < cur_score:
                    h, cur_score, cur_lag = h_tmp, sc_tmp, lag_tmp
                    if sc_tmp < best_score:
                        best_score, best_h = sc_tmp, h_tmp.copy()
        it += 1
        if best_score <= 0.3808:
            break
    best_score = compute_c5_fft(best_h)
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # simple Thue–Morse start (64 points)
        def thue_morse(n):
            seq = np.empty(n, dtype=float)
            for i in range(n):
                seq[i] = (bin(i).count('1') & 1)
            return seq
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base
    best_score = compute_c5_fft(best_h)

    scales = [128, 256, 512, 1024]
    weights = {128: 0.12, 256: 0.18, 512: 0.30, 1024: 0.40}

    for n in scales:
        if time_left() <= 0.0:
            break
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        alloc = min(time_left(), weights[n] * budget_s)
        if alloc <= 0.0:
            continue
        half = alloc / 2.0

        # descent from current best
        cand_h1, cand_score1 = gradient_descent(best_h, half, rng)
        if cand_score1 < best_score:
            best_h, best_score = cand_h1, cand_score1

        # descent from a random start
        if half > 0.5:
            rand_start = random_balanced_binary(n, rng)
            cand_h2, cand_score2 = gradient_descent(rand_start, half, rng)
            if cand_score2 < best_score:
                best_h, best_score = cand_h2, cand_score2

    # final refinement with any remaining budget
    while time_left() > 0.6:
        n = best_h.shape[0]
        # random balanced binary perturbation
        hb = random_balanced_binary(n, rng)
        hb = project_onto_box_sum(hb, n / 2.0)
        sc = compute_c5_fft(hb)
        if sc < best_score:
            best_h, best_score = hb, sc
        # small local transfer move
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0.0:

            h_new[j] += delta
            h_new = project_onto_box_sum(h_new, n / 2.0)
            sc_new = compute_c5_fft(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new, sc_new
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.shape[0] / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.shape[0]
