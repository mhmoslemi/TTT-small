def binary_local_search(h, max_iter=10000, time_left=None):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 0
    while True:
        if time_left is not None and time_left() <= 0:
            break
        # compute auto-correlation for non-negative lags
        corr = np.correlate(h, h, mode='full')
        S = corr[n-1:]  # lengths n
        # compute C5 correlation = (2/n)*( (n/2) - S ) = 1 - (2/n)*S
        # But we can compute C5 via compute_c5.
        # Find worst shift: min S (smallest)
        l_min = np.argmin(S)
        # If l_min == 0, that's trivial because S_0 = sum h_i^2 = sum h_i = n/2 (since binary). So C5=1 - (2/n)*(n/2) = 0.0? Actually S_0 = sum h_i^2 = sum h_i = n/2 > 0. Not worst.
        # Typically worst shifts are >0.
        # Compute neighbor_sum for each index
        if l_min == 0:
            # nothing to improve, break?
            break
        # compute neighbor_sum quickly using vectorized slices
        ns = np.zeros_like(h, dtype=int)
        if l_min < n:
            ns[:n-l_min] += h[l_min:]
            ns[l_min:] += h[:n-l_min]
        # Determine best swap
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        # neighbor_sum values for ones and zeros
        ns_ones = ns[ones_idx]
        ns_zeros = ns[zeros_idx]
        i_min_idx_local = np.argmin(ns_ones)
        j_max_idx_local = np.argmax(ns_zeros)
        i = ones_idx[i_min_idx_local]
        j = zeros_idx[j_max_idx_local]
        delta = int(ns[j] - ns[i])
        if abs(i-j) == l_min:
            delta -= 1
        if delta <= 0:
            # no improvement for this worst shift; try next worst shift?
            # Option: find next worst shift (second min) and try improvement.
            # Or break.
            # Let's try to detect if any shift can be improved:
            # Compute for all shifts sorted by S.
            # For each shift in sorted_lags (excluding 0 and l_min) attempt similar swap.
            sorted_lags = np.argsort(S)
            improved = False
            for l in sorted_lags[1:]:
                if l == 0:
                    continue
                ns_temp = np.zeros_like(h, dtype=int)
                if l < n:
                    ns_temp[:n-l] += h[l:]
                    ns_temp[l:] += h[:n-l]
                # compute best swap for this l
                ns_ones_temp = ns_temp[ones_idx]
                ns_zeros_temp = ns_temp[zeros_idx]
                i2 = ones_idx[np.argmin(ns_ones_temp)]
                j2 = zeros_idx[np.argmax(ns_zeros_temp)]
                delta2 = int(ns_temp[j2] - ns_temp[i2])
                if abs(i2-j2) == l:
                    delta2 -= 1
                if delta2 > 0:
                    # perform swap and set improved flag, update h, break out
                    h[i2] = 0
                    h[j2] = 1
                    improved = True
                    break
            if not improved:
                # no improving swap for any shift, break
                break
            else:
                # continue loop after swap
                continue
        # perform swap
        h[i] = 0
        h[j] = 1
        it += 1
        # compute score, update best
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        if it >= max_iter:
            break
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random jitter
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

            i2 = ones_idx[np.argmin(ns_ones2)]

                    best_h = h.astype(float).copy()
                improved = True
                break

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_local_search(h_float, max_iter, rng, time_left):
    h = np.rint(h_float).astype(int)
    n = h.size
    target_sum = n // 2
    cur_sum = h.sum()
    if cur_sum != target_sum:
        while cur_sum > target_sum:
            idx = np.where(h == 1)[0]
            i = rng.choice(idx)
            h[i] = 0
            cur_sum -= 1
        while cur_sum < target_sum:
            idx = np.where(h == 0)[0]
            i = rng.choice(idx)
            h[i] = 1
            cur_sum += 1
    best_h = h.astype(float).copy()
    best_score = compute_c5(best_h)
    it = 0
    while it < max_iter:
        if time_left() <= 0:
            break
        corr = np.correlate(h, h, mode='full')
        S = corr[n - 1:]  # S[l] for l>=0
        if n <= 1:
            break
        min_l = np.argmin(S[1:]) + 1
        l = min_l
        ns = np.zeros(n, dtype=int)
        if l < n:
            ns[:n - l] += h[l:]
            ns[l:] += h[:n - l]
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        ns_ones = ns[ones_idx]
        ns_zeros = ns[zeros_idx]
        i = ones_idx[np.argmin(ns_ones)]
        j = zeros_idx[np.argmax(ns_zeros)]
        delta = int(ns[j] - ns[i])
        if abs(i - j) == l:

        if delta > 0:
            h[i] = 0

            it += 1
            score = compute_c5(h.astype(float))
            if score < best_score:

                ns2[l2:] += h[:n - l2]

            j2 = zeros_idx[np.argmax(ns_zeros2)]

    return best_h, best_score
