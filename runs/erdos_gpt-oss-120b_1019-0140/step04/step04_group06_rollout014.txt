def compute_c5(h):
    n = h.shape[0]
    L = 2*n - 1
    fh = np.fft.rfft(h, L)
    fh_rev = np.fft.rfft(h[::-1], L)

    max_corr = np.max(corr)
    return 1.0 - (2.0 / n) * max_corr

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project vector v onto the box [lo,hi]^n while fixing sum(v)=s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for step function h."""
    n = h.shape[0]
    L = 2 * n - 1
    # correlation of h with itself (no reversal needed for np.correlate behavior)
    fh = np.fft.rfft(h, L)
    fh_rev = np.fft.rfft(h[::-1], L)
    corr = np.fft.irfft(fh * fh_rev, L)
    max_corr = np.max(corr)
    c5 = 1.0 - (2.0 / n) * max_corr
    return float(c5) if c5 > 0 else 0.0

def max_shifts(h):
    """Return list of integer shifts where correlation attains its maximum."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.rfft(h, L)
    fh_rev = np.fft.rfft(h[::-1], L)
    corr = np.fft.irfft(fh * fh_rev, L)
    maxc = np.max(corr)
    idx = np.where(np.isclose(corr, maxc, atol=1e-9))[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    """Gradient of the overlap integral for given max shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    """Linearly upsample arr (assumed on [0,2]) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Return a binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

            if cur_score < best_score:
                best_score, best_h = cur_score, cur_h.copy()
                if best_score <= 0.3808:
                    break

        # occasional aggressive perturbation
        if rng.random() < 0.02:
            # perform a short gradient step from current binary state
            shifts = max_shifts(cur_h)
            grad = compute_gradients_for_shifts(cur_h, shifts)
            step = 0.1
            h_tmp = cur_h - step * grad
            h_tmp = project_onto_box_sum(h_tmp, n / 2.0)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < cur_score:
                cur_h, cur_score = h_tmp, sc_tmp
                if sc_tmp < best_score:
                    best_score, best_h = sc_tmp, h_tmp.copy()

    # final projection and score
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
