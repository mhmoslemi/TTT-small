def anneal_binary(h0, rng, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    start = time.time()
    # schedule: T0 ~ 0.1, T_end ~ 1e-4
    # we will use total time = time_limit.
    # inside while time left > some small threshold
    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        # compute T based on fraction
        frac = (time.time() - start) / time_limit
        T = 0.1 * (1e-4 / 0.1) ** frac
        # choose random 1-index i and 0-index j.
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # try swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_score = compute_c5(h_new)
        dE = new_score - cur_score
        accept = False
        if dE < 0:
            accept = True
        else:
            # accept with prob exp(-dE/T)
            if rng.random() < math.exp(-dE / T):
                accept = True
        if accept:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
    return best_h, best_score

import time, math, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_balance(b, target):
    cur = int(b.sum())
    if cur == target:
        return b
    indices = np.arange(b.size)
    if cur > target:
        # turn off cur-target ones
        ones = indices[b == 1]
        # use rng? We'll pass rng as global? Could use np.random default.
        # Use np.random.default_rng
        rng = np.random.default_rng()
        to_flip = rng.choice(ones, size=cur-target, replace=False)
        b[to_flip] = 0
    else:
        zeros = indices[b == 0]
        rng = np.random.default_rng()
        to_flip = rng.choice(zeros, size=target-cur, replace=False)
        b[to_flip] = 1
    return b

def binary_from_continuous(v, rng):
    b = (v > 0.5).astype(float)
    target = v.size // 2
    cur = int(b.sum())
    if cur > target:
        ones = np.where(b == 1)[0]
        to_flip = rng.choice(ones, size=cur-target, replace=False)
        b[to_flip] = 0
    elif cur < target:
        zeros = np.where(b == 0)[0]
        to_flip = rng.choice(zeros, size=target-cur, replace=False)
        b[to_flip] = 1
    return b

def random_swap_move(h, rng):
    n = h.size
    while True:
        i = rng.integers(n)
        if h[i] == 1:
            break
    while True:
        j = rng.integers(n)
        if h[j] == 0:
            break
    return i, j

def anneal_binary(h0, rng, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    start = time.time()
    # parameters for schedule
    T0 = 0.1
    T_end = 1e-4
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        i, j = random_swap_move(h, rng)
        # apply swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5(h)
        dE = new_score - cur_score
        # schedule
        frac = elapsed / time_limit
        T = T0 * (T_end / T0) ** frac
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best_h, best_score

def upsample_repeat(v, new_n):
    factor = new_n // v.size
    return np.repeat(v, factor)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def annealing_with_restarts(h0, rng, time_limit):

import time, math, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_balance(b, target, rng):
    cur = int(b.sum())
    if cur == target:
        return b
    if cur > target:
        ones = np.where(b == 1)[0]
        to_flip = rng.choice(ones, size=cur - target, replace=False)
        b[to_flip] = 0
    else:
        zeros = np.where(b == 0)[0]
        to_flip = rng.choice(zeros, size=target - cur, replace=False)
        b[to_flip] = 1
    return b

def binary_from_continuous(v, rng):
    b = (v > 0.5).astype(float)
    target = v.size // 2
    cur = int(b.sum())
    if cur > target:
        ones = np.where(b == 1)[0]
        to_flip = rng.choice(ones, size=cur - target, replace=False)
        b[to_flip] = 0
    elif cur < target:
        zeros = np.where(b == 0)[0]
        to_flip = rng.choice(zeros, size=target - cur, replace=False)
        b[to_flip] = 1
    return b

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_swap_move(h, rng):
    n = h.size
    while True:
        i = rng.integers(n)
        if h[i] == 1:
            break
    while True:
        j = rng.integers(n)
        if h[j] == 0:
            break
    return i, j

def anneal_binary(h0, rng, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    start = time.time()
    T0, T_end = 0.1, 1e-4
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        i, j = random_swap_move(h, rng)
        h[i] = 0
        h[j] = 1
        new_score = compute_c5(h)
        dE = new_score - cur_score
        frac = elapsed / time_limit
        T = T0 * (T_end / T0) ** frac
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_score

def annealing_with_restarts(h0, rng, total_time):
    start = time.time()
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    restarts = 5
    for _ in range(restarts):
        elapsed = time.time() - start
        remaining = total_time - elapsed
        if remaining <= 5:
            break
        # diversify start
        h = best_h.copy()
        for _ in range(10):
            i, j = random_swap_move(h, rng)
            h[i] = 0
            h[j] = 1
        h_res, sc_res = anneal_binary(h, rng, remaining)
        if sc_res < best_score:
            best_score, best_h = sc_res, h_res.copy()
    return best_h, best_score

def greedy_refine(h, rng, max_iter=2000):
    cur_score = compute_c5(h)
    no_improve = 0
    for _ in range(max_iter):
        i, j = random_swap_move(h, rng)
        h[i] = 0
        h[j] = 1
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            no_improve = 0
        else:
            h[i] = 1
            h[j] = 0
            no_improve += 1
        if no_improve >= 500:
            break
    return h, cur_score

def upsample_repeat(v, new_n):
    factor = new_n // v.size
    return np.repeat(v, factor)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = np.clip(base, 0.0, 1.0)
        base = binary_from_continuous(base, rng)
    else:
        base = random_binary_balanced(64, rng)

    base = repair_balance(base, base.size // 2, rng)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # scale schedule (sizes and allocated seconds)
    scales = [64, 128, 256, 512, 1024]
    alloc = {64: 150, 128: 200, 256: 250, 512: 300, 1024: 200}

    for n in scales:
        if time_left() <= 5:
            break
        if best_h.size != n:
            if n % best_h.size == 0:
                best_h = upsample_repeat(best_h, n)
            else:
                # fallback linear interpolation then binarize
                x_old = np.linspace(0.0, 1.0, best_h.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                best_h = np.interp(x_new, x_old, best_h)
                best_h = binary_from_continuous(best_h, rng)
        best_h = repair_balance(best_h, n // 2, rng)

        # run annealing with possible restarts
        t_alloc = min(alloc.get(n, 100), time_left())
        h_opt, sc_opt = annealing_with_restarts(best_h, rng, t_alloc)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # final greedy refinement if time permits
    if time_left() > 5:
        best_h, best_score = greedy_refine(best_h, rng, max_iter=5000)

    best_h = repair_balance(best_h, best_h.size // 2, rng)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
