    return np.interp(x_new, x_old, arr)

import time, math, numpy as np

def time_remaining(start, budget):
    return max(0.0, budget - (time.time() - start))

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5(h):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * np.conj(G), n=L)[:2 * n - 1]
    return np.max(conv) * (2.0 / n)

def corr_and_worst_shifts(h):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * np.conj(G), n=L)[:2 * n - 1]
    max_corr = np.max(conv)
    idx = np.where(conv == max_corr)[0]
    shifts = (idx - (n - 1)).tolist()
    return conv, max_corr, shifts

def grad_worst_shifts(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += h[k:]
            grad[k:] += h[:n - k]
        else:
            s = -k
            grad[s:] += h[:n - s]
            grad[:n - s] += h[s:]
    grad *= -(2.0 / n)
    return grad

def grad_descent(h0, max_iter, step0, beta, rng, start, budget):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    cur_score = best_score
    for it in range(1, max_iter + 1):
        if time_remaining(start, budget) <= 0:
            break
        _, _, shifts = corr_and_worst_shifts(h)
        grad = grad_worst_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score, best_h = sc_new, h_new.copy()
        h, cur_score = h_new, sc_new
        if best_score <= 0.3808:
            break
    return best_h, best_score

def simulated_anneal(h0, score0, rng, start, budget):
    h = h0.copy()
    best_h = h.copy()
    best_score = score0
    cur_score = score0
    T0, T1 = 0.01, 1e-5
    while time_remaining(start, budget) > 0:
        elapsed = time.time() - start
        progress = elapsed / budget
        T = T0 * (T1 / T0) ** progress
        i, j = rng.choice(h.size, size=2, replace=False)
        delta = min(h[i], 1.0 - h[j]) * rng.uniform(0.1, 0.5)
        if delta <= 0:

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            h, cur_score = h_new, sc_new
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
