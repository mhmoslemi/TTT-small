def compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_h_rev = np.fft.rfft(h[::-1], n=N)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_h_rev * fft_one_minus_h, n=N)
    corr = conv[:2*n - 1]
    return corr.max() * (2.0 / n)  # dx=2/n

def crossover(p1, p2, rng):
    # uniform crossover
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2).astype(float)
    # repair to exactly n/2 ones
    ones = np.sum(child)
    target = p1.shape[0] // 2
    diff = int(ones - target)
    if diff > 0:
        # need to turn diff ones to zero
        idx_ones = np.where(child == 1)[0]
        # randomly select diff of them to become 0
        remove = rng.choice(idx_ones, size=diff, replace=False)
        child[remove] = 0.0
    elif diff < 0:
        # need to turn -diff zeros to one
        idx_zeros = np.where(child == 0)[0]
        add = rng.choice(idx_zeros, size=-diff, replace=False)
        child[add] = 1.0
    return child

import time, numpy as np

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_h_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_h_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n); term1[:n_s] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n); term1[s:] = h[:n_s]
        term2 = np.zeros(n); term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_h_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_h_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h, best_s = h.copy(), compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def crossover(p1, p2, rng):
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2).astype(float)
    target = p1.shape[0] // 2
    diff = int(child.sum() - target)
    if diff > 0:
        idx = np.where(child == 1)[0]
        child[rng.choice(idx, size=diff, replace=False)] = 0.0
    elif diff < 0:
        idx = np.where(child == 0)[0]
        child[rng.choice(idx, size=-diff, replace=False)] = 1.0
    return child

def mutate(child, rng):
    idx_one = np.where(child == 1)[0]
    idx_zero = np.where(child == 0)[0]
    if idx_one.size == 0 or idx_zero.size == 0:
        return child
    i = rng.choice(idx_one)
    j = rng.choice(idx_zero)
    child[i], child[j] = 0.0, 1.0
    return child

def tournament_select(fitness, rng, tournament_size=3):
    idx = rng.choice(fitness.size, size=tournament_size, replace=False)
    return idx[np.argmin(fitness[idx])]
