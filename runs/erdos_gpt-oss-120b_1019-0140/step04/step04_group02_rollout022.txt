def binary_hill_climb(b, rng, time_left):
    n = b.shape[0]
    cur_score = compute_c5_fft(b)
    while time_left() > 0.1:
        i = rng.choice(np.where(b == 1)[0])
        j = rng.choice(np.where(b == 0)[0])
        # swap
        b[i] = 0
        b[j] = 1
        new_score = compute_c5_fft(b)
        if new_score < cur_score:
            cur_score = new_score
        else:
            # revert
            b[i] = 1
            b[j] = 0
    return b, cur_score

def gradient_descent(h0, n_iter, step0, beta, epsilon, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        # Binary penalty
        if epsilon > 0:
            grad += epsilon * (1.0 - 2.0 * h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random perturbation
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(b, rng, time_left):
    b = b.copy()
    n = b.shape[0]
    score = compute_c5_fft(b)
    # Precompute indices of ones and zeros? We'll recompute each iteration but ensure stable.
    while time_left() > 0.1:
        ones = np.where(b > 0.5)[0]  # binary assumed 0 or 1, but use >0.5
        zeros = np.where(b <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        b[i] = 0.0
        b[j] = 1.0
        new_score = compute_c5_fft(b)
        if new_score < score:
            score = new_score
        else:
            # revert
            b[i] = 1.0
            b[j] = 0.0
    return b, score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

import time
import numpy as np
import scipy.optimize as opt


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as float."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) via FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.fft(h, L)
    Hc = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(Hc)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Sub‑gradient of the max‑correlation term (averaged over all max shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent(h0, n_iter, step0, beta, epsilon, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        if epsilon > 0:
            grad += epsilon * (1.0 - 2.0 * h)  # binary‑penalty term
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def binary_hill_climb(b, rng, time_left):
    """Simple hill‑climbing on binary vectors (swap 1↔0 if it improves C5)."""
    b = b.copy()
    n = b.shape[0]
    score = compute_c5_fft(b)
    while time_left() > 0.1:
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        b[i] = 0.0
        b[j] = 1.0
        new_score = compute_c5_fft(b)
        if new_score < score:
            score = new_score
        else:
            b[i] = 1.0
            b[j] = 0.0
    return b, score


def de_optimize(n, rng, time_left):
    """Light differential‑evolution global search with sum‑penalty."""
    bounds = [(0.0, 1.0)] * n
    penalty = 500.0

    def obj(x):
        return compute_c5_fft(x) + penalty * abs(np.sum(x) - n / 2)

    try:
        res = opt.differential_evolution(
            obj, bounds, maxiter=30, popsize=15, polish=False, disp=False
        )
        cand = project_onto_box_sum(res.x, n / 2)
        return cand, compute_c5_fft(cand)
    except Exception:
        h = random_balanced_binary(n, rng)
        return h, compute_c5_fft(h)
