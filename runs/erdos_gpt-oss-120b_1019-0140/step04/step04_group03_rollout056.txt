def compute_corr_fft(h):
    n = h.shape[0]
    d = 2.0 / n
    a = h.astype(np.float64)
    b = 1.0 - a
    L = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    corr = conv[:2*n-1] * d
    return corr

def compute_gradient(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_opt(h0, max_time, rng):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    # momentum
    beta = 0.9
    momentum = np.zeros_like(h)
    it = 0
    while time.time() - start < max_time:
        it += 1
        # compute correlation and worst shifts
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        # tolerance to capture all max shifts
        tol = 1e-12
        shift_idxs = np.where(corr >= max_corr - tol)[0]
        shifts = shift_idxs - (n - 1)
        # compute gradient
        grad = compute_gradient(h, shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # learning rate schedule
        eta = 0.5 / np.sqrt(it)  # could also multiply by something
        h_new = h - eta * momentum
        # occasionally add noise
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project to feasible set
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h0, max_time, rng):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    # schedule parameters
    total_steps = int(min(200000, max(5000, (max_time) * 20000)))  # steps per second approx
    for step in range(total_steps):
        if time.time() - start >= max_time:
            break
        # temperature decreasing
        frac = step / total_steps
        T = 0.01 * (1 - frac) + 1e-6
        i, j = rng.choice(n, size=2, replace=False)
        # move mass from i to j
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # evaluate
        new_score = compute_c5_fft(h_new)
        # accept if better or with probability
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if new_score < best_score:

import time
import numpy as np

TARGET = 0.3808  # early stop target


def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """Projection of v onto [lo,hi]^n with sum(v)=target_sum via bisection."""
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lo_lam = np.min(v) - hi
    hi_lam = np.max(v) - lo
    for _ in range(max_iter):
        lam = 0.5 * (lo_lam + hi_lam)
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if s > target_sum:
            lo_lam = lam
        else:
            hi_lam = lam
        if abs(s - target_sum) < 1e-12:
            break
    return np.clip(v - lam, lo, hi)


def compute_corr_fft(h):
    """Linear correlation corr = np.correlate(h, 1-h, mode='full') using FFT."""
    n = h.shape[0]
    d = 2.0 / n
    a = h.astype(np.float64)
    b = (1.0 - a)[::-1]                     # reversed second argument
    L = 1 << ((2 * n - 1).bit_length())     # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    corr = conv[: 2 * n - 1][::-1] * d       # reverse to match np.correlate order
    return corr


def compute_c5_fft(h):
    """Maximum overlap value C5 for a given h."""
    return compute_corr_fft(h).max()


def compute_gradient(h, shifts):
    """Subgradient of max_{s in shifts} corr(s) w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad


def thue_morse(n):
    """Thue‑Morse binary sequence of length n (n power of two)."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def random_continuous_balanced(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def gradient_descent_opt(h0, max_time, rng):
    """Projected momentum gradient descent on C5 for at most max_time seconds."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    it = 0
    while time.time() - start < max_time:
        it += 1
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        shifts = shift_idxs - (n - 1)
        grad = compute_gradient(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = 0.5 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
            if best_score <= TARGET:
                return best_h, best_score
        h = h_new
    return best_h, best_score


def simulated_annealing(h0, max_time, rng):
    """Simple pairwise mass‑transfer annealing."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    cur_score = compute_c5_fft(h)
    best_h, best_score = h.copy(), cur_score
    total_steps = int(min(200000, max(5000, max_time * 20000)))
    for step in range(total_steps):
        if time.time() - start >= max_time:
            break
        frac = step / total_steps
        T = 0.01 * (1.0 - frac) + 1e-6
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h, cur_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= TARGET:
                    return best_h, best_score
    return best_h, best_score

    rng = np.random.default_rng(seed)
    start_global = time.time()
    # initial base vector

    best_h = None

        if time.time() - start_global >= budget_s:

    best_score = compute_c5_fft(best_h)
