def compute_c5_fft(h):
    n = h.shape[0]
    L = 2*n - 1
    # compute FFT length L
    H = np.fft.rfft(h, n=L)
    H1 = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(H1), n=L)
    return np.max(corr) * (2.0 / n)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=float)
    h[perm[:n // 2]] = 1.0
    return h


def compute_c5_fft(h):
    """Maximum overlap integral C5 for discrete h using FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    H1 = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(H1), n=L)
    return np.max(corr) * (2.0 / n)


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) (linear, no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    H1 = np.fft.rfft(1.0 - h, n=L)
    return np.fft.irfft(H * np.conj(H1), n=L)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient of max‑correlation term, averaged over all max shifts."""
    n = h.size
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on C5."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    rng = np.random.default_rng(seed)
