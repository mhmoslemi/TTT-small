def simulated_anneal(h0, score0, rng, time_left):
    h = h0.copy()
    best_h = h.copy()
    best_score = score0
    cur_score = score0
    start = time.time()
    # schedule: linear cooling from T0 to T1 over total time
    T0 = 0.01
    T1 = 1e-5
    while time_left() > 0:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed)  # approximate progress
        T = T0 * (T1/T0) ** frac
        # propose swap
        ones = np.where(h > 0.5)[0]  # approximate; h may be fractional but close to 0/1
        zeros = np.where(h <= 0.5)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        # swap values
        h_new[i], h_new[j] = h_new[j], h_new[i]
        sc_new = compute_c5(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score)/T):
            h = h_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
        # occasional small gradient nudge
        if rng.random() < 0.02:
            # perturb a few random entries slightly (continuous)
            h_tmp = h.copy()
            idx = rng.choice(h.size, size=3, replace=False)
            h_tmp[idx] = np.clip(h_tmp[idx] + rng.normal(scale=0.1, size=3), 0, 1)
            # reproject to maintain sum

Define compute_gradients(h, epsilon=1e-6):

    corr = np.correlate(h, 1-h, mode='full')
    maxc = corr.max()
    # find indices where corr >= maxc - epsilon
    idxs = np.where(corr >= maxc - epsilon)[0]
    # for each shift compute gradient, then average
    shift_idxs = idxs - (h.size - 1)
    grad = np.zeros_like(h)
    for shift in shift_idxs:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:h.size - shift] = h[shift:]
            term2[shift:] = h[:h.size - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:h.size - s]
            term2[:h.size - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shift_idxs) > 0:
        grad /= len(shift_idxs)
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, time_left, epsilon=1e-6):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        grad = compute_gradients(h, epsilon)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        # add occasional random perturbation
        if it % 2000 == 0 and rng.random() < 0.5:
            # random jitter: add small noise
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure that f(lam_low) >= target_sum and f(lam_high) <= target_sum
    # adjust if needed
    if f(lam_low) < target_sum:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):

def compute_gradients(h, epsilon=1e-8):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    # choose shifts with correlation within epsilon of maxc
    idxs = np.where(corr >= maxc - epsilon)[0]
    shifts = idxs - (n - 1)  # shift values
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1: h[shift:] -> aligned as term1[i]=h[i+shift] for i = 0..n-shift-1
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, time_left, epsilon=1e-8):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        grad = compute_gradients(h, epsilon)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise
        if it % 2000 == 0 and rng.random() < 0.5:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_anneal(h0, score0, rng, time_left):
    h = h0.copy()
    best_h = h.copy()
    best_score = score0
    cur_score = score0
    start = time.time()
    T0, T1 = 5e-3, 1e-5
    n = h.size
    target_sum = n / 2.0
    while time_left() > 0:
        elapsed = time.time() - start
        # linear fraction approximated as elapsed / total_time? need total time? we can use fraction of elapsed / (elapsed + remaining)
        remain = time_left()
        frac = elapsed / (elapsed + remain) if (elapsed + remain) > 0 else 0.0
        T = T0 * (T1 / T0) ** frac
        # propose swap-like move
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(h[i], 1.0 - h[j]) * rng.uniform(0.1, 0.5)
        if delta > 0:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
                h, cur_score = h_new, sc_new
                if sc_new < best_score:
                    best_score, best_h = sc_new, h_new.copy()
        # occasional gradient nudge
        if rng.random() < 0.02:
            grad = compute_gradients(h, epsilon=1e-6)
            step = 0.02
            h_tmp = h - step * grad
            h_tmp = project_onto_box_sum(h_tmp, target_sum)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < cur_score or rng.random() < math.exp(-(sc_tmp - cur_score) / T):
                h, cur_score = h_tmp, sc_tmp
                if sc_tmp < best_score:
                    best_score, best_h = sc_tmp, h_tmp.copy()
        # break if perfect small bound
        if best_score <= 0.3808:
            break
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time, math, numpy as np

def remaining_time(start, budget_s):
    return max(0.0, budget_s - (time.time() - start))

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < target_sum:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients(h, epsilon=1e-8):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idxs = np.where(corr >= maxc - epsilon)[0]
    shifts = idxs - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, start, budget_s, epsilon=1e-8):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if remaining_time(start, budget_s) <= 0:
            break
        grad = compute_gradients(h, epsilon)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0 and rng.random() < 0.5:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_anneal(h0, score0, rng, start, budget_s):
    h = h0.copy()
    best_h = h.copy()
    best_score = score0
    cur_score = score0
    n = h.size
    target_sum = n / 2.0
    T0, T1 = 5e-3, 1e-5
    anneal_start = time.time()
    while remaining_time(start, budget_s) > 0:
        elapsed = time.time() - anneal_start
        remain = remaining_time(start, budget_s)
        denom = elapsed + remain
        frac = elapsed / denom if denom > 0 else 0.0
        T = T0 * (T1 / T0) ** frac
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(h[i], 1.0 - h[j]) * rng.uniform(0.1, 0.5)
        if delta > 0:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
                h, cur_score = h_new, sc_new
                if sc_new < best_score:
                    best_score, best_h = sc_new, h_new.copy()
        if rng.random() < 0.02:
            grad = compute_gradients(h, epsilon=1e-6)
            step = 0.02
            h_tmp = h - step * grad
            h_tmp = project_onto_box_sum(h_tmp, target_sum)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < cur_score or rng.random() < math.exp(-(sc_tmp - cur_score) / T):
                h, cur_score = h_tmp, sc_tmp
                if sc_tmp < best_score:
                    best_score, best_h = sc_tmp, h_tmp.copy()
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != 64:
            base = upsample(base, 64)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]
    for n in scales:
        if remaining_time(start, budget_s) <= 0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        if remaining_time(start, budget_s) > 5:
            iters = int(min(60000, max(8000, remaining_time(start, budget_s) * (3000.0 / n))))
            h_gd, sc_gd = grad_descent(base, n_iter=iters, step0=0.5, beta=0.9,
                                         rng=rng, start=start, budget_s=budget_s)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()
            base = h_gd

        if remaining_time(start, budget_s) > 5:
            h_ann, sc_ann = simulated_anneal(base, compute_c5(base), rng, start, budget_s)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()
            base = h_ann

    n_final = best_h.size
    if n_final < 512 and remaining_time(start, budget_s) > 30:
        base = upsample(best_h, 512)
        base = project_onto_box_sum(base, 512 / 2.0)
        best_h = base.copy()
        best_score = compute_c5(best_h)
        n_final = 512

    while remaining_time(start, budget_s) > 8:
        h_rand = random_binary_balanced(n_final, rng)
        iters = int(min(12000, max(2000, remaining_time(start, budget_s) * 5)))
        h_rand, sc_rand = grad_descent(h_rand, n_iter=iters, step0=0.4,
                                       beta=0.9, rng=rng, start=start, budget_s=budget_s)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.1, 0.3)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
