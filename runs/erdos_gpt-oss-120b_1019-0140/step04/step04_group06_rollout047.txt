def optimize_binary_h(h0, rng, time_left):
    n = h0.shape[0]
    cur_h = h0.copy()
    corr = compute_corr_fft(cur_h)
    cur_score = np.max(corr)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.02
    T1 = 1e-5
    it = 0
    while time_left() > 0.005:
        it += 1
        elapsed = time.time() - start
        total = max(1e-6, time_left() + elapsed)
        T = T0 * (T1 / T0) ** (elapsed / total)
        # find shift
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        shift = max_idx - (n - 1)
        # targeted swap
        if shift >= 0:
            overlap_i = np.arange(0, n - shift)
            nonoverlap_j = np.arange(n - shift, n)
            i_cand = overlap_i[(cur_h[overlap_i] > 0.5) & (cur_h[overlap_i + shift] < 0.5)]
            j_cand = nonoverlap_j[cur_h[nonoverlap_j] < 0.5]
        else:
            s = -shift
            overlap_i = np.arange(s, n)
            nonoverlap_j = np.arange(0, s)
            i_cand = overlap_i[(cur_h[overlap_i] > 0.5) & (cur_h[overlap_i - s] < 0.5)]
            j_cand = nonoverlap_j[cur_h[nonoverlap_j] < 0.5]
        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
            cur_h[i], cur_h[j] = 0.0, 1.0
            corr = compute_corr_fft(cur_h)
            cur_score = np.max(corr)
            if cur_score < best_score:
                best_score, best_h = cur_score, cur_h.copy()
            continue
        # random swap
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if cur_h[i] == cur_h[j]:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_corr = compute_corr_fft(cur_h)

import time, math, numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def binary_from_vals(vals, n):
    idx = np.argsort(vals)[::-1]
    h = np.zeros(n, dtype=float)
    h[idx[:n // 2]] = 1.0
    return h

def compute_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    H = np.fft.rfft(h, n=L)
    F = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(F), n=L)[:2 * n - 1]
    corr *= (2.0 / n)                              # dx factor
    return corr

def compute_c5_fft(h):
    return np.max(compute_corr_fft(h))

def optimize_binary_h(h0, rng, time_left):
    n = h0.shape[0]
    cur_h = h0.copy()
    corr = compute_corr_fft(cur_h)
    cur_score = np.max(corr)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    T0, T1 = 2e-2, 1e-5  # temperature bounds
    while time_left() > 0.005:
        elapsed = time.time() - start
        total = max(1e-6, time_left() + elapsed)
        T = T0 * (T1 / T0) ** (elapsed / total)

        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)

        if shift >= 0:
            overlap_i = np.arange(0, n - shift)
            nonoverlap_j = np.arange(n - shift, n)
            i_cand = overlap_i[(cur_h[overlap_i] > 0.5) &
                               (cur_h[overlap_i + shift] < 0.5)]
            j_cand = nonoverlap_j[cur_h[nonoverlap_j] < 0.5]
        else:
            s = -shift
            overlap_i = np.arange(s, n)
            nonoverlap_j = np.arange(0, s)
            i_cand = overlap_i[(cur_h[overlap_i] > 0.5) &
                               (cur_h[overlap_i - s] < 0.5)]
            j_cand = nonoverlap_j[cur_h[nonoverlap_j] < 0.5]

        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
            cur_h[i], cur_h[j] = 0.0, 1.0
            corr = compute_corr_fft(cur_h)
            cur_score = np.max(corr)
            if cur_score < best_score:

            continue

        # random swap of a 0 and a 1

        j = rng.integers(n - 1)

            j += 1
        if cur_h[i] == cur_h[j]:
