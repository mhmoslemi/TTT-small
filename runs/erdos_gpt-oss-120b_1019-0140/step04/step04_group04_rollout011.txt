                base = np.asarray(initial_h_values, dtype=float)
                if base.shape[0] != n:
                    base = upsample(base, n)
                base_cont = base
            else:
                base_cont = balanced_binary_random(n, rng).astype(float)

        h = binary_from_continuous(base_cont, n, rng)
        cur_score = compute_c5_fft(h)
        if cur_score < best_score:
            best_h, best_score = h.copy(), cur_score

        h_sa, score_sa = simulated_annealing_binary(h, n, cur_score, start, budget_s, rng)
        if score_sa < best_score:
            best_h, best_score = h_sa.copy(), score_sa

        h_hc, score_hc = hill_climb_binary(best_h, start, budget_s, rng)
        if score_hc < best_score:
            best_h, best_score = h_hc.copy(), score_hc

    final_score = compute_c5_fft(best_h)
    return best_h, final_score, best_h.size
