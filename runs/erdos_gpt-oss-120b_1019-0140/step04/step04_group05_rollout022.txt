def compute_c5(h):
    n = h.shape[0]
    # next power of 2 for convolution length 2*n-1.
    N = 1 << ((2*n-1).bit_length())
    fh = np.fft.rfft(h, n=N)
    fg = np.fft.rfft(1-h, n=N)
    conv = np.fft.irfft(fh*fg, n=N)[:2*n-1]
    return np.max(conv) * (2.0 / n)

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    # use FFT to compute correlation efficiently; reverse second argument
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft((1.0 - h)[::-1], n=N)
    conv = np.fft.irfft(H * G, n=N)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    interpolated = np.interp(x_new, x_old, arr)
    return project_onto_box_sum(interpolated, new_n / 2.0)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count("1") & 1)
    return seq

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial base vector
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    # grid schedule
    grids = [128, 256, 512]
    fractions = {128: 0.15, 256: 0.35, 512: 0.45}
    best_h = base.copy()
    best_score = compute_c5(best_h)

    total_budget = budget_s

    for n in grids:
        if time_left() <= 0.1:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        alloc = fractions[n] * total_budget
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        cand_h, cand_score = simulated_annealing(best_h, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        if best_score <= 0.3808:
            break

    # final refinement with remaining time
    remaining = time_left()
    if remaining > 2.0:
        # try a short SA then greedy refinement
        extra_h, extra_score = simulated_annealing(best_h, min(5.0, remaining / 2.0), rng)
        if extra_score < best_score:
            best_score = extra_score
            best_h = extra_h.copy()
        remaining = time_left()
        if remaining > 1.0:
            best_h, best_score = final_refinement(best_h, rng, remaining)

    # final projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
