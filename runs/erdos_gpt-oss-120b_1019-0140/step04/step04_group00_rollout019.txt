def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n/2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left()>0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s
```

def binary_swap_anneal(h0, rng, time_left, max_iters=400000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = 0.2
    T_min = 1e-5
    decay = 0.99998
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s
```

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # find lambda by binary search
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    # expand bounds if necessary
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n); term1[:n_s] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n); term1[s:] = h[:n_s]
        term2 = np.zeros(n); term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def binary_swap_anneal(h0, rng, time_left, max_iters=400000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = 0.2
    T_min = 1e-5
    decay = 0.99998
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def thue_morse_binary(n):
    # generate using bit parity
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_top_half(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    hb = np.zeros_like(h)
    hb[idx] = 1.0
    return hb

    candidates.append(thue_morse_binary(n_small))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(3):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, target_sum_small))

    best_h_small = None
    best_s_small = np.inf

    # quick gradient descent on each candidate
    for cand in candidates:
        if time_left() < 2.0:
            break
        h_opt, s_opt = gradient_descent_momentum(cand, time_left,
                                                 step0=0.5, beta=0.9,
                                                 max_iter=20000)
        if s_opt < best_s_small:
            best_s_small = s_opt
            best_h_small = h_opt.copy()
        # early exit if already good
        if best_s_small <= 0.3808:
            return best_h_small, best_s_small, n_small

    if best_h_small is None:

        best_s_small = compute_c5_fast(best_h_small)

    # upscale to larger resolution if time permits
    best_h = best_h_small
    best_s = best_s_small
    n_best = n_small

    if time_left() > 60.0:

        target_sum_big = n_big / 2.0
        h_big = resample(best_h, n_big)
        h_big = project_onto_box_sum(h_big, target_sum_big)
        # long gradient descent
        h_opt_big, s_opt_big = gradient_descent_momentum(
            h_big, time_left,
            step0=0.4, beta=0.9,
            max_iter=200000)
        # optional binary refinement
        h_bin = binarize_top_half(h_opt_big)
        s_bin = compute_c5_fast(h_bin)
        if s_bin < s_opt_big:
            h_opt_big, s_opt_big = h_bin, s_bin
        # annealing on binary if time left
        if time_left() > 5.0:
            h_ann, s_ann = binary_swap_anneal(h_opt_big, rng, time_left,
                                              max_iters=400000)
            if s_ann < s_opt_big:
                h_opt_big, s_opt_big = h_ann, s_ann
        # keep best
        if s_opt_big < best_s:
            best_s = s_opt_big
            best_h = h_opt_big
            n_best = n_big

    # final projection and binary check
    best_h = project_onto_box_sum(best_h, best_h.shape[0] / 2.0)
    best_s = compute_c5_fast(best_h)

    # try binary version of final continuous solution
    h_bin_final = binarize_top_half(best_h)
    s_bin_final = compute_c5_fast(h_bin_final)
    if s_bin_final < best_s:
        best_h, best_s = h_bin_final, s_bin_final

    return best_h, best_s, n_best
