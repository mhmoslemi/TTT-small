            return best_h, best_c, n

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50, tol=1e-12):
    """Project v onto the intersect of box [lo, hi]^n and affine sum=v."""
    n = v.size
    # Trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # Initialize lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # Helper to compute sum after clipping with a given lambda
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Expand bounds until they bracket s
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    # Bisection
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) <= tol:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    """Return max overlap C5 for step function represented by h."""
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    """Gradient of overlap w.r.t. h for a specific shift achieving the max."""
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n); term1[:n_s] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n); term1[s:] = h[:n_s]
        term2 = np.zeros(n); term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    """Gradient averaged over all shifts achieving the current max overlap."""
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    """Projected gradient descent with momentum on continuous h."""
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        c_new = compute_c5_fast(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c

def binary_swap_anneal(h0, rng, time_left, max_iters=800000,
                      T0=1.0, T_min=1e-7, decay=0.999995):
    """Simulated annealing on binary vectors using random swaps."""
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_c = h.copy(), cur
    T = T0
    n = h.shape[0]
    it = 0
    while it < max_iters and time_left() > 0.0:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or h[i] == h[j]:
            continue
        # perform swap
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_c:
                best_c, best_h = new, h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_c

def thue_morse_binary(n):
    """Generate Thue–Morse sequence of length n (0/1)."""
    # vectorized popcount parity
    i = np.arange(n, dtype=np.uint32)
    parity = np.bitwise_xor.reduce(np.unpackbits(i[:, None].view(np.uint8), axis=1), axis=1) & 1
    return 1.0 - parity.astype(float)  # 0 -> 1, 1 -> 0 gives binary pattern

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def resample(arr, new_n):
    """Resample array arr (assumed periodic on [0,1)) to new_n points."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
