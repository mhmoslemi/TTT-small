import time
import numpy as np

def upsample(arr, new_n):
    """Linearly upsample a 1‑D array to length new_n."""
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    """Exact C5 for a binary (or real) step function h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # --- target resolution -------------------------------------------------
    target_n = 512                     # finer grid gives tighter bound
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.size != target_n:
            base = upsample(base, target_n)
    else:
        base = rng.random(target_n)    # random start if no parent construction

    # --- binary initialization (exactly half ones) -------------------------
    half = target_n // 2
    idx = np.argpartition(base, -half)[-half:]   # indices of the largest half
    h = np.zeros(target_n, dtype=float)
    h[idx] = 1.0

    # --- pre‑compute correlation vector ------------------------------------
    n = target_n
    L = 2 * n - 1
    offset = n - 1
    factor = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    current_score = np.max(c) * factor

    best_h = h.copy()
    best_score = current_score

    # --- simulated annealing (pairwise swaps) -----------------------------
    T0 = 0.003                     # initial temperature
    delta = np.empty(L, dtype=float)

    while time_left() > 0.5:
        # pick a random 1‑position i and a random 0‑position j
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)

        # compute correlation update caused by swapping i↔j
        delta.fill(0.0)
        arr = 1.0 - h

        # term from corr(delta_h, arr)
        sj = offset - j
        si = offset - i
        delta[sj:sj + n] += arr
        delta[si:si + n] -= arr

        # term from -corr(h, delta_h)
        delta[j:j + n] -= h
        delta[i:i + n] += h

        # term from -corr(delta_h, delta_h)
        delta[offset] -= 2.0
        shift = i - j
        delta[offset + shift] += 1.0
        delta[offset - shift] += 1.0

        # new correlation and score
        new_c = c + delta
        new_score = np.max(new_c) * factor

        # temperature (linear cooling)
        T = T0 * max(0.0, time_left() / budget_s)

        # Metropolis acceptance
        if new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / (T + 1e-12)):
            # accept move
            c = new_c
            h[i] = 0.0
            h[j] = 1.0
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= 0.3808:
                    break

    # final accurate evaluation
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
