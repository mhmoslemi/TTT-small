def continuous_refinement(h, time_limit, rng):
    n = h.size
    cur = h.copy()
    cur_score = compute_c5(cur)
    best_h = cur.copy()
    best_score = cur_score

        frac = elapsed / time_limit
        temp = T0 * (1 - frac)

        if j >= i:

        cur[j] += delta

            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1

import time
import numpy as np

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - total) < 1e-12:
            return proj
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample_continuous(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def adjust_to_balance(arr, rng):
    """Force binary array to have exactly n/2 ones."""
    n = arr.size
    target = n // 2
    current = int(arr.sum())
    needed = target - current
    if needed > 0:
        zeros = np.where(arr == 0)[0]
        if zeros.size >= needed:
            chosen = rng.choice(zeros, size=needed, replace=False)
            arr[chosen] = 1.0
    elif needed < 0:
        ones = np.where(arr == 1)[0]
        if ones.size >= -needed:
            chosen = rng.choice(ones, size=-needed, replace=False)
            arr[chosen] = 0.0
    return arr

def binary_sa(h0, time_limit, rng):
    """Simulated annealing on binary vectors with swaps."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    max_iters = int(5e7)
    T0 = 0.02
    it = 0
    while it < max_iters and time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = T0 * (1.0 - frac)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            it += 1
            continue
        h[i], h[j] = h[j], h[i]  # swap
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        elif temp > 0:
            if rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= 0.3808:
                    break
        else:
            h[i], h[j] = h[j], h[i]  # revert
        it += 1
    return best_h, best_score

def continuous_refinement(h0, time_limit, rng):
    """Small mass‑transfer local search on continuous vectors."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    max_iters = int(5e7)
    T0 = 0.004
    it = 0
    while it < max_iters and time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = T0 * (1.0 - frac)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        delta_sc = new_score - cur_score
        accept = False
        if delta_sc < 0:
            accept = True
        elif temp > 0:
            if rng.random() < np.exp(-delta_sc / temp):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= 0.3808:
                    break
        else:
            h[i] += delta
            h[j] -= delta
        it += 1
    return best_h, best_score
