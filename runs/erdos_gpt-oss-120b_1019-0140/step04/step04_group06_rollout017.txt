def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import math
import numpy as np

# ---------- helper functions ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project vector v onto the box [lo,hi]^n while fixing sum(v)=s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure the root bracket
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    v[idx] = 1.0
    return v

def repair_binary(v, n, rng):
    """Force binary 0/1 and exact sum n/2."""
    target = n // 2
    # round to binary
    v_bin = (v >= 0.5).astype(float)
    cur = int(v_bin.sum())
    if cur > target:
        ones_idx = np.where(v_bin == 1)[0]
        rng.shuffle(ones_idx)
        v_bin[ones_idx[:cur - target]] = 0.0
    elif cur < target:
        zeros_idx = np.where(v_bin == 0)[0]
        rng.shuffle(zeros_idx)
        v_bin[zeros_idx[:target - cur]] = 1.0
    return v_bin

def tournament_select(pop, scores, rng, size=3):
    """Select one individual from pop using tournament."""
    candidates = rng.choice(len(pop), size=size, replace=False)
    best = candidates[0]
    best_score = scores[best]
    for c in candidates[1:]:
        if scores[c] < best_score:
            best = c
            best_score = scores[c]
    return pop[best]

def crossover(p1, p2, rng):
    """Single-point crossover."""
    n = len(p1)
    point = rng.integers(1, n)  # 1 .. n-1
    child = np.concatenate([p1[:point], p2[point:]]).astype(float)
    return child

def mutate(v, rng, rate):
    """Flip each bit with probability rate."""
    mask = rng.random(v.shape) < rate
    v[mask] = 1.0 - v[mask]
    return v

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, time_left):
    """Projected gradient descent on the continuous problem."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

    # Target resolution (try to improve within 256 points)

    # Track global best
    best_idx = int(np.argmin(pop_scores))
    best_h = population[best_idx].copy()
    best_score = pop_scores[best_idx]

    # GA loop
    crossover_prob = 0.8
    mutation_rate = 0.02
    tournament_size = 3

    while time_left() > 6.0:

        children = []
        child_scores = []
        for _ in range(pop_size):

            p2 = tournament_select(population, pop_scores, rng, tournament_size)
            child = p1.copy()

            child = mutate(child, rng, mutation_rate)
            child = repair_binary(child, n, rng)
            sc = compute_c5(child)
            children.append(child)
            child_scores.append(sc)

        combined = population + children
        combined_scores = pop_scores + child_scores
        idx_sorted = np.argsort(combined_scores)
        population = [combined[i] for i in idx_sorted[:pop_size]]
        pop_scores = [combined_scores[i] for i in idx_sorted[:pop_size]]

        # update global best
        if pop_scores[0] < best_score:
            best_score = pop_scores[0]
            best_h = population[0].copy()
            if best_score <= 0.3808:

        j = rng.choice(zeros)

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
