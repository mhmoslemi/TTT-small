def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000, T0=0.5, decay=0.99997, Tmin=1e-6):
    h = h0.copy()
    # Ensure binary (0/1) and sum = n/2
    h = (h > 0.5).astype(float)
    n = h.size
    # ensure correct sum - if not, adjust by converting some.
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    k = n // 2
    if ones.size > k:
        # turn excess ones into zeros (choose random)
        excess = ones.size - k
        idx_rm = rng.choice(ones, size=excess, replace=False)
        h[idx_rm] = 0
    elif zeros.size > k:
        excess = zeros.size - k
        idx_rm = rng.choice(zeros, size=excess, replace=False)
        h[idx_rm] = 1
    # recompute ones and zeros after adjustment
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    # compute current score
    corr = compute_corr_scaled_fft(h)
    current_score = np.max(corr)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap: flip i to 0, j to 1
        h[i] = 0.0
        h[j] = 1.0
        # update arrays (could recalc each time)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        corr = compute_corr_scaled_fft(h)
        new_score = np.max(corr)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
```

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # forward shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

def upsample(arr, new_n):

def binary_swap_anneal(h0, rng, time_left_func, max_iters=1000000, T0=0.5, decay=0.99997, Tmin=1e-6):
    # Ensure binary pattern with correct sum
    h = (h0 > 0.5).astype(float)
    n = h.size
    k = n // 2
    # Adjust sum if needed
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if ones.size > k:
        excess = ones.size - k
        idx_rm = rng.choice(ones, size=excess, replace=False)
        h[idx_rm] = 0.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    elif zeros.size > k:
        excess = zeros.size - k
        idx_rm = rng.choice(zeros, size=excess, replace=False)
        h[idx_rm] = 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    # compute initial score
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        # random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new score
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # update index sets
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # temperature schedule
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def best_binary_from_continuous(h, rng, attempts=5):
    n = h.size
    k = n // 2
    best_sc = np.inf
    best_bin = None
    for _ in range(attempts):
        rot = rng.integers(0, n)
        h_rot = np.roll(h, rot)
        idx = np.argpartition(-h_rot, k)[:k]
        h_bin = np.zeros_like(h_rot)
        h_bin[idx] = 1.0
        sc = compute_c5_fft(h_bin)
        if sc < best_sc:
            best_sc = sc
            best_bin = h_bin.copy()
    return best_bin, best_sc

def gradient_descent_momentum(h0, max_iters, time_left_func):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    momentum = np.zeros_like(h)
    corr = compute_corr_scaled_fft(h)
    best_score = np.max(corr)
    best_h = h.copy()
    target = 0.3808
    for it in range(1, max_iters + 1):
        if time_left_func() <= 0:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        if max_corr < best_score:
            best_score = max_corr
            best_h = h.copy()
            if best_score <= target:
                break
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = 0.9 * momentum + 0.1 * grad
        eta = 0.5 / np.sqrt(it)
        h_new = h - eta * momentum
        h = project_onto_box_sum(h_new, s)
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=1000000, T0=0.5, decay=0.99997, Tmin=1e-6):
    h = (h0 > 0.5).astype(float)
    n = h.size
    k = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if ones.size > k:
        excess = ones.size - k
        idx_rm = rng.choice(ones, size=excess, replace=False)
        h[idx_rm] = 0.0
    elif zeros.size > k:
        excess = zeros.size - k
        idx_rm = rng.choice(zeros, size=excess, replace=False)
        h[idx_rm] = 1.0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
