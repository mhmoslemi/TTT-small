import time, numpy as np, math

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n: return arr.copy()
    old_x = np.linspace(0.0, 1.0, m, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, arr)

def binary_random_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def hill_climb_binary(h, max_iters, rng):
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    for _ in range(max_iters):
        # pick random swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones)==0 or len(zeros)==0: break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        if new_score < cur_score or rng.random() < 0.01:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

def simulated_annealing_binary(h0, rng, time_left):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    start = time.time()
    # schedule: T0 set as maybe 0.02 * cur_score
    T0 = 0.02 * cur_score + 1e-3
    # total allowed time ~ time_left
    while time_left() > 0.1:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed)
        T = T0 * (0.001 ** frac)  # exponential cooling.
        # random swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

def project_to_binary(h):
    # enforce exactly half ones: set top half values to 1
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    out = np.zeros_like(h)
    out[idx] = 1.0
    return out

def real_gradient_descent(h0, n_iters, step0, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iters + 1):
        if time_left() <= 0:
            break
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        # gradient for max shift
        grad = np.zeros_like(h)
        if shift >= 0:
            grad[:n - shift] = 1 - h[shift:]
            grad[shift:] += -h[:n - shift]
        else:
            sft = -shift
            grad[sft:] = 1 - h[:n - sft]
            grad[:n - sft] += -h[sft:]
        # momentum
        momentum = 0.9 * momentum + 0.1 * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        # clip to [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)
        # simple sum correction (optional)
        # compute current sum
        s_target = n / 2.0
        diff = s_target - h_new.sum()
        if abs(diff) > 1e-8:
            mask = (h_new > 0) & (h_new < 1)
            nfree = mask.sum()
            if nfree > 0:
                h_new[mask] += diff / nfree
                h_new = np.clip(h_new, 0.0, 1.0)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    # final projection to binary
    h_bin = project_to_binary(h)
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        return h_bin, sc_bin
    else:
        return best_h, best_score

        base = upsample(base, target_n)
    base = project_to_binary(base)

    best_h, best_score = base.copy(), compute_c5(base)

    while time_left() > 5:
        if time_left() > 200:
            h0 = binary_random_balanced(target_n, rng)
            iters = int(min(10000, time_left() * 10))
            h_opt, sc_opt = hill_climb_binary(h0, iters, rng)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
        else:
            h_opt, sc_opt = simulated_annealing_binary(best_h, rng, time_left)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
            if time_left() > 30:
                n_iter = int(min(20000, time_left() * 5))
                h_opt2, sc_opt2 = real_gradient_descent(best_h, n_iter, 0.5, rng, time_left)
                if sc_opt2 < best_score:
                    best_h, best_score = h_opt2.copy(), sc_opt2
                best_h = project_to_binary(best_h)
                best_score = compute_c5(best_h)
        if best_score <= 0.3808:
            break

    best_h = project_to_binary(best_h)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
