def compute_grad_smooth(h, alpha):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def adam_optimize_level(h0, n_iters, alpha_start, alpha_max, step0, rng, time_left, idx_plus, idx_minus):
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iters+1):
        if time_left() <= 0:
            break
        # Compute correlation (scaled)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        # compute weights
        alpha = alpha_start + (alpha_max - alpha_start) * t / n_iters
        w = np.exp(alpha * corr)
        w_sum = w.sum()
        w_norm = w / w_sum
        # gradient via precomputed indices
        term = (1 - h[idx_plus]) - h[idx_minus]  # shape (n,2n-1)
        grad = dx * (term @ w_norm)  # vector
        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        # step size
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        h_new = h - eta * m_hat
        # project to constraints
        h_new = project_onto_box_sum(h_new, n/2.0)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type':'eq','fun':eq_sum},
            {'type':'ineq','fun':ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def adam_optimize_level(h0, max_iter, alpha_start, alpha_max, step0, rng, time_left, idx_plus, idx_minus):
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, max_iter+1):
        if time_left() <= 0:
            break
        # correlation scaled
        corr = np.correlate(h, 1 - h, mode='full') * dx
        # schedule alpha
        alpha = alpha_start + (alpha_max - alpha_start) * t / max_iter
        w = np.exp(alpha * corr)
        w_sum = w.sum()
        w_norm = w / w_sum
        # gradient via matrix multiplication
        term = (1 - h[idx_plus]) - h[idx_minus]  # shape (n,2n-1)
        grad = dx * (term @ w_norm)  # shape (n,)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        eta = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - eta * m_hat
        # Project
        h_new = project_onto_box_sum(h_new, n/2.0)
        # Evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

    eps = 1e-8
    for t in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        alpha = alpha_start + (alpha_max - alpha_start) * t / max_iter
        w = np.exp(alpha * corr)
        w_norm = w / w.sum()
        term = (1 - h[idx_plus]) - h[idx_minus]      # (n,2n-1)
        grad = dx * (term @ w_norm)                  # (n,)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - eta * m_hat
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 64
        base = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        base[idx] = 1.0

    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    sizes = [64, 128, 256, 512]
    iter_schedule = {64: 8000, 128: 4000, 256: 2000, 512: 2000}

    for n_target in sizes:
        if time_left() <= 5:
            break
        if base.shape[0] != n_target:
            base = upsample(base, n_target)
        base = project_onto_box_sum(base, n_target / 2.0)
        sc = compute_c5(base)
        if sc < best_score:
            best_h, best_score = base.copy(), sc

        idx_plus, idx_minus = precompute_roll_indices(n_target)
        max_iter = min(iter_schedule.get(n_target, 1000), int(time_left() * 5))
        if max_iter <= 0:
            continue
        restarts = 3 if n_target >= 256 else 2
        step0 = 0.5 if n_target <= 256 else 0.3

        for _ in range(restarts):
            h0 = base + rng.normal(scale=0.05, size=n_target)
            h0 = project_onto_box_sum(h0, n_target / 2.0)
            h_opt, sc_opt = adam_optimize_level(
                h0, max_iter, alpha_start=2.0, alpha_max=20.0,
                step0=step0, time_left=time_left,
                idx_plus=idx_plus, idx_minus=idx_minus)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
        base = best_h.copy()

    if time_left() > 5:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=500)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    if time_left() > 5:
        n_final = best_h.size
        idx_plus, idx_minus = precompute_roll_indices(n_final)
        max_iter = min(2000, int(time_left() * 5))
        h_opt, sc_opt = adam_optimize_level(
            best_h, max_iter,
            alpha_start=5.0, alpha_max=30.0,
            step0=0.2, time_left=time_left,
            idx_plus=idx_plus, idx_minus=idx_minus)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    while time_left() > 2:
        n_final = best_h.size
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.01, 0.5)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new, sc_new
        if rng.random() < 0.3:
            pert = rng.normal(scale=0.02, size=n_final)
            h_new = best_h + pert
            h_new = project_onto_box_sum(h_new, n_final / 2.0)
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new, sc_new
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
