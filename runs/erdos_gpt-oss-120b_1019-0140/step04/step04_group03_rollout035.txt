import time
import numpy as np

def upsample(arr, new_n):
    """Linear up‑sampling of a 1‑D array to length new_n."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (n a power of two)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=np.int8, count=n)

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """Box [lo,hi] projection with exact sum constraint via bisection."""
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C₅ value for a (float) step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_cross_corr(h):
    """Integer self‑correlation of a binary vector."""
    return np.correlate(h, h, mode='full')

def delta_cross_swap(i, j, h, delta):
    """Fill delta = cross_corr(new) - cross_corr(old) for swapping
    a 1 at i with a 0 at j (binary h)."""
    n = h.shape[0]
    delta.fill(0)
    # -h[i + s]
    delta[:n - i] -= h[i:]
    # +h[j + s]
    delta[:n - j] += h[j:]
    # -h[i - s]  (use reversed view)
    delta[i:i + n] -= h[::-1]
    # +h[j - s]
    delta[j:j + n] += h[::-1]
    # constant terms
    delta[n - 1] += 2                     # s = 0
    diff = j - i
    delta[diff + n - 1] -= 1              # s = j-i
    delta[-diff + n - 1] -= 1             # s = i-j
    return delta

def hill_climb_binary(h0, time_budget, rng):
    """Binary hill climbing that maximises the minimal self‑correlation."""
    n = h0.shape[0]
    cross = compute_cross_corr(h0)
    cur_min = cross.min()
    cur_h = h0.copy()

    # maintain index lists and position maps
    ones = list(np.where(cur_h == 1)[0])
    zeros = list(np.where(cur_h == 0)[0])
    pos_one = np.empty(n, dtype=np.int32)
    pos_zero = np.empty(n, dtype=np.int32)
    for idx, v in enumerate(ones):
        pos_one[v] = idx
    for idx, v in enumerate(zeros):
        pos_zero[v] = idx

    best_h = cur_h.copy()
    best_min = cur_min
    delta = np.empty(2 * n - 1, dtype=np.int32)

    start = time.time()
    no_imp = 0
    RESTART = 20000

    while time.time() - start < time_budget:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        delta_cross_swap(i, j, cur_h, delta)
        new_cross = cross + delta
        new_min = new_cross.min()
        if new_min > cur_min:
            # accept swap
            cur_h[i] = 0
            cur_h[j] = 1
            cross = new_cross
            cur_min = new_min
            # update index structures
            pi = pos_one[i]
            pj = pos_zero[j]
            zeros[pj] = i
            pos_zero[i] = pj
            ones[pi] = j
            pos_one[j] = pi
            no_imp = 0
            if new_min > best_min:
                best_min = new_min
                best_h = cur_h.copy()
        else:
            no_imp += 1

        if no_imp >= RESTART:
            # restart from best solution (fast reset)
            cur_h = best_h.copy()
            cross = compute_cross_corr(cur_h)
            cur_min = cross.min()
            ones = list(np.where(cur_h == 1)[0])
            zeros = list(np.where(cur_h == 0)[0])
            for idx, v in enumerate(ones):
                pos_one[v] = idx
            for idx, v in enumerate(zeros):
                pos_zero[v] = idx
            no_imp = 0

    return best_h, best_min

def compute_gradients_for_shifts(h, shifts):
    """Gradient for the max‑overlap objective (continuous version)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    """Projected momentum gradient descent with a hard time limit."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    n = 512                     # primary resolution
    # ----- initialise base binary candidate -----
    if 'initial_h_values' in globals():
        base = upsample(np.asarray(initial_h_values, dtype=float), n)
        base = (base > 0.5).astype(np.int8)
    else:
        base = thue_morse(n)

    # ensure exact half‑ones
    diff = int(base.sum() - n // 2)
    if diff > 0:
        idx = np.where(base == 1)[0][:diff]
        base[idx] = 0
    elif diff < 0:
        idx = np.where(base == 0)[0][: -diff]
        base[idx] = 1

    # ----- binary hill climbing (major part of budget) -----
    elapsed = time.time() - start_total
    binary_budget = max(0.0, budget_s - elapsed - 10.0)  # reserve ~10 s for refinement
    best_bin, best_min = hill_climb_binary(base, binary_budget, rng)
    best_score = 1.0 - (2.0 * best_min) / n        # C₅ from min cross‑correlation
    best_h = best_bin.astype(float)

    # ----- optional continuous refinement if time permits -----
    remaining = budget_s - (time.time() - start_total)
    if remaining > 5.0:
        # start from the best binary vector
        h0 = best_h.copy()
        # a modest number of iterations, limited by remaining time
        max_iters = int(min(200000, max(50000, remaining * 50000.0 / n)))
        refined_h, refined_score = gradient_descent_momentum_custom(
            h0, n_iter=max_iters, step0=0.2, beta=0.9,
            rng=rng, max_time=remaining)
        # keep the better of binary and refined solutions
        if refined_score < best_score:
            best_score = refined_score
            best_h = refined_h

    # final projection to guarantee constraints
    best_h = project_onto_box_sum(best_h, n / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
