def compute_gradient_softmax(h, alpha):
    corr = compute_corr_scaled_fft(h)  # shape (2*n-1)
    # compute softmax weights
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    w = exp_vals / np.sum(exp_vals)
    # compute FFT of w and wr
    # But need w and w_rev in arrays of length 2*n (M) as described.
    n = h.size
    M = 2 * n
    w_padded = np.zeros(M, dtype=np.float64)
    w_padded[:2*n-1] = w
    w_rev_padded = np.zeros(M, dtype=np.float64)
    w_rev_padded[:2*n-1] = w[::-1]
    H = np.fft.rfft(h, n=M)
    W = np.fft.rfft(w_padded, n=M)
    Wr = np.fft.rfft(w_rev_padded, n=M)
    conv = np.fft.irfft(H * W, n=M)
    conv_rev = np.fft.irfft(H * Wr, n=M)
    # slice
    start = n - 1
    grad = 1.0 - (conv_rev[start:start+n] + conv[start:start+n])
    # return gradient
    return grad

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the interval brackets the target sum
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Exact C5 = max_{k} ∫ h(x)(1-h(x+k)) dx via FFT."""

    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)
