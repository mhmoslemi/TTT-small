        size = 1 << (int(np.ceil(np.log2(L + n - 1))))

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def smooth_weights(corr_raw, alpha):
    """Stable soft‑max weights for smoothing max."""
    max_corr = corr_raw.max()
    exp_vals = np.exp(alpha * (corr_raw - max_corr))
    sum_exp = exp_vals.sum()
    return exp_vals / sum_exp


def gradient_smooth_max(h, w):
    """Gradient of the smooth‑max objective."""
    n = h.size
    L = 2 * n - 1
    size = 1 << (int(np.ceil(np.log2(L + n - 1))))
    w_fft = np.fft.fft(w, size)
    w_rev_fft = np.fft.fft(w[::-1], size)
    h_fft = np.fft.fft(h, size)
    one_m_h_fft = np.fft.fft(1.0 - h, size)
    conv_w = np.real(np.fft.ifft(w_fft * h_fft, size))
    conv_rev = np.real(np.fft.ifft(w_rev_fft * one_m_h_fft, size))
    start = n - 1
    return conv_rev[start:start + n] - conv_w[start:start + n]

    best_score = compute_c5(best_h)

    # Adam parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    eta = 0.5
    alpha = 1.0
    alpha_max = 20.0
    it = 0

    # main optimisation loop
    while time.time() - start < budget_s * 0.95:
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        w = smooth_weights(corr_raw, alpha)
        grad = gradient_smooth_max(h, w)

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        h_new = h - eta * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target_n / 2)

        if it % 200 == 0:
            cur_score = compute_c5(h_new)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_new.copy()
                if alpha < alpha_max:
                    alpha = min(alpha_max, alpha * 1.1)
            # occasional random jitter if stagnant
            if it % 2000 == 0 and cur_score > best_score * 1.0005:
                noisy = h_new + rng.normal(scale=0.01, size=target_n)
                noisy = project_onto_box_sum(noisy, target_n / 2)
                noisy_score = compute_c5(noisy)
                if noisy_score < best_score:
                    best_score = noisy_score
                    best_h = noisy.copy()
                    h_new = noisy

        h = h_new
        if best_score <= 0.3808:
            break

    # final local refinement with simple transfers
    while time.time() - start < budget_s * 0.99:
        n = best_h.size
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            cand = best_h.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = compute_c5(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand

        # occasional binary restart
        if rng.random() < 0.01:
            cand = random_balanced_binary(n, rng)
            cand_score = compute_c5(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand

    best_h = project_onto_box_sum(best_h, best_h.size / 2)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
