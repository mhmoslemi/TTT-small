def hill_climb_binary(h0, rng, time_left, max_iters=10**9):

    h = h0.copy()

    current_score = compute_c5_fft(h)

    best_h = h.copy()
    best_score = current_score

    it = 0

    while it < max_iters and time_left() > 0.01:

        # find worst shift(s)
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # If max_corr is already best we have no improvement potential but still may need to check other shifts; but we have current_score = max_corr; it's equal.

        # find shift(s) achieving max_corr
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shift = idxes[rng.integers(len(idxes))] - (h.size - 1)

        # compute mismatch sets
        n = h.size
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            s = -shift
            i_range = np.arange(s, n)
            j_range = i_range - s

        hi = h[i_range]
        hj = h[j_range]

        mask_A = (hi == 1) & (hj == 0)
        mask_B = (hi == 0) & (hj == 1)

        if mask_A.any() and mask_B.any():
            # pick random idx from each mask
            i1 = rng.choice(i_range[mask_A])
            i2 = rng.choice(i_range[mask_B])
            # swap
            h[i1], h[i2] = h[i2], h[i1]
        else:
            # fallback: random swap of any 1 and 0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i1 = rng.choice(ones)
            i2 = rng.choice(zeros)
            h[i1], h[i2] = h[i2], h[i1]

        new_score = compute_c5_fft(h)

        if new_score < current_score - 1e-12:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i1], h[i2] = h[i2], h[i1]
        it += 1

    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # expand bounds if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def hill_climb_binary(h0, rng, time_left):
    h = h0.copy()
    n = h.size
    corr = compute_corr_scaled_fft(h)
    cur_score = np.max(corr)
    best_h = h.copy()
    best_score = cur_score
    stagnation = 0
    while time_left() > 0.01:
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            s = -shift
            i_range = np.arange(s, n)
            j_range = i_range - s
        hi = h[i_range]
        hj = h[j_range]
        mask_A = (hi == 1) & (hj == 0)
        mask_B = (hi == 0) & (hj == 1)
        if mask_A.any() and mask_B.any():
            i1 = rng.choice(i_range[mask_A])
            i2 = rng.choice(i_range[mask_B])
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i1 = rng.choice(ones)
            i2 = rng.choice(zeros)
        h[i1], h[i2] = h[i2], h[i1]
        corr = compute_corr_scaled_fft(h)
        new_score = np.max(corr)
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            stagnation = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i1], h[i2] = h[i2], h[i1]
            stagnation += 1
            if stagnation >= 3000:
                break
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_scaled_fft(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    n = len(initial_h_values) if 'initial_h_values' in globals() else 512

    candidates = []
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        k = n // 2
        idx = np.argpartition(-init_arr, k)[:k]
        h_bin = np.zeros_like(init_arr)
        h_bin[idx] = 1.0
        candidates.append(h_bin)

    tm = thue_morse(n)
    if tm.sum() != n / 2:
        tm = project_onto_box_sum(tm, n / 2.0)
    candidates.append(tm)

    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))

    best_h = None
    best_score = np.inf

    for cand in candidates:
        if time_left() < 0.5:
            break
        h_opt, sc_opt = hill_climb_binary(cand, rng, time_left)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
        if best_score <= target:
            return best_h, best_score, n
        if time_left() > 5.0:
            h_ref, sc_ref = slsqp_optimize_candidate(best_h, max_iter=200)
            if sc_ref < best_score:
                best_h, best_score = h_ref, sc_ref
            if best_score <= target:
                return best_h, best_score, n

    while time_left() > 10.0:
        h_rand = random_binary_balanced(n, rng)
        h_opt, sc_opt = hill_climb_binary(h_rand, rng, time_left)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
        if best_score <= target:
            break
        if time_left() > 5.0:
            h_ref, sc_ref = slsqp_optimize_candidate(best_h, max_iter=200)
            if sc_ref < best_score:
                best_h, best_score = h_ref, sc_ref
            if best_score <= target:
                break

    if best_h is None:
        best_h = random_binary_balanced(n, rng)
    best_h = best_h.astype(float)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, n
