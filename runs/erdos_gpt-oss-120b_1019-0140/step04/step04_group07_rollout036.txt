    # zero pad to next power of 2 maybe to avoid circular convolution? For linear correlation, we can do convolution using FFT of length 2n.
    # For linear correlation h with (1-h): define a = h, b = 1-h.
    # Full correlation can be obtained via FFT of padded arrays length N = 2*n.
    N = 1 << ((2*n-1).bit_length())
    # zero-pad both to length N

def binarize_and_adjust(h):
    n = h.shape[0]
    half = n // 2
    # initial binary via threshold
    binary = (h > 0.5).astype(float)
    cur = int(binary.sum())
    if cur > half:
        # need to reduce number of ones: find indices where h is smallest among those set to 1.
        ones_idx = np.where(binary == 1)[0]
        # sort these indices by h value ascending
        sorted_ones = ones_idx[np.argsort(h[ones_idx])]
        to_zero = sorted_ones[:cur - half]
        binary[to_zero] = 0
    elif cur < half:
        zeros_idx = np.where(binary == 0)[0]
        sorted_zeros = zeros_idx[np.argsort(-h[zeros_idx])]  # highest h among zeros
        to_one = sorted_zeros[:half - cur]
        binary[to_one] = 1
    # ensure exact sum
    return binary

def compute_c5_fft(h):

    # length for full convolution: 2*n - 1
    N = 1 << ((2*n - 1).bit_length())
    # pad h and 1-h
    a = np.fft.rfft(np.concatenate([h, np.zeros(N - n)]))

    corr = np.fft.irfft(a * np.conj(b))
    corr = corr[: 2*n - 1]

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def full_correlation_fft(a, b):
    """Linear correlation a * b (full mode) via FFT; returns array of length 2n-1."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    a_fft = np.fft.rfft(np.concatenate([a, np.zeros(N - n)]))
    b_rev_fft = np.fft.rfft(np.concatenate([b[::-1], np.zeros(N - n)]))
    conv = np.fft.irfft(a_fft * np.conj(b_rev_fft))
    return conv[: 2 * n - 1][::-1]               # reverse to obtain correlation


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = full_correlation_fft(h, 1.0 - h)
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

    h = h0.copy()

    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score

    for it in range(1, max_iters + 1):

        max_corr = np.max(corr)

        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum

        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    # multiscale refinement
    scales = [64, 128, 256, 512]
    if time_left() > 200:
        scales.append(1024)

    for n in scales:
        if time_left() <= 5:
            break
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)

        max_iters = int(min(50000, max(5000, time_left() * (8000.0 / n))))
        h_gd, sc_gd = gradient_descent_adaptive(best_h, max_iters,
                                                step0=0.8, beta=0.9,
                                                rng=rng, time_left=time_left)
        if sc_gd < best_score:
            best_h, best_score = h_gd, sc_gd
        best_h = h_gd

    # binary refinement if time permits
    if time_left() > 10:
        # round to binary while keeping exact sum
        n = best_h.shape[0]
        half = n // 2
        binary_h = (best_h > 0.5).astype(float)
        cur = int(binary_h.sum())
        if cur > half:
            ones_idx = np.where(binary_h == 1)[0]
            to_zero = ones_idx[np.argsort(best_h[ones_idx])][:cur - half]
            binary_h[to_zero] = 0
        elif cur < half:
            zeros_idx = np.where(binary_h == 0)[0]
            to_one = zeros_idx[np.argsort(-best_h[zeros_idx])][:half - cur]
            binary_h[to_one] = 1

        # anneal
        h_bin_best, sc_bin_best = binary_anneal(binary_h, rng, time_left)
        if sc_bin_best < best_score:
            best_h, best_score = h_bin_best, sc_bin_best

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
