        if s >= 0:

            hj = h[s:]
            start_i = 0

            hi = h[s_abs:]
            hj = h[:n-s_abs]

        i_local = np.argmax(contributions)

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse(n):
    """Thue–Morse binary sequence of length n."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def project_to_binary(v, half):
    """Project onto the set of exactly `half` ones (balanced binary)."""
    out = np.zeros_like(v, dtype=float)
    if half <= 0:
        return out
    idx = np.argpartition(v, -half)[-half:]
    out[idx] = 1.0
    return out


def greedy_flow(h, time_limit, rng):
    """Greedy pairwise mass transfer targeting the current worst shift."""
    start = time.time()
    n = h.size
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        s = idx_max - (n - 1)
        if s == 0:
            break  # cannot transfer mass for zero shift
        if s > 0:
            hi = h[:n - s]
            hj = h[s:]
            start_i = 0
        else:
            d = -s
            hi = h[d:]
            hj = h[:n - d]
            start_i = d
        contrib = hi * (1 - hj)
        # best contributing pair
        i_local = np.argmax(contrib)
        i = start_i + i_local
        j = i + s
        max_delta = min(h[i], 1 - h[j])
        if max_delta < 1e-12:
            # try a few other top contributors
            order = np.argsort(-contrib)[:min(10, contrib.size)]
            found = False
            for ii in order:
                i2 = start_i + ii
                j2 = i2 + s
                d2 = min(h[i2], 1 - h[j2])
                if d2 > 1e-12:
                    i, j, max_delta = i2, j2, d2
                    found = True
                    break
            if not found:
                break
        delta = max_delta * rng.uniform(0.2, 0.5)
        h[i] -= delta
        h[j] += delta
    return h


def improve_binary(h, time_limit, rng):
    """Local search on binary vectors using random swaps."""
    start = time.time()
    n = h.size
    half = n // 2
    best_h = h.copy()
    best_score = compute_c5(best_h)
    # maintain ones/zeros indices for quick sampling
    while time.time() - start < time_limit:
        # recompute index lists
        ones = np.flatnonzero(best_h > 0.5)
        zeros = np.flatnonzero(best_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cand = best_h.copy()
        cand[i], cand[j] = 0.0, 1.0
        sc = compute_c5(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand
            if best_score <= 0.3808:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # ---- Initial construction ----
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = project_to_binary(base, base.size // 2)
    else:
        base = thue_morse(64)
        base = project_to_binary(base, base.size // 2)

    # try a few random balanced binaries to possibly beat the parent
    best_h = base
    best_score = compute_c5(best_h)
    for _ in range(4):
        if time_left() <= 0:
            break
        cand = random_balanced_binary(best_h.size, rng)
        sc = compute_c5(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand

    # ---- Multi-resolution improvement ----
    target_sizes = [64, 128, 256, 512]
    for n in target_sizes:
        if time_left() <= 0:
            break
        if best_h.size != n:
            # Upsample and re-binarize
            best_h = upsample(best_h, n)
            best_h = project_to_binary(best_h, n // 2)

        # allocate time: half for greedy flow, half for binary swaps
        alloc = time_left()
        if alloc <= 0.2:
            break
        t_greedy = alloc * 0.5
        t_binary = alloc - t_greedy

        # Greedy continuous refinement
        best_h = greedy_flow(best_h.copy(), t_greedy, rng)

        # Re-binarize
        best_h = project_to_binary(best_h, n // 2)

        # Binary local search
        best_h, cand_score = improve_binary(best_h, t_binary, rng)
        if cand_score < best_score:
            best_score = cand_score

        if best_score <= 0.3808:
            break

    # final projection to guarantee exact constraints
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
