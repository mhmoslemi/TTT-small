import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

    shifts = idx - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            n_s = n - s
            t1 = np.zeros(n); t1[:n_s] = h[s:]
            t2 = np.zeros(n); t2[s:] = h[:n_s]
        else:
            s2 = -s
            n_s = n - s2
            t1 = np.zeros(n); t1[s2:] = h[:n_s]
            t2 = np.zeros(n); t2[:n_s] = h[s2:]
        grad += 1.0 - t1 - t2
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h, best_s = h.copy(), compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def slsqp_optimize_candidate(h0, max_iter=2000):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5_fast(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x): return np.sum(x[:n]) - n / 2.0
    def ineq(x):
        h, t = x[:n], x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    except Exception:
        h_opt = h0
    return h_opt, compute_c5_fast(h_opt)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def smooth_objective_and_grad(h, alpha, dx):
    n = h.size
    # correlation via FFT
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1] * dx
    c_max = corr.max()
    exp_terms = np.exp(alpha * (corr - c_max))
    sum_exp = exp_terms.sum()
    f = c_max + np.log(sum_exp) / alpha
    w = exp_terms / sum_exp          # normalized weights
    w_rev = w[::-1]
    M = 1 << ((3 * n - 2).bit_length())
    fft_w_rev = np.fft.rfft(w_rev, n=M)
    fft_one_h = np.fft.rfft(1.0 - h, n=M)
    fft_h = np.fft.rfft(h, n=M)
    conv1 = np.fft.irfft(fft_w_rev * fft_one_h, n=M)
    conv2 = np.fft.irfft(fft_w_rev * fft_h, n=M)
    start = n - 1
    grad = dx * (conv1[start:start + n] - conv2[start:start + n])
    return f, grad

def smooth_descent(h0, target_sum, time_left, phases):
    n = h0.size
    dx = 2.0 / n
    best_h, best_s = h0.copy(), compute_c5_fast(h0)
    h = h0.copy()
    for alpha, steps, lr0 in phases:
        for it in range(1, steps + 1):
            if time_left() <= 0.0:
                return best_h, best_s
            _, grad = smooth_objective_and_grad(h, alpha, dx)
            lr = lr0 / np.sqrt(it)
            h = project_onto_box_sum(h - lr * grad, target_sum)
            if it % 100 == 0:
                s = compute_c5_fast(h)
                if s < best_s:
                    best_s, best_h = s, h.copy()
                if best_s <= 0.3808:
                    return best_h, best_s
    s = compute_c5_fast(h)
    if s < best_s:
        best_s, best_h = s, h.copy()
    return best_h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))

    n = 384
    target_sum = n / 2.0
    target_c5 = 0.3808
    inits = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        inits.append(project_onto_box_sum(init, target_sum))
    inits.append(thue_morse_binary(n))
    for _ in range(2):
        inits.append(random_binary_balanced(n, rng))
    for _ in range(2):
        v = rng.random(n)
        inits.append(project_onto_box_sum(v, target_sum))

    phases = [(10, 2000, 0.5), (30, 2000, 0.3), (70, 2000, 0.2), (150, 2000, 0.1)]

    best_h, best_s = None, np.inf
    for init in inits:
        if time_left() <= 5.0:
            break
        h_opt, s_opt = smooth_descent(init, target_sum, time_left, phases)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= target_c5:
            return best_h, best_s, n

    while time_left() > 10.0:
        v = rng.random(n)
        v = project_onto_box_sum(v, target_sum)
        h_opt, s_opt = smooth_descent(v, target_sum, time_left, phases)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= target_c5:
            return best_h, best_s, n

    if best_h is not None and time_left() > 5.0:
        h_opt, s_opt = slsqp_optimize_candidate(best_h, max_iter=3000)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= target_c5:
            return best_h, best_s, n

    if best_h is not None and time_left() > 2.0:
        h_ref, s_ref = gradient_descent_momentum(best_h, time_left, step0=0.5, beta=0.9, max_iter=200000)
        if s_ref < best_s:
            best_s, best_h = s_ref, h_ref.copy()
        if best_s <= target_c5:
            return best_h, best_s, n

    best_h = project_onto_box_sum(best_h, target_sum)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s, n
