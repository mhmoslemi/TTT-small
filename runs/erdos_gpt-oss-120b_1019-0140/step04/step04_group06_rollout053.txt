def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_repair(h):
    n = h.shape[0]
    target = n // 2
    # threshold
    bh = (h >= 0.5).astype(int)
    cur = bh.sum()
    if cur > target:
        # need to flip some 1->0
        ones_idx = np.where(bh == 1)[0]
        # randomly select cur-target to flip
        flip = rng.choice(ones_idx, size=cur-target, replace=False)
        bh[flip] = 0
    elif cur < target:
        zeros_idx = np.where(bh == 0)[0]
        flip = rng.choice(zeros_idx, size=target-cur, replace=False)
        bh[flip] = 1
    return bh.astype(float)

import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a binary or continuous step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear up‑sampling of a 1‑D array to length new_n (periodic domain [0,2))."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_repair(h, rng):
    """Project onto the set of binary vectors with exactly half ones."""
    n = h.shape[0]
    target = n // 2
    bh = (h >= 0.5).astype(int)
    cur = bh.sum()
    if cur > target:
        ones = np.where(bh == 1)[0]
        flip = rng.choice(ones, size=cur - target, replace=False)
        bh[flip] = 0
    elif cur < target:
        zeros = np.where(bh == 0)[0]
        flip = rng.choice(zeros, size=target - cur, replace=False)
        bh[flip] = 1
    return bh.astype(float)

def greedy_improve(h, time_left, rng, target_D):
    """Greedy local improvement: swap bits to reduce the maximal overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_h = h.copy()
    no_improve = 0
    max_no_improve = 5000

    while time_left() > 0:
        if best_score <= target_D:
            break

        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_corr = corr.max()
        cur_score = max_corr * dx

        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # pick a shift attaining the maximum
        shift_idxs = np.where(corr == max_corr)[0]
        shift_idx = rng.choice(shift_idxs)
        k = shift_idx - (n - 1)  # actual shift (can be negative)

        # find a 1‑bit that can be moved forward/backward to a 0‑bit at distance k
        idx = np.arange(n)
        j = idx + k
        valid = (j >= 0) & (j < n)
        candidates = idx[valid & (cur_h[idx] == 1) & (cur_h[j] == 0)]

        if candidates.size > 0:
            i = rng.choice(candidates)
            j = i + k
            cur_h[i], cur_h[j] = 0.0, 1.0
        else:
            # fallback: random 1↔0 swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur_h[i], cur_h[j] = 0.0, 1.0

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    target_D = 0.3808

    # initialise base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 64
        base = np.zeros(n0, dtype=float)
        base[rng.choice(n0, size=n0 // 2, replace=False)] = 1.0

    # resolution schedule
    resolutions = [64, 128, 256, 512]
    best_h = None
    best_score = float('inf')

    for n in resolutions:
        if time_left() < 5.0:
            break

        # up‑sample to the current resolution
        if base.shape[0] != n:
            base = upsample(base, n)

        # enforce binary and exact mass
        base = binary_repair(base, rng)

        # greedy improvement at this resolution
        base, score = greedy_improve(base, time_left, rng, target_D)

        if score < best_score:
            best_score = score
            best_h = base.copy()

        if best_score <= target_D:
            break

    # final evaluation
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.shape[0]
