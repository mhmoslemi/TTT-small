def compute_delta_swap(h, i, j, shift, zero_idx):
    n = h.size
    delta = np.zeros(2*n-1, dtype=int)
    # A mask
    mask_A = (i + shift >= 0) & (i + shift < n)
    idx_A = i + shift[mask_A]
    delta[mask_A] -= (1 - h[idx_A])

    # B mask
    mask_B = (j + shift >= 0) & (j + shift < n)
    idx_B = j + shift[mask_B]
    vals_B = 1 - h[idx_B]
    # handle idx_B == i
    eq_i = (idx_B == i)
    if np.any(eq_i):
        vals_B[eq_i] = 1
    # handle idx_B == j
    eq_j = (idx_B == j)
    if np.any(eq_j):
        vals_B[eq_j] = 0
    delta[mask_B] += vals_B

    # C mask
    mask_C = (i - shift >= 0) & (i - shift < n)
    idx_C = i - shift[mask_C]
    # note: for shift=0, idx_C = i; this would give delta +h[i]; will be corrected later
    delta[mask_C] += h[idx_C]

    # D mask
    mask_D = (j - shift >= 0) & (j - shift < n)
    idx_D = j - shift[mask_D]
    delta[mask_D] -= h[idx_D]

    # correct double count at shift s1=j-i
    idx1 = (j - i) + n - 1
    delta[idx1] += 1

    # ensure zero shift zero
    delta[zero_idx] = 0

    return delta

def repair_binary(h, target, rng):
    n = h.size
    cur = int(h.sum())
    if cur == target:
        return h
    diff = cur - target
    if diff > 0:
        # too many ones, flip diff ones to zero
        ones_idx = np.flatnonzero(h)
        flip = rng.choice(ones_idx, diff, replace=False)
        h[flip] = 0
    else:
        zeros_idx = np.flatnonzero(1 - h)
        flip = rng.choice(zeros_idx, -diff, replace=False)
        h[flip] = 1
    return h

def upsample_binary(h, new_n, rng):
    old_n = h.size
    if new_n == old_n:
        return h.copy()
    # Interpolate linearly
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    h_float = np.interp(x_new, x_old, h.astype(float))
    h_bin = (h_float > 0.5).astype(np.int8)
    # repair to target count
    target = new_n // 2
    h_bin = repair_binary(h_bin, target, rng)
    return h_bin

import time, math, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_binary(h, target, rng):
    cur = int(h.sum())
    if cur == target:
        return h
    diff = cur - target
    if diff > 0:
        ones = np.flatnonzero(h)
        flip = rng.choice(ones, diff, replace=False)
        h[flip] = 0
    else:
        zeros = np.flatnonzero(1 - h)
        flip = rng.choice(zeros, -diff, replace=False)
        h[flip] = 1
    return h

def upsample_binary(h, new_n, rng):
    old_n = h.size
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    h_float = np.interp(x_new, x_old, h.astype(float))
    h_bin = (h_float > 0.5).astype(np.int8)
    target = new_n // 2
    return repair_binary(h_bin, target, rng)

    target_c5 = 0.3808
