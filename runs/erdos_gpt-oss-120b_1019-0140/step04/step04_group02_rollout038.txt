def weighted_gradient(w, h):
    n = h.shape[0]
    L = 2 * n - 1
    # w is length L
    # compute conv1 = convolution of w_rev with (1 - h)
    # compute conv2 = convolution of w with h

    M = 1 << ((n + L - 1).bit_length())  # next power of two
    # but we can simply use n + L -1 = 3n - 2 ; compute next pow2
    w_rev = w[::-1]
    f_w_rev = np.fft.fft(w_rev, M)
    f_one_minus_h = np.fft.fft(1.0 - h, M)
    conv1 = np.fft.ifft(f_w_rev * f_one_minus_h).real

    f_w = np.fft.fft(w, M)
    f_h = np.fft.fft(h, M)
    conv2 = np.fft.ifft(f_w * f_h).real

    # Slice the central n entries: conv indices correspond to shift starting at 0
    # conv result length M. Since we convolve sequences of length L and n, the "zero lag" (where index = 0) corresponds to index where no shift, i.e., the sequence start. However we need conv1[j] = sum_i w_rev[i] * (1 - h)[j - i] for j in range(n) as before.
    # Standard convolution result has first element conv[0] = w_rev[0] * (1 - h)[0] (since other terms zero). That matches j=0.
    # So we can simply take conv1[0:n] and conv2[0:n].

    g = conv1[:n] - conv2[:n]
    return g
```

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # lam find threshold via binary search
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure bounds proper
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2*n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def weighted_gradient(w, h):
    n = h.shape[0]
    L = 2*n - 1
    M = 1 << ((n + L - 1).bit_length())
    w_rev = w[::-1]
    f_wr = np.fft.fft(w_rev, M)
    f_one_minus_h = np.fft.fft(1.0 - h, M)
    conv1 = np.fft.ifft(f_wr * f_one_minus_h).real
    f_w = np.fft.fft(w, M)
    f_h = np.fft.fft(h, M)
    conv2 = np.fft.ifft(f_w * f_h).real
    # take first n entries
    return conv1[:n] - conv2[:n]
```

def smooth_descent(h0, time_left, target, rng):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    step0 = 0.3
    beta = 0.9
    alpha = 5.0
    it = 0
    # schedule for alpha increase
    while time_left() > 1.0:
        it += 1
        # increase alpha slowly
        if it % 2000 == 0 and alpha < 200.0:
            alpha *= 1.5
        corr = compute_corr_full_fft(h)
        max_corr = np.max(corr)
        # compute softmax weights
        exp_arg = alpha * (corr - max_corr)  # max 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient
        grad = weighted_gradient(w, h)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 3000 == 0:
            # occasional jitter
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= target:
                break
        h = h_new
    return best_h, best_score

    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def weighted_gradient(w, h):
    """Gradient of smooth max objective: g = conv(w_rev,1-h) - conv(w,h)."""
    n = h.shape[0]
    L = 2 * n - 1
    M = 1 << ((n + L - 1).bit_length())  # next power of two
    w_rev = w[::-1]
    f_wr = np.fft.fft(w_rev, M)
    f_om = np.fft.fft(1.0 - h, M)
    conv1 = np.fft.ifft(f_wr * f_om).real
    f_w = np.fft.fft(w, M)
    f_h = np.fft.fft(h, M)
    conv2 = np.fft.ifft(f_w * f_h).real
    return conv1[:n] - conv2[:n]


def smooth_descent(h0, time_left, target, rng):
    """Gradient descent on smooth approximation of max correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    step0 = 0.3
    beta = 0.9
    alpha = 5.0
    it = 0
    while time_left() > 1.0:
        it += 1
        if it % 2000 == 0 and alpha < 200.0:
            alpha *= 1.5
        corr = compute_corr_full_fft(h)
        max_corr = np.max(corr)
        exp_arg = alpha * (corr - max_corr)           # max at 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        grad = weighted_gradient(w, h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 3000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
            if best_score <= target:
                break
        h = h_new
    return best_h, best_score


def simulated_annealing(h0, init_score, time_left, rng):
    cur = h0.copy()
    cur_score = init_score
    best_h = cur.copy()
    best_score = cur_score
    n = cur.size
    total = time_left()
    start = time.time()
    while time_left() > 0.5:
        elapsed = time.time() - start
        T = max(1e-5, 0.02 * (1.0 - elapsed / total))
        if rng.random() < 0.25:
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            delta = min(cur[i], 1.0 - cur[j]) * rng.uniform(0.001, 0.1)
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        new_score = compute_c5_fft(new)
        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / T):
            cur, cur_score = new, new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, cur.copy()
                if best_score <= 0.3808:
                    break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    TARGET = 0.3808

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128)

    best_h = None
    best_score = np.inf

    for n in (128, 256, 512):
        if time_left() <= 2.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        h_opt, sc_opt = smooth_descent(base, time_left, TARGET, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= TARGET:
                break
        base = h_opt

    # final refinement with simulated annealing
    if best_h is not None and time_left() > 5.0:
        best_h, best_score = simulated_annealing(best_h, best_score, time_left, rng)

    # occasional random binary search if still above target
    while best_h is not None and best_score > TARGET and time_left() > 2.0:
        h_bin = random_binary_balanced(best_h.size, rng)
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
            if best_score <= TARGET:
                break

    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
        n_ret = best_h.size
    else:
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, n_ret
