import time, math, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)

def upsample_to_binary(base, n):
    m = base.shape[0]
    if m == n:
        return base.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    vals = np.interp(x_new, x_old, base.astype(float))
    idx = np.argsort(-vals)
    h = np.zeros(n, dtype=int)
    h[idx[:n // 2]] = 1
    return h

def compute_delta_swap(i_remove, j_add, h):
    """i_remove has h[i]=1 and will become 0, j_add has h[j]=0 and will become 1."""
    n = h.shape[0]
    delta = np.zeros(n, dtype=int)
    if i_remove + 1 < n:
        delta[1:n - i_remove] -= h[i_remove + 1:]
    if i_remove > 0:
        delta[1:i_remove + 1] -= h[i_remove - 1::-1]
    if j_add + 1 < n:
        delta[1:n - j_add] += h[j_add + 1:]
    if j_add > 0:
        delta[1:j_add + 1] += h[j_add - 1::-1]
    return delta

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial base vector (binary)
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=int)
    else:
        base = thue_morse(64)

    best_h = None
    best_score = float('inf')
    best_n = None
    target_c5 = 0.3808
    scales = [128, 256, 512]

    for n in scales:
        # produce binary vector of length n
        h = upsample_to_binary(base, n)

        # auto-correlation p_s (including shift 0)
        p = np.correlate(h, h, mode='full')[n - 1:].astype(int)
        min_p = p[1:].min() if n > 1 else p[0]
        cur_score = 1.0 - (2.0 * min_p / n)
        if cur_score < best_score:
            best_score, best_h, best_n = cur_score, h.copy(), n

        # hill‑climbing / simulated annealing
        start_T, end_T = 0.5, 1e-5
        while time_left() > 0.1:
            # shift with minimal overlap (ignore shift 0)
            s_min = int(np.argmin(p[1:]) + 1)  # 1 <= s_min <= n-1
            improved = False
            region = n - s_min
            if region > 0:
                h0 = h[:region]
                h1 = h[s_min:]
                mask_i = (h0 == 0) & (h1 == 1)   # zero that could become one
                mask_j = (h0 == 1) & (h1 == 0)   # one that could become zero
                idx_i = np.flatnonzero(mask_i)
                idx_j = np.flatnonzero(mask_j)
                if idx_i.size and idx_j.size:
                    i = rng.choice(idx_i)        # will become 1
                    j = rng.choice(idx_j)        # will become 0
                    delta = compute_delta_swap(j, i, h)  # remove j, add i
                    p += delta
                    h[i] = 1
                    h[j] = 0
                    min_p = p[1:].min()
                    cur_score = 1.0 - (2.0 * min_p / n)
                    if cur_score < best_score:
                        best_score, best_h, best_n = cur_score, h.copy(), n
                    improved = True

            if not improved:
                # random swap with SA acceptance
                ones = np.flatnonzero(h)
                zeros = np.flatnonzero(1 - h)
                if ones.size == 0 or zeros.size == 0:
                    break
                i_rm = rng.choice(ones)   # will become 0
                j_ad = rng.choice(zeros)  # will become 1
                delta = compute_delta_swap(i_rm, j_ad, h)
                new_p = p + delta
                new_min_p = new_p[1:].min()
                elapsed = time.time() - start_time
                frac = elapsed / budget_s
                T = start_T * (end_T / start_T) ** frac
                accept = False
                if new_min_p >= min_p:
                    accept = True
                else:
                    if rng.random() < math.exp((new_min_p - min_p) / T):
                        accept = True
                if accept:
                    p = new_p
                    min_p = new_min_p
                    h[i_rm] = 0
                    h[j_ad] = 1
                    cur_score = 1.0 - (2.0 * min_p / n)
                    if cur_score < best_score:
                        best_score, best_h, best_n = cur_score, h.copy(), n

            if best_score <= target_c5:
                break

        if best_score <= target_c5:
            break

        # prepare for next scale
        base = best_h.copy()  # use current best as seed for upsampling

    final_h = best_h.astype(float)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, final_h.size
