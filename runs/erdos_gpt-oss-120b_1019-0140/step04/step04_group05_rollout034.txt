    start = time.time()

    # initial base
    if 'initial_h_values' in globals():

    base = project_onto_box_sum(base, target_n/2)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    if sa_time > 5.0:

def simulated_annealing(h0, time_limit, rng):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    cur_score = compute_c5(h)
    best_score = cur_score

    start = time.time()
    init_temp = 0.02
    final_temp = 1e-5
    # number of iterations estimate based on time limit and simple loop
    # We'll iterate while time left.
    it = 0
    while time.time() - start < time_limit:
        # linear schedule based on fraction of time used
        t_frac = (time.time() - start) / time_limit
        temp = init_temp * (1 - t_frac) + final_temp * t_frac
        # random i, j distinct
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        # delta scaled by temperature: as temp higher, larger moves
        scale = temp / init_temp
        delta = max_delta * rng.random() * scale
        if delta <= 0:
            continue
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # accept with prob

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def simulated_annealing(h0, time_limit, rng):
    """Projected SA with occasional swaps."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    cur_score = compute_c5(h)
    best_score = cur_score

    start = time.time()
    init_temp = 0.02
    final_temp = 1e-5
    swap_prob = 0.01

    while True:
        if time.time() - start >= time_limit:
            break
        # temperature linear in elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = init_temp * (1.0 - frac) + final_temp * frac

        if rng.random() < swap_prob:
            i, j = rng.choice(n, size=2, replace=False)
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                # Metropolis for swaps as well
                if temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[j], h[i]
            continue

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * (temp / init_temp)
        if delta <= 0.0:
            continue
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            if temp > 0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_score = new_score
                else:
                    h[i] += delta
                    h[j] -= delta
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_score


def targeted_refinement(h, score, time_limit, rng):
    """Greedy moves aimed at the current worst lag."""
    n = h.size
    best_h = h.copy()
    best_score = score
    start = time.time()
    while True:
        if time.time() - start >= time_limit:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)
        if lag == 0:
            break
        if lag > 0:
            shifted = np.concatenate([np.zeros(lag), best_h[:n - lag]])
        else:
            s = -lag
            shifted = np.concatenate([best_h[s:], np.zeros(s)])
        overlapped = best_h * shifted
        i_max = int(np.argmax(overlapped))
        cand = np.where((shifted < 0.2) & (best_h < 1.0))[0]
        if cand.size == 0:
            j = rng.integers(n)
            while j == i_max:
                j = rng.integers(n)
        else:
            j = int(rng.choice(cand))
            if j == i_max:
                j = rng.integers(n)
        max_delta = min(best_h[i_max], 1.0 - best_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.5
        if delta <= 0.0:
            continue
        new_h = best_h.copy()
        new_h[i_max] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h, best_score = new_h, new_score
            if best_score <= 0.3808:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = rng.random(64)
    base = project_onto_box_sum(base, base.size / 2.0)

    target_n = 512
    if base.size != target_n:
        base = upsample(base, target_n)
    base = project_onto_box_sum(base, target_n / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # Main SA phase (≈80% of budget)
    sa_time = time_left() * 0.8
    if sa_time > 5.0:
        h_sa, sc_sa = simulated_annealing(best_h, sa_time, rng)
        if sc_sa < best_score:
            best_h, best_score = h_sa, sc_sa

    # Targeted refinement (remaining budget)
    while time_left() > 1.0 and best_score > 0.3808:
        rem = time_left()
        # allocate small chunk for each refinement call
        chunk = min(20.0, rem)
        best_h, best_score = targeted_refinement(best_h, best_score, chunk, rng)

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
