def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    # handle trivial cases
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    # lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - total) <= 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    # final projection
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):

    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - total) <= 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def max_corr_and_grad(h):
    """Return max correlation value, its gradient w.r.t. h, and shift k."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n-1
    a = np.empty(L, dtype=float)
    b = np.empty(L, dtype=float)
    a[:n] = h
    a[n:] = 0.0
    b[:n] = np.flip(1.0 - h)
    b[n:] = 0.0
    fa = np.fft.rfft(a)
    fb = np.fft.rfft(b)
    conv = np.fft.irfft(fa * fb, n=L)[:2 * n - 1]
    idx = np.argmax(conv)
    max_corr = conv[idx]
    k = (n - 1) - idx  # because convolution is reversed correlation
    grad = np.zeros_like(h)
    if k >= 0:
        ki = int(k)
        grad[: n - ki] = 1.0 - h[ki:]
        grad[ki:] -= h[: n - ki]
    else:
        ki = -int(k)
        grad[ki:] = 1.0 - h[: n - ki]
        grad[: n - ki] -= h[ki:]
    return max_corr, grad, k


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    max_corr, _, _ = max_corr_and_grad(h)
    return max_corr * (2.0 / n)


def gradient_descent(h0, time_limit, rng):
    """Projected gradient descent on the max‑correlation objective."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    init_step = 0.5
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        remaining = time_limit - elapsed
        max_corr, grad, _ = max_corr_and_grad(h)
        cur_score = max_corr * (2.0 / n)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        # step size decays with remaining time
        step = init_step * (remaining / time_limit) ** 0.6
        if step < 1e-5:
            step = 1e-5
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, n / 2.0)
        h = h_new
        if best_score <= 0.3808:
            break
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score


def upsample(arr, new_n):
    """Linear interpolation to a larger array."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (as floats)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count("1") & 1)
    return seq


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # base construction
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    # work at 512 points (chosen for a good trade‑off)
    target_n = 512
    h = upsample(base, target_n)
    h = project_onto_box_sum(h, target_n / 2.0)

    # primary gradient descent on the current solution
    total = time_left()
    primary_time = total * 0.70
    best_h, best_score = gradient_descent(h, primary_time, rng)

    # stochastic restarts while time permits
    while time_left() > 1.0:
        n = target_n
        idx = rng.choice(n, size=n // 2, replace=False)
        h_rand = np.zeros(n, dtype=float)
        h_rand[idx] = 1.0
        # short descent from the random start
        chunk = min(time_left() * 0.6, 8.0)
        h_loc, sc_loc = gradient_descent(h_rand, chunk, rng)
        if sc_loc < best_score:
            best_score = sc_loc
            best_h = h_loc
            if best_score <= 0.3808:
                break

    best_h = project_onto_box_sum(best_h, target_n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, target_n
