def compute_c5_fft(h, m):
    n = h.size
    # conv using FFT
    rev = (1 - h)[::-1]
    H = np.fft.rfft(h, n=m)
    R = np.fft.rfft(rev, n=m)
    conv = np.fft.irfft(H * R, n=m)[:2*n-1]
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):

    idx = rng.choice(n, size=n // 2, replace=False)

            grad[: n - shift] -= h[shift:]
            grad[shift:] -= h[: n - shift]
        else:
            s = -shift
            grad[s:] -= h[: n - s]
            grad[: n - s] -= h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_adaptive_fft(h0, max_iters, step0, beta, rng, time_left, m):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h, m)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    eps = 1e-12
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = full_corr_fft(h, m)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fft(h_new, m)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-8:
                step = step0 * 0.1
        if it % 1000 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5_fft(h, m)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def local_pair_moves(h, rng, time_left, m):
    best_h = h.copy()
    best_score = compute_c5_fft(best_h, m)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5_fft(h_new, m)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5_fft(h_new2, m)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left, m):
    best_h = h.copy()
    best_score = compute_c5_fft(best_h, m)
    n = best_h.size
    eps = 1e-12
    while time_left() > 0:
        corr = full_corr_fft(best_h, m)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[: n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[: n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5_fft(h_new, m)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    n_target = 512
    if base.shape[0] != n_target:
        base = upsample(base, n_target)
    base = project_onto_box_sum(base, n_target / 2.0)
    m_target = 1 << ((2 * n_target - 1).bit_length())

    best_h = base.copy()
    best_score = compute_c5_fft(best_h, m_target)

    if time_left() > 5:
        max_iters = int(min(200000, max(50000, time_left() * (50000.0 / n_target))))
        h_gd, sc_gd = gradient_descent_adaptive_fft(best_h, max_iters,
                                                    step0=0.5, beta=0.85,
                                                    rng=rng, time_left=time_left,
                                                    m=m_target)
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd

    if time_left() > 5:
        h_ms, sc_ms = improve_via_max_shift(best_h, rng, time_left, m_target)
        if sc_ms < best_score:
            best_h, best_score = h_ms.copy(), sc_ms

    if time_left() > 5:
        h_pair, sc_pair = local_pair_moves(best_h, rng, time_left, m_target)
        if sc_pair < best_score:
            best_h, best_score = h_pair.copy(), sc_pair

    while time_left() > 10:
        h_rand = random_continuous_balanced(n_target, rng)
        max_iters_rand = int(min(20000, max(1000, time_left() * (5000.0 / n_target))))
        h_rand_ref, sc_rand = gradient_descent_adaptive_fft(h_rand, max_iters_rand,
                                                            step0=0.5, beta=0.85,
                                                            rng=rng, time_left=time_left,
                                                            m=m_target)
        if sc_rand < best_score:
            best_h, best_score = h_rand_ref.copy(), sc_rand
        if time_left() > 5:
            h_rand_pair, sc_rand_pair = local_pair_moves(h_rand_ref, rng, time_left, m_target)
            if sc_rand_pair < best_score:
                best_h, best_score = h_rand_pair.copy(), sc_rand_pair

    if time_left() > 30:
        n2 = 1024
        h_up = upsample(best_h, n2)
        h_up = project_onto_box_sum(h_up, n2 / 2.0)
        m2 = 1 << ((2 * n2 - 1).bit_length())
        max_iters2 = int(min(80000, max(20000, time_left() * 30)))
        h_gd2, sc_gd2 = gradient_descent_adaptive_fft(h_up, max_iters2,
                                                      step0=0.3, beta=0.85,
                                                      rng=rng, time_left=time_left,
                                                      m=m2)
        if sc_gd2 < best_score:
            best_h, best_score = h_gd2.copy(), sc_gd2
            if time_left() > 5:
                h_pair2, sc_pair2 = local_pair_moves(best_h, rng, time_left, m2)
                if sc_pair2 < best_score:
                    best_h, best_score = h_pair2.copy(), sc_pair2
            if time_left() > 5:
                h_ms2, sc_ms2 = improve_via_max_shift(best_h, rng, time_left, m2)
                if sc_ms2 < best_score:
                    best_h, best_score = h_ms2.copy(), sc_ms2

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    m_final = 1 << ((2 * best_h.size - 1).bit_length())
    best_score = compute_c5_fft(best_h, m_final)
    return best_h, best_score, best_h.size
