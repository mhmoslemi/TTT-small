def adam_optimize(h0, max_iters, target_sum, start_time, budget_s, rng):
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters+1):
        # time check
        if time.time() - start_time >= budget_s - 3:
            break
        # compute full correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # gradient for shifts achieving max (within tolerance)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                sz = n - shift
                term1 = np.empty(n, dtype=h.dtype)
                term1[:sz] = h[shift:]
                term1[sz:] = 0.0
                term2 = np.empty(n, dtype=h.dtype)
                term2[shift:] = h[:sz]
                term2[:shift] = 0.0
            else:
                s = -shift
                sz = n - s
                term1 = np.empty(n, dtype=h.dtype)
                term1[s:] = h[:sz]
                term1[:s] = 0.0
                term2 = np.empty(n, dtype=h.dtype)
                term2[:sz] = h[s:]
                term2[sz:] = 0.0
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / np.sqrt(t)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        # project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        # occasional noise injection (every 1000)
        if t % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_box_sum(h_new, target_sum)
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        s = w.sum()
        if abs(s - target_sum) < 1e-12:
            return w
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def max_shift_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[s:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[s:] = h[:sz]
            term2[:s] = 0.0
        else:
            s2 = -s
            sz = n - s2
            term1 = np.empty(n, dtype=h.dtype)
            term1[s2:] = h[:sz]
            term1[:s2] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s2:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad, max_corr

def adam_optimize(h0, max_iters, target_sum, start, budget, rng):
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.2
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters + 1):
        if time.time() - start >= budget - 3.0:
            break
        grad, _ = max_shift_gradient(h)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / np.sqrt(t)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, target_sum)
        if t % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_refine(h, cur_score, start, budget, rng):
    n = h.size
    best_h, best_score = h.copy(), cur_score
    remaining = budget - (time.time() - start)
    total_steps = int(min(200000, max(20000, remaining * 200)))
    for step in range(total_steps):
        if time.time() - start >= budget:
            break
        if step % 1000 == 0 and (time.time() - start) >= budget:
            break
        T = 0.01 * (1.0 - step / total_steps) + 1e-6
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h, cur_score = h_new, new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_final = 1024
    scales = [64, 128, 256, 512, n_final]
    best_h = None
    best_score = np.inf

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = rng.random(64)
    base = project_onto_box_sum(base, 64 / 2.0)

    for idx, n in enumerate(scales):
        if time.time() - start >= budget_s - 5.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        target = n / 2.0
        base = project_onto_box_sum(base, target)

        # random balanced continuous samples
        r_samples = 30 if n <= 256 else 10
        for _ in range(r_samples):
            if time.time() - start >= budget_s - 5.0:
                break
            h_rand = rng.random(n)
            h_rand = project_onto_box_sum(h_rand, target)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score, best_h = sc, h_rand.copy()

        # Adam gradient descent restarts
        for r in range(3):
            if time.time() - start >= budget_s - 5.0:
                break
            h0 = base.copy()
            if r > 0:
                h0 += rng.normal(scale=0.05, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, target)
            remaining = budget_s - (time.time() - start)
            max_iters = int(min(50000, max(3000, remaining * 30)))
            h_opt, sc_opt = adam_optimize(h0, max_iters, target, start, budget_s, rng)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt

        # Simulated annealing refinement at this scale
        if best_h is not None and best_h.size == n and (time.time() - start) < budget_s - 5.0:
            cur_sc = compute_c5(best_h)
            h_ref, sc_ref = simulated_annealing_refine(best_h, cur_sc, start, budget_s, rng)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()
            base = best_h

        # prepare for next scale
        if idx < len(scales) - 1:
            base = best_h if best_h is not None else base

    # final Adam refinement on the best found
    if best_h is not None and (time.time() - start) < budget_s - 5.0:
        n = best_h.size
        target = n / 2.0
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(200000, max(50000, remaining * 30)))
        best_h, best_score = adam_optimize(best_h, max_iters, target, start, budget_s, rng)
        best_h = project_onto_box_sum(best_h, target)
        best_score = compute_c5(best_h)

    # fallback if nothing succeeded
    if best_h is None:
        n = n_final
        best_h = project_onto_box_sum(rng.random(n), n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
