def compute_c5(h):
    n = h.shape[0]
    # compute linear correlation via FFT with zero padding length = 2n
    N = 2*n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * G, n=N)
    # full linear correlation length 2n-1, but we got length N, we use first (2n-1)
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)
```

def gradient_mass_transfer_climb(h0, steps, rng):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for step in range(steps):
        # compute worst shifts
        corr = np.fft.irfft(np.fft.rfft(h, n=2*n) * np.fft.rfft(1.0 - h, n=2*n), n=2*n)[:2*n-1]
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n-1)
        grad = compute_gradients_for_shifts(h, shifts)
        i = np.argmax(grad)
        j = np.argmin(grad)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        h[i] -= delta
        h[j] += delta
        # optional small random perturb occasionally
        if step % 1000 == 0 and rng.random() < 0.05:
            idx = rng.integers(n)
            h[idx] = np.clip(h[idx] + rng.normal(scale=0.01), 0.0, 1.0)
            h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def simulated_annealing(h0, steps, rng):
    h = h0.copy()
    n = h.shape[0]
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T0 = 0.01
    T_final = 1e-6
    for step in range(steps):
        T = T0 * ((steps - step) / steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def full_corr(h):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * G, n=N)[:2 * n - 1]
    return corr

def compute_c5(h):
    n = h.shape[0]
    return full_corr(h).max() * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, max_iters, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time.time() - start >= max_time:
            break
        corr = full_corr(h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        if it % 500 == 0:
            i = np.argmax(grad)
            j = np.argmin(grad)
            move = min(h_new[i], 1.0 - h_new[j])
            if move > 0:
                delta = move * rng.random()
                h_new[i] -= delta
                h_new[j] += delta
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def mass_transfer_climb(h0, steps, rng):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    for step in range(steps):
        corr = full_corr(h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        i = np.argmax(grad)
        j = np.argmin(grad)
        move = min(h[i], 1.0 - h[j])
        if move <= 0:
            continue
        delta = move * rng.random()
        h[i] -= delta
        h[j] += delta
        if step % 2000 == 0 and rng.random() < 0.1:
            idx = rng.integers(n)
            h[idx] = np.clip(h[idx] + rng.normal(scale=0.01), 0.0, 1.0)
            h = project_onto_box_sum(h, n / 2.0)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def simulated_annealing(h0, steps, rng):
    h = h0.copy()
    n = h.shape[0]
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T0 = 0.01
    T_final = 1e-6
    for step in range(steps):
        T = T0 * ((steps - step) / steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        move = min(h[i], 1.0 - h[j])
        if move <= 0:
            continue
        delta = move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)
