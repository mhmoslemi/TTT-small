def compute_c5_and_shift(h):
    n = h.shape[0]
    g = 1.0 - h
    g_rev = g[::-1]
    N = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, N)
    G_rev = np.fft.rfft(g_rev, N)
    conv = np.fft.irfft(H * G_rev, N)[:2*n-1]
    idx = np.argmax(conv)
    max_corr = conv[idx]
    shift = (n - 1) - idx
    return max_corr * (2.0 / n), shift

def subgradient_descent(h0, time_limit, rng):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_score,_ = compute_c5_and_shift(h)  # compute once
    iter_cnt = 0
    while time.time() - start < time_limit:
        # compute current c5 and shift
        c5, shift = compute_c5_and_shift(h)
        # if c5 < best_score: update
        if c5 < best_score:
            best_score = c5
            best_h = h.copy()
            # optional early exit
            if best_score <= 0.3808:
                break
        # compute gradient
        n = h.shape[0]
        grad = np.zeros_like(h)
        if shift >= 0:
            # grad[:n-shift] = 1 - h[shift:]
            # grad[shift:] -= h[:n-shift]
            end = n - shift
            grad[:end] = 1.0 - h[shift:]
            grad[shift:] -= h[:end]
        else:
            s = -shift
            end = n - s
            grad[s:] = 1.0 - h[:end]
            grad[:end] -= h[s:]
        # step size schedule
        # alpha = initial_alpha / sqrt(iter+1)
        iter_cnt += 1
        alpha = 0.5 / np.sqrt(iter_cnt)  # maybe 0.5 initial step
        # update
        h_new = h - alpha * grad
        # project onto box-sum constraints
        h = project_onto_box_sum(h_new, n/2.0)
    # final best
    # compute final best score (ensuring projection)
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_c5_and_shift(h):
    """Return (C5, shift) using FFT‑based linear correlation."""
    n = h.shape[0]
    g = 1.0 - h
    g_rev = g[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    G_rev = np.fft.rfft(g_rev, N)
    conv = np.fft.irfft(H * G_rev, N)[: 2 * n - 1]
    idx = np.argmax(conv)
    max_corr = conv[idx]
    shift = (n - 1) - idx
    return max_corr * (2.0 / n), shift


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    g = 1.0 - h
    g_rev = g[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    G_rev = np.fft.rfft(g_rev, N)
    conv = np.fft.irfft(H * G_rev, N)[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def subgradient_descent(h0, time_limit, rng):
    """Subgradient method for min max_k ∫h(1−h_k); returns best found."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 0
    while time.time() - start < time_limit:
        c5, shift = compute_c5_and_shift(h)
        if c5 < best_score:
            best_score = c5
            best_h = h.copy()
            if best_score <= 0.3808:
                break
        # gradient of the worst shift
        grad = np.zeros_like(h)
        if shift >= 0:
            end = n - shift
            grad[:end] = 1.0 - h[shift:]
            grad[shift:] -= h[:end]
        else:
            s = -shift
            end = n - s
            grad[s:] = 1.0 - h[:end]
            grad[:end] -= h[s:]
        it += 1
        alpha = 0.5 / np.sqrt(it)
        h_new = h - alpha * grad
        h = project_onto_box_sum(h_new, n / 2.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score
