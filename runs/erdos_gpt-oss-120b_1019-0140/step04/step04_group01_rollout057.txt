    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)   # actual max corr scaled
    v = np.zeros_like(h)
    iteration = 0
    alpha = alp_init
    while time_left_func() > 0.01 and iteration < max_iter:
        # compute gradient
        grad = compute_softmax_gradient(h, alpha)  # returns gradient unscaled
        v = beta * v + (1 - beta) * grad
        eta = step0 / np.sqrt(iteration + 1)
        h_new = h - eta * v
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # update alpha gradually
        if iteration % 200 == 0 and alpha < alp_max:
            alpha = min(alp_max, alpha * 1.1)
        iteration += 1
    return best_h, best_score

def compute_softmax_gradient(h, alpha):
    n = h.shape[0]
    # compute unscaled corr
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * G, n=M)[:2 * n - 1]
    # compute softmax
    max_c = np.max(corr)
    exp_alpha = np.exp(alpha * (corr - max_c))
    sum_exp = np.sum(exp_alpha)
    p = exp_alpha / sum_exp
    # reversed p
    p_rev = p[::-1]
    # convolution p_rev and h
    L = 3 * n - 2
    M2 = 1 << ((L - 1).bit_length())
    P = np.fft.rfft(p_rev, n=M2)
    Hh = np.fft.rfft(h, n=M2)
    conv_full = np.fft.irfft(P * Hh, n=M2)[:L]
    weighted = conv_full[2 * n - 2 : 2 * n - 2 + n]
    grad = 1.0 - 2.0 * weighted
    return grad
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # broaden bounds if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C₅, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * G, n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def resample_to_n(arr, n):
    """Linear interpolation to length n."""

    if old_n == n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_softmax_gradient(h, alpha):
    """Gradient of smoothed max overlap (softmax) w.r.t. h."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * G, n=M)[:2 * n - 1]          # unscaled correlation
    max_c = np.max(corr)
    exp_alpha = np.exp(alpha * (corr - max_c))
    p = exp_alpha / np.sum(exp_alpha)                  # softmax weights, sum(p)=1
    p_rev = p[::-1]
    L = 3 * n - 2
    M2 = 1 << ((L - 1).bit_length())
    P = np.fft.rfft(p_rev, n=M2)
    Hh = np.fft.rfft(h, n=M2)
    conv_full = np.fft.irfft(P * Hh, n=M2)[:L]
    weighted = conv_full[2 * n - 2: 2 * n - 2 + n]
    grad = 1.0 - 2.0 * weighted
    return grad

def smooth_gradient_descent(h0, time_left_func, max_iter=20000,
                           step0=0.3, beta=0.9,
                           alpha_start=10.0, alpha_max=150.0):
    """Projected gradient descent on the softmax‑smoothed max overlap."""
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    v = np.zeros_like(h)
    alpha = alpha_start
    it = 0
    while it < max_iter and time_left_func() > 0.01:
        grad = compute_softmax_gradient(h, alpha)
        v = beta * v + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it + 1.0)
        h_new = h - eta * v
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
        # slowly increase smoothing parameter
        if (it % 200) == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.1)
    return best_h, best_score

        else:

            h[j] = 0.0

    return best_h, best_score
