def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.shape[0]
    if s <= lo*n:
        return np.full_like(v, lo)
    if s >= hi*n:
        return np.full_like(v, hi)
    # Bisection on lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        x = np.clip(v - lam_mid, lo, hi)
        sumx = x.sum()
        if sumx < s:
            lam_high = lam_mid
        else:
            lam_low = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

def compute_gradients_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # overlapping region: i=0..n-shift-1
            grad[:n-shift] += 1.0 - h[shift:]
            grad[shift:] += -h[:n-shift]
        else:
            s = -shift
            # overlapping region: i=s..n-1 (i refers to original index)
            grad[s:] += 1.0 - h[:n-s]
            grad[:n-s] += -h[s:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {x: lo <= x_i <= hi, sum x_i = s} using bisection."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi   # sum at lam_low >= s
    lam_high = np.max(v) - lo  # sum at lam_high <= s
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam_mid, lo, hi)
        total = x.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return np.max(conv) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from old size to new_n."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Binary Thue–Morse sequence of length n as floats."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def random_continuous_balanced(n, rng):
    """Continuous vector in [0,1] with sum = n/2."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def compute_gradients_for_shifts(h, shifts):
    """Gradient of the worst‑shift correlation term (averaged over shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            k = n - shift
            grad[:k] += 1.0 - h[shift:]
            grad[shift:] += -h[:k]
        else:
            s = -shift
            k = n - s
            grad[s:] += 1.0 - h[:k]
            grad[:k] += -h[s:]
    if shifts:
        grad /= len(shifts)
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum and adaptive step."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break

        corr = np.convolve(h, (1.0 - h)[::-1], mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)

        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.01)
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            no_improve += 1
            if step < 1e-7:
                step = step0 * 0.1

        if no_improve >= 2000:
            # add small jitter to escape flat regions
            jitter = rng.normal(scale=0.02, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            no_improve = 0

    return best_h, best_score

                best_h, best_score = h_rand.copy(), score_rand
            if time_left() > 2:
                h_rand, score_rand = local_pair_moves(h_rand, rng, time_left)
                if score_rand < best_score:
                    best_h, best_score = h_rand.copy(), score_rand

    # Final intensive refinement on the global best
    if time_left() > 5 and best_score > target:
        max_iters_final = int(min(120000, max(40000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(
            best_h, max_iters_final, step0=0.3, beta=0.9, rng=rng, time_left=time_left
        )

    # One more round of targeted and pair moves
    if time_left() > 2:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    # Final projection and exact evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
