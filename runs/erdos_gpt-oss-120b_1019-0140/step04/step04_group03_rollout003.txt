def adam_optimize(h0, steps, lr, beta1, beta2, eps, time_limit, start_time, target_score=None):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    n = len(h)
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, steps+1):
        now = time.time()
        if now - start_time >= time_limit:
            break
        # compute correlation and worst shifts
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr_full)
        # find shifts achieving max_corr within tolerance
        tol = 1e-12
        inds = np.where(corr_full >= max_corr - tol)[0]
        shifts = inds - (n -1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # projection
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if target_score is not None and best_score <= target_score:
                return best_h, best_score
        h = h_new
    return best_h, best_score

    n = len(h)
    target_sum = n/2.0
    best_h = h.copy()

    # Temperature schedule: T0 maybe 0.02, T_end 1e-6.
    # We'll compute total steps = steps.
    for t in range(steps):
        now = time.time()
        if now - start_time >= time_limit:
            break
        # compute corr and worst shifts
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr_full)
        inds = np.where(corr_full >= max_corr - 1e-12)[0]
        # choose shift randomly among them
        shift = inds[rng.integers(len(inds))] - (n-1)
        # Determine feasible i
        if shift >= 0:
            # i ranges 0..n-shift-1 for valid i+shift
            idx_i = rng.integers(n - shift)
            i = idx_i
            j = i + shift
        else:
            s = -shift
            idx_i = rng.integers(n - s)
            i = idx_i + s
            j = i - s
        # Ensure i != j and feasible move
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            # fallback random pair
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 0.0:
                continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # No need to project: mass conserved and bounds satisfied
        # Evaluate
        score = compute_c5(h_new)
        # Temperature schedule
        # Compute T decreasing linearly on steps
        T0 = 0.02

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad


def adam_optimize(h0, max_steps, lr, beta1, beta2, eps,
                  time_limit, start_time, rng, target_score=None):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_steps + 1):
        if time.time() - start_time >= time_limit:
            break
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr_full)
        idxes = np.where(corr_full >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
            if target_score is not None and best_score <= target_score:
                return best_h, best_score
        h = h_new
    return best_h, best_score


def anneal_refine(h0, max_steps, time_limit, start_time, rng, target_score=None):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    T0 = 0.02
    T_end = 1e-6
    for step in range(max_steps):
        if time.time() - start_time >= time_limit:
            break
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr_full)
        idxes = np.where(corr_full >= max_corr - 1e-12)[0]
        shift = idxes[rng.integers(len(idxes))] - (n - 1)

        if shift >= 0:
            i = rng.integers(n - shift)
            j = i + shift
        else:
            s = -shift
            i = rng.integers(n - s) + s
            j = i - s

        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 0.0:
                continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        new_score = compute_c5(h_new)
        T = T0 * (1 - step / max_steps) + T_end
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / T):
            h = h_new
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if target_score is not None and best_score <= target_score:
                    return best_h, best_score
    return best_h, best_score
