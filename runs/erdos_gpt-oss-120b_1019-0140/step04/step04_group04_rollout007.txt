def gradient_descent_momentum_custom(h0, max_iter, step0, beta, rng, time_left):

    best_h = h.copy()

        max_corr = np.max(corr)

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter,
                                                  'ftol': 1e-12, 'disp': False})
    if not res.success:
        # fallback to original
        h_opt = project_onto_box_sum(h0, n / 2.0)
        t_opt = compute_c5(h_opt)
        return h_opt, t_opt
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
