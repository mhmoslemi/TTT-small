import time
import math
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from size len(arr) to new_n (both assume periodic domain)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_c5(h):
    """Fast C5 via FFT: max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    g_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(g_rev, n=m)
    conv = np.fft.irfft(H * G, n=m)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def random_balanced_binary(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def anneal_grid(h0, time_limit, rng):
    """Simulated annealing on grid h0 for given time_limit seconds."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.05
    Tmin = 1e-6
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        temp = max(Tmin, T0 * (1.0 - elapsed / time_limit))
        # decide move type
        if rng.random() < 0.65:  # swap move (preserves binary nature)
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            hi, hj = h[i], h[j]
            if hi == hj:
                continue
            # swap
            h[i], h[j] = hj, hi
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                diff = new_score - cur_score
                if diff / temp > 50:
                    accept = False
                else:
                    accept = rng.random() < math.exp(-diff / temp)
                if accept:
                    cur_score = new_score
                else:
                    h[i], h[j] = hi, hj
        else:  # small delta transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            scale = temp / T0
            delta = max_delta * rng.random() * scale
            if delta <= 0.0:
                continue
            h[i] -= delta
            h[j] += delta
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                diff = new_score - cur_score
                if diff / temp > 50:
                    accept = False
                else:
                    accept = rng.random() < math.exp(-diff / temp)
                if accept:
                    cur_score = new_score
                else:
                    h[i] += delta
                    h[j] -= delta
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
