def softmax_gradient_descent(h0, time_left, max_iters=5000, beta=20.0, lr0=0.8):

    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_val = compute_c5_fast(h)
    beta_cur = beta
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        grad, _ = softmax_grad(h, beta_cur)
        lr = lr0 / np.sqrt(it)
        h = project_onto_box_sum(h - lr * grad, target)
        cur = compute_c5_fast(h)
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
        # optionally increase beta every few iterations to sharpen
        if it % 1000 == 0:
            beta_cur = min(beta_cur * 1.2, 200.0)
        # early stop if best_val <= target_bound
        # but target bound is 0.3808; we can check outside.
    return best_h, best_val

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

def softmax_grad(h, beta):
    n = h.shape[0]
    corr = compute_corr_raw(h)  # length 2n-1
    max_corr = corr.max()
    w = np.exp(beta * (corr - max_corr))
    sum_w = w.sum()
    # w_center at index n-1
    w_center = w[n-1]
    # w_pos from index n-1 to end, length n
    w_pos = w[n-1:]            # shape (n,)
    # w_neg from index n-1 down to 0 (inclusive), length n
    w_neg = w[n-1::-1]         # shape (n,)

    grad = grad_raw / sum_w
    # return gradient

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # expand bounds if needed
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def compute_corr_raw(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    return conv[:2 * n - 1]

def softmax_grad(h, beta):
    n = h.shape[0]
    corr = compute_corr_raw(h)                # raw correlations
    max_corr = corr.max()
    w = np.exp(beta * (corr - max_corr))      # non‑negative weights
    sum_w = w.sum()
    # split weights into positive and negative shift components
    w_pos = w[n - 1:]                         # shifts 0 .. n-1
    w_neg = w[n - 1::-1]                      # shifts 0 .. -(n-1)
    # A = sum_{k>=0} w_k * h_{i+k}
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_hrev = np.fft.rfft(h_rev, n=N)
    fft_wpos = np.fft.rfft(w_pos, n=N)
    conv_rev = np.fft.irfft(fft_hrev * fft_wpos, n=N)[:2 * n - 1]
    A = conv_rev[n - 1::-1][:n]               # reversed slice of length n
    # B = sum_{k>=0} w_{-k} * h_{i-k}
    fft_h = np.fft.rfft(h, n=N)
    fft_wneg = np.fft.rfft(w_neg, n=N)
    conv_fwd = np.fft.irfft(fft_h * fft_wneg, n=N)[:2 * n - 1]
    B = conv_fwd[:n]
    grad_raw = sum_w - A - B
    grad = grad_raw / sum_w
    return grad

def softmax_gradient_descent(h0, time_left, max_iters=5000, beta=15.0, lr0=0.8):
    n = h0.shape[0]
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_val = compute_c5_fast(h)
    beta_cur = beta
    for it in range(1, max_iters + 1):
        if time_left() <= 0.0:
            break
        grad = softmax_grad(h, beta_cur)
        lr = lr0 / np.sqrt(it)
        h = project_onto_box_sum(h - lr * grad, target)
        cur = compute_c5_fast(h)
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
        if it % 1000 == 0:
            beta_cur = min(beta_cur * 1.2, 200.0)
    return best_h, best_val

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # base resolution
    n = 256
    target_sum = n / 2.0
    best_h = None
    best_val = np.inf

    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            init = resample(init, n)
        candidates.append(project_onto_box_sum(init, target_sum))
    candidates.append(thue_morse_binary(n))
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))

    # refine each candidate
    for c in candidates:
        if time_left() < 1.0:
            break
        h_opt, v_opt = softmax_gradient_descent(c, time_left,
                                                max_iters=5000,
                                                beta=15.0,
                                                lr0=0.8)
        if v_opt < best_val:
            best_val, best_h = v_opt, h_opt.copy()
        if best_val <= 0.3808:
            return best_h, best_val, n

    # random restarts
    while time_left() > 5.0:
        v = rng.normal(loc=0.5, scale=0.2, size=n)
        v = np.clip(v, 0.0, 1.0)
        v = project_onto_box_sum(v, target_sum)
        h_opt, v_opt = softmax_gradient_descent(v, time_left,
                                                max_iters=5000,
                                                beta=15.0,
                                                lr0=0.8)
        if v_opt < best_val:
            best_val, best_h = v_opt, h_opt.copy()
        if best_val <= 0.3808:
            return best_h, best_val, n

    # if still above target, upscale to finer grid and refine
    if time_left() > 5.0:
        n2 = 512
        target2 = n2 / 2.0
        if best_h is None:
            best_h = np.full(n, 0.5)
        best_h2 = resample(best_h, n2)
        best_h2 = project_onto_box_sum(best_h2, target2)
        h_opt2, v_opt2 = softmax_gradient_descent(best_h2, time_left,
                                                  max_iters=8000,
                                                  beta=20.0,
                                                  lr0=0.6)
        if v_opt2 < best_val:
            best_h, best_val, n = h_opt2, v_opt2, n2

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_val = compute_c5_fast(best_h)
    return best_h, best_val, best_h.size
