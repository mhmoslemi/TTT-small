def gradient_descent(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        # momentum
        momentum = beta*momentum + (1-beta)*grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h0, rng, time_left):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score
    # Track start_time for temperature schedule

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    # Determine base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128).astype(float)  # maybe start with 128 points
    # Ensure base sum constraint
    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = None
    best_score = np.inf
    best_n = base.size
    
    # scales
    scales = [128, 256, 512]
    
    for n in scales:
        if time_left() <= 2.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        # project after upsample
        base = project_onto_box_sum(base, n/2.0)
        
        # perform several gradient descent restarts
        restarts = 4
        for r in range(restarts):
            if time_left() <= 1.0:
                break
            # initial h0: start near base with noise
            h0 = base + rng.normal(scale=0.05, size=n)
            h0 = project_onto_box_sum(h0, n/2.0)
            # gradient descent
            max_iter = 150000
            h_opt, sc_opt = gradient_descent(
                h0, max_iter=max_iter, step0=0.6, beta=0.9,
                rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()
                best_n = n
                if best_score <= 0.3808:
                    break
        # if we have a best_h at this n, try binary hill climb
        if best_h is not None and best_h.size == n and time_left() > 2.0:
            # Round to binary
            # Use ordering: top n/2 values to 1
            idx_sorted = np.argsort(-best_h)  # descending
            bin_h = np.zeros(n, dtype=float)
            bin_h[idx_sorted[:n//2]] = 1.0
            # hill climb
            bin_h_opt, bin_sc_opt = binary_hill_climb(bin_h, rng, time_left)
            if bin_sc_opt < best_score:
                best_score = bin_sc_opt
                best_h = bin_h_opt.copy()
                best_n = n
                if best_score <= 0.3808:
                    break
        # set base for next scale
        base = best_h.copy() if best_h is not None else base
        # continue
        
    # Final fallback if nothing found
    if best_h is None:
        n_ret = 256
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret/2.0)
        best_score = compute_c5_fft(best_h)
        best_n = n_ret
    else:
        # Ensure final projection
        best_h = project_onto_box_sum(best_h, best_h.size/2.0)
        best_score = compute_c5_fft(best_h)
        best_n = best_h.size
    
    return best_h, best_score, best_n

import time

import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_full_fft(h):
    """Full linear correlation of h with (1‑h) via FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Sub‑gradient of max‑correlation term (averaged over all maximising shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent(h0, max_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def binary_hill_climb(h0, rng, time_left):
    """Simple hill‑climbing on a balanced binary vector (swap 1↔0)."""
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score
    start = time.time()
    while time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        sc_new = compute_c5_fft(h_new)
        if sc_new < cur_score:
            h = h_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
        else:
            # occasional uphill move with tiny probability
            if rng.random() < 0.001:
                h = h_new
                cur_score = sc_new
        # optional early exit if excellent bound reached
        if best_score <= 0.3808:
            break
    return best_h, best_score
