def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # find lam bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure proper bounds
    # adjust bounds if needed
    if f(lam_low) < target_sum:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

Now implement compute_gradient:

def compute_gradient(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            s = shift
            grad[:n - s] += 1.0 - h[s:]
            grad[s:] += -h[:n - s]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] += -h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

# -------------------- helper functions -------------------- #
def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < target_sum:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target_sum:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_fft_raw(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, n=L)
    G = np.fft.fft(1.0 - h, n=L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    return corr


def compute_c5_fft(h):
    n = h.shape[0]
    return np.max(compute_corr_fft_raw(h)) * (2.0 / n)


def compute_gradient(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            s = shift
            grad[: n - s] += 1.0 - h[s:]
            grad[s:] += -h[: n - s]
        else:
            s = -shift
            grad[s:] += 1.0 - h[: n - s]
            grad[: n - s] += -h[s:]
    if shifts:
        grad /= len(shifts)
    return grad


def optimize_h(h_init, n, time_left, rng):
    target_sum = n / 2.0
    h = project_onto_box_sum(h_init.copy(), target_sum)
    best_h = h.copy()
    best_score = compute_c5_fft(h)

    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.5
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    while time_left() > 0.05:
        corr_raw = compute_corr_fft_raw(h)
        max_val = np.max(corr_raw)
        c5 = max_val * (2.0 / n)

        if c5 < best_score:
            best_score = c5
            best_h = h.copy()
            if best_score <= 0.3808:
                break

        max_idxs = np.where(corr_raw == max_val)[0]
        shifts = (max_idxs - (n - 1)).tolist()
        grad = compute_gradient(h, shifts)

        # Adam update
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = alpha / (np.sqrt(v_hat) + eps)

        h_new = h - step * m_hat
        h_new = project_onto_box_sum(h_new, target_sum)

        # occasional noise
        if it % 400 == 0:
            noise = rng.normal(scale=0.005, size=n)
            h_new = project_onto_box_sum(h_new + noise, target_sum)

        # occasional pairwise tweak
        if it % 1000 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(h_new[i], 1.0 - h_new[j]) * rng.uniform(0.1, 0.3)
            if delta > 0:
                h_tmp = h_new.copy()
                h_tmp[i] -= delta
                h_tmp[j] += delta
                c5_tmp = compute_c5_fft(h_tmp)
                if c5_tmp < best_score:
                    best_score = c5_tmp
                    best_h = h_tmp.copy()
                    h_new = h_tmp

        h = h_new

    return best_h, best_score


# -------------------- main entry point -------------------- #
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # base vector
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    scales = [64, 128, 256, 512]
    alloc_fractions = [0.12, 0.18, 0.30, 0.40]  # sum ≈ 1

    overall_best_h = None
    overall_best_score = float("inf")

    for idx, n in enumerate(scales):
        if time_left() <= 5.0:
            break

        # allocate time for this scale
        alloc = budget_s * alloc_fractions[idx]
        remaining = time_left()
        alloc = min(alloc, remaining * 0.9)  # keep a margin
        scale_start = time.time()
        scale_end = scale_start + alloc
        scale_time_left = lambda: max(0.0, scale_end - time.time())

        # upsample base if needed and project
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # main optimization on current scale
        h_opt, sc_opt = optimize_h(base, n, scale_time_left, rng)
        if sc_opt < overall_best_score:
            overall_best_score = sc_opt
            overall_best_h = h_opt.copy()

        # short random restarts while time permits
        while scale_time_left() > 5.0:
            h_rand = random_binary_balanced(n, rng)
            # allocate a small budget for this restart
            sub_alloc = min(30.0, scale_time_left() * 0.3)
            sub_end = time.time() + sub_alloc
            sub_time_left = lambda: max(0.0, sub_end - time.time())
            h_rand_opt, sc_rand_opt = optimize_h(h_rand, n, sub_time_left, rng)
            if sc_rand_opt < overall_best_score:
                overall_best_score = sc_rand_opt
                overall_best_h = h_rand_opt.copy()
            if overall_best_score <= 0.3808:
                break

        if overall_best_score <= 0.3808:
            break

        base = overall_best_h.copy()

    if overall_best_h is None:
        overall_best_h = project_onto_box_sum(base, base.size / 2.0)
        overall_best_score = compute_c5_fft(overall_best_h)

    # final refinement with any leftover time
    if time_left() > 5.0:
        h_ref, sc_ref = optimize_h(overall_best_h, overall_best_h.size,
                                   time_left, rng)
        if sc_ref < overall_best_score:
            overall_best_h = h_ref
            overall_best_score = sc_ref

    overall_best_h = project_onto_box_sum(overall_best_h,
                                          overall_best_h.size / 2.0)
    overall_best_score = compute_c5_fft(overall_best_h)
    return overall_best_h, overall_best_score, overall_best_h.size
