def adam_optimize(h0, time_left, max_iters=50000):
    n = h0.size
    m = np.zeros_like(h0)
    v = np.zeros_like(h0)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_h = h0.copy()
    best_score = compute_c5(h0)
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h0)
        grad = compute_gradients_for_shifts(h0, shifts)
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = 0.5 * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = h0 - step
        # occasional noise
        if it % 2000 == 0:
            h_new += np.random.RandomState(it).normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, n/2.0)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h0 = h_new
    return best_h, best_score

    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)
    # Determine initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        # ensure sum constraint exactly by projecting
        base = project_onto_box_sum(base, base.size/2.0)
    else:
        # generate random binary balanced vector as base
        n0 = 64
        base = np.zeros(n0)
        idx = rng.choice(n0, size=n0//2, replace=False)
        base[idx] = 1.0
    best_h = base.copy()
    best_score = compute_c5(best_h)
    # scale sequence
    scales = [64, 128, 256, 512]
    for n in scales:
        if time_left() <= 5: break
        # upscale base to current n if needed
        if best_h.size != n:
            # upsample using linear interpolation
            x_old = np.linspace(0, 2, best_h.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            base = np.interp(x_new, x_old, best_h)
        else:
            base = best_h.copy()

        base = project_onto_box_sum(base, n/2.0)
        # Run Adam optimizer
        max_iters = int(min(200000, max(20000, time_left() * (1000.0 / n))))  # approximate
        h_opt, sc_opt = adam_optimize(base, time_left, max_iters=max_iters, rng=rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

        # maybe also do occasional random restart with small steps
        # After each scale, we can do some random binary restarts and short Adam

            # Random binary restart

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure the bracket contains the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def max_shifts(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def adam_optimize(h0, time_left, rng, max_iters=50000):

    beta2 = 0.999

        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)

        h_new[j] += delta
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        it += 1
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    import time
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # Initial base vector
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = project_onto_box_sum(base, base.size / 2.0)
    else:
        n0 = 64
        base = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        base[idx] = 1.0

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]
    for n in scales:
        if time_left() <= 5:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        # Adam optimization
        remaining = time_left()
        max_iters = int(min(120000, max(20000, remaining * (60000.0 / n))))
        h_opt, sc_opt = adam_optimize(best_h, time_left, rng, max_iters=max_iters)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

        # Local refinement
        if time_left() > 5:
            h_ref, sc_ref = local_refine(best_h, time_left, rng, max_iter=3000)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()

        # Random binary restart with short Adam
        if time_left() > 10:
            h_rand = np.zeros(n, dtype=float)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_rand[idx] = 1.0
            max_iters2 = int(min(8000, max(2000, time_left() * (2000.0 / n))))
            h_rand_opt, sc_rand_opt = adam_optimize(h_rand, time_left, rng, max_iters=max_iters2)
            if sc_rand_opt < best_score:
                best_score, best_h = sc_rand_opt, h_rand_opt.copy()

    # Final binary rounding check
    n = best_h.size
    k = n // 2
    binary_h = np.zeros_like(best_h)
    if k > 0:
        idx = np.argpartition(best_h, -k)[-k:]
        binary_h[idx] = 1.0
    binary_score = compute_c5(binary_h)
    if binary_score < best_score:
        best_h, best_score = binary_h, binary_score

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
