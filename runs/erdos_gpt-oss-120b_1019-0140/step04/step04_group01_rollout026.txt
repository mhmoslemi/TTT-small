def binary_anneal_aggressive(h0, rng, time_left, max_iters=None, T0=1.0, decay=0.99999, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    # compute initial score
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    # determine max_iters based on time_left if not given
    # while time left
    while time_left() > 0.01:
        if max_iters is not None and it >= max_iters:
            break
        # random swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > Tmin and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        # temperature update
        T = max(Tmin, T * decay)
        it += 1
        # occasional local greedy (e.g., every 5000 steps)
        if it % 5000 == 0 and time_left() > 0.1:
            # perform local hill climbing: try up to 1000 random swaps and accept any improvement
            for _ in range(200):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i] = 0.0
                h[j] = 1.0
                sc = compute_c5_fft(h)
                if sc < current_score:
                    current_score = sc
                    if sc < best_score:
                        best_score = sc
                        best_h = h.copy()
                    # update ones/zeros arrays
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h <= 0.5)[0]
                    # break early after improvement
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    return best_h, best_score

def gradient_descent_blocks(b0, block_len, time_left, step0=0.5, beta=0.9, max_iter=200000):
    b = b0.copy()
    m = b.size
    s = m/2.0
    best_b = b.copy()
    best_score = np.inf
    momentum = np.zeros_like(b)
    it = 1
    while it <= max_iter and time_left()>0:
        h = np.repeat(b, block_len)
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad_h = compute_gradients_for_shifts(h, shifts)
        # map to block gradient
        grad_b = grad_h.reshape(m, block_len).mean(axis=1)
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad_b
        eta = step0 / np.sqrt(it)
        b_new = b - eta * momentum
        b_new = project_onto_box_sum(b_new, s)
        # compute score
        h_new = np.repeat(b_new, block_len)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_b = b_new.copy()
        b = b_new
        it += 1
    return best_b, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max-overlap term w.r.t. h (averaged over given shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func,
                                        step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_descent_blocks(b0, block_len, time_left_func,
                            step0=0.5, beta=0.9, max_iter=200000):
    b = b0.copy()
    m = b.size
    s = m / 2.0
    best_b = b.copy()
    best_score = np.inf
    momentum = np.zeros_like(b)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        h = np.repeat(b, block_len)
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad_h = compute_gradients_for_shifts(h, shifts)
        grad_b = grad_h.reshape(m, block_len).mean(axis=1)
        momentum = beta * momentum + (1.0 - beta) * grad_b
        eta = step0 / np.sqrt(it)
        b_new = b - eta * momentum
        b_new = project_onto_box_sum(b_new, s)
        h_new = np.repeat(b_new, block_len)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_b = b_new.copy()
        b = b_new
        it += 1
    return best_b, best_score

def slsqp_optimize_blocks(b0, block_len, max_iter=1000):
    m = b0.size
    x0 = np.concatenate([b0, [compute_c5_fft(np.repeat(b0, block_len))]])
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:m]) - m / 2.0
    def ineq_fun(x):
        b = x[:m]
        h = np.repeat(b, block_len)
        corr = compute_corr_scaled_fft(h)
        return x[-1] - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return b0, compute_c5_fft(np.repeat(b0, block_len))
    b_opt = project_onto_box_sum(res.x[:m], m / 2.0)
    h_opt = np.repeat(b_opt, block_len)
    t_opt = compute_c5_fft(h_opt)
    return b_opt, t_opt

def binary_anneal_aggressive(h0, rng, time_left_func,
                             T0=1.0, decay=0.99999, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > Tmin and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
        if it % 5000 == 0 and time_left_func() > 0.1:
            # targeted attempt on worst shift
            corr = compute_corr_scaled_fft(h)
            max_corr = np.max(corr)
            shift_idx = np.argmax(corr)
            shift = shift_idx - (n - 1)
            idx = np.arange(n)
            if shift >= 0:
                valid = idx + shift < n
            else:
                valid = idx + shift >= 0
            over1 = (h == 1.0) & valid & ((1 - h)[idx + shift] == 1.0)
            over0 = (h == 0.0) & valid & ((1 - h)[idx + shift] == 0.0)
            cand_i = np.where(over1)[0]
            cand_j = np.where(over0)[0]
            improved = False
            for _ in range(5):
                if cand_i.size and cand_j.size:
                    ii = rng.choice(cand_i)
                    jj = rng.choice(cand_j)
                else:
                    break
                h[ii] = 0.0
                h[jj] = 1.0
                sc = compute_c5_fft(h)
                if sc < current_score:
                    current_score = sc
                    if sc < best_score:
                        best_score = sc
                        best_h = h.copy()
                    improved = True
                    break
                h[ii] = 1.0
                h[jj] = 0.0
            if improved:
                continue
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    best_h = None
    best_score = np.inf

    # Determine base resolution
    if 'initial_h_values' in globals():
        base_arr = np.asarray(initial_h_values, dtype=float)
        n = base_arr.size
    else:
        n = 512
        base_arr = rng.random(n)
        base_arr = project_onto_box_sum(base_arr, n / 2.0)

    # ----- Phase 0: coarse block optimization -----
    # choose block size dividing n
    block_len = 8
    if n % block_len != 0:
        block_len = 4
    m = n // block_len
    # initialise blocks by averaging
    b0 = base_arr.reshape(m, block_len).mean(axis=1)
    b0 = project_onto_box_sum(b0, m / 2.0)

    # block gradient descent
    if time_left() > 30.0:
        b_opt, sc_opt = gradient_descent_blocks(b0, block_len, time_left,
                                                step0=0.8, beta=0.95, max_iter=200000)
        h_opt = np.repeat(b_opt, block_len)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # block SLSQP refinement
    if time_left() > 20.0:
        b_ref, sc_ref = slsqp_optimize_blocks(b_opt if best_h is not None else b0,
                                              block_len, max_iter=800)
        h_ref = np.repeat(b_ref, block_len)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 1: fine continuous gradient descent -----
    if best_h is None:
        best_h = base_arr.copy()
        best_score = compute_c5_fft(best_h)

    if time_left() > 50.0:
        h_fine, sc_fine = gradient_descent_momentum_with_time(best_h, time_left,
                                                              step0=0.6, beta=0.9,
                                                              max_iter=200000)
        if sc_fine < best_score:
            best_score, best_h = sc_fine, h_fine.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 2: binary conversion and aggressive annealing -----
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5_fft(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()

    if time_left() > 30.0:
        h_bin_ref, sc_bin_ref = binary_anneal_aggressive(best_h, rng, time_left,
                                                         T0=1.0, decay=0.99999, Tmin=1e-6)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 3: upsample to higher resolution (optional) -----
    if time_left() > 20.0:
        n_high = 1024
        h_up = upsample(best_h, n_high)
        h_up = project_onto_box_sum(h_up, n_high / 2.0)
        # fine gradient descent at higher res
        h_up_ref, sc_up_ref = gradient_descent_momentum_with_time(h_up, time_left,
                                                                  step0=0.6, beta=0.9,
                                                                  max_iter=150000)
        if sc_up_ref < best_score:
            best_score, best_h = sc_up_ref, h_up_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size
        # binary anneal high-res
        if time_left() > 15.0:
            h_up_bin, sc_up_bin = binary_anneal_aggressive(best_h, rng, time_left,
                                                           T0=1.0, decay=0.99999, Tmin=1e-6)
            if sc_up_bin < best_score:
                best_score, best_h = sc_up_bin, h_up_bin.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ----- Phase 4: random restarts (binary + anneal) -----
    while time_left() > 8.0:
        n_rand = 512
        v = rng.random(n_rand)
        v = project_onto_box_sum(v, n_rand / 2.0)
        h_rand, sc_rand = gradient_descent_momentum_with_time(v, time_left,
                                                              step0=0.5, beta=0.9,
                                                              max_iter=5000)
        # binary conversion
        idx = np.argpartition(-h_rand, n_rand // 2)[:n_rand // 2]
        h_bin_rand = np.zeros_like(h_rand)
        h_bin_rand[idx] = 1.0
        # short anneal
        h_bin_rand, sc_bin_rand = binary_anneal_aggressive(h_bin_rand, rng, time_left,
                                                           T0=0.8, decay=0.9999, Tmin=1e-5)
        if sc_bin_rand < best_score:
            best_score, best_h = sc_bin_rand, h_bin_rand.copy()
        if best_score <= target:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
