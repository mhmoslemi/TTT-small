def gradient_descent_nesterov(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    step = step0
    for it in range(max_iters):
        if time_left() <= 0: break
        # Compute gradient of max correlation term (average over argmax shifts)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Nesterov update
        v = beta * v + grad
        h_next = h - step * v
        h_next = project_onto_box_sum(h_next, s)
        new_score = compute_c5(h_next)
        if new_score <= best_score:
            best_h, best_score = h_next.copy(), new_score
            # maybe increase step
            step = min(step0, step * 1.01)
        else:
            step *= 0.5
        h = h_next
    return best_h, best_score
```

def gradient_descent_nesterov(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    step = step0
    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # Determine shifts achieving nearly max
        # Use tolerance relative to max_corr.
        tol = 1e-12
        shifts = np.where(corr >= max_corr - tol)[0] - (n - 1)
        # Compute gradient average over shifts
        grad = compute_gradients_for_shifts(h, shifts)
        v = beta * v + grad
        h_new = h - step * v
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score < best_score - 1e-12:
            best_h = h_new.copy()
            best_score = new_score
        # Simple step adaptation
        if new_score <= compute_c5(h):  # accept
            h = h_new
            # increase step if improvement
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
    return best_h, best_score

def simulated_annealing(h0, rng, time_left, initial_temp=0.01, final_temp=1e-6):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    # total allocation time:
    total = time_left()
    start = time.time()
    while True:
        remain = time_left()
        if remain <= 0:
            break
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / total
        temp = final_temp + (initial_temp - final_temp) * max(0.0, 1.0 - frac)
        # propose move: random pair i,j, random delta
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5

def gradient_descent_nesterov(h0, max_iters, step0, beta, rng, start, budget):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    v = np.zeros_like(h)

    best_h = h.copy()
    best_score = cur_score
    step = step0
    for it in range(max_iters):
        remaining = budget - (time.time() - start)
        if remaining <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts achieving max

        tol = 1e-12

                step = step0 * 0.1

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def simulated_annealing_delta(h0, rng, start, budget, initial_temp=0.01, final_temp=1e-6):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()

    # total time allocated = budget - (time.time() - start)
    # We'll loop until time exhausted.
    while True:
        remaining = budget - (time.time() - start)
        if remaining <= 0:
            break

        elapsed = time.time() - start
        frac = elapsed / budget
        temp = final_temp + (initial_temp - final_temp) * max(0.0, 1.0 - frac)
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:

        h_new[j] += delta

        if dE < 0 or rng.random() < math.exp(-dE / temp):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

import time
import math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Approximate C5 = max_k ∫ h(x)(1-h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from size m to new_n (both >0)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as float."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def gradient_descent_nesterov(h0, max_iters, step0, beta, rng, start, budget):
    """Projected Nesterov gradient descent on the C5 objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    v = np.zeros_like(h)
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    step = step0
    for _ in range(max_iters):
        if budget - (time.time() - start) <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        tol = 1e-12
        shifts = np.where(corr >= max_corr - tol)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        v = beta * v + grad
        h_new = h - step * v
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if cur_score < best_score:
            best_h = h.copy()
            best_score = cur_score
    return best_h, best_score

def simulated_annealing_delta(h0, rng, start, budget, initial_temp=0.01, final_temp=1e-6):
    """Transfer a small mass between random positions, accepting uphill moves."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    while True:
        if budget - (time.time() - start) <= 0:
            break
        elapsed = time.time() - start
        frac = elapsed / budget
        temp = final_temp + (initial_temp - final_temp) * max(0.0, 1.0 - frac)
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / temp):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
    return best_h, best_score

def simulated_annealing_binary_swap(h0, rng, start, budget, initial_temp=0.01, final_temp=1e-6):
    """Swap a 1 and a 0 in a binary vector, accepting uphill moves."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    while True:
        if budget - (time.time() - start) <= 0:
            break

        frac = elapsed / budget
        temp = final_temp + (initial_temp - final_temp) * max(0.0, 1.0 - frac)
        ones = np.where(h > 0.5)[0]

        if len(ones) == 0 or len(zeros) == 0:

        j = rng.choice(zeros)
        h_new = h.copy()

    return best_h, best_score
