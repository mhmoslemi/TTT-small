def compute_smoothed_grad(h, alpha):
    n = h.shape[0]
    # compute unscaled correlation between h and (1-h)
    corr = compute_corr_full_fft(h)  # length 2n-1
    max_corr = np.max(corr)
    # compute weights
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # cross-correlation of w with (1-h) and h
    L = 1 << ((3 * n - 2 - 1).bit_length())  # next power of two
    fw = np.fft.fft(w, L)
    f_one_minus_h = np.fft.fft(1.0 - h, L)
    f_h = np.fft.fft(h, L)
    cc_w_one_minus_h = np.fft.ifft(fw * np.conj(f_one_minus_h)).real
    cc_w_h = np.fft.ifft(fw * np.conj(f_h)).real
    # slice central n entries (indices n-1 ... n-1 + n -1)
    start = n - 1
    grad = cc_w_one_minus_h[start:start + n] - cc_w_h[start:start + n]
    return grad
```

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

        else:

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_smoothed_grad(h, alpha):
    """Gradient of the smooth max (log-sum-exp) of the correlation."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)               # unscaled correlation
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    L = next_pow2(3 * n - 2)
    fw = np.fft.fft(w, L)
    f_one_minus_h = np.fft.fft(1.0 - h, L)
    f_h = np.fft.fft(h, L)

    cc_w_one_minus_h = np.fft.ifft(fw * np.conj(f_one_minus_h)).real
    cc_w_h = np.fft.ifft(fw * np.conj(f_h)).real

    start = n - 1
    grad = cc_w_one_minus_h[start:start + n] - cc_w_h[start:start + n]
    return grad

    """Projected gradient descent on the smoothed objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0

    for alpha, n_iter in phases:

                return best_h, best_score

                best_h = h.copy()

    n = v.size
    half = n // 2

                phases = [(a, max(1, int(it * factor))) for a, it in phases]

            h_opt, sc_opt = smooth_descent(h0, phases,
                                           step0=2.0, beta=0.9,
                                           rng=rng, time_left=time_left)

            # binary projection and local binary optimization
            h_bin = binary_projection(h_opt)
            sc_bin = compute_c5(h_bin)
            if sc_bin < best_score:
                best_score, best_h = sc_bin, h_bin.copy()
                if best_score <= TARGET:
                    break
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_score <= TARGET:
            break
        # prepare base for finer scale
        if best_h is not None:
            base = best_h.copy()
        else:
            base = base

    # final binary hill climbing on the best solution found
    if best_h is not None and time_left() > 4.0:
        best_h = hill_climb_binary(best_h, rng, time_left, max_no_improve=3000)
        best_score = compute_c5(best_h)

    # safety fallback
    if best_h is None:
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5(best_h)
    else:
        n_ret = best_h.shape[0]
        best_h = project_onto_box_sum(best_h, n_ret / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, n_ret
