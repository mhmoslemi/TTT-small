def adam_optimize(h0, max_iters, lr, beta1, beta2, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = max_corr(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    eps = 1e-8
    for t in range(1, max_iters+1):
        if time_left() <= 0.0:
            break
        corr = fft_corr_full(h)
        max_c = np.max(corr)
        grad = compute_grad_max_corr(h, corr, max_c)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        # projection onto [0,1] & sum
        h = project_onto_box_sum(h, s)
        # score
        score = max_corr(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            # optional: reduce lr or early break? Not needed.
    return best_h, best_score
```

def local_search(h, score, time_left, rng):
    n = h.size
    best_h = h
    best_score = score
    while time_left() > 0.5:
        i, j = rng.choice(n, size=2, replace=False)
        # allowed delta from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * rng.uniform(0.01, 0.2) # random fraction
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # no projection needed as sum unchanged, bounds okay
        new_score = max_corr(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new
    return best_h, best_score
```

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # Determine starting vector
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
    else:
        init = thue_morse(128)  # fallback
    # Ensure integral constraint
    n0 = init.shape[0]
    init = project_onto_box_sum(init, n0 / 2.0)

    scales = [128, 256, 512, 1024]
    best_h = None
    best_score = np.inf
    TARGET = 0.3808

    base = init
    for n in scales:
        if time_left() <= 2.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        # number of restarts based on remaining time
        # allocate roughly 0.2*remaining_time per restart maybe.
        restarts = 4
        for r in range(restarts):
            if time_left() <= 1.0:
                break
            # random initialization around base
            h0 = base + rng.normal(scale=0.08, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)
            # run Adam
            max_iters = 200000  # but maybe adjust based on time
            h_opt, sc_opt = adam_optimize(h0, max_iters, lr=0.5, beta1=0.9, beta2=0.999, time_left=time_left, rng=rng)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_score <= TARGET:
            break
        # local search on best_h at this resolution
        if best_h is not None and best_h.size == n and time_left() > 2.0:
            best_h, best_score = local_search(best_h, best_score, time_left, rng)
        # Prepare base for next finer scale
        if best_h is not None:
            base = best_h.copy()
        else:
            base = base

    # If time still left, attempt binary balanced search
    if best_h is not None and time_left() > 2.0:

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def fft_corr_full(h):
    """Full cross‑correlation of h with (1‑h) via FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def max_corr(h):
    """Maximum overlap integral C₅ for discretized h."""
    n = h.shape[0]
    corr = fft_corr_full(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_corr(h, corr, max_c, tol=1e-12):
    """Subgradient of max_k ∫ h(1‑h(·+k)) for the shifts attaining the max."""
    n = h.size
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts):
        grad /= len(shifts)
    return grad


def adam_optimize(h0, max_iters, lr, beta1, beta2, time_left, rng):
    """Projected Adam optimizer for the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = max_corr(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    eps = 1e-8
    beta1_pow = beta1
    beta2_pow = beta2

    for t in range(1, max_iters + 1):
        if time_left() <= 0.0:
            break

        corr = fft_corr_full(h)
        max_c = np.max(corr)
        grad = compute_grad_max_corr(h, corr, max_c)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step

        if t % 5000 == 0:
            h = h + rng.normal(scale=0.02, size=n)

        h = project_onto_box_sum(h, s)

        score = max_corr(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()

        beta1_pow *= beta1
        beta2_pow *= beta2
    return best_h, best_score


def local_search(h, score, time_left, rng):
    """Simple hill‑climbing by random mass transfers."""
    n = h.size
    best_h = h
    best_score = score
    while time_left() > 0.5:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * rng.uniform(0.01, 0.2)
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = max_corr(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new
    return best_h, best_score

    start = time.time()
