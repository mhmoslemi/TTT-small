Implementation details for compute_corr:

def compute_corr(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1] * (2.0 / n)
    return corr

def smooth_optimize(h0, time_left, total_steps=2000, betas=[2,5,10,20,50,100]):

    h = project_onto_box_sum(h0, h0.size/2.0)  # initial projection

    best_h, best_val = h.copy(), compute_c5_fast(h) # actual max C5, not smoothed.

    for beta in betas:
        # Adam / momentum parameters
        lr = 0.05  # or maybe adjust per beta
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        t = 0
        while t < total_steps and time_left() > 0.1:
            t += 1
            val, grad = grad_smooth(h, beta)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h - step

def grad_smooth(h, beta):
    n = h.shape[0]
    dx = 2.0 / n
    corr = compute_corr(h)  # returns array of length 2n-1 multiplied by dx
    max_corr = corr.max()
    # use stable softmax
    exps = np.exp(beta * (corr - max_corr))
    w = exps / exps.sum()
    # compute convolution terms via FFT
    N = 1 << ((3 * n - 2).bit_length())
    # FFT of h reversed
    H_rev = np.fft.rfft(h[::-1], n=N)
    # FFT of w
    W = np.fft.rfft(w, n=N)
    conv1 = np.fft.irfft(H_rev * W, n=N)
    term1 = conv1[(n-1):(n-1+n)]
    # FFT of w reversed
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv2 = np.fft.irfft(H_rev * W_rev, n=N)
    term2 = conv2[(n-1):(n-1+n)]
    grad = dx * (1.0 - term1 - term2)
    val = (w * corr).sum()
    return val, grad

def smooth_optimize(h0, time_left):
    n = h0.size
    target_sum = n / 2.0
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_val = compute_c5_fast(h) # actual bound
    betas = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    # define total steps per beta maybe 4000
    steps_per_beta = 3000
    # Initiate Adam parameters
    for beta in betas:
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 0
        lr = 0.03  # perhaps smaller for higher beta?
        while t < steps_per_beta and time_left() > 0.2:
            t += 1
            val, grad = grad_smooth(h, beta)
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_new = h - step
            h_new = project_onto_box_sum(h_new, target_sum)
            cur = compute_c5_fast(h_new)
            if cur < best_val:
                best_val, best_h = cur, h_new.copy()
                if best_val <= 0.3808:
                    return best_h, best_val
            h = h_new
    return best_h, best_val

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    # expand bounds until they bracket s
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):

    conv = np.fft.irfft(fft_rev * fft_one, n=N)

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            init = resample(init, n)
        candidates.append(project_onto_box_sum(init, target_sum))
    # Thue–Morse binary pattern
    candidates.append(np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                                 dtype=float, count=n))
    # Random balanced binary vectors
    for _ in range(3):
        bin_vec = np.zeros(n, dtype=float)
        bin_vec[rng.choice(n, size=n // 2, replace=False)] = 1.0
        candidates.append(bin_vec)
    # Random continuous vectors
    for _ in range(3):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))
    best_h = None
    best_val = np.inf
    for c in candidates:
        if time_left() < 5.0:
            break
        h_opt, v_opt = smooth_optimize(c, time_left)
        if v_opt < best_val:
            best_val, best_h = v_opt, h_opt.copy()
        if best_val <= target_val:
            return best_h, best_val, n
    # If time permits, upscale to a finer grid
    if time_left() > 60.0:
        n2 = 512
        target_sum2 = n2 / 2.0
        if best_h is not None:
            h_up = resample(best_h, n2)
        else:
            h_up = np.full(n2, 0.5)
        h_up = project_onto_box_sum(h_up, target_sum2)
        h_opt2, v_opt2 = smooth_optimize(h_up, time_left)
        if v_opt2 < best_val:
            best_val, best_h, n = v_opt2, h_opt2.copy(), n2
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), n / 2.0)
        best_val = compute_c5_fast(best_h)
    else:
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_val = compute_c5_fast(best_h)
    return best_h, best_val, n
