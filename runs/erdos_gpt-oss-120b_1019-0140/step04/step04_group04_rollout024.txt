def compute_c5_fft(h):
    n = len(h)
    H = np.fft.rfft(h, n*2) # zero-pad to length 2n for full correlation without wrap. Actually to get linear convolution (full) we need length 2n-1, zero-pad to 2n.
    G = np.fft.rfft(1 - h, n*2)
    corr = np.fft.irfft(H * G, n*2)[:2*n-1]  # linear convolution
    return np.max(corr) * (2.0 / n)
```

def thue_morse_bin(n, rng=None):
    # compute parity bits
    arr = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    # arr is 0/1; for n power of two it's balanced. For non-power-of-two adjust using random selection to ensure sum=n/2.
    ones_needed = n // 2
    ones = np.where(arr == 1)[0]
    if ones.size >= ones_needed:
        sel = rng.choice(ones, size=ones_needed, replace=False) if rng is not None else np.random.choice(ones, size=ones_needed, replace=False)
        result = np.zeros(n, dtype=float)
        result[sel] = 1.0
    else:
        # fill all ones from arr then random select remaining from zeros
        zeros = np.where(arr == 0)[0]
        need = ones_needed - ones.size
        sel0 = rng.choice(zeros, size=need, replace=False) if rng is not None else np.random.choice(zeros, size=need, replace=False)
        result = arr.copy()
        result[sel0] = 1.0
    return result

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr, _ = compute_corr_and_c5(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_and_c5(h):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * G, N)[:2 * n - 1]
    c5 = np.max(corr) * (2.0 / n)
    return corr, c5

def compute_c5(h):
    _, c5 = compute_corr_and_c5(h)
    return c5

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_bin(n, rng=None):
    arr = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                      dtype=float, count=n)
    need = n // 2
    cur = int(arr.sum())
    if cur > need:
        idx = np.where(arr == 1)[0]
        to_zero = rng.choice(idx, size=cur - need, replace=False) if rng else np.random.choice(idx, size=cur - need, replace=False)
        arr[to_zero] = 0.0
    elif cur < need:
        idx = np.where(arr == 0)[0]
        to_one = rng.choice(idx, size=need - cur, replace=False) if rng else np.random.choice(idx, size=need - cur, replace=False)
        arr[to_one] = 1.0
    return arr

def binary_hill_climb(h_init, rng, time_left):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.shape[0]
    max_iters = min(100000, int(time_left() * 2000))
    for _ in range(max_iters):
        if time_left() <= 0:
            break
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        score = compute_c5(h_new)
        if score < best_score:
            h = h_new
            best_h = h_new.copy()
            best_score = score
            if best_score <= 0.3808:
                break
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts.size:
        grad /= shifts.size
    return grad

def adam_optimize(h0, n_iter, lr, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    b1_pow = 1.0
    b2_pow = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr, _ = compute_corr_and_c5(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        b1_pow *= beta1
        b2_pow *= beta2
        m_hat = m / (1 - b1_pow)
        v_hat = v / (1 - b2_pow)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= 0.3808:
                break
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr, _ = compute_corr_and_c5(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Base vector (64 samples) if provided
    if 'initial_h_values' in globals():
        base64 = np.asarray(initial_h_values, dtype=float)
    else:
        base64 = thue_morse_bin(64, rng)

    # ---------- Binary hill climbing at n=256 ----------
    n_bin = 256
    binary_start = thue_morse_bin(n_bin, rng)
    best_bin_h, best_bin_score = binary_hill_climb(binary_start, rng, time_left)

    # ---------- Upsample to n=512 ----------
    n_up = 512
    h_up = upsample(best_bin_h, n_up)
    h_up = project_onto_box_sum(h_up, n_up / 2.0)

    # ---------- Main Adam refinement ----------
    remaining = time_left()
    max_main_iters = int(min(200000, remaining * 500))
    if max_main_iters < 1:
        max_main_iters = 1
    h_ref, sc_ref = adam_optimize(h_up, n_iter=max_main_iters,
                                  lr=0.30, beta1=0.90, beta2=0.999,
                                  eps=1e-8, rng=rng, time_left=time_left)

    best_h, best_score = h_ref, sc_ref

    # ---------- Secondary Adam with finer step ----------
    if time_left() > 5:
        max_fine_iters = int(min(100000, time_left() * 400))
