def gradient_descent_with_time(h0, time_left_func, max_iter=200000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        # compute correlation
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        # find worst shifts (including ties)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # gradient from all worst shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.empty_like(h)
                term2 = np.empty_like(h)
                term1[:n-shift] = h[shift:]
                term1[n-shift:] = 0.0
                term2[shift:] = h[:n-shift]
                term2[:shift] = 0.0
            else:
                sft = -shift
                term1 = np.empty_like(h)
                term2 = np.empty_like(h)
                term1[sft:] = h[:n-sft]
                term1[:sft] = 0.0
                term2[:n-sft] = h[sft:]
                term2[n-sft:] = 0.0
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection onto box and sum
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score
```

def compute_corr_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2*n-1]
    return corr * (2.0 / n)
```

def binary_swap_anneal(h0, rng, time_left_func, max_iters=1000000, T0=0.2, decay=0.99995, Tmin=1e-6):
    h = (h0 > 0.5).astype(float)
    n = h.size
    # initialize ones and zeros list
    ones = list(np.flatnonzero(h).astype(int))
    zeros = list(np.flatnonzero(1 - h).astype(int))
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    T = T0
    it = 0
    while time_left_func() > 0.0 and it < max_iters:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - best_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # update lists
            # replace i in ones with j
            idx_i = ones.index(i)
            ones[idx_i] = j
            # replace j in zeros with i
            idx_j = zeros.index(j)
            zeros[idx_j] = i
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # cooling
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    """Full correlation of h with (1‑h) scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    """Maximum overlap integral C₅."""
    return np.max(compute_corr_fft(h))

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_with_time(h0, time_left_func, max_iter=200000, step0=0.5, beta=0.9):
    n = h0.shape[0]
    s = n / 2.0
    h = project_onto_box_sum(h0, s)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # --- initial solution -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        h0 = project_onto_box_sum(h0, n0 / 2.0)
    else:
        n0 = 512
        h0 = random_binary_balanced(n0, rng)

    best_h = h0
    best_score = compute_c5_fft(best_h)
    target = 0.3808

    # --- Phase A: continuous refinement ----------------------------------
    if time_left() > 5.0:
        h_ref, sc_ref = gradient_descent_with_time(best_h, time_left,
                                                   max_iter=200000)
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref

    # --- Phase B: binary conversion & annealing -------------------------
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5_fft(h_bin)
    if sc_bin < best_score:
        best_h, best_score = h_bin, sc_bin

    if time_left() > 5.0:
        T0 = 0.1 * best_score
        h_bin_ref, sc_bin_ref = binary_swap_anneal(best_h, rng, time_left,
                                                  max_iters=500000,
                                                  T0=T0, decay=0.99996,
                                                  Tmin=1e-7)
        if sc_bin_ref < best_score:
            best_h, best_score = h_bin_ref, sc_bin_ref

    # --- Phase C: random restarts with short gradient descents -----------
    while time_left() > 10.0:
        n = best_h.size
        if rng.random() < 0.5:
            h_rand = random_binary_balanced(n, rng)
        else:
            v = rng.random(n)
            h_rand = project_onto_box_sum(v, n / 2.0)
        max_iter = int(min(40000, time_left() * 5))
        h_rand_ref, sc_rand = gradient_descent_with_time(h_rand, time_left,
                                                         max_iter=max_iter)
        if sc_rand < best_score:
            best_h, best_score = h_rand_ref, sc_rand
            if best_score <= target:
                break

    # --- Phase D: final fine‑tune ----------------------------------------
    if time_left() > 2.0:
        h_fin, sc_fin = gradient_descent_with_time(best_h, time_left,
                                                   max_iter=100000)
        if sc_fin < best_score:
            best_h, best_score = h_fin, sc_fin

    # --- Optional: higher resolution upsample ----------------------------
    if time_left() > 8.0:
        n_big = best_h.size * 2
        h_big = upsample(best_h, n_big)
        h_big = project_onto_box_sum(h_big, n_big / 2.0)
        sc_big = compute_c5_fft(h_big)
        if sc_big < best_score:
            best_h, best_score = h_big, sc_big

    # final project and evaluate
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
