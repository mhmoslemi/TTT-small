def project_onto_box_sum(v, s_val, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum(x_i)=s_val}."""
    n = v.size
    if s_val <= lo * n:
        return np.full_like(v, lo)
    if s_val >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def clip_and_sum(lam):
        return np.clip(v - lam, lo, hi).sum()

    if clip_and_sum(lam_low) < s_val:
        lam_low = np.min(v) - 10 * hi

    if clip_and_sum(lam_high) > s_val:
        lam_high = np.max(v) - 10 * lo

    for i in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        try:
            f_mid = clip_and_sum(lam_mid)
            if abs(f_mid - s_val) < 1e-12:
                break
            if f_mid < s_val:
                lam_low = lam_mid
            else:
                lam_high = lam_mid
        except:
            # Handle any exceptional case gracefully
            break
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]

        return arr.copy()

    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
