def weighted_grad_step(h, eta, beta, momentum, alpha=20.0):
    # compute correlation via FFT
    n = h.size
    # compute corr: linear correlation of h and (1-h)
    # Use FFT with zero padding length L = next power of 2 >= 2*n - 1
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[:2*n-1]
    maxC = corr.max()
    # weights
    w = np.exp(alpha * (corr - maxC))
    W = w.sum()
    # conv1: correlation of w_rev and h
    # conv via FFT: w_rev = w[::-1]
    # Use same L maybe bigger because w length 2n-1, h length n => conv length 3n-2.
    # compute conv using FFT of length L2 = power of two >= (2n-1)+n-1 = 3n-2
    L2 = 1 << ((3 * n - 2).bit_length())
    w_arr = w
    w_rev = w_arr[::-1]
    H_h = np.fft.rfft(h, n=L2)
    W_rev = np.fft.rfft(w_rev, n=L2)
    W_arr = np.fft.rfft(w_arr, n=L2)
    conv1 = np.fft.irfft(W_rev * H_h, n=L2)  # length L2
    conv2 = np.fft.irfft(W_arr * H_h, n=L2)
    # slice indices
    mid = n - 1
    conv1_slice = conv1[mid:mid + n]
    conv2_slice = conv2[mid:mid + n]
    accum = conv1_slice + conv2_slice
    grad = 1.0 - accum / W
    # momentum update
    momentum = beta * momentum + (1 - beta) * grad
    h_new = h - eta * momentum
    h_new = project_onto_box_sum(h_new, n / 2.0)
    return h_new, momentum, corr

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_fft(h):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[:2 * n - 1]
    return corr

def weighted_grad_step(h, eta, beta, momentum, alpha=20.0):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    maxC = np.max(corr)
    w = np.exp(alpha * (corr - maxC))
    W = np.sum(w)
    # convolution lengths
    L2 = 1 << ((3 * n - 2).bit_length())
    h_fft = np.fft.rfft(h, n=L2)
    w_rev_fft = np.fft.rfft(w[::-1], n=L2)
    w_fft = np.fft.rfft(w, n=L2)
    conv1 = np.fft.irfft(w_rev_fft * h_fft, n=L2)
    conv2 = np.fft.irfft(w_fft * h_fft, n=L2)
    mid = n - 1
    conv1_slice = conv1[mid:mid + n]
    conv2_slice = conv2[mid:mid + n]
    accum = conv1_slice + conv2_slice
    grad = 1.0 - accum / W
    momentum = beta * momentum + (1.0 - beta) * grad
    h_new = h - eta * momentum
    h_new = project_onto_box_sum(h_new, n / 2.0)
    return h_new, momentum, corr

def perturb(h, rng, scale=0.02):

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(256)

    scales = [256, 512, 1024]
    best_h, best_score = None, float('inf')

    for n in scales:
        if time_left() < 5.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        momentum = np.zeros(n, dtype=float)
        it = 0
        step0 = 0.5
        beta = 0.9
        alpha = 20.0

        cur_score = compute_c5(base)
        if cur_score < best_score:
            best_score, best_h = cur_score, base.copy()

        # weighted gradient descent
        while time_left() > 2.0:
            it += 1
            eta = step0 / math.sqrt(it)
            base, momentum, corr = weighted_grad_step(base, eta, beta, momentum, alpha=alpha)
            cur_score = np.max(corr) * (2.0 / n)
            if cur_score < best_score:
                best_score, best_h = cur_score, base.copy()
                if best_score <= 0.3808:
                    break
            if it % 3000 == 0:
                base = perturb(base, rng, scale=0.01)

        if best_score <= 0.3808:
            break

        # short annealing phase
        if time_left() > 2.0:
            h_ann, sc_ann = simulated_anneal(base, compute_c5(base), rng, time_left)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()
            base = h_ann

        # occasional random restart with brief descent
        if time_left() > 2.0:
            h_rand = random_binary_balanced(n, rng)
            momentum = np.zeros(n, dtype=float)
            it_rand = 0
            while time_left() > 1.0 and it_rand < 2000:
                it_rand += 1
                eta = 0.4 / math.sqrt(it_rand)
                h_rand, momentum, corr = weighted_grad_step(h_rand, eta, beta, momentum, alpha=alpha)
                sc_rand = np.max(corr) * (2.0 / n)
                if sc_rand < best_score:
                    best_score, best_h = sc_rand, h_rand.copy()
                    if best_score <= 0.3808:
                        break

        base = best_h.copy()

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
