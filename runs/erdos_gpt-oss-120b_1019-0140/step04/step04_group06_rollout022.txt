def compute_c5_max_shift(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx] * (2.0 / n)

    return max_val, max_idx - (n - 1)

def subgradient_max_shift(h, k):
    n = len(h)
    grad = np.zeros_like(h)
    # compute contributions:
    if k >= 0:
        # first term: (1 - h[i+k]) for i=0..n-1-k
        grad[:n - k] += 1 - h[k:]  # i part
        # second term: -h[i-k] for i=k..n-1
        grad[k:] -= h[:n - k]
    else:
        # k negative: equivalently shift by -k in other direction, we can treat symmetrically.
        # Use C(k) = sum_{i=0}^{n-1+k} h[i - k] * (1 - h[i])
        # Actually for negative k (shift left), we can compute by swapping roles.
        # Simpler: compute same formula for k_neg = -k as if shifting h opposite.
        # When k<0, C = sum_{i=0}^{n-1-(-k)} h[i+(-k)]*(1 - h[i])
        # Equivalent to for i=0..n-1+k: h[i - k] * (1 - h[i])  (since -k positive)
        pos_k = -k
        # first term: (1 - h[i]) with i from 0..n-1-pos_k, and use h[i+pos_k] as first factor?
        # Let's derive: C(-pos_k) = sum_{i=0}^{n-1-pos_k} h[i+pos_k] * (1 - h[i]).
        # So h[i+pos_k] is factor h[j] where j = i+pos_k, and (1-h[i]) factor is (1-h[i]).
        # So grad wrt h[j] (j = i+pos_k): if j >= pos_k, correspond to i=j-pos_k
        # term1: + (1 - h[i]) for i=j-pos_k, for j=pos_k..n-1
        # term2: -h[i+pos_k]?? Actually derivative wrt h[i] appears also.
        # For derivative wrt h[i] (i from 0..n-1-pos_k): derivative from (1 - h[i]) factor: derivative = -h[i+pos_k].
        # So final: grad for j in pos_k..n-1: + (1 - h[j - pos_k])
        # grad for i in 0..n-1-pos_k: - h[i + pos_k]
        grad[pos_k:] += 1 - h[:n - pos_k]

def gd_subgradient(h, max_iters, step0, momentum_coef, rng, time_left):
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        max_val, shift = compute_c5_max_shift(h)
        # if this is less than best, update
        if max_val < best_score:
            best_score = max_val
            best_h = h.copy()
        # compute subgradient
        grad = subgradient_max_shift(h, shift)
        # momentum update
        momentum = momentum_coef * momentum + (1 - momentum_coef) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        # optionally add small random noise occasionally
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project onto box-sum
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
        # early exit if best_score < target
        if best_score <= 0.3808:
            break
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def grad_descent(h0, max_iters, step0, beta, rng, start_time, budget_s):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        # time check
        if time.time() - start_time >= budget_s:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        # occasional random noise
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project onto box and sum
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
        if best_score <= 0.3808:
            break
    return best_h, best_score

def binary_hill_climb(h, attempts, start_time, budget_s, rng):
    # h is binary (0/1) with sum n/2.
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for _ in range(attempts):
        if time.time() - start_time >= budget_s:
            break
        # choose random i where h[i]==1, j where h[j]==0
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    return best_h, best_score

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def max_shifts(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += 1.0 - h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            k = -shift
            grad[k:] += 1.0 - h[:n - k]
            grad[:n - k] -= h[k:]
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent(h0, max_iters, step0, beta, rng, start_time, budget_s):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time.time() - start_time >= budget_s:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h.copy()
        if best_score <= 0.3808:
            break
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def round_to_binary(h):
    n = h.shape[0]
    idx = np.argsort(-h)
    bh = np.zeros_like(h)
    bh[idx[:n // 2]] = 1.0
    return bh

def binary_hill_climb(h, attempts, start_time, budget_s, rng):
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for _ in range(attempts):
        if time.time() - start_time >= budget_s:
            break
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # starting vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != 64:
            base = upsample(base, 64)
    else:
        base = np.fromiter(((bin(i).count('1') & 1) for i in range(64)), dtype=float, count=64)
    base = project_onto_box_sum(base, 64 / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)
    best_n = best_h.size
    scales = [64, 128, 256, 512]

    for n in scales:
        if time.time() - start_time >= budget_s - 5:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        time_rem = budget_s - (time.time() - start_time)
        max_iters = int(min(60000, max(12000, time_rem * (5000.0 / n))))
        h_gd, sc_gd = grad_descent(best_h, max_iters, step0=0.7, beta=0.8,
                                   rng=rng, start_time=start_time, budget_s=budget_s)
        if sc_gd < best_score:
            best_score, best_h, best_n = sc_gd, h_gd.copy(), n

        time_rem = budget_s - (time.time() - start_time)
        for _ in range(5):
            if time.time() - start_time >= budget_s - 5:
                break
            h_rand = random_binary_balanced(n, rng)
            time_rem2 = budget_s - (time.time() - start_time)
            max_iters2 = int(min(15000, max(3000, time_rem2 * (2000.0 / n))))
            h_rand_gd, sc_rand = grad_descent(h_rand, max_iters2, step0=0.5, beta=0.85,
                                              rng=rng, start_time=start_time, budget_s=budget_s)
            if sc_rand < best_score:

            h_bin = round_to_binary(best_h)

            if sc_hc < best_score:
                best_score, best_h, best_n = sc_hc, h_hc.copy(), n

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
