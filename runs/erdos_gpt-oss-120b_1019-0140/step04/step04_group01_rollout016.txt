def binary_swap_anneal(h0, rng, time_left_func, max_iters=2000000, T0=0.5, decay=0.99999, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 512
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
    h0 = project_onto_box_sum(h0, n / 2.0)
    best_h = h0.copy()
    best_score = compute_c5_fft(best_h)

    # Continuous refinement (Adam)
    if time_left() > 15.0:
        h_cont, sc_cont = adam_optimize(best_h, time_left, max_iter=200000, alpha=0.3)
        if sc_cont < best_score:
            best_h, best_score = h_cont, sc_cont
        if best_score <= target:
            return best_h, best_score, n

    # Binary conversion
    if time_left() > 5.0:
        idx_top = np.argpartition(-best_h, n // 2)[: n // 2]
        h_bin = np.zeros_like(best_h)
        h_bin[idx_top] = 1.0
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_h, best_score = h_bin, sc_bin

    # Binary annealing
    if time_left() > 10.0:
        h_bin_opt, sc_bin_opt = binary_swap_anneal(best_h, rng, time_left,
                                                  max_iters=2000000,
                                                  T0=0.5, decay=0.99999, Tmin=1e-6)
        if sc_bin_opt < best_score:
            best_h, best_score = h_bin_opt, sc_bin_opt
        if best_score <= target:
            return best_h, best_score, n

    # Final continuous refinement
    if time_left() > 10.0:
        h_ref, sc_ref = adam_optimize(best_h, time_left, max_iter=100000, alpha=0.2)
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref

    # SLSQP refinement (if time permits)
    if time_left() > 5.0:
        try:
            h_slsqp, sc_slsqp = slsqp_optimize_candidate(best_h, max_iter=5000)
            if sc_slsqp < best_score:
                best_h, best_score = h_slsqp, sc_slsqp
        except Exception:
            pass

    # Random restarts
    while time_left() > 25.0:
        h_rand = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h_rand[idx] = 1.0
        h_rand_opt, sc_rand_opt = adam_optimize(h_rand, time_left, max_iter=5000, alpha=0.3)
        idx_top = np.argpartition(-h_rand_opt, n // 2)[: n // 2]
        h_bin2 = np.zeros_like(h_rand_opt)
        h_bin2[idx_top] = 1.0
        h_bin2_opt, sc_bin2_opt = binary_swap_anneal(h_bin2, rng, time_left,
                                                    max_iters=200000,
                                                    T0=0.5, decay=0.99995, Tmin=1e-6)
        if sc_bin2_opt < best_score:
            best_h, best_score = h_bin2_opt, sc_bin2_opt
        if best_score <= target:
            break

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, n
