def smooth_obj_and_grad(h, alpha):
    n = h.shape[0]
    raw_corr = np.correlate(h, 1 - h, mode='full')
    # raw_corr shape 2n-1
    beta = alpha * (2.0 / n)
    # Compute softmax; for stability subtract max
    rc = raw_corr * beta
    rc_max = rc.max()
    exp_rc = np.exp(rc - rc_max)
    sum_exp = exp_rc.sum()
    w = exp_rc / sum_exp  # shape 2n-1
    # Compute weighted sum of gradients:
    # conv_plus = np.convolve(w[::-1], h, mode='full')

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds contain the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
