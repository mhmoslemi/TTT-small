import time, math, numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def generate_m_sequence():
    # 6-stage LFSR with primitive polynomial x^6 + x + 1
    state = [1,1,1,1,1,1]  # list of bits
    seq = []
    for _ in range(63):
        out = state[-1]
        seq.append(out)
        newbit = state[5] ^ state[0]
        state = [newbit] + state[:-1]
    return np.array(seq, dtype=float)

def initial_vector():
    # Try to use provided initial_h_values, otherwise m-sequence + zero
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.size != 64:
            base = upsample(base, 64)
        # keep top half values as ones
        k = base.size // 2
        idx = np.argpartition(base, -k)[-k:]
        h = np.zeros(base.size, dtype=float)
        h[idx] = 1.0
        return h
    else:
        seq63 = generate_m_sequence()
        # append zero to length 64
        return np.concatenate([seq63, [0.0]])

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target_c5 = 0.3808
    n = 64
    # start from an initial vector
    best_h = initial_vector()
    best_c5 = compute_c5(best_h)
    # Ensure binary weight
    if best_h.sum() != n/2:
        # repair to exact weight
        best_h = random_balanced_binary(n, rng)
        best_c5 = compute_c5(best_h)
    # Main search loop with restarts
    while time_left() > 5.0:
        # random restart or perturbation of current best
        if rng.random() < 0.2:
            # perturb best
            h = best_h.copy()
            # apply a few random swaps
            for _ in range(rng.integers(5, 30)):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i], h[j] = 0.0, 1.0
        else:

                        best_c5 = new_c5

    # final projection to ensure exact weight (should be already)
    # recompute best C5 to be safe
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5), int(n)

import time, math, numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def generate_m_sequence():
    # 6‑stage LFSR with primitive polynomial x^6 + x + 1
    state = [1, 1, 1, 1, 1, 1]
    seq = []
    for _ in range(63):
        out = state[-1]
        seq.append(out)
        newbit = state[5] ^ state[0]
        state = [newbit] + state[:-1]
    return np.array(seq, dtype=float)

def initial_vector():
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.size != 64:
            base = upsample(base, 64)
        k = base.size // 2
        idx = np.argpartition(base, -k)[-k:]
        h = np.zeros(base.size, dtype=float)
        h[idx] = 1.0
        return h
    else:
        seq63 = generate_m_sequence()
        return np.concatenate([seq63, [0.0]])

def run(seed=42, budget_s=1100, **kwargs):
