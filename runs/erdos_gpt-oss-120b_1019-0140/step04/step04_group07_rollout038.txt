def compute_c5_fft(h):
    n = h.shape[0]
    # compute convolution via FFT
    size = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    B = np.fft.rfft((1 - h)[::-1], n=size)
    conv = np.fft.irfft(H * B, n=size)
    conv = conv[:2*n-1]
    return np.max(conv) * (2.0 / n)

def compute_grad_subgrad(h):
    n = h.size
    # compute correlation via fft as above
    size = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    B = np.fft.rfft((1 - h)[::-1], n=size)
    corr = np.fft.irfft(H * B, n=size)[:2*n-1]
    max_corr = np.max(corr)
    # tolerance: treat values close to max within epsilon (relative)
    # Use absolute tolerance maybe 1e-12 * max_corr
    eps = 1e-12
    active_shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
    # compute gradient average
    grad = np.zeros_like(h)
    if len(active_shifts) == 0:
        # no active? shouldn't happen
        return grad
    for shift in active_shifts:
        if shift == 0:
            grad += 1 - 2*h
        elif shift > 0:
            grad[shift:] += 1 - h[shift:] - h[:n - shift]
            grad[:n - shift] += 1 - h[:n - shift] - h[shift:]
        else:
            s = -shift
            grad[:n - s] += 1 - h[:n - s] - h[s:]
            grad[s:] += 1 - h[s:] - h[:n - s]
    grad /= len(active_shifts)
    return grad

def gradient_descent(h0, max_iters, step0, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score
    m = np.zeros_like(h) # momentum first moment
    v = np.zeros_like(h) # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        grad = compute_grad_subgrad(h)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fft(h_new)
        # Accept if improvement or with small probability
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(1e-6, step0)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        # Occasionally jitter to escape
        if it % 2000 == 0:
            # jitter
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5_fft(h)
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # move from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5_fft(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        # also opposite direction
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5_fft(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def sturmian_sequence(n, alpha, phase=0.0):
    i = np.arange(n)
    vals = ((i * alpha + phase) % 1) < 0.5
    return vals.astype(float)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def sturmian_sequence(n, alpha, phase=0.0):
    """Binary Sturmian (characteristic) sequence of length n."""
    i = np.arange(n)
    vals = ((i * alpha + phase) % 1) < 0.5
    return vals.astype(float)


def compute_corr_fft(h):
    """Full correlation of h with (1-h) using FFT (mode='full')."""
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    B = np.fft.rfft((1 - h)[::-1], n=size)
    corr = np.fft.irfft(H * B, n=size)
    return corr[: 2 * n - 1]


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_subgrad(h):
    """Subgradient of C5 (max overlap) w.r.t. h."""
    n = h.size
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    eps = 1e-12
    shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)

    grad = np.zeros_like(h)
    for shift in shifts:
        if shift == 0:
            grad += 1 - 2 * h
        elif shift > 0:
            grad[shift:] += 1 - h[shift:] - h[: n - shift]
            grad[: n - shift] += 1 - h[: n - shift] - h[shift:]
        else:
            s = -shift
            grad[: n - s] += 1 - h[: n - s] - h[s:]
            grad[s:] += 1 - h[s:] - h[: n - s]
    if shifts.size:
        grad /= shifts.size
    return grad


def gradient_descent(h0, max_iters, step0, rng, time_left):
    """Projected Adam-like descent on C5."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score

    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        grad = compute_grad_subgrad(h)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 * (m_hat / (np.sqrt(v_hat) + eps))
        step = np.clip(step, -step0, step0)

        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fft(h_new)

        # accept if improvement or with modest probability
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(step0, 1e-6)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score

        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5_fft(h)
            if cur_score < best_score:
                best_h, best_score = h.copy(), cur_score

    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Random mass transfers between two points, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)

        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5_fft(h_new)
            if score < best_score:
                best_h, best_score = h_new, score

        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5_fft(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    """Transfer mass directly from a point that worsens the worst shift."""
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = compute_corr_fft(best_h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)

        if shift >= 0:
            contrib = best_h[: n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[: n - s])
            i = np.argmax(contrib) + s
            j = i - s

        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = sturmian_sequence(64, (np.sqrt(5) - 1) / 2)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if time_left() < 5:
            break

        # upsample current best to current scale
        if best_h.shape[0] != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)

        # gradient descent from upsampled best
        max_iters = int(min(20000, max(2000, time_left() * 5)))
        h_gd, sc_gd = gradient_descent(base, max_iters, step0=0.5, rng=rng, time_left=time_left)
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd

        # a few random Sturmian restarts
        for _ in range(3):
            if time_left() < 5:
                break
            alpha = rng.random()
            cand = sturmian_sequence(n, alpha)
            cand = project_onto_box_sum(cand, n / 2.0)
            max_iters2 = int(min(5000, max(500, time_left() * 2)))
            cand_gd, sc_cand = gradient_descent(cand, max_iters2, step0=0.4, rng=rng, time_left=time_left)
            if sc_cand < best_score:
                best_h, best_score = cand_gd.copy(), sc_cand

        # local pairwise moves
        if time_left() > 5:
            h_pair, sc_pair = local_pair_moves(best_h, rng, time_left)
            if sc_pair < best_score:
                best_h, best_score = h_pair.copy(), sc_pair

        # targeted max‑shift improvement
        if time_left() > 5:
            h_ms, sc_ms = improve_via_max_shift(best_h, rng, time_left)
            if sc_ms < best_score:
                best_h, best_score = h_ms.copy(), sc_ms

    # final intensive refinement
    if time_left() > 5:
        max_iters_final = int(min(50000, max(20000, time_left() * 30)))
        best_h, best_score = gradient_descent(best_h, max_iters_final, step0=0.3, rng=rng, time_left=time_left)

    # final polishing
    if time_left() > 2:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
