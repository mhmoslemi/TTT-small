def gradient_descent_adam(h0, n_iter, step0, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # compute shifts achieving max
        # tolerance small
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        shifts = shift_idxs - (n - 1)
        # compute gradient
        grad = compute_gradients_for_shifts(h, shifts)
        # update moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        # bias correction
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size schedule
        eta = step0 / np.sqrt(it)
        h_new = h - eta * m_hat / (np.sqrt(v_hat) + eps)
        # project
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new.copy(), score
        # acceptance criteria: accept if improves, else maybe occasionally accept to escape
        if score < compute_c5(h) or rng.random() < 0.001:
            h = h_new
        # occasional noise
        if it % 1000 == 0:
            h = h + rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h, s)
    return best_h, best_score

def simulated_annealing_binary(h0, max_iters, rng, time_left):
    h = h0.copy()
    n = h.size
    target = n // 2
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    # find indices of 0 and 1 groups
    zero_idxs = np.where(h == 0)[0]
    one_idxs = np.where(h == 1)[0]
    # temperature schedule: from T0 to T_end
    T0 = cur_score * 0.1 + 0.001  # some small
    T_end = 1e-6
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        # pick random i from zeros, j from ones
        if zero_idxs.size == 0 or one_idxs.size == 0:

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        grad += 1.0
        if shift >= 0:
            k = shift
            if k < n:
                grad[:n - k] -= h[k:]
                grad[k:] -= h[:n - k]
        else:
            k = -shift
            if k < n:
                grad[k:] -= h[:n - k]
                grad[:n - k] -= h[k:]
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_adam(h0, n_iter, step0, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        shifts = shift_idxs - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        eta = step0 / np.sqrt(it)
        h_new = h - eta * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_h, best_score = h_new.copy(), new_score
        if new_score < cur_score or rng.random() < 0.001:
            h, cur_score = h_new, new_score
        if it % 1000 == 0:
            h = h + rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_h, best_score = h.copy(), cur_score
    return best_h, best_score

def simulated_annealing_binary(h0, max_iters, rng, time_left):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    zero_idxs = np.where(h == 0)[0]
    one_idxs = np.where(h == 1)[0]
    T0 = cur_score * 0.1 + 0.001
    T_end = 1e-6
    for it in range(1, max_iters + 1):
        if time_left() <= 0:

        if zero_idxs.size == 0 or one_idxs.size == 0:
