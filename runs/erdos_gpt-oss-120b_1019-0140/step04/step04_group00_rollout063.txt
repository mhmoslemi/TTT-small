def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()

    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

Better to implement a more efficient projection: The problem is projection onto intersection of box [0,1] and hyperplane sum = target. This is known as "clipping and then adjusting by adding constant delta to all coordinates not at bounds". The algorithm: Let v be unprojected vector. Compute w = np.clip(v, 0, 1). Let sum_w = w.sum(). Then we need to adjust a subset of variables not at bounds to reach target sum. If sum_w != target, we need to distribute delta = target - sum_w across free variables (those not at bounds) while ensuring they stay within bounds. This can be done via simple iterative water-filling.

We can implement a function project_onto_box_sum_fast(v, target) that uses a while loop updating free set until all delta distribution fits bounds. Since n is small (256), this will be fast.

def project_onto_box_sum(v, target, lo=0.0, hi=1.0):
    # initial clip
    x = np.clip(v, lo, hi)
    # compute sum
    s = x.sum()
    # if close enough, return x
    if abs(s - target) < 1e-12:
        return x
    # active mask: indices where 0 < x < 1 initially (i.e., not at bounds)

def compute_corr_raw(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    # compute convolution of h_rev and (1 - h)
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one_minus = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one_minus, n=N)
    corr = conv[:2 * n - 1]
    return corr  # raw C(s) for each shift

Now gradient_softmax:

def gradient_softmax_corr(h, tau):
    n = h.shape[0]
    # raw correlation:
    raw_corr = compute_corr_raw(h)
    # compute weights w = softmax(raw_corr / tau)
    # stable
    max_val = raw_corr.max()
    exp_vals = np.exp((raw_corr - max_val) / tau)
    w = exp_vals / exp_vals.sum()
    # compute forward sum = convolution of w_rev and h (full conv)
    w_rev = w[::-1]
    # Determine FFT size
    N = 1 << ((3 * n - 2).bit_length())  # since len(w_rev)=2n-1, len(h)=n => conv length <=3n-2
    fft_w_rev = np.fft.rfft(w_rev, n=N)
    fft_h = np.fft.rfft(h, n=N)
    conv_fwd_full = np.fft.irfft(fft_w_rev * fft_h, n=N)
    conv_fwd = conv_fwd_full[:n]  # only first n entries correspond to forward sum for i=0..n-1
    # backward sum = convolution of w and h
    fft_w = np.fft.rfft(w, n=N)
    conv_bwd_full = np.fft.irfft(fft_w * fft_h, n=N)
    conv_bwd = conv_bwd_full[:n]
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

Check `conv_fwd` and `conv_bwd` indices correctness: For i from 0 to n-1, conv_fwd[i] = sum_j w_rev[j] * h[i - j] (with zero padding). Since w_rev[j] = w[-j], where indexing j corresponds to shift -j. So conv_fwd[i] = sum_j w[-j] * h[i - j] = sum_s w[s] * h[i + s] (setting s = -j). That's forward sum (i + s). Good.

Similarly conv_bwd[i] = sum_j w[j] * h[i - j] = sum_s w[s] * h[i - s] = backward sum. Perfect.

Thus gradient formula holds.

def adam_optimize(h0, time_left, max_iter=20000, lr=0.01, tau0=0.05, tau_min=0.005, tau_decay=0.9995):
    n = h0.shape[0]
    target_sum = n / 2.0
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5_fast(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, max_iter+1):
        if time_left() < 0.0:
            break
        tau = max(tau_min, tau0 * (tau_decay**t))
        grad = gradient_softmax_corr(h, tau)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h - step, target_sum)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # early exit if below target bound:
            # but we can continue to improve
        h = h_new
    return best_h, best_c5

def compute_corr_raw(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    return conv[:2 * n - 1]

def compute_c5_fast(h):
    n = h.shape[0]
    return compute_corr_raw(h).max() * (2.0 / n)

def gradient_softmax_corr(h, tau):
    n = h.shape[0]
    raw = compute_corr_raw(h)
    maxv = raw.max()
    expv = np.exp((raw - maxv) / tau)
    w = expv / expv.sum()
    w_rev = w[::-1]
    N = 1 << ((3 * n - 2).bit_length())
    fft_h = np.fft.rfft(h, n=N)
    fft_w_rev = np.fft.rfft(w_rev, n=N)
    fft_w = np.fft.rfft(w, n=N)
    conv_fwd = np.fft.irfft(fft_w_rev * fft_h, n=N)[:n]
    conv_bwd = np.fft.irfft(fft_w * fft_h, n=N)[:n]
    return 1.0 - conv_fwd - conv_bwd

def adam_optimize(h0, time_left, max_iter=20000, lr=0.02,
                  tau0=0.05, tau_min=0.002, tau_decay=0.9998):
    n = h0.shape[0]
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h, best_c5 = h.copy(), compute_c5_fast(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    for t in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        tau = max(tau_min, tau0 * (tau_decay ** t))
        grad = gradient_softmax_corr(h, tau)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h - step, target)
        c5 = compute_c5_fast(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n); term1[:n_s] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n); term1[s:] = h[:n_s]
        term2 = np.zeros(n); term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idx = np.where(corr == mx)[0]
    shifts = idx - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    return grad / len(shifts)

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h, best_s = h.copy(), compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

                                      lr=0.02, tau0=0.05,
                                      tau_min=0.002, tau_decay=0.9998)
        if c5_ref < best_c5:
            best_c5, best_h = c5_ref, h_ref.copy()
            if best_c5 <= target_c5:
                return best_h, best_c5, n

    while time_left() > 5.0:
        v = rng.random(n)
        cand = project_onto_box_sum(v, target_sum)
        h_pre, c5_pre = gradient_descent_momentum(cand, time_left,
                                                  step0=0.5, beta=0.9,
                                                  max_iter=2000)
        if c5_pre < best_c5:
            best_c5, best_h = c5_pre, h_pre.copy()
            if best_c5 <= target_c5:
                return best_h, best_c5, n
        if time_left() < 2.0:
            break
        max_iter = int(min(15000, time_left() * 10))
        h_ref, c5_ref = adam_optimize(h_pre, time_left,
                                      max_iter=max_iter,
                                      lr=0.02, tau0=0.05,
                                      tau_min=0.002, tau_decay=0.9998)
        if c5_ref < best_c5:
            best_c5, best_h = c5_ref, h_ref.copy()
            if best_c5 <= target_c5:
                return best_h, best_c5, n

    if time_left() > 20.0:
        n2 = 512
        if best_h is not None:
            h_up = resample(best_h, n2)
        else:
            h_up = np.full(n2, 0.5)
        target_sum2 = n2 / 2.0
        h_up = project_onto_box_sum(h_up, target_sum2)
        max_iter = int(min(20000, time_left() * 10))
        h_up_opt, c5_up = adam_optimize(h_up, time_left,
                                         max_iter=max_iter,
                                         lr=0.015, tau0=0.05,
                                         tau_min=0.002, tau_decay=0.9998)
        if c5_up < best_c5:
            best_c5, best_h, n = c5_up, h_up_opt.copy(), n2
            best_h = project_onto_box_sum(best_h, n / 2.0)
            best_c5 = compute_c5_fast(best_h)
            return best_h, best_c5, n

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), target_sum)
    else:
        best_h = project_onto_box_sum(best_h, n / 2.0)
    best_c5 = compute_c5_fast(best_h)
    return best_h, best_c5, n
