def compute_corr_full_fft(h):
    n = h.size
    L = next_pow2(2*n)
    # pad to length L
    hp = np.zeros(L, dtype=float)
    gp = np.zeros(L, dtype=float)
    hp[:n] = h
    gp[:n] = 1.0 - h
    f = np.fft.rfft(hp)
    g = np.fft.rfft(gp)
    conv = np.fft.irfft(f * np.conj(g), n=L)
    # linear correlation size 2*n -1 corresponds to conv[:2*n-1]
    return conv[:2*n-1]

def compute_gradient_for_shift(h, shift):
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        # (1 - h[i+shift]) term for i in 0..n-shift-1
        grad[:n-shift] += 1.0 - h[shift:]
        # -h[i-shift] term for i in shift..n-1
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        # shift negative: gradient over positions where i+shift >=0 (i>=s)
        grad[s:] += 1.0 - h[:n-s]
        # -h[i-shift] term: shift negative => subtract h[i+s] for i in 0..n-s-1
        grad[:n-s] -= h[s:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Get base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # generate thue-morse of length 64
        def thue_morse(n):
            return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
        base = thue_morse(64)

    # Ensure sum = n/2 via projection
    n0 = base.size
    s0 = n0 / 2.0
    base = project_onto_box_sum(base, s0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # Define scales
    scales = [64, 128, 256, 512, 1024]
    # allocate iterations per scale based on remaining time
    for n in scales:
        if time_left() <= 5:
            break
        # Upsample if needed
        if best_h.size != n:
            # linear interpolation
            x_old = np.linspace(0, 1, best_h.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            best_h = np.interp(x_new, x_old, best_h)
        # project to sum = n/2
        best_h = project_onto_box_sum(best_h, n / 2.0)
        # evaluate
        cur_score = compute_c5(best_h)
        if cur_score < best_score:
            best_h, best_score = best_h.copy(), cur_score

        # Determine iteration budget: allocate proportionally to n and remaining time
        # Roughly 0.5 * time_left seconds for gradient descent at each scale
        iter_budget = int(min(50000, max(2000, time_left() * (n/512) * 100)))  # adjust
        # Ensure at least minimal
        if iter_budget > 0:
            # Choose step0: larger for coarser scales
            step0 = 0.6 if n <= 256 else 0.4 if n <= 512 else 0.3
            beta = 0.9
            # run gradient descent
            h_opt, sc_opt = gradient_descent(best_h, n_iter=iter_budget, step0=step0, beta=beta, rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
            # update best_h for next scale
            best_h = h_opt

    # After hierarchical descent, perform random restart hill climbing for remaining time
    n_final = best_h.size
    s_final = n_final / 2.0
    # Precompute random swap loop
    while time_left() > 2:
        # generate random binary candidate and maybe local move
        h_rand = random_binary_balanced(n_final, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_h, best_score = h_rand.copy(), sc_rand

        # local move: transfer delta from i to j
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.01, 0.5)
        if delta > 1e-12:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new.copy(), sc_new

        # occasional perturbation
        if rng.random() < 0.3:
            pert = rng.normal(scale=0.02, size=n_final)
            h_pert = best_h + pert
            h_pert = project_onto_box_sum(h_pert, s_final)
            sc_pert = compute_c5(h_pert)
            if sc_pert < best_score:
                best_h, best_score = h_pert.copy(), sc_pert

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np
import scipy.optimize as _opt

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def _corr_full_fft(h):
    n = h.size
    L = _next_pow2(2 * n)
    hp = np.zeros(L, dtype=float)
    gp = np.zeros(L, dtype=float)
    hp[:n] = h
    gp[:n] = 1.0 - h
    f = np.fft.rfft(hp)
    g = np.fft.rfft(gp)
    conv = np.fft.irfft(f * np.conj(g), n=L)
    return conv[:2 * n - 1]

def _compute_c5(h):
    n = h.size
    return np.max(_corr_full_fft(h)) * (2.0 / n)

def _grad_shift(h, shift):
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        g[:n - shift] += 1.0 - h[shift:]
        g[shift:] -= h[:n - shift]
    else:
        s = -shift
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    return g

def _gd_max_shift(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    s = h.size / 2.0
    best_h, best_score = h.copy(), _compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = _corr_full_fft(h)
        shift = np.argmax(corr) - (h.size - 1)
        grad = _grad_shift(h, shift)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        if it % 500 == 0:
            h_new += rng.normal(scale=0.005, size=h.size)
        h_new = _project_onto_box_sum(h_new, s)
        sc = _compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def _gd_random_shift(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    s = h.size / 2.0
    best_h, best_score = h.copy(), _compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = _corr_full_fft(h)
        max_idxs = np.where(corr == np.max(corr))[0]
        shift = rng.choice(max_idxs) - (h.size - 1)
        grad = _grad_shift(h, shift)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=h.size)
        h_new = _project_onto_box_sum(h_new, s)
        sc = _compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def _random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def _slsqp_refine(h0, max_iter, rng):
    n = h0.size
    x0 = np.concatenate([h0, [_compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = _corr_full_fft(h) * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = _opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = _project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, _compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.fromiter(((bin(i).count('1') & 1) for i in range(64)), dtype=float, count=64)

    n0 = base.size
    base = _project_onto_box_sum(base, n0 / 2.0)
    best_h, best_score = base.copy(), _compute_c5(base)

    scales = [64, 128, 256, 512, 1024]
    target_iters = {64: 20000, 128: 20000, 256: 25000, 512: 20000, 1024: 15000}
    step0_map = {64: 0.6, 128: 0.6, 256: 0.5, 512: 0.4, 1024: 0.3}
    beta = 0.9

    for n in scales:
        if time_left() <= 5:
            break
        if best_h.size != n:
            x_old = np.linspace(0, 1, best_h.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            best_h = np.interp(x_new, x_old, best_h)
        best_h = _project_onto_box_sum(best_h, n / 2.0)
        cur_sc = _compute_c5(best_h)
        if cur_sc < best_score:
            best_h, best_score = best_h.copy(), cur_sc

        max_it = min(target_iters[n], int(time_left() * 30))
        if max_it > 0:
            h_opt, sc_opt = _gd_random_shift(best_h, max_it,
                                             step0=step0_map[n],
                                             beta=beta, rng=rng,
                                             time_left=time_left)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
            best_h = h_opt

    # Deterministic refinement on final scale
    if time_left() > 5:
        it_ref = int(min(20000, time_left() * 30))
        h_opt, sc_opt = _gd_max_shift(best_h, it_ref,
                                      step0=0.3, beta=0.8,
                                      rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_h, best_score = h_opt.copy(), sc_opt

    # Optional SLSQP polish if time permits
    if time_left() > 10:
        h_opt, sc_opt = _slsqp_refine(best_h, max_iter=200, rng=rng)
        if sc_opt < best_score:
            best_h, best_score = h_opt.copy(), sc_opt

    n_final = best_h.size
    s_final = n_final / 2.0

    # Random hill climbing until budget expires
    while time_left() > 2:
        h_rand = _random_binary_balanced(n_final, rng)
        sc_rand = _compute_c5(h_rand)
        if sc_rand < best_score:
            best_h, best_score = h_rand.copy(), sc_rand
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.01, 0.5)
        if delta > 1e-12:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = _compute_c5(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new.copy(), sc_new
        if rng.random() < 0.3:
            pert = rng.normal(scale=0.02, size=n_final)
            h_pert = best_h + pert
            h_pert = _project_onto_box_sum(h_pert, s_final)
            sc_pert = _compute_c5(h_pert)
            if sc_pert < best_score:
                best_h, best_score = h_pert.copy(), sc_pert

    best_h = _project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = _compute_c5(best_h)
    return best_h, best_score, best_h.size
