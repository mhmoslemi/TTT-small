def two_flip_mutation(individual):
    idx1 = np.random.randint(len(individual))
    idx2 = np.random.randint(len(individual))
    # Flip one 0 and one 1
    if individual[idx1] == 0:
        individual[idx1] = 1
    else:
        individual[idx1] = 0
    if individual[idx2] == 0:
        individual[idx2] = 1
    else:
        individual[idx2] = 0

def one_point_crossover(parent1, parent2):
    crossover_point = np.random.randint(1, len(parent1)-1)
    child1 = parent1.copy()
    child2 = parent2.copy()
    child1[:crossover_point+1] = parent2[:crossover_point+1]
    child2[:crossover_point+1] = parent1[:crossover_point+1]
    # Repair child to have n/2 ones
    target_sum1 = len(child1) // 2
    current_sum1 = np.sum(child1)
    if current_sum1 < target_sum1:
        num_ones_to_add = target_sum1 - current_sum1
        positive_indices = [i for i, val in enumerate(child1) if val == 1][:num_ones_to_add]
        for i in positive_indices:
            child1[i] = 0
    elif current_sum1 > target_sum1:
        num_zeros_to_add = target_sum1 - current_sum1
        negative_indices = [i for i, val in enumerate(child1) if val == 0][:num_zeros_to_add]
        for i in negative_indices:
            child1[i] = 1
    # Repeat for child2
    target_sum2 = len(child2) // 2
    current_sum2 = np.sum(child2)
    if current_sum2 < target_sum2:
        num_ones_to_add = target_sum2 - current_sum2
        positive_indices = [i for i, val in enumerate(child2) if val == 1][:num_ones_to_add]
        for i in positive_indices:
            child2[i] = 0
    elif current_sum2 > target_sum2:
        num_zeros_to_add = target_sum2 - current_sum2
        negative_indices = [i for i, val in enumerate(child2) if val == 0][:num_zeros_to_add]
        for i in negative_indices:
            child2[i] = 1
    return child1, child2

def compute_objective(h, n):
