def gradient_descent_smooth(h0, n_iter, step0, beta, alpha, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # compute correlation
        corr = compute_corr_full_fft(h)
        # compute smooth max
        w = np.exp(alpha * corr - np.max(alpha * corr))  # subtract max for numeric stability
        w_sum = w.sum()
        w_norm = w / w_sum
        # compute gradient
        grad = np.zeros_like(h)
        offset = n - 1
        # loop over shifts
        # Option: vectorized? But loop is okay.
        for idx in range(len(corr)):
            wk = w_norm[idx]
            if wk == 0:
                continue
            k = idx - offset
            if k >= 0:
                grad[:n - k] += wk * (1.0 - h[k:])
                grad[k:] -= wk * h[:n - k]
            else:
                m = -k
                grad[m:] += wk * (1.0 - h[:n - m])
                grad[:n - m] -= wk * h[m:]
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def local_pairwise_search(h, moves, rng, time_left):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    for _ in range(moves):
        if time_left() <= 0.0:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if best_h[i] <= 0.0 or best_h[j] >= 1.0:
            continue
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.random() * 0.2
        if delta < 1e-12:
            continue
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_onto_box_sum(cand, n / 2.0)
        sc = compute_c5_fft(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand
    return best_h, best_score

def binary_swap_search(h, rng, time_left):
    n = h.size
    idx = np.argsort(-h)
    bin_h = np.zeros_like(h)
    bin_h[idx[:n // 2]] = 1.0
    bin_h = project_onto_box_sum(bin_h, n / 2.0)
    best_h = bin_h.copy()
    best_score = compute_c5_fft(best_h)
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h <= 0.5)[0]
    it = 0
    max_iter = 200000
    while time_left() > 0.1 and it < max_iter:
        temp = 0.01 * (1 - it / max_iter) + 1e-5 * (it / max_iter)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        sc = compute_c5_fft(cand)
        delta = sc - best_score
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            best_h = cand
            best_score = sc
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h <= 0.5)[0]
        it += 1
    return best_h, best_score

def slsqp_refine(h, score, rng, time_left):
    if time_left() <= 5.0:
        return h, score
    n = h.size
    x0 = np.concatenate([h, [score]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        hh = x[:n]
        t = x[-1]
        corr = np.correlate(hh, 1.0 - hh, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-12,
                                    'disp': False})
        h_s = res.x[:n]
        h_s = project_onto_box_sum(h_s, n / 2.0)
        sc_s = compute_c5_fft(h_s)
        if sc_s < score:
            return h_s, sc_s
    except Exception:
        pass
    return h, score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(128, rng)

    scales = [128, 256, 512]
    best_h = None
    best_score = np.inf
    TARGET = 0.3808

    for n in scales:
        if time_left() <= 1.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        restarts = 4
        for r in range(restarts):
            if time_left() <= 0.5:
                break
            h0 = base + rng.normal(scale=0.05, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)
            iters = 20000
            h_opt, sc_opt = gradient_descent(h0, n_iter=iters,
                                             step0=0.6, beta=0.9,
                                             rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_score <= TARGET:
            break
        # light local continuous refinement
        if best_h is not None and time_left() > 5.0:
            best_h, best_score = slsqp_refine(best_h, best_score, rng, time_left)
        # prepare base for next scale
        base = best_h.copy() if best_h is not None else base

    # final continuous local search at highest resolution
    if best_h is not None and time_left() > 5.0:
        best_h, best_score = local_pairwise_search(best_h, moves=20000,
                                                   rng=rng, time_left=time_left)

    # binary conversion and swap search
    if best_h is not None and time_left() > 2.0:
        bin_h, bin_score = binary_swap_search(best_h, rng, time_left)
        if bin_score < best_score:
            best_h, best_score = bin_h, bin_score

    # final safety projection
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
        n_ret = best_h.size
    else:
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, n_ret
