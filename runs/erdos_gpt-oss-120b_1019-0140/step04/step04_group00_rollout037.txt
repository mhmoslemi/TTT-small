def compute_c5_fast(h):
    n = h.shape[0]
    hx = h.astype(np.float64)
    h_rev = hx[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - hx, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

            best_score = scores[i]
    return pop[best_i]

def two_point_crossover(p1, p2, rng):
    n = p1.shape[0]
    i, j = sorted(rng.choice(n, size=2, replace=False))
    child1 = np.concatenate([p1[:i], p2[i:j], p1[j:]])
    child2 = np.concatenate([p2[:i], p1[i:j], p2[j:]])
    return child1, child2

def swap_mutation(ind, rng):
    ones = np.where(ind == 1)[0]
    zeros = np.where(ind == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return ind
    i = rng.choice(ones)
    j = rng.choice(zeros)
    ind[i] = 0
    ind[j] = 1
    return ind

Repair balance:

def repair_balance(ind, target):
    # target is number of ones
    cur = ind.sum()
    if cur > target:
        excess = cur - target
        # turn excess ones to zero
        ones = np.where(ind == 1)[0]
        idx = rng.choice(ones, size=excess, replace=False)
        ind[idx] = 0
    elif cur < target:
        deficit = target - cur
        zeros = np.where(ind == 0)[0]
        idx = rng.choice(zeros, size=deficit, replace=False)
        ind[idx] = 1
    return ind

def resample_arr(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def hill_climb_binary(h, time_left, max_iter=5000):
    n = h.size
    cur = compute_c5_fast(h)
    best_h = h.copy()
    best_c = cur
    it = 0
    while it < max_iter and time_left() > 0.1:
        # sample random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # apply swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5_fast(h)
        if new_c < cur:
            cur = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        it += 1
    return best_h, best_c

def compute_c5_and_shifts(h):
    n = h.shape[0]
    hx = h.astype(np.float64)
    h_rev = hx[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - hx, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    max_val = np.max(corr)
    # compute shift indices
    idxes = np.where(corr == max_val)[0]
    shifts = idxes - (n - 1)
    return max_val * (2.0 / n), shifts

def compute_c5_fast(h):
    n = h.shape[0]
    hx = h.astype(np.float64)
    h_rev = hx[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - hx, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_c5_and_shifts(h):
    n = h.shape[0]
    hx = h.astype(np.float64)
    h_rev = hx[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - hx, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    max_val = np.max(corr)
    idxes = np.where(corr == max_val)[0]
    shifts = idxes - (n - 1)
    return max_val * (2.0 / n), shifts

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n); term1[:n_s] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n); term1[s:] = h[:n_s]
        term2 = np.zeros(n); term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    _, shifts = compute_c5_and_shifts(h)
    grad = np.zeros_like(h, dtype=np.float64)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < target:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > target:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target) < 1e-12:
            break
        if fm > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def random_binary_balanced(n, rng):

    old_n = arr.shape[0]

def repair_balance(ind, target, rng):
    cur = ind.sum()

        excess = cur - target
        ones = np.where(ind == 1)[0]
        idx = rng.choice(ones, size=excess, replace=False)
        ind[idx] = 0
    elif cur < target:
        deficit = target - cur
        zeros = np.where(ind == 0)[0]
        idx = rng.choice(zeros, size=deficit, replace=False)

    best_score = scores[best_i]
    for i in idxs[1:]:
        if scores[i] < best_score:

    return pop[best_i]

        rng = np.random.default_rng()
    cur = compute_c5_fast(h)

        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_c = compute_c5_fast(h)
        if new_c < cur:
            cur = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        it += 1

import time
import numpy as np

def compute_c5_fast(h):
    n = h.shape[0]
    hx = h.astype(np.float64)
    h_rev = hx[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - hx, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_c5_and_shifts(h):
    n = h.shape[0]
    hx = h.astype(np.float64)
    h_rev = hx[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - hx, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    max_val = np.max(corr)
    idxes = np.where(corr == max_val)[0]
    shifts = idxes - (n - 1)
    return max_val * (2.0 / n), shifts

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n); term1[:n_s] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n); term1[s:] = h[:n_s]
        term2 = np.zeros(n); term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    _, shifts = compute_c5_and_shifts(h)
    grad = np.zeros_like(h, dtype=np.float64)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < target:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > target:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target) < 1e-12:
            break
        if fm > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=int)

    return h

def thue_morse_binary(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def resample_arr(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def continuous_to_binary(v):
    n = v.size
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    bin_vec = np.zeros(n, dtype=int)
    bin_vec[idx] = 1
    return bin_vec

def two_point_crossover(p1, p2, rng):
    n = p1.size
    i, j = sorted(rng.choice(n, size=2, replace=False))
    child1 = np.concatenate([p1[:i], p2[i:j], p1[j:]])
    child2 = np.concatenate([p2[:i], p1[i:j], p2[j:]])
    return child1, child2

def repair_balance(ind, target, rng):
    cur = ind.sum()
    if cur > target:
        excess = cur - target
        ones = np.where(ind == 1)[0]
        idx = rng.choice(ones, size=excess, replace=False)
        ind[idx] = 0
    elif cur < target:
        deficit = target - cur
        zeros = np.where(ind == 0)[0]
        idx = rng.choice(zeros, size=deficit, replace=False)
        ind[idx] = 1
    return ind

def swap_mutation(ind, rng):
    ones = np.where(ind == 1)[0]
    zeros = np.where(ind == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return ind
    i = rng.choice(ones)
    j = rng.choice(zeros)

    ind[j] = 1
    return ind

            best_score = scores[i]

            h[j] = 0
