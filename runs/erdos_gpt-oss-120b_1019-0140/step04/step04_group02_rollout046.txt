def compute_grad_smooth_max(h, corr, max_c, eps=0.01, alpha=50.0):
    n = h.size
    # find indices where corr within eps of max
    idxes = np.where(corr >= max_c - eps)[0]
    if len(idxes) == 0:
        # fallback to max only
        idxes = np.where(corr == max_c)[0]
    shifts = idxes - (n - 1)
    # compute weights: w_i = exp(alpha*(corr_i - max_c))
    weights = np.exp(alpha * (corr[idxes] - max_c))
    # sum of weights for normalization
    w_sum = np.sum(weights)
    grad = np.zeros_like(h)
    for shift, w in zip(shifts, weights):
        if shift >= 0:
            end = n - shift
            grad[:end] += w * (1.0 - h[shift:])
            grad[shift:] -= w * h[:end]
        else:
            m = -shift
            end = n - m
            grad[m:] += w * (1.0 - h[:end])
            grad[:end] -= w * h[m:]
    # normalized gradient (dividing by total weight)
    grad /= w_sum
    return grad

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Enlarge bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_smooth_max(h, corr, max_c, eps=0.02, alpha=50.0):
    """
    Gradient of a smooth approximation of max_k C(k).

    Uses only shifts whose correlation is within eps of the current max.
    Weights them by exp(alpha*(corr - max_c)).
    """
    n = h.size
    idxes = np.where(corr >= max_c - eps)[0]
    if idxes.size == 0:
        idxes = np.where(corr == max_c)[0]
    shifts = idxes - (n - 1)
    # weights bounded by 1 (since corr <= max_c)
    w = np.exp(alpha * (corr[idxes] - max_c))
    w_sum = w.sum()
    grad = np.zeros_like(h)
    for shift, weight in zip(shifts, w):
        if shift >= 0:
            end = n - shift
            grad[:end] += weight * (1.0 - h[shift:])
            grad[shift:] -= weight * h[:end]
        else:
            m = -shift
            end = n - m
            grad[m:] += weight * (1.0 - h[:end])
            grad[:end] -= weight * h[m:]
    grad /= w_sum
    return grad


def gradient_descent_smooth(h0, n_iter, step0, beta, eps, alpha, rng, time_left):
    """
    Projected momentum gradient descent on a smooth max‑correlation objective.
    Stops early if time runs out.
    """
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_smooth_max(h, corr, max_c, eps=eps, alpha=alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # Initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128).astype(float)
    n0 = base.size
    base = project_onto_box_sum(base, n0 / 2.0)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)
    TARGET = 0.3808

    scales = [128, 256, 512, 1024]
    iter_map = {128: 120000, 256: 150000, 512: 200000, 1024: 250000}
    for n in scales:
        if time_left() <= 2.0:
            break
        # Upsample to current scale
        if best_h.size != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)

        restarts = 3
        for r in range(restarts):
            if time_left() <= 1.0:
                break
            h0 = base + rng.normal(scale=0.05, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)
            h_opt, sc_opt = gradient_descent_smooth(
                h0,
                n_iter=iter_map.get(n, 150000),
                step0=0.5,
                beta=0.9,
                eps=0.02,
                alpha=50.0,
                rng=rng,
                time_left=time_left
            )
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        # Local stochastic mass‑transfer hill climbing
        while time_left() > 2.0:
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(best_h[i], 1.0 - best_h[j])
            if max_delta <= 1e-12:
                continue
            delta = max_delta * rng.uniform(0.05, 0.2)
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5_fft(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                if best_score <= TARGET:
                    break
            else:
                # Simulated‑annealing style acceptance
                elapsed = time.time() - start
                T = max(0.001, 0.1 * (budget_s - elapsed) / budget_s)
                if rng.random() < np.exp(-(sc_new - best_score) / T):
                    best_h, best_score = h_new, sc_new
        # Re‑project after local search
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
        if best_score <= TARGET:
            break

    if best_h is None:
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5_fft(best_h)
    else:
        n_ret = best_h.size

    return best_h, best_score, n_ret
