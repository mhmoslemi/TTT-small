def anneal_binary(h0, rng, time_left_func, T0=0.2, decay=0.9999, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    T = T0
    while time_left_func() > 0.01:
        # pick swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
    return best_h, best_score
```

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx = 2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    k = n // 2
    idx = rng.choice(n, size=k, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def make_binary(v):
    n = v.size
    k = n // 2
    idx = np.argpartition(-v, k)[:k]
    bin_arr = np.zeros_like(v, dtype=float)
    bin_arr[idx] = 1.0
    return bin_arr

def anneal_binary(h0, rng, time_left_func, T0=1.0, decay=0.99998, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    T = T0
    while time_left_func() > 0.0:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    end = start + budget_s
    def time_left():
        return max(0.0, end - time.time())

    target = 0.3808

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        n = init_arr.size
        if n % 2 != 0:  # enforce even size
            init_arr = init_arr[:n-1]
            n -= 1
        h0 = init_arr.copy()
    else:
        n = 512
        h0 = random_binary_balanced(n, rng)

    # Phase 1: continuous refinement (gradient descent) - ~20% of budget
    phase1_end = start + budget_s * 0.2
    def tl_phase1():
        return max(0.0, phase1_end - time.time())
    h_gd, sc_gd = gradient_descent_momentum_with_time(h0, tl_phase1,
                                                      step0=0.5, beta=0.9,
                                                      max_iter=200000)
    best_h = h_gd.copy()
    best_score = sc_gd
    if best_score <= target:
        return best_h, best_score, n

    # Phase 2: binary annealing - ~70% of budget (leaving ~10% for final refinement)
    phase2_end = start + budget_s * 0.9
    def tl_phase2():
        return max(0.0, phase2_end - time.time())
    # start annealing from binary version of current best
    start_bin = make_binary(best_h)
    start_bin = project_onto_box_sum(start_bin, n/2.0)
    h_ann, sc_ann = anneal_binary(start_bin, rng, tl_phase2,
                                  T0=1.0, decay=0.99998, Tmin=1e-6)
    if sc_ann < best_score:
        best_h = h_ann.copy()
        best_score = sc_ann
    if best_score <= target:
        return best_h, best_score, n

    # Phase 3: final continuous refinement with remaining time
    if time_left() > 0.5:
        h_final, sc_final = gradient_descent_momentum_with_time(best_h, time_left,
                                                                step0=0.3, beta=0.9,
                                                                max_iter=200000)
        if sc_final < best_score:
            best_h = h_final.copy()
            best_score = sc_final

    # Ensure feasibility and final evaluation
    best_h = project_onto_box_sum(best_h, n/2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, n
