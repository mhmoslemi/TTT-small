def thue_morse_binary(n):
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float)
    # make balanced
    ones = int(seq.sum())
    target = n // 2
    if ones > target:
        # turn some ones to zero
        idx_ones = np.where(seq == 1)[0]
        to_flip = rng.choice(idx_ones, size=ones-target, replace=False)
        seq[to_flip] = 0
    elif ones < target:
        idx_zeros = np.where(seq == 0)[0]
        to_flip = rng.choice(idx_zeros, size=target-ones, replace=False)
        seq[to_flip] = 1
    return seq

def binary_anneal(h0, rng, time_left_func, max_iters=1000000, T0=0.1, decay=0.9995, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    current_score = best_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.5:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5_fft(h_new)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > Tmin and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            h = h_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term1[n-shift:] = 0.0
            term2[shift:] = h[:n-shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term1[:s] = 0.0
            term2[:n-s] = h[s:]
            term2[n-s:] = 0.0
        grad += 1.0 - term1 - term2
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        # compute correlation to find worst shifts
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # find indices with max correlation within tolerance
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # adaptive step
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)
```

def compute_c5_fft(h):

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {lo <= x_i <= hi, sum x_i = s} using bisection."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap C5 using FFT, scaled by dx = 2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n, rng):
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    target = n // 2
    cur = int(seq.sum())
    if cur > target:
        idx_ones = np.where(seq == 1)[0]
        to_flip = rng.choice(idx_ones, size=cur - target, replace=False)
        seq[to_flip] = 0.0
    elif cur < target:
        idx_zeros = np.where(seq == 0)[0]
        to_flip = rng.choice(idx_zeros, size=target - cur, replace=False)
        seq[to_flip] = 1.0
    return seq

def to_binary_balanced(v):
    n = v.shape[0]
    idx = np.argpartition(-v, n // 2)[:n // 2]
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += 1.0 - term1 - term2
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_anneal(h0, rng, time_left_func, max_iters=500000, T0=0.1, decay=0.9995, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    current_score = best_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.5:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5_fft(h_new)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > Tmin and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            h = h_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
