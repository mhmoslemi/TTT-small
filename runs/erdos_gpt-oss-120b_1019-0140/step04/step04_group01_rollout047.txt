def thue_morse_binary(n):
    # generate bits: parity of popcount, result 0 for even popcount? But we need values 0/1. The generator earlier in previous code returned 0.0/1.0.

import time
import math
import numpy as np

def compute_corr(h):
    """Full correlation of binary h with (1-h) using integer arithmetic."""
    return np.correlate(h, 1 - h, mode="full")

def compute_max_corr(h):
    """Return (max_corr, full_corr_array)."""
    corr = compute_corr(h)
    return int(corr.max()), corr

def thue_morse_binary(n):
    """Balanced Thue–Morse sequence for n (power of 2)."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                       dtype=np.int8, count=n)

def random_balanced_binary(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1
    return h

def greedy_optimize_binary(h_init, time_left_func, target_max, rng):
    """Targeted simulated annealing on binary vectors."""
    n = h_init.shape[0]
    h = h_init.copy()
    best_h = h.copy()
    best_max, _ = compute_max_corr(h)
    # annealing schedule
    T0 = 0.5
    decay = 0.9999
    it = 0
    while time_left_func() > 0.01:
        it += 1
        T = max(1e-12, T0 * (decay ** it))
        cur_max, corr_arr = compute_max_corr(h)
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
            if best_max <= target_max:
                break
        # pick a worst shift (random among ties)
