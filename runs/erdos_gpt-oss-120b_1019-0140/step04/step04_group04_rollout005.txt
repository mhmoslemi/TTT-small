import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 300 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_projection(v):
    n = v.shape[0]
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    h = np.zeros_like(v, dtype=float)
    h[idx] = 1.0
    return h

def hill_climb_binary(h_start, score_start, time_left, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_score = score_start
    cur_score = score_start
    n = h.shape[0]
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    while time_left() > 0.5:
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    target_n = 512
    if base.shape[0] != target_n:
        base = upsample(base, target_n)
    base = project_onto_box_sum(base, target_n / 2.0)

    # Continuous gradient descent
    max_iters = int(min(180000, time_left() * 30))
    if max_iters > 0:
        cont_h, cont_score = gradient_descent_momentum_custom(
            base, n_iter=max_iters, step0=0.3, beta=0.95, rng=rng,
            time_left=time_left
        )
    else:
        cont_h, cont_score = base, compute_c5(base)

    # Choose best continuous candidate
    best_h = cont_h.copy()
    best_score = cont_score

    # Binary projection and improvement
    bin_h = binary_projection(cont_h)
    bin_score = compute_c5(bin_h)
    if bin_score < best_score:
        best_h, best_score = bin_h.copy(), bin_score

    if np.all((best_h == 0) | (best_h == 1)):
        best_h, best_score = hill_climb_binary(best_h, best_score, time_left, rng)

    # Random restarts with hill climbing
    while time_left() > 6.0:
        rand_h = random_binary_balanced(target_n, rng)
        rand_score = compute_c5(rand_h)
        if rand_score < best_score:
            best_h, best_score = hill_climb_binary(rand_h, rand_score, time_left, rng)
        if best_score <= 0.3808:
            break

    # Final feasibility projection
    best_h = project_onto_box_sum(best_h, target_n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, target_n
