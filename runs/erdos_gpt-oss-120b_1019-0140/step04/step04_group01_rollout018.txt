def gradient_smoothed(h, beta):
    n = h.size
    corr = compute_corr_scaled_fft(h)  # length 2n-1
    max_corr = np.max(corr)  # maybe not needed
    # compute weights
    # To avoid overflow, subtract max before exp
    cmax = np.max(beta * corr)
    exp_vals = np.exp(beta * corr - cmax)
    w = exp_vals / np.sum(exp_vals)  # length 2n-1
    grad = np.zeros_like(h)
    # loop over shifts
    # shift index: shift = idx - (n - 1)
    for idx, shift in enumerate(range(-(n-1), n)):
        weight = w[idx]
        if weight == 0:
            continue
        if shift >= 0:
            # i in [0, n-shift-1]
            grad[:n-shift] += weight * (1.0 - h[shift:])
            # second term: i in [shift, n-1] subtract weight * h[:n-shift]
            grad[shift:] -= weight * h[:n-shift]
        else:
            s = -shift
            # i in [s, n-1]
            grad[s:] += weight * (1.0 - h[:n-s])
            # second term: i in [0, n-s-1]
            grad[:n-s] -= weight * h[s:]
    return grad

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False) # domain length 2
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_smoothed(h, beta):
    n = h.size
    corr = compute_corr_scaled_fft(h)
    # compute weights via softmax
    mu = beta * corr
    mu_max = np.max(mu)
    exp_shifted = np.exp(mu - mu_max)
    w = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros_like(h)
    shift0 = -(n-1)
    for idx, shift in enumerate(range(shift0, n)):
        weight = w[idx]
        if weight == 0.0:
            continue
        if shift >= 0:
            if n - shift > 0:
                grad[:n-shift] += weight * (1.0 - h[shift:])
                grad[shift:] -= weight * h[:n-shift]
        else:
            s = -shift
            if n - s > 0:
                grad[s:] += weight * (1.0 - h[:n-s])
                grad[:n-s] -= weight * h[s:]
    return grad

def binary_swap_anneal(h0, rng, time_left_func, max_iters=200000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    # Ensure binary - h0 expected binary but may contain floats; we treat >0.5 as 1
    h = (h > 0.5).astype(float)
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        # pick a random 1 and a random 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > Tmin and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure the interval brackets the target
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx = 2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    """Maximum overlap integral C5."""
    return np.max(compute_corr_scaled_fft(h))

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_smoothed(h, beta):
    """Gradient of the soft‑max surrogate of max_k corr."""
    n = h.size
    corr = compute_corr_scaled_fft(h)
    mu = beta * corr
    mu_max = np.max(mu)
    exp_shifted = np.exp(mu - mu_max)
    w = exp_shifted / np.sum(exp_shifted)  # length 2n‑1
    grad = np.zeros_like(h)
    shift0 = -(n - 1)
    for idx, shift in enumerate(range(shift0, n)):
        weight = w[idx]
        if weight == 0.0:
            continue
        if shift >= 0:
            if n - shift > 0:
                grad[:n - shift] += weight * (1.0 - h[shift:])
                grad[shift:] -= weight * h[:n - shift]
        else:
            s = -shift
            if n - s > 0:
                grad[s:] += weight * (1.0 - h[:n - s])
                grad[:n - s] -= weight * h[s:]
    return grad

def compute_gradients_for_shifts(h, shifts):

    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            if n - shift > 0:
                grad[:n - shift] += (1.0 - h[shift:])
                grad[shift:] -= h[:n - shift]
        else:

                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] -= h[s:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

            accept = True
        else:
            if T > Tmin and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    n = 512                     # target resolution
    s = n / 2.0                 # sum constraint (since dx = 2/n)

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float).ravel()
        if init_arr.size != n:
            init_arr = upsample(init_arr, n)
        h = init_arr.copy()
    else:
        h = rng.random(n)
    h = project_onto_box_sum(h, s)

    best_h = h.copy()
    best_score = compute_c5_fft(best_h)

    # ----- smoothed gradient descent (soft‑max) ----------------------
    beta_schedule = [5.0, 15.0, 35.0, 70.0, 120.0]
    momentum = np.zeros_like(h)
    step0 = 0.5
    beta_mom = 0.9
    it_global = 1

    for beta in beta_schedule:
        max_iter_beta = 20000
        for it in range(max_iter_beta):
            if time_left() <= 0.0:
                break
            grad = gradient_smoothed(h, beta)
            momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
            step = step0 / np.sqrt(it_global)
            h_new = h - step * momentum
            h_new = project_onto_box_sum(h_new, s)
            score = compute_c5_fft(h_new)
            if score < best_score:
                best_score, best_h = score, h_new.copy()
            h = h_new
            it_global += 1
        if time_left() <= 0.0:
            break

    # ----- refinement using true max (subgradient) --------------------
    momentum = np.zeros_like(h)
    max_iter_ref = 20000
    for it in range(max_iter_ref):
        if time_left() <= 0.0:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        step = step0 / np.sqrt(it_global)
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it_global += 1

    # ----- binary conversion (top half) ------------------------------
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    score_bin = compute_c5_fft(h_bin)
    if score_bin < best_score:
        best_score, best_h = score_bin, h_bin.copy()

    # ----- binary annealing -----------------------------------------
    if time_left() > 0.2:
        h_bin_opt, score_bin_opt = binary_swap_anneal(
            best_h, rng, time_left,
            max_iters=200000, T0=0.2, decay=0.99995, Tmin=1e-5)
        if score_bin_opt < best_score:
            best_score, best_h = score_bin_opt, h_bin_opt.copy()

    # ----- final high‑beta smoothed refinement -----------------------
    if time_left() > 0.2:
        momentum = np.zeros_like(best_h)
        beta = 150.0
        max_iter_final = 5000
        for it in range(max_iter_final):
            if time_left() <= 0.0:
                break
            grad = gradient_smoothed(best_h, beta)
            momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
            step = step0 / np.sqrt(it_global + 1)
            h_new = best_h - step * momentum
            h_new = project_onto_box_sum(h_new, s)
            score = compute_c5_fft(h_new)
            if score < best_score:
                best_score, best_h = score, h_new.copy()
            it_global += 1

    # ----- random restarts if time permits -------------------------
    while time_left() > 5.0:
        h_rand = rng.random(n)
        h_rand = project_onto_box_sum(h_rand, s)
        momentum = np.zeros_like(h_rand)
        beta = 70.0
        for i in range(2000):
            if time_left() <= 0.0:
                break
            grad = gradient_smoothed(h_rand, beta)
            momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
            step = step0 / np.sqrt(i + 1)
            h_rand = project_onto_box_sum(h_rand - step * momentum, s)
        score = compute_c5_fft(h_rand)
        if score < best_score:
            best_score, best_h = score, h_rand.copy()
            # optional quick binary tweak
            idx = np.argpartition(-best_h, k)[:k]
            h_bin = np.zeros_like(best_h)
            h_bin[idx] = 1.0
            sc = compute_c5_fft(h_bin)
            if sc < best_score:
                best_score, best_h = sc, h_bin.copy()

    best_h = project_onto_box_sum(best_h, s)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
