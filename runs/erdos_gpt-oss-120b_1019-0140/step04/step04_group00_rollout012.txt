def binary_swap_anneal(h0, rng, time_left, max_iters=200000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99995
    it = 0
    n = h.size
    while it < max_iters and time_left() > 0:
        # choose random indices 1 and 0
        i = rng.integers(n)
        # ensure we choose a 1 and a 0
        if h[i] == 1.0:
            # find zero index
            zeros = np.where(h == 0.0)[0]
            if zeros.size == 0: break
            j = rng.choice(zeros)
        else:
            ones = np.where(h == 1.0)[0]
            if ones.size == 0: break
            j = rng.choice(ones)
        # swap
        h[i], h[j] = h[j], h[i]
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def compute_c5_fast(h):
    n = h.shape[0]
    # we can compute convolution of h_rev and (1-h) using FFT
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

import time
import numpy as np

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def thue_morse_binary(n):
    i = np.arange(n, dtype=np.uint32)
    bits = np.unpackbits(i[:, None].view(np.uint8), axis=1)
    parity = bits.sum(axis=1) & 1
    return parity.astype(float)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample_array(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_continuous(arr, n):
    if arr.size != n:
        arr = resample_array(arr, n)
    target = n // 2
    idx = np.argpartition(-arr, target)[:target]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h

def remaining_time(start, budget_s):
    return max(0.0, budget_s - (time.time() - start))

    best_h = h.copy()
