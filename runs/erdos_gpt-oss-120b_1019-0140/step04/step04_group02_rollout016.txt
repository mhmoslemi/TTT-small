def compute_c5_fft(h):
    n = h.shape[0]
    L = 2*n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return np.max(corr) * (2.0 / n)

import time
import numpy as np

TARGET = 0.3808  # early stopping target


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the interval brackets the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation of arr (size m) to new_n points in [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) via FFT (linear, not circular)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient of the max‑correlation term (averaged over ties)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def binary_hill_climb(n, rng, deadline):
    """Simulated‑annealing hill climb on a balanced binary vector."""
    # initial random balanced binary
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    cur_h = h.copy()
    cur_score = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # maintain index lists and position maps for O(1) swaps
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    pos_one = {idx: i for i, idx in enumerate(ones)}
    pos_zero = {idx: i for i, idx in enumerate(zeros)}

    it = 0
    max_it = 200000
    while time.time() < deadline and it < max_it:
        it += 1
        # linear temperature schedule
        T = 0.02 * (1.0 - it / max_it) + 1e-4
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = compute_c5_fft(new_h)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_h = new_h
            cur_score = new_score
            # update structures
            oi = pos_one[i]
            zj = pos_zero[j]
            ones[oi] = j
            zeros[zj] = i
            pos_one[j] = oi
            pos_zero[i] = zj
            del pos_one[i]
            del pos_zero[j]

            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
    return best_h, best_score


def gradient_descent(h0, deadline):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 0
    max_it = 200000
    while time.time() < deadline and it < max_it:
        it += 1
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = 0.9 * momentum + 0.1 * grad
        eta = 0.5 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        if best_score <= TARGET:
            break
    return best_h, best_score

    best_h = None
    best_score = np.inf
    best_n = None

    # resolutions to explore
    resolutions = [128, 256, 512]

        if n == base_n:

        else:

                start_h = upsample(base, n)

        # binary hill‑climbing restarts
        restarts = 5

        best_h = project_onto_box_sum(best_h, n_ret / 2.0)
        best_score = compute_c5_fft(best_h)
