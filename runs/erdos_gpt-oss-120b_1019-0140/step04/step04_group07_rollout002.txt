def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        corr = compute_corr_full(h) # shape 2n-1
        max_corr = corr.max()
        # shifts achieving max:
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n-1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1-beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        # jitter injection: maybe every 1000 iterations
        if it % 1000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = compute_corr_full(best_h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n-1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[:n-shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])

        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:

        max_delta = min(best_h[i], 1.0 - best_h[j])

import time
import numpy as np

# Global timing variables for helper functions
RUN_START = 0.0
RUN_BUDGET = 0.0


def time_left():
    """Remaining time in seconds."""
    return RUN_BUDGET - (time.time() - RUN_START)


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure the root is bracketed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_corr_full(h):
    """Full linear correlation of h with (1-h) (zero‑padded)."""
    n = h.shape[0]
    m = 2 * n - 1
    b_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, m)
    B = np.fft.rfft(b_rev, m)
    corr = np.fft.irfft(H * B, m)
    return corr


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    return compute_corr_full(h).max() * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max‑correlation term summed over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng):
    """Projected gradient descent with momentum and adaptive step size."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        # correlation for gradient direction
        corr = compute_corr_full(h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 1000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def improve_via_max_shift(h, rng, max_iters):
    """Transfer mass from a point that contributes most to the worst lag."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    for _ in range(max_iters):
        if time_left() <= 0:
            break
        corr = compute_corr_full(best_h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[: n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[: n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def local_pair_moves(h, rng, max_iters):
    """Transfer small mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    for _ in range(max_iters):
        if time_left() <= 0:
            break
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

    rng = np.random.default_rng(seed)
    RUN_START = time.time()
    RUN_BUDGET = budget_s

    # Initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # Multi‑scale refinement
    scales = [64, 128, 256, 512, 1024]
    for n in scales:
        if time_left() <= 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Gradient descent on current scale
        remaining = time_left()
        max_iters_gd = int(min(80000, max(5000, remaining * (15000.0 / n))))
        step0 = 0.8 if n <= 256 else 0.5
        h_gd, sc_gd = gradient_descent_adaptive(base, max_iters_gd, step0=step0,
                                                beta=0.9, rng=rng)
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd
        base = h_gd

        # Targeted max‑shift improvement
        if time_left() > 5:
            max_iters_ms = int(min(4000, time_left() * 2))
            h_ms, sc_ms = improve_via_max_shift(base, rng, max_iters_ms)
            if sc_ms < best_score:
                best_h, best_score = h_ms.copy(), sc_ms
            base = h_ms

        # Local pair‑wise moves
        if time_left() > 5:
            max_iters_pair = int(min(8000, time_left() * 3))
            h_pair, sc_pair = local_pair_moves(base, rng, max_iters_pair)
            if sc_pair < best_score:
                best_h, best_score = h_pair.copy(), sc_pair
            base = h_pair

        # Random binary restarts with brief refinement
        if time_left() > 5:
            restarts = 4
            for _ in range(restarts):
                if time_left() < 6:
                    break
                h_rand = random_binary_balanced(n, rng)
                # short gradient descent
                max_iters_rand = int(min(10000, max(2000, time_left() * (4000.0 / n))))
                h_rand_ref, sc_rand = gradient_descent_adaptive(h_rand, max_iters_rand,
                                                               step0=0.5, beta=0.9,
                                                               rng=rng)
                if sc_rand < best_score:
                    best_h, best_score = h_rand_ref.copy(), sc_rand
                if time_left() > 5:
                    max_iters_rand_pair = int(min(4000, time_left() * 2))
                    h_rand_pair, sc_rand_pair = local_pair_moves(h_rand_ref, rng,
                                                                 max_iters_rand_pair)
                    if sc_rand_pair < best_score:
                        best_h, best_score = h_rand_pair.copy(), sc_rand_pair

        # Prepare base for next scale
        base = best_h

    # Final intensive refinement on the global best
    if time_left() > 5:
        max_iters_final = int(min(150000, max(40000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(best_h, max_iters_final,
                                                      step0=0.3, beta=0.9,
                                                      rng=rng)

    # One more round of targeted and pair moves
    if time_left() > 2:
        max_iters_ms = int(min(3000, time_left() * 2))
        best_h, best_score = improve_via_max_shift(best_h, rng, max_iters_ms)
    if time_left() > 2:
        max_iters_pair = int(min(5000, time_left() * 2))
        best_h, best_score = local_pair_moves(best_h, rng, max_iters_pair)

    # Final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
