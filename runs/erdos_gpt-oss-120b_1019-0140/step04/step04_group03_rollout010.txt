def simulated_annealing(h, total_steps, T0, T_final, rng):
    cur = h.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    n = len(h)
    for step in range(total_steps):
        # schedule temperature
        t = T0 * ((total_steps - step) / total_steps) + T_final
        # move type: with 0.7 probability delta move else swap
        if rng.random() < 0.7:
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(cur[i], 1.0 - cur[j])
            if max_move <= 0:
                continue
            delta = max_move * rng.random()
            # apply
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        else:
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        new_score = compute_c5(new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / t):
            cur = new
            cur_score = new_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
    return best, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h, total_steps, T0, T_final, rng, max_time=None):
    n = h.shape[0]
    cur = h.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    for step in range(total_steps):
        if max_time is not None and time.time() - start >= max_time:

                best_score = cur_score
    return best, best_score

def process_scale(h, target_n, rng, start_time, budget_s):
    # Upsample and project
    h_up = upsample(h, target_n)
    h_up = project_onto_box_sum(h_up, target_n / 2.0)
    best_h = h_up
    best_score = compute_c5(best_h)

    # Perturbation candidate
    rem = budget_s - (time.time() - start_time)
    if rem > 5.0:
        cand = best_h + rng.normal(scale=0.02, size=target_n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_box_sum(cand, target_n / 2.0)
        sc = compute_c5(cand)
        if sc < best_score:
            best_h, best_score = cand, sc

    # Gradient descent restarts
    for r in range(2):
        rem = budget_s - (time.time() - start_time)
        if rem < 3.0:
            break
        h_start = best_h.copy()
        if r > 0:
            h_start += rng.normal(scale=0.05, size=target_n)
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = project_onto_box_sum(h_start, target_n / 2.0)
        max_iters = int(min(200000, max(50000, rem * 20000)))
        h_opt, sc_opt = gradient_descent_momentum_custom(
            h_start, max_iters, step0=0.5, beta=0.9, rng=rng, max_time=rem)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    # Simulated annealing
    rem = budget_s - (time.time() - start_time)
    if rem > 3.0:
        steps = int(min(300000, max(50000, rem * 25000)))
        T0 = 0.02 if target_n <= 256 else 0.015
        best_h, best_score = simulated_annealing(best_h, steps, T0, 1e-6, rng, max_time=rem)

    # Short gradient refinement
    rem = budget_s - (time.time() - start_time)
    if rem > 2.0:
        max_iters = int(min(50000, max(5000, rem * 10000)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, max_iters, step0=0.3, beta=0.9, rng=rng, max_time=rem)

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    n_cur = base.shape[0]
    base = project_onto_box_sum(base, n_cur / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # Random binary sampling at current resolution
    samples = 5000 if n_cur <= 128 else 2000
    for _ in range(samples):
        if time.time() - start >= budget_s - 2.0:
            break
        h_rand = random_binary_balanced(n_cur, rng)
        sc = compute_c5(h_rand)
        if sc < best_score:
            best_score, best_h = sc, h_rand.copy()

    # Thue-Morse candidate
    thue = thue_morse(n_cur)
    thue = project_onto_box_sum(thue, n_cur / 2.0)
    sc = compute_c5(thue)
    if sc < best_score:
        best_score, best_h = sc, thue.copy()

    # Gradient descent on current scale
    rem = budget_s - (time.time() - start)
    if rem > 5.0:
        max_iters = int(min(300000, max(50000, rem * 25000)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, max_iters, step0=0.5, beta=0.9, rng=rng, max_time=rem)

    # Multiscale refinement
    for target_n in (128, 256, 512):
        if time.time() - start >= budget_s - 2.0:
            break
        best_h, best_score = process_scale(best_h, target_n, rng, start, budget_s)

    # Final polishing (if time permits)
    rem = budget_s - (time.time() - start)
    if rem > 5.0 and best_h.size >= 512:
        steps = int(min(400000, max(80000, rem * 30000)))
        best_h, best_score = simulated_annealing(best_h, steps, 0.01, 1e-6, rng, max_time=rem)

    # Ensure feasibility
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
