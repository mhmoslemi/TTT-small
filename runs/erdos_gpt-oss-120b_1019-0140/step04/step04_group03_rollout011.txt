def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = np.max(corr)
        # find max shift indices
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # add occasional noise
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project to [0,1] sum constraint
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # Ensure brackets
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
