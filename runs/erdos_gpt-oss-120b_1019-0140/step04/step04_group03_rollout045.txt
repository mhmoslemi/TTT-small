def differential_evolution_optimize(n, time_limit, rng, init_h=None):
    start = time.time()
    best_h = None
    best_score = np.inf
    pop_size = 30
    F = 0.8
    CR = 0.9
    
    # initial population
    pop = np.empty((pop_size, n), dtype=float)
    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        if i == 0 and init_h is not None:
            h = init_h.copy()
        else:
            # random binary balanced
            h = random_binary_balanced(n, rng)
        h = project_onto_box_sum(h, n/2)
        pop[i] = h
        scores[i] = compute_c5(h)
        if scores[i] < best_score:
            best_score = scores[i]
            best_h = h.copy()
    
    # main loop
    while time.time() - start < time_limit - 1.0:  # keep 1 sec margin
        for i in range(pop_size):
            if time.time() - start >= time_limit - 0.5:
                break
            # select three distinct indices a,b,c != i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, n/2)
            # crossover
            cross_mask = rng.random(n) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = project_onto_box_sum(trial, n/2)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

def gradient_descent_momentum_custom(h0, max_iters, step0, beta, rng, time_limit):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def simulated_annealing_refine(h, cur_score, time_limit, rng):
    start = time.time()
    n = h.size
    best_h = h.copy()
    best_score = cur_score
    T0 = 0.01
    T_final = 1e-6
    # total steps choose based on time
    # steps = int(time_limit * 5000)  # as per earlier code
    steps = int(min(200000, max(5000, time_limit * 5000)))
    for step in range(steps):
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        T = T0 * ((steps - step) / steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        # delta from i to j
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # ensure values in range (they will be)
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda such that sum clip(v - lambda) = target_sum
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        projected = np.clip(v - lam_mid, lo, hi)
        s = projected.sum()
        if abs(s - target_sum) < 1e-12:
            return projected
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    # after max_iter, return projection with last lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def gradient_descent_momentum_custom(h0, max_iters, step0, beta, rng, time_limit):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time.time() - start >= time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def differential_evolution_optimize(n, time_limit, rng, init_h=None):
    start = time.time()
    pop_size = 30
    F = 0.8
    CR = 0.9
    target = n / 2.0
    pop = np.empty((pop_size, n), dtype=float)
    scores = np.empty(pop_size, dtype=float)
    best_h = None
    best_score = np.inf
    for i in range(pop_size):
        if i == 0 and init_h is not None:
            h = init_h.copy()
        else:
            h = random_binary_balanced(n, rng)
        h = project_onto_box_sum(h, target)
        pop[i] = h
        sc = compute_c5(h)
        scores[i] = sc
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    while time.time() - start < time_limit - 0.5:
        for i in range(pop_size):
            if time.time() - start >= time_limit - 0.2:
                break
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, target)
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(0, n)] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_onto_box_sum(trial, target)
            trial_sc = compute_c5(trial)
            if trial_sc <= scores[i]:
                pop[i] = trial
                scores[i] = trial_sc
                if trial_sc < best_score:
                    best_score = trial_sc
                    best_h = trial.copy()
    return best_h, best_score

def simulated_annealing_refine(h, cur_score, time_limit, rng):
    start = time.time()
    n = h.size
    best_h = h.copy()
    best_score = cur_score
    T0 = 0.01
    T_final = 1e-6
    steps = int(min(200000, max(5000, time_limit * 5000)))
    for step in range(steps):
        if time.time() - start >= time_limit:
            break
        T = T0 * ((steps - step) / steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Quick fallback for tiny budgets
    if budget_s < 5.0:
        n0 = 64
        h0 = random_binary_balanced(n0, rng)
        h0 = project_onto_box_sum(h0, n0 / 2.0)
        return h0, compute_c5(h0), n0

    n = 1024  # final resolution
    # Prepare initial vector
    if 'initial_h_values' in globals():
        init_h = upsample(np.asarray(initial_h_values, dtype=float), n)
    else:
        init_h = random_binary_balanced(n, rng)
    init_h = project_onto_box_sum(init_h, n / 2.0)

    # Differential Evolution (≈70% of budget)
    remaining = budget_s - (time.time() - start)
    de_time = max(0.0, min(remaining * 0.7, remaining - 5.0))
    best_h, best_score = differential_evolution_optimize(n, de_time, rng, init_h)

    # Gradient descent refinement
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        max_iters = int(min(300000, max(50000, remaining * 5000)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, max_iters, step0=0.5, beta=0.9, rng=rng, time_limit=remaining)

    # Simulated annealing final polishing
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        best_h, best_score = simulated_annealing_refine(best_h, best_score, remaining, rng)

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
