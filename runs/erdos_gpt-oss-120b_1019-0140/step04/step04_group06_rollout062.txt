def smooth_max_gradient_descent(h0, n_iter, step0, beta, alpha_start, alpha_end, rng, time_left):

   h = h0.copy()

   n = h.size

   dx = 2.0 / n

   target_sum = n / 2.0

   best_h = h.copy()

   best_score = compute_c5(h)

   m = np.zeros_like(h)

   for it in range(1, n_iter+1):

        if time_left() <= 0: break

        corr = np.correlate(h, 1.0 - h, mode='full')

        score = np.max(corr) * dx

        if score < best_score:

            best_score = score

            best_h = h.copy()

        # schedule alpha

        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (n_iter - 1) if n_iter>1 else alpha_end

        # smoothed weights

        max_corr = np.max(corr)

        w = np.exp(alpha * (corr - max_corr))

        w_sum = np.sum(w)

        w = w / w_sum

        # compute A and B

        # correlate w with h: using np.correlate as described

        conv_wh = np.correlate(w, h, mode='full')

        A = conv_wh[n-1 : n-1 + n]

        conv_wrev_h = np.correlate(w[::-1], h, mode='full')

        B = conv_wrev_h[n-1 : n-1 + n]

        grad = dx * (1.0 - A - B)

        m = beta * m + (1.0 - beta) * grad

        eta = step0 / np.sqrt(it)

        h_new = h - eta * m

        # occasional noise

        if rng.random() < 0.02:

            h_new += rng.normal(scale=0.02, size=n)

        h_new = project_onto_box_sum(h_new, target_sum)

        h = h_new

   # final eval

   final_score = compute_c5(best_h)

   return best_h, final_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project vector v onto the box [lo,hi]^n while forcing sum(v)=s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10.0 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10.0 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    # binary Thue–Morse sequence (0/1)
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def smooth_max_gradient_descent(h0, n_iter, step0, beta,
                               alpha_start, alpha_end, rng, time_left):
    """Projected gradient descent on a smoothed max objective."""
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0
    m = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        score = np.max(corr) * dx
        if score < best_score:
            best_score = score
            best_h = h.copy()

        # schedule smoothing parameter
        if n_iter > 1:
            alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (n_iter - 1)
        else:
            alpha = alpha_end

        # softmax weights for all shifts
        max_corr = np.max(corr)
        w = np.exp(alpha * (corr - max_corr))
        w /= np.sum(w)           # now sum(w)=1

        # convolution terms A = sum_s w_s * h_{i+s}, B = sum_s w_s * h_{i-s}
        conv_wh = np.correlate(w, h, mode='full')
        A = conv_wh[n - 1:n - 1 + n]
        conv_wrev_h = np.correlate(w[::-1], h, mode='full')
        B = conv_wrev_h[n - 1:n - 1 + n]

        grad = dx * (1.0 - A - B)   # gradient of smoothed max
        m = beta * m + (1.0 - beta) * grad

        eta = step0 / np.sqrt(it)
        h_new = h - eta * m

        # occasional small random perturbation
        if rng.random() < 0.02:
            h_new += rng.normal(scale=0.02, size=n)

        h = project_onto_box_sum(h_new, target_sum)

    # final evaluation of the best seen
    final_score = compute_c5(best_h)
    return best_h, final_score


def local_pairwise_refine(h, best_score, rng, time_left):
    """Simple pairwise swaps and occasional gradient nudges."""
    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0

    while time_left() > 0.5:
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(h[i], 1.0 - h[j]) * rng.uniform(0.05, 0.3)
        if delta > 0.0:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score = sc_new
                h = h_new

        # occasional gradient step with high smoothing (sharp max)
        if rng.random() < 0.1:
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = np.max(corr)
            alpha = 200.0
            w = np.exp(alpha * (corr - max_corr))
            w /= np.sum(w)
            conv_wh = np.correlate(w, h, mode='full')
            A = conv_wh[n - 1:n - 1 + n]
            conv_wrev_h = np.correlate(w[::-1], h, mode='full')
            B = conv_wrev_h[n - 1:n - 1 + n]
            grad = dx * (1.0 - A - B)
            h_tmp = h - 0.01 * grad
            h_tmp = project_onto_box_sum(h_tmp, target_sum)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < best_score:
                best_score = sc_tmp
                h = h_tmp

    return h, best_score
