def sa_optimize(h0, time_limit, rng):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # temperature schedule using linear decay to near zero
    init_temp = 0.02
    # calculate total allowed iterations approx: loop while time left > 0
    it = 0
    # precompute max_iters maybe large
    while time.time() - start < time_limit:
        # compute temp
        elapsed = time.time() - start
        proportion = elapsed / time_limit
        temp = init_temp * max(0.0, 1.0 - proportion) # linear decay
        # pick move type
        if rng.random() < 0.2:
            # swap move
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            # swap
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            # accept always? We could compute delta; but swapping doesn't affect sum or bounds.
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                # acceptance with probability exp(-(new_score-cur_score)/temp)
                if temp > 0 and rng.random() < np.exp(-(new_score - cur_score)/temp):
                    cur_score = new_score
                else:
                    # revert swap
                    h[i], h[j] = h[j], h[i]
        else:
            # transfer move
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i: j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            # scale delta based on temp
            # scaling factor up to 1
            scale = temp / init_temp # between 0 and 1
            delta = max_delta * rng.random() * scale
            if delta <= 0:

            # apply
            h[i] -= delta
            h[j] += delta
            new_score = compute_c5(h)
            if new_score < cur_score:

                    # revert
                    h[i] += delta
                    h[j] -= delta
        it += 1
        # maybe occasional occasional heavy moves
        # maybe every 5000 it, try random mass move between 3 indexes
        # optional
    return best_h, best_score

    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # get initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # generate a random balanced binary of length 64
        n0 = 64
        base = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, n0 // 2, replace=False)
        base[idx] = 1.0

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {x: lo <= x_i <= hi, sum x_i = total} (Euclidean)."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return discrete C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to size new_n, preserving mean via projection."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    interp = np.interp(x_new, x_old, arr)
    return interp


def random_balanced_binary(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    vec = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    vec[idx] = 1.0
    return vec


def sa_optimize(h0, time_limit, rng):
    """Simple simulated annealing preserving box and sum constraints."""
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    end = start + time_limit
    init_temp = 0.05

    while time.time() < end:
        now = time.time()
        prop = (now - start) / time_limit
        temp = init_temp * max(0.0, 1.0 - prop)

        if rng.random() < 0.1:                     # swap move
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                else:
                    h[i], h[j] = h[j], h[i]

        else:                                      # transfer move
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            scale = max(temp / init_temp, 0.01)     # keep minimal activity
            delta = max_delta * rng.random() * scale
            if delta <= 0.0:
                continue
            h[i] -= delta
            h[j] += delta
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                else:
                    h[i] += delta
                    h[j] -= delta
    return best_h, best_score


def targeted_local_refinement(h, rng, time_limit):
    """Iteratively reduce the worst shift by moving mass between mismatched positions."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    start = time.time()
    end = start + time_limit

    while time.time() < end:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        shift_idx = int(np.argmax(corr))
        shift = shift_idx - (n - 1)                     # shift in index units

        # overlapping index ranges
        if shift >= 0:
            i_start = 0
            i_end = n - shift
        else:
            i_start = -shift
            i_end = n
        i_range = np.arange(i_start, i_end, dtype=int)
        i_shifted = i_range + shift

        donor_mask = (best_h[i_range] > 0.0) & (best_h[i_shifted] < 1.0)
        receiver_mask = (best_h[i_range] < 1.0) & (best_h[i_shifted] > 0.0)
        donors = i_range[donor_mask]
        receivers = i_range[receiver_mask]

        if donors.size == 0 or receivers.size == 0:
            break

        i = rng.choice(donors)
        j = rng.choice(receivers)
        if i == j:

        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0.0:

        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    return best_h, best_score
