```
def compute_grad_softmax(h, beta):
    n = h.size
    corr = compute_corr_full_fft(h)  # shape (2n-1)
    # shift indexing: lag k = idx - (n-1)
    # weights:
    max_c = np.max(corr)  # for stability: subtract max
    w = np.exp(beta * (corr - max_c))
    w /= w.sum()
    # compute gradient as sum_k w_k * grad_k
    grad = np.zeros_like(h)
    # compute contributions for k >= 0
    for shift in range(-(n-1), n):
        idx = shift + (n-1)
        weight = w[idx]
        if weight == 0:
            continue
        k = shift
        if k >= 0:
            # add (1 - h[i+k]) to grad[i] for i=0..n-k-1
            grad[:n-k] += weight * (1 - h[k:])
            # subtract h[i] from grad[i+k] for i=0..n-k-1
            grad[k:] -= weight * h[:n-k]
        else:
            m = -k
            grad[m:] += weight * (1 - h[:n-m])
            grad[:n-m] -= weight * h[m:]
    return grad
```

```
def gradient_descent_softmax(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        grad = compute_grad_softmax(h, beta=beta)
        momentum = 0.9 * momentum + grad   # using fixed momentum coefficient
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random perturb
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=h.size)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full linear cross‑correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    return np.max(compute_corr_full_fft(h)) * (2.0 / n)


def compute_grad_softmax(h, beta):
    """Gradient of soft‑max approximation of max‑correlation."""
    n = h.size
    corr = compute_corr_full_fft(h)
    maxc = np.max(corr)
    w = np.exp(beta * (corr - maxc))
    w /= w.sum()
    grad = np.zeros_like(h)
    for shift in range(-(n - 1), n):
        idx = shift + (n - 1)
        weight = w[idx]
        if weight == 0.0:
            continue
        if shift >= 0:
            k = shift
            grad[:n - k] += weight * (1.0 - h[k:])
            grad[k:] -= weight * h[:n - k]
        else:
            m = -shift
            grad[m:] += weight * (1.0 - h[:n - m])
            grad[:n - m] -= weight * h[m:]
    return grad


def gradient_descent_softmax(h0, n_iter, step0, beta_softmax, rng, time_left):
    """Projected gradient descent using soft‑max gradient."""
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        grad = compute_grad_softmax(h, beta_softmax)
        momentum = 0.9 * momentum + grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=h.size)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    idx = rng.choice(n, size=n // 2, replace=False)
