import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from a smaller to a larger array."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Average subgradient of the max‑correlation term over given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum and occasional jitter."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    """Transfer small mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    """Transfer mass from points contributing most to the worst lag."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift == 0:
            continue
        if shift > 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score

def anneal_h(h, rng, time_left, max_iters=None):
    """Simulated annealing using pairwise mass transfers."""
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.shape[0]
    T = cur_score * 0.02 + 1e-4
    decay = 0.9995
    it = 0
    if max_iters is None:
        max_iters = int(time_left() * 5)
    while time_left() > 0 and it < max_iters:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_score = compute_c5(new_h)
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                cur_h = new_h
                cur_score = new_score
                if new_score < best_score:
                    best_h = new_h.copy()
                    best_score = new_score
        T = max(T * decay, 1e-8)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]

    for n in scales:
        if time_left() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
            base = project_onto_box_sum(base, n / 2.0)

        # Subgradient descent
        if time_left() > 10:
            max_iters = int(min(80000, max(2000, time_left() * (25000.0 / n))))
            base, sc = gradient_descent_adaptive(base, max_iters,
                                                 step0=0.5, beta=0.95,
                                                 rng=rng, time_left=time_left)
            if sc < best_score:
                best_h, best_score = base.copy(), sc

        # Simulated annealing
        if time_left() > 5:
            max_iters = int(min(20000, time_left() * 5))
            base, sc = anneal_h(base, rng, time_left, max_iters=max_iters)
            if sc < best_score:
                best_h, best_score = base.copy(), sc

        # Max‑shift improvement
        if time_left() > 5:
            base, sc = improve_via_max_shift(base, rng, time_left)
            if sc < best_score:
                best_h, best_score = base.copy(), sc

        # Local pair moves
        if time_left() > 5:
            base, sc = local_pair_moves(base, rng, time_left)
            if sc < best_score:
                best_h, best_score = base.copy(), sc

        # Random balanced binary restarts
        if time_left() > 10:
            for _ in range(2):
                if time_left() < 2:
                    break
                h_rand = random_binary_balanced(n, rng)
                max_iters_rand = int(min(6000, max(500, time_left() * 2)))
                h_rand, sc_rand = gradient_descent_adaptive(h_rand, max_iters_rand,
                                                            step0=0.4, beta=0.9,
                                                            rng=rng, time_left=time_left)
                if sc_rand < best_score:
                    best_h, best_score = h_rand.copy(), sc_rand
                if time_left() > 3:
                    h_rand, sc_rand = local_pair_moves(h_rand, rng, time_left)
                    if sc_rand < best_score:
                        best_h, best_score = h_rand.copy(), sc_rand

    # Final intensive refinement
    if time_left() > 5:
        max_iters_final = int(min(150000, max(50000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(best_h, max_iters_final,
                                                      step0=0.3, beta=0.96,
                                                      rng=rng, time_left=time_left)

    # Final annealing
    if time_left() > 5:
        best_h, best_score = anneal_h(best_h, rng, time_left,
                                      max_iters=int(time_left() * 4))

    # Final targeted and pair moves
    if time_left() > 2:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
