def anneal_binary(h, rng, time_left, max_iters=10**6, init_temp=0.5, decay=0.9995, min_temp=1e-8):
    cur_h = h.copy()
    cur_val = compute_c5_fast(cur_h)
    best_h, best_val = cur_h.copy(), cur_val
    T = init_temp
    i=0
    while i < max_iters and time_left() > 0.0 and T > min_temp:
        # choose swap
        ones = np.where(cur_h==1)[0]
        zeros = np.where(cur_h==0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # perform swap
        cur_h[i1] = 0.0
        cur_h[i0] = 1.0
        new_val = compute_c5_fast(cur_h)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i1] = 1.0
            cur_h[i0] = 0.0
        T = max(min_temp, T * decay)
        i += 1
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))

    n = 512
    target_sum = n // 2
    best_h = None
    best_val = np.inf

    # Load initial if available
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        candidates.append(project_onto_box_sum(init, target_sum))
    # Add Thue-Morse
    candidates.append(thue_morse_binary(n))
    # Add a few random balanced
    for _ in range(4):
        candidates.append(random_binary_balanced(n, rng))

    # Evaluate each candidate directly (compute c5) and maybe apply refinement
    for c in candidates:
        if time_left() < 0.5: break
        # binary pattern may be already best
        c_val = compute_c5_fast(c)
        if c_val < best_val:

        # Optionally apply annealing from this starting point
        if time_left() > 10.0:
            h_ann, v_ann = anneal_binary(c.copy(), rng, time_left, max_iters=300000, init_temp=0.5, decay=0.9997, min_temp=1e-6)

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    # If target outside feasible range, return clamped array
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    # λ such that sum(clip(v-λ, lo, hi)) = target
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # expand bounds if needed
    while f(lam_low) > target:
        lam_low -= (hi - lo) * 10
    while f(lam_high) < target:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):

def anneal_binary(h0, rng, time_left, max_iters=400000, init_temp=0.5, decay=0.9998, min_temp=1e-6):

    cur_val = compute_c5_fast(h)
    best_h = h.copy()
    best_val = cur_val
    T = init_temp
    it = 0
    while it < max_iters and time_left() > 0.0 and T > min_temp:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # swap bits
        h[i1] = 0.0
        h[i0] = 1.0
        new_val = compute_c5_fast(h)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val

                best_h = h.copy()
        else:
            # revert swap
            h[i1] = 1.0
            h[i0] = 0.0
        T = max(min_temp, T * decay)
        it += 1
    return best_h, best_val

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)

    return best_h, best

def thue_morse_binary(n):
    arr = np.empty(n, dtype=float)
    for i in range(n):
        arr[i] = (bin(i).count('1') & 1)
    return arr

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    n = 512
    target_sum = n / 2.0
    cand = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        cand.append(project_onto_box_sum(init, target_sum))

    cand.append(thue_morse_binary(n))
    for _ in range(3):
        cand.append(random_binary_balanced(n, rng))

    best_h = None
    best_val = np.inf
    target_val = 0.3808

    for c in cand:
        if time_left() < 0.5:
            break
        val = compute_c5_fast(c)
        if val < best_val:
            best_val, best_h = val, c.copy()
            if best_val <= target_val:
                return best_h, best_val, n
        if time_left() > 5.0:
            h_ann, v_ann = anneal_binary(c.copy(), rng, time_left,
                                         max_iters=200000,
                                         init_temp=0.5,
                                         decay=0.9997,
                                         min_temp=1e-6)
            if v_ann < best_val:
                best_val, best_h = v_ann, h_ann.copy()
                if best_val <= target_val:
                    return best_h, best_val, n

    while time_left() > 20.0:
        h0 = random_binary_balanced(n, rng)
        max_iters = int(min(300000, time_left() * 300))
        h_ann, v_ann = anneal_binary(h0, rng, time_left,
                                     max_iters=max_iters,
                                     init_temp=0.6,
                                     decay=0.9996,
                                     min_temp=1e-7)
        if v_ann < best_val:
            best_val, best_h = v_ann, h_ann.copy()
            if best_val <= target_val:
                break

    if best_h is not None and time_left() > 5.0:
        # small continuous refinement
        h_ref, s_ref = gradient_descent_momentum(best_h, time_left,
                                                 step0=0.5,
                                                 max_iter=50000)
        if s_ref < best_val:
            best_val, best_h = s_ref, h_ref

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), target_sum)
        best_val = compute_c5_fast(best_h)
    else:
        best_h = project_onto_box_sum(best_h, target_sum)
        best_val = compute_c5_fast(best_h)

    return best_h, best_val, n
