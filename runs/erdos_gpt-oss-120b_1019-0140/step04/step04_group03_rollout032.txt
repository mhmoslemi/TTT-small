import time, numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # Binary search for lambda such that sum(clipped(v - lam)) = target_sum
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift == 0:
            grad -= 2 * h
        elif shift > 0:
            grad[shift:] -= h[:-shift]
            grad[:n-shift] -= h[shift:]
        else:
            s = -shift

            grad[s:] -= h[:n-s]

def worst_shifts(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    idxes = np.where(corr >= max_val - 1e-12)[0]
    shifts = idxes - (n - 1)
    return shifts

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        shifts = worst_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift == 0:
            grad -= 2 * h
        elif shift > 0:
            grad[shift:] -= h[:-shift]
            grad[:n - shift] -= h[shift:]
        else:
            s = -shift
            grad[:n - s] -= h[s:]
            grad[s:] -= h[:n - s]
    if len(shifts) > 0:
        grad /= len(shifts)
    grad += 1.0
    return grad

def worst_shifts(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    idxes = np.where(corr >= max_val - 1e-12)[0]
    return idxes - (n - 1)

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        shifts = worst_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_score = np.inf
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    scales = [64, 128, 256, 512, 1024]
    for idx, n in enumerate(scales):
        if time.time() - start >= budget_s - 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # random binary sampling
        sample_limit = 500 if n >= 256 else 500
        for _ in range(sample_limit):
            if time.time() - start >= budget_s - 1:
                break
            h_rand = random_binary_balanced(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score, best_h = sc, h_rand.copy()

        # gradient descent restarts
        restarts = 3 if n <= 256 else 2
        for r in range(restarts):
            if time.time() - start >= budget_s - 5:
                break
            h0 = base.copy()
            if r > 0:
                h0 += rng.normal(scale=0.04, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, n / 2.0)
            remaining = budget_s - (time.time() - start)
            max_iters = int(min(40000, max(15000, remaining * 25000 / max(64, n))))
            h_opt, sc_opt = gradient_descent_momentum_custom(
                h0, n_iter=max_iters, step0=0.5, beta=0.9,
                rng=rng, max_time=remaining)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt

        # simulated annealing refinement
        if best_h is not None and best_h.size == n and time.time() - start < budget_s - 5:
            h_cur = best_h.copy()
            cur_score = best_score
            remaining = budget_s - (time.time() - start)
            steps = int(min(30000, max(8000, remaining * 800)))
            T0, T_final = 0.02, 1e-6
            for s in range(steps):
                if time.time() - start >= budget_s:
                    break
                T = T0 * ((steps - s) / steps) + T_final
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(h_cur[i], 1.0 - h_cur[j])
                if max_move <= 0:
                    continue
                delta = max_move * rng.random()
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_score = compute_c5(h_new)
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                    h_cur = h_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score, best_h = cur_score, h_cur.copy()
            base = best_h

        if idx < len(scales) - 1:
            base = best_h if best_h is not None else base

    # final refinement
    if best_h is not None and budget_s - (time.time() - start) > 5:
        n = best_h.size
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(200000, max(50000, remaining * 50000 / max(64, n))))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=max_iters, step0=0.2, beta=0.9,
            rng=rng, max_time=remaining)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)
        # hill climbing until time expires
        while time.time() - start < budget_s - 0.1:
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(best_h[i], 1.0 - best_h[j])
            if max_move <= 0:
                continue
            delta = max_move * rng.random()
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            new_score = compute_c5(h_new)
            if new_score < best_score:
                best_h, best_score = h_new, new_score

    if best_h is None:
        n = 512
        thue = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
        best_h = project_onto_box_sum(thue, n / 2.0)
        best_score = compute_c5(best_h)
        n = best_h.size

    return best_h, best_score, best_h.size
