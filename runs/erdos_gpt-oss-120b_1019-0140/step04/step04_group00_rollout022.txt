def compute_c5_fast(h):
    n = h.shape[0]
    # compute correlation h with (1-h) using FFT convolution for faster speed
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_h_rev = np.fft.rfft(h_rev, n=N)
    fft_one_h = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_h_rev * fft_one_h, n=N)
    corr = conv[:2*n-1]
    return corr.max() * (2.0 / n)

def hill_climb_swap(h, best_score, rng, max_iter, time_left):
    cur = best_score
    h_cur = h.copy()
    n = h.shape[0]
    ones = np.where(h_cur==1)[0]
    zeros = np.where(h_cur==0)[0]
    for it in range(max_iter):
        if time_left() <= 0:
            break
        # choose random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h_cur[i], h_cur[j] = 0, 1
        new_score = compute_c5_fast(h_cur)
        if new_score < cur:
            cur = new_score
            ones = np.where(h_cur==1)[0]
            zeros = np.where(h_cur==0)[0]
        else:
            # revert
            h_cur[i], h_cur[j] = 1, 0
    return h_cur, cur

def genetic_algorithm(rng, time_left, n=512, pop_size=40, target=0.3808):

    pop = []
    # If initial_h_values available:
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # Resample to n if needed:
        if init.size != n:
            init = resample(init, n)
        # threshold to binary:
        idx = np.argpartition(init, -w)[-w:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        pop.append(h_bin)
    # Add Thue-Morse
    try:
        h_tm = thue_morse_binary(n)
        pop.append(h_tm.astype(float))
    except Exception:
        pass
    # Fill rest random
    while len(pop) < pop_size:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=w, replace=False)] = 1.0
        pop.append(h)
    # Compute fitness
    scores = np.array([compute_c5_fast(h) for h in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # GA loop
    tournament_size = 3
    crossover_prob = 0.7
    mutation_prob = 0.1
    while time_left() > 0.5:
        new_pop = []
        while len(new_pop) < pop_size:
            # tournament selection
            idxs = rng.choice(pop_size, size=tournament_size, replace=False)
            parent1 = pop[idxs[np.argmin(scores[idxs])]]
            idxs = rng.choice(pop_size, size=tournament_size, replace=False)
            parent2 = pop[idxs[np.argmin(scores[idxs])]]
            child = parent1.copy()
            if rng.random() < crossover_prob:
                # two-point crossover
                p1, p2 = sorted(rng.integers(0, n, size=2))
                # swap segments between parents
                child[p1:p2] = parent2[p1:p2]
            # repair to balance

            # mutation
            if rng.random() < mutation_prob:
                # swap two bits
                ones = np.where(child==1)[0]
                zeros = np.where(child==0)[0]
                if ones.size>0 and zeros.size>0:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    child[i], child[j] = 0, 1
            new_pop.append(child)
        # evaluate new_pop
        new_scores = np.array([compute_c5_fast(h) for h in new_pop])
        # combine populations? // For simplicity, replace old pop with new
        pop = new_pop
        scores = new_scores
        # update best

        cur_best_score = scores[cur_best_idx]

            best_score = cur_best_score
            best_h = pop[cur_best_idx].copy()
        # early exit if below target

import time, numpy as np

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_h_rev = np.fft.rfft(h_rev, n=N)
    fft_one_h = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_h_rev * fft_one_h, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def repair_to_balance(arr, weight, rng):
    arr = np.where(arr >= 0.5, 1.0, 0.0)
    cur = int(arr.sum())
    if cur > weight:
        ones = np.where(arr == 1)[0]
        to_zero = rng.choice(ones, size=cur - weight, replace=False)
        arr[to_zero] = 0.0
    elif cur < weight:
        zeros = np.where(arr == 0)[0]
        to_one = rng.choice(zeros, size=weight - cur, replace=False)
        arr[to_one] = 1.0
    return arr

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def hill_climb_swap(h, cur_score, rng, max_iter, time_left):
    n = h.shape[0]
    h_cur = h.copy()
    cur = cur_score
    ones = np.where(h_cur == 1)[0]
    zeros = np.where(h_cur == 0)[0]
    for _ in range(max_iter):
        if time_left() <= 0.0:
            break
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_cur[i], h_cur[j] = 0.0, 1.0
        new = compute_c5_fast(h_cur)
        if new < cur:
            cur = new
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
        else:
            h_cur[i], h_cur[j] = 1.0, 0.0
    return h_cur, cur

def binary_swap_anneal(h0, rng, time_left, max_iters=200000):

    cur = compute_c5_fast(h)

        if ones.size == 0 or zeros.size == 0:
            break

        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0

        if delta < 0 or rng.random() < np.exp(-delta / T):

            if new < best_s:
                best_s, best_h = new, h.copy()

def genetic_algorithm(rng, time_left, n=512, pop_size=40, target=0.3808):
    w = n // 2
    pop = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            init = resample(init, n)
        idx = np.argpartition(init, -w)[-w:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        pop.append(h_bin)
    try:
        pop.append(thue_morse_binary(n).astype(float))
    except Exception:
        pass
    while len(pop) < pop_size:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=w, replace=False)] = 1.0
        pop.append(h)
    scores = np.array([compute_c5_fast(h) for h in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    tournament = 3
    pcross = 0.7
    pmut = 0.1
    while time_left() > 0.5:
        new_pop = []
        while len(new_pop) < pop_size:
            idxs = rng.choice(pop_size, size=tournament, replace=False)
            p1 = pop[idxs[np.argmin(scores[idxs])]]
            idxs = rng.choice(pop_size, size=tournament, replace=False)
