def compute_corr_fft(h):
    n = h.size
    N = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(h, N)
    g = np.fft.rfft((1.0 - h)[::-1], N)
    corr = np.fft.irfft(f * g, N)[:2*n-1]
    # due to floating errors, may have small negative values; ok.
    return corr

def compute_c5_and_shift(h):
    corr = compute_corr_fft(h)
    max_idx = np.argmax(corr)
    max_val = corr[max_idx] * (2.0 / h.size)  # dx scaling
    # shift = max_idx - (n-1)
    return max_val, max_idx

Now gradient:

def gradient_max_shift(h, max_idx):
    n = h.size
    shift = max_idx - (n - 1)
    g = np.zeros_like(h)
    dx = 2.0 / n
    if shift >= 0:
        # i from 0 to n-1-shift
        i = np.arange(0, n - shift)
        g[i] += (1.0 - h[i + shift])
        g[i + shift] += (-h[i])
    else:
        t = -shift
        i = np.arange(0, n - t)
        # term h[i+t] * (1 - h[i])
        g[i + t] += (1.0 - h[i])
        g[i] += (-h[i + t])
    return g * dx

Note: This gradient is for the correlation term (without dx scaling, then we multiply). Verified.

def gradient_descent(h, time_limit, start_alpha=0.1):
    start = time.time()
    best_h = h.copy()
    best_val, _ = compute_c5_and_shift(best_h)
    iteration = 0
    while time.time() - start < time_limit:
        # dynamic alpha: maybe decrease linearly
        elapsed = time.time() - start
        frac = elapsed / time_limit
        alpha = start_alpha * (1 - frac)  # linear decay
        max_val, max_idx = compute_c5_and_shift(h)
        g = gradient_max_shift(h, max_idx)
        h_new = h - alpha * g
        h_new = project_onto_box_sum(h_new, h_new.size/2.0)
        # record if improved
        new_val, _ = compute_c5_and_shift(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        h = h_new
        iteration += 1
        # optionally break if alpha becomes tiny
        if alpha < 1e-6:
            break
    return best_h, best_val

def binary_sa(h_bin, time_limit, rng):
    n = h_bin.size
    current = h_bin.copy()
    cur_val, _ = compute_c5_and_shift(current)
    best = current.copy()
    best_val = cur_val

    start = time.time()
    # parameters
    max_iter = int(1e7)
    # start temperature maybe 0.01?
    init_temp = 0.01
    for it in range(max_iter):
        if time.time() - start >= time_limit:
            break
        # temperature schedule exponential
        temp = init_temp * (0.995 ** it)
        if temp < 1e-6:
            temp = 0
        # pick random i with 1, j with 0
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        current[i] = 0.0
        current[j] = 1.0
        new_val, _ = compute_c5_and_shift(current)
        # accept?
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = current.copy()
        else:
            # Metropolis
            if temp > 0 and rng.random() < np.exp(-(new_val - cur_val) / temp):
                cur_val = new_val
            else:
                # revert swap
                current[i] = 1.0
                current[j] = 0.0
        # early break if near target
        if best_val <= 0.3808:
            break
    return best, best_val

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points on a periodic interval."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Generate Thue‑Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def compute_corr_fft(h):
    """Correlation h ⋆ (1‑h) using FFT, matching np.correlate(..., 'full')."""
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())  # next power of two
    f_fft = np.fft.rfft(h, N)
    g_rev = (1.0 - h)[::-1]
    g_fft = np.fft.rfft(g_rev, N)
    conv = np.fft.irfft(f_fft * g_fft, N)[:2 * n - 1]
    return conv[::-1]  # reverse to get correlation order


def compute_c5_and_shift(h):
    """Return (C5 value, index of maximal correlation)."""
    n = h.size
    corr = compute_corr_fft(h)
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx] * (2.0 / n)
    return max_val, max_idx


def gradient_max_shift(h, max_idx):
    """Subgradient of the maximal correlation term."""
    n = h.size
    shift = max_idx - (n - 1)
    g = np.zeros_like(h)
    dx = 2.0 / n
    if shift >= 0:
        i = np.arange(0, n - shift)
        g[i] += (1.0 - h[i + shift])
        g[i + shift] += (-h[i])
    else:
        t = -shift
        i = np.arange(0, n - t)
        g[i + t] += (1.0 - h[i])
        g[i] += (-h[i + t])
    return g * dx


def gradient_descent(h0, time_limit, start_alpha=0.1):
    """Projected gradient descent on the max‑correlation objective."""
    h = h0.copy()
    best_h = h.copy()
    best_val, _ = compute_c5_and_shift(h)
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        alpha = start_alpha * (1.0 - frac)
        if alpha < 1e-6:
            break
        cur_val, max_idx = compute_c5_and_shift(h)
        g = gradient_max_shift(h, max_idx)
        h_new = project_onto_box_sum(h - alpha * g, h.size / 2.0)
        new_val, _ = compute_c5_and_shift(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val


def to_balanced_binary(h, rng):
    """Return a 0/1 vector with exactly half ones (top half of h)."""
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k - 1)[:k]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def binary_sa(h0, time_limit, rng):
    """Simulated annealing on balanced binary vectors."""
    n = h0.size
    cur = h0.copy()
    cur_val, _ = compute_c5_and_shift(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    init_temp = 0.02
    it = 0
    while time.time() - start < time_limit:
        it += 1
        temp = init_temp * (0.995 ** it)
        if temp < 1e-6:
            temp = 0.0
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0.0
        cur[j] = 1.0
        new_val, _ = compute_c5_and_shift(cur)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            if temp > 0.0 and rng.random() < np.exp(-(new_val - cur_val) / temp):
                cur_val = new_val
            else:
                cur[i] = 1.0
                cur[j] = 0.0
        if best_val <= 0.3808:
            break
    return best, best_val
