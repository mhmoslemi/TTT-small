    # b = 1 - a
    # c: current correlation (np.ndarray size 2*n-1)

    # compute d_a_i = -delta, d_a_j = +delta
    # compute corr(d_a, b)
    # Since d_a is sparse, can compute:
    # For shift s, contribution = -delta * b[i + s] + delta * b[j + s]
    # This is vectorizable: we can compute b shifted by i and j via indexing.
    # For shift index from -(n-1) to n-1; let idx = s + (n-1). So we need array of length 2*n-1.
    # b_i_shift = b shifted by -i: That is b[i + s] for each s such that i+s in [0,n-1]; else 0.

    # We can compute using convolution with one-hot vectors.
    # But we can just compute via np.correlate on d_a and b directly (fast enough O(n)).
    d_a = np.zeros(n, dtype=a.dtype)
    d_a[i] = -delta
    d_a[j] = delta
    c1 = np.correlate(d_a, b)   # (2n-1)
    c2 = np.correlate(a, d_a)   # (2n-1)
    # c3

    c3[center] = 2 * delta * delta
    shift = j - i
    c3[center + shift] = -delta * delta
    c3[center - shift] = -delta * delta
    c_prime = c + c1 - c2 - c3
    # update a and b
    a[i] -= delta
    a[j] += delta
    b[i] = 1 - a[i]
    b[j] = 1 - a[j]
    # compute new score
    new_score = np.max(c_prime) * (2.0 / n)
    return new_score, c_prime

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size

    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure brackets
    if f(lam_lo) < s:

        if time_left() <= 5:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)
        # first annealing phase
        h_opt, sc_opt = anneal_optimize(best_h, time_left, rng)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
        # second annealing if time permits
        if time_left() > 5:
            h_opt2, sc_opt2 = anneal_optimize(best_h, time_left, rng)
            if sc_opt2 < best_score:
                best_h, best_score = h_opt2, sc_opt2
        # quick SLSQP polish
        if time_left() > 8:
            h_opt3, sc_opt3 = slsqp_polish(best_h, max_iter=150)
            if sc_opt3 < best_score:
                best_h, best_score = h_opt3, sc_opt3

    # final refinement on the best resolution
    if time_left() > 10:
        h_opt, sc_opt = anneal_optimize(best_h, time_left, rng, max_iters=int(time_left()*100))
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
        if time_left() > 5:
            h_opt, sc_opt = slsqp_polish(best_h, max_iter=200)
            if sc_opt < best_score:
                best_h, best_score = h_opt, sc_opt

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
