import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to size new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def generate_random_balanced_binary(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def simulated_annealing(h0, time_limit, rng):
    """Projected SA that preserves sum and stays in [0,1]."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.015  # initial temperature

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # linear temperature decay
        frac = elapsed / time_limit
        temp = init_temp * max(0.0, 1.0 - frac)

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue

        # scale delta by temperature (larger early, smaller later)
        scale = temp / init_temp + 0.01
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            continue

        # propose move
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            if temp > 0.0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_score = new_score
                else:
                    h[i] += delta
                    h[j] -= delta
            else:
                h[i] += delta
                h[j] -= delta

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # target resolution (must be even and <1000)
    target_n = 512

    # obtain base solution
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = upsample(base, target_n)
        base = project_onto_box_sum(base, target_n / 2.0)
    else:
        base = generate_random_balanced_binary(target_n, rng)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # Main multi-start SA loop
    runs_done = 0
    max_runs = 30
    while time_left() > 2.0 and runs_done < max_runs:
        remaining_runs = max_runs - runs_done
        # allocate a generous slice of the remaining time
        alloc = time_left() * 0.9 / remaining_runs
        if alloc < 2.0:
            break

        if runs_done == 0:
            h0 = best_h.copy()
        else:
            # alternate between perturbed best and fresh binary
            if runs_done % 2 == 0:
                h0 = best_h.copy()
                noise = (rng.random(target_n) - 0.5) * 0.2  # [-0.1,0.1]
                h0 = np.clip(h0 + noise, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, target_n / 2.0)
            else:
                h0 = generate_random_balanced_binary(target_n, rng)

        cand_h, cand_score = simulated_annealing(h0, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        runs_done += 1

    # final refinement with any remaining budget
    remaining = time_left()
    if remaining > 1.0:
        best_h, best_score = simulated_annealing(best_h, remaining, rng)

    # final projection to guarantee feasibility
    best_h = project_onto_box_sum(best_h, target_n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, target_n
