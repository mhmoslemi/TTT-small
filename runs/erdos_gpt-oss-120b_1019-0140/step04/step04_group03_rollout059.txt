def gradient_descent_momentum_proj(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time.time() - start >= max_time:
            break
        # compute corr, maxcorr
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts where corr close to max, tolerance maybe 1e-12
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # gradient
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise injection every N steps maybe 5000
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        # score
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_refine(h, start_time, budget_s, rng):
    n = h.shape[0]
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    total_steps = int(min(50000, max(10000, (budget_s - (time.time() - start_time)) * 2000.0)))  # approximate
    T0 = 0.01
    T_final = 1e-6
    for step in range(total_steps):
        if time.time() - start_time >= budget_s:
            break
        fraction = step / total_steps
        T = T0 * (1 - fraction) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        # determine max possible transfer
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    # max over k of ∫ h(x)(1-h(x+k))dx
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_proj(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_refine(h, start_time, budget_s, rng):
    n = h.shape[0]
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    total_steps = int(min(50000, max(10000,
                     (budget_s - (time.time() - start_time)) * 2000.0)))
    T0, T_final = 0.01, 1e-6
    for step in range(total_steps):
        if time.time() - start_time >= budget_s:
            break
        frac = step / total_steps
        T = T0 * (1 - frac) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
    return best_h, best_score
