def correlation_full(a, b):
    n = a.size
    m = b.size
    M = 1 << ((n+m-1).bit_length())
    A = np.fft.rfft(a, M)
    B = np.fft.rfft(b, M)
    corr = np.fft.irfft(A * np.conj(B), M)[:n+m-1]

def gradient_descent_softmax_momentum(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000, beta_soft=10.0):

    n = h.size

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, M)
    G = np.fft.rfft(1.0 - h, M)
    corr = np.fft.irfft(H * np.conj(G), M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, M)
    G = np.fft.rfft(1.0 - h, M)
    corr = np.fft.irfft(H * np.conj(G), M)[:2 * n - 1]
    return corr * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def softmax_gradient(h, beta_soft):
    """Gradient of a soft‑max smoothed max‑overlap."""
    n = h.size
    # raw correlation (no dx scaling)
    raw_corr = compute_corr_scaled_fft(h) / (2.0 / n)
    max_c = raw_corr.max()
    w = np.exp(beta_soft * (raw_corr - max_c))
    sum_w = w.sum()
    if sum_w == 0:
        return np.zeros_like(h)
    # correlation of w with h
    L = w.size
    M = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, M)
    H = np.fft.rfft(h, M)
    corr1 = np.fft.irfft(W * np.conj(H), M)[:L + n - 1]
    offset = L - 1
    conv1 = corr1[offset:offset + n]
    # correlation of reversed w with h
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, M)
    corr2 = np.fft.irfft(W_rev * np.conj(H), M)[:L + n - 1]
    conv2 = corr2[offset:offset + n]
    grad = (sum_w - conv1 - conv2) / sum_w
    return grad

def gradient_descent_softmax_momentum(h0, time_left_func, step0=0.5,
                                     mom=0.9, max_iter=200000, beta_soft=10.0):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        grad = softmax_gradient(h, beta_soft)
        momentum = mom * momentum + (1.0 - mom) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func,
                      max_iters=300000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    best_h = None
    best_score = np.inf

    # Resolutions
    n_small, n_mid, n_large = 64, 128, 256

    # ----- Phase 1: small resolution candidates -----
    cand_small = []
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.size != n_small:
            init_arr = downsample(init_arr, n_small)
        cand_small.append(project_onto_box_sum(init_arr, n_small / 2.0))
    cand_small.append(thue_morse(n_small))
    for _ in range(3):
        cand_small.append(random_binary_balanced(n_small, rng))
    for _ in range(3):
        v = rng.random(n_small)
        cand_small.append(project_onto_box_sum(v, n_small / 2.0))

    for cand in cand_small:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = gradient_descent_softmax_momentum(
            cand, time_left, step0=0.5, mom=0.9,
            max_iter=20000, beta_soft=15.0)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 2: medium resolution -----
    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(n_mid), n_mid / 2.0)
        best_score = compute_c5_fft(best_h)
    else:
        best_h = upsample(best_h, n_mid)
        best_h = project_onto_box_sum(best_h, n_mid / 2.0)
        best_score = compute_c5_fft(best_h)

    if time_left() > 5.0:
        h_opt, sc_opt = gradient_descent_softmax_momentum(
            best_h, time_left, step0=0.5, mom=0.9,
            max_iter=30000, beta_soft=20.0)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 3: large resolution -----
    best_h = upsample(best_h, n_large)
    best_h = project_onto_box_sum(best_h, n_large / 2.0)
    best_score = compute_c5_fft(best_h)

    if time_left() > 5.0:
        h_opt, sc_opt = gradient_descent_softmax_momentum(
            best_h, time_left, step0=0.5, mom=0.9,
            max_iter=40000, beta_soft=30.0)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 4: binary threshold -----
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5_fft(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # ----- Phase 5: binary annealing -----
    if time_left() > 2.0:
        h_bin_ref, sc_bin_ref = binary_swap_anneal(
            best_h, rng, time_left,
            max_iters=250000, T0=0.3, decay=0.9999, Tmin=1e-6)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 6: final continuous refinement -----
    if time_left() > 0.5:
        h_ref, sc_ref = gradient_descent_softmax_momentum(
            best_h, time_left, step0=0.2, mom=0.9,
            max_iter=20000, beta_soft=30.0)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 7: random restarts -----
    while time_left() > 5.0:
        v = rng.random(n_large)
        v = project_onto_box_sum(v, n_large / 2.0)
        h_tmp, sc_tmp = gradient_descent_softmax_momentum(
            v, time_left, step0=0.5, mom=0.9,
            max_iter=10000, beta_soft=20.0)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
            if best_score <= target:
                break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
