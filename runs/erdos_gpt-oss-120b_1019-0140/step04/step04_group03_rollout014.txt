def compute_corr_array(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    conv = np.fft.irfft(H * G, L)
    return conv[:2*n-1]

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

def simulated_annealing_refine(h, cur_score, time_limit, rng):
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_score = cur_score
    h_cur = h.copy()
    cur_score = cur_score
    # schedule: compute total steps based on time_limit
    total_steps = int(max(1000, time_limit * 2000)) # step count
    for step in range(total_steps):
        if time.time() - start > time_limit:
            break
        # compute grad for active shifts
        shifts, max_corr = active_shifts(h_cur)
        grad = compute_gradients_for_shifts(h_cur, shifts)
        # donors indices where grad > 0 (decrease them)
        donors = np.where((grad > 1e-8) & (h_cur > 1e-8))[0]
        receivers = np.where((grad < -1e-8) & (h_cur < 1.0 - 1e-8))[0]
        if donors.size == 0 or receivers.size == 0:
            # fallback random
            i, j = rng.choice(n, size=2, replace=False)
        else:
            i = rng.choice(donors)
            j = rng.choice(receivers)
        max_move = min(h_cur[i], 1.0 - h_cur[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        h_new = h_cur.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # projection not needed because we respect bounds exactly.
        new_score = compute_c5(h_new)
        # temperature schedule: linear cooling
        T0 = 0.01
        T_final = 1e-6
        T = T0 * (1 - step / total_steps) + T_final
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h_cur = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
    return best_h, best_score
```

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        # compute active shifts
        shifts, max_corr = active_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random noise
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project to box and sum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_array(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    conv = np.fft.irfft(H * G, L)
    return conv[: 2 * n - 1]

def compute_c5(h):
    n = h.shape[0]
    corr = compute_corr_array(h)
    return np.max(corr) * (2.0 / n)

def active_shifts(h, eps=1e-12):
    corr = compute_corr_array(h)
    max_corr = np.max(corr)
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (h.shape[0] - 1)
    return shifts, max_corr

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        shifts, _ = active_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_refine(h, cur_score, time_limit, rng):
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_score = cur_score
    h_cur = h.copy()
    cur_score = cur_score
    total_steps = int(max(2000, time_limit * 2000))
    T0 = 0.01
    T_final = 1e-6
    for step in range(total_steps):
        if time.time() - start > time_limit:
            break
        shifts, _ = active_shifts(h_cur)
        grad = compute_gradients_for_shifts(h_cur, shifts)
        donors = np.where((grad > 1e-8) & (h_cur > 1e-8))[0]
        receivers = np.where((grad < -1e-8) & (h_cur < 1.0 - 1e-8))[0]
        if donors.size and receivers.size:
            i = rng.choice(donors)
            j = rng.choice(receivers)
        else:
            i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h_cur[i], 1.0 - h_cur[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h_cur.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        T = T0 * (1 - step / total_steps) + T_final
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h_cur = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
    return best_h, best_score
