def gradient_descent(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h0.size
    s = n/2
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    iter = 0
    while iter < max_iter and time_left() > 0.1:
         iter += 1
         corr = compute_corr_full_fft(h)
         max_c = np.max(corr)
         grad = compute_grad_max_shifts(h, corr, max_c)
         momentum = beta * momentum + (1-beta) * grad
         eta = step0 / np.sqrt(iter)
         h_new = h - eta * momentum
         # occasional noise
         if iter % 3000 == 0:
             h_new += rng.normal(scale=0.02, size=n)
         h_new = project_onto_box_sum(h_new, s)
         score = compute_c5_fft(h_new)
         if score < best_score:
             best_score = score
             best_h = h_new.copy()
         h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128).astype(float)  # define thue_morse function

    # ensure base sum is n/2: project
    base = project_onto_box_sum(base, base.shape[0]/2.0)

    # define scales; we start with 256,512,1024
    scales = [256, 512, 1024]
    best_h = None
    best_score = np.inf
    TARGET = 0.3808

    for n in scales:
        if time_left() < 2.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
            base = project_onto_box_sum(base, n/2.0)
        # Greedy restarts
        # Determine number of restarts based on n and remaining time
        restarts = max(1, int( (time_left() // 30) ))  # ~30 seconds per restart?
        # but also cap
        restarts = min(restarts, 30)
        for r in range(restarts):
            if time_left() < 5.0:
                break
            # perturb
            h0 = base + rng.normal(scale=0.07, size=n)
            h0 = project_onto_box_sum(h0, n/2.0)
            # gradient descent iterations: assign based on n
            max_iter = int(5e5) if n <= 512 else int(6e5)
            # adjust maybe lower for speed if time left small
            # allocate proportionally: maybe allocate half time to each scale
            # but we use time_left check inside.
            h_opt, sc_opt = gradient_descent(
                h0, max_iter=max_iter, step0=0.6, beta=0.9,
                rng=rng, time_left=time_left
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_score <= TARGET:
            break

    # binary rounding and local search
    if best_h is not None and time_left() > 10.0:
        n = best_h.size
        # round to binary: set top n/2 entries to 1
        idx = np.argpartition(best_h, -n//2)[-n//2:]
        binary_h = np.zeros(n, dtype=float)
        binary_h[idx] = 1.0
        binary_score = compute_c5_fft(binary_h)
        if binary_score < best_score:
            best_score, best_h = binary_score, binary_h.copy()
        # hill climbing swaps
        ones = set(np.where(binary_h == 1)[0].tolist())
        zeros = set(np.where(binary_h == 0)[0].tolist())
        # limit swaps
        max_swaps = 5000
        for s in range(max_swaps):
            if time_left() < 2.0:
                break
            i = rng.choice(list(ones))
            j = rng.choice(list(zeros))
            # swap
            binary_h[i] = 0.0
            binary_h[j] = 1.0
            new_score = compute_c5_fft(binary_h)
            if new_score < best_score:
                best_score = new_score
                best_h = binary_h.copy()
                # update sets
                ones.remove(i); zeros.remove(j)
                ones.add(j); zeros.add(i)
            else:
                # revert
                binary_h[i] = 1.0
                binary_h[j] = 0.0
            # optional early stop if bestScore <= TARGET
            if best_score <= TARGET:
                break

    # final projection
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
        n_ret = best_h.size
    else:
        # fallback to default 512 thue-morse
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret/2.0)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # broaden interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as float."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq


def random_binary_balanced(n, rng):
    """Balanced binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_full_fft(h):
    """Full correlation of h with (1-h) using FFT (real output)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """Approximation of C5 via correlation on the discrete grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient of the max‑correlation term (average over all max shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if idxes.size > 0:
        grad /= idxes.size
    return grad


def gradient_descent(h0, max_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    it = 0
    while it < max_iter and time_left() > 0.5:
        it += 1
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 3000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
