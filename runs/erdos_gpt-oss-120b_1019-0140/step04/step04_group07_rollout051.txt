import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {x: lo <= x_i <= hi, sum x_i = s} (Euclidean)."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated on grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from size len(arr) to new_n (periodic on [0,1))."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Binary Thue–Morse sequence (0/1)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                      dtype=float, count=n)
    return seq


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of C(s) over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size:
        grad /= shifts.size
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum and adaptive step."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 2500 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)

    return best_h, best_score


def binary_round_h(h):
    """Return binary vector with exactly n/2 ones, taking largest entries."""
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    out = np.zeros_like(h)
    out[idx] = 1.0
    return out


def binary_swap_local_search(h, rng, time_left, max_no_improve=2000):
    """Random swap of a 1 and a 0, keep improvements."""
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h < 0.5)[0]
    no_improve = 0
    while time_left() > 0 and no_improve < max_no_improve:
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        score = compute_c5(h_new)
        if score < best_score:
            best_h = h_new
            best_score = score
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)

    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if time_left() <= 2:
            break
        if base.size != n:

        # Gradient descent stage

        h_gd, sc_gd = gradient_descent_adaptive(base, max_iters,
                                                step0=0.8, beta=0.9,
                                                rng=rng, time_left=time_left)
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd
        base = h_gd

        # Binary rounding & swap search
        h_bin = binary_round_h(base)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_h, best_score = h_bin.copy(), sc_bin
        if time_left() > 2:
            h_swap, sc_swap = binary_swap_local_search(h_bin, rng, time_left)
            if sc_swap < best_score:
                best_h, best_score = h_swap.copy(), sc_swap
            base = h_swap
        else:
            base = h_bin

    # Final intensive refinement on the best candidate
    if time_left() > 5:
        max_iters = int(min(80000, max(20000, time_left() * 20)))
        best_h, best_score = gradient_descent_adaptive(best_h, max_iters,
                                                       step0=0.3, beta=0.9,
                                                       rng=rng, time_left=time_left)

    # Final binary cleanup
    if time_left() > 2:
        h_bin = binary_round_h(best_h)
        sc_bin = compute_c5(h_bin)

            if sc_swap < best_score:

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
