def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000, T0=0.2, decay=0.99995, Tmin=1e-6):

    h = h0.copy()

    current_score = compute_c5_fft(h)

    best_h = h.copy()

    best_score = current_score

    T = T0

    it = 0

    while it < max_iters and time_left_func() > 0.01:

        ones = np.where(h > 0.5)[0]

        zeros = np.where(h <= 0.5)[0]

        if ones.size == 0 or zeros.size == 0: break

        i = rng.choice(ones)

        j = rng.choice(zeros)

        # swap

        h[i] = 0.0

        h[j] = 1.0

        new_score = compute_c5_fft(h)

        delta = new_score - current_score

        accept = False

        if delta < 0:

            accept = True

        else:

            if T > 0 and rng.random() < np.exp(-delta / T):

                accept = True

        if accept:

            current_score = new_score

            if new_score < best_score:

                best_score = new_score

                best_h = h.copy()

        else:

            # revert

            h[i] = 1.0

            h[j] = 0.0

        T = max(Tmin, T * decay)

        it += 1

    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    """Full correlation of h with (1-h) scaled by dx = 2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    """Maximum overlap integral C5."""
    return np.max(compute_corr_scaled_fft(h))

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_active_gradient(h, corr):
    """Gradient of max_k f_k with respect to h (averaged over active shifts)."""
    n = h.size
    max_corr = np.max(corr)
    tol = max_corr - 1e-9
    active_idx = np.where(corr >= tol)[0]
    if active_idx.size == 0:
        active_idx = np.array([np.argmax(corr)], dtype=int)
    shifts = active_idx - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            term1 = np.empty_like(h)
            term1[:n - s] = h[s:]
            term1[n - s:] = 0.0
            term2 = np.empty_like(h)
            term2[s:] = h[:n - s]
            term2[:s] = 0.0
        else:
            ss = -s
            term1 = np.empty_like(h)
            term1[ss:] = h[:n - ss]
            term1[:ss] = 0.0
            term2 = np.empty_like(h)
            term2[:n - ss] = h[ss:]
            term2[n - ss:] = 0.0
        grad += 1.0 - term1 - term2
    grad /= float(shifts.size)
    return grad

def adam_optimize(h0, time_left_func, max_iter=200000, lr=0.02,
                  beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam-like projected optimization for C5."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    while t < max_iter and time_left_func() > 0.01:
        t += 1
        corr = compute_corr_scaled_fft(h)
        grad = compute_active_gradient(h, corr)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fft(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func,
                      max_iters=800000, T0=0.5, decay=0.99995, Tmin=1e-6):
    """Simulated annealing on binary vectors (swap 1↔0)."""
    h = h0.copy()
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
