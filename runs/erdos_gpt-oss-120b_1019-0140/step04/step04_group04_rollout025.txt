def hill_climb_binary(h_init, rng, time_left, max_iters=None):
    n = h_init.shape[0]
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    # zero indices
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    iter_count = 0
    # set temperature schedule
    # Determine max_iters
    if max_iters is None:
        max_iters = int(time_left() * 2000)  # arbitrary scaling
    while iter_count < max_iters and time_left() > 0.01:
        iter_count += 1
        # choose move
        # With probability 0.8, do random swap; else targeted swap.
        if rng.random() < 0.8:
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            # targeted: choose worst shift from current correlation
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            # get shift indices
            idxes = np.where(corr == max_corr)[0]
            # pick one shift
            shift_idx = rng.choice(idxes)
            s = shift_idx - (n - 1)  # integer shift
            if s >= 0:
                # overlapping pairs where both ones
                overlap_mask = (h[:n - s] == 1) & (h[s:] == 1)
                overlap_idx = np.where(overlap_mask)[0]
                if overlap_idx.size == 0:
                    # fallback random swap
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                else:
                    i = overlap_idx[rng.integers(overlap_idx.size)]
                    # choose a zero somewhere not overlapping maybe?
                    # Simple: choose any zero
                    j = rng.choice(zeros)
            else:
                s_abs = -s
                overlap_mask = (h[s_abs:] == 1) & (h[:n - s_abs] == 1)
                overlap_idx = np.where(overlap_mask)[0] + s_abs  # i index where overlap occurs at i, i-s
                if overlap_idx.size == 0:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                else:
                    i = overlap_idx[rng.integers(overlap_idx.size)]
                    j = rng.choice(zeros)
        # perform swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        # compute new score
        new_score = compute_c5(h_new)
        # accept if better or with probability
        delta = new_score - current_score
        # Temperature schedule: linear from T0 to T_min
        # Let's set T0 = best_score * 0.1
        # T_min = 1e-5
        # T = T0 * (1 - iter_count / max_iters) + T_min
        T0 = best_score * 0.1
        T_min = 1e-6
        T = T0 * max(0.0, (max_iters - iter_count) / max_iters) + T_min
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            current_score = new_score
            # update zeros/ones
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        # else reject, keep h unchanged
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise to escape local minima
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climb_binary(h0, rng, time_left, max_iters=None):
    n = h0.shape[0]
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    if max_iters is None:
        max_iters = int(min(time_left() * 2000, 200000))
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        # decide move type
        if rng.random() < 0.8:
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            # targeted: compute correlation and find worst shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = np.max(corr)
            idxes = np.where(corr == max_corr)[0]
            shift_idx = rng.choice(idxes)
            s = shift_idx - (n - 1)
            if s >= 0:
                mask = (h[:n - s] == 1) & (h[s:] == 1)
                overlap = np.where(mask)[0]
                if overlap.size == 0:
                    i = rng.choice(ones)
                else:
                    i = overlap[rng.integers(overlap.size)]
            else:
                s_abs = -s
                mask = (h[s_abs:] == 1) & (h[:n - s_abs] == 1)
                overlap = np.where(mask)[0] + s_abs
                if overlap.size == 0:
                    i = rng.choice(ones)
                else:
                    i = overlap[rng.integers(overlap.size)]
            # pick any zero j
            j = rng.choice(zeros)
        # swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        # temperature schedule
        # linear cooling: T0 = best_score * 0.1, Tmin = 1e-6
        T0 = best_score * 0.1
        Tmin = 1e-6
        T = T0 * max(0.0, (max_iters - it) / max_iters) + Tmin
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            current_score = new_score
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        # early exit if best_score already below target
        if best_score <= 0.3808:
            break
    return best_h, best_score

def thue_morse_bits(n):
    # generate binary Thue-Morse sequence of length n
    # works for any n, but for n power of 2, it's balanced.
    # using vectorized popcount via binary string conversion could be slow; maybe use np.unpackbits.
    # Simpler loop.
    arr = np.empty(n, dtype=float)
    for i in range(n):
        # parity of bits
        arr[i] = (bin(i).count('1') & 1)
    return arr

            max_corr = np.max(corr)
            idxes = np.where(corr == max_corr)[0]
            shift_idx = rng.choice(idxes)
            s = shift_idx - (n - 1)
            if s >= 0:
                mask = (h[:n - s] == 1) & (h[s:] == 1)
                overlap = np.where(mask)[0]
                if overlap.size == 0:
                    i = rng.choice(ones)
                else:
                    i = overlap[rng.integers(overlap.size)]
            else:
                s_abs = -s
                mask = (h[s_abs:] == 1) & (h[:n - s_abs] == 1)
                overlap = np.where(mask)[0] + s_abs
                if overlap.size == 0:
                    i = rng.choice(ones)
                else:
                    i = overlap[rng.integers(overlap.size)]
            j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        T0 = best_score * 0.1
        Tmin = 1e-6
        T = T0 * max(0.0, (max_iters - it) / max_iters) + Tmin
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            current_score = new_score
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        if best_score <= 0.3808:
            break
    return best_h, best_score

def thue_morse_bits(n):
    arr = np.empty(n, dtype=float)
    for i in range(n):
        arr[i] = (bin(i).count('1') & 1)
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    best_h = None
    best_score = np.inf

    if time_left() > 800:
        n = 512
    elif time_left() > 400:
        n = 256
    elif time_left() > 150:
        n = 128
    else:
        n = 64

    h0 = thue_morse_bits(n)
    init_score = compute_c5(h0)
    if init_score < best_score:
        best_h, best_score = h0.copy(), init_score

    while time_left() > 8:
        if rng.random() < 0.3:
            h_start = thue_morse_bits(n)
        else:
            h_start = np.zeros(n, dtype=float)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_start[idx] = 1.0

        max_iters = int(min(time_left() * 2000, 100000))
        h_local, sc_local = hill_climb_binary(h_start, rng, time_left, max_iters=max_iters)
        if sc_local < best_score:
            best_h, best_score = h_local.copy(), sc_local

        if time_left() > 5:
            gd_iters = int(min(20000, time_left() * 30))
            h_ref, sc_ref = gradient_descent_momentum_custom(best_h, n_iter=gd_iters,
                                                             step0=0.2, beta=0.9,
                                                             rng=rng, time_left=time_left)
            if sc_ref < best_score:
                best_h, best_score = h_ref.copy(), sc_ref

        if best_score <= 0.3808:
            break

    if time_left() > 5:
        final_iters = int(min(50000, time_left() * 30))
        best_h, best_score = gradient_descent_momentum_custom(best_h, n_iter=final_iters,
                                                             step0=0.1, beta=0.9,
                                                             rng=rng, time_left=time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
