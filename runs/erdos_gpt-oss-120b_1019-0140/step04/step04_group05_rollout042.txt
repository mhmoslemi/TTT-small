def subgradient_descent(h, time_limit):
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(best_h)
    start = time.time()
    iteration = 0
    # initial step:
    init_alpha = 0.5  # try moderate
    while time.time() - start < time_limit:
        # compute correlation and find worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute shift arrays
        if k >= 0:
            shift_plus = np.empty_like(h)
            shift_plus[:n - k] = h[k:]
            if k > 0:
                shift_plus[n - k:] = 0.0
            shift_minus = np.empty_like(h)
            shift_minus[k:] = h[:n - k]
            if k > 0:
                shift_minus[:k] = 0.0
        else:
            kk = -k
            shift_plus = np.empty_like(h)
            shift_plus[kk:] = h[:n - kk]
            shift_plus[:kk] = 0.0
            shift_minus = np.empty_like(h)
            shift_minus[:n - kk] = h[kk:]
            shift_minus[n - kk:] = 0.0
        grad = dx * (1.0 - (shift_plus + shift_minus))
        # step size
        alpha = init_alpha / np.sqrt(iteration + 1)
        # update
        h_new = h - alpha * grad
        # project onto feasible set
        h_new = project_onto_box_sum(h_new, n/2.0)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        iteration += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # Determine base vector from initial_h_values if exists
    if 'initial_h_values' in globals():
        base_initial = np.asarray(initial_h_values, dtype=float)
    else:
        base_initial = random_balanced_binary(64, rng)  # fallback

    best_h = None
    best_score = np.inf

    # parameters
    n_list = [128, 256, 512, 1024]
    fractions = [0.12, 0.2, 0.4, 0.28]  # sum ~1
    for idx, n in enumerate(n_list):
        if time_left() <= 0.0:
            break
        alloc = fractions[idx] * budget_s
        if alloc > time_left():
            alloc = time_left()
        # prepare base h for this resolution
        if base_initial.shape[0] != n:
            base = upsample(base_initial, n)
        else:
            base = base_initial.copy()
        base = project_onto_box_sum(base, n/2.0)
        # run subgradient descent
        h_opt, sc_opt = subgradient_descent(base, alloc, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        # update base_initial for next resolution (upsampled from best)
        base_initial = best_h.copy()
    # final refinement with SA if time remains
    while time_left() > 2.0:
        # allocate small chunk
        chunk = min(5.0, time_left())
        h_ref, sc_ref = simulated_annealing(best_h, chunk, rng)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()
        # break if good enough
        if best_score <= 0.3808:
            break

            shift_plus[:n - k] = h[k:]
            if k > 0:

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    else:
        lam_mid = (lam_low + lam_high) * 0.5
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def subgradient_descent(h0, time_limit, rng):
    """Projected subgradient descent on max_k C(k)."""
    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    total = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    it = 0
    init_alpha = 2.0
    while time.time() - start < time_limit:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        # compute shifted copies (zero outside)
        if k >= 0:
            shift_plus = np.empty_like(h)
            shift_plus[:n - k] = h[k:]
            if k > 0:
                shift_plus[n - k:] = 0.0
            shift_minus = np.empty_like(h)
            shift_minus[k:] = h[:n - k]
            if k > 0:
                shift_minus[:k] = 0.0
        else:
            kk = -k
            shift_plus = np.empty_like(h)
            shift_plus[kk:] = h[:n - kk]
            if kk > 0:
                shift_plus[:kk] = 0.0
            shift_minus = np.empty_like(h)
            shift_minus[:n - kk] = h[kk:]
            if kk > 0:
                shift_minus[n - kk:] = 0.0

        grad = dx * (1.0 - (shift_plus + shift_minus))
        alpha = init_alpha / np.sqrt(it + 1)
        h_new = h - alpha * grad
        h_new = project_onto_box_sum(h_new, total)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1

        # occasional jitter to escape flat regions
        if it % 2000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            h = np.clip(h + noise, 0.0, 1.0)
            h = project_onto_box_sum(h, total)
    return best_h, best_score


def simulated_annealing(h0, time_limit, rng):
    """Simple SA with mass‑transfer moves, respecting sum and box constraints."""
    h = h0.copy()
    n = h.size
    total = n / 2.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    max_iters = int(5e6)
    init_temp = 0.03
    for it in range(max_iters):
        if time.time() - start >= time_limit:
            break
        temp = init_temp * (1.0 - it / max_iters)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        if new_score < cur_score or (temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta

        # occasional binary perturbation
        if it % 5000 == 0:
            h_bin = random_balanced_binary(n, rng)
            sc_bin = compute_c5(h_bin)
            if sc_bin < best_score:
                best_score = sc_bin
                best_h = h_bin.copy()
                cur_score = sc_bin
                h = h_bin.copy()
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # start from provided construction if any
    if 'initial_h_values' in globals():
        base_initial = np.asarray(initial_h_values, dtype=float)
    else:
        base_initial = random_balanced_binary(64, rng)

    best_h = None
    best_score = np.inf

    # resolution schedule
    n_list = [128, 256, 512, 1024]
    fractions = [0.12, 0.20, 0.40, 0.28]

    for idx, n in enumerate(n_list):
        if time_left() <= 0.0:
            break
        alloc = fractions[idx] * budget_s
        if alloc > time_left():
            alloc = time_left()
        # prepare base vector at this resolution
        if base_initial.shape[0] != n:
            base = upsample(base_initial, n)
        else:
            base = base_initial.copy()
        base = project_onto_box_sum(base, n / 2.0)

        # subgradient descent
        h_opt, sc_opt = subgradient_descent(base, alloc, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        base_initial = best_h.copy()  # warm‑start next resolution

    # final refinement with SA if time permits
    while time_left() > 2.0:
        chunk = min(5.0, time_left())
        h_ref, sc_ref = simulated_annealing(best_h, chunk, rng)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()
        if best_score <= 0.3808:
            break

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
