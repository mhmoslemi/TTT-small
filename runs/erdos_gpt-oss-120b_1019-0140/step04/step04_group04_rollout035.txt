def smoothed_max_c5_and_grad(h, alpha, penalty_lambda):
    n = len(h)
    sum_h = h.sum()
    O = np.correlate(h, h, mode='full')
    C = sum_h - O   # length 2n-1
    # smoothed max:
    logsum = scipy.special.logsumexp(alpha * C)
    smoothed = logsum / alpha
    # weights:
    w = np.exp(alpha * C - logsum)   # sum(w)=1
    # compute conv of w and h:
    conv = np.convolve(w, h, mode='full')
    conv_center = conv[n-1:n-1+n]  # shape n
    # reverse w:
    w_rev = w[::-1]
    conv_rev = np.convolve(w_rev, h, mode='full')
    conv_rev_center = conv_rev[n-1:n-1+n]
    grad = np.ones(n) - (conv_center + conv_rev_center)
    # penalty term:
    grad += 2 * penalty_lambda * (sum_h - n/2)
    # objective to minimize: smoothed (which approximates max C) + penalty? but we can directly minimize smoothed + penalty*...?
    obj = smoothed + penalty_lambda * (sum_h - n/2)**2

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0., 1., m, endpoint=False)
    x_new = np.linspace(0., 1., new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

def subgradient_max_corr(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # find shift(s) achieving max; take first
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.ones(n)
    # shift right contribution: indices i where i+shift valid
    if shift >= 0:
        grad[:n-shift] -= h[shift:]  # h_{i+shift} for i in 0..n-shift-1
        grad[shift:] -= h[:n-shift]  # h_{i-shift} for i in shift..n-1 (i-shift >=0)
    else:
        s = -shift
        grad[s:] -= h[:n-s]  # h_{i+shift} for i>=s
        grad[:n-s] -= h[s:]  # h_{i-shift} for i < n-s
    return grad
```

def simulated_annealing(h0, time_left, best_score):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_sc = best_score
    current_sc = best_sc
    # initial temperature as fraction of best_sc (say 0.01)
    T = best_sc * 0.02 + 1e-4
    while time_left() > 2:
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * 0.5
        if delta == 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        d = sc_new - current_sc
        if d < 0 or np.exp(-d / T) > np.random.rand():
            h = h_new
            current_sc = sc_new
            if sc_new < best_sc:
                best_sc, best_h = sc_new, h_new.copy()
        # anneal temperature
        T *= 0.9995
        # occasionally perturb globally
        if np.random.rand() < 0.02:
            pert = np.random.normal(scale=0.01, size=n)
            h_pert = np.clip(h + pert, 0.0, 1.0)
            h_pert = project_onto_box_sum(h_pert, n/2)
            sc_pert = compute_c5(h_pert)
            if sc_pert < best_sc:
                best_sc, best_h = sc_pert, h_pert.copy()
                h = h_pert
                current_sc = sc_pert
        # break condition if not improving and temperature low
        if T < 1e-6:
            break
    return best_h, best_sc

def upsample(arr, new_n):

def subgradient_max_corr(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.ones(n)
    if shift >= 0:
        if shift < n:
            grad[:n-shift] -= h[shift:]
            grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        if s < n:
            grad[s:] -= h[:n-s]
            grad[:n-s] -= h[s:]
    return grad
```

def smoothed_obj_and_grad(h, alpha):
    n = h.shape[0]
    sum_h = h.sum()
    O = np.correlate(h, h, mode='full')
    C = sum_h - O
    logsum = sps.logsumexp(alpha * C)
    smoothed = logsum / alpha
    w = np.exp(alpha * C - logsum)  # sums to 1
    conv = np.convolve(w, h, mode='full')
    conv_center = conv[n-1:n-1+n]
    w_rev = w[::-1]
    conv_rev = np.convolve(w_rev, h, mode='full')
    conv_rev_center = conv_rev[n-1:n-1+n]
    grad = np.ones(n) - (conv_center + conv_rev_center)
    return smoothed, grad
```

            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def project_onto_box_sum_simple(v, s):
    """Fast approximate projection onto [0,1]^n with sum(v)=s."""
    n = v.size
    v = np.clip(v, 0.0, 1.0)
    delta = v.sum() - s
    if abs(delta) > 1e-12:
        v = v - delta / n
        v = np.clip(v, 0.0, 1.0)
    return v

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def subgradient_max_corr(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.ones(n)
    if shift >= 0:
        if shift < n:
            grad[:n - shift] -= h[shift:]
            grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            grad[s:] -= h[:n - s]
            grad[:n - s] -= h[s:]
    return grad

def smoothed_obj_and_grad(h, alpha):
    n = h.shape[0]
    sum_h = h.sum()
    O = np.correlate(h, h, mode='full')
    C = sum_h - O
    logsum = sps.logsumexp(alpha * C)
    smoothed = logsum / alpha
    w = np.exp(alpha * C - logsum)  # sums to 1
    conv = np.convolve(w, h, mode='full')
    conv_center = conv[n - 1:n - 1 + n]
    w_rev = w[::-1]
    conv_rev = np.convolve(w_rev, h, mode='full')
    conv_rev_center = conv_rev[n - 1:n - 1 + n]
    grad = np.ones(n) - (conv_center + conv_rev_center)
    return smoothed, grad

def adam_subgradient_optimize(h0, n_iter, lr, beta1, beta2, eps, time_left, best_score):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_sc = best_score
    s_target = h.size / 2.0
    for t in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        grad = subgradient_max_corr(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum_simple(h, s_target)
        if t % 500 == 0 or t == n_iter:
            sc = compute_c5(h)
            if sc < best_sc:
                best_sc, best_h = sc, h.copy()
    return best_h, best_sc

def adam_optimize(h0, n_iter, lr, beta1, beta2, eps, alpha, time_left, best_score):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_sc = best_score
    s_target = h.size / 2.0
    for t in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        obj, grad = smoothed_obj_and_grad(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum_simple(h, s_target)
        if t % 200 == 0 or t == n_iter:
            sc = compute_c5(h)
            if sc < best_sc:
                best_sc, best_h = sc, h.copy()
    return best_h, best_sc

def simulated_annealing(h0, time_left, best_score, rng):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_sc = best_score
    current_sc = best_sc
    T = best_sc * 0.02 + 1e-4
    while time_left() > 2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        if delta == 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        d = sc_new - current_sc
        if d < 0 or np.exp(-d / max(T, 1e-12)) > rng.random():
            h = h_new
            current_sc = sc_new
            if sc_new < best_sc:
                best_sc, best_h = sc_new, h_new.copy()
        T *= 0.9995
        if rng.random() < 0.02:
            pert = rng.normal(scale=0.01, size=n)
            h_pert = np.clip(h + pert, 0.0, 1.0)
            h_pert = project_onto_box_sum_simple(h_pert, n / 2.0)
            sc_pert = compute_c5(h_pert)
            if sc_pert < best_sc:
                best_sc, best_h = sc_pert, h_pert.copy()
                h = h_pert
                current_sc = sc_pert
        if T < 1e-6:
            break
    return best_h, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 64
        base = np.fromiter((bin(i).count('1') % 2 for i in range(n0)),
                           dtype=float, count=n0)

    best_h = None
    best_score = np.inf

    for n in (64, 128, 256, 512):
        if time_left() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum_simple(base, n / 2.0)
        sc = compute_c5(base)
        if sc < best_score:
            best_h, best_score = base.copy(), sc

        remaining = time_left()
        max_iter = int(min(20000, remaining * 25))
        if max_iter <= 0:
            continue
        lr = 0.2 if n <= 128 else (0.1 if n == 256 else 0.05)
        h_opt, sc_opt = adam_subgradient_optimize(
            base, n_iter=max_iter, lr=lr, beta1=0.9, beta2=0.999,
            eps=1e-8, time_left=time_left, best_score=best_score)
        if sc_opt < best_score:
            best_h, best_score = h_opt.copy(), sc_opt
        base = h_opt

    if best_h is not None and time_left() > 10:
        alphas = [0.5, 2.0, 5.0, 10.0, 20.0]
        for alpha in alphas:
            if time_left() <= 2:
                break
            iters = int(min(2000, time_left() * 20))
            if iters <= 0:
                continue
            best_h, best_score = adam_optimize(
                best_h, n_iter=iters, lr=0.02, beta1=0.9, beta2=0.999,
                eps=1e-8, alpha=alpha, time_left=time_left,
                best_score=best_score)

    if best_h is not None and time_left() > 5:
        best_h, best_score = simulated_annealing(
            best_h, time_left, best_score, rng)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
