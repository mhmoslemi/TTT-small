import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation to a larger grid."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def simulated_annealing(h0, time_budget, rng):
    """SA that respects box and sum constraints."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # Estimate a reasonable initial temperature
    init_temp = 0.02
    n = h.size

    while time.time() - start < time_budget:
        # linear cooling based on remaining fraction
        remain = max(0.0, time_budget - (time.time() - start))
        temp = init_temp * (remain / time_budget) ** 0.5

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue

        # scale delta by temperature (larger moves early)
        delta = max_delta * rng.random() * (temp / init_temp)
        if delta <= 0.0:
            continue

        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            if temp > 0.0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_score = new_score
                else:
                    h[i] += delta
                    h[j] -= delta
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_balanced_binary(64, rng)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # schedule of grid sizes and time fractions
    fractions = {64: 0.05, 128: 0.10, 256: 0.20, 512: 0.35, 1024: 0.30}
    sizes = [64, 128, 256, 512, 1024]

    total_budget = budget_s
    for n in sizes:
        if time_left() <= 0.0:
            break
        # bring best_h to current size
        if best_h.size != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

        # allocate dedicated budget for this resolution
        alloc = total_budget * fractions.get(n, 0.0)
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        # run several SA restarts within allocated time
        start_scale = time.time()
        while time.time() - start_scale < alloc:
            remaining = alloc - (time.time() - start_scale)
            run_time = min(5.0, remaining)

            cand_h, cand_score = simulated_annealing(best_h, run_time, rng)

                rand_score = compute_c5(rand_h)
                if rand_score < best_score:

            h_new[j] += delta

                best_h = h_new.copy()
        if best_score <= 0.3808:

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
