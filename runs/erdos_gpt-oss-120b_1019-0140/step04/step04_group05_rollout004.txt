def local_hill_climb(h, time_budget, rng):
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    n = len(h)
    start = time.time()
    while time.time() - start < time_budget:
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i: j += 1
        max_delta = min(h[i], 1-h[j])

        delta = max_delta * rng.random() * 0.2  # small step
        h[i] -= delta
        h[j] += delta
        new = compute_c5(h)
        if new < cur:
            cur = new
            if new < best:

        # Periodically re-project
        if rng.integers(1000) == 0:
            h = project_onto_box_sum(h, n/2)
            cur = compute_c5(h)
    return best_h, best

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to size new_n, preserving domain [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.arange(m) / m
    x_new = np.arange(new_n) / new_n
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def random_feasible(n, rng):
    """Random vector in [0,1] with exact sum n/2."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def simulated_annealing(h0, time_limit, rng):
    """SA preserving sum, box constraints. Returns best vector and its C5."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    n = h.size
    start = time.time()
    start_temp = 0.02
    end_temp = 0.0001
    it = 0
    # main loop
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        # exponential temperature decay
        t_frac = elapsed / time_limit
        temp = start_temp * (end_temp / start_temp) ** t_frac

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta > 0.0:
            # delta scales with temperature
            delta = max_delta * rng.random() * (temp / start_temp) * 0.5
            if delta > 0.0:
                h[i] -= delta
                h[j] += delta
                new_score = compute_c5(h)
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h.copy()
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta

        # occasional swap move
        if rng.random() < 0.01:
            a = rng.integers(n)
            b = rng.integers(n)
            if a != b:
                h[a], h[b] = h[b], h[a]
                new_score = compute_c5(h)
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h.copy()
                else:
                    h[a], h[b] = h[b], h[a]

        # periodic projection to clean numeric drift
        if it % 500 == 0:
            h = project_onto_box_sum(h, n / 2.0)
            cur_score = compute_c5(h)

        it += 1
        if best_score <= 0.3808:
            break

    return best_h, best_score


def local_hill_climb(h, time_limit, rng):
    """Deterministic hill climbing with small transfers; returns improved vector."""
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    n = h.size
    start = time.time()
    while time.time() - start < time_limit:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.2
        if delta <= 0.0:
            continue
        h[i] -= delta
        h[j] += delta
        new = compute_c5(h)
        if new < cur:
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta

        if rng.integers(1000) == 0:
            h = project_onto_box_sum(h, n / 2.0)
            cur = compute_c5(h)

    return best_h, best


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    target_n = 512

    # obtain initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != target_n:
            base = upsample(base, target_n)
    else:
        base = thue_morse(target_n).astype(float)

    base = project_onto_box_sum(base, target_n / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # build a small set of diverse starters
    starters = [best_h]
    starters.append(random_balanced_binary(target_n, rng))
    starters.append(random_feasible(target_n, rng))
    for _ in range(2):
        noisy = best_h + rng.normal(scale=0.1, size=target_n)
        starters.append(project_onto_box_sum(noisy, target_n / 2.0))

    # main SA phase
    while time_left() > 30.0:
        # allocate slice (keep ~30s for final refinement)
        slice_time = min(20.0, time_left() - 30.0)
        if slice_time <= 0.0:
            break
        init = starters[rng.integers(len(starters))]
        cand_h, cand_score = simulated_annealing(init, slice_time, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        if best_score <= 0.3808:
            break

    # final local refinement with remaining budget
    remaining = time_left()
    if remaining > 0.5:
        best_h, best_score = local_hill_climb(best_h, remaining, rng)

    # final projection and exact scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
