                diff = A - g
                max_diff = np.max(diff)
                # worst shifts (max diff)
                ks = np.where(np.abs(diff - max_diff) < 1e-12)[0] - (n - 1)
                grad = np.zeros_like(h)
                for k in ks:
                    if k >= 0:
                        i_slice = slice(0, n - k)
                        j_slice = slice(k, n)
                        hi = h[i_slice]
                        hj = h[j_slice]
                        grad[i_slice] += 1.0 - hj
                        grad[j_slice] += -hi
                    else:
                        kk = -k
                        i_slice = slice(kk, n)
                        j_slice = slice(0, n - kk)
                        hi = h[i_slice]
                        hj = h[j_slice]
                        grad[i_slice] += 1.0 - hj
                        grad[j_slice] += -hi
                momentum = beta * momentum + (1.0 - beta) * grad
                eta = step0 / np.sqrt(it)
                h_new = h - eta * momentum
                h_new = project_onto_box_sum(h_new, n / 2.0)
                cur_c5 = compute_c5(h_new)
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h_new.copy()
                h = h_new
            base = h
        # simulated annealing refinement at current scale
        if best_h is not None and best_h.size == n and time_left() > 5.0:
            h_cur = best_h.copy()
            cur_c5 = best_c5
            total_steps = int(min(30000, max(5000, time_left() * 500.0)))
            T0 = 0.02
            T_final = 1e-6
            for step in range(total_steps):
                if time_left() <= 0.0:
                    break
                T = T0 * (1 - step / total_steps) + T_final
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(h_cur[i], 1.0 - h_cur[j])
                if max_move <= 0:
                    continue
                delta = max_move * rng.random()
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_c5 = compute_c5(h_new)
                if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                    h_cur = h_new
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5, best_h = cur_c5, h_cur.copy()
            base = upsample(h_cur, scales[idx + 1]) if idx + 1 < len(scales) else h_cur
    # final gradient descent at the finest scale
    if best_h is not None and time_left() > 5.0:
        n = best_h.size
        h = best_h.copy()
        momentum = np.zeros(n, dtype=float)
        beta = 0.9
        step0 = 0.2
        max_iters = int(min(200000, max(50000, time_left() * 50000.0 / n)))
        for it in range(1, max_iters + 1):
            if it % 5000 == 0 and time_left() <= 5.0:
                break
            A, g = compute_corrs_fft(h)
            diff = A - g
            max_diff = np.max(diff)
            ks = np.where(np.abs(diff - max_diff) < 1e-12)[0] - (n - 1)
            grad = np.zeros_like(h)
            for k in ks:
                if k >= 0:
                    i_slice = slice(0, n - k)
                    j_slice = slice(k, n)
                    hi = h[i_slice]
                    hj = h[j_slice]
                    grad[i_slice] += 1.0 - hj
                    grad[j_slice] += -hi
                else:
                    kk = -k
                    i_slice = slice(kk, n)
                    j_slice = slice(0, n - kk)
                    hi = h[i_slice]
                    hj = h[j_slice]
                    grad[i_slice] += 1.0 - hj
                    grad[j_slice] += -hi
            momentum = beta * momentum + (1.0 - beta) * grad
            eta = step0 / np.sqrt(it)
            h_new = h - eta * momentum
            h_new = project_onto_box_sum(h_new, n / 2.0)
            cur_c5 = compute_c5(h_new)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h_new.copy()
            h = h_new
    # fallback if nothing found
    if best_h is None:
        n = n_final
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_h.size
