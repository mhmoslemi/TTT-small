def gradient_descent_softmax(h0, max_iters, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    # Momentum
    mom = np.zeros_like(h)
    beta = 0.9
    step = 0.8
    # softmax alpha schedule
    alpha = 5.0
    alpha_inc = 0.02
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        # compute gradient
        grad = softmax_gradient(h, alpha)
        # momentum
        mom = beta * mom + (1-beta) * grad
        # update
        h_new = h - step * mom
        # projection
        h_new = project_onto_box_sum(h_new, s)
        # compute score
        score = compute_c5(h_new)
        # accept if improvement
        if score < best_score:
            best_h, best_score = h_new.copy(), score
        # acceptance criteria: if score <= current score, accept; else maybe accept with probability? But keep simple.
        # We'll maintain current h as the one we just moved (even if not improving) because momentum may need path.
        # But if score is worse, we may reduce step.
        if score <= compute_c5(h):
            h = h_new
            # increase step a bit

def softmax_gradient(h, alpha):
    n = h.size
    corr = np.correlate(h, 1-h, 'full')
    # shift range
    # compute softmax weights
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # loops
    for idx, L in enumerate(range(-(n-1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if L >= 0:
            # forward
            grad[:n-L] += w * (1 - h[L:])
            grad[L:]   -= w * h[:n-L]
        else:
            s = -L
            grad[s:]   += w * (1 - h[:n-s])
            grad[:n-s] -= w * h[s:]
    return grad

    lam_low = np.min(v) - hi

        if fm > s:

        else:

    return np.clip(v - lam_mid, lo, hi)

    corr = np.correlate(h, 1.0 - h, mode='full')

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 2
    while f(lam_high) > s:
        lam_high += (hi - lo) * 2
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def softmax_gradient(h, alpha):
    n = h.size
    corr = np.correlate(h, 1 - h, 'full')
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    lag_start = -(n - 1)
    for idx, L in enumerate(range(lag_start, n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if L >= 0:
            grad[:n - L] += w * (1 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            s = -L
            grad[s:] += w * (1 - h[:n - s])
            grad[:n - s] -= w * h[s:]
    return grad

class Timer:
    def __init__(self, budget):
        self.start = time.time()
        self.budget = budget
    def time_left(self):
        return self.budget - (time.time() - self.start)

def gradient_descent_softmax(h0, max_iters, rng, timer):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_score = best_score
    mom = np.zeros_like(h)
    beta = 0.9
    step = 0.8
    alpha = 1.0
    alpha_inc = 0.03
    for it in range(1, max_iters + 1):
        if timer.time_left() <= 0:
            break
        grad = softmax_gradient(h, alpha)
        mom = beta * mom + (1 - beta) * grad
        h_new = h - step * mom
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_h, best_score = h_new.copy(), new_score
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step * 1.02, 1.5)
        else:
            step = max(step * 0.5, 1e-5)
            if step < 1e-3:
                jitter = rng.normal(scale=0.005, size=n)
                h = project_onto_box_sum(h + jitter, s)
                cur_score = compute_c5(h)
        alpha = min(alpha + alpha_inc, 30.0)
        if it % 800 == 0:
            jitter = rng.normal(scale=0.003, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
    return best_h, best_score
