def compute_c5_fast(h):
    n = h.size
    # Use FFT of length next power of two >= 2*n-1
    m = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    H1 = np.fft.rfft(1 - h, n=m)
    corr = np.fft.irfft(H * np.conj(H1), n=m)[:2*n-1]
    # Multiply by dx = 2/n
    return np.max(corr) * (2.0 / n)

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    b_rev = (1.0 - h)[::-1]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    B = np.fft.rfft(b_rev, n=m)
    conv = np.fft.irfft(H * B, n=m)[:2 * n - 1]
    return np.max(conv) * (2.0 / n)

def full_corr_fast(h):
    n = h.shape[0]
    b_rev = (1.0 - h)[::-1]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    B = np.fft.rfft(b_rev, n=m)
    return np.fft.irfft(H * B, n=m)[:2 * n - 1]

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_adam(h0, max_iters, step0, rng, max_time):
    start = time.time()
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    eps = 1e-8
    for it in range(1, max_iters + 1):
        if time.time() - start >= max_time:
            break
        corr = full_corr_fast(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        h_new = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fast(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)
