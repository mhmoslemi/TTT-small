def optimize_h(h, budget_s, rng):
    start = time.time()
    best_h = h.copy()
    best_score = compute_c5(h)
    # we can compute a once
    a = compute_a(h)
    while time.time() - start < budget_s:
        # compute min_a and low ks
        min_a = a.min()
        # threshold for low ks: within 0.001% maybe:
        eps = 1e-6
        low_mask = a <= min_a * (1 + eps)
        low_ks = np.where(low_mask)[0] - (len(h)-1)
        # compute weights w
        n = len(h)
        w = np.zeros(n, dtype=float)
        for k in low_ks:
            if k >= 0:
                w[:n-k] += h[k:]
                w[k:] += h[:n-k]
            else:
                kk = -k
                w[kk:] += h[:n-kk]
                w[:n-kk] += h[kk:]
        # find i_low, i_high
        # exclude indices where they cannot move
        i_low_candidates = np.where(h > 1e-9)[0]
        i_high_candidates = np.where(h < 1 - 1e-9)[0]
        if i_low_candidates.size == 0 or i_high_candidates.size == 0:
            break
        i_low = i_low_candidates[np.argmin(w[i_low_candidates])]
        i_high = i_high_candidates[np.argmax(w[i_high_candidates])]
        if w[i_high] <= w[i_low]:
            # no beneficial move, break
            break
        # amount to transfer
        max_delta = min(h[i_low], 1 - h[i_high])
        if max_delta <= 0:
            break
        delta = max_delta * step_factor
        # apply
        h[i_low] -= delta
        h[i_high] += delta
        # recompute a (full recompute)
        a = compute_a(h)
        # compute new C5
        new_score = compute_c5_from_a(a, len(h))
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
        # optionally break if best_score below target
        if best_score <= 0.3808:
            break
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto the box [lo,hi] with sum constraint total."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n (both on [0,2] interval)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_c5_from_a(a, n):
    """C5 = 1 - (2/n) * min autocorrelation."""
    return 1.0 - (2.0 / n) * a.min()


def optimize_h_with_budget(h, budget_s, rng):
    """Deterministic / stochastic local search with time budget."""
    n = h.shape[0]
    a = np.correlate(h, h, mode='full')
    best_h = h.copy()
    best_score = compute_c5_from_a(a, n)
    TARGET = 0.3808
    start = time.time()
    while time.time() - start < budget_s:
        # low-autocorrelation shifts
        min_a = a.min()
        low_idx = np.where(a <= min_a + 1e-12)[0]
        ks = low_idx - (n - 1)

        # weight vector w_i = sum_{k in low} (h_{i+k} + h_{i-k})
        w = np.zeros(n, dtype=float)
        for k in ks:
            if k >= 0:
                w[:n - k] += h[k:]
                w[k:] += h[:n - k]
            else:
                kk = -k
                w[kk:] += h[:n - kk]
                w[:n - kk] += h[kk:]

        # decide move type
        if rng.random() < 0.8:
            mask_low = h > 1e-9
            mask_high = h < 1 - 1e-9
            if not mask_low.any() or not mask_high.any():
                break
            low_idxs = np.where(mask_low)[0]
            high_idxs = np.where(mask_high)[0]
            i_low = low_idxs[np.argmin(w[low_idxs])]
            i_high = high_idxs[np.argmax(w[high_idxs])]
            max_delta = min(h[i_low], 1.0 - h[i_high])
            if max_delta > 0:
                delta = max_delta * rng.uniform(0.3, 0.7)
                h[i_low] -= delta
                h[i_high] += delta
        else:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta > 0:
                delta = max_delta * rng.random()
                h[i] -= delta
                h[j] += delta

        a = np.correlate(h, h, mode='full')
        cur_score = compute_c5_from_a(a, n)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
            if best_score <= TARGET:
                break
    return best_h, best_score


def binary_hill_climb_with_budget(h_bin, budget_s, rng):
    """Swap 1/0 pairs to improve C5."""
    n = h_bin.shape[0]
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    TARGET = 0.3808
    start = time.time()
    while time.time() - start < budget_s:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc = compute_c5(h_new)
        if sc < best_score:
            best_h = h_new
            best_score = sc
            if best_score <= TARGET:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # base vector from parent or Thue–Morse
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    scales = [64, 128, 256, 512]
    total_n = sum(scales)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    for n in scales:
        if time_left() <= 0.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # time allocation proportional to n
        time_n = budget_s * n / total_n
        time_n = min(time_n, time_left())
        if n == 512:
            time_base = time_n * 0.6
            time_rand = time_n * 0.4
        else:
            time_base = time_n
            time_rand = 0.0

        # optimize starting from current base
        h_opt, sc_opt = optimize_h_with_budget(base.copy(), time_base, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        base = h_opt

        # optional random binary start for final scale
        if time_rand > 0.0:
            h_rand = random_balanced_binary(n, rng)
            h_rand = project_onto_box_sum(h_rand, n / 2.0)
            h_opt2, sc_opt2 = optimize_h_with_budget(h_rand, time_rand, rng)
            if sc_opt2 < best_score:
                best_score = sc_opt2
                best_h = h_opt2.copy()
                base = h_opt2

    # final binary refinement if time permits
    if time_left() > 0.5:
        n = best_h.shape[0]
        # round to exact half ones
        idx = np.argpartition(best_h, -n // 2)[-n // 2:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        h_bin = project_onto_box_sum(h_bin, n / 2.0)
        t_bin = time_left()
        h_bin_opt, sc_bin_opt = binary_hill_climb_with_budget(h_bin, t_bin, rng)
        if sc_bin_opt < best_score:
            best_score = sc_bin_opt
            best_h = h_bin_opt.copy()

    best_h = project_onto_box_sum(best_h, best_h.shape[0] / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.shape[0]
