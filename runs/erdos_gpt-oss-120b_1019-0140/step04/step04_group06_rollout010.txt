def simulated_annealing_binary(h_init, init_score, time_left, rng):
    h = h_init.copy()
    best_h = h.copy()
    best_score = init_score
    cur_score = init_score
    start = time.time()
    # Define schedule: exponential cooling from T0 to Tf over total time
    T0 = 0.1  # initial temperature for acceptance of higher scores
    Tf = 1e-4
    total_time = time_left()
    while time_left() > 0:
        elapsed = time.time() - start
        frac = elapsed / total_time if total_time > 0 else 1.0
        T = T0 * (Tf / T0) ** frac
        # propose swap
        # select i where h[i]==1 and j where h[j]==0
        idx_one = np.where(h > 0.5)[0]
        idx_zero = np.where(h <= 0.5)[0]
        if len(idx_one) == 0 or len(idx_zero) == 0:
            break
        i = rng.choice(idx_one)
        j = rng.choice(idx_zero)
        # swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        # compute new score
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()

import time, math, numpy as np, cvxpy as cp

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def repair_binary(vec, target_n, rng):
    target_ones = target_n // 2
    cur = int(vec.sum())
    if cur > target_ones:
        idx = np.where(vec > 0.5)[0]
        if idx.size:
            flip = rng.choice(idx, size=cur - target_ones, replace=False)
            vec[flip] = 0.0
    elif cur < target_ones:
        idx = np.where(vec <= 0.5)[0]
        if idx.size:
            flip = rng.choice(idx, size=target_ones - cur, replace=False)
            vec[flip] = 1.0
    return vec

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def solve_ilp(n, init_h, time_limit):
    # variables
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    t = cp.Variable(integer=True)

    cons = [cp.sum(h) == n // 2]

    # product linearization
    for i in range(n):
        hi = h[i]
        for j in range(n):
            cons.append(w[i, j] <= hi)
            cons.append(w[i, j] <= h[j])
            cons.append(w[i, j] >= hi + h[j] - 1)

    # shift constraints
    for d in range(-n + 1, n):
        terms = []
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                terms.append(w[i, j])
        if terms:
            cons.append(cp.sum(cp.hstack(terms)) >= t)

    # warm start
    h.value = init_h
    w.value = np.outer(init_h, init_h)
    t.value = int(np.min([np.sum(np.outer(init_h, init_h)[i, i+d]
                               for i in range(n) if 0 <= i+d < n)
                         for d in range(-n+1, n)]))

    prob = cp.Problem(cp.Maximize(t), cons)
    try:
        prob.solve(solver=cp.CBC, maxTime=time_limit, warm_start=True, verbose=False)
    except Exception:
        return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    h_sol = np.round(h.value).astype(float)
    t_sol = int(round(t.value))
    return h_sol, t_sol

def simulated_annealing_binary(h0, score0, time_left, rng):
    h = h0.copy()
    best_h, best_score = h.copy(), score0
    cur_score = score0
    start = time.time()
    total = time_left()
    if total <= 0:
        return best_h, best_score
    T0, Tf = 0.2, 1e-4
    while time_left() > 0:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        T = T0 * (Tf / T0) ** frac

        # propose a swap of a 1 and a 0
        idx_one = np.where(h > 0.5)[0]
        idx_zero = np.where(h <= 0.5)[0]
        if idx_one.size == 0 or idx_zero.size == 0:
            break
        i = rng.choice(idx_one)
        j = rng.choice(idx_zero)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()

        # occasional random perturbation
        if rng.random() < 0.01:
            a, b = rng.choice(len(h), size=2, replace=False)
            if h[a] != h[b]:
                h_tmp = h.copy()
                h_tmp[a], h_tmp[b] = h_tmp[b], h_tmp[a]
                s_tmp = compute_c5(h_tmp)
                d2 = s_tmp - cur_score
                if d2 < 0 or rng.random() < math.exp(-d2 / T):
                    h, cur_score = h_tmp, s_tmp
                    if cur_score < best_score:
                        best_score, best_h = cur_score, h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # target resolution
    n = 64
    # obtain base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != n:
            base = upsample(base, n)
    else:
        base = thue_morse(n)

    # binary conversion and repair
    h_bin = (base > 0.5).astype(float)
    h_bin = repair_binary(h_bin, n, rng)

    # try ILP if enough time
    best_h, best_score = h_bin, compute_c5(h_bin)
    if time_left() > 30.0:
        ilp_time = max(5.0, time_left() - 5.0)
        ilp_res = solve_ilp(n, h_bin, ilp_time)
        if ilp_res is not None:
            h_ilp, t_ilp = ilp_res
            c5_ilp = 1.0 - (2.0 / n) * t_ilp
            if c5_ilp < best_score:
                best_h, best_score = h_ilp, c5_ilp
            if best_score <= 0.3808:
                return best_h, best_score, n

    # fallback to simulated annealing
    if time_left() > 5.0:
        best_h, best_score = simulated_annealing_binary(best_h, best_score,
                                                        time_left, rng)

    # final repair and score recomputation
    best_h = repair_binary(best_h, n, rng)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
