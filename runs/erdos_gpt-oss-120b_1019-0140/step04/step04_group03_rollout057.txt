def adam_optimize(h0, n_iter, step, beta1, beta2, rng, max_time):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    target_sum = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    for t in range(1, n_iter+1):
        if time.time() - start >= max_time:
            break
        # compute gradient of worst shift(s)
        grad = compute_gradients_for_shifts(h, worst_shifts_of_h(h))
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        # add small noise occasionally
        if t % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=h.size)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def worst_shifts_of_h(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    idxes = np.where(corr >= max_corr - 1e-12)[0]
    shifts = idxes - (n - 1)

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # binary search on lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        w = np.clip(v - lam, lo, hi)
        s = w.sum()
        if s > target_sum:
            lam_low = lam
        else:
            lam_high = lam
        if abs(s - target_sum) < 1e-12:
            break
    return np.clip(v - lam, lo, hi)
```

def upsample(arr, new_n):

def adam_optimize(h0, max_iters, step, beta1, beta2, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters + 1):
        if time.time() - start >= max_time:
            break
        # gradient of worst shift(s)
        shifts = worst_shifts_of_h(h)
        grad = compute_gradients_for_shifts(h, shifts)
        # adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        # occasional noise
        if t % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def targeted_annealing(h0, max_iters, T0, T_final, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(max_iters):
        if time.time() - start >= max_time:
            break
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        # counterpart values
        counterpart = np.zeros(n, dtype=h.dtype)
        if lag >= 0:
            counterpart[:n - lag] = h[lag:]
        else:
            s = -lag
            counterpart[s:] = h[:n - s]
        # choose i with h[i] > 0 and counterpart[i] low
        low_mask = (h > 1e-12) & (counterpart < 0.5)
        high_mask = (h < 1 - 1e-12) & (counterpart > 0.5)
        if low_mask.any() and high_mask.any():
            i = rng.choice(np.where(low_mask)[0])
            j = rng.choice(np.where(high_mask)[0])
        else:
            i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        # temperature schedule linear
        T = T0 * (1 - it / max_iters) + T_final * (it / max_iters)
        delta = max_move * rng.random() * (T / T0)  # scale with temperature
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # after move, sum unchanged; still in box
        score = compute_c5(h_new)
        if score < best_score or rng.random() < np.exp(-(score - best_score) / (T + 1e-12)):
            h = h_new
            if score < best_score:
                best_score = score
                best_h = h.copy()
    return best_h, best_score
```

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

        return np.full_like(v, hi)

    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        w = np.clip(v - lam, lo, hi)
        s = w.sum()

            lam_low = lam
        else:
            lam_high = lam

    # optional: if sum differs, perform a tiny correction
    diff = target_sum - w.sum()
    if abs(diff) > 1e-10:

        mask = (w > lo + 1e-12) & (w < hi - 1e-12)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, lo, hi)

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam, lo, hi)
        s = w.sum()
        if s > target_sum:
            lam_low = lam
        else:
            lam_high = lam
        if abs(s - target_sum) < 1e-12:
            break
    w = np.clip(v - lam, lo, hi)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-10:
        mask = (w > lo + 1e-12) & (w < hi - 1e-12)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, lo, hi)
    return w

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def worst_shifts_of_h(h, tol=1e-12):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = corr.max()
    idxes = np.where(corr >= mx - tol)[0]
    return idxes - (n - 1)

def adam_optimize(h0, max_iters, step, beta1, beta2, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.size
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters + 1):
        if time.time() - start >= max_time:
            break
        shifts = worst_shifts_of_h(h)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        if t % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def targeted_annealing(h0, max_iters, T0, T_final, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    for it in range(max_iters):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        counterpart = np.zeros(n, dtype=h.dtype)
        if lag >= 0:
            counterpart[: n - lag] = h[lag:]
        else:
            s = -lag
            counterpart[s:] = h[: n - s]
        low_mask = (h > 1e-12) & (counterpart < 0.5)
        high_mask = (h < 1 - 1e-12) & (counterpart > 0.5)
        if low_mask.any() and high_mask.any():
            i = rng.choice(np.where(low_mask)[0])
            j = rng.choice(np.where(high_mask)[0])
        else:
            i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        T = T0 * (1 - it / max_iters) + T_final * (it / max_iters)
        delta = max_move * rng.random() * (T / T0)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
    return best_h, best_score
