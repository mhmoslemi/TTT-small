def anneal_binary(h, current_score, time_left, max_iters=20000):
    n=h.size
    best_h = h.copy()
    best_score = current_score
    T0=0.1; T_min=1e-5; decay=0.9999
    T=T0
    it=0
    while time_left()>0.01 and it < max_iters:
        # choose random 1 and 0 indices
        ones = np.where(h>0.5)[0]
        zeros = np.where(h<=0.5)[0]
        if ones.size==0 or zeros.size==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i]=0.0; h[j]=1.0
        new_score=compute_c5_fft(h)
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta/T):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i]=1.0; h[j]=0.0
        T = max(T_min, T*decay)
        it+=1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def adam_optimize(h0, time_left, max_iters=4000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_sc = compute_c5_fft(h)
    for t in range(1, max_iters + 1):
        if time_left() < 0.1:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = corr.max()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        eta = 0.5 / np.sqrt(t)
        h = h - eta * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, s)
        sc = compute_c5_fft(h)
        if sc < best_sc:
            best_sc, best_h = sc, h.copy()
    return best_h, best_sc

def anneal_binary(h, cur_sc, time_left, max_iters=50000):
    n = h.size
    rng = np.random.default_rng()
    best_h = h.copy()
    best_sc = cur_sc
    T = 0.2
    T_min = 1e-5
    decay = 0.9999
    it = 0
    while time_left() > 0.01 and it < max_iters:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_sc = compute_c5_fft(h)
        delta = new_sc - cur_sc
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_sc
