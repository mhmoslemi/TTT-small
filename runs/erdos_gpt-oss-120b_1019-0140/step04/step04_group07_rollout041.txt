import time
import numpy as np
import cvxpy as cp

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max correlation w.r.t. h (average over worst shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            contrib = h[:n - shift] * (1.0 - h[shift:])
            grad[:n - shift] += 1.0 - h[shift:]
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            contrib = h[s:] * (1.0 - h[:n - s])
            grad[s:]       += 1.0 - h[:n - s]
            grad[:n - s]   -= h[s:]
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_adaptive(h0, step0, beta, rng, time_left):
    """Projected gradient descent with momentum and adaptive step."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    it = 0
    while time_left() > 0:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    """Transfer a small mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    """Move mass from a point that contributes most to the worst shift."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = int(rng.choice(shifts))
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = int(np.argmax(contrib))
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = int(np.argmax(contrib) + s)
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score

def solve_lp(n, time_left, rng):
    """Solve LP maximizing min_k sum_i min(h_i, h_{i+k}). Returns h and LP value."""
    if time_left() < 10.0:
        return None, None
    h = cp.Variable(n)
    m = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    for k in range(-(n - 1), n):
        length = n - abs(k)
        u = cp.Variable(length, nonneg=True)
        constraints.append(cp.sum(u) >= m)
        if k >= 0:
            i_slice = slice(0, n - k)
            j_slice = slice(k, n)
        else:
            i_slice = slice(-k, n)
            j_slice = slice(0, n + k)
        constraints.append(u <= h[i_slice])
        constraints.append(u <= h[j_slice])
    prob = cp.Problem(cp.Maximize(m), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, warm_start=True, verbose=False)
    except Exception:
        return None, None
    if h.value is None:
        return None, None
    h_opt = np.clip(h.value, 0.0, 1.0)
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, prob.value

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # start from provided initial if any
    best_h = None
    best_score = np.inf
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = project_onto_box_sum(init, init.shape[0] / 2.0)
        best_h = init.copy()
        best_score = compute_c5(best_h)

    # --- LP stage (n=128) -------------------------------------------------
    n_lp = 128
    h_lp, lp_val = solve_lp(n_lp, time_left, rng)
    if h_lp is not None:
        c5_lp = compute_c5(h_lp)
        if c5_lp < best_score:
            best_h, best_score = h_lp, c5_lp

    # If LP failed, fall back to a simple balanced binary start
    if best_h is None:
        best_h = np.zeros(n_lp, dtype=float)
        best_h[rng.choice(n_lp, size=n_lp // 2, replace=False)] = 1.0
        best_h = project_onto_box_sum(best_h, n_lp / 2.0)
        best_score = compute_c5(best_h)

    # --- refinement on current scale ---------------------------------------
    # Adaptive GD
    if time_left() > 5.0:
        gd_h, gd_score = gradient_descent_adaptive(
            best_h, step0=0.8, beta=0.9, rng=rng, time_left=time_left)
        if gd_score < best_score:
            best_h, best_score = gd_h, gd_score

    # targeted max-shift moves
    if time_left() > 5.0:
        ms_h, ms_score = improve_via_max_shift(best_h, rng, time_left)
        if ms_score < best_score:
            best_h, best_score = ms_h, ms_score

    # local pair moves
    if time_left() > 5.0:
        lpair_h, lpair_score = local_pair_moves(best_h, rng, time_left)
        if lpair_score < best_score:
            best_h, best_score = lpair_h, lpair_score

    # --- upscale to 256 if time permits ------------------------------------
    target_scales = [256, 512]
    for n_target in target_scales:
        if time_left() < 10.0:
            break
        # upscale
        best_h = upsample(best_h, n_target)
        best_h = project_onto_box_sum(best_h, n_target / 2.0)

        # GD on new scale
        if time_left() > 5.0:
            step0 = 0.5 if n_target == 256 else 0.3
            gd_h, gd_score = gradient_descent_adaptive(
                best_h, step0=step0, beta=0.9, rng=rng, time_left=time_left)
            if gd_score < best_score:
                best_h, best_score = gd_h, gd_score
            best_h = gd_h

        # max‑shift improvement
        if time_left() > 5.0:
            ms_h, ms_score = improve_via_max_shift(best_h, rng, time_left)
            if ms_score < best_score:
                best_h, best_score = ms_h, ms_score
            best_h = ms_h

        # local moves
        if time_left() > 5.0:
            lp_h, lp_score = local_pair_moves(best_h, rng, time_left)
            if lp_score < best_score:
                best_h, best_score = lp_h, lp_score
            best_h = lp_h

        # occasional random binary restart + short refinement (only at 256)
        if n_target == 256 and time_left() > 5.0:
            for _ in range(2):
                if time_left() < 2:
                    break
                rnd = np.zeros(n_target, dtype=float)
                idx = rng.choice(n_target, size=n_target // 2, replace=False)
                rnd[idx] = 1.0
                rnd = project_onto_box_sum(rnd, n_target / 2.0)
                gd_h2, gd_score2 = gradient_descent_adaptive(
                    rnd, step0=0.5, beta=0.9, rng=rng, time_left=time_left)
                if gd_score2 < best_score:
                    best_h, best_score = gd_h2, gd_score2
                # a brief local move

                    if lp_score2 < best_score:

    # final projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
