def compute_c5_and_shift(h):
    n = h.shape[0]
    size = 2 * n
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(H * np.conj(G), size)
    # shift zero to center
    corr_centered = np.fft.ifftshift(corr)
    # remove the first element (lag -n)
    c_full = corr_centered[1:2*n]  # length 2n-1
    max_idx = int(np.argmax(c_full))
    max_corr = c_full[max_idx]
    shift = max_idx - (n - 1)
    return max_corr * (2.0 / n), shift

def compute_c5_and_shift(h):
    n = h.shape[0]
    size = 2 * n
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(H * np.conj(G), size)
    corr_centered = np.fft.ifftshift(corr)
    c_full = corr_centered[1:2*n]  # length 2n-1
    max_idx = int(np.argmax(c_full))
    max_corr = c_full[max_idx]
    shift = max_idx - (n - 1)
    # Integrate scaling
    c5 = max_corr * (2.0 / n)
    return c5, shift

def compute_c5(h):
    # simple wrapper
    c5, _ = compute_c5_and_shift(h)
    return c5

def upsample(arr, target_n):
    m = arr.shape[0]
    if m == target_n:

    x_old = np.linspace(0.0, 1.0, m, endpoint=False)

    interp = np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5_and_shift(h):
    """Return (C5, shift) using FFT for speed."""
    n = h.shape[0]
    size = 2 * n
    H = np.fft.rfft(h, size)

    corr = np.fft.irfft(H * np.conj(G), size)
    corr_centered = np.fft.ifftshift(corr)

    max_idx = int(np.argmax(c_full))

    shift = max_idx - (n - 1)

    return c5, shift

def upsample(arr, target_n):
    """Linear interpolation from arr (len m) to target_n, then project to sum constraint."""
    m = arr.shape[0]
    if m == target_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
    interp = np.interp(x_new, x_old, arr)
    return project_onto_box_sum(interp, target_n / 2.0)


def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def gradient_descent(h0, time_limit, rng, target_score=0.3808):
    """Deterministic pairwise mass transfer guided by gradient of the active shift."""
    start = time.time()
    n = h0.size
    idx_array = np.arange(n)

    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    while time.time() - start < time_limit:
        cur_score, k_star = compute_c5_and_shift(cur_h)
        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h.copy()
            if best_score <= target_score:
                break

        # gradient-like vector: (1 - h_{i+k}) - h_{i-k}, with zero padding
        idx_plus = idx_array + k_star
        idx_minus = idx_array - k_star
        term1 = np.where((idx_plus >= 0) & (idx_plus < n), 1.0 - cur_h[idx_plus], 1.0)
        term2 = np.where((idx_minus >= 0) & (idx_minus < n), cur_h[idx_minus], 0.0)
        grad = term1 - term2

        i_dec = int(np.argmax(grad))   # decreasing hi (grad > 0)
        j_inc = int(np.argmin(grad))   # increasing hj (grad < 0)

        if grad[i_dec] <= 0 or grad[j_inc] >= 0:
            # no improving pair; random local perturbation
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta > 0:
                delta = max_delta * rng.random() * 0.5
                cur_h[i] -= delta
                cur_h[j] += delta
                cur_score = compute_c5(cur_h)
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = cur_h.copy()
            continue

        max_delta = min(cur_h[i_dec], 1.0 - cur_h[j_inc])
        improved = False
        for frac in (1.0, 0.5, 0.25, 0.125):
            delta = max_delta * frac
            if delta <= 0:
                continue
            cur_h[i_dec] -= delta
            cur_h[j_inc] += delta
            new_score = compute_c5(cur_h)
            if new_score < cur_score:
                cur_score = new_score
                improved = True
                break
            cur_h[i_dec] += delta
            cur_h[j_inc] -= delta
        if not improved:
            # small random move as fallback
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta2 = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta2 > 0:
                delta = max_delta2 * rng.random() * 0.5
                cur_h[i] -= delta
                cur_h[j] += delta
                cur_score = compute_c5(cur_h)
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = cur_h.copy()
    return best_h, best_score


def simulated_annealing(h0, time_limit, rng, target_score=0.3808):
    """Simple SA preserving sum and box constraints."""
    start = time.time()
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    init_temp = cur_score * 0.1 + 1e-4

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = init_temp * (1.0 - frac)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target_score:
                    break
        else:
            if temp > 0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_score = new_score
                else:
                    h[i] += delta
                    h[j] -= delta
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # target grid size
    n_target = 512

    # base construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_balanced_binary(64, rng)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    # upscale to target resolution
    if base.shape[0] != n_target:
        base = upsample(base, n_target)
    base = project_onto_box_sum(base, n_target / 2.0)

    # Phase 1: gradient descent (≈50% of budget)
    gd_budget = time_left() * 0.5
    if gd_budget > 5.0:
        best_h, best_score = gradient_descent(base, gd_budget, rng, target_score=0.3808)
    else:
        best_h = base.copy()
        best_score = compute_c5(best_h)

    # Phase 2: simulated annealing with remaining time
    remaining = time_left()
    if remaining > 5.0:
        h_sa, s_sa = simulated_annealing(best_h, remaining, rng, target_score=0.3808)
        if s_sa < best_score:
            best_h, best_score = h_sa, s_sa

    # final projection and exact evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
