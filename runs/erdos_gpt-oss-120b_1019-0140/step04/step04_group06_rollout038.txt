def compute_c5(h):
    n = h.shape[0]

```
def max_shifts(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()
```

def simulated_annealing_binary(h0, time_left, target=0.3808, rng=None):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # temperature schedule depending on elapsed time
    while time_left() > 0.01:
        elapsed = time.time() - start
        # compute fraction of allowed time
        # Use linear cooling from T0 to T1
        T0 = 0.1
        T1 = 0.001
        T = T0 * (T1 / T0) ** (elapsed / (time_left() + elapsed))
        # random swap
        ones_idx = np.flatnonzero(h == 1)
        zeros_idx = np.flatnonzero(h == 0)
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5(h)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # optionally do occasional greedy local search
        if rng.random() < 0.01:
            h, cur_score = greedy_swap_local_search(h, time_left, target)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        if best_score <= target:
            break
    return best_h, best_score

            rng = np.arange(-k, n)
            sel = rng[(h[rng] == 1) & (h[rng + k] == 0)]
        if sel.size:
            pos.append(sel)
    return np.concatenate(pos) if pos else np.empty(0, dtype=int)

def binarize(v):
    n = v.shape[0]
    ones = n // 2
    idx = np.argpartition(v, -ones)[-ones:]
    h = np.zeros_like(v, dtype=float)
    h[idx] = 1.0
    return h

def time_remaining(start, budget):
    return max(0.0, budget - (time.time() - start))

def greedy_swap_local_search(h, start, budget, target, rng):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while time_remaining(start, budget) > 0.02:
        shifts = max_shifts(best_h)
        i_cand = max_overlap_positions(best_h, shifts)
        if i_cand.size == 0:
            i_cand = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if i_cand.size == 0 or zeros.size == 0:
            break
        ns = min(200, i_cand.size, zeros.size)
        i_samp = rng.choice(i_cand, size=ns, replace=False)
        j_samp = rng.choice(zeros, size=ns, replace=False)
        improved = False
        best_delta = 0.0
        best_swap = None
        for i, j in zip(i_samp, j_samp):
            trial = best_h.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            c5 = compute_c5(trial)
            delta = best_score - c5
            if delta > best_delta:
                best_delta = delta
                best_swap = (i, j)
                improved = True
        if not improved:
            break
        i, j = best_swap
        best_h[i] = 0.0
        best_h[j] = 1.0
        best_score = compute_c5(best_h)
        if best_score <= target:
            break
    return best_h, best_score

def simulated_annealing_binary(h0, start, budget, target, rng):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T0, T1 = 0.1, 0.001
    while time_remaining(start, budget) > 0.01:
        frac = (time.time() - start) / budget
        T = T0 * ((T1 / T0) ** frac)
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        dE = new_score - cur_score
        accept = dE < 0 or rng.random() < math.exp(-dE / T)
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                if best_score <= target:
                    break
        else:
            h[i] = 1.0
            h[j] = 0.0
        if rng.random() < 0.01:
            h, cur_score = greedy_swap_local_search(h, start, budget, target, rng)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    target = 0.3808

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.shape[0]
        if n0 % 2 != 0:
            n0 -= 1
            init = init[:n0]
        h0 = binarize(init)
    else:
        n0 = 64
        h0 = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        h0[idx] = 1.0

    best_h = h0
    best_score = compute_c5(best_h)

    # random restarts at base size
    while time_remaining(start, budget_s) > 20 and best_score > target:
        h_rand = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        h_rand[idx] = 1.0
        h_opt, sc_opt = simulated_annealing_binary(h_rand, start, budget_s, target, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        h_opt2, sc_opt2 = greedy_swap_local_search(best_h, start, budget_s, target, rng)
        if sc_opt2 < best_score:
            best_score, best_h = sc_opt2, h_opt2.copy()

    current_h = best_h
    # scale up twice (64→128→256)
    for factor in (2, 2):
        if time_remaining(start, budget_s) <= 5:
            break
        current_h = np.repeat(current_h, factor)
        n_cur = current_h.size
        # small random perturbation
        swaps = max(1, n_cur // 200)
        for _ in range(swaps):
            ones = np.flatnonzero(current_h == 1)
            zeros = np.flatnonzero(current_h == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            current_h[i] = 0.0
            current_h[j] = 1.0
        # greedy then annealing then greedy
        current_h, cur_sc = greedy_swap_local_search(current_h, start, budget_s, target, rng)
        if cur_sc < best_score:
            best_score, best_h = cur_sc, current_h.copy()
        current_h, cur_sc = simulated_annealing_binary(current_h, start, budget_s, target, rng)
        if cur_sc < best_score:
            best_score, best_h = cur_sc, current_h.copy()
        current_h, cur_sc = greedy_swap_local_search(current_h, start, budget_s, target, rng)
        if cur_sc < best_score:
            best_score, best_h = cur_sc, current_h.copy()

    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
