def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def gradient_descent(h0, time_left, max_iters=200000, step0=0.5, beta=0.9):

    h = h0.copy()
    target_sum = h.size/2
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iters and time_left()>0:
        # compute correlation full
        corr = compute_corr_scaled_fft(h) # returns full corr *dx (same scaling)
        # compute max correlation
        max_corr = np.max(corr)
        # get shifts where correlation close to max
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        # compute gradient
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum and step
        momentum = beta*momentum + (1-beta)*grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project onto feasible box sum
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5_fft(h_new)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_anneal(h0, rng, time_left, max_iters=500000, T0=0.5, decay=0.9999, Tmin=1e-6):

    # Ensure binary: threshold top half just once (or h0 assumed binary)
    n = h0.size
    k = n//2
    # Ensure exactly k ones:
    idx = np.argpartition(-h0, k)[:k]
    h = np.zeros_like(h0, dtype=float)
    h[idx] = 1.0
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left()>0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def fractional_swap_anneal(h0, rng, time_left, max_iters=200000, T0=0.1, decay=0.99995, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.02:
        # select i from indices where h[i] > 1e-12, j where h[j] < 1-1e-12
        pos = np.where(h > 1e-12)[0]
        neg = np.where(h < 1 - 1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        # compute max allowable delta
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5  # random fraction up to 0.5 of allowed amount
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5_fft(h)
        delta_score = new_score - current_score
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta_score / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto { lo <= x_i <= hi, sum x_i = s }."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap C5 from h using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array of h with (1‑h), scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max‑correlation w.r.t. h for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, time_left, max_iters=300000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iters and time_left() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_anneal(h0, rng, time_left, max_iters=600000,
                 T0=0.5, decay=0.9999, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def fractional_swap_anneal(h0, rng, time_left,
                          max_iters=200000, T0=0.1,
                          decay=0.9999, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.02:
        pos = np.where(h > 1e-12)[0]
        neg = np.where(h < 1.0 - 1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5_fft(h)
        dscore = new_score - current_score
        accept = dscore < 0 or (T > 0 and rng.random() < np.exp(-dscore / T))
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # target resolution (prefer ≤1k points, power of two for FFT)
    n_target = 1024

    # initialise h
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.size != n_target:
            init_arr = upsample(init_arr, n_target)
        h0 = project_onto_box_sum(init_arr, n_target / 2.0)
    else:
        h0 = random_binary_balanced(n_target, rng)

    best_h = h0.copy()
    best_score = compute_c5_fft(best_h)

    # Phase 1: continuous gradient descent
    if time_left() > 10.0:
        h_cont, sc_cont = gradient_descent(best_h, time_left)
        if sc_cont < best_score:
            best_score, best_h = sc_cont, h_cont.copy()

    # Phase 2: convert to exact binary (half ones)
    if time_left() > 5.0:
        n = best_h.size
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()

    # Phase 3: binary simulated annealing
    if time_left() > 5.0:
        n = best_h.size
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        h_ann, sc_ann = binary_anneal(h_bin, rng, time_left)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()

    # Phase 4: random binary restarts with annealing
    while time_left() > 30.0:
        n = best_h.size
        idx = rng.choice(n, size=n // 2, replace=False)
        h_rand = np.zeros(n, dtype=float)
        h_rand[idx] = 1.0
        h_opt, sc_opt = binary_anneal(h_rand, rng, time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= 0.3808:
                break

    # Phase 5: fine‑grained fractional swaps
    if time_left() > 5.0:
        h_frac, sc_frac = fractional_swap_anneal(best_h, rng, time_left)
        if sc_frac < best_score:
            best_score, best_h = sc_frac, h_frac.copy()

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
