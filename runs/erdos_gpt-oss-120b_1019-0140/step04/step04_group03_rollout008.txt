def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    m = 2 * n - 1
    b_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=m)
    B = np.fft.rfft(b_rev, n=m)
    c = np.fft.irfft(H * B, n=m)
    return np.max(c) * (2.0 / n)

def correlation_fft(h):
    n = h.shape[0]
    m = 2 * n - 1
    b_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=m)
    B = np.fft.rfft(b_rev, n=m)
    return np.fft.irfft(H * B, n=m)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        c = correlation_fft(h)
        max_corr = c.max()
        shifts = np.where(c >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def to_binary_balanced(h):
    n = h.shape[0]
    half = n // 2
    idx = np.argpartition(-h, half - 1)[:half]
    hb = np.zeros(n, dtype=float)
    hb[idx] = 1.0
    return hb

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def simulated_annealing_binary(h0, start_T, steps, rng):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.shape[0]
    for step in range(steps):
        T = max(start_T * (1.0 - step / steps), 1e-12)
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n // 2
    best_h = None
    best_score = np.inf

    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, target_sum)
        base_score = compute_c5(base)
        best_h, best_score = base, base_score

    # Main binary simulated annealing loop
    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 10.0:
            break
        # start from a random binary balanced vector
        h0 = random_binary_balanced(n, rng)
        # steps estimated from remaining time (≈800 steps per second)
        steps = int(min(200000, max(20000, remaining * 800)))
        h_sa, sc_sa = simulated_annealing_binary(h0, start_T=0.5, steps=steps, rng=rng)
        if sc_sa < best_score:
            best_score, best_h = sc_sa, h_sa.copy()
        # also try SA from binary version of current best if different
        if best_h is not None:
            h_bin = to_binary_balanced(best_h)
            h_sa2, sc_sa2 = simulated_annealing_binary(h_bin, start_T=0.5, steps=steps // 2, rng=rng)
            if sc_sa2 < best_score:
                best_score, best_h = sc_sa2, h_sa2.copy()
        if best_score <= 0.3808:
            break

    # Final continuous refinement if time permits
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0 and best_h is not None:
        max_iters = int(min(200000, max(50000, remaining * 300)))  # ≈300 iterations per second
        best_h, best_score = gradient_descent_momentum_custom(

            rng=rng, max_time=remaining)
        best_h = project_onto_box_sum(best_h, target_sum)
        best_score = compute_c5(best_h)

        best_h = project_onto_box_sum(thue_morse(n).astype(float), target_sum)
