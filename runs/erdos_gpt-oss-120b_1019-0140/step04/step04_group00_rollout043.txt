def corr_full(h):
    # using FFT for init only
    N = 1 << ((2*n - 1).bit_length())

function run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start_time))

    n = 256
    target_sum = n // 2

    # If initial_h_values provided, use it.
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # Resample if needed
        if init.shape[0] != n:
            # use linear interpolation as in previous code.
            x_old = np.linspace(0, 1, init.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        # Ensure sum constraint exactly via projection
        # Use project_onto_box_sum from previous code or implement quickly.
        # We'll implement a simple projection function for box constraints & sum.
        # We'll reuse the project_onto_box_sum from earlier code if imported? 
        # It may not be pre-imported. We'll define our own, based on the earlier.
    else:
        # random balanced binary
        init = np.zeros(n, dtype=float)
        init[rng.choice(n, size=target_sum, replace=False)] = 1.0

    # we need to create binary h for annealing: binarize init (if fractional).
    # We'll do: idx_top = np.argpartition(init, -target_sum)[-target_sum:] ; set them 1.
    if init.dtype != np.float64:
        init = init.astype(float) # ensure float

    # If init contains non-binary values (not exactly 0 or 1), convert:
    if not np.all((init == 0) | (init == 1)):
        # Use top half rule
        idx_top = np.argpartition(init, -target_sum)[-target_sum:]
        h0 = np.zeros(n, dtype=float)
        h0[idx_top] = 1.0
    else:
        h0 = init.copy()

    # Ensure sum exactly target: might be off due to duplicates etc.
    # If sum not equal target_sum, adjust:
    current_sum = int(h0.sum())
    if current_sum != target_sum:
        # adjust by flipping
        diff = target_sum - current_sum
        if diff > 0:
            # need to add diff ones: pick zero positions with largest init values
            zeros = np.where(h0 == 0)[0]
            # among zeros, choose those with largest init values
            if zeros.size > 0:
                # sort zeros by init descending
                sorted_zeros = zeros[np.argsort(-init[zeros])]
                h0[sorted_zeros[:diff]] = 1.0
        else:
            # need to remove -diff ones: pick one positions with smallest init values
            ones = np.where(h0 == 1)[0]
            sorted_ones = ones[np.argsort(init[ones])]
            h0[sorted_ones[: -diff]] = 0.0

    # pre-allocate data structures
    n2 = 2*n - 1
    # compute initial correlation via FFT
    def compute_corr_fft(arr):
        N = 1 << ((2*n - 1).bit_length())
        fft_h = np.fft.rfft(arr, n=N)
        fft_one = np.fft.rfft(1.0 - arr, n=N)
        conv = np.fft.irfft(fft_h * fft_one, n=N)
        return conv[:n2]

    corr = compute_corr_fft(h0)
    cur_max_raw = corr.max()
    best_h = h0.copy()
    best_max_raw = cur_max_raw
    best_corr = corr.copy()
    # pre-alloc delta buffer
    delta_buf = np.empty(n2, dtype=float)

    # annealing parameters
    T = 0.5  # raw temperature
    T_min = 1e-6
    decay = 0.99999  # per iteration, adjust to shrink slowly

    # Precompute ones complement array: one_minus_h = 1 - h
    one_minus_h = 1.0 - h0

    # Get positions of zeros and ones as arrays for fast sampling
    zeros = np.where(h0 == 0)[0]
    ones = np.where(h0 == 1)[0]

    # main loop
    iter_count = 0
    while time_left() > 0.1:
        iter_count += 1
        # occasionally recompute zeros/ones lists if they change infrequently
        if iter_count % 1000 == 0:
            zeros = np.where(h0 == 0)[0]
            ones = np.where(h0 == 1)[0]
        # randomly select a swap
        if zeros.size == 0 or ones.size == 0:
            break
        i0 = rng.choice(zeros)  # zero -> becomes 1
        i1 = rng.choice(ones)   # one -> becomes 0
        # compute delta
        delta_buf.fill(0.0)
        # i1: add (1 - h) shifted
        start = n - 1 - i1
        delta_buf[start:start + n] += one_minus_h
        # i1: subtract h shifted
        delta_buf[i1:i1 + n] -= h0
        # i0: subtract (1 - h) shifted
        start0 = n - 1 - i0
        delta_buf[start0:start0 + n] -= one_minus_h
        # i0: add h shifted
        delta_buf[i0:i0 + n] += h0

        new_max_raw = np.max(corr + delta_buf)
        # acceptance
        improve = new_max_raw < cur_max_raw
        if improve or (rng.random() < np.exp(-(new_max_raw - cur_max_raw) / T)):
            # accept swap
            # update h, one_minus_h, corr, cur_max_raw
            h0[i0] = 1.0
            h0[i1] = 0.0
            one_minus_h[i0] = 0.0
            one_minus_h[i1] = 1.0
            corr += delta_buf
            cur_max_raw = new_max_raw
            # update best
            if cur_max_raw < best_max_raw:
                best_max_raw = cur_max_raw
                best_h = h0.copy()
                best_corr = corr.copy()
                # early exit if below target
                if best_max_raw * (2.0/n) <= 0.3808:
                    return best_h, best_max_raw * (2.0/n), n
        # update temperature
        T = max(T_min, T * decay)

    # final best result based on best_h
    c5_bound = best_max_raw * (2.0/n)
    return best_h, c5_bound, n

def anneal_start(h0, time_limit):
    # h0 binary vector
    n = h0.shape[0]
    n2 = 2*n-1
    # compute initial corr
    corr = compute_corr_fft(h0)
    cur_max_raw = corr.max()
    best_h_local = h0.copy()
    best_max_raw_local = cur_max_raw
    delta_buf = np.empty(n2, dtype=float)
    one_minus_h = 1.0 - h0
    zeros = np.where(h0 == 0)[0]
    ones = np.where(h0 == 1)[0]
    rng = np.random.default_rng() # use global rng maybe

    # parameters
    T = 0.5
    T_min = 1e-6
    decay = 0.99995 # maybe a bit slower

    start0 = time.time()
    it = 0
    while time.time() - start0 < time_limit:
        # sample swap
        if zeros.size == 0 or ones.size == 0:
            break
        a = rng.choice(zeros) # from zero to one
        b = rng.choice(ones)  # from one to zero

        # compute delta
        delta_buf.fill(0.0)
        start_a = n - 1 - a
        delta_buf[start_a:start_a + n] += one_minus_h
        delta_buf[a:a + n] -= h0
        start_b = n - 1 - b
        delta_buf[start_b:start_b + n] -= one_minus_h
        delta_buf[b:b + n] += h0
        delta_buf[n - 1] -= 2.0
        delta_buf[(b - a) + (n - 1)] += 1.0
        delta_buf[(a - b) + (n - 1)] += 1.0

        new_corr = corr + delta_buf

import time, numpy as np

def _corr_fft(h):
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    fft_h = np.fft.rfft(h, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_h * fft_one, n=N)
    return conv[:2 * n - 1]

def _binarize_top(arr, half):
    idx = np.argpartition(arr, -half)[-half:]
    h = np.zeros(arr.shape[0], dtype=float)
    h[idx] = 1.0
    return h

def _anneal(h_start, time_limit, rng):
    n = h_start.size
    n2 = 2 * n - 1
    h = h_start.copy()
    one_minus = 1.0 - h
    corr = _corr_fft(h)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    delta = np.empty(n2, dtype=float)
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    T, T_min, decay = 0.5, 1e-6, 0.99995
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        if zeros.size == 0 or ones.size == 0:
            break
        a = rng.choice(zeros)      # 0 → 1
        b = rng.choice(ones)       # 1 → 0
        delta.fill(0.0)
        # a: add (1‑h) shifted, subtract h shifted
        sa = n - 1 - a
        delta[sa:sa + n] += one_minus
        delta[a:a + n] -= h
        # b: subtract (1‑h) shifted, add h shifted
        sb = n - 1 - b
        delta[sb:sb + n] -= one_minus
        delta[b:b + n] += h
        # double‑swap correction
        delta[n - 1] -= 2.0
        delta[(b - a) + (n - 1)] += 1.0
        delta[(a - b) + (n - 1)] += 1.0
        new_corr = corr + delta
        new_max = new_corr.max()
        improve = new_max < cur_max
        if improve or rng.random() < np.exp(-(new_max - cur_max) / T):
            # accept
            h[a] = 1.0
            h[b] = 0.0
            one_minus[a] = 0.0
            one_minus[b] = 1.0
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        T = max(T_min, T * decay)
        it += 1
        if it % 1000 == 0:
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
    return best_h, best_max

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))
    n = 256
    half = n // 2
    # initial construction
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            x_old = np.linspace(0, 1, init.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        h0 = _binarize_top(init, half)
    else:
        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, size=half, replace=False)] = 1.0
    # first annealing (major portion of time)
    first_limit = time_left() * 0.6
    if first_limit > 1.0:
        best_h, best_raw = _anneal(h0, first_limit, rng)
    else:
        best_h = h0.copy()
        best_raw = _corr_fft(best_h).max()
    best_c5 = best_raw * (2.0 / n)
    target = 0.3808
    if best_c5 <= target:
        return best_h, best_c5, n
    # additional random restarts
    while time_left() > 5.0:
        h_rand = np.zeros(n, dtype=float)
        h_rand[rng.choice(n, size=half, replace=False)] = 1.0
        limit = min(30.0, time_left() * 0.3)
        h_loc, raw_loc = _anneal(h_rand, limit, rng)
        c5_loc = raw_loc * (2.0 / n)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc.copy()
            if best_c5 <= target:
                return best_h, best_c5, n
    return best_h, best_c5, n
