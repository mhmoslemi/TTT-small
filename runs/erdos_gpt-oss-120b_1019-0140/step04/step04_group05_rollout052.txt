def simulated_annealing(h0, time_limit, rng):
    start = time.time()
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start_temp = 0.2
    end_temp = 0.001
    max_iter = 2000000  # set large
    for it in range(max_iter):
        if time.time() - start >= time_limit:
            break
        # temperature decay
        temp = start_temp * ((end_temp / start_temp) ** (it / max_iter))
        # Choose move
        if rng.random() < 0.3:
            # swap
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
            if h[i] == h[j]:
                continue
            # swap
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            # acceptance
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                if temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                    # note: best_score unchanged
                else:
                    # revert swap
                    h[i], h[j] = h[j], h[i]
        else:
            # transfer
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            # scale delta by temperature
            scale = temp / start_temp
            delta = max_delta * rng.random() * scale
            if delta <= 0:
                continue
            # apply
            h[i] -= delta
            h[j] += delta
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                if temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta
        # periodic reproject to correct numeric drift
        if (it & 1023) == 0:
            # optionally reproject
            # small drift might accumulate
            h = project_onto_box_sum(h, n/2.0)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def elapsed():
        return time.time() - start_time
    def time_left():
        return max(0.0, budget_s - elapsed())
    # target bound
    target = 0.3808
    # desired resolution
    n = 512
    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # default to balanced binary random
        base = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n//2, replace=False)
        base[idx] = 1.0
    # upsample to n if needed
    if base.shape[0] != n:
        base = upsample(base, n)
    # project onto sum constraint
    base = project_onto_box_sum(base, n/2.0)
    best_h = base
    best_score = compute_c5(best_h)
    # warm-up SA on best_h using portion of time
    warm_time = min(0.2 * budget_s, time_left())
    if warm_time > 5.0:
        cand_h, cand_score = simulated_annealing(best_h, warm_time, rng)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
    # main loop
    while time_left() > 5.0:
        # decide starting solution
        if rng.random() < 0.5:
            # perturb best_h slightly
            h0 = best_h.copy()
            # apply few random transfers
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n-1)
                if j >= i:
                    j += 1
                max_delta = min(h0[i], 1.0 - h0[j])
                if max_delta > 0:
                    delta = max_delta * rng.random() * 0.2
                    h0[i] -= delta
                    h0[j] += delta
            # ensure feasibility
            h0 = project_onto_box_sum(h0, n/2.0)
        else:
            # random feasible vector
            h0 = rng.random(n)
            h0 = project_onto_box_sum(h0, n/2.0)
        # allocate time for this run
        alloc = min(8.0, time_left())
        cand_h, cand_score = simulated_annealing(h0, alloc, rng)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
            if best_score <= target:
                break
        # occasionally test random binary pattern
        if rng.random() < 0.1:
            # balanced binary vector
            idx = rng.choice(n, size=n//2, replace=False)

                best_h, best_score = h_bin, sc
                if best_score <= target:

    # final projection and scoring
    best_h = project_onto_box_sum(best_h, n/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n

def upsample(arr, new_n):
    """Linear interpolation to new_n points preserving domain [0,2)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    # coordinate arrays in [0,1) normalized
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo≤x_i≤hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on a uniform grid."""
    n = h.shape[0]
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1].astype(np.float64)
    size = 2 * n
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    max_val = conv[:2 * n - 1].max()
    return max_val * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points on [0,2)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def simulated_annealing(h0, time_limit, rng):
    """Projected SA preserving box and sum constraints."""
    start = time.time()
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start_temp = 0.2
    end_temp = 0.001
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        temp = start_temp * ((end_temp / start_temp) ** frac)
        # move selection
        if rng.random() < 0.3:
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                else:
                    h[i], h[j] = h[j], h[i]
        else:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:

            scale = temp / start_temp

            if delta <= 0.0:

            h[i] -= delta
            h[j] += delta
            new_score = compute_c5(h)
            if new_score < cur_score:
