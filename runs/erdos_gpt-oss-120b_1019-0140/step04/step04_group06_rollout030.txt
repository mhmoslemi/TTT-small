def local_search_binary(n, rng, time_left):
    # initialize random balanced binary array
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1
    # compute cyclic overlap using FFT
    fft_h = np.fft.rfft(h)
    overlap = np.fft.irfft(fft_h * np.conj(fft_h), n=n).round().astype(int)
    # overlap[0] = n//2
    min_overlap = overlap.min()
    # We'll keep best h and best min_overlap
    best_h = h.copy()
    best_min = min_overlap
    # iterative improvement
    while time_left() > 0.5:
        # sample a number of random swaps
        for _ in range(20):
            i = np.where(h==1)[0][rng.integers(h.sum())]   # pick random 1
            j = np.where(h==0)[0][rng.integers(n - h.sum())] # random 0
            # evaluate delta
            # compute delta overlap for all shifts k: delta = -h[(i+k)%n] + h[(j+k)%n]
            # That's like negative of h shifted by -i plus h shifted by -j.
            # we can compute new_overlap = overlap + delta
            # compute delta array:
            # using numpy roll
            delta = -np.roll(h, -i) + np.roll(h, -j)
            new_overlap = overlap + delta
            new_min = new_overlap.min()
            if new_min > min_overlap:
                # accept swap
                h[i] = 0
                h[j] = 1
                overlap = new_overlap
                min_overlap = new_min
                if new_min > best_min:
                    best_min = new_min
                    best_h = h.copy()
                break   # break inner for loop to restart sampling
        else:
            # if no improving swap after attempts, break outer
            break
    # after loop, compute C5 bound
    c5 = (n/2 - best_min) * (2.0 / n)
    return best_h.astype(float), c5, n

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project vector v onto the box [lo,hi] and onto the hyperplane sum=v==s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure the interval brackets the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, h.dtype)
            term2 = np.empty(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, h.dtype)
            term2 = np.empty(n, h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def optimize_at_scale(h0, n, rng, time_left):
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    K = 5
    no_improve = 0
    max_no_improve = 5000
    while time_left() > 4.0 and no_improve < max_no_improve:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = np.max(corr) * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # top‑K shifts
        if K >= corr.size:
            idx = np.arange(corr.size)
        else:
            idx = np.argpartition(corr, -K)[-K:]
        shifts = (idx - (n - 1)).tolist()
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.1 / np.sqrt(t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, target_sum)
    return best_h, best_score

def hill_climb(h, rng, time_left):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while time_left() > 0.5:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.uniform(0.05, 0.2)
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    return best_h, best_score

def upsample_by_factor(arr, factor):
    if factor == 1:
        return arr.copy()
    return np.repeat(arr, factor)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # base pattern
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.zeros(64, dtype=float)
        base[rng.choice(64, 32, replace=False)] = 1.0
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]
    for scale in scales:
        if base.shape[0] != scale:
            factor = scale // base.shape[0]
            base = upsample_by_factor(base, factor)
            base = project_onto_box_sum(base, scale / 2.0)

        # number of random restarts at this scale
        restarts = 2 if scale <= 256 else 1
        for _ in range(restarts):
            if time_left() < 5.0:
                break
            h0 = base + rng.normal(scale=0.02, size=scale)
            h0 = project_onto_box_sum(h0, scale / 2.0)
            h_opt, sc_opt = optimize_at_scale(h0, scale, rng, time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt.copy()

        # brief local hill‑climb refinement
        if time_left() > 2.0:
            h_hc, sc_hc = hill_climb(best_h.copy(), rng, time_left)
            if sc_hc < best_score:
                best_score, best_h = sc_hc, h_hc.copy()
            base = best_h.copy()

        if best_score <= 0.3808:
            break

    # final extensive random search at the finest resolution obtained
    final_n = best_h.shape[0]
    while time_left() > 1.0:
        h_rand = np.zeros(final_n, dtype=float)
        h_rand[rng.choice(final_n, final_n // 2, replace=False)] = 1.0
        h_rand = project_onto_box_sum(h_rand, final_n / 2.0)
        h_opt, sc_opt = optimize_at_scale(h_rand, final_n, rng, time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.shape[0] / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.shape[0]
