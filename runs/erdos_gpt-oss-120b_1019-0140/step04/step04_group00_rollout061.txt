def compute_gradient_smooth_max(h, alpha):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation unscaled
    h_rev = h[::-1]
    Ncorr = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=Ncorr)
    fft_one = np.fft.rfft(1.0 - h, n=Ncorr)
    conv = np.fft.irfft(fft_rev * fft_one, n=Ncorr)
    c = conv[:2 * n - 1]  # unscaled correlation

    max_c = c.max()
    # compute weight
    exp_arg = alpha * (c - max_c)
    # to avoid overflow/underflow extremes, cap max exponent maybe?
    w = np.exp(exp_arg)  # shape (2n-1)
    W = w.sum()

    # convolution w * h
    Nconv = 1 << ((3 * n - 2).bit_length())
    fft_h = np.fft.rfft(h, n=Nconv)
    fft_w = np.fft.rfft(w, n=Nconv)
    conv_w_h = np.fft.irfft(fft_w * fft_h, n=Nconv)
    # term2 weighted
    term2 = conv_w_h[n - 1 : n - 1 + n]  # length n

    # w_rev convolution
    w_rev = w[::-1]
    fft_wrev = np.fft.rfft(w_rev, n=Nconv)
    conv_wrev_h = np.fft.irfft(fft_wrev * fft_h, n=Nconv)
    term1 = conv_wrev_h[2 * n - 2 : 2 * n - 2 + n]  # length n

    grad = dx * (W - term1 - term2) / W
    return grad

def optimize_h_adam(h0, time_left, max_iters=4000, target_sum=None):
    n = h0.shape[0]
    if target_sum is None:
        target_sum = n / 2.0
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_val = compute_c5_fast(h)
    # Adam state
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha_start = 5.0
    alpha_end = 80.0
    # optional base learning rate
    base_lr = 0.3  # tweak
    for it in range(1, max_iters + 1):
        if time_left() < 0.2:
            break
        # schedule alpha
        frac = it / max_iters
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        grad = compute_gradient_smooth_max(h, alpha)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = base_lr / np.sqrt(it)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        val = compute_c5_fast(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            # early exit if good
            if best_val <= 0.3808:
                return best_h, best_val
        h = h_new
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    # try n=256 first
    n = 256
    target_sum = n / 2.0
    # Build initial candidates
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # ensure correct size, use resample if needed
        if init.size != n:
            # resample via interpolation over [0,1)
            x_old = np.linspace(0, 1, init.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        candidates.append(project_onto_box_sum(init, target_sum))
    # Add thue-morse binary
    def thue_morse_binary(n):
        # generate using bit parity
        seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
        return seq
    candidates.append(thue_morse_binary(n))
    # Add random binary balances
    for _ in range(2):
        h_bin = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, size=n//2, replace=False)
        h_bin[ones_idx] = 1.0
        candidates.append(h_bin)
    # Add random real vectors projected
    for _ in range(3):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))
    # Optimization loop over candidates
    best_h = None
    best_val = np.inf
    # allocate time for each candidate: maybe dynamic
    for h0 in candidates:
        if time_left() < 5.0:
            break
        # allocate some max_iters based on time left: cap at 5000
        # for fairness, keep max_iters like 3000
        max_iters = 3000
        # Reduce max_iters if time limited
        if time_left() < 30.0:
            max_iters = max(200, int(max_iters * time_left() / 30.0))
        h_opt, v_opt = optimize_h_adam(h0, time_left, max_iters=max_iters, target_sum=target_sum)
        if v_opt < best_val:
            best_val = v_opt
            best_h = h_opt.copy()
            if best_val <= 0.3808:
                return best_h, best_val, n
    # After initial candidates, try perturbations around best
    if best_h is not None:
        # generate perturbations
        for _ in range(5):
            if time_left() < 5.0:
                break
            # small Gaussian noise (mean 0, std 0.05)
            noise = rng.normal(0.0, 0.05, n)
            h_pert = best_h + noise
            # clip and project
            h_pert = project_onto_box_sum(h_pert, target_sum)
            h_opt, v_opt = optimize_h_adam(h_pert, time_left, max_iters=2000, target_sum=target_sum)
            if v_opt < best_val:
                best_val = v_opt
                best_h = h_opt.copy()
                if best_val <= 0.3808:
                    return best_h, best_val, n
    # If best still not <= target and time left allow, upscale to n=512
    if time_left() > 50.0 and best_h is not None:
        # Upsample to double resolution via interpolation
        new_n = n * 2
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, new_n, endpoint=False)
        best_res = np.interp(x_new, x_old, best_h)
        # Project onto box sum for new_n
        target_sum_new = new_n / 2.0
        best_res = project_onto_box_sum(best_res, target_sum_new)
        # Optimize at higher resolution
        h_opt, v_opt = optimize_h_adam(best_res, time_left, max_iters=4000, target_sum=target_sum_new)
        if v_opt < best_val:
            best_val = v_opt
            best_h = h_opt.copy()
            n = new_n
            if best_val <= 0.3808:
                return best_h, best_val, n
    # final ensure projection and compute best C5
    if best_h is None:
        # fallback uniform
        best_h = project_onto_box_sum(np.full(n, 0.5), target_sum)
        best_val = compute_c5_fast(best_h)
    else:
        best_h = project_onto_box_sum(best_h, n/2.0)
        best_val = compute_c5_fast(best_h)
    return best_h, best_val, n

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def compute_gradient_smooth_max(h, alpha):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation (unscaled) via FFT
    h_rev = h[::-1]
    Ncorr = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=Ncorr)
    fft_one = np.fft.rfft(1.0 - h, n=Ncorr)
    conv_corr = np.fft.irfft(fft_rev * fft_one, n=Ncorr)
    c = conv_corr[:2 * n - 1]                     # length 2n-1
    max_c = c.max()
    w = np.exp(alpha * (c - max_c))               # weights, max weight = 1
    W = w.sum()
    # convolution for term2: w * h
    Nconv = 1 << ((3 * n - 2).bit_length())
    fft_h = np.fft.rfft(h, n=Nconv)
    fft_w = np.fft.rfft(w, n=Nconv)
    conv_w_h = np.fft.irfft(fft_w * fft_h, n=Nconv)
    term2 = conv_w_h[n - 1 : n - 1 + n]
    # convolution for term1: w_rev * h
    w_rev = w[::-1]
    fft_wrev = np.fft.rfft(w_rev, n=Nconv)
    conv_wrev_h = np.fft.irfft(fft_wrev * fft_h, n=Nconv)
    term1 = conv_wrev_h[n - 1 : n - 1 + n]
    grad = dx * (W - term1 - term2) / W
    return grad

def optimize_h_adam(h0, time_left, max_iters=4000, target_sum=None):
    n = h0.shape[0]
    if target_sum is None:
        target_sum = n / 2.0
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_val = compute_c5_fast(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.3
    alpha_start = 5.0
    alpha_end = 80.0
    for it in range(1, max_iters + 1):
        if time_left() < 0.2:
            break
        frac = it / max_iters
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        grad = compute_gradient_smooth_max(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = base_lr / np.sqrt(it)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target_sum)
        val = compute_c5_fast(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            if best_val <= 0.3808:
                return best_h, best_val
        h = h_new
    return best_h, best_val

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
