def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    start_time = time.time()

    def time_left():
         return max(0.0, budget_s - (time.time() - start_time))

    # Load initial h values if provided else use thue_morse 64
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # use thue_morse
        base = np.array([ (bin(i).count('1') & 1 ) for i in range(64)], dtype=float)

    base = project_onto_box_sum(base, base.shape[0]/2.0)

    # Upsample progressively to final n = 512
    target_n = 512
    if base.shape[0] != target_n:
        base = upsample(base, target_n)
        base = project_onto_box_sum(base, target_n/2.0)

    current = base.copy()
    cur_score = compute_c5(current)
    best_h = current.copy()
    best_score = cur_score

    # allow a few phases: e.g., 0.6 budget for SA, remaining for random restarts and refinement

    sa_budget = 0.6 * budget_s
    # but ensure at least 30 seconds for refinement
    if sa_budget > budget_s - 30:
        sa_budget = budget_s - 30

    # Do SA
    h_sa, score_sa = simulated_annealing_improved(current, sa_budget, rng)
    if score_sa < best_score:
        best_score = score_sa
        best_h = h_sa.copy()
    current = h_sa
    cur_score = score_sa

    # Now refinement: use local moves with low temperature (downhill only) and restarts until time runs out
    # We'll loop while time left > some threshold:
    while time_left() > 2.0:
        # random restart: generate random binary balanced vector
        n = best_h.size
        h_rand = random_balanced_binary(n, rng)
        # Optionally apply a few local improvements (hill climbing)
        h_rand, rand_score = hill_climb(h_rand, time_left() * 0.5, rng)
        if rand_score < best_score:
            best_score = rand_score
            best_h = h_rand.copy()
        # also try small perturbations around best_h
        h_pert = best_h.copy()
        # perform some small moves
        h_pert, pert_score = hill_climb(h_pert, min(5.0, time_left()), rng)
        if pert_score < best_score:
            best_score = pert_score
            best_h = h_pert.copy()
        # break if we have reached target bound
        if best_score <= 0.3808:
            break

    # final projection and compute best_score again
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size

def hill_climb(h0, time_limit, rng):
    h = h0.copy()
    cur_score = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
         # attempt a transfer move with random i,j and delta; accept only if improves.
         i = rng.integers(len(h))
         j = rng.integers(len(h)-1)
         if j >= i: j += 1
         max_delta = min(h[i], 1.0 - h[j])
         if max_delta <= 0: continue
         delta = max_delta * rng.random() * 0.5
         h_candidate = h.copy()
         h_candidate[i] -= delta
         h_candidate[j] += delta
         new_score = compute_c5(h_candidate)
         if new_score < cur_score:
             h = h_candidate
             cur_score = new_score
         # optionally also try swap move
         if rng.random() < 0.2:
             i2 = rng.integers(len(h))
             j2 = rng.integers(len(h))
             while j2 == i2: j2 = rng.integers(len(h))
             h_candidate = h.copy()
             h_candidate[i2], h_candidate[j2] = h_candidate[j2], h_candidate[i2]
             new_score = compute_c5(h_candidate)
             if new_score < cur_score:
                 h = h_candidate
                 cur_score = new_score
    return h, cur_score

def simulated_annealing_improved(h0, time_limit, rng):

    n = h0.shape[0]
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()

    # Estimate initial temperature:
    # Do 200 random moves and record delta costs
    deltas = []
    for _ in range(200):
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_temp = h.copy()
        h_temp[i] -= delta
        h_temp[j] += delta
        deltas.append(abs(compute_c5(h_temp) - cur_score))
    if deltas:
        avg_delta = np.mean(deltas)
        init_temp = max(avg_delta, 1e-6) * 4.0  # multiply for acceptance
    else:
        init_temp = 0.001

    # SA loop
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        fraction = elapsed / time_limit
        temp = init_temp * (1.0 - fraction)  # linear cooling to 0

        # choose move type: weighted
        if rng.random() < 0.75:
            # Transfer
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            # scaling delta based on temperature: early bigger steps
            scale = 0.5 + 0.5 * (1 - fraction)  # start 1, end 0.5?
            delta = max_delta * rng.random() * scale
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # Swap
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        new_score = compute_c5(h_new)
        # Acceptance
        if new_score < cur_score or (temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        # Also break if best_score <= target
        # we cannot refer to target globally? Define constant TARGET=0.3808
        if best_score <= 0.3808:
            break

    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = np.clip(v - lam_mid, lo, hi).sum()
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def simulated_annealing_improved(h0, time_limit, rng):
    n = h0.size
    h = h0.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur
    start = time.time()
    # estimate initial temperature
    deltas = []
    for _ in range(200):
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_tmp = h.copy()
        h_tmp[i] -= delta
        h_tmp[j] += delta
        deltas.append(abs(compute_c5(h_tmp) - cur))
    init_temp = max(np.mean(deltas) if deltas else 0.0, 1e-6) * 4.0

    TARGET = 0.3808
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = init_temp * (1.0 - frac)
        if rng.random() < 0.75:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            scale = 0.5 + 0.5 * (1.0 - frac)
            delta = max_delta * rng.random() * scale
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        new = compute_c5(h_new)
        if new < cur or (temp > 0 and rng.random() < np.exp(-(new - cur) / temp)):
            h = h_new
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
                if best <= TARGET:
                    break
    return best_h, best

def hill_climb(h0, time_limit, rng):
    h = h0.copy()
    cur = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        i = rng.integers(h.size)
        j = rng.integers(h.size - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_candidate = h.copy()
            h_candidate[i] -= delta
            h_candidate[j] += delta
            new = compute_c5(h_candidate)
            if new < cur:
                h = h_candidate
                cur = new
        if rng.random() < 0.2:
            i2 = rng.integers(h.size)
            j2 = rng.integers(h.size)
            while j2 == i2:
                j2 = rng.integers(h.size)
            h_candidate = h.copy()
            h_candidate[i2], h_candidate[j2] = h_candidate[j2], h_candidate[i2]
            new = compute_c5(h_candidate)
            if new < cur:
                h = h_candidate
                cur = new
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.array([float((bin(i).count('1') & 1)) for i in range(64)], dtype=float)

    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    target_n = 512
    if base.shape[0] != target_n:
        base = upsample(base, target_n)
        base = project_onto_box_sum(base, target_n / 2.0)

    current = base.copy()
    best_h = current.copy()
    best_score = compute_c5(best_h)

    # allocate majority of time to SA, leave at least 30s for refinement
    sa_time = min(0.6 * budget_s, max(0.0, budget_s - 30.0))
    if time_left() > sa_time:
        h_sa, sc_sa = simulated_annealing_improved(current, sa_time, rng)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa.copy()
        current = h_sa
    # refinement phase
    while time_left() > 2.0:
        n = best_h.size
        # random balanced start
        h_rand = random_balanced_binary(n, rng)
        h_rand, sc_rand = hill_climb(h_rand, min(5.0, time_left() * 0.5), rng)
        if sc_rand < best_score:
            best_score = sc_rand
            best_h = h_rand.copy()
        # small perturbation of best
        h_pert = best_h.copy()
        h_pert, sc_pert = hill_climb(h_pert, min(5.0, time_left()), rng)
        if sc_pert < best_score:
            best_score = sc_pert
            best_h = h_pert.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
