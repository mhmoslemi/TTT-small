import cvxpy as cp

def solve_milp(n, time_limit):
    h = cp.Variable(n, integer=True) # boolean but integer
    constraints = [h >= 0, h <= 1, cp.sum(h) == n // 2]

    t = cp.Variable(integer=True)
    constraints.append(t >= 0)

    for k in range(n):
        # length len = n - k
        length = n - k
        if length <= 0:
            continue
        # y_{i,k} for i=0..length-1
        y = cp.Variable(length, integer=True)
        constraints.append(y >= 0)
        constraints.append(y <= 1)
        # y >= h[i] - h[i+k]
        constraints.append(y >= h[:length] - h[k:k+length])
        # y <= h[i]
        constraints.append(y <= h[:length])
        # y <= 1 - h[i+k]
        constraints.append(y <= 1 - h[k:k+length])
        # sum(y) <= t
        constraints.append(cp.sum(y) <= t)

    # objective minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    # solve with time limit using CBC
    prob.solve(solver=cp.CBC, time_limit=time_limit, warm_start=True, verbose=False)

    # after solve, extract values
    h_opt = np.rint(h.value).astype(float) # should be 0/1
    t_opt = int(round(t.value)) if t.value is not None else None
    return h_opt, t_opt

def solve_milp(n, time_limit):
    # Setup problem
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n//2, t >= 0]
    for k in range(n):
        length = n - k
        if length <= 0:
            continue
        y = cp.Variable(length, boolean=True)
        constraints += [y >= h[:length] - h[k:k+length],
                        y <= h[:length],
                        y <= 1 - h[k:k+length],
                        cp.sum(y) <= t]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit, warm_start=True, verbose=False)
    except Exception:
        return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.USER_LIMIT]:
        # if time limit reached maybe status = cp.USER_LIMIT
        pass
    h_opt = h.value
    if h_opt is None:
        return None, None
    # Round h_opt
    h_opt = np.rint(h_opt).astype(float)
    # Ensure sum correct (maybe adjust)
    # Project onto constraints for safety
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    # Compute t value from variable
    t_val = t.value
    if t_val is None:
        # compute t from h_opt
        t_val = max(np.correlate(h_opt, 1-h_opt, mode='full')) * (n/2.0)
        # Actually need integer count
    return h_opt, t_val

def simulated_annealing2(h0, time_limit, rng, target=None):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # Determine number of iterations based on time left
    # We'll just loop until time limit
    # Temperature schedule: exponential decay
    init_temp = 0.01
    # We'll compute total_iters approx 1e6
    total_iters = int(1e7)  # large
    it = 0
    while time.time() - start < time_limit:
        # temperature
        temp = init_temp * (0.9995 ** it)  # decay factor
        # could also clamp
        # Choose move type
        if rng.random() < 0.5:
            # swap move if binary-like
            # find indices with h > 0.5 and < 0.5
            i_candidates = np.where(h > 0.5)[0]
            j_candidates = np.where(h < 0.5)[0]
            if i_candidates.size == 0 or j_candidates.size == 0:
                continue
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
            # swap values
            new_h = h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
        else:
            # mass transfer move
            n = h.size
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            # scale delta based on temperature maybe
            delta = max_delta * rng.random() * (temp / init_temp)
            if delta <= 1e-12:
                continue
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < cur_score or (temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))):
            h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                if target is not None and best_score <= target:
                    break
        it += 1
    return best_h, best_score

def solve_milp(n, time_limit):
    # Build model
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n//2, t >= 0]

        y = cp.Variable(length, boolean=True)
        # constraints

        constraints.append(y <= h[:length])
        constraints.append(y <= 1 - h[k:k+length])
        constraints.append(cp.sum(y) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)

    h_opt = np.rint(h.value).astype(float)
    t_opt = t.value
    return h_opt, t_opt

def simulated_annealing_binary_fast(h0, time_limit, rng, target=None):
    h = h0.copy()
    n = h.shape[0]
    # initial corr
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = np.max(corr)
    cur_score = cur_max * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.2
    it = 0
    while time.time() - start < time_limit:
        temp = init_temp * (0.9995 ** it)
        # choose random i with h[i]==1, j with h[j]==0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute new corr using delta
        # compute delta as described
        # optimization: precompute pos2 = np.arange(n-1, -1, -1) outside loop
        # We'll define pos2 = np.arange(n-1, -1, -1) once.
        delta = np.zeros_like(corr)
        # i first
        start_k = -i
        end_k = n - i
        idx_start = (n-1) + start_k
        idx_end = (n-1) + end_k
        delta[idx_start:idx_end] -= (1 - h[i+np.arange(0, n - i)])  # i + k = pos
        # i second
        start_k2 = i - (n - 1)
        end_k2 = i + 1
        idx_start2 = (n-1) + start_k2
        idx_end2 = (n-1) + end_k2
        delta[idx_start2:idx_end2] += h[pos2]  # pos2 length n
        # j first
        start_k = -j
        end_k = n - j
        idx_start = (n-1) + start_k
        idx_end = (n-1) + end_k
        delta[idx_start:idx_end] += (1 - h[j+np.arange(0, n - j)])
        # j second
        start_k2 = j - (n - 1)
        end_k2 = j + 1
        idx_start2 = (n-1) + start_k2
        idx_end2 = (n-1) + end_k2
        delta[idx_start2:idx_end2] -= h[pos2]
        new_corr = corr + delta
        new_max = np.max(new_corr)
        new_score = new_max * (2.0 / n)
        # acceptance
        if new_score < cur_score or (temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp)):
            # accept
            h[i], h[j] = 0.0, 1.0
            corr = new_corr
            cur_score = new_score
            cur_max = new_max
            if new_score < best_score:
                best_h = h.copy()
                best_score = new_score
                if target is not None and best_score <= target:
                    break
        it += 1
    return best_h, best_score

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection onto {lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def solve_milp(n, time_limit):
    """Mixed‑integer linear program for binary h with minimal max overlap."""
    if cp is None:
        return None, None
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n // 2, t >= 0]

    for k in range(n):
        length = n - k
        if length == 0:
            continue
        y = cp.Variable(length, boolean=True)
        constraints.append(y >= h[:length] - h[k:k + length])
        constraints.append(y <= h[:length])
        constraints.append(y <= 1 - h[k:k + length])
        constraints.append(cp.sum(y) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit, warm_start=True, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, warm_start=True, verbose=False)
        except Exception:
            return None, None
    if h.value is None:
        return None, None
    h_opt = np.rint(np.array(h.value)).astype(float)
    # ensure exact sum
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_val = t.value
    return h_opt, t_val


def simulated_annealing_binary(h0, time_limit, rng, target=None):
    """SA with swap moves on binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_score = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.2
    it = 0
    while time.time() - start < time_limit:
        temp = init_temp * (0.9995 ** it)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute new correlation via full recompute (n small)
        h[i], h[j] = 0.0, 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_score = np.max(new_corr) * (2.0 / n)
        accept = False
        if new_score < cur_score:
            accept = True
        elif temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
            accept = True
        if accept:
            corr = new_corr
            cur_score = new_score
            if new_score < best_score:
                best_h = h.copy()
                best_score = new_score
                if target is not None and best_score <= target:
                    break
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
        it += 1
    return best_h, best_score

        return max(0.0, budget_s - (time.time() - start_time))

    target_c5 = 0.3808
    n = 128  # moderate size, good for MILP and fast SA

    # initial vector (use parent if present)
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != n:
            base = upsample(base, n)
    else:
        base = random_balanced_binary(n, rng)

    # turn initial vector into a binary start
    base = to_binary_by_top(base, n)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # --- attempt MILP ---
    milp_alloc = min(120, int(time_left() * 0.1))
    if milp_alloc > 5 and cp is not None:
        milp_h, _ = solve_milp(n, milp_alloc)
        if milp_h is not None:
            milp_score = compute_c5(milp_h)
            if milp_score < best_score:
                best_h = milp_h
                best_score = milp_score

    # --- stochastic optimisation ---
    while time_left() > 5:
        # allocate a chunk (max 60 s)
        chunk = min(60.0, time_left() * 0.6)
        # occasionally restart from a fresh random binary vector
        if rng.random() < 0.25:
            start_h = random_balanced_binary(n, rng)
        else:
            start_h = best_h.copy()
        cand_h, cand_score = simulated_annealing_binary(start_h, chunk, rng, target=target_c5)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
        # short greedy polishing
        if time_left() > 5:
            hp_h, hp_score = hill_climbing_binary(best_h, min(5.0, time_left()), rng, target=target_c5)
            if hp_score < best_score:
                best_h, best_score = hp_h, hp_score
        if best_score <= target_c5:
            break

    # final projection for safety
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
