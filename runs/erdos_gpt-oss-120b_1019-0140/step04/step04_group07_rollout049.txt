def improve_at_scale(h, n, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    # parameters
    max_bal_iters = int(4000 * n / 1024) + 100
    for i in range(max_bal_iters):
        if time_left() <= 0:
            break
        s, _ = get_worst_shift(best_h)
        # apply balance
        new_h = balance_shift(best_h, s, frac=0.5)
        cur_score = compute_c5(new_h)
        if cur_score < best_score:
            best_h = new_h
            best_score = cur_score

    if time_left() > 5:

def get_worst_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    # find shift(s)
    idxs = np.where(corr >= max_corr - 1e-14)[0]
    # pick first
    shift = idxs[0] - (n - 1)
    return shift, max_corr

def balance_shift(h, s, frac=0.5):
    n = h.shape[0]
    h_new = h.copy()
    if s >= 0:
        low = h_new[:n - s]
        high = h_new[s:]
        diff = high - low
        mask = diff > 1e-12
        if not np.any(mask):
            return h_new
        delta = np.minimum.reduce([diff[mask] * frac, 1 - low[mask], high[mask]])
        low[mask] += delta
        high[mask] -= delta
    else:
        s = -s
        low = h_new[s:]
        high = h_new[:n - s]
        diff = high - low  # here high is original at earlier indices
        mask = diff > 1e-12
        if not np.any(mask):
            return h_new
        delta = np.minimum.reduce([diff[mask] * frac, 1 - low[mask], high[mask]])
        low[mask] += delta
        high[mask] -= delta
    return h_new
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def get_worst_shift(h):
    """Return (shift, max_corr) where shift ∈ [-(n-1), n-1] gives maximal correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    idx = np.where(corr >= max_corr - 1e-14)[0][0]
    shift = int(idx - (n - 1))
    return shift, max_corr


def balance_shift(h, s, frac=0.5):
    """Transfer mass from higher to lower side across shift s (fraction of the difference)."""
    n = h.shape[0]
    h_new = h.copy()
    if s >= 0:
        low = h_new[: n - s]
        high = h_new[s:]
        diff = high - low
        mask = diff > 1e-12
        if not np.any(mask):
            return h_new
        delta = np.minimum.reduce([diff[mask] * frac, 1.0 - low[mask], high[mask]])
        low[mask] += delta
        high[mask] -= delta
    else:
        s = -s
        low = h_new[s:]
        high = h_new[: n - s]
        diff = high - low
        mask = diff > 1e-12
        if not np.any(mask):
            return h_new
        delta = np.minimum.reduce([diff[mask] * frac, 1.0 - low[mask], high[mask]])
        low[mask] += delta
        high[mask] -= delta
    return h_new


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with adaptive step size and momentum."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Transfer a small mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

    """Perform a suite of refinements at a given resolution."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    target_sum = n / 2.0

    # --- Balance worst shift -------------------------------------------------
    max_bal_iters = int(2000 * n / 512) + 200

    for _ in range(max_bal_iters):
