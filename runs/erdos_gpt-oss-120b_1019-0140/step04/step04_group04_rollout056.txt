def max_corr_of_h(h, N):
    # h is 0/1 float array length n
    # compute correlation with 1-h
    # Use FFT method.
    h_fft = np.fft.fft(h, N)
    w_fft = np.fft.fft(1.0 - h, N)
    conv = np.fft.ifft(h_fft * np.conj(w_fft)).real
    # Since we used zero padding to length N >= 2n-1, linear convolution is first 2n-1 entries.
    # The max over all shifts is max of conv[:2n-1] (since conv length N).
    # But we can just take max over conv[:2*n-1] (or whole conv).
    max_corr = int(np.rint(conv[:2*h.shape[0] - 1]).max())
    return max_corr

def adam_optimize(h0, max_iters, lr, rng, start_time, budget_s):
    # h0: initial vector
    n = h0.shape[0]
    target_sum = n/2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_max = compute_max_corr(h)
    for t in range(1, max_iters+1):
        # time check
        if time.time() - start_time >= budget_s:
            break
        grad = compute_subgradient(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # project onto box and sum
        h_new = project_onto_box_sum(h_new, target_sum)
        # compute max_corr
        cur_max = compute_max_corr(h_new)
        if cur_max < best_max:
            best_max, best_h = cur_max, h_new.copy()
        h = h_new
    return best_h, best_max

import time
import numpy as np

# cache for FFT lengths
_fft_len_cache = {}

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def upsample_repeat(arr, new_n):
    factor = new_n // arr.shape[0]
    return np.repeat(arr, factor)

def round_to_binary(v):
    n = v.size
    idx = np.argpartition(-v, n // 2)[: n // 2]
    bin_vec = np.zeros_like(v, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_max_fft(h):
    n = h.shape[0]
    N = _fft_len_cache.get(n)
    if N is None:
        N = 1 << ((2 * n - 1).bit_length())
        _fft_len_cache[n] = N
    h_fft = np.fft.fft(h, N)
    w_fft = np.fft.fft(1.0 - h, N)
    conv = np.fft.ifft(h_fft * np.conj(w_fft)).real
    corr = conv[: 2 * n - 1]
    corr_round = np.rint(corr)
    max_corr = int(corr_round.max())
    return corr_round, max_corr

def compute_max_corr(h):
    _, mc = compute_corr_max_fft(h)
    return mc

def compute_c5_from_max(n, max_corr):
    return (2.0 / n) * max_corr

def compute_subgradient(h):
    n = h.shape[0]
    corr, max_corr = compute_corr_max_fft(h)
    shifts = np.where(corr == max_corr)[0] - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def adam_optimize(h0, max_iters, lr, rng, start_time, budget_s):
    n = h0.shape[0]
    target_sum = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_max = compute_max_corr(best_h)
    t = 1
    while t <= max_iters:
        if (t % 1000) == 0:
            if time.time() - start_time >= budget_s:
                break
        grad = compute_subgradient(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, target_sum)
        cur_max = compute_max_corr(h_new)
        if cur_max < best_max:
            best_max, best_h = cur_max, h_new.copy()
        h = h_new
        t += 1
    return best_h, best_max
