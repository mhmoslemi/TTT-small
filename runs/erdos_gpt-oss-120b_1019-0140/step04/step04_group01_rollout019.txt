def binary_anneal(h0, rng, time_left, max_iters=None, T0=0.5, decay=0.99998, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    it = 0
    T = T0
    while True:
        if max_iters is not None and it >= max_iters: break
        if time_left() <= 0: break
        # Choose random 1 and 0 to swap (or randomly flip two bits)
        ones = np.where(h > 0.5)[0]  # binary h; should be exactly 0/1
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # probabilistic acceptance
            if T > Tmin and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
        it += 1
        # update temperature
        T = max(Tmin, T * decay)
    return best_h, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    """Full correlation of h with 1-h, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    """Maximum overlap integral C5 for step function h."""
    return np.max(compute_corr_fft(h))

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:

    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    return upsample(arr, new_n)

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=1500):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_fft(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

def binary_anneal(h0, rng, time_left, max_iters=None, T0=0.5, decay=0.999985, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    it = 0
    T = T0
    while True:
        if max_iters is not None and it >= max_iters:
            break
        if time_left() <= 0.0:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > Tmin and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        it += 1
        T = max(Tmin, T * decay)
    return best_h, best_score

    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    best_h = None
    best_score = np.inf

    # --- Phase 0: initial vector (if provided) ---
    init_arr = None
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)

    # --- Phase 1: small candidates (n=64) ---
    n_small = 64
    candidates = []
    if init_arr is not None:
        init_small = downsample(init_arr, n_small) if init_arr.size != n_small else init_arr.copy()
        init_small = project_onto_box_sum(init_small, n_small / 2.0)
        candidates.append(init_small)
    candidates.append(thue_morse(n_small))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(3):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    for cand in candidates:
        if time_left() < 4.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=1500)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Phase 2: medium resolution (n=256) ---
    n_mid = 256
    if best_h is None:
        best_h = upsample(thue_morse(n_small), n_mid)
    else:
        best_h = upsample(best_h, n_mid)
    best_h = project_onto_box_sum(best_h, n_mid / 2.0)
    sc_mid = compute_c5_fft(best_h)
    if sc_mid < best_score:
        best_score = sc_mid
        best_h = best_h.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    if time_left() > 2.0:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=1200)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    if time_left() > 5.0:
        h_ref, sc_ref = gradient_descent_momentum_with_time(best_h, time_left, max_iter=50000)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Phase 3: large resolution (n=512) ---
    n_large = 512
    best_h = upsample(best_h, n_large)
    best_h = project_onto_box_sum(best_h, n_large / 2.0)
    sc_large = compute_c5_fft(best_h)
    if sc_large < best_score:
        best_score = sc_large
        best_h = best_h.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    if time_left() > 2.0:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=800)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    if time_left() > 5.0:
        h_ref, sc_ref = gradient_descent_momentum_with_time(best_h, time_left, max_iter=40000)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Phase 4: binary conversion ---
    n = best_h.size
    k = n // 2
    idx = np.argsort(best_h)[::-1][:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5_fft(h_bin)
    if sc_bin < best_score:
        best_score = sc_bin
        best_h = h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # --- Phase 5: binary annealing (extensive) ---
    if time_left() > 2.0:
        remaining = time_left()
        max_iters = int(5e5) if remaining > 30 else int(1e5 * remaining)
        b_h, b_score = binary_anneal(best_h, rng, time_left,
                                    max_iters=max_iters,
                                    T0=0.5, decay=0.999985, Tmin=1e-6)
        if b_score < best_score:
            best_score = b_score
            best_h = b_h.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Phase 6: greedy binary hill climbing ---
    if time_left() > 1.0:
        max_iters = int(200000)
        if time_left() < 10.0:
            max_iters = int(50000 * time_left())
        b_h, b_score = binary_hill_climb(best_h, time_left, max_iters=max_iters, rng=rng)
        if b_score < best_score:
            best_score = b_score
            best_h = b_h.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Phase 7: random binary restarts with hill climbing ---
    while time_left() > 5.0:
        h_rand = random_binary_balanced(n, rng)
        b_h, b_score = binary_hill_climb(h_rand, time_left, max_iters=50000, rng=rng)
        if b_score < best_score:
            best_score = b_score
            best_h = b_h.copy()
            if best_score <= target:
                break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
