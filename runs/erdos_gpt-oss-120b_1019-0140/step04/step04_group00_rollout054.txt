def anneal_binary(h, rng, time_left, max_iters=None):
    n = h.shape[0]
    cur = compute_c5_fast(h)
    best_h = h.copy()
    best = cur
    # Determine max_iters based on time_left if not specified.
    if max_iters is None:
        # Rough estimate: each iteration takes ~ 0.000012 seconds? Let's allocate maybe time_left * 0.9 / (cost per iteration)
        # We'll just set max_iters = int(time_left() / 0.00001) as upper bound.
        max_iters = int(time_left() * 1000)  # 1000 it per second approx.
    T = 0.3
    T_min = 1e-5
    decay = 0.99996
    it = 0
    while it < max_iters and time_left() > 0.0:
        # choose random 1 and 0 positions
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best
```

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)[:2 * n - 1]
    return conv.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.concatenate([h[shift:], np.zeros(shift)])
        term2 = np.concatenate([np.zeros(shift), h[:n_s]])
    else:
        s = -shift
        n_s = n - s
        term1 = np.concatenate([np.zeros(s), h[:n_s]])
        term2 = np.concatenate([h[s:], np.zeros(s)])
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)[:2 * n - 1]
    mx = conv.max()
    idxes = np.where(conv == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h, best_s = h.copy(), compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_correlation(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)[:2 * n - 1]
    return conv

def reduce_max_overlap(h):
    n = h.size
    corr = compute_correlation(h)
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    if shift >= 0:
        i_range = np.arange(0, n - shift)
        cond_i = (h[i_range] == 1) & (h[i_range + shift] == 0)
        cond_j = (h[i_range] == 0) & (h[i_range + shift] == 1)
    else:
        s = -shift
        i_range = np.arange(s, n)
        cond_i = (h[i_range] == 1) & (h[i_range - s] == 0)
        cond_j = (h[i_range] == 0) & (h[i_range - s] == 1)
    i_candidates = i_range[cond_i]
    j_candidates = i_range[cond_j]
    if i_candidates.size and j_candidates.size:
        i_swap = i_candidates[0]
        j_swap = j_candidates[0]
        h[i_swap] = 0.0
        h[j_swap] = 1.0
        return True
    return False

def anneal_binary(h, rng, time_left, max_iters=None):
    n = h.shape[0]
    cur = compute_c5_fast(h)
    best_h, best = h.copy(), cur
    if max_iters is None:
        max_iters = int(time_left() * 1000) + 1000
    T, T_min, decay = 0.3, 1e-5, 0.99996
    it = 0
    while it < max_iters and time_left() > 0.0:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i, j = rng.choice(ones), rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best:
                best, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best

def improve_binary_h(h, time_left, rng):
    best_h, best_val = h.copy(), compute_c5_fast(h)
    # deterministic reduction
    while time_left() > 0.2:
        changed = reduce_max_overlap(h)
        if not changed:
            break
        cur = compute_c5_fast(h)
        if cur < best_val:
            best_val, best_h = cur, h.copy()
    # annealing fallback
    if time_left() > 0.5:
        h2, v2 = anneal_binary(h, rng, time_left)
        if v2 < best_val:
            best_h, best_val = h2, v2
    return best_h, best_val

def adjust_binary_sum(h, target_sum, rng):
    cur = int(h.sum())
    diff = int(target_sum - cur)
    if diff > 0:
        zeros = np.where(h == 0)[0]
        if zeros.size >= diff:
            h[rng.choice(zeros, size=diff, replace=False)] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        if ones.size >= -diff:
            h[rng.choice(ones, size=-diff, replace=False)] = 0.0
    return h

def optimize_candidate(h0, rng, time_left):
    n = h0.size
    target = n / 2.0
    h = h0.astype(float, copy=True)
    # ensure sum constraint
    if np.all((h == 0) | (h == 1)):
        h = adjust_binary_sum(h, target, rng)
        best_h, best_val = improve_binary_h(h, time_left, rng)
        return best_h, best_val
    else:
        h = project_onto_box_sum(h, target)
        best_h, best_val = gradient_descent_momentum(h, time_left, max_iter=200000)
        # try binary refinement
        h_bin = (best_h >= 0.5).astype(float)
        h_bin = adjust_binary_sum(h_bin, target, rng)
        if time_left() > 0.5:
            h_bin_opt, val_bin_opt = improve_binary_h(h_bin, time_left, rng)
            if val_bin_opt < best_val:
                return h_bin_opt, val_bin_opt
        return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))
    n = 256
    target_sum = n / 2.0
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            init = resample(init, n)
        candidates.append(init.copy())
    candidates.append(thue_morse_binary(n))
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))
    best_h, best_val = None, np.inf
    for h0 in candidates:
        if time_left() < 5.0:
            break
        h_opt, val_opt = optimize_candidate(h0, rng, time_left)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()
            if best_val <= 0.3808:
                return best_h, best_val, n
    if best_h is None:
        best_h = random_binary_balanced(n, rng)
        best_val = compute_c5_fast(best_h)
    # final polishing with annealing if time permits
    if time_left() > 1.0 and np.all((best_h == 0) | (best_h == 1)):
        h_final, v_final = anneal_binary(best_h, rng, time_left)
        if v_final < best_val:
            best_h, best_val = h_final, v_final
    best_h = project_onto_box_sum(best_h, target_sum)
    best_val = compute_c5_fast(best_h)
    return best_h, best_val, n
