def local_search(h, total_sum, time_limit, rng):
    n = h.shape[0]
    offset = n - 1
    pad = np.concatenate([np.zeros(offset), h, np.zeros(offset)])
    corr = np.correlate(h, h, mode='full')
    best_h = h.copy()
    best_min = np.min(corr)
    best_corr = corr.copy()
    start = time.time()
    init_temp = 0.006  # can tune
    # maybe schedule decays with elapsed fraction
    iteration = 0
    # pre-alloc maybe ell? Not needed.
    while time.time() - start < time_limit:
        # time based temp
        elapsed = time.time() - start
        fraction = elapsed / time_limit
        temp = init_temp * (1.0 - fraction)
        # sample indices
        i = rng.integers(n)
        # ensure j != i
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        # delta scaling: allow full move sometimes
        scale = min(1.0, (temp / init_temp) * 1.5 + 0.5)  # somewhat high
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            continue
        # compute slices
        i_start = i
        i_plus = pad[i_start : i_start + 2*n - 1]
        i_minus = pad[i_start + 2*(n-1) : i_start - 1 : -1]
        j_start = j
        j_plus = pad[j_start : j_start + 2*n - 1]
        j_minus = pad[j_start + 2*(n-1) : j_start - 1 : -1]
        delta_corr = delta * (j_plus + j_minus - i_plus - i_minus)
        cand_corr = corr + delta_corr
        cand_min = np.min(cand_corr)
        cur_min = np.min(corr)
        # desire larger min
        if cand_min > cur_min or (temp > 0 and rng.random() < np.exp((cand_min - cur_min) / temp)):
            # accept
            h[i] -= delta
            h[j] += delta
            pad[i + offset] = h[i]
            pad[j + offset] = h[j]
            corr = cand_corr
            # track best
            if cand_min > best_min:
                best_min = cand_min
                best_h = h.copy()
                best_corr = corr.copy()
        iteration += 1
        # periodic reproject to avoid drift
        if (iteration & 0xFFF) == 0:  # every 4096
            # reproject to box sum
            h = project_onto_box_sum(h, total_sum)
            pad[offset:offset+n] = h
            corr = np.correlate(h, h, mode='full')
            cur_min = np.min(corr)
            if cur_min > best_min:
                best_min = cur_min
                best_h = h.copy()
                best_corr = corr.copy()
    return best_h, best_corr

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def local_search(h, total_sum, time_limit, rng, init_temp=0.006):
    """
    Stochastic local search that tries to maximize the minimum autocorrelation.
    Returns the best vector found and its autocorrelation array.
    """
    n = h.shape[0]
    offset = n - 1
    pad = np.concatenate([np.zeros(offset), h, np.zeros(offset)])
    corr = np.correlate(h, h, mode='full')
    best_h = h.copy()
    best_corr = corr.copy()
    best_min = np.min(corr)
    start = time.time()
    it = 0
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        # temperature linearly decays to zero
        frac = (now - start) / time_limit
        temp = init_temp * (1.0 - frac)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue

        # scale move size with temperature, keep a small baseline
        scale = max(temp / init_temp, 0.05)
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            continue

        # slices for efficient delta computation
        i_start = i
        i_plus = pad[i_start : i_start + 2 * n - 1]
        i_minus = pad[i_start + 2 * (n - 1) : i_start - 1 : -1]
        j_start = j
        j_plus = pad[j_start : j_start + 2 * n - 1]
        j_minus = pad[j_start + 2 * (n - 1) : j_start - 1 : -1]

        delta_corr = delta * (j_plus + j_minus - i_plus - i_minus)
        cand_corr = corr + delta_corr
        cand_min = np.min(cand_corr)
        cur_min = np.min(corr)

        accept = False
        if cand_min > cur_min:
            accept = True
        elif temp > 0.0:
            # Metropolis acceptance for worsening moves
            if rng.random() < np.exp((cand_min - cur_min) / temp):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            pad[i + offset] = h[i]
            pad[j + offset] = h[j]
            corr = cand_corr
            if cand_min > best_min:
                best_min = cand_min
                best_h = h.copy()
                best_corr = corr.copy()

        it += 1
        # occasional re‑projection to correct numeric drift
        if (it & 0x3FFF) == 0:
            h = project_onto_box_sum(h, total_sum)
            pad[offset:offset + n] = h
            corr = np.correlate(h, h, mode='full')
            cur_min = np.min(corr)
            if cur_min > best_min:
                best_min = cur_min
                best_h = h.copy()
                best_corr = corr.copy()
    return best_h, best_corr
