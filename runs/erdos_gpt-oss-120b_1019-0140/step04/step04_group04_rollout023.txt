def project_onto_box_sum(v, s, lo=0., hi=1., max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust bounds for safety
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def simulated_annealing(h0, max_iter, rng, time_left):
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    # initial temperature as fraction of cur_score
    T0 = cur_score * 0.05  # 5% of current C5
    Tmin = cur_score * 0.0005
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        # proposal
        i, j = rng.integers(0, n, 2)
        while i == j:
            j = rng.integers(0, n)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        # delta scaled down early and later
        scale = 0.5 * (1 - it / max_iter) + 0.02  # decaying, min ~0.02
        delta = max_delta * rng.random() * scale
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = compute_c5(cur_h)
        T = T0 * (1 - it / max_iter) + Tmin
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_score

    iters_done = 0
    # loop with phases
    while time_left() > 5 and iters_done < total_iters:

        if rng.random() < 0.08 and time_left() > 10:

            idx = rng.choice(n, size=n//2, replace=False)
            h[idx] = 1.0

        if phase_iters < 1000:

        # simulated annealing phase
        h_sa, score_sa = simulated_annealing(h, phase_iters, rng, time_left)
        iters_done += phase_iters
        if score_sa < best_score:

        # short gradient refinement

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    n = h.shape[0]
    g_rev = (1.0 - h)[::-1]
    m = 2 * n - 1
    L = 1 << ((m - 1).bit_length())
    H = np.fft.fft(h, n=L)
    G = np.fft.fft(g_rev, n=L)
    conv = np.fft.ifft(H * G)[:m].real
    return np.max(conv) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h0, max_iter, rng, time_left):
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    T0 = cur_score * 0.05
    Tmin = cur_score * 0.0005
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        i, j = rng.integers(0, n, 2)
        while i == j:
            j = rng.integers(0, n)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        scale = 0.5 * (1 - it / max_iter) + 0.02
        delta = max_delta * rng.random() * scale
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = compute_c5(cur_h)
        T = T0 * (1 - it / max_iter) + Tmin
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_score

def optimize_at_resolution(h_start, n, rng, time_left):
    h = upsample(h_start, n)
    h = project_onto_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    total_iters = int(max(20000, time_left() * 30))
    iters_done = 0
    while time_left() > 5 and iters_done < total_iters:
        if rng.random() < 0.08 and time_left() > 10:
            h = np.zeros(n)
            idx = rng.choice(n, size=n // 2, replace=False)
            h[idx] = 1.0
        phase_iters = int(min(time_left() * 20, total_iters - iters_done, 40000))
        if phase_iters < 1000:
            break
        h_sa, score_sa = simulated_annealing(h, phase_iters, rng, time_left)
        iters_done += phase_iters
        if score_sa < best_score:
            best_score, best_h = score_sa, h_sa.copy()
        gd_iters = int(min(5000, time_left() * 5))
        if gd_iters >= 500:
            h_gd, score_gd = gradient_descent_momentum_custom(best_h,
                                                             n_iter=gd_iters,
                                                             step0=0.3,
                                                             beta=0.95,
                                                             rng=rng,
                                                             time_left=time_left)
            if score_gd < best_score:
                best_score, best_h = score_gd, h_gd.copy()
            h = h_gd
        else:
            h = best_h
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.fromiter(((bin(i).count('1') & 1) for i in range(64)), dtype=float)
        base = project_onto_box_sum(base, 32.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    for n in (128, 256, 512):
        if time_left() <= 8:
            break
        best_h, best_score = optimize_at_resolution(best_h, n, rng, time_left)
        if best_score <= 0.3808:
            break

    if time_left() > 10:
        final_iters = int(min(80000, time_left() * 30))
        best_h, best_score = gradient_descent_momentum_custom(best_h,
                                                             n_iter=final_iters,
                                                             step0=0.2,
                                                             beta=0.95,
                                                             rng=rng,
                                                             time_left=time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
