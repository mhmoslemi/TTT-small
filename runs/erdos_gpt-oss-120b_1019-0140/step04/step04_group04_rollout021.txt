def upsample_binary(arr, new_n):
    old_n = arr.shape[0]
    reps = new_n // old_n
    rem = new_n % old_n
    h = np.repeat(arr, reps)
    if rem > 0:
        h = np.concatenate([h, arr[:rem]])
    # If now sum not exactly new_n/2, adjust:
    target = new_n // 2
    current = int(h.sum())
    if current > target:
        # turn off some ones
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, size=current-target, replace=False)
        h[idx] = 0
    elif current < target:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(zeros, size=target-current, replace=False)
        h[idx] = 1
    return h.astype(float)
```

def compute_c5_fft(h):
    n = h.shape[0]
    # compute correlation using FFT:
    # pad to length 2*n
    N = 2*n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    # corr length N, linear cross-correlation; last element corresponds to shift n-1? We need only first 2*n-1 values (0..2n-2). The IRFFT returns array length N.

    # We'll take corr[:2*n-1]
    c = corr[:2*n-1]
    # compute max * (2/n)
    return np.max(c) * (2.0 / n)
```

def upsample_binary(arr, new_n, rng):
    old_n = arr.shape[0]
    reps = new_n // old_n
    rem = new_n % old_n
    h = np.repeat(arr, reps)
    if rem:
        h = np.concatenate([h, arr[:rem]])
    # adjust sum to target
    target = new_n // 2
    cur = int(h.sum())
    if cur > target:
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, size=cur - target, replace=False)
        h[idx] = 0
    elif cur < target:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(zeros, size=target - cur, replace=False)
        h[idx] = 1
    return h.astype(float)
```

def gradient_descent_refine(h0, time_left, rng):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 0
    # set total iterations limit based on time left
    max_iters = int(time_left() * 1500)  # estimate 1500 it/s
    beta = 0.95
    step0 = 0.3
    while it < max_iters and time_left() > 0.05:
        it += 1
        # compute correlation using FFT
        N = 2 * n
        H = np.fft.rfft(h, n=N)
        G = np.fft.rfft(1.0 - h, n=N)
        corr = np.fft.irfft(H * np.conj(G), n=N)[:2*n-1]
        max_corr = np.max(corr)
        # get shifts achieving max
        eps = 1e-12
        shift_idxs = np.where(corr >= max_corr - eps)[0]
        shifts = shift_idxs - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        if score <= compute_c5_fft(h) or rng.random() < 0.01:
            h = h_new
        # else keep old h

                best_h = h.copy()

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample_binary(arr, new_n, rng):
    old_n = arr.shape[0]
    reps = new_n // old_n
    rem = new_n % old_n
    h = np.repeat(arr, reps)
    if rem:
        h = np.concatenate([h, arr[:rem]])
    target = new_n // 2
    cur = int(h.sum())
    if cur > target:
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, size=cur - target, replace=False)
        h[idx] = 0
    elif cur < target:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(zeros, size=target - cur, replace=False)
        h[idx] = 1
    return h.astype(float)

def compute_gradients_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.ones_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[: n - shift] -= h[shift:]
            grad[shift:] -= h[: n - shift]
        else:
            s = -shift
            grad[s:] -= h[: n - s]
            grad[: n - s] -= h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def simulated_annealing_binary(h0, time_left, rng):
    h = h0.copy()
    n = h.shape[0]
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    max_iters = int(time_left() * 20000)
    max_iters = min(max_iters, 2_000_000)
    T0 = cur_score * 0.02
    T_end = cur_score * 0.0005
    no_improve = 0
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        T = T_end + (T0 - T_end) * (1 - it / max_iters)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        if no_improve >= 10000 and time_left() > 5:
            h = random_binary_balanced(n, rng)
            cur_score = compute_c5_fft(h)
            no_improve = 0
        if best_score <= 0.3808:
            break
    return best_h, best_score

def gradient_descent_refine(h0, time_left, rng):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 0
    max_iters = int(time_left() * 1500)
    max_iters = min(max_iters, 800_000)
    beta = 0.95
    step0 = 0.3
    score_h = best_score
    while it < max_iters and time_left() > 0.05:
        it += 1
        N = 2 * n
        H = np.fft.rfft(h, n=N)
        G = np.fft.rfft(1.0 - h, n=N)
        corr = np.fft.irfft(H * np.conj(G), n=N)[:2 * n - 1]
        max_corr = np.max(corr)
        eps = 1e-12
        shift_idxs = np.where(corr >= max_corr - eps)[0]
        shifts = shift_idxs - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score_new = compute_c5_fft(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
        if score_new <= score_h or rng.random() < 0.01:
            h = h_new
            score_h = score_new
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # Primary resolution
    n_target = 256
    if 'initial_h_values' in globals():

        base = np.rint(base).astype(float)

        h0 = random_binary_balanced(n_target, rng)

    # Binary simulated annealing
    best_h, best_score = simulated_annealing_binary(h0, time_left, rng)

    # Continuous refinement if time permits
    if time_left() > 10:
        best_h, best_score = gradient_descent_refine(best_h, time_left, rng)

    # If still not good enough and time remains, try a larger scale
    if time_left() > 30 and best_score > 0.3808:

        if score_big < best_score:
            best_h, best_score = h_big, score_big
        if time_left() > 10:
            best_h, best_score = gradient_descent_refine(best_h, time_left, rng)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
