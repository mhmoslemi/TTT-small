- momentum = np.zeros_like(h)

   it += 1

   corr = np.correlate(h, 1.0 - h, mode='full'); C_arr = corr * (2.0 / n); maxC = C_arr.max()

   # find active shifts where C_arr >= maxC - 1e-12

   # compute gradient via compute_gradients_for_shifts

   g = compute_gradients_for_shifts(h, shift_idxs)

   score = compute_c5(h_new)

- return best_h, best_score

def targeted_annealing(h0, max_time, rho=0.5, T0=0.01, T_final=1e-6):

- start = time.time()

- h = h0.copy(); n = len(h); target_sum = n/2

- cur_score = compute_c5(h)

- best_h, best_score = h.copy(), cur_score

- steps = 0

- while time.time() - start < max_time:

   steps += 1

   # compute C array

   corr = np.correlate(h, 1.0 - h, mode='full'); C_arr = corr * (2.0 / n)

   maxC = C_arr.max()

   # find active shifts

   idxes = np.where(C_arr >= maxC - 1e-12)[0]; if idxes.size == 0: break

   shift_idx = rng.choice(idxes)  # random among max

   s = shift_idx - (n - 1)

   # compute contributions

   cont = np.zeros_like(h)

   if s >= 0:

       # overlapping region: i from 0 to n-1-s

       cont[:n-s] = h[:n-s] * (1.0 - h[s:])

   else:

       s_neg = -s

       cont[s_neg:] = h[s_neg:] * (1.0 - h[:n-s_neg])

   # find source and target

   # source: high contribution and h > small

   src_candidates = np.where((cont > 0) & (h > 1e-8))[0]

   if src_candidates.size == 0:

       # fallback to random source

       src = rng.choice(np.where(h > 1e-8)[0]) if np.any(h > 1e-8) else None

   else:

       src = src_candidates[np.argmax(cont[src_candidates])]

   # target: low contribution and h < 1

   tgt_candidates = np.where((cont < 1e-8) & (h < 1 - 1e-8))[0]

   if tgt_candidates.size == 0:

       tgt = rng.choice(np.where(h < 1 - 1e-8)[0]) if np.any(h < 1) else None

   else:

       tgt = tgt_candidates[np.argmin(cont[tgt_candidates])]

   if src is None or tgt is None or src == tgt:

       # random transfer of small mass

       i, j = rng.choice(n, size=2, replace=False)

       max_move = min(h[i], 1.0 - h[j])

       if max_move <= 0: continue

       delta = max_move * rng.random() * rho

       h_new = h.copy(); h_new[i] -= delta; h_new[j] += delta

   else:

       max_move = min(h[src], 1.0 - h[tgt])

       if max_move <= 0: continue

       delta = max_move * rng.random() * rho

       h_new = h.copy(); h_new[src] -= delta; h_new[tgt] += delta

   # projection not needed due to delta definition ensures box constraints and sum preserved

   new_score = compute_c5(h_new)

   # temperature schedule: linear

   elapsed = time.time() - start

   remaining = max_time - elapsed

   # compute current T as linear decay based on steps within total allowed steps

   # Since we don't have total steps, implement T = max(T0 * (remaining / max_time), T_final)

   T = max(T0 * (remaining / max_time), T_final)

   if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):

       h = h_new; cur_score = new_score

       if cur_score < best_score:

           best_score = cur_score; best_h = h.copy()

- return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # n must be power of 2
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def subgradient_descent(h0, max_time, rng=None, step0=0.5, beta=0.9):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while True:
        if time.time() - start >= max_time:
            break
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        C_arr = corr * (2.0 / n)
        maxC = C_arr.max()
        idxes = np.where(C_arr >= maxC - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if (it % 2000 == 0) and (rng is not None):
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def targeted_annealing(h0, max_time, rng, rho=0.5, T0=0.01, T_final=1e-6):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    while True:
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        C_arr = corr * (2.0 / n)
        maxC = C_arr.max()
        idxes = np.where(C_arr >= maxC - 1e-12)[0]
        if idxes.size == 0:
            break
        shift_idx = rng.choice(idxes)
        s = shift_idx - (n - 1)
        cont = np.zeros_like(h)
        if s >= 0:
            cont[:n - s] = h[:n - s] * (1.0 - h[s:])
        else:
            sneg = -s
            cont[sneg:] = h[sneg:] * (1.0 - h[:n - sneg])
        src_candidates = np.where((cont > 0) & (h > 1e-8))[0]
        if src_candidates.size == 0:
            src = None
        else:
            src = src_candidates[np.argmax(cont[src_candidates])]
        tgt_candidates = np.where((cont < 1e-8) & (h < 1.0 - 1e-8))[0]
        if tgt_candidates.size == 0:
            tgt = None
        else:
            tgt = tgt_candidates[np.argmin(cont[tgt_candidates])]
        if src is None or tgt is None or src == tgt:
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 0:
                continue
            delta = max_move * rng.random() * rho
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            max_move = min(h[src], 1.0 - h[tgt])
            if max_move <= 0:
                continue
            delta = max_move * rng.random() * rho
            h_new = h.copy()
            h_new[src] -= delta
            h_new[tgt] += delta
        new_score = compute_c5(h_new)
        elapsed = time.time() - start
        remaining = max_time - elapsed
        T = max(T0 * (remaining / max_time), T_final)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h, cur_score = h_new, new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
    return best_h, best_score
