def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bracketing
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

    return np.max(corr) * (2.0 / n)

    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def max_iter_for_n(n):
    if n <= 64:
        return 20000
    if n <= 128:
        return 15000
    if n <= 256:
        return 12000
    return 8000

def step0_for_n(n):
    if n <= 128:
        return 0.8
    if n <= 256:
        return 0.5
    return 0.3

def gradient_descent_backtrack(h0, time_left, max_iter, step0, rng):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    eps_shift = 1e-12
    min_step = 1e-7
    stagnation = 0
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps_shift)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        step = step0 / math.sqrt(it)
        improved = False
        while step > min_step:
            h_new = project_onto_box_sum(h - step * grad, target_sum)
            new_score = compute_c5(h_new)
            if new_score < best_score - 1e-13:
                best_h, best_score = h_new.copy(), new_score
                h = h_new
                improved = True
                break
            step *= 0.5
        if not improved:
            stagnation += 1
            if stagnation >= 5:
                break
        else:
            stagnation = 0
        if it % 500 == 0:
            h = project_onto_box_sum(h + rng.normal(scale=0.01, size=n), target_sum)
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    target_sum = n / 2.0
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - target_sum
    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, target_sum)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def final_refine(h, time_left, rng):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    # optional upsample to larger size if time permits
    while time_left() > 30 and n < 768:
        new_n = min(768, n * 2)
        h_up = upsample(best_h, new_n)
        h_up = project_onto_box_sum(h_up, new_n / 2.0)
        max_it = int(min(8000, time_left() * 10))
        if max_it <= 0:
            break
        h_up, s_up = gradient_descent_backtrack(h_up, time_left, max_it,
                                                step0_for_n(new_n), rng)
        if s_up < best_score:
            best_h, best_score = h_up.copy(), s_up
            n = new_n
        else:
            break
    # local random moves
    target_sum = n / 2.0
    while time_left() > 2:
        # swap mass between two positions
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j:
            delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.01, 0.5)
            if delta > 0:
                h_new = best_h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_h, best_score = h_new, sc
        # Gaussian perturbation
        if rng.random() < 0.3:
            pert = rng.normal(scale=0.015, size=n)
            h_pert = best_h + pert
            h_pert = project_onto_box_sum(h_pert, target_sum)
            sc = compute_c5(h_pert)
            if sc < best_score:
                best_h, best_score = h_pert, sc
        # occasional random binary restart
        if rng.random() < 0.1:
            h_rand = random_binary_balanced(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_h, best_score = h_rand, sc
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)
    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)
    # resolution schedule
    for n in (64, 128, 256, 512):
        if time_left() <= 5:
            break
        # upsample current best to n
        if best_h.size != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)
        base_score = compute_c5(base)
        if base_score < best_score:
            best_h, best_score = base.copy(), base_score
        # gradient descent phase
        max_iters = min(max_iter_for_n(n), int(time_left() * 20))
        if max_iters > 0:
            h_ref, sc_ref = gradient_descent_backtrack(base, time_left,
                                                       max_iters,
                                                       step0_for_n(n), rng)
            if sc_ref < best_score:
                best_h, best_score = h_ref.copy(), sc_ref
        # SLSQP refinement
        if time_left() > 5:
            h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=300)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
        # occasional random restart with short GD
        if time_left() > 10:
            for _ in range(2):
                h_rand = random_binary_balanced(n, rng)
                h_rand = project_onto_box_sum(h_rand.astype(float), n / 2.0)
                max_it = min(max_iter_for_n(n) // 3, int(time_left() * 5))
                h_tmp, sc_tmp = gradient_descent_backtrack(h_rand, time_left,
                                                           max_it,
                                                           step0_for_n(n) * 0.7,
                                                           rng)
                if sc_tmp < best_score:
                    best_h, best_score = h_tmp.copy(), sc_tmp
    # final refinement (possible upsample, local moves)
    if time_left() > 2:
        best_h, best_score = final_refine(best_h, time_left, rng)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
