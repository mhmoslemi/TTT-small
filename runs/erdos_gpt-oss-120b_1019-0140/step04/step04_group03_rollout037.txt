def compute_grad_for_shift(h, shift):
    n = h.size
    # pad zeros on both ends for out-of-range indices
    # left term: h[i - shift] if i - shift >= 0 else 0
    if shift >= 0:
        left = np.empty_like(h)
        left[:shift] = 0.0
        left[shift:] = h[:-shift]  # h[i - shift]
        right = np.empty_like(h)
        right[:n-shift] = h[shift:]
        right[n-shift:] = 0.0
    else:
        s = -shift
        left = np.empty_like(h)
        left[:n-s] = h[s:]  # i - (-s) = i + s
        left[n-s:] = 0.0
        right = np.empty_like(h)
        right[:s] = 0.0
        right[s:] = h[:n-s]
    grad = 1.0 - left - right
    # Note indices where left and right both zeros get grad=1.
    return grad

def adam_optimize(h0, max_iters, lr, beta1, beta2, epsilon, rng, remaining_time_func):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    n = h.size
    target_sum = n/2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_score = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    for t in range(1, max_iters+1):
        # time check periodically
        if t % 500 == 0:
            if remaining_time_func() <= 0:
                break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # compute gradient for max shifts
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = h - step
        # optional small random perturbation per certain steps
        if t % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        # compute new score
        new_corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_score = np.max(new_corr) * dx
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # initialize lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # make sure the sum at bounds encloses target
    # compute f(lam) = sum clip(v - lam, lo, hi)
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        # shift lam_low further left
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):

            term1 = np.empty_like(h)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty_like(h)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def adam_optimize(h0, max_iters, lr, beta1, beta2, epsilon, rng, start_time, budget_s):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    dx = 2.0 / n
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    # precompute powers for efficiency? We can compute beta1_pow = beta1, etc.
    beta1_pow = beta1
    beta2_pow = beta2
    for t in range(1, max_iters + 1):
        if t % 500 == 0:
            if time.time() - start_time >= budget_s - 5.0:
                break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        # bias correction
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = h - step
        if t % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        new_score = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        h = h_new
        # update powers
        beta1_pow *= beta1
        beta2_pow *= beta2
    return best_h, best_score

def hill_climb_swaps(h, cur_score, n_swaps, rng, start_time, budget_s):
    n = h.size
    dx = 2.0 / n
    cur_h = h.copy()
    best_h = cur_h.copy()
    best_score = cur_score
    for _ in range(n_swaps):
        if time.time() - start_time >= budget_s - 5.0:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur_h[i] == cur_h[j]:
            continue
        new_h = cur_h.copy()
        new_h[i], new_h[j] = cur_h[j], cur_h[i]
        new_score = np.max(np.correlate(new_h, 1.0 - new_h, mode='full')) * dx
        if new_score < cur_score:
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
    return best_h, best_score

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
