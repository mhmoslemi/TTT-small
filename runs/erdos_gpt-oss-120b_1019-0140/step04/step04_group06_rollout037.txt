def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    idx = np.where(corr == max_val)[0]
    # return list of shifts
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h, dtype=float)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=float)
            term2 = np.zeros(n, dtype=float)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=float)
            term2 = np.zeros(n, dtype=float)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def get_bad_good(h, s):
    n = h.size
    if s >= 0:
        i = np.arange(0, n - s)
        partner = i + s
    else:
        s_abs = -s
        i = np.arange(s_abs, n)
        partner = i - s_abs
    h_i = h[i]
    h_partner = h[partner]
    bad = i[(h_i == 1) & (h_partner == 0)]
    good = i[(h_i == 0) & (h_partner == 1)]
    return bad, good

def simulated_annealing_binary(h0, time_left, rng):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    n = len(h)
    start = time.time()
    # Choose schedule: total time = time_left()
    total = time_left()
    # while time left
    while time_left() > 0:
        # compute time fraction passed
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        # Temperature decreasing from Tmax to Tmin
        Tmax = 1e-2
        Tmin = 1e-6
        T = Tmax * ((Tmin/Tmax) ** frac)
        # random swap
        # choose random 1 and 0
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

def simulated_annealing_continuous(h0, time_left, rng):

import time, math, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h, dtype=float)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=float)
            term2 = np.zeros(n, dtype=float)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=float)
            term2 = np.zeros(n, dtype=float)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)

def round_to_binary(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    bin_h = np.zeros_like(h, dtype=float)
    bin_h[idx] = 1.0
    return bin_h

def get_bad_good(h, s):
    n = h.size
    if s >= 0:
        i = np.arange(0, n - s)
        partner = i + s
    else:
        s_abs = -s
        i = np.arange(s_abs, n)
        partner = i - s_abs
    h_i = h[i]
    h_p = h[partner]
    bad = i[(h_i == 1) & (h_p == 0)]
    good = i[(h_i == 0) & (h_p == 1)]
    return bad, good

def local_search_binary(h, time_left, rng):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    while time_left() > 0.1:
        corr = np.correlate(h, 1 - h, mode='full')
        s = int(corr.argmax() - (n - 1))
        bad, good = get_bad_good(h, s)
        improved = False
        if bad.size and good.size:
            attempts = min(10, bad.size * good.size)
            for _ in range(attempts):
                i_bad = int(bad[rng.integers(bad.size)])
                i_good = int(good[rng.integers(good.size)])
                h[i_bad], h[i_good] = h[i_good], h[i_bad]
                new_score = compute_c5(h)
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_h = h.copy()
                    improved = True
                    break
                h[i_bad], h[i_good] = h[i_good], h[i_bad]
        if not improved:
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(ones[rng.integers(ones.size)])
            j = int(zeros[rng.integers(zeros.size)])
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h.copy()
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return best_h, best_score

def gradient_descent_projected(h0, max_iter, step0, beta, time_left):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_binary(h0, time_left, rng):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    n = h.size
    start = time.time()
    total = time_left()
    while time_left() > 0:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        Tmax, Tmin = 5e-3, 1e-6
        T = Tmax * ((Tmin / Tmax) ** frac)
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

def simulated_annealing_continuous(h0, time_left, rng):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    n = h.size
    start = time.time()
    total = time_left()
    while time_left() > 0:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        Tmax, Tmin = 5e-3, 1e-6
        T = Tmax * ((Tmin / Tmax) ** frac)
        i, j = rng.choice(n, size=2, replace=False)
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0:
            continue
        delta = delta_max * rng.uniform(0.1, 0.5)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < math.exp(-delta_score / T):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        if rng.random() < 0.02:
            shifts = max_shifts(h)
            grad = compute_gradients_for_shifts(h, shifts)
            step = 0.02
            h_tmp = h - step * grad
            h_tmp = project_onto_box_sum(h_tmp, n / 2.0)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < cur_score:

                if sc_tmp < best_score:
                    best_score = sc_tmp

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    rng = np.random.default_rng(seed)

    # Choose problem size based on remaining time
    if time_left() > 500:
        n = 512
    elif time_left() > 300:
        n = 256
    else:
        n = 128

    # Initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != n:
            base = upsample(base, n)
    else:
        base = thue_morse(n)
    base = project_onto_box_sum(base, n / 2.0)

    # Gradient descent
    max_iters = int(min(20000, max(5000, time_left() * 10)))
    h_cont, best_score_cont = gradient_descent_projected(base, max_iters, step0=0.8, beta=0.9, time_left=time_left)
    best_h = h_cont.copy()
    best_score = best_score_cont

    # Binary rounding + local search
    h_bin = round_to_binary(h_cont)
    if time_left() > 5:
        h_bin, score_bin = local_search_binary(h_bin, time_left, rng)
        if score_bin < best_score:
            best_h = h_bin.copy()
            best_score = score_bin

    # Simulated annealing refinement
    if time_left() > 30:
        if np.all((best_h == 0) | (best_h == 1)):
            h_sa, score_sa = simulated_annealing_binary(best_h, time_left, rng)
        else:
            h_sa, score_sa = simulated_annealing_continuous(best_h, time_left, rng)
        if score_sa < best_score:
            best_h = h_sa.copy()
            best_score = score_sa

    # Final projection and evaluation
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
