def compute_c5(h):
    n = h.shape[0]
    w = 1.0 - h
    L = 1 << (2*n-1).bit_length()  # next power of two for speed

    corr = np.fft.ifft(H * W_rev).real[:2*n-1]
    C5 = corr.max() * (2.0 / n)
    return C5

import time
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_balanced_from_continuous(arr):
    """Return a 0/1 vector with exactly half ones, keeping largest entries."""
    n = arr.shape[0]
    half = n // 2
    idx = np.argpartition(-arr, half)[:half]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out

def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    """Full correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    w_rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, L)
    W = np.fft.fft(w_rev, L)
    corr = np.fft.ifft(H * W).real[: 2 * n - 1]
    return corr

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    return compute_corr_fft(h).max() * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of the worst‑shift terms of C5 w.r.t. h (ignoring sum constraint)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_continuous(h0, max_iters, rng, time_left):
    """Projected gradient descent with momentum on the continuous problem."""
    h = h0.astype(float).copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    step = 0.5
    momentum = np.zeros_like(h)
    beta = 0.9

    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step * 1.02, 1.0)
        else:
            step *= 0.5
            if step < 1e-6:
                step = 0.5
        if it % 1000 == 0 and time_left() > 0.5:
            jitter = rng.normal(scale=0.01, size=n)
            h_j = project_onto_box_sum(h + jitter, target_sum)
            score_j = compute_c5(h_j)
            if score_j < cur_score:
                h, cur_score = h_j, score_j
                if score_j < best_score:
                    best_score = score_j
                    best_h = h_j.copy()
    return best_h, best_score

def simulated_annealing_binary(h0, rng, time_left, max_iters):
    """Binary swap annealing guided by C5."""
    n = h0.shape[0]
    h = (h0 > 0.5).astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    T0, T_end = 0.1, 0.001
    for it in range(max_iters):
        if time_left() <= 0:
            break
        t = it / max_iters
        T = T0 * (1 - t) + T_end * t
        i = rng.integers(n)
        j = rng.integers(n)
        while h[i] == h[j]:
            i = rng.integers(n)
            j = rng.integers(n)
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n0 = 64
        best_h = np.zeros(n0, dtype=float)
        best_h[: n0 // 2] = 1.0
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    # random balanced binary restarts at base resolution
    n0 = best_h.shape[0]
    restarts = 0
    while time_left() > 3.0 and restarts < 20:
        h_rand = random_balanced_binary(n0, rng)
        remaining = time_left()
        max_iters = int(min(15000, max(3000, remaining * 4000 / n0)))
        h_opt, sc_opt = gradient_descent_continuous(h_rand, max_iters, rng, time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        restarts += 1

    # multiscale refinement
    scales = [n0]
    while scales[-1] * 2 <= 2048:
        scales.append(scales[-1] * 2)
    for n in scales[1:]:
        if time_left() <= 0:
            break
        best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        # continuous gradient descent
        if time_left() > 2.0:
            remaining = time_left()
            max_iters = int(min(80000, max(5000, remaining * 20000 / n)))
            best_h, best_score = gradient_descent_continuous(best_h, max_iters, rng, time_left)

        # binary annealing as a polishing step
        if time_left() > 2.0:
            h_bin = binary_balanced_from_continuous(best_h)
            remaining = time_left()
            max_iters = int(min(200000, max(5000, remaining * 5000 / n)))
            h_bin_opt, sc_bin_opt = simulated_annealing_binary(h_bin, rng, time_left, max_iters)
            if sc_bin_opt < best_score:
                best_h = h_bin_opt.astype(float)
                best_score = sc_bin_opt

        best_h = project_onto_box_sum(best_h, n / 2.0)
        cur = compute_c5(best_h)
        if cur < best_score:
            best_score = cur

    # final intensive refinement
    if time_left() > 2.0:
        max_iters = int(min(80000, max(20000, time_left() * 30)))
        best_h, best_score = gradient_descent_continuous(best_h, max_iters, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
