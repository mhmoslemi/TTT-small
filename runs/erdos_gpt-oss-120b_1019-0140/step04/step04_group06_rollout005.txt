def repair(v, target_ones, rng):
    # v is 0/1 int array
    cur = v.sum()
    if cur > target_ones:
        # need to turn some 1s to 0
        idx = np.where(v == 1)[0]
        to_zero = rng.choice(idx, size=cur - target_ones, replace=False)
        v[to_zero] = 0
    elif cur < target_ones:
        idx = np.where(v == 0)[0]
        to_one = rng.choice(idx, size=target_ones - cur, replace=False)
        v[to_one] = 1
    return v
```

def tournament(pop, scores, tournament_size, rng):
    idx = rng.choice(len(pop), size=tournament_size, replace=False)
    best = idx[0]
    best_score = scores[best]
    for i in idx[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]

def ga_optimize(n, initial_h, rng, time_left):

    pop_size = 100  # maybe dynamic
    # adjust pop size based on n? For n=64 perhaps 100, for 256 maybe 150

        pop_size = 150
    # Ensure at least have initial_h
    population = []
    # start with initial_h if provided shape matches n
    if initial_h is not None and initial_h.shape[0] == n:

        population.append(v)

    while len(population) < pop_size:
        v = np.zeros(n, dtype=np.int8)
        idx = rng.choice(n, size=target_ones, replace=False)

        population.append(v)

    cache = {}

        else:

            cache[key] = s
            scores[i] = s
    best_idx = np.argmin(scores)
    best = population[best_idx].copy()
    best_score = scores[best_idx]
    generation = 0
    # parameters
    tournament_size = 5
    crossover_prob = 0.9

def local_swap_improve(h, rng, time_left):
    n = h.size
    best = h.copy()
    best_score = compute_c5(best)
    while time_left() > 0.1:
        # choose random pair 1 and 0
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        i = rng.choice(ones)
        j = rng.choice(zeros)

        new = best.copy()
        new[i] = 0
        new[j] = 1

        if s < best_score:
            best = new
            best_score = s

            # maybe accept with some low prob? Could skip.

    return best, best_score

def compute_c5(h):
    n = h.shape[0]
    h = h.astype(np.float64, copy=False)
    # compute linear correlation using FFT
    # pad to length 2*n
    N = 2 * n
    f = np.fft.rfft(h, n=N)
    g = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(f * np.conj(g), n=N)
    # correlation length 2*n-1 relevant
    corr = corr[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

import time
import numpy as np

def compute_c5(h):
    """Linear correlation based C5 using FFT (fast)."""
    n = h.shape[0]
    h = h.astype(np.float64, copy=False)
    N = 2 * n
    f = np.fft.rfft(h, n=N)
    g = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(f * np.conj(g), n=N)
    corr = corr[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """Project onto [lo,hi] with exact sum target_sum."""
    v = np.clip(v, lo, hi)
    cur = v.sum()
    if abs(cur - target_sum) < 1e-12:
        return v
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # enlarge bounds if needed
    if f(lam_low) < target_sum:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target_sum:
        lam_high = np.max(v) + 10 * hi
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def repair(v, target_ones, rng):
    """Force binary vector to have exactly target_ones ones."""
    cur = int(v.sum())
    if cur > target_ones:
        idx = np.where(v == 1)[0]
        to_zero = rng.choice(idx, size=cur - target_ones, replace=False)
        v[to_zero] = 0
    elif cur < target_ones:
        idx = np.where(v == 0)[0]
        to_one = rng.choice(idx, size=target_ones - cur, replace=False)
        v[to_one] = 1
    return v

def mutate(v, rate, rng):
    """Flip bits with given probability."""
    mask = rng.random(v.shape) < rate
    v[mask] = 1 - v[mask]
    return v

def crossover(p1, p2, rng):
    """Uniform crossover."""
    mask = rng.random(p1.shape) < 0.5
    child = np.where(mask, p1, p2)
    return child

def tournament(pop, scores, tsize, rng):
    """Select best of tsize random individuals."""
    idx = rng.choice(len(pop), size=tsize, replace=False)
    best = idx[0]
    best_score = scores[best]
    for i in idx[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return pop[best]

def ga_optimize(n, init_vec, rng, time_left):
    """Binary GA with repair and caching."""
    target_ones = n // 2
    pop_size = 120 if n < 256 else 150
    # build initial population
    population = []
    if init_vec is not None and init_vec.shape[0] == n:
        v = repair(init_vec.copy(), target_ones, rng)
        population.append(v)
    while len(population) < pop_size:
        v = np.zeros(n, dtype=np.int8)
        v[rng.choice(n, size=target_ones, replace=False)] = 1
        population.append(v)
    population = np.array(population, dtype=np.int8)
    cache = {}
    scores = np.empty(pop_size, dtype=np.float64)
    for i in range(pop_size):
        key = population[i].tobytes()
        if key in cache:
            scores[i] = cache[key]
        else:
            s = compute_c5(population[i])
            cache[key] = s
            scores[i] = s
    best_idx = int(np.argmin(scores))
    best = population[best_idx].copy()
    best_score = scores[best_idx]
    tournament_size = 5
    crossover_prob = 0.9
    mutation_rate = 0.02
    generation = 0
    while time_left() > 0.5:
        generation += 1
        new_pop = [best.copy()]  # elitism
        while len(new_pop) < pop_size:
            p1 = tournament(population, scores, tournament_size, rng)
            p2 = tournament(population, scores, tournament_size, rng)
            if rng.random() < crossover_prob:
                child = crossover(p1, p2, rng)
            else:
                child = p1.copy()
            child = mutate(child, mutation_rate, rng)
            child = repair(child, target_ones, rng)
            new_pop.append(child)
        population = np.array(new_pop, dtype=np.int8)
        scores = np.empty(pop_size, dtype=np.float64)
        for i in range(pop_size):
            key = population[i].tobytes()
            if key in cache:
                scores[i] = cache[key]
            else:
                s = compute_c5(population[i])
                cache[key] = s
                scores[i] = s
        cur_best = int(np.argmin(scores))
        cur_score = scores[cur_best]
        if cur_score < best_score:
            best_score = cur_score
            best = population[cur_best].copy()
    return best.astype(np.float64), best_score

def local_swap_improve(h, rng, time_left):
    """Random pairwise swaps that strictly improve C5."""
    best = h.copy()
    best_score = compute_c5(best)
    n = h.size
    ones = np.where(best == 1)[0]
    zeros = np.where(best == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return best, best_score
    while time_left() > 0.1:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        sc = compute_c5(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
    return best, best_score

def refine_continuous(h0, rng, time_left):
    """Projected gradient descent starting from a binary vector."""
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    best = h.copy()
    best_score = compute_c5(best)
    momentum = np.zeros_like(h)
    step0 = 0.35
    beta = 0.9
    it = 0
    while time_left() > 0.2:
        it += 1
        # compute gradient for the current maximal shift(s)
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = corr.max()
        shifts = np.where(corr == maxc)[0] - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.concatenate((np.zeros(shift), h[shift:]))
                term2 = np.concatenate((h[:n - shift], np.zeros(shift)))
            else:
                s = -shift
                term1 = np.concatenate((h[:n - s], np.zeros(s)))
                term2 = np.concatenate((np.zeros(s), h[s:]))
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best = h_new.copy()
        h = h_new
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # initial construction if provided
    init_h = None
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=np.float64)

    # scales to explore
    scales = [64, 128, 256, 512]
    best_h = None
    best_score = float('inf')

    for n in scales:
        if time_left() < 5.0:
            break

        # obtain an upsampled starting vector
        if best_h is not None:
            m = best_h.shape[0]
            x_old = np.linspace(0.0, 1.0, m, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            start_vec = np.interp(x_new, x_old, best_h)
        elif init_h is not None and init_h.shape[0] == n:
            start_vec = init_h.copy()
        else:
            # random balanced binary as fallback
            v = np.zeros(n, dtype=np.float64)
            v[rng.choice(n, size=n // 2, replace=False)] = 1.0
            start_vec = v

        # binary version for GA
        bin_vec = (start_vec > 0.5).astype(np.int8)
        bin_vec = repair(bin_vec, n // 2, rng)

        # GA optimization
        h_opt, sc_opt = ga_optimize(n, bin_vec, rng, time_left)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt

        # local pairwise improvement
        if time_left() > 5.0:
            h_loc, sc_loc = local_swap_improve(best_h, rng, time_left)
            if sc_loc < best_score:
                best_score = sc_loc
                best_h = h_loc

        # continuous refinement
        if time_left() > 3.0:
            h_ref, sc_ref = refine_continuous(best_h, rng, time_left)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.shape[0] / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
