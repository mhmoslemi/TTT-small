```python
def subgradient_max_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # Identify shifts where correlation within tolerance
    idxes = np.where(corr >= max_corr - 1e-12)[0]
    shifts = idxes - (n - 1)  # shift k
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            # i from 0 to n-k-1: grad[i] += 1 - h[i+k]
            grad[:n - k] += 1 - h[k:]
            # i from k to n-1: grad[i] += 1 - h[i-k]
            grad[k:] += 1 - h[:n - k]
        else:
            s = -k
            # shift negative => equivalent to shift positive s but switched roles
            # For negative k: the formula still holds with k negative:
            # grad[:n - s] += 1 - h[s:]
            # grad[s:] += 1 - h[:n - s]
            grad[:n - s] += 1 - h[s:]
            grad[s:] += 1 - h[:n - s]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

Note: For k negative, shift representation is same as s positive. So the same code works if we treat s = abs(k). Indeed we can just set k_abs = abs(k).

```python
def simulated_annealing(h0, init_score, max_time, rng):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    cur_score = init_score
    best_score = init_score
    n = h.shape[0]
    total_steps = int(20000 * max_time / 100.0)  # scaling?
    # Let's say we allocate a fixed number like 20000 steps or proportion to time.
    # Use max of 2000 steps if time small.
    if max_time < 1.0:
        total_steps = 2000
    else:
        total_steps = int(min(50000, max(5000, max_time * 5000)))  # as previous code
    T0 = 0.01
    T_final = 1e-6
    for step in range(total_steps):
        now = time.time()
        if now - start >= max_time:
            break
        # linear schedule
        T = T0 * (1 - step / total_steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        # Move a random fraction up to max_move
        delta = max_move * rng.random()
        if delta == 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score
```

def solve_lp(n):
    # variables
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    # For each shift k >=0
    for k in range(n):
        # overlapping length
        m = n - k
        if m <= 0:
            continue
        # w_k variables
        w = cp.Variable(m)
        h_i = h[:m]
        h_j = h[k:k+m]  # h[k:]
        constraints += [w <= h_i,
                        w <= h_j,
                        w >= h_i + h_j - 1,
                        w >= 0,
                        cp.sum(w) >= t]
    # maximize t
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=2000, verbose=False)
    except Exception:
        prob.solve(solver=cp.ECOS, verbose=False)
    if h.value is None:
        # fallback to random binary
        return None
    h_val = np.clip(np.array(h.value, dtype=float), 0.0, 1.0)
    # project to exact sum
    target_sum = n / 2.0
    h_val = project_onto_box_sum(h_val, target_sum)
    return h_val

import time
import numpy as np

try:
    import cvxpy as cp
    _CVXPY = True
except Exception:
    _CVXPY = False


def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        s = w.sum()
        if abs(s - target_sum) < 1e-12:
            return w
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_subgradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    idxes = np.where(corr >= max_corr - 1e-12)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        s = abs(k)
        if s == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[: n - s] += 1.0 - h[s:]
            grad[s:] += 1.0 - h[: n - s]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def adam_optimize(h0, max_time, rng, lr=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 0
    while True:
        if time.time() - start >= max_time:
            break
        it += 1
        grad = compute_subgradient(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def simulated_annealing(h0, init_score, max_time, rng):
    start = time.time()
    h = h0.copy()
    cur_score = init_score
    best_h = h.copy()
    best_score = init_score
    n = h.shape[0]
    total_steps = int(min(50000, max(5000, max_time * 5000)))
    T0 = 0.01
    T_final = 1e-6
    for step in range(total_steps):
        if time.time() - start >= max_time:
            break
        T = T0 * ((total_steps - step) / total_steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        if delta == 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def solve_lp(n):
    if not _CVXPY:
        return None
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    for k in range(n):
        m = n - k
        if m <= 0:
            continue
        w = cp.Variable(m)
        hi = h[:m]
        hj = h[k:k + m]
        constraints += [w <= hi,
                        w <= hj,
                        w >= hi + hj - 1,
                        w >= 0,
                        cp.sum(w) >= t]
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=2000, verbose=False)
    except Exception:
        prob.solve(solver=cp.ECOS, verbose=False)
    if h.value is None:
        return None
    h_val = np.clip(np.array(h.value, dtype=float), 0.0, 1.0)
    h_val = project_onto_box_sum(h_val, n / 2.0)
    return h_val

        if base.shape[0] != n_target:
            base = upsample(base, n_target)
        base = project_onto_box_sum(base, n_target / 2.0)
        sc = compute_c5(base)
        best_h, best_score = base, sc

    # LP-based seed (small scale)
    remaining = budget_s - (time.time() - start)
    if remaining > 30.0:
        h_lp_small = solve_lp(n_lp)
        if h_lp_small is not None:
            h_lp_big = upsample(h_lp_small, n_target)
            h_lp_big = project_onto_box_sum(h_lp_big, n_target / 2.0)
            sc_lp = compute_c5(h_lp_big)
            if sc_lp < best_score:
                best_h, best_score = h_lp_big, sc_lp

    # fallback random binary
    if best_h is None:
        h_rand = random_binary_balanced(n_target, rng)
        best_h, best_score = h_rand, compute_c5(h_rand)

    # allocate time slices
    remaining = budget_s - (time.time() - start)
    if remaining <= 0:
        return best_h, best_score, n_target
    time_adam = remaining * 0.6
    time_anneal = remaining * 0.3
    time_extra = remaining - time_adam - time_anneal

    # Adam optimization
    h_adam, sc_adam = adam_optimize(best_h, time_adam, rng, lr=0.25)
    if sc_adam < best_score:
        best_h, best_score = h_adam, sc_adam

    # Simulated annealing
    h_ann, sc_ann = simulated_annealing(best_h, best_score, time_anneal, rng)
    if sc_ann < best_score:
        best_h, best_score = h_ann, sc_ann

    # Extra short restarts with jitter
    extra_start = time.time()
    while time.time() - extra_start < time_extra:
        h_jit = best_h + rng.normal(scale=0.05, size=n_target)
        h_jit = np.clip(h_jit, 0.0, 1.0)
        h_jit = project_onto_box_sum(h_jit, n_target / 2.0)
        t_rem = time_extra - (time.time() - extra_start)
        if t_rem <= 0:
            break
        h_opt, sc_opt = adam_optimize(h_jit, min(5.0, t_rem), rng, lr=0.2)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    best_h = project_onto_box_sum(best_h, n_target / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_target
