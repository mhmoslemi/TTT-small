def adam_projected(h0, n_iter, lr, beta1,beta2, epsilon, rng, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s = h.shape[0]/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for t in range(1, n_iter+1):
        if time_left() <= 0: break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        # compute gradient using all active shifts average
        grad = compute_grad_max_shifts(h, corr, max_c)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - step
        if t % 5000 == 0:
            # add jitter
            h += rng.normal(scale=0.02, size=h.shape)
        h = project_onto_box_sum(h, s)
        score = compute_c5_fft(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts):
        grad /= len(shifts)
    return grad

        h_grid, best_score = adam_projected(
            h_grid,
            n_iter=50000,
            lr=0.5,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            rng=rng,
            time_left=time_left,
        )
    else:
        # ensure projection before returning
        h_grid = project_onto_box_sum(h_grid, n_ret / 2.0)
        best_score = compute_c5_fft(h_grid)

    return h_grid, best_score, n_ret
