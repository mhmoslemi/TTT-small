function gradient_descent(h0, time_left, step0=0.5, beta=0.9, max_iters=200000):
    h = h0.copy()
    n = h.size
    s = n/2
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    m = np.zeros_like(h)
    i = 1
    while i <= max_iters and time_left() > 0.01:
        corr = compute_corr_scaled_fft(h)  # scaled
        max_corr = np.max(corr)
        # update best if improvement
        if max_corr < best_score:
            best_score = max_corr
            best_h = h.copy()
        # find indices of max correlation (maybe more than one)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum
        m = beta * m + (1 - beta) * grad
        # step size scaling: step0 / sqrt(i)
        eta = step0 / np.sqrt(i)
        h = h - eta * m
        # projection
        h = project_onto_box_sum(h, s)
        i += 1
    return best_h, best_score

def gradient_descent_adam(h0, time_left_func, max_iters=200000, step0=0.05, beta1=0.9, beta2=0.999, eps=1e-8):

    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    while t < max_iters and time_left_func() > 0.0:
        t += 1
        # compute correlation
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        if max_corr < best_score:
            best_score = max_corr
            best_h = h.copy()
        # find worst shifts
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # gradient
        grad = compute_gradients_for_shifts(h, shifts)
        # adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        h = project_onto_box_sum(h, s)
    # final best
    final_score = compute_c5_fft(best_h)
    return best_h, final_score

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # broaden bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 (scaled by dx=2/n)."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max correlation w.r.t. h for the given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_adam(h0, time_left_func, max_iters=200000,
                         step0=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam-like projected gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    b1_pow = beta1
    b2_pow = beta2
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        it += 1
        corr = compute_corr_scaled_fft(h)
        cur = np.max(corr)
        if cur < best_score:
            best_score = cur
            best_h = h.copy()
        idxes = np.where(corr >= cur - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - b1_pow)
        v_hat = v / (1.0 - b2_pow)
        h = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, target_sum)
        # update bias‑correction powers
        b1_pow *= beta1
        b2_pow *= beta2
    final_score = compute_c5_fft(best_h)
    return best_h, final_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=400000,
                      T0=0.5, decay=0.9999, Tmin=1e-5):
    """Swap‑based simulated annealing on binary vectors."""
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        # choose a 1‑to‑0 and a 0‑to‑1 swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    final_score = compute_c5_fft(best_h)
    return best_h, final_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 512
        h0 = project_onto_box_sum(rng.random(n), n / 2.0)

    best_h = h0.copy()
    best_score = compute_c5_fft(best_h)
    target = 0.3808

    # Phase 1: long Adam descent on the current best
    if time_left() > 10.0:
        h_opt, sc_opt = gradient_descent_adam(best_h, time_left,
                                              max_iters=200000, step0=0.05)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
        if best_score <= target:
            best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
            return best_h, best_score, best_h.size

    # Phase 2: stochastic restarts with shorter Adam runs
    while time_left() > 5.0:
        v = rng.random(n)
        h_start = project_onto_box_sum(v, n / 2.0)
        # allocate roughly 0.1 s per iteration (adjustable)
        max_iters = min(20000, int(time_left() * 2000))
        h_opt, sc_opt = gradient_descent_adam(h_start, time_left,
                                              max_iters=max_iters, step0=0.05)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
            if best_score <= target:
                break

    # Phase 3: binary conversion + annealing if time permits
    if time_left() > 5.0:
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        h_bin_opt, sc_bin_opt = binary_swap_anneal(h_bin, rng, time_left,
                                                   max_iters=400000,
                                                   T0=0.5, decay=0.9999,
                                                   Tmin=1e-5)
        if sc_bin_opt < best_score:
            best_h, best_score = h_bin_opt, sc_bin_opt

    # Final projection and score recomputation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
