def gradient_descent_momentum(h0, n_iter, init_lr, beta, top_k, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # get top_k shifts
        idx_sorted = np.argpartition(-corr, range(min(top_k, corr.size)))[:min(top_k, corr.size)]
        # But argpartition does not guarantee sorted order but we will get top top_k.
        # But they may be same value or close. Use idx_sorted.
        shifts = idx_sorted - (n - 1)
        # compute gradient from these shifts
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        lr = init_lr / np.sqrt(it)
        h_new = h - lr * momentum
        # occasionally noise injection
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score_new = compute_c5(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter):
    n = h0.size
    s = n/2.0
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x, n=n):
        return np.sum(x[:n]) - n/2.0

    def ineq_fun(x, n=n):
        h = x[:n]
        t = x[-1]
        # compute correlation scaled
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr) * (2.0 / n)
        # want t >= max_corr => constraint: t - max_corr >= 0
        return t - max_corr

    # use functools.partial for constraints
    constraints = [{'type': 'eq', 'fun': functools.partial(eq_sum, n=n)},
                   {'type': 'ineq', 'fun': functools.partial(ineq_fun, n=n)}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, s)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

import time
import numpy as np
import scipy.optimize as opt
import functools

# ----- helper functions -----------------------------------------------------

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Project v onto the box [lo,hi] while enforcing sum(v)=s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # enlarge interval if needed
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def top_shifts(corr, top_k):
    k = min(top_k, corr.size)
    idx = np.argpartition(-corr, k - 1)[:k]
    return idx - (corr.size // 2)

def gradient_descent_momentum(h0, n_iter, init_lr, beta, top_k, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        shifts = top_shifts(corr, top_k)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        lr = init_lr / np.sqrt(it)
        h_new = h - lr * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def objective_last(x):
    return x[-1]

def eq_sum_constraint(x, n):
    return np.sum(x[:n]) - n / 2.0

def ineq_max_corr(x, n):
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr) * (2.0 / n)
    return t - max_corr

def slsqp_optimize_candidate(h0, max_iter):
    n = h0.size
    s = n / 2.0
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': functools.partial(eq_sum_constraint, n=n)},
        {'type': 'ineq', 'fun': functools.partial(ineq_max_corr, n=n)}
    ]
    res = opt.minimize(objective_last, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, s)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

# ----- main entry point ----------------------------------------------------

    start = time.time()

    # initial base vector

    TARGET = 0.3808

    # progressive refinements
    for n in (64, 128, 256, 512):

        if best_h.size != n:

        cur_score = compute_c5(h_cur)

        # pool for interpolations (keep up to 6 best)
        pool = [(h_cur, cur_score)]

            # choose initialization strategy

            cur_score = best_score
            h_cur = best_h.copy()

    n = best_h.size
    if time_left() > 10:

        if sc_ref < best_score:

        if time_left() > 6:
            h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=800)
            if sc_opt < best_score:
                best_h, best_score = h_opt, sc_opt

    while time_left() > 2:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or best_h[i] <= 0.0 or best_h[j] >= 1.0:

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
