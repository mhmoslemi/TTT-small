def compute_c5_fft(h):
    n = h.shape[0]
    L = 2*n - 1
    # compute correlation directly with fft.
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return np.max(corr) * (2.0 / n)

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=int, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector of length n (exactly n/2 ones)."""
    b = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=n // 2, replace=False)
    b[idx] = 1
    return b


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,2)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def balanced_binary_from_values(v):
    """Convert a real vector v to a balanced binary vector (half 1's)."""
    n = v.shape[0]
    half = n // 2
    idx = np.argsort(v)[::-1]           # descending
    b = np.zeros(n, dtype=int)
    b[idx[:half]] = 1
    return b


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on a uniform grid."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return np.max(corr) * (2.0 / n)


def contrib_vector(b, idx):
    """Contribution of a 1‑entry at position idx to the full autocorrelation."""
    n = b.shape[0]
    out = np.zeros(2 * n - 1, dtype=int)
    out[idx:idx + n] = b
    out[idx:idx + n] += b[::-1]
    out[n - 1] -= b[idx]          # correct double count of self‑pair
    return out


def hill_climb_binary(rng, b, time_left, max_no_improve=20000):
    """
    Simple local search on a balanced binary vector.
    Swaps a 1 and a 0 if it strictly increases the minimum autocorrelation.
    """
    n = b.shape[0]
    autocorr = np.correlate(b, b, mode='full')
    cur_min = autocorr.min()
    ones = np.flatnonzero(b)
    zeros = np.flatnonzero(1 - b)
    no_improve = 0

    while time_left() > 0.05 and no_improve < max_no_improve:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)      # currently 1
        j = rng.choice(zeros)     # currently 0

        # Remove contribution of i
        rem = contrib_vector(b, i)

        # Add contribution of j (based on current b, then delete i‑j pair)
        add = contrib_vector(b, j)
        add[n - 1] += 1           # self‑pair for new 1 at j
        # delete the i‑j pair that would be counted erroneously
        idx1 = i - j + (n - 1)
        idx2 = j - i + (n - 1)
        add[idx1] -= 1
        add[idx2] -= 1

        new_autocorr = autocorr - rem + add
        new_min = new_autocorr.min()

        if new_min > cur_min:
            # accept swap
            b[i] = 0
            b[j] = 1
            autocorr = new_autocorr
            cur_min = new_min
            no_improve = 0
            ones = np.flatnonzero(b)
            zeros = np.flatnonzero(1 - b)
        else:
            no_improve += 1

    return b, cur_min


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    TARGET = 0.3808
    best_h = None
    best_score = np.inf

    # Choose a moderate resolution where binary search is still cheap
    n = 256
    min_time_per_hill = 5.0   # we need at least this much time for a hill‑climb

    # Prepare a first deterministic start (from parent if any, else Thue‑Morse)
    if 'initial_h_values' in globals():
        init_vals = np.asarray(initial_h_values, dtype=float)
        if init_vals.shape[0] != n:
            init_vals = upsample(init_vals, n)
        b_start = balanced_binary_from_values(init_vals)
    else:
        b_start = balanced_binary_from_values(thue_morse(n))

    # First hill‑climb on the deterministic start
    if time_left() > min_time_per_hill:
        b_opt, _ = hill_climb_binary(rng, b_start.copy(), time_left)
        h_opt = b_opt.astype(float)
        score = compute_c5_fft(h_opt)
        if score < best_score:
            best_score, best_h = score, h_opt.copy()
        if best_score <= TARGET:
            return best_h, best_score, n

    # Random restarts until time runs out
    while time_left() > min_time_per_hill:
        b0 = random_binary_balanced(n, rng)
        b_opt, _ = hill_climb_binary(rng, b0, time_left)
        h_opt = b_opt.astype(float)
        score = compute_c5_fft(h_opt)
        if score < best_score:
            best_score, best_h = score, h_opt.copy()
        if best_score <= TARGET:
            break

    # Safety fallback
    if best_h is None:
        b_fallback = balanced_binary_from_values(thue_morse(n))
        best_h = b_fallback.astype(float)
        best_score = compute_c5_fft(best_h)

    # Final projection to guarantee exact sum (binary already satisfies)
    # Return the best found construction
    return best_h, best_score, n
