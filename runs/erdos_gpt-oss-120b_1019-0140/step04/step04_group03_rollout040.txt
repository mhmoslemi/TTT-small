    n = len(h)
    offset = n - 1
    # i term first
    s1 = -i
    e1 = n-1-i
    idx1 = slice(s1+offset, e1+offset+1)  # inclusive
    h_i_shift = h[i + np.arange(s1, e1+1)]

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        s = w.sum()
        if abs(s - target_sum) < 1e-12:
            return w
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / h.size)

def fast_local_search(h0, time_budget, rng):
    h = h0.copy()
    n = h.size
    offset = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_corr = corr.max()
    best_h = h.copy()
    eps = 1e-12
    start = time.time()
    while time.time() - start < time_budget:
        cur_max = corr.max()
        max_idx = np.argmax(corr)
        shift = max_idx - offset  # integer shift with max correlation

        # candidates for decreasing (i) and increasing (j) mass
        if shift >= 0:
            lim = n - shift
            i_range = np.arange(lim)
            i_mask = (h[i_range] > eps) & (h[i_range + shift] < 1.0 - eps)
            j_mask = (h[i_range] < 1.0 - eps) & (h[i_range + shift] > eps)
            i_cand = i_range[i_mask]
            j_cand = i_range[j_mask]
        else:
            s = -shift
            lim = n - s
            i_range = np.arange(s, n)
            i_mask = (h[i_range] > eps) & (h[i_range - s] < 1.0 - eps)
            j_mask = (h[i_range] < 1.0 - eps) & (h[i_range - s] > eps)
            i_cand = i_range[i_mask]
            j_cand = i_range[j_mask]

        if i_cand.size == 0:
            i = rng.integers(n)
        else:
            i = rng.choice(i_cand)
        if j_cand.size == 0:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
        else:
            j = rng.choice(j_cand)
        if i == j:
            continue

        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        if delta <= 0.0:
            continue

        # prepare linear part
        one_minus_h = 1.0 - h
        rev_h = h[::-1]

        # linear updates (four slices)
        start_i1 = offset - i
        corr[start_i1:start_i1 + n] -= delta * one_minus_h
        start_j1 = offset - j
        corr[start_j1:start_j1 + n] += delta * one_minus_h
        corr[i:i + n] += delta * rev_h
        corr[j:j + n] -= delta * rev_h

        # quadratic corrections
        d2 = delta * delta
        idx0 = offset
        corr[idx0] -= 2.0 * d2
        idx_j_i = idx0 + (j - i)
        corr[idx_j_i] += d2
        idx_i_j = idx0 + (i - j)
        corr[idx_i_j] += d2

        # apply move
        h[i] -= delta
        h[j] += delta

        new_max = corr.max()
        if new_max <= cur_max + 1e-12:
            # accept
            cur_max = new_max
            if new_max < best_corr:
                best_corr = new_max
                best_h = h.copy()
        else:
            # revert move
            h[i] += delta
            h[j] -= delta
            corr[start_i1:start_i1 + n] += delta * one_minus_h
            corr[start_j1:start_j1 + n] -= delta * one_minus_h
            corr[i:i + n] -= delta * rev_h
            corr[j:j + n] += delta * rev_h
            corr[idx0] += 2.0 * d2
            corr[idx_j_i] -= d2
            corr[idx_i_j] -= d2
    return best_h, best_corr * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_final = 1024
    # initial construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = upsample(base, n_final)
    else:
        base = thue_morse(n_final)
    base = project_onto_box_sum(base, n_final / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # a few random restarts
    for _ in range(8):
        if budget_s - (time.time() - start) < 1.0:
            break
        h_rand = random_binary_balanced(n_final, rng)
        h_rand = project_onto_box_sum(h_rand, n_final / 2.0)
        sc = compute_c5(h_rand)
        if sc < best_score:
            best_score, best_h = sc, h_rand.copy()

    # fast local refinement
    remaining = budget_s - (time.time() - start) - 0.2
    if remaining > 0.1:
        h_opt, sc_opt = fast_local_search(best_h, remaining, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    best_h = project_onto_box_sum(best_h, n_final / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_final
