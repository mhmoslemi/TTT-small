def binary_swap_anneal(h0, rng, time_left, max_iters=10_000_000, T0=0.2, decay=0.99999, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    current_score = best_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.0:
        # Choose random 1 and 0 positions
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]   # ensures zeros
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        # accept with probability
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        # update temperature
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
```

    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s_target = h.size / 2.0
    for t in range(1, max_iter+1):
        if time_left() <= 0:
            break
        # compute correlation and max shifts
        corr = compute_corr_scaled_fft(h)

        shifts = np.where(np.abs(corr - max_corr) < 1e-12)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, s_target)
    return h

def compute_gradients_for_shifts(h, shifts):

def adam_optimize(h0, rng, time_left, max_iter=2000, lr=0.1):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    target_sum = h.size / 2.0
    for t in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # determine shifts achieving max
        shifts = np.where(np.abs(corr - max_corr) < 1e-12)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, target_sum)
    return h
```

def binary_swap_anneal(h0, rng, time_left, max_iters=10_000_000, T0=0.2, decay=0.99999, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    current_score = best_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.0:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def adam_optimize(h0, rng, time_left, max_iter=2000, lr=0.1):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    target_sum = h.size / 2.0
    for t in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        shifts = np.where(np.abs(corr - max_corr) < 1e-12)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, target_sum)
    return h

def binary_swap_anneal(h0, rng, time_left, max_iters=10_000_000, T0=0.2, decay=0.99999, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    current_score = best_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.0:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    n = 1024

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = upsample(init, n)
    else:
        init = rng.random(n)
    init = project_onto_box_sum(init, n / 2.0)

    if time_left() > 1.0:
        h_cont = adam_optimize(init, rng, lambda: time_left(), max_iter=1500, lr=0.15)
    else:
        h_cont = init

    idx = np.argpartition(-h_cont, n // 2)[:n // 2]
    h_bin = np.zeros_like(h_cont)
    h_bin[idx] = 1.0

    best_h = h_bin
    best_score = compute_c5_fft(best_h)

    # evaluate thue‑morse baseline
    h_tm = project_onto_box_sum(thue_morse(n), n / 2.0)
    sc_tm = compute_c5_fft(h_tm)
    if sc_tm < best_score:
        best_h, best_score = h_tm, sc_tm

    if best_score <= target:
        return best_h, best_score, n

    while time_left() > 2.0:
        remaining = time_left()
        run_time = min(40.0, remaining * 0.6)
        run_start = time.time()
        def inner_time_left():
            return max(0.0, run_time - (time.time() - run_start))
        if rng.random() < 0.3:
            h0 = best_h.copy()
        else:
            h0 = random_binary_balanced(n, rng)
        h_cand, sc_cand = binary_swap_anneal(
            h0, rng, inner_time_left,
            max_iters=10_000_000, T0=0.2, decay=0.99999, Tmin=1e-6
        )
        if sc_cand < best_score:
            best_h, best_score = h_cand.copy(), sc_cand
        if best_score <= target:
            break

    if time_left() > 1.0:
        h_final = adam_optimize(best_h, rng, lambda: time_left(), max_iter=800, lr=0.1)
        sc_final = compute_c5_fft(h_final)
        if sc_final < best_score:
            best_h, best_score = h_final.copy(), sc_final

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, n
