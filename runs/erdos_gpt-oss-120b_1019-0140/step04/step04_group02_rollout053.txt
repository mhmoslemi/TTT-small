        max_corr = np.max(corr)

        if max_corr <= target_score:

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    return np.fft.ifft(fh * np.conj(f1mh)).real

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.size
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def local_move_optimization(h, cur_score, time_left, rng, target):
    n = h.size
    best_h = h.copy()
    best_score = cur_score
    while time_left() > 2.0:
        corr = compute_corr_full_fft(best_h)
        max_unscaled = np.max(corr)
        max_scaled = max_unscaled * (2.0 / n)
        if max_scaled < best_score:
            best_score = max_scaled
            best_h = best_h.copy()
        if max_scaled <= target:
            break
        tol = max_unscaled * 1e-6 if max_unscaled > 0 else 1e-12
        idxes = np.where(corr >= max_unscaled - tol)[0]
        rng.shuffle(idxes)
        improved = False
        for idx in idxes:
            shift = idx - (n - 1)
            if shift >= 0:
                if n - shift < 2:
                    continue
                i_range = np.arange(n - shift)
                diff = best_h[i_range] - best_h[i_range + shift]
            else:
                m = -shift
                if n - m < 2:
                    continue
                i_range = np.arange(m, n)
                diff = best_h[i_range] - best_h[i_range - m]
            max_diff = diff.max()
            if max_diff <= 1e-6:
                continue
            i_rel = diff.argmax()
            i = i_range[i_rel]
            j = i + shift if shift >= 0 else i - m
            delta = min(max_diff * 0.5, 0.05, best_h[i], 1.0 - best_h[j])
            if delta < 1e-8:
                continue
            new_h = best_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_corr = compute_corr_full_fft(new_h)
            new_max = np.max(new_corr) * (2.0 / n)
            if new_max < max_scaled - 1e-9:
                best_h, best_score = new_h, new_max
                improved = True
                break
        if not improved:
            i, j = rng.choice(n, size=2, replace=False)
            if best_h[i] > 1e-8 and best_h[j] < 1 - 1e-8:
                delta = min(best_h[i] * 0.5, (1 - best_h[j]) * 0.5, 0.02)
                new_h = best_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
                new_max = np.max(compute_corr_full_fft(new_h)) * (2.0 / n)
                if new_max < max_scaled - 1e-9:
                    best_h, best_score = new_h, new_max
                    continue
            new_h = best_h + rng.normal(scale=0.01, size=n)
            new_h = project_onto_box_sum(new_h, n / 2.0)
            new_max = np.max(compute_corr_full_fft(new_h)) * (2.0 / n)
            if new_max < max_scaled - 1e-9:
                best_h, best_score = new_h, new_max
    return best_h, best_score
