def simulated_annealing(h0, max_iters, time_left, rng):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    T = cur_score * 0.05 + 1e-6
    alpha = 0.9999
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # choose move
        if rng.random() < 0.65:
            # pair transfer
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random() * 0.5
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # worst shift move
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            max_corr = np.max(corr_full)
            worst_idxs = np.where(corr_full >= max_corr - 1e-12)[0]
            shift = worst_idxs[rng.integers(len(worst_idxs))] - (n - 1)
            if shift >= 0:
                region_len = n - shift
                contrib = cur_h[:region_len] * (1.0 - cur_h[shift:])
                i_local = np.argmax(contrib)
                i = i_local
                j = i + shift
            else:
                s = -shift
                region_len = n - s
                contrib = cur_h[s:] * (1.0 - cur_h[:region_len])
                i_local = np.argmax(contrib)
                i = i_local + s
                j = i - s
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random() * 0.5
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        new_score = compute_c5(new_h)
        dE = new_score - cur_score
        if dE <= 0 or rng.random() < np.exp(-dE / T):
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
        T *= alpha
        if T < 1e-12:
            T = 1e-12

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) / 2.0
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:

def simulated_annealing(h0, max_iters, time_left, rng):
    """Simple SA that respects box+sum constraints."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    T = max(cur_score * 0.05, 1e-8)
    alpha = 0.9999

    for it in range(max_iters):
        if time_left() <= 0:
            break
        if rng.random() < 0.65:
            # pairwise mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random() * 0.5
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move mass along a worst shift
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            max_corr = np.max(corr_full)
            worst_idxs = np.where(corr_full >= max_corr - 1e-12)[0]
            shift = worst_idxs[rng.integers(len(worst_idxs))] - (n - 1)
            if shift >= 0:
                region_len = n - shift
                contrib = cur_h[:region_len] * (1.0 - cur_h[shift:])
                i_local = np.argmax(contrib)
                i = i_local
                j = i + shift
            else:
                s = -shift
                region_len = n - s
                contrib = cur_h[s:] * (1.0 - cur_h[:region_len])
                i_local = np.argmax(contrib)
                i = i_local + s
                j = i - s
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random() * 0.5
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_score = compute_c5(new_h)
        dE = new_score - cur_score
        if dE <= 0 or rng.random() < np.exp(-dE / T):
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
        T *= alpha
        if T < 1e-12:
            T = 1e-12
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]

    for n in scales:
        if time_left() <= 2:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Quick gradient descent refinement
        if time_left() > 5:
            max_iters_gd = int(min(4000, max(800, time_left() * (2000.0 / n))))
            h_gd, sc_gd = gradient_descent_adaptive(
                base, max_iters_gd, step0=0.8, beta=0.9, rng=rng, time_left=time_left
            )
            base = h_gd
            if sc_gd < best_score:
                best_h, best_score = h_gd.copy(), sc_gd

        # Simulated annealing runs
        max_sa_runs = 5 if n <= 256 else 3
        for _ in range(max_sa_runs):
            if time_left() <= 5:
                break
            max_iters_sa = int(min(20000, max(2000, time_left() * 12)))
            h_sa, sc_sa = simulated_annealing(base, max_iters_sa, time_left, rng)
            if sc_sa < best_score:
                best_h, best_score = h_sa.copy(), sc_sa
                base = h_sa
            # Short gradient descent after SA
            if time_left() > 4:
                max_iters_ref = int(min(8000, max(1500, time_left() * 5)))
                h_ref, sc_ref = gradient_descent_adaptive(
                    base, max_iters_ref, step0=0.5, beta=0.9, rng=rng, time_left=time_left
                )
                base = h_ref
                if sc_ref < best_score:
                    best_h, best_score = h_ref.copy(), sc_ref

        # Random balanced binary search at this scale
        if time_left() > 2:
            num_rand = min(200, int(time_left() * 20))
            for _ in range(num_rand):
                h_rand = random_balanced_binary(n, rng)
                sc_rand = compute_c5(h_rand)
                if sc_rand < best_score:
                    best_h, best_score = h_rand.copy(), sc_rand
                    base = h_rand

    # Final intensive refinement on the global best
    if time_left() > 5:
        max_iters_final = int(min(80000, max(20000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(
            best_h, max_iters_final, step0=0.3, beta=0.9, rng=rng, time_left=time_left
        )

    # Final small SA for fine-tuning
    if time_left() > 3:
        max_iters_final_sa = int(min(20000, max(5000, time_left() * 15)))
        best_h, best_score = simulated_annealing(
            best_h, max_iters_final_sa, time_left, rng
        )

    # Make absolutely sure constraints hold
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
