def solve_milp_binary(n, time_limit):
    # Build MILP and solve with GLPK Mi.
    import cvxpy as cp
    x = cp.Variable(n, boolean=True)
    M = cp.Variable()
    constraints = [cp.sum(x) == n//2, M >= 0]

    for t in range(1, n):
        y = cp.Variable(n - t, boolean=True)
        # product constraints
        constraints.append(y <= x[:n - t])
        constraints.append(y <= x[t:])
        constraints.append(y >= x[:n - t] + x[t:] - 1)
        # prefix and suffix sums
        prefix_sum = cp.sum(x[:n - t])
        suffix_sum = cp.sum(x[t:])
        constraints.append(prefix_sum - cp.sum(y) <= M)
        constraints.append(suffix_sum - cp.sum(y) <= M)
    prob = cp.Problem(cp.Minimize(M), constraints)
    # Solve with time limit
    # Use GLPK_MI with tmlim option
    try:
        prob.solve(solver=cp.GLPK_MI, glpk_options=["--tmlim", str(int(time_limit))],
                   warm_start=False, verbose=False)
    except Exception as e:
        # maybe solver not installed
        return None, None

    if x.value is None:
        return None, None
    # extract binary solution rounding
    x_sol = np.rint(np.array(x.value)).astype(float)
    M_val = M.value
    return x_sol, M_val

def local_descent(h, time_left):
    n = h.shape[0]
    while time_left() > 0.2:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        if max_val * (2.0 / n) <= 0.3808:
            break
        idx = np.where(corr == max_val)[0][0]  # first
        k = idx - (n - 1)
        # find candidate i:
        if k >= 0:
            # overlap region indices 0..n-1-k
            region = np.arange(0, n - k)
        else:
            # k negative: shift left, overlap region indices -k..n-1
            region = np.arange(-k, n)
        # find i where h[i]==1 and h[i+k]==0
        i_candidates = region[(h[region] > 0.5) & (h[(region + k) % n] <= 0.5)]   # Wait zero padding not wrapping; need to index only within region where i+k valid. Since for k>=0 region i+k within [k,n-1]; for k negative region i+k within [0,n-1+k].
        # Actually region is defined accordingly, so i+k always within valid index range.

        if i_candidates.size == 0:
            # can't improve with directed move; break
            break
        i = i_candidates[0]  # pick first

        # find j where h[j]==0 and j not equal i (any zero)
        zero_indices = np.where(h <= 0.5)[0]
        # Remove i if zero (but it's 1)
        if zero_indices.size == 0:
            break
        j = zero_indices[0] if zero_indices[0] != i else (zero_indices[1] if zero_indices.size>1 else None)
        if j is None:
            break
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new score
        new_max = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
        if new_max < max_val * (2.0 / n):
            # improved, continue
            continue
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            # break or try another i or j
            break

    return h

import time, math, numpy as np

# Helper: projection onto box [lo,hi] with exact sum s (convex).
def _project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # Binary search on lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # enlarge bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 value for a given step vector h (numpy 1‑d)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def random_binary_balanced(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


# MILP formulation for binary h of length n.
def _solve_milp_binary(n, time_limit):
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    # variables
    x = cp.Variable(n, boolean=True)
    M = cp.Variable()
    constraints = [cp.sum(x) == n // 2, M >= 0]

    # product variables and overlap constraints
    for t in range(1, n):
        y = cp.Variable(n - t, boolean=True)
        constraints += [
            y <= x[: n - t],
            y <= x[t:],
            y >= x[: n - t] + x[t:] - 1,
        ]
        prefix = cp.sum(x[: n - t])
        suffix = cp.sum(x[t:])
        constraints += [
            prefix - cp.sum(y) <= M,
            suffix - cp.sum(y) <= M,
        ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    # GLPK_MI respects '--tmlim' (seconds)
    try:
        prob.solve(solver=cp.GLPK_MI,
                   glpk_options=["--tmlim", str(int(time_limit))],
                   warm_start=False,
                   verbose=False)
    except Exception:
        return None, None

    if x.value is None:
        return None, None
    x_sol = np.rint(np.array(x.value)).astype(float)
    M_val = M.value
    return x_sol, M_val
