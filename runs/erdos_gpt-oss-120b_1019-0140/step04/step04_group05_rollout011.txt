def compute_c5(h):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)
    conv = np.fft.ifft(H * G).real
    # linear convolution result length 2n-1 at indices 0..2n-2
    max_val = np.max(conv[:2*n-1])
    return max_val * (2.0 / n)

def simulated_annealing(h0, time_limit, rng, init_temp=0.2, final_temp=1e-5):
    h = h0.copy()
    cur_score = compute_c5(h)
    best = h.copy()
    best_score = cur_score
    start = time.time()
    iter = 0
    # we may periodically project to correct rounding
    proj_interval = 5000
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        t = elapsed / time_limit  # progress 0..1
        temp = init_temp * ((final_temp / init_temp) ** t)
        n = h.size
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # compute max delta
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        # scale delta by temperature fraction (more diff)
        # Could multiply by (temp / init_temp)
        scale = temp / init_temp
        # random fraction
        delta = max_delta * rng.random() * scale
        if delta <= 0:
            continue
        # apply move
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        if new_score < best_score:
            best_score = new_score
            best = h.copy()
        if new_score < cur_score:
            cur_score = new_score
        else:
            # Metropolis
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert
                h[i] += delta
                h[j] -= delta
        iter += 1
        if iter % proj_interval == 0:
            # correct rounding
            h = project_onto_box_sum(h, n/2.0)
            cur_score = compute_c5(h)  # recompute after projection
    return best, best_score

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Approximate C5 = max_k ∫ h(x)(1‑h(x+k))dx on the grid."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    conv = np.fft.irfft(H * G, N)
    max_val = np.max(conv[:2 * n - 1])
    return max_val * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Binary Thue–Morse sequence of length n (values 0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def simulated_annealing(h0, time_limit, rng, init_temp=0.2, final_temp=1e-5):
    """Continuous SA preserving sum by transferring mass between two positions."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    proj_interval = 5000
    it = 0

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        # temperature schedule (exponential decay)
        t = elapsed / time_limit
        temp = init_temp * ((final_temp / init_temp) ** t)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue

        scale = temp / init_temp
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            continue

        # apply move (preserves sum)
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                cur_score = new_score
            else:
                h[i] += delta
                h[j] -= delta

        it += 1
        if it % proj_interval == 0:
            h = project_onto_box_sum(h, n / 2.0)
            cur_score = compute_c5(h)

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # base construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    # multiscale schedule (total budget fractions sum ≤ 1)
    scales = [128, 256, 512, 1024]
    fractions = {128: 0.05, 256: 0.10, 512: 0.25, 1024: 0.55}
    best_h = base
    best_score = compute_c5(best_h)

    for n in scales:
        if time_left() <= 0.0:
            break
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        cur_score = compute_c5(best_h)
        if cur_score < best_score:
            best_score, best_h = cur_score, best_h.copy()

        alloc = fractions[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        cand_h, cand_score = simulated_annealing(best_h, alloc, rng)
        if cand_score < best_score:
            best_score, best_h = cand_score, cand_h.copy()
        # prepare for next scale
        best_h = cand_h

    # final refinement with remaining budget
    rem = time_left()
    if rem > 0.1:
        # short low‑temp SA for fine tuning
        cand_h, cand_score = simulated_annealing(best_h, rem, rng,
                                                init_temp=0.05, final_temp=1e-6)
        if cand_score < best_score:
            best_score, best_h = cand_score, cand_h.copy()

        # random binary perturbations as last resort
        while time_left() > 0.05:
            h_rand = random_balanced_binary(best_h.size, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score, best_h = sc, h_rand.copy()
                if best_score <= 0.3808:
                    break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
