def gradient_smooth(h, p):
    n = h.size
    L = 2 * n - 1
    corr = compute_corr_full_fft(h)   # size L, not scaled
    corr_scaled = corr * (2.0 / n)    # scale to integral
    max_corr = np.max(corr_scaled)
    # compute softmax weights for numerical stability
    exp_term = np.exp(p * (corr_scaled - max_corr))
    w = exp_term / np.sum(exp_term)   # shape L
    N = L + n - 1
    # compute correlation of w and h (linear)
    conv = np.fft.ifft(np.fft.fft(w, N) * np.conj(np.fft.fft(h, N))).real
    conv_forward = conv[:n]   # length n
    grad = 1.0 - 2.0 * conv_forward
    return grad, max_corr  # maybe return objective

                            beta1, beta2, eps, rng, time_left):

    n = h.size

        # gradient

def gradient_smooth(h, p):
    n = h.shape[0]
    L = 2*n - 1
    # correlation between h and (1-h)
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    corr_scaled = corr * (2.0 / n)
    # softmax weights
    max_corr = np.max(corr_scaled)
    # exponent
    exp_term = np.exp(p * (corr_scaled - max_corr))
    w = exp_term / np.sum(exp_term)
    # compute convolution w * h
    N = L + n - 1
    # FFT of h padded to N
    fhN = np.fft.fft(h, N)
    fw = np.fft.fft(w, N)
    conv = np.fft.ifft(fw * np.conj(fhN)).real
    conv_fwd = conv[:n]  # length n
    grad = 1.0 - 2.0 * conv_fwd
    return grad, max_corr
```

    scales = [n0]
    if n0 < 256:
        scales.append(256)
    if scales[-1] < 512:
        scales.append(512)

    for n in scales:
        if time_left() <= 2.0:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)

        max_iter = int(150000 * 256 / n)
        max_iter = max(50000, min(max_iter, 200000))
        step0 = 0.6
        p_initial, p_final = 5.0, 50.0
        restarts = 2 if n <= 256 else 1

        for _ in range(restarts):
            if time_left() <= 2.0:
                break
            h0 = best_h + rng.normal(scale=0.07, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)
            h_opt, sc_opt = smooth_gradient_descent(
                h0, max_iter,
                p_initial, p_final,
                step0, 0.9, 0.999, 1e-8,
                rng, time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_score <= TARGET:
            break

        if time_left() > 1.0:
            n_cur = best_h.size
            for _ in range(5):
                h_bin = random_binary_balanced(n_cur, rng)
                sc_bin = compute_c5_fft(h_bin)
                if sc_bin < best_score:
                    best_score, best_h = sc_bin, h_bin.copy()
                    if best_score <= TARGET:
                        break
                i, j = rng.choice(n_cur, size=2, replace=False)
                delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.2)
                if delta > 1e-12:
                    h_new = best_h.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    sc_new = compute_c5_fft(h_new)
                    if sc_new < best_score:
                        best_score, best_h = sc_new, h_new.copy()
                        if best_score <= TARGET:
                            break
                if best_score <= TARGET:
                    break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)

    # optional SLSQP refinement if time permits
    if time_left() > 5.0:
        try:
            n_dim = best_h.size
            x0 = np.concatenate([best_h, [best_score]])
            bounds = [(0.0, 1.0)] * n_dim + [(0.0, 1.0)]

            def eq_sum(x):
                return np.sum(x[:n_dim]) - n_dim / 2.0

            def ineq_fun(x):
                h = x[:n_dim]
                t = x[-1]
                corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n_dim)
                return t - corr

            cons = [{'type': 'eq', 'fun': eq_sum},
                    {'type': 'ineq', 'fun': ineq_fun}]
            res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
            if res.success:
                h_sls = project_onto_box_sum(res.x[:n_dim], n_dim / 2.0)
                sc_sls = compute_c5_fft(h_sls)
                if sc_sls < best_score:
                    best_score, best_h = sc_sls, h_sls.copy()
        except Exception:
            pass

    return best_h, best_score, best_h.size
