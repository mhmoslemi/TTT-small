def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n); term1[:n_s] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n); term1[s:] = h[:n_s]
        term2 = np.zeros(n); term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

Check correctness: The correlation C(k) = ∫ h(x)*(1 - h(x+k))dx discretized as sum_i h[i] * (1 - h[i+k]) where index offset modulo??? Since periodic? Actually correlation uses full mode, which includes shifts from -(n-1) to (n-1). In the discrete convolution, h and (1-h) are convolved with full mode, which corresponds to non-circular correlation (zero padded). For shift >= 0, C[k] = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k]) . Then derivative wrt h at index i: currently h[i] appears in term h[i] * (1 - h[i+k]) for shift >=0 (if i <= n-1-k) plus appears in h[i-k] * (1 - h[i]) (if i-k >=0). The gradient formula yields contributions: derivative of C wrt h[i] = (1 - h[i+shift]) for i <= n-1-shift, plus -h[i-shift] for i >= shift. The implemented gradient_of_shift returns 1 - term1 - term2 where term1 maps shifted indices of h[shift:] onto term1[:n_s] (i from 0 to n-shift-1) and term2 maps h[:n_s] onto term2[shift:] (i from shift to n-1). So term1[i] = h[i+shift] (if i < n_s else 0), and term2[i] = h[i-shift] (if i >= shift else 0). Then gradient = 1 - (h[i+shift] + h[i-shift]), which matches derivative of C (due to both terms). Actually derivative of C with respect to h[i] is (1 - h[i+shift]) + (-h[i-shift]) = 1 - h[i+shift] - h[i-shift], matches gradient_of_shift. For shift negative, similar.

When shift >=0, we also need to consider that for indices beyond overlap (i.e., term1 or term2 truncated) we have zeros, gradient = 1 - term - term2 may be right. However for the region where overlap exists, full expression applies. Means gradient is correct.

def compute_gradient_max_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad
```

However using exact equality may only pick exactly equal indices; but since we want the shift(s) that achieve maximum, using np.where(corr == mx) works because mx is precisely a entry from corr array. It's okay.

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n); term1[:n_s] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n); term1[s:] = h[:n_s]
        term2 = np.zeros(n); term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

        if time_left() <= 0.0:
            break
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        mx_new, grad_new = compute_corr_max_and_grad(h_new)
        s_new = mx_new * (2.0 / n)
        if s_new + 1e-12 < best_s:
            best_s, best_h = s_new, h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h, grad = h_new, grad_new
        if no_improve >= patience:
            break
    return best_h, best_s

def jitter_refine(h, time_left, rng, target, sigma_init=0.05, decay=0.99):
    cur = compute_c5_fast(h)
    best_h, best_c = h.copy(), cur
    sigma = sigma_init
    while time_left() > 0.01 and sigma > 1e-5:
        i = rng.integers(0, h.size)
        h_new = best_h.copy()
        h_new[i] = np.clip(h_new[i] + rng.normal(0.0, sigma), 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, target)
        c_new = compute_c5_fast(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        sigma *= decay
    return best_h, best_c

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # primary resolution
    n = 512
    target = n / 2.0
    cand = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n)
        cand.append(project_onto_box_sum(init, target))

    cand.append(thue_morse_binary(n))
    for _ in range(2):
        cand.append(random_binary_balanced(n, rng))
    for _ in range(2):
        v = rng.normal(0.5, 0.2, size=n)
        cand.append(project_onto_box_sum(v, target))
    for freq in (2, 3, 5):
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / 2)
        cand.append(project_onto_box_sum(h, target))

    best_h = None
    best_s = np.inf

    # refined search on initial candidates
    for c in cand:
        if time_left() < 5.0:
            break
        h_opt, s_opt = gradient_descent_momentum(
            c, time_left, step0=0.3, beta=0.9, max_iter=20000, patience=2000)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= 0.3808:
            return best_h, best_s, n

    # random restarts
    while time_left() > 8.0:
        v = rng.normal(0.5, 0.2, size=n)
        v = project_onto_box_sum(v, target)
        h_opt, s_opt = gradient_descent_momentum(
            v, time_left, step0=0.3, beta=0.9, max_iter=5000, patience=1000)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= 0.3808:
            break

    # optional upscaling if time permits
    if best_h is not None and time_left() > 200.0:
        n2 = 1024
        target2 = n2 / 2.0
        h2 = resample(best_h, n2)
        h2 = project_onto_box_sum(h2, target2)
        h2_opt, s2_opt = gradient_descent_momentum(
            h2, time_left, step0=0.3, beta=0.9, max_iter=20000, patience=2000)
        if s2_opt < best_s:
            best_s, best_h, n = s2_opt, h2_opt, n2
            target = target2

    # final jitter refinement
    if best_h is not None and time_left() > 1.0:
        h_ref, s_ref = jitter_refine(best_h, time_left, rng, target)
        if s_ref < best_s:
            best_s, best_h = s_ref, h_ref

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, target)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s, best_h.shape[0]
