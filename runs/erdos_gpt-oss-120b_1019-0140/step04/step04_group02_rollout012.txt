def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    # trivial cases:
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # initial lam bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # define f(lam)
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure f(lam_low) >= s and f(lam_high) <= s
    # bisection
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if np.abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)
```

But be careful: f is decreasing in lam: as lam increases, v-lam decreases, so sum decreases. Actually f(lam) = sum_i clip(v_i - lam, lo, hi). At lam low, v_i - lam high => may be >hi, so clip to hi, sum ~ hi*n. At lam high, v_i - lam low => clip to lo, sum ~ lo*n. So f(lam) goes from hi*n down to lo*n as lam increases. So we should have f(lam_low) >= s >= f(lam_high). Good.

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    start_time = time.time()
    target = 0.3808

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128)

    nlist = [64, 128, 256]
    best_h = None
    best_score = np.inf

    for n in nlist:
        if time.time() - start_time >= budget_s - 5.0:
            break

        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        if n == 64:
            n_iter = 150000
            restarts = 30
            step0 = 0.8
        elif n == 128:
            n_iter = 200000
            restarts = 20
            step0 = 0.6
        else:  # 256
            n_iter = 250000
            restarts = 12
            step0 = 0.5

        for r in range(restarts):
            if time.time() - start_time >= budget_s - 5.0:
                break
            p = rng.random()
            if p < 0.2:
                h0 = random_balanced_binary(n, rng)
            elif p < 0.4:
                h0 = binary_balanced_from_continuous(base)
            else:
                h0 = base + rng.normal(scale=0.07, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)

            h_opt, sc_opt = gradient_descent(
                h0, n_iter, step0, 0.9, rng, budget_s, start_time
            )
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= target:
                    break
        if best_score <= target:
            break

        # extra refinement on current best at this scale
        if best_h is not None and best_h.size == n:
            elapsed = time.time() - start_time
            if elapsed < budget_s - 5.0:
                extra_iter = max(50000, n_iter // 2)
                h_ref, sc_ref = gradient_descent(
                    best_h, extra_iter, step0 * 0.7, 0.9, rng, budget_s, start_time
                )
                if sc_ref < best_score:
                    best_score, best_h = sc_ref, h_ref.copy()

        # prepare base for next finer resolution
        base = best_h if best_h is not None else base

    # final binary hill climbing if time permits
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if best_h is not None and remaining > 4.0:
        bin_start = binary_balanced_from_continuous(best_h)
        deadline = start_time + remaining * 0.5
        h_bin, sc_bin = hill_climb_binary(bin_start, deadline, rng)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()

    if best_h is None:
        n_ret = 256
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5(best_h)
    else:
        n_ret = best_h.size
        best_h = project_onto_box_sum(best_h, n_ret / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, n_ret
