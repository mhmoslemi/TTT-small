    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    it = 1

        max_corr = np.max(corr)
        # Find shifts achieving max

            if shift >= 0:
                grad[:n-shift] += 1.0 - h[shift:]
                grad[shift:] += -h[:n-shift]

            best_score = score
            best_h = h_new.copy()

    corr = compute_corr_fft(h)  # unscaled
    max_corr = corr.max()
    w = np.exp(alpha * (corr - max_corr))

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_corr_fft(h):
    """Full correlation array (unscaled)."""
    n = h.shape[0]
    M = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += 1.0 - h[shift:]
            grad[shift:] += -h[:n - shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] += -h[s:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def smooth_max_gradient(h, alpha):
    """Gradient of log-sum-exp approximation to max C5 (scaled by dx)."""
    n = h.shape[0]
    corr = compute_corr_fft(h)                # unscaled correlation
    max_corr = corr.max()
    w = np.exp(alpha * (corr - max_corr))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    len_w = w.shape[0]
    conv_len = len_w + n - 1
    M2 = 1 << ((conv_len).bit_length())
    W = np.fft.rfft(w, n=M2)
    rev_one_minus_h = np.flip(1.0 - h)
    A_full = np.fft.irfft(W * np.conj(np.fft.rfft(rev_one_minus_h, n=M2)),
                          n=M2)[:conv_len]
    B_full = np.fft.irfft(W * np.conj(np.fft.rfft(h, n=M2)),
                          n=M2)[:conv_len]
    start = n - 1
    A = A_full[start:start + n]
    B = B_full[start:start + n]
    grad = (A - B) * (2.0 / n)               # scale by dx
    return grad, max_corr * (2.0 / n)        # return also current C5 estimate

def subgradient_descent(h0, time_left, max_iter=200000):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    it = 1
    while it <= max_iter and time_left() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = 0.5 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def smooth_gradient_descent(h0, time_left, max_iter=20000):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    base_alpha = 5.0
    alpha_max = 200.0
    step_alpha = 2000
    it = 1
    while it <= max_iter and time_left() > 0.0:
        alpha = base_alpha * (1 + (it // step_alpha))
        if alpha > alpha_max:
            alpha = alpha_max
        grad, _ = smooth_max_gradient(h, alpha)
        momentum = beta * momentum + (1 - beta) * grad
        eta = 0.5 / np.sqrt(it)

        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score
