import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # Ensure bounds contain solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def fft_convolve(a, b):
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:n]
    return conv

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    return np.max(compute_corr_scaled_fft(h))

def smooth_grad_descent(h0, time_left, max_iter=20000,
                       alpha_start=2.0, alpha_end=30.0,
                       step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        frac = it / max_iter
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)

        p = exp_vals / sum_exp
        L = np.dot(p, corr)

        w_center = n - 1

        u = 1.0 - h

        term1 = conv1[w_center:w_center + n]

        conv3 = fft_convolve(w_neg_full, u)
        term3 = conv3[:n]
        conv4 = fft_convolve(w_neg_full[::-1], h)

        sc = compute_c5_fft(h)
        if sc < best_score:

            best_h = h.copy()
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left, max_iters=500000,
                      T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def fft_convolve(a, b):
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:n]
    return conv

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    return np.max(compute_corr_scaled_fft(h))

def smooth_grad_descent(h0, time_left, max_iter=20000,
                       alpha_start=2.0, alpha_end=30.0,
                       step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_sc = compute_c5_fft(h)
    mom = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        frac = it / max_iter
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        corr = compute_corr_scaled_fft(h)
        m = np.max(corr)
        expv = np.exp(alpha * (corr - m))
        p = expv / expv.sum()
        L = np.dot(p, corr)
        w = p * (1.0 + alpha * (corr - L))
        w_center = n - 1
        w_pos = w[w_center:]
        w_neg_full = np.concatenate([np.array([0.0]), w[:w_center][::-1]])
        u = 1.0 - h
        conv1 = fft_convolve(w_pos[::-1], u)
        t1 = conv1[w_center:w_center + n]
        conv2 = fft_convolve(w_pos, h)
        t2 = -conv2[:n]
        conv3 = fft_convolve(w_neg_full, u)
        t3 = conv3[:n]
        conv4 = fft_convolve(w_neg_full[::-1], h)
        t4 = -conv4[w_center:w_center + n]
        grad = (t1 + t2 + t3 + t4) * dx
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h = project_onto_box_sum(h - eta * mom, target)
        sc = compute_c5_fft(h)
        if sc < best_sc:
            best_sc, best_h = sc, h.copy()
    return best_h, best_sc

def binary_swap_anneal(h0, rng, time_left, max_iters=500000,
                      T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    cur_sc = compute_c5_fft(h)
    best_h, best_sc = h.copy(), cur_sc
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_sc = compute_c5_fft(h)
        delta = new_sc - cur_sc
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_h = new_sc, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 512
        h0 = rng.random(n)
        h0 = project_onto_box_sum(h0, n / 2.0)
    best_h, best_sc = h0.copy(), compute_c5_fft(h0)
    # Phase 1: smooth gradient descent
    h_sgd, sc_sgd = smooth_grad_descent(best_h, time_left)
    if sc_sgd < best_sc:
        best_h, best_sc = h_sgd, sc_sgd
    if best_sc <= target:
        return best_h, best_sc, best_h.size
    # Phase 2: perturbation + short runs
    while time_left() > 30.0:
        pert = best_h + rng.normal(scale=0.1, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        pert = project_onto_box_sum(pert, n / 2.0)
        h2, sc2 = smooth_grad_descent(pert, time_left,
                                      max_iter=4000,
                                      alpha_start=5.0,
                                      alpha_end=25.0)
        if sc2 < best_sc:
            best_h, best_sc = h2, sc2
            if best_sc <= target:
                return best_h, best_sc, best_h.size
        # occasional binary conversion & anneal
        if time_left() > 5.0:
            k = n // 2
            idx = np.argpartition(-best_h, k)[:k]
            h_bin = np.zeros_like(best_h)
            h_bin[idx] = 1.0
            sc_bin = compute_c5_fft(h_bin)
            if sc_bin < best_sc:
                best_h, best_sc = h_bin, sc_bin
                if best_sc <= target:
                    return best_h, best_sc, best_h.size
            if time_left() > 2.0:
                h_ann, sc_ann = binary_swap_anneal(best_h, rng, time_left)
                if sc_ann < best_sc:
                    best_h, best_sc = h_ann, sc_ann
                    if best_sc <= target:
                        return best_h, best_sc, best_h.size
    # Phase 3: final binary anneal
    if best_h.max() - best_h.min() > 1e-6 and time_left() > 2.0:
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        best_h = h_bin
        best_sc = compute_c5_fft(best_h)
    if time_left() > 1.0:
        h_ann, sc_ann = binary_swap_anneal(best_h, rng, time_left)
        if sc_ann < best_sc:
            best_h, best_sc = h_ann, sc_ann
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_sc = compute_c5_fft(best_h)
    return best_h, best_sc, best_h.size
