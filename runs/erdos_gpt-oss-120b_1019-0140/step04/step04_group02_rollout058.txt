def optimize_h(h0, time_left, target=None):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    # Adam parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it_total = 0
    alphas = [1.0, 5.0, 20.0, 80.0]
    # compute steps per alpha based on remaining time?
    for alpha in alphas:
        # set step size: maybe decreasing with alpha
        step0 = 0.3 / np.sqrt(alpha)  # arbitrary
        # run many iterations
        for _ in range(20000):
            if time_left() < 0.02:
                break
            it_total += 1
            # compute corr, weights, gradient
            corr = compute_corr_full_fft(h)
            max_c = np.max(corr)
            # optional early break if best_score <= target
            if target is not None and best_score <= target:
                break
            # compute gradient via softmax
            w = np.exp(alpha * (corr - max_c))
            w_sum = np.sum(w)
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            # gradient
            grad = grad_softmax(h, w)   # need to compute using convs
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad**2)
            m_hat = m / (1 - beta1**it_total)
            v_hat = v / (1 - beta2**it_total)
            step = step0 / np.sqrt(it_total)
            h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_onto_box_sum(h_new, s)
            # evaluate score occasionally
            if it_total % 20 == 0:
                score = compute_c5_fft(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new.copy()
            h = h_new

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1−h) using FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def grad_softmax(h, w):
    """Gradient of (1/α)·log‑sum‑exp(α·corr) = Σ p·∇corr, using FFT convolutions."""
    n = h.shape[0]
    a = 1.0 - h
    w_rev = w[::-1]
    L = 1 << ((3 * n - 2 - 1).bit_length())
    fw_rev = np.fft.fft(w_rev, L)
    fa = np.fft.fft(a, L)
    conv1 = np.real(np.fft.ifft(fw_rev * fa))
    fw = np.fft.fft(w, L)
    fh = np.fft.fft(h, L)
    conv2 = np.real(np.fft.ifft(fw * fh))
    offset = 2 * n - 2
    grad = conv1[offset:offset + n] - conv2[offset:offset + n]
    return grad

def optimize_h(h0, time_left, rng, target=None):
    """Projected Adam on a smoothed max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    alphas = [1.0, 5.0, 20.0, 80.0]
    for alpha in alphas:
        step0 = 0.5 / np.sqrt(alpha + 1e-12)
        max_iters = 20000
        for _ in range(max_iters):
            if time_left() < 0.02:
                return best_h, best_score
            it += 1
            corr = compute_corr_full_fft(h)
            max_c = np.max(corr)
            # softmax weights
            w = np.exp(alpha * (corr - max_c))
            w_sum = np.sum(w)
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            grad = grad_softmax(h, w)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = step0 / np.sqrt(it)
            h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_onto_box_sum(h_new, s)
            # occasional evaluation
            if it % 20 == 0:
                score = compute_c5_fft(h_new)
                if score < best_score:
                    best_score, best_h = score, h_new.copy()
                    if target is not None and best_score <= target:
                        return best_h, best_score
            # occasional noise injection to escape plateaus
            if it % 2000 == 0:
                h_new = project_onto_box_sum(h_new + rng.normal(scale=0.01, size=n), s)
            h = h_new
        # start next alpha from current best
        h = best_h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    TARGET = 0.3808

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128)

    scales = [128, 256, 512]
    best_h = None
    best_score = np.inf

    for n in scales:
        if time_left() < 1.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # multiple random restarts
        restarts = 3
        for _ in range(restarts):
            if time_left() < 0.5:
                break
            h0 = base + rng.normal(scale=0.04, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)
            h_opt, sc_opt = optimize_h(h0, time_left, rng, target=TARGET)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_score <= TARGET:
            break
        # prepare for next finer scale
        base = best_h.copy() if best_h is not None else base

    # final refinement on the finest scale if time permits
    if best_h is not None and time_left() > 5.0:
        best_h, best_score = optimize_h(best_h, time_left, rng, target=TARGET)

    # optional cyclic‑shift search
    if best_h is not None and time_left() > 2.0:
        n = best_h.size
        for shift in range(1, n):
            if time_left() < 0.02:
                break
            h_shift = np.roll(best_h, shift)
            sc = compute_c5_fft(h_shift)
            if sc < best_score:
                best_score, best_h = sc, h_shift.copy()
                if best_score <= TARGET:
                    break

    # last‑minute random binary / small transfer tweaks
    while best_h is not None and time_left() > 4.0:
        n = best_h.size
        # random balanced binary attempt
        h_bin = random_binary_balanced(n, rng)
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
            if best_score <= TARGET:
                break
        # tiny mass transfer between two positions
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.02, 0.15)
        if delta > 1e-12:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5_fft(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                if best_score <= TARGET:
                    break

    # final safety projection
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
        n_ret = best_h.size
    else:
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, n_ret
