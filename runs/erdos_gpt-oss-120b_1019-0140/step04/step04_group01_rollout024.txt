def gradient_softmax(h, alpha):
    n = h.size
    s = n/2.0
    corr = compute_corr_scaled_fft(h)
    # Compute softmax weights
    a_corr = alpha * corr
    max_a = np.max(a_corr)
    w = np.exp(a_corr - max_a)
    w_sum = np.sum(w)
    w /= w_sum
    # w_arr length 2n-1
    w_arr = w
    # Reverse for forward
    w_rev = w_arr[::-1]
    # FFT length
    M = 1 << ((3 * n - 2 - 1).bit_length())
    # FFTs
    W = np.fft.rfft(w_arr, n=M)
    Wr = np.fft.rfft(w_rev, n=M)
    H = np.fft.rfft(h, n=M)
    conv_back = np.fft.irfft(W * H, n=M)
    conv_fwd = np.fft.irfft(Wr * H, n=M)
    # extract
    start = n - 1
    term_back = conv_back[start:start + n]
    term_fwd = conv_fwd[start:start + n]
    grad = 1.0 - term_fwd - term_back
    return grad

def softmax_gd_momentum(h0, time_left_func, max_iter=200000, step0=0.5, beta=0.9, alpha0=0.1, alpha_factor=1.0005, alpha_max=15.0):

    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    alpha = alpha0
    while it <= max_iter and time_left_func() > 0.1:
        # compute gradient via softmax
        grad = gradient_softmax(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
        # occasionally evaluate real C5
        if it % 50 == 0:
            sc = compute_c5_fft(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
        # increase alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)
        it += 1
    # final evaluation
    final_score = compute_c5_fft(h)
    if final_score < best_score:
        best_score = final_score
        best_h = h.copy()
    return best_h, best_score

def binary_rounding(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    h_bin = np.zeros_like(h)
    h_bin[idx] = 1.0
    return h_bin

def gradient_softmax(h, alpha):
    n = h.size
    dx = 2.0 / n
    corr = compute_corr_scaled_fft(h)   # scaled
    a_corr = alpha * corr
    max_a = np.max(a_corr)
    w = np.exp(a_corr - max_a)
    w_sum = np.sum(w)
    w /= w_sum
    w_arr = w
    w_rev = w_arr[::-1]
    M = 1 << ((3 * n - 2 - 1).bit_length())
    W = np.fft.rfft(w_arr, n=M)
    Wr = np.fft.rfft(w_rev, n=M)
    H = np.fft.rfft(h, n=M)
    conv_back = np.fft.irfft(W * H, n=M)
    conv_fwd = np.fft.irfft(Wr * H, n=M)
    start = n - 1
    term_back = conv_back[start:start + n]
    term_fwd = conv_fwd[start:start + n]
    grad = (1.0 - term_fwd - term_back) / dx
    return grad

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gd_active_momentum(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):

def binary_swap_anneal(h0, rng, time_left_func, max_iters=400000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        # Ensure binary values (0 or 1) maybe h is binary but may have 0/1 values, ignore tolerance
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # adjust bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import numpy as np
import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    return np.max(compute_corr_scaled_fft(h))

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gd_active_momentum(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    final_score = compute_c5_fft(h)
    if final_score < best_score:
        best_score = final_score
        best_h = h.copy()
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func,
                      max_iters=400000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def binary_rounding(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    h_bin = np.zeros_like(h)
    h_bin[idx] = 1.0
    return h_bin

def gradient_softmax(h, alpha):
    n = h.size
    corr = compute_corr_scaled_fft(h)          # scaled by dx
    a_corr = alpha * corr
    max_a = np.max(a_corr)
    w = np.exp(a_corr - max_a)
    w /= w.sum()
    w_arr = w
    w_rev = w_arr[::-1]
    M = 1 << ((3 * n - 2 - 1).bit_length())
    W = np.fft.rfft(w_arr, n=M)
    Wr = np.fft.rfft(w_rev, n=M)
    H = np.fft.rfft(h, n=M)
    conv_back = np.fft.irfft(W * H, n=M)
    conv_fwd = np.fft.irfft(Wr * H, n=M)
    start = n - 1
    term_back = conv_back[start:start + n]
    term_fwd = conv_fwd[start:start + n]
    grad = 1.0 - term_fwd - term_back
    return grad

def softmax_gd_momentum(h0, time_left_func, max_iter=200000,
                        step0=0.5, beta=0.9,
                        alpha0=0.2, alpha_factor=1.001, alpha_max=20.0):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    alpha = alpha0
    dx = 2.0 / n
    while it <= max_iter and time_left_func() > 0.1:
        grad = gradient_softmax(h, alpha)               # unscaled gradient
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 * dx / np.sqrt(it)                  # scale step by dx
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
        if it % 50 == 0:
            sc = compute_c5_fft(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)
        it += 1
    final_score = compute_c5_fft(h)
    if final_score < best_score:
        best_score = final_score
        best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # initial construction
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.ndim != 1:
            h0 = h0.ravel()
        n_target = 512
        if h0.size != n_target:
            h0 = downsample(h0, n_target)
    else:
        n_target = 512
        h0 = random_binary_balanced(n_target, rng)
    h0 = project_onto_box_sum(h0, n_target / 2.0)

    best_h = h0.copy()
    best_score = compute_c5_fft(best_h)

    # Phase 1: smoothed gradient descent
    if time_left() > 5.0:
        h_s, sc_s = softmax_gd_momentum(best_h, time_left,
                                        max_iter=200000,
                                        step0=0.5, beta=0.9,
                                        alpha0=0.2, alpha_factor=1.001,
                                        alpha_max=20.0)
        if sc_s < best_score:
            best_score, best_h = sc_s, h_s.copy()
        if best_score <= 0.3808:
            return best_h, best_score, best_h.size

    # Phase 2: binary rounding
    if time_left() > 2.0:
        h_bin = binary_rounding(best_h)
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
        if best_score <= 0.3808:
            return best_h, best_score, best_h.size

    # Phase 3: binary annealing
    if time_left() > 1.0:
        h_ann, sc_ann = binary_swap_anneal(best_h, rng, time_left,
                                          max_iters=400000,
                                          T0=0.2, decay=0.99995, Tmin=1e-5)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()
        if best_score <= 0.3808:
            return best_h, best_score, best_h.size

    # Phase 4: active-shift gradient descent
    if time_left() > 0.5:
        h_ref, sc_ref = gd_active_momentum(best_h, time_left,
                                           step0=0.5, beta=0.9,
                                           max_iter=200000)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= 0.3808:
            return best_h, best_score, best_h.size

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
