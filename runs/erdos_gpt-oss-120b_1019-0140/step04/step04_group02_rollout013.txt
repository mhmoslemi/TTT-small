def hill_climb_local(h, max_iters, rng, time_left):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    for it in range(max_iters):
        if time_left() <= 0.0:
            break
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # choose delta as fraction of min
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta < 1e-12:
            continue
        delta = max_delta * rng.random() * 0.2  # up to 20% of allowable
        if delta <= 0:
            continue
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5_fft(h_new)
        if sc_new < best_score - 1e-9:
            best_h, best_score = h_new, sc_new
    return best_h, best_score
```

def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)  # returns full correlation (no scaling)
        # compute max correlation value (scaled)
        max_c = np.max(corr)
        # gradient
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasionally add noise
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def compute_corr_full_fft(h):
    n = len(h)
    L = 2*n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr
```

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = len(h)
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if shifts.size > 0:
        grad /= shifts.size
    return grad
```

import time
import numpy as np

def project_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the interval brackets the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1 as floats)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector of length n (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) (unscaled)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient averaged over all shifts attaining the maximal correlation."""
    n = h.size
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if shifts.size > 0:
        grad /= shifts.size
    return grad


def compute_grad_random_max_shift(h, corr, max_c, rng):
    """Gradient for a randomly chosen shift achieving the maximal correlation."""
    n = h.size
    idxes = np.where(corr >= max_c - 1e-12)[0]
    chosen = rng.choice(idxes)
    k = chosen - (n - 1)
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] = 1.0 - h[k:]
        grad[k:] = -h[:n - k]
    else:
        m = -k
        grad[m:] = 1.0 - h[:n - m]
        grad[:n - m] = -h[m:]
    return grad


def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    """Projected sub‑gradient descent with momentum."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, max_iters + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        # combine average and random max‑shift gradients
        grad_avg = compute_grad_max_shifts(h, corr, max_c)
        grad_rand = compute_grad_random_max_shift(h, corr, max_c, rng)
        grad = 0.5 * grad_avg + 0.5 * grad_rand

        grad_norm = np.linalg.norm(grad) + 1e-12
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / (np.sqrt(it) * grad_norm)
        h_new = h - eta * momentum

        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def hill_climb_local(h, max_iters, rng, time_left):
    """Simple pairwise mass‑transfer hill climbing."""
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)

    for _ in range(max_iters):
        if time_left() <= 0.0:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * rng.random() * 0.2  # up to 20% of allowable mass
        if delta <= 0:
            continue
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5_fft(h_new)
        if sc_new < best_score - 1e-12:
            best_h, best_score = h_new, sc_new

        # occasional swap of values (full exchange)
        if rng.random() < 0.02:
            i2, j2 = rng.integers(0, n, size=2)
            if i2 != j2:
                h_sw = best_h.copy()
                h_sw[i2], h_sw[j2] = h_sw[j2], h_sw[i2]
                sc_sw = compute_c5_fft(h_sw)
                if sc_sw < best_score - 1e-12:
                    best_h, best_score = h_sw, sc_sw

    return best_h, best_score
