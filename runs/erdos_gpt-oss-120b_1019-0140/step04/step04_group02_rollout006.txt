def compute_corr_full_fft(h):
    n = h.shape[0]

    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def simulated_annealing(h_start, max_iters, temp0, decay, step_scale, time_left):
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    t = temp0
    i = 0
    while i < max_iters and time_left() > 0:
        i += 1
        # pick two distinct indices
        i_idx, j_idx = rng.integers(0, n, size=2, endpoint=False)
        if i_idx == j_idx:
            continue
        # compute max delta feasible
        max_delta = min(h[i_idx], 1.0 - h[j_idx])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * step_scale
        # apply move
        h_new = h.copy()
        h_new[i_idx] -= delta
        h_new[j_idx] += delta
        new_score = compute_c5(h_new)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / max(t, 1e-12)):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()

                    break
        # update temperature
        t *= decay
        # occasional global perturbation
        if i % 5000 == 0:
            # add small Gaussian noise and project

def adam_optimize(h0, max_iters, step0, beta1, beta2, eps, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        # compute gradient
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad*grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - step * m_hat
        # project onto constraints
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= TARGET:
                break
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()

import time
import numpy as np

def remaining_time(start, budget):
    return max(0.0, budget - (time.time() - start))

def project_onto_box_sum(v, s):
    """Project onto {0<=x_i<=1, sum x_i = s} via binary search."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - s) < 1e-12:
        return v
    # search lambda such that sum(clip(v - lam)) = s
    lam_low, lam_high = -1.0, 1.0
    for _ in range(60):
        lam = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        total = w.sum()
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, 0.0, 1.0)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idx = np.where(corr >= max_c - tol)[0]
    shifts = idx - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts):
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def simulated_annealing(h_start, max_iters, temp0, decay, step_scale,
                        rng, start, budget, target):
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    cur_score = compute_c5(h)
    best_score = cur_score
    t = temp0
    for it in range(1, max_iters + 1):
        if remaining_time(start, budget) <= 0.0:
            break
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * step_scale
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / max(t, 1e-12)):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                if best_score <= target:
                    break
        t = max(t * decay, 1e-14)
        # occasional global noise
        if it % 5000 == 0:
            h_noise = h + rng.normal(scale=0.01, size=n)
            h_noise = project_onto_box_sum(h_noise, n / 2.0)
            noise_score = compute_c5(h_noise)
            dEn = noise_score - cur_score
            if dEn < 0.0 or rng.random() < np.exp(-dEn / max(t, 1e-12)):
                h = h_noise
                cur_score = noise_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
                    if best_score <= target:
                        break
    return best_h, best_score

def adam_optimize(h0, max_iters, step0, beta1, beta2, eps,
                  start, budget, target):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters + 1):
        if remaining_time(start, budget) <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - step * m_hat
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= target:
                break
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    TARGET = 0.3808

    # --- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():
        base_init = np.asarray(initial_h_values, dtype=float)
    else:
        base_init = thue_morse(128)

    # --- coarse scale (256) ---------------------------------------------
    n_coarse = 256
    base = upsample(base_init, n_coarse)
    base = project_onto_box_sum(base, n_coarse / 2.0)

    # one SA run on coarse grid
    max_iter_coarse = 300000
    h_coarse, sc_coarse = simulated_annealing(
        base, max_iter_coarse, temp0=0.001, decay=0.99998,
        step_scale=0.5, rng=rng, start=start, budget=budget_s, target=TARGET)

    best_h = h_coarse
    best_score = sc_coarse

    # --- random restarts on coarse grid (if time permits) ---------------
    while remaining_time(start, budget_s) > 6.0 and best_score > TARGET:
        # random balanced binary start
        h0 = random_binary_balanced(n_coarse, rng)
        h0 = project_onto_box_sum(h0, n_coarse / 2.0)
        h_tmp, sc_tmp = simulated_annealing(
            h0, 80000, temp0=0.001, decay=0.9999,
            step_scale=0.5, rng=rng, start=start, budget=budget_s, target=TARGET)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
        if best_score <= TARGET:
            break

    # --- fine scale (512) ------------------------------------------------
    if best_score > TARGET and remaining_time(start, budget_s) > 6.0:
        n_fine = 512
        h_fine = upsample(best_h, n_fine)
        h_fine = project_onto_box_sum(h_fine, n_fine / 2.0)
        max_iter_fine = 500000
        h_fine_opt, sc_fine_opt = simulated_annealing(
            h_fine, max_iter_fine, temp0=0.001, decay=0.99999,
            step_scale=0.5, rng=rng, start=start, budget=budget_s, target=TARGET)
        if sc_fine_opt < best_score:
            best_score, best_h = sc_fine_opt, h_fine_opt.copy()

        # occasional random binary restart on fine grid
        while remaining_time(start, budget_s) > 4.0 and best_score > TARGET:
            h0 = random_binary_balanced(n_fine, rng)
            h0 = project_onto_box_sum(h0, n_fine / 2.0)
            h_tmp, sc_tmp = simulated_annealing(
                h0, 100000, temp0=0.001, decay=0.9999,
                step_scale=0.5, rng=rng, start=start, budget=budget_s, target=TARGET)
            if sc_tmp < best_score:
                best_score, best_h = sc_tmp, h_tmp.copy()
            if best_score <= TARGET:
                break

    # --- final Adam refinement -------------------------------------------
    if best_score > TARGET and remaining_time(start, budget_s) > 5.0:
        max_iter_adam = 50000
        h_adam, sc_adam = adam_optimize(
            best_h, max_iter_adam, step0=0.5, beta1=0.9,
            beta2=0.999, eps=1e-8, start=start, budget=budget_s, target=TARGET)
        if sc_adam < best_score:
            best_score, best_h = sc_adam, h_adam.copy()

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    n_ret = best_h.size
    return best_h, best_score, n_ret
