def subgradient_step(h, k_max, step_size):
    # compute gradient g_i as described
    n = len(h)
    g = np.zeros_like(h)
    # contributions when i + k_max < n:
    i1 = np.arange(n - k_max)
    g[i1] += (1 - h[i1 + k_max])
    # contributions when i - k_max >= 0:
    i2 = np.arange(k_max, n)
    g[i2] -= h[i2 - k_max]
    # choose i (largest positive gradient) and j (largest negative gradient)
    i = np.argmax(g)  # high positive
    j = np.argmin(g)  # most negative
    if i == j:
        return h # nothing
    max_delta = min(h[i], 1 - h[j])
    delta = min(max_delta, step_size)
    if delta > 0:
        h[i] -= delta
        h[j] += delta
    return h

def simulated_annealing(h0, time_alloc, rng):
    n = len(h0)
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.01  # maybe larger
    while time.time() - start < time_alloc:
        elapsed = time.time() - start
        # fractional progress
        frac = elapsed / time_alloc
        # temperature decays exponentially
        temp = init_temp * (0.5 ** frac)  # can also use exp(-frac*ln2)
        # sample i,j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i: j += 1
        # delta
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        # scale delta by temperature fraction
        # ratio = temp / init_temp in [0.5,1] initially? Actually at start frac=0, temp=init_temp.
        # As frac increases, temp halves each half of time. For delta scaling, we can just use rng.random() * max_delta * (temp / init_temp) (bounded <=max_delta)
        delta = max_delta * rng.random() * (temp / init_temp)
        if delta <= 0.0:
            continue

        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

                # compute acceptance prob
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_score = new_score
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta
            else:
                # revert
                h[i] += delta
                h[j] -= delta
    return best_h, best_score

def subgradient_move(h, k_max, step):
    n = len(h)
    # compute gradient
    g = np.zeros_like(h)
    # i+k < n
    idx = np.arange(n - k_max)
    g[idx] += (1 - h[idx + k_max])
    # i-k >= 0
    idx2 = np.arange(k_max, n)
    g[idx2] -= h[idx2 - k_max]
    # find i_max positive, i_min negative
    i_pos = np.argmax(g)
    i_neg = np.argmin(g)
    # ensure we can transfer from pos (where we want decrease) to neg (increase)
    # Actually if g[i_pos] <= 0, all gradients <=0, no beneficial move; return h unchanged.
    if g[i_pos] <= 0:
        return h, False
    # we want to decrease h[i_pos] and increase h[i_neg]
    max_delta = min(h[i_pos], 1.0 - h[i_neg])
    delta = min(max_delta, step)
    if delta <= 0.0:
        return h, False
    h_new = h.copy()
    h_new[i_pos] -= delta
    h_new[i_neg] += delta
    # note: sum preserved
    return h_new, True

def local_search(h_best, time_alloc, rng):
    n = len(h_best)
    best_h = h_best.copy()
    best_score = compute_c5(best_h)
    start = time.time()
    # step size schedule: maybe start with 0.01 and reduce
    init_step = 0.02  # max delta per step
    while time.time() - start < time_alloc:
        elapsed = time.time() - start
        frac = elapsed / time_alloc
        step = init_step * (0.5 ** frac)  # decreasing
        # compute correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (n - 1)
        # try subgradient move
        h_new, moved = subgradient_move(best_h.copy(), k_max, step)
        if moved:
            new_score = compute_c5(h_new)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new
                continue
        # else try random move similar to SA but without acceptance (only improvement)
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i: j += 1
        max_delta = min(best_h[i], 1 - best_h[j])
        if max_delta <= 0: continue
        delta = max_delta * rng.random() * (step / init_step)  # step scaled
        if delta <= 0: continue
        h2 = best_h.copy()
        h2[i] -= delta
        h2[j] += delta
        new_score2 = compute_c5(h2)
        if new_score2 < best_score:
            best_score = new_score2
            best_h = h2
            continue
        # optionally try random balanced binary restart if stuck and time left > something
        if rng.random() < 0.001:  # occasional restart
            h_rand = random_balanced_binary(n, rng)
            # project (should already satisfy sum)
            sc_rand = compute_c5(h_rand)
            if sc_rand < best_score:
                best_score = sc_rand
                best_h = h_rand
    return best_h, best_score

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - total) < 1e-12:
            return proj
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):

    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_for_shift(h, k):
    """Gradient of C_k = Σ h_i (1‑h_{i+k}) w.r.t. h, for a given shift k."""

def subgradient_move(h, k, max_step):
    """Move mass from a positive‑gradient index to a negative‑gradient index."""
    g = gradient_for_shift(h, k)
    i_pos = np.argmax(g)
    i_neg = np.argmin(g)
    if g[i_pos] <= 0:
        return h, False
    max_delta = min(h[i_pos], 1.0 - h[i_neg])
    if max_delta <= 0:
        return h, False
    delta = min(max_delta, max_step)
    if delta <= 0:
        return h, False
    h_new = h.copy()
    h_new[i_pos] -= delta
    h_new[i_neg] += delta
    return h_new, True

                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                # revert
                h[i], h[j] = h[j], h[i]
            continue

        # regular mass‑transfer move
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * (temp / init_temp)
        if delta <= 0.0:
            continue
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            if temp > 0.0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_score = new_score
                else:
                    h[i] += delta
                    h[j] -= delta
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_score


def local_search(h_start, time_alloc, rng):
    """Deterministic subgradient + small random moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    init_step = 0.05

    while time.time() - start < time_alloc:
        if best_score <= TARGET_C5:
            break
        elapsed = time.time() - start
        frac = elapsed / time_alloc
        step = init_step * (0.5 ** frac)

        # current worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr) - (h.size - 1)

        # subgradient move
        h_new, moved = subgradient_move(h, k_max, step)
        if moved:
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new
                continue

        # small random mass‑transfer (only accept improvements)
        i = rng.integers(h.size)
        j = rng.integers(h.size - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * (step / init_step)
            if delta > 0:
                h2 = h.copy()
                h2[i] -= delta
                h2[j] += delta
                sc2 = compute_c5(h2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h2.copy()
                    h = h2
                    continue

        # occasional restart with a random balanced binary vector
        if rng.random() < 0.001:
            h_rand = random_balanced_binary(h.size, rng)
            sc_rand = compute_c5(h_rand)
            if sc_rand < best_score:
                best_score = sc_rand
                best_h = h_rand.copy()
                h = h_rand
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial vector
    if 'initial_h_values' in globals():
        cur_h = np.asarray(initial_h_values, dtype=float)
    else:
        cur_h = thue_morse(64)
    cur_h = project_onto_box_sum(cur_h, cur_h.size / 2.0)

    best_h = cur_h.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]
    fractions = {64: 0.07, 128: 0.13, 256: 0.20, 512: 0.30, 1024: 0.30}

    for n in scales:
        if time_left() <= 0.0:
            break
        if cur_h.size != n:
            cur_h = upsample(cur_h, n)
        cur_h = project_onto_box_sum(cur_h, n / 2.0)

        alloc = fractions[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        cand_h, cand_score = simulated_annealing(cur_h, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
            cur_h = cand_h
        else:
            cur_h = best_h  # keep the best found so far

        if best_score <= TARGET_C5:
            break

    # Ensure best_h is at the highest resolution processed
    if best_h.size < cur_h.size:
        best_h = upsample(best_h, cur_h.size)
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)

    # final refinement with remaining budget
    remaining = time_left()
    if remaining > 0.0 and best_score > TARGET_C5:
        best_h, best_score = local_search(best_h, remaining, rng)

    # final safeguard projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
