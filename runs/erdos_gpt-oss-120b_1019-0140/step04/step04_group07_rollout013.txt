def full_correlation_fft(a, b):
    n = a.shape[0]
    L = 2 * n - 1
    # Next power of two for FFT speed
    L_fft = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, n=L_fft)
    B = np.fft.rfft(b, n=L_fft)
    corr = np.fft.irfft(A * np.conj(B), n=L_fft)
    # Return only first L elements (full correlation)
    return corr[:L]

    step = step0

        if time_left() <= 0:

        corr = full_correlation_fft(h, 1.0 - h)
        max_corr = np.max(corr)

        new_score = compute_c5_fft(h_new)
        if new_score <= cur_score:

def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = n = best_h.size
    while time_left() > 0:
        corr = full_correlation_fft(best_h, 1.0 - best_h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            # contributions for each i where i+shift < n
            i_range = np.arange(n - shift)
            contrib = best_h[i_range] * (1.0 - best_h[i_range + shift])
            i_local = np.argmax(contrib)
            i = i_local
            j = i_local + shift
        else:
            s = -shift
            i_range = np.arange(s, n)
            contrib = best_h[i_range] * (1.0 - best_h[i_range - s])
            i_local = np.argmax(contrib) + s
            i = i_local
            j = i_local - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)
        if new_score < best_score:
            best_h, best_score = h_new, new_score
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure the search interval brackets the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def full_correlation_fft(a, b):
    """Full (linear) correlation of real vectors a and b via FFT."""
    n = a.shape[0]
    L = 2 * n - 1
    L_fft = 1 << (L - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=L_fft)
    B = np.fft.rfft(b, n=L_fft)
    corr = np.fft.irfft(A * np.conj(B), n=L_fft)
    return corr[:L]


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = full_correlation_fft(h, 1.0 - h)
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n as floats."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            i = np.arange(n - shift)
            grad[i] += (1.0 - h[i + shift])
            grad[i + shift] -= h[i]
        else:
            s = -shift
            i = np.arange(s, n)
            grad[i] += (1.0 - h[i - s])
            grad[i - s] -= h[i]
    if shifts.size > 0:
        grad /= shifts.size
    return grad

            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, target_sum)
            cur_score = compute_c5_fft(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Small mass transfers between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # Transfer from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5_fft(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        # Transfer from j to i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5_fft(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    """Transfer mass from a point that contributes most to a worst lag."""
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = full_correlation_fft(best_h, 1.0 - best_h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            i_range = np.arange(n - shift)
            contrib = best_h[i_range] * (1.0 - best_h[i_range + shift])
            i_local = np.argmax(contrib)
            i = i_local
            j = i_local + shift
        else:
            s = -shift
            i_range = np.arange(s, n)
            contrib = best_h[i_range] * (1.0 - best_h[i_range - s])
            i_local = np.argmax(contrib) + s
            i = i_local
            j = i_local - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)
        if new_score < best_score:
            best_h, best_score = h_new, new_score
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Initial vector (64 points)
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        # Upsample if needed
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Gradient descent on this scale
        if time_left() > 5:
            max_iters = int(min(50000,
                                max(5000, time_left() * (20000.0 / n))))
            h_gd, sc_gd = gradient_descent_adaptive(base, max_iters,
                                                    step0=0.8, beta=0.9,
                                                    rng=rng, time_left=time_left)
            if sc_gd < best_score:
                best_h, best_score = h_gd.copy(), sc_gd
            base = h_gd

        # Targeted max‑shift improvement
        if time_left() > 5:
            h_ms, sc_ms = improve_via_max_shift(base, rng, time_left)
            if sc_ms < best_score:
                best_h, best_score = h_ms.copy(), sc_ms
            base = h_ms

        # Local pair‑wise moves
        if time_left() > 5:
            h_pair, sc_pair = local_pair_moves(base, rng, time_left)
            if sc_pair < best_score:
                best_h, best_score = h_pair.copy(), sc_pair
            base = h_pair

        # Random balanced binary restarts with brief refinement
        if time_left() > 5:
            restarts = 2 if n <= 512 else 1
            for _ in range(restarts):
                if time_left() < 2:
                    break
                h_rand = random_binary_balanced(n, rng)
                max_iters_rand = int(min(4000,
                                         max(400, time_left() * (3000.0 / n))))
                h_rand_ref, sc_rand = gradient_descent_adaptive(
                    h_rand, max_iters_rand, step0=0.5, beta=0.9,
                    rng=rng, time_left=time_left)
                if sc_rand < best_score:
                    best_h, best_score = h_rand_ref.copy(), sc_rand
                if time_left() > 5:
                    h_rand2, sc_rand2 = local_pair_moves(h_rand_ref, rng, time_left)
                    if sc_rand2 < best_score:
                        best_h, best_score = h_rand2.copy(), sc_rand2

    # Final intensive refinement on the global best
    if time_left() > 5:
        max_iters_final = int(min(80000,
                                  max(20000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(
            best_h, max_iters_final, step0=0.3, beta=0.9,
            rng=rng, time_left=time_left)

    # One more round of targeted and pair moves
    if time_left() > 2:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
