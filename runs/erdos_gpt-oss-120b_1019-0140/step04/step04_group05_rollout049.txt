def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam_mid, lo, hi)
        s = x.sum()
        if np.abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def simulated_annealing(h0, time_limit, rng, init_temp=0.05, decay_power=1.0):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # Determine max_iters as int of time limit * something? We'll loop while time left > small epsilon.
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        # temperature schedule: T = init_temp * (1 - t_frac) ** decay_power
        T = init_temp * (1 - t_frac) ** decay_power
        # delta factor schedule: we can use same as T, or a higher early delta.
        delta_factor = (1 - t_frac) ** 0.5  # decays slower
        # pick i,j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * delta_factor
        if delta <= 0.0:
            continue
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        # accept if better, else probabilistic
        if new_score < cur_score or (T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T)):
            cur_score = new_score
            # update best
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        # early exit if best_score <= target
        # we could use global target constant?
    return best_h, best_score

def local_refine(h, rng, max_iter=1000):

        j = rng.integers(n-1)
        if j >= i:

            best_score = sc
            best_h = tmp_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # generate Thue-Morse of length 64 as fallback
        def thue_morse(n):
            seq = np.empty(n, dtype=float)
            for i in range(n):
                seq[i] = (bin(i).count('1') & 1)
            return seq
        base = thue_morse(64)
    # Project base to satisfy constraints
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # resolution sequence and time fractions
    resolutions = [64, 128, 256, 512, 1024]  # we may stop early.
    # time allocation per resolution: we allocate proportionally to size (more time for higher res) but also ensure remaining time accounted.
    # Let's allocate equal fractions initially and then use leftover for refinement.
    total_levels = len(resolutions)
    for idx, n in enumerate(resolutions):
        if time_left() <= 0:
            break
        # Upsample or downsample best_h to size n
        if best_h.size != n:
            # interpolation of best_h
            # Use np.interp
            old_n = best_h.size
            x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            best_h = np.interp(x_new, x_old, best_h)
        # ensure constraints
        best_h = project_onto_box_sum(best_h, n / 2.0)
        # compute score
        cur_score = compute_c5(best_h)
        if cur_score < best_score:
            best_score = cur_score
            best_h = best_h.copy()
        # allocate time for this level
        # we assign fraction = remaining time * (1/(remaining_levels+1)) maybe
        remaining_levels = total_levels - idx
        alloc = time_left() * 0.7 / remaining_levels  # allocate 70% of remaining time across remaining levels
        # if not enough, allocate remaining time
        if alloc > time_left():
            alloc = time_left()
        # do several random restarts with SA
        restarts = 5
        # ensure per restart time
        per_restart = max(alloc / restarts, 0.01)
        for r in range(restarts):
            if time_left() <= 0.01:
                break
            # generate random balanced continuous vector
            rand_vec = rng.random(n)
            rand_vec = project_onto_box_sum(rand_vec, n / 2.0)
            # run SA
            sa_time = min(per_restart, time_left())
            cand_h, cand_score = simulated_annealing(rand_vec, sa_time, rng,
                                                     init_temp=0.07, decay_power=1.2)
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand_h.copy()
                # early exit if target achieved
                if best_score <= 0.3808:
                    break
        if best_score <= 0.3808:
            break
        # also refine current best at this resolution
        # allocate small leftover time for local refine
        refine_time = min(0.2 * time_left(), 5.0)  # up to 5 seconds
        # we can repeatedly apply local_refine until time exhausted
        refine_start = time.time()
        while time.time() - refine_start < refine_time and time_left() > 0.1:
            # small local refine
            best_h, best_score = local_refine(best_h, rng, max_iter=200)
            if best_score <= 0.3808:
                break
        if best_score <= 0.3808:
            break
    # final projection
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam_mid, lo, hi)
        s = x.sum()
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid via FFT."""
    n = h.shape[0]
    m = 2 * n
    H = np.fft.fft(h, m)
    G = np.fft.fft(1.0 - h, m)
    corr = np.fft.ifft(H * np.conj(G)).real
    # linear correlation length = 2n-1, ignore the trailing zero
    return np.max(corr[:2 * n - 1]) * (2.0 / n)


def simulated_annealing(h0, time_limit, rng, init_temp=0.07, decay_power=0.8):
    """SA that preserves sum and box constraints, with occasional swaps."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    swap_prob = 0.1
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        T = init_temp * (1.0 - t_frac) ** decay_power
        delta_factor = T / init_temp if init_temp > 0 else 0.0
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if rng.random() < swap_prob:
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            if new_score < cur_score or (T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T)):
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * delta_factor
        if delta <= 0.0:
            continue
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        if new_score < cur_score or (T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
    return best_h, best_score


def local_refine(h, rng, max_iter=500):
    """Greedy small transfers to improve current best."""
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for _ in range(max_iter):
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.05
        if delta <= 0.0:
            continue
        tmp = best_h.copy()
        tmp[i] -= delta
        tmp[j] += delta
        sc = compute_c5(tmp)
        if sc < best_score:
            best_score = sc
            best_h = tmp
            if best_score <= 0.3808:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # Thue–Morse fallback (length 64)
        def thue_morse(n):
            seq = np.empty(n, dtype=float)
            for i in range(n):
                seq[i] = (bin(i).count('1') & 1)
            return seq
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    resolutions = [64, 128, 256, 512, 1024]
    total_levels = len(resolutions)

    for lvl, n in enumerate(resolutions):
        if time_left() <= 0.0:
            break
        if best_h.size != n:
            # linear upsampling preserving shape on [0,1) interval
            old_n = best_h.size
            x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            best_h = np.interp(x_new, x_old, best_h)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        cur = compute_c5(best_h)
        if cur < best_score:
            best_score = cur
            best_h = best_h.copy()

        remaining_levels = total_levels - lvl
        alloc = time_left() * 0.7 / remaining_levels
        if alloc > time_left():
            alloc = time_left()
        # number of SA restarts
        restarts = 5
        per_restart = max(alloc / restarts, 0.02)
        init_temp = 0.1 if n >= 256 else 0.07
        for _ in range(restarts):
            if time_left() <= 0.01:
                break
            rand_vec = rng.random(n)
            rand_vec = project_onto_box_sum(rand_vec, n / 2.0)
            sa_time = min(per_restart, time_left())
            cand_h, cand_score = simulated_annealing(
                rand_vec, sa_time, rng, init_temp=init_temp, decay_power=0.8)
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand_h.copy()
                if best_score <= 0.3808:
                    break
        if best_score <= 0.3808:
            break

        # local refinement with remaining budget for this level
        refine_budget = min(0.2 * time_left(), 5.0)
        refine_start = time.time()
        while time.time() - refine_start < refine_budget and time_left() > 0.1:
            best_h, best_score = local_refine(best_h, rng, max_iter=200)
            if best_score <= 0.3808:
                break
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
