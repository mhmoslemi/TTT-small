def compute_c5(h):
    n = h.shape[0]
    # compute correlation = np.correlate(h, 1-h, mode='full')
    # using FFT zero-padded to 2*n
    a = np.fft.rfft(h, n*2)

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

            total_steps = int(min(50000, max(10000, remaining * 400.0)))
            T0 = 0.02
            T_final = 1e-6
            for step in range(total_steps):
                if budget_s - (time.time() - start) <= 0.0:
                    break
                T = T0 * ((total_steps - step) / total_steps) + T_final
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(h_cur[i], 1.0 - h_cur[j])
                if max_move <= 0.0:
                    continue
                delta = max_move * rng.random()
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_score = compute_c5(h_new)
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                    h_cur = h_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score, best_h = cur_score, h_cur.copy()
            base = best_h

        # prepare for next scale
        if idx < len(scales) - 1:
            if best_h is not None:
                base = best_h.copy()
            # next loop will upsample as needed

    # final refinement at full resolution
    if best_h is not None and best_h.size == n_final and (budget_s - (time.time() - start)) > 2.0:
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(500000, max(50000, remaining * 3000)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=max_iters, step0=0.3, beta=0.9,
            rng=rng, max_time=remaining)
        # final annealing pass
        if budget_s - (time.time() - start) > 2.0:
            h_cur = best_h.copy()
            cur_score = best_score
            remaining = budget_s - (time.time() - start)
            total_steps = int(min(30000, max(5000, remaining * 250.0)))
            T0 = 0.015
            T_final = 1e-6
            for step in range(total_steps):
                if budget_s - (time.time() - start) <= 0.0:
                    break
                T = T0 * ((total_steps - step) / total_steps) + T_final
                i, j = rng.choice(n_final, size=2, replace=False)
                max_move = min(h_cur[i], 1.0 - h_cur[j])
                if max_move <= 0.0:
                    continue
                delta = max_move * rng.random()
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_score = compute_c5(h_new)

                    h_cur = h_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score, best_h = cur_score, h_cur.copy()
        best_h = project_onto_box_sum(best_h, n_final / 2.0)
        best_score = compute_c5(best_h)

    if best_h is None:
        n = n_final
        best_h = project_onto_box_sum(random_binary_balanced(n, rng), n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
