import time
import numpy as np


def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def gradient_of_corr(h, lag):
    """Gradient of the correlation term at a given lag."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if lag >= 0:
        # i = 0 .. n-lag-1, i2 = i+lag
        grad[: n - lag] += 1.0 - h[lag:]
        grad[lag:] -= h[: n - lag]
    else:
        lag = -lag
        # i = lag .. n-1, i2 = i-lag
        grad[lag:] += 1.0 - h[: n - lag]
        grad[: n - lag] -= h[lag:]
    return grad


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def subgradient_descent(h0, time_limit, rng):
    """Projected subgradient descent on the max-overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    start = time.time()
    it = 0
    step0 = 0.3
    beta = 0.9
    velocity = np.zeros_like(h)

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        cur_score = max_corr * (2.0 / n)

        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()

        lag = max_idx - (n - 1)
        grad = gradient_of_corr(h, lag)

        it += 1
        step = step0 / np.sqrt(it)
        velocity = beta * velocity - step * grad
        h_new = h + velocity
        h_new = project_onto_box_sum(h_new, n / 2.0)
        h = h_new

    return best_h, best_score
