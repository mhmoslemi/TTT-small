def compute_c5_fft(h):
    n = h.shape[0]
    N = 2*n
    h_pad = np.zeros(N, dtype=h.dtype)
    one_minus_h_pad = np.zeros(N, dtype=h.dtype)
    h_pad[:n] = h
    one_minus_h_pad[:n] = 1.0 - h
    # FFT
    H = np.fft.rfft(h_pad)
    G = np.fft.rfft(one_minus_h_pad)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    # Use full correlation: length 2*n-1; this includes indices 0..2*n-2
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_fft(h):
    """Linear convolution of h with (1-h) using FFT (no scaling)."""
    n = h.shape[0]
    N = 2 * n
    h_pad = np.empty(N, dtype=h.dtype)
    t_pad = np.empty(N, dtype=h.dtype)
    h_pad[:n] = h
    h_pad[n:] = 0.0
    t_pad[:n] = 1.0 - h
    t_pad[n:] = 0.0
    H = np.fft.rfft(h_pad)
    T = np.fft.rfft(t_pad)
    corr = np.fft.irfft(H * np.conj(T), n=N)
    return corr[:2 * n - 1]


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with adaptive step size."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fft(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5_fft(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    """Transfer mass from a point that contributes most to the worst lag."""
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = compute_corr_fft(best_h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Transfer small mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5_fft(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5_fft(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def simulated_annealing(h0, rng, time_left):
    """Simple SA moving mass between two points while preserving sum."""
    h = h0.copy()
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    total = time_left()
    T0 = cur_score * 0.5 if cur_score > 0 else 0.1

    while time_left() > 0.2:
        elapsed = time.time() - start
        fraction = elapsed / total if total > 0 else 1.0
        T = max(0.0, T0 * (1.0 - fraction))
        i, j = rng.integers(0, h.shape[0], size=2)
        if i == j:
            continue
        max_delta_i = h[i]
        max_delta_j = 1.0 - h[j]
        if max_delta_i <= 0 or max_delta_j <= 0:
            continue
        delta = min(max_delta_i, max_delta_j) * rng.random() * 0.5
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5_fft(h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)

    # target resolution
    n_target = 1024
    if time_left() < 40.0:
        n_target = 512
    base = upsample(base, n_target)
    base = project_onto_box_sum(base, n_target / 2.0)

    # gradient descent phase
    if time_left() > 5.0:
        remaining = time_left()
        max_iters = int(min(30000, max(5000, remaining * 30)))
        base, _ = gradient_descent_adaptive(base, max_iters, step0=0.6, beta=0.9,
                                            rng=rng, time_left=time_left)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    # simulated annealing phase
    if time_left() > 5.0:
        best_h, best_score = simulated_annealing(best_h, rng, time_left)

    # targeted max‑shift improvements
    if time_left() > 5.0:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)

    # local pair‑wise fine tuning
    if time_left() > 5.0:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
