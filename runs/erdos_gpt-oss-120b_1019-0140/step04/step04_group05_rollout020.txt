    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.size/2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)
    scales = [64, 128, 256, 512, 1024]
    # fractions for each scale in terms of total budget
    scale_fractions = {64:0.05, 128:0.07, 256:0.12, 512:0.3, 1024:0.46}
    for n in scales:
        if time_left() <= 0.0:
            break
        # upscale if needed
        if base.shape[0] != n:
            base = upsample(base, n)
        # project to ensure sum exactly
        base = project_onto_box_sum(base, n/2.0)
        # DE optimization on this scale
        alloc = scale_fractions[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        if alloc < 5.0:  # need minimal time
            continue
        # solve using DE
        h_opt, score_opt = de_optimize(base, n, alloc, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
        base = h_opt
    # final local improvement with SA for leftover time
    while time_left() > 0.6:
        # random perturb and local hill climbing
        h_new = best_h.copy()
        # random pair transfer
        n = h_new.size
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_delta = min(h_new[i], 1.0 - h_new[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_onto_box_sum(h_new, n/2.0)

    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    rev = (1.0 - h)[::-1]
    n_fft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    R = np.fft.rfft(rev, n=n_fft)
    conv = np.fft.irfft(H * R, n=n_fft)
    # take first 2*n-1 values.
    val = np.max(conv[:2*n-1])
    return val * dx

def de_optimize(base, n, time_budget, rng):
    start = time.time()
    popsize = min(40, max(10, n // 25))
    # initial population
    pop = np.empty((popsize, n), dtype=float)
    for i in range(popsize):
        # add small noise
        noise = rng.random(n) * 0.2  # amplitude
        vec = base + noise - 0.1  # shift noise to [-0.1,0.1]
        pop[i] = project_onto_box_sum(vec, n/2.0)
    scores = np.empty(popsize, dtype=float)
    for i in range(popsize):
        scores[i] = compute_c5(pop[i])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # DE parameters
    CR = 0.9
    while time.time() - start < time_budget:
        # choose target
        target = rng.integers(popsize)
        # choose distinct a,b,c
        idxs = rng.choice(popsize, size=3, replace=False)
        while target in idxs:
            idxs = rng.choice(popsize, size=3, replace=False)
        a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
        F = 0.5 + 0.3 * rng.random()
        mutant = a + F * (b - c)
        # crossover
        j_rand = rng.integers(n)
        mask = rng.random(n) < CR
        mask[j_rand] = True
        trial = np.where(mask, mutant, pop[target])
        # enforce bounds and sum
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_onto_box_sum(trial, n/2.0)
        trial_score = compute_c5(trial)
        if trial_score < scores[target]:
            pop[target] = trial
            scores[target] = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_h = trial.copy()
        # early exit if reach target bound
        if best_score <= 0.3808:
            break
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap C5 for a discrete h on [0,2]. Uses FFT for speed."""
    n = h.shape[0]
    dx = 2.0 / n
    rev = (1.0 - h)[::-1]
    n_fft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    R = np.fft.rfft(rev, n=n_fft)
    conv = np.fft.irfft(H * R, n=n_fft)
    return np.max(conv[: 2 * n - 1]) * dx


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def de_optimize(base, n, time_budget, rng):
    """Differential Evolution with projection, returns best (h,score)."""
    start = time.time()
    popsize = min(40, max(10, n // 25))
    pop = np.empty((popsize, n), dtype=float)
    for i in range(popsize):
        # small perturbation around base
        pert = base + (rng.random(n) - 0.5) * 0.2
        pop[i] = project_onto_box_sum(pert, n / 2.0)
    scores = np.empty(popsize, dtype=float)
    for i in range(popsize):
        scores[i] = compute_c5(pop[i])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    CR = 0.9
    while time.time() - start < time_budget:
        target = rng.integers(popsize)

        # choose mutation scheme
        if rng.random() < 0.3:  # DE/best/1
            # select two distinct donors not equal to target
            idxs = rng.choice(popsize, size=2, replace=False)
            while target in idxs:
                idxs = rng.choice(popsize, size=2, replace=False)
            b, c = pop[idxs[0]], pop[idxs[1]]
            a = best_h
        else:  # DE/rand/1
            idxs = rng.choice(popsize, size=3, replace=False)
            while target in idxs:
                idxs = rng.choice(popsize, size=3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]

        F = 0.5 + 0.3 * rng.random()
        mutant = a + F * (b - c)

        # binomial crossover
        j_rand = rng.integers(n)
        mask = rng.random(n) < CR
        mask[j_rand] = True
        trial = np.where(mask, mutant, pop[target])

        trial = np.clip(trial, 0.0, 1.0)
        trial = project_onto_box_sum(trial, n / 2.0)
        trial_score = compute_c5(trial)

        if trial_score < scores[target]:
            pop[target] = trial
            scores[target] = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_h = trial.copy()
                if best_score <= 0.3808:
                    break
        if best_score <= 0.3808:
            break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # base construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]
    scale_fractions = {64: 0.05, 128: 0.07, 256: 0.12, 512: 0.30, 1024: 0.46}

    for n in scales:
        if time_left() <= 0.0 or best_score <= 0.3808:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        alloc = scale_fractions[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        if alloc < 5.0:
            continue

        h_opt, sc_opt = de_optimize(base, n, alloc, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        base = h_opt

    # final short local search with remaining budget
    while time_left() > 0.6 and best_score > 0.3808:
        n = best_h.size
        # small mass transfer
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0.0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_onto_box_sum(h_new, n / 2.0)
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

        # occasional binary restart
        if rng.random() < 0.02:
            h_rand = random_balanced_binary(n, rng)
            sc_rand = compute_c5(h_rand)
            if sc_rand < best_score:
                best_score = sc_rand
                best_h = h_rand

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
