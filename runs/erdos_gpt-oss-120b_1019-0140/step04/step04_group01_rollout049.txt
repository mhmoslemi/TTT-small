def adam_gradient_descent(h0, time_left_func, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=10**7):

    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)

    it = 1
    while time_left_func()>0 and it <= max_iter:
        # compute correlation
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # get shifts within epsilon of max_corr
        eps_shift = 1e-9  # or a relative tolerance
        shifts = np.where(corr >= max_corr - eps_shift)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)  # will average among them

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h_new = h - step
        # project
        h_new = project_onto_box_sum(h_new, h.size/2.0)

        cur_score = compute_c5_fft(h_new)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h_new.copy()
        h = h_new
        it += 1
        # maybe adjust lr if needed

    return best_h, best_score

    grad = np.zeros_like(h)
    if shifts.size == 0:

    for shift in shifts:
        if shift >= 0:

            term1[:n - shift] = h[shift:]

            # backward term: h[i-shift] for i >= shift

            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)

    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h = h.copy()

    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # include shifts within small absolute tolerance

        m = beta1 * m + (1 - beta1) * grad

        h = h_new
        it += 1

                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto { lo <= x_i <= hi, sum x_i = s }."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # Find initial brackets for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # Ensure brackets contain root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 for step function defined by samples h."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array (scaled by dx) for h and 1-h."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max correlation with respect to h for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    if shifts.size == 0:
        return grad
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    grad /= shifts.size
    return grad

def adam_gradient_descent(h0, time_left_func, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=10**7):
    """Projected Adam optimizer on C5 (max correlation) subgradient."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # include shifts whose correlation is within a tiny absolute tolerance
        eps_shift = 1e-8
        shift_idxs = np.where(corr >= max_corr - eps_shift)[0]
        shifts = shift_idxs - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=10**9, T0=0.5, decay=0.9999, Tmin=1e-8):
    """Simulated annealing on binary vectors with swap moves preserving sum."""
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                # avoid overflow in exp
                exp_arg = -delta / T
                if exp_arg > -700:
                    if rng.random() < np.exp(exp_arg):
                        accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto { lo ≤ x_i ≤ hi, Σx_i = s }."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C₅ for a step function sampled by h."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array (scaled by dx) between h and 1‑h."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of the max‑correlation objective for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    if shifts.size == 0:
        return grad
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    grad /= shifts.size
    return grad

def adam_gradient_descent(h0, time_left_func, lr=0.05, beta1=0.9, beta2=0.999,
                         eps=1e-8, max_iter=10**7):
    """Projected Adam optimizer on the C₅ sub‑gradient."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    b1_pow, b2_pow = 1.0, 1.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        eps_shift = 1e-8
        shift_idx = np.where(corr >= max_corr - eps_shift)[0]
        shifts = shift_idx - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        b1_pow *= beta1
        b2_pow *= beta2
        m_hat = m / (1 - b1_pow)
        v_hat = v / (1 - b2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

            best_h = h_new.copy()
        h = h_new
        it += 1

def binary_swap_anneal(h0, rng, time_left_func, max_iters=10**9,
                      T0=0.5, decay=0.9999, Tmin=1e-8):

    best_h = h.copy()
    best_score = cur_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        h[j] = 1.0
        new_score = compute_c5_fft(h)
