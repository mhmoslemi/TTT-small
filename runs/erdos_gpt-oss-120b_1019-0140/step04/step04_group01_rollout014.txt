def compute_gradient(h, alpha, M):
    # compute H = rfft(h, n=M)
    H = np.fft.rfft(h, n=M)
    # J = correlation of h with h
    corr = np.fft.irfft(H * np.conj(H), n=M)[:2*len(h)-1]
    J = corr
    # softmin weights p:
    # to avoid overflow we compute exponent = -alpha * J, subtract max.
    exp_vals = np.exp(-alpha * J - np.max(-alpha * J))
    p = exp_vals / np.sum(exp_vals)
    # compute convolution of p with h (full) using FFT
    P = np.fft.rfft(p, n=M)
    conv_full = np.fft.irfft(P * H, n=M)
    conv_slice = conv_full[len(h)-1:len(h)-1+len(h)]
    # reverse p
    p_rev = p[::-1]
    P_rev = np.fft.rfft(p_rev, n=M)
    conv_rev_full = np.fft.irfft(P_rev * H, n=M)
    conv_rev_slice = conv_rev_full[len(h)-1:len(h)-1+len(h)]
    grad = - (conv_slice + conv_rev_slice)
    return grad

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Exact C5 via FFT (dx = 2/n)."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def next_pow2(x):
    return 1 << (x - 1).bit_length()

def compute_gradient(h, alpha, M):
    """Gradient of smooth min of correlations (negative direction)."""
    n = h.shape[0]
    H = np.fft.rfft(h, n=M)
    # full correlation h with h
    corr_full = np.fft.irfft(H * np.conj(H), n=M)
    J = corr_full[:2 * n - 1]                       # length 2n-1
    # soft-min weights
    exp_arg = -alpha * J
    max_arg = np.max(exp_arg)
    p = np.exp(exp_arg - max_arg)
    p_sum = p.sum()
    if p_sum == 0:
        p = np.full_like(p, 1.0 / p.size)
    else:
        p /= p_sum
    # convolution p * h
    P = np.fft.rfft(p, n=M)
    conv_full = np.fft.irfft(P * H, n=M)
    conv = conv_full[n - 1:n - 1 + n]
    # convolution reverse(p) * h
    p_rev = p[::-1]
    P_rev = np.fft.rfft(p_rev, n=M)
    conv_rev_full = np.fft.irfft(P_rev * H, n=M)
    conv_rev = conv_rev_full[n - 1:n - 1 + n]
    return - (conv + conv_rev)

def binary_swap_anneal(h0, rng, time_left, max_iters=300000,
                      T0=0.2, decay=0.9999, Tmin=1e-5):
    h = h0.copy()
    cur_score = compute_c5_fft(h)
    best_h, best_score = h.copy(), cur_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.5:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

        if best_score <= target:

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, n
