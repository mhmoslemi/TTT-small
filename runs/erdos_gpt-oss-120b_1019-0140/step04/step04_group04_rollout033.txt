def gradient_descent_momentum(h0, time_limit, max_iter=200000, step0=0.5, beta=0.95):
    start = time.time()
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 0
    while time.time() - start < time_limit and it < max_iter:
        it += 1
        grad = compute_gradient_max_corr(h)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

        if delta > 0.0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5_fft(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new, sc_new
        # occasional Gaussian perturbation
        if rng.random() < 0.2:
            pert = rng.normal(scale=0.02, size=n_curr)
            h_new = best_h + pert
            h_new = project_onto_box_sum(h_new, n_curr / 2.0)
            sc_new = compute_c5_fft(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new, sc_new

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, float(best_score), best_h.size
