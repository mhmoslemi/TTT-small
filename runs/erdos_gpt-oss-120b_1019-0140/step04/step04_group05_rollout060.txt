def compute_c5(h):
    n = h.shape[0]
    hx = h
    one_minus_h = 1.0 - h
    # perform convolution via FFT of length 2n-1 (next power of 2)
    size = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(hx, n=size)
    O = np.fft.rfft(one_minus_h, n=size)
    conv = np.fft.irfft(H * O, n=size)[:2*n-1]
    return np.max(conv) * (2.0 / n)

    # base h

    else:

    schedule = [128, 256, 512, 1024]

        if time_left() < 1.0:
            break
        # upsample
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n/2.0)

            # simple local random move loop
            # perform a few random transfer moves
            moves = int(remaining * 2000)  # controls moves per second approx
            h = best_h.copy()
            cur_score = best_score
            for _ in range(moves):
                i = rng.integers(n)
                j = rng.integers(n-1)
                if j >= i:
                    j += 1
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0.0:
                    continue
                delta = max_delta * rng.random() * 0.5
                h[i] -= delta
                h[j] += delta
                sc = compute_c5(h)

                    cur_score = sc
                    if sc < best_score:
                        best_score = sc
                        best_h = h.copy()
                else:

                    h[i] += delta
                    h[j] -= delta
                if best_score <= 0.3808:
                    break

import time, numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, n=size)
    O = np.fft.fft(1.0 - h, n=size)
    conv = np.fft.ifft(H * O).real[:2 * n - 1]
    return np.max(conv) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def autocorr_full(h):
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, n=size)
    conv = np.fft.ifft(H * H.conj()).real[:2 * n - 1]
    return conv * (2.0 / n)

def gradient_ascent(h, time_limit, rng):
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    init_alpha = 0.02
    while time.time() - start < time_limit:
        A = autocorr_full(h)
        min_val = A.min()
        mask = np.isclose(A, min_val, atol=1e-12)
        shifts = np.where(mask)[0] - (n - 1)
        g = np.zeros_like(h)
        for k in shifts:
            if k >= 0:
                i = np.arange(0, n - k, dtype=int)
                g[i] += h[i + k]
                g[i + k] += h[i]
            else:
                kp = -k
                i = np.arange(kp, n, dtype=int)
                i2 = i - kp
                g[i] += h[i2]
                g[i2] += h[i]
        elapsed = time.time() - start
        remain = time_limit - elapsed
        alpha = init_alpha * (remain / time_limit) if remain > 0 else 1e-6
        alpha = max(alpha, 1e-6)
        h_new = h + alpha * g
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 64
        base = rng.random(n0)
        base = project_onto_box_sum(base, n0 / 2.0)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)
    sizes = [128, 256, 512, 1024]
    fractions = {128: 0.15, 256: 0.20, 512: 0.25, 1024: 0.40}
    for n in sizes:
        if time_left() < 0.6:
            break
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        alloc = fractions[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        cand_h, cand_score = gradient_ascent(best_h, alloc, rng)
        if cand_score < best_score:
            best_score, best_h = cand_score, cand_h.copy()
        remaining = time_left()
        if remaining > 0.5 and best_score > 0.3808:
            moves = int(remaining * 2000)
            h = best_h.copy()
            cur_score = best_score
            for _ in range(moves):
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0.0:
                    continue
                delta = max_delta * rng.random() * 0.5
                h[i] -= delta
                h[j] += delta
                sc = compute_c5(h)
                if sc < cur_score:
                    cur_score = sc
                    if sc < best_score:
                        best_score, best_h = sc, h.copy()
                else:
                    h[i] += delta
                    h[j] -= delta
                if best_score <= 0.3808:
                    break
        if best_score <= 0.3808:
            break
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
