def slsqp_refine(h0, time_left):
    n = h0.size
    # define objective and gradient
    def obj(h):
        return compute_c5(h)
    def jac(h):
        # gradient = compute_gradients_for_shifts(h, shifts) * (2.0 / n)
        n = h.size
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        shift_indices = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        shifts = shift_indices - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts) * (2.0 / n)
        return grad
    # constraints
    cons = {'type':'eq', 'fun': lambda h: np.sum(h) - n/2.0,
            'jac': lambda h: np.ones_like(h)}
    bounds = [(0.0, 1.0)] * n
    # Use minimize
    from scipy.optimize import minimize
    # set maxiter based on time left
    maxiter = int(min(500, max(50, time_left() * 100)))
    res = minimize(obj, h0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    score = compute_c5(h_opt)
    return h_opt, score

import time
import numpy as np
from scipy.optimize import minimize

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            part1 = np.concatenate([h[shift:], np.zeros(shift)])
            part2 = np.concatenate([np.zeros(shift), h[:n - shift]])
        else:
            s = -shift
            part1 = np.concatenate([np.zeros(s), h[:n - s]])
            part2 = np.concatenate([h[s:], np.zeros(s)])
        grad += 1.0 - part1 - part2
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def slsqp_refine(h0, time_left):
    n = h0.size
    def obj(h):
        return compute_c5(h)
    def jac(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shift_idx = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        shifts = shift_idx - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts) * (2.0 / n)
        return grad
    cons = {'type': 'eq',
            'fun': lambda h: np.sum(h) - n / 2.0,
            'jac': lambda h: np.ones_like(h)}
    bounds = [(0.0, 1.0)] * n
    maxiter = int(min(400, max(30, time_left() * 80)))
    res = minimize(obj, h0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]
    for n in scales:
        if time_left() <= 2:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)
        max_iters = int(min(20000, max(2000, time_left() * (5000.0 / n))))
        best_h, best_score = gradient_descent_adaptive(best_h, max_iters,
                                                       step0=0.6, beta=0.9,
                                                       rng=rng, time_left=time_left)
        if time_left() > 2:
            best_h, best_score = local_pair_moves(best_h, rng, time_left)
        for _ in range(2):
            if time_left() < 3:
                break
            h_rand = random_binary_balanced(n, rng)
            h_rand = project_onto_box_sum(h_rand, n / 2.0)
            iters = int(min(5000, max(500, time_left() * (2000.0 / n))))
            h_rand, sc_rand = gradient_descent_adaptive(h_rand, iters,
                                                        step0=0.5, beta=0.9,
                                                        rng=rng, time_left=time_left)
            if sc_rand < best_score:
                best_h, best_score = h_rand, sc_rand

    if time_left() > 5:
        best_h, best_score = slsqp_refine(best_h, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
