def solve_milp_binary(n, time_limit):
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == n/2]
    for s in range(n):
        # overlapping length
        length = n - s
        if length == 0:
            # No overlap, correlation 0. So t >= 0.
            constraints.append(t >= 0)
            continue
        y = cp.Variable(length, boolean=True)
        # linking constraints
        constraints.append(y <= x[:length])
        constraints.append(y <= x[s:])
        constraints.append(y >= x[:length] + x[s:] - 1)
        weight_overlap = cp.sum(x[:length])
        constraints.append(t >= weight_overlap - cp.sum(y))
    prob = cp.Problem(cp.Minimize(t), constraints)
    # solver selection
    best_result = None
    best_status = None
    for solver in (cp.CBC, cp.GLPK, cp.ECOS_BB):
        try:
            opts = {}
            if solver == cp.CBC:
                opts = {"max_seconds": time_limit}
            elif solver == cp.GLPK:
                # GLPK expects tm_lim in milliseconds
                opts = {"tm_lim": int(time_limit * 1000)}
            # else ECOS_BB may be slower; set max_iters
            elif solver == cp.ECOS_BB:
                opts = {"max_iters": int(time_limit * 1e5)}  # guess
            prob.solve(solver=solver, **opts)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                best_result = (x.value, t.value)
                best_status = prob.status
                break
        except Exception:
            continue
    if best_result is None:
        return None, None
    x_val, t_val = best_result
    h = np.rint(x_val).astype(float)
    return h, t_val
```

def gd_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score
```

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            length = n - shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:length] = h[shift:]
            term2[shift:] = h[:length]
        else:
            s = -shift
            length = n - s
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:length]
            term2[:length] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gd_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def solve_milp_binary(n, time_limit):
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == n / 2.0]
    for s in range(1, n):
        length = n - s
        y = cp.Variable(length, boolean=True)
        # product linking
        constraints.append(y <= x[:length])
        constraints.append(y <= x[s:])
        constraints.append(y >= x[:length] + x[s:] - 1)
        weight_pos = cp.sum(x[:length])
        weight_neg = cp.sum(x[s:])
        constraints.append(t >= weight_pos - cp.sum(y))
        constraints.append(t >= weight_neg - cp.sum(y))
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in (cp.CBC, cp.GLPK, cp.ECOS_BB):
        try:
            opts = {}
            if solver == cp.CBC:
                opts = {"max_seconds": time_limit}
            elif solver == cp.GLPK:
                opts = {"tm_lim": int(time_limit * 1000)}
            prob.solve(solver=solver, **opts)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                xv = x.value
                if xv is None:
                    continue
                h = np.rint(xv).astype(float)
                return h, t.value
        except Exception:
            continue
    return None, None
