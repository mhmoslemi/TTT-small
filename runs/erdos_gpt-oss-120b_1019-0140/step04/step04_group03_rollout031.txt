def gradient_descent(h0, max_iters, step0, beta, eps_factor, rng, max_time):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time.time() - start >= max_time:
            break
        # compute correlations
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = np.max(corr)
        # select shifts within epsilon
        eps = max_corr * eps_factor
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n-1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # Optional small random noise occasionally
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h0, cur_score, total_steps, rng, start, max_time):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = cur_score
    for step in range(total_steps):
        if time.time() - start >= max_time:
            break
        # Temperature linear schedule
        T = 0.01 * (1 - step/total_steps) + 1e-6
        i, j = rng.choice(n, size=2, replace=False)
        # compute maximum allowed delta
        max_move = min(h[i], 1 - h[j])  # moving from i to j (decrease i, increase j)
        if max_move <= 0:
            continue
        # propose delta scaling
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score
```

def compute_corr_fft(h):
    n = h.size
    g_rev = (1.0 - h)[::-1]
    size = 2 * n - 1
    fft_h = np.fft.fft(h, size)
    fft_rev = np.fft.fft(g_rev, size)
    corr = np.fft.ifft(fft_h * fft_rev).real
    return corr

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - target_sum) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    n = h.size
    rev_g = (1.0 - h)[::-1]
    size = 2 * n - 1
    fft_h = np.fft.fft(h, size)
    fft_rev = np.fft.fft(rev_g, size)
    corr = np.fft.ifft(fft_h * fft_rev).real
    return corr

def compute_c5(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, max_iters, step0, beta, eps_factor, rng, start_time, max_time):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while it < max_iters:
        if time.time() - start_time >= max_time:
            break
        it += 1
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        eps = max_corr * eps_factor
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h0, cur_score, total_steps, rng, start_time, max_time):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = cur_score
    for step in range(total_steps):
        if time.time() - start_time >= max_time:
            break
        T = 0.01 * (1.0 - step / total_steps) + 1e-6
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score
