def gradient_refine(h, max_iter, step0, beta, rng, time_left):
    n = h.size; target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts with max_corr (within tolerance)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros(n, dtype=h.dtype)
                term1[:n - shift] = h[shift:]
                term2 = np.zeros(n, dtype=h.dtype)
                term2[shift:] = h[:n - shift]
            else:
                s = -shift
                term1 = np.zeros(n, dtype=h.dtype)
                term1[s:] = h[:n - s]
                term2 = np.zeros(n, dtype=h.dtype)
                term2[:n - s] = h[s:]
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project onto box and sum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

        score = compute_c5(h_new)

        h = h_new

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_c5_binary(h):
    n = h.shape[0]
    s = n // 2
    A = np.correlate(h, h, mode='full')
    return s - A.min()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_refine(h, max_iter, step0, beta, rng, time_left):
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    h[idx] = 1
    return h

    cur_A = np.correlate(cur_h, cur_h, mode='full')
    best_h = cur_h.copy()
    best_score = s - cur_A.min()
    cur_score = best_score
    L = np.arange(-(n - 1), n, dtype=int)
    d = np.empty(2 * n - 1, dtype=int)
    zero_idx = n - 1
    no_improve = 0
    while time_left() > 0.5:
        p = rng.integers(n)
        while cur_h[p] == 0:
            p = rng.integers(n)
        q = rng.integers(n)
        while cur_h[q] == 1:
            q = rng.integers(n)
        d.fill(0)
        idx = p + L
        mask = (idx >= 0) & (idx < n)
        d[mask] -= cur_h[idx[mask]]
        idx = p - L
        mask = (idx >= 0) & (idx < n)
        d[mask] -= cur_h[idx[mask]]
        idx = q + L
        mask = (idx >= 0) & (idx < n)
        d[mask] += cur_h[idx[mask]]
        idx = q - L
        mask = (idx >= 0) & (idx < n)
        d[mask] += cur_h[idx[mask]]
        d[zero_idx] = 0
        new_A = cur_A + d
        new_score = s - new_A.min()
        if new_score < cur_score - 1e-12:
            cur_h[p] = 0
            cur_h[q] = 1
            cur_A = new_A
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            no_improve += 1
            if no_improve >= 5000:
                for _ in range(5):
                    p2 = rng.integers(n)
                    while cur_h[p2] == 0:
                        p2 = rng.integers(n)
                    q2 = rng.integers(n)
                    while cur_h[q2] == 1:
                        q2 = rng.integers(n)
                    cur_h[p2] = 0
                    cur_h[q2] = 1
                cur_A = np.correlate(cur_h, cur_h, mode='full')
                cur_score = s - cur_A.min()
                no_improve = 0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        base = (base > 0.5).astype(np.int8)
        n0 = base.size
        target0 = n0 // 2
        diff = base.sum() - target0
        if diff > 0:
            idx = np.flatnonzero(base == 1)
            base[rng.choice(idx, size=diff, replace=False)] = 0
        elif diff < 0:
            idx = np.flatnonzero(base == 0)
            base[rng.choice(idx, size=-diff, replace=False)] = 1
        h = base
    else:
        n0 = 64
        h = np.fromiter(((bin(i).count('1') & 1) for i in range(n0)),
                        dtype=np.int8, count=n0)

    best_h = h.copy()
    best_score = compute_c5_binary(best_h)

    scales = [64, 128, 256, 512]
    for n in scales:
        if time_left() < 5:
            break
        if best_h.size != n:
            repeat = n // best_h.size
            best_h = np.repeat(best_h, repeat)
        target = n // 2
        diff = best_h.sum() - target
        if diff > 0:
            idx = np.flatnonzero(best_h == 1)
            best_h[rng.choice(idx, size=diff, replace=False)] = 0
        elif diff < 0:
            idx = np.flatnonzero(best_h == 0)
            best_h[rng.choice(idx, size=-diff, replace=False)] = 1
        best_h, best_score = local_swap_optimize(best_h, time_left, rng)

    if time_left() > 5:
        h_float = best_h.astype(float)
        max_it = int(min(5000, time_left() * 10))
        h_ref, sc_ref = gradient_refine(h_float, max_iter=max_it,
                                        step0=0.3, beta=0.95,
                                        rng=rng, time_left=time_left)
        if sc_ref < best_score:
            best_h = h_ref
            best_score = sc_ref

    while time_left() > 1:
        h_rand = random_balanced_binary(best_h.size, rng)
        sc_rand = compute_c5_binary(h_rand)
        if sc_rand < best_score:
            best_h = h_rand
            best_score = sc_rand

    final_h = best_h.astype(float)
    final_score = compute_c5(final_h)
    return final_h, final_score, final_h.size
