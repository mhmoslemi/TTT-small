def repair_to_target(arr, target_sum, rng):
    # arr is float array 0/1; ensure sum = target_sum (float but integer)
    cur = int(arr.sum())
    target = int(target_sum)
    diff = target - cur
    if diff > 0:
        # need to add ones: select diff zeros randomly
        zeros = np.where(arr == 0)[0]
        if zeros.size >= diff:
            idx = rng.choice(zeros, size=diff, replace=False)
            arr[idx] = 1.0
        else:
            # if not enough zeros (should not happen), set all zeros to ones
            arr[zeros] = 1.0
    elif diff < 0:
        # need to remove ones
        ones = np.where(arr == 1)[0]
        idx = rng.choice(ones, size=-diff, replace=False)
        arr[idx] = 0.0
    return arr

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2*n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n - 1]
    return corr.max() * (2.0 / n)

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n: return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradient_max_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

        if elapsed >= budget_s:
            break

def binary_swap_anneal(h0, rng, start, budget_s, max_iters=400000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    while it < max_iters:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.01:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def repair_to_target(arr, target_sum, rng):
    # arr is binary (0/1) float
    cur = int(arr.sum())
    target = int(target_sum)
    diff = target - cur
    if diff > 0:
        zeros = np.where(arr == 0)[0]
        if zeros.size >= diff:
            idx = rng.choice(zeros, size=diff, replace=False)
            arr[idx] = 1.0
        else:
            # set all zeros to 1
            arr[zeros] = 1.0
    elif diff < 0:
        ones = np.where(arr == 1)[0]
        idx = rng.choice(ones, size=-diff, replace=False)
        arr[idx] = 0.0
    return arr

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

            break
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def binary_swap_anneal(h0, rng, start, budget_s, max_iters=400000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    while it < max_iters:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.01:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def repair_to_target(arr, target_sum, rng):
    cur = int(arr.sum())
    target = int(target_sum)
    diff = target - cur
    if diff > 0:
        zeros = np.where(arr == 0)[0]
        if zeros.size >= diff:
            idx = rng.choice(zeros, size=diff, replace=False)
            arr[idx] = 1.0
        else:
            arr[zeros] = 1.0
    elif diff < 0:
        ones = np.where(arr == 1)[0]
        idx = rng.choice(ones, size=-diff, replace=False)
        arr[idx] = 0.0
    return arr

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    arr = np.zeros(n, dtype=float)
    arr[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return arr

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    n = 512
    target_sum = n / 2
    target_bound = 0.3808
    best_h = None
    best_s = np.inf
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.shape[0] != n:
            init = resample(init, n)
        bin_init = (init > 0.5).astype(float)
        bin_init = repair_to_target(bin_init, target_sum, rng)
        candidates.append(bin_init)
    candidates.append(thue_morse_binary(n))
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))
    for h in candidates:
        if time.time() - start >= budget_s - 1.0:
            break
        s = compute_c5_fast(h)
        if s < best_s:
            best_s, best_h = s, h.copy()
        if best_s <= target_bound:
            return best_h, best_s, n
    pop_size = 30
    elite_size = 2
    pc = 0.7
    pm = 0.1
    tournament_k = 3
    pop = []
    if best_h is not None:
        bin_best = (best_h > 0.5).astype(float)
        bin_best = repair_to_target(bin_best, target_sum, rng)
        pop.append(bin_best)
    while len(pop) < pop_size:
        pop.append(random_binary_balanced(n, rng))
    generation = 0
    max_gens = 10000
    while time.time() - start < budget_s - 2.0 and generation < max_gens:
        fitness = np.empty(pop_size)
        for i in range(pop_size):
            fitness[i] = compute_c5_fast(pop[i])
        for i in range(pop_size):
            if fitness[i] < best_s:
                best_s, best_h = fitness[i], pop[i].copy()
        if best_s <= target_bound:
            return best_h, best_s, n
        elite_idx = np.argsort(fitness)[:elite_size]
        new_pop = [pop[i].copy() for i in elite_idx]
        while len(new_pop) < pop_size:
            participants = rng.choice(pop_size, size=tournament_k, replace=False)
            p1_idx = participants[np.argmin(fitness[participants])]
            participants = rng.choice(pop_size, size=tournament_k, replace=False)
            p2_idx = participants[np.argmin(fitness[participants])]
            parent1 = pop[p1_idx]
            parent2 = pop[p2_idx]
            child = parent1.copy()
            if rng.random() < pc:
                points = rng.choice(n, size=2, replace=False)
                low, high = sorted(points)
                child = np.concatenate([parent1[:low], parent2[low:high], parent1[high:]])
            child = repair_to_target(child, target_sum, rng)
            if rng.random() < pm:
                zeros = np.where(child == 0)[0]
                ones = np.where(child == 1)[0]
                if zeros.size and ones.size:
                    i = rng.choice(zeros)
                    j = rng.choice(ones)
                    child[i], child[j] = 1.0, 0.0
            new_pop.append(child)
        pop = new_pop
        generation += 1
    if best_h is not None and time.time() - start < budget_s - 5.0:
        h_ref, s_ref = gradient_descent_momentum(best_h, start, budget_s, step0=0.5, beta=0.9, max_iter=200000)
        if s_ref < best_s:
            best_s, best_h = s_ref, h_ref.copy()
    if best_h is not None and time.time() - start < budget_s - 2.0:
        h_ann, s_ann = binary_swap_anneal(best_h, rng, start, budget_s, max_iters=200000)
        if s_ann < best_s:
            best_s, best_h = s_ann, h_ann.copy()
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, target_sum)
        best_s = compute_c5_fast(best_h)
        return best_h, best_s, n
    fallback = project_onto_box_sum(np.full(n, 0.5), target_sum)
    return fallback, compute_c5_fast(fallback), n
