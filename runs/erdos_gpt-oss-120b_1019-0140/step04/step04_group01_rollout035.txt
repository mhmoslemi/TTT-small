def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)

def adam_descent(h0, time_left_func, max_iter=200000, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for t in range(1, max_iter+1):
        if time_left_func() <= 0.0:
            break
        # Compute correlation and get max shifts
        # Use compute_corr_scaled_fft (scaled) for detection
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # gradient
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # positions where i+shift < n
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            # term1 = h[shift:] placed at start indices 0..n-shift-1
            term1[:n - shift] = h[shift:]
            # term2 = h[:n - shift] placed at indices shift..n-1
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def adam_descent(h0, time_left_func, max_iter=200000, lr=0.5,
                 beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for t in range(1, max_iter + 1):
        if time_left_func() <= 0.0:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_from_continuous(h, k):
    idx = np.argpartition(-h, k)[:k]
    h_bin = np.zeros_like(h)
    h_bin[idx] = 1.0
    return h_bin

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def local_swap_search(h_bin, rng, time_left_func, max_no_improve=2000):
    """Hill‑climbing on a binary vector by swapping a 0–1 pair that reduces the max overlap."""
    h = h_bin.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    no_improve = 0
    while time_left_func() > 0.0 and no_improve < max_no_improve:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        s = rng.choice(shifts)  # pick one of the worst shifts
        if s >= 0:
            i_cands = np.where((h[:n - s] == 0) & (h[s:] == 1))[0]
        else:
            s_abs = -s
            i_cands = np.where((h[s_abs:] == 0) & (h[:n - s_abs] == 1))[0] + s_abs
        if i_cands.size == 0:
            no_improve += 1
            continue
        max_samples = 100
        if i_cands.size > max_samples:
            sample_i = rng.choice(i_cands, size=max_samples, replace=False)
        else:
            sample_i = i_cands
        best_local_score = max_corr
        best_swap = None
        for i in sample_i:
            j = i + s
            if j < 0 or j >= n:
                continue
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5_fft(h)
            h[i], h[j] = h[j], h[i]  # revert
            if new_score < best_local_score - 1e-12:
                best_local_score = new_score
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            if best_local_score < best_score - 1e-12:
                best_score = best_local_score
                best_h = h.copy()
            no_improve = 0
            continue
        no_improve += 1
        if no_improve >= max_no_improve // 2:
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if zeros.size and ones.size:
                i = rng.choice(zeros)
                j = rng.choice(ones)
                h[i], h[j] = h[j], h[i]
                new_score = compute_c5_fft(h)
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_h = h.copy()
                no_improve = 0
    return best_h, best_score

            if sc_up_ref < best_score:
                best_score, best_h = sc_up_ref, h_up_ref.copy()
                if best_score <= target:
                    return best_h, best_score, best_h.size
        # binary rounding at higher resolution
        h_up_bin = binary_from_continuous(best_h, best_h.size // 2)
        sc_up_bin = compute_c5_fft(h_up_bin)
        if sc_up_bin < best_score:
            best_score, best_h = sc_up_bin, h_up_bin.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size
        # local swap at higher resolution
        if time_left() > 0.5:
            h_up_local, sc_up_local = local_swap_search(best_h, rng, time_left, max_no_improve=2000)
            if sc_up_local < best_score:
                best_score, best_h = sc_up_local, h_up_local.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ----- Phase 5: random restarts with short refinements -----
    while time_left() > 10.0:
        n_cur = best_h.size if best_h is not None else n
        h_rand = np.zeros(n_cur, dtype=float)
        idx = rng.choice(n_cur, size=n_cur // 2, replace=False)
        h_rand[idx] = 1.0
        # brief Adam refinement (continuous)
        h_rand_ref, sc_rand_ref = adam_descent(h_rand, time_left, max_iter=10000, lr=0.5)
        if sc_rand_ref < best_score:
            best_score, best_h = sc_rand_ref, h_rand_ref.copy()
            if best_score <= target:
                break
        # binary rounding
        h_rand_bin = binary_from_continuous(h_rand_ref, n_cur // 2)
        sc_rand_bin = compute_c5_fft(h_rand_bin)
        if sc_rand_bin < best_score:
            best_score, best_h = sc_rand_bin, h_rand_bin.copy()
            if best_score <= target:
                break
        # local swap
        if time_left() > 5.0:
            h_rand_loc, sc_rand_loc = local_swap_search(best_h, rng, time_left, max_no_improve=1000)
            if sc_rand_loc < best_score:
                best_score, best_h = sc_rand_loc, h_rand_loc.copy()
                if best_score <= target:
                    break

    # final projection to guarantee feasibility
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
