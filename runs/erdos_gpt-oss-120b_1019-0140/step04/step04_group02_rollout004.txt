def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.size
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.size
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if shifts.size:
        grad /= shifts.size
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h_start, rng, time_left, target=None):
    n = h_start.size
    idx = np.argpartition(h_start, -n // 2)[-n // 2:]
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx] = 1.0
    best_h = h_bin.copy()
    best_score = compute_c5_fft(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    T = 0.1
    steps = 0
    while time_left() > 0.5:
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        trial = cur_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        trial_score = compute_c5_fft(trial)
        delta = trial_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_score = trial, trial_score
            if trial_score < best_score:
                best_score, best_h = trial_score, trial.copy()
                if target is not None and best_score <= target:
                    break
        steps += 1
        if steps % 2000 == 0:
            T = max(T * 0.95, 1e-5)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    TARGET = 0.3808
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128)
    best_h = None
    best_score = np.inf
    scales = [128, 256, 512]
    for n in scales:
        if time_left() <= 2.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        # Determine feasible restarts
        max_restarts = max(1, int(time_left() // 20))
        if n == 128:
            max_restarts = min(max_restarts, 30)
        elif n == 256:
            max_restarts = min(max_restarts, 15)
        else:
            max_restarts = min(max_restarts, 10)
        for r in range(max_restarts):
            if time_left() <= 2.0:
                break
            sigma = 0.1 if r < max_restarts // 2 else 0.05
            h0 = base + rng.normal(scale=sigma, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)
            iters = 200000 if n == 128 else 300000 if n == 256 else 400000
            h_opt, sc_opt = gradient_descent(h0, n_iter=iters, step0=0.6, beta=0.9, rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_score <= TARGET:
            break
        # Further refinement from current best
        if best_h is not None and time_left() > 5.0:
            if best_h.size != n:
                best_h = upsample(best_h, n)
                best_h = project_onto_box_sum(best_h, n / 2.0)
            h_opt, sc_opt = gradient_descent(best_h, n_iter=600000, step0=0.4, beta=0.9, rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
        base = best_h.copy() if best_h is not None else base
    # Binary hill climbing
    if best_h is not None and time_left() > 5.0:
        h_bin, sc_bin = binary_hill_climb(best_h, rng, time_left, target=TARGET)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
    # Optional final continuous refinement
    if best_h is not None and time_left() > 5.0:
        h_opt, sc_opt = gradient_descent(best_h, n_iter=300000, step0=0.4, beta=0.9, rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
    # Final projection and evaluation
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
        n_ret = best_h.size
    else:
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5_fft(best_h)
    return best_h, best_score, n_ret
