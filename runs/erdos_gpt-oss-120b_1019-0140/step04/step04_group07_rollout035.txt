def softmax_gradient_descent(h0, max_iters, step0, mom_beta, beta0, beta_factor, beta_max, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    step = step0
    beta = beta0
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        # softmax weight
        max_corr = np.max(corr)
        # compute weights = exp(beta * (corr - max_corr))
        # to avoid overflow/underflow
        w = np.exp(beta * (corr - max_corr))
        w_sum = np.sum(w)
        # compute weighted term1_sum, term2_sum
        term1 = np.zeros_like(h)
        term2 = np.zeros_like(h)
        # loop over shifts
        n_shift = n
        L = 2*n_shift - 1
        for idx_shift in range(L):
            shift = idx_shift - (n_shift - 1)
            weight = w[idx_shift]
            if weight == 0.0:
                continue
            if shift >= 0:
                # term1 at positions 0..n-shift-1
                term1[:n_shift-shift] += weight * h[shift:]
                # term2 at positions shift..n-1
                term2[shift:] += weight * h[:n_shift-shift]
            else:
                sft = -shift
                term1[sft:] += weight * h[:n_shift-sft]
                term2[:n_shift-sft] += weight * h[sft:]
        # Grad = (1) - (term1+term2)/w_sum
        grad = np.full_like(h, 1.0) - (term1 + term2) / w_sum
        # Momentum update
        momentum = mom_beta * momentum + (1.0 - mom_beta) * grad
        # Update
        h_new = h - step * momentum
        # Project
        h_new = project_onto_box_sum(h_new, s)
        # Compute new score (actual objective)
        new_score = compute_c5(h_new)
        # Possibly adapt step size: if new_score <= best_score, accept, increase step slightly; else reduce step.
        if new_score <= best_score * (1 + 1e-6):  # allow tiny tolerance
            h = h_new
            best_h = h.copy()
            best_score = new_score
            # increase step
            step = min(step0, step * 1.02)
        else:
            # reject, keep old h, reduce step
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        # update beta
        beta = min(beta_max, beta * beta_factor)
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def softmax_gradient_descent(h0, step0, mom_beta, beta0, beta_factor, beta_max,
                             max_iters, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    step = step0
    beta = beta0
    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        w = np.exp(beta * (corr - max_corr))
        w_sum = np.sum(w)
        term1 = np.zeros_like(h)
        term2 = np.zeros_like(h)
        L = 2 * n - 1
        for idx_shift in range(L):
            shift = idx_shift - (n - 1)
            weight = w[idx_shift]
            if weight == 0.0:
                continue
            if shift >= 0:
                term1[:n - shift] += weight * h[shift:]
                term2[shift:] += weight * h[:n - shift]
            else:
                sft = -shift
                term1[sft:] += weight * h[:n - sft]
                term2[:n - sft] += weight * h[sft:]
        grad = np.full_like(h, 1.0) - (term1 + term2) / w_sum
        momentum = mom_beta * momentum + (1.0 - mom_beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= best_score * (1 + 1e-9):
            h = h_new
            best_h = h.copy()
            best_score = new_score
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        beta = min(beta_max, beta * beta_factor)
        if it % 5000 == 0 and time_left() > 0:
            jitter = rng.normal(scale=0.005, size=n)
            h = project_onto_box_sum(h + jitter, s)
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc = compute_c5(h_new)
            if sc < best_score:
                best_h, best_score = h_new, sc
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            sc2 = compute_c5(h_new2)
            if sc2 < best_score:
                best_h, best_score = h_new2, sc2
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shift_idxs.size == 0:
            break
        shift = rng.choice(shift_idxs)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            sft = -shift
            contrib = best_h[sft:] * (1.0 - best_h[:n - sft])
            i = np.argmax(contrib) + sft
            j = i - sft
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc = compute_c5(h_new)
        if sc < best_score:
            best_h, best_score = h_new, sc
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)
    rng = np.random.default_rng(seed)

    n_target = 512
    s_target = n_target / 2.0

    def get_initial():
        if 'initial_h_values' in globals():
            base = np.asarray(initial_h_values, dtype=float)
        else:
            base = thue_morse(64).astype(float)
        base = upsample(base, n_target)
        base = project_onto_box_sum(base, s_target)
        return base

    best_h = get_initial()
    best_score = compute_c5(best_h)

    # Continuous softmax gradient descent
    if time_left() > 10:
        max_iters = int(min(200000, max(5000, time_left() * 100)))
        h_opt, sc_opt = softmax_gradient_descent(
            best_h,
            step0=0.8,
            mom_beta=0.9,
            beta0=0.05,
            beta_factor=1.001,
            beta_max=6.0,
            max_iters=max_iters,
            rng=rng,
            time_left=time_left,
        )
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    # Local pairwise refinement
    if time_left() > 5:
        h_ref, sc_ref = local_pair_moves(best_h, rng, time_left)
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref

    # Targeted max-shift improvement
    if time_left() > 5:
        h_ref2, sc_ref2 = improve_via_max_shift(best_h, rng, time_left)
        if sc_ref2 < best_score:
            best_h, best_score = h_ref2, sc_ref2

    # Random binary restarts with short refinement
    while time_left() > 30:
        h_rand = random_binary_balanced(n_target, rng)
        max_iters = int(min(5000, max(500, time_left() * 5)))
        h_rand_opt, sc_rand_opt = softmax_gradient_descent(
            h_rand,
            step0=0.5,
            mom_beta=0.9,
            beta0=0.05,
            beta_factor=1.002,
            beta_max=6.0,
            max_iters=max_iters,
            rng=rng,
            time_left=time_left,
        )
        if sc_rand_opt < best_score:
            best_h, best_score = h_rand_opt, sc_rand_opt
            if time_left() > 5:
                h_rand_ref, sc_rand_ref = local_pair_moves(best_h, rng, time_left)
                if sc_rand_ref < best_score:
                    best_h, best_score = h_rand_ref, sc_rand_ref
            if time_left() > 5:
                h_rand_ref2, sc_rand_ref2 = improve_via_max_shift(best_h, rng, time_left)
                if sc_rand_ref2 < best_score:
                    best_h, best_score = h_rand_ref2, sc_rand_ref2

    best_h = project_onto_box_sum(best_h, s_target)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_target
