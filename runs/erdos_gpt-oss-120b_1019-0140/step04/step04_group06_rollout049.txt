Define solve_binary_mip(n, time_limit):

    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    const = n / 2.0
    w = cp.Variable((n, n), boolean=True)

    constraints = [cp.sum(h) == const, t >= 0, t <= 1]

    # w[i,i] = h[i]
    for i in range(n):
        constraints.append(w[i,i] == h[i])

    # product constraints:
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(w[i,j] <= h[i])
            constraints.append(w[i,j] <= h[j])
            constraints.append(w[i,j] >= h[i] + h[j] - 1)

    # shift constraints:
    for s in range(1, n):
        sum_w = cp.sum(w[:n-s, s:n])  # w[i,i+s] for i=0..n-s-1
        constraints.append(sum_w + const * t >= const)

    # objective
    objective = cp.Minimize(t)

    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.CBC, verbose=False, **{'seconds': time_limit})
    except Exception as e:
        # maybe solver not available; fallback.
        return None, None

    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE, cp.FEASIBLE_INACCURATE]:
        return None, None

    h_val = np.array(h.value).flatten()
    # Clip to [0,1] round to nearest integer
    h_bin = np.rint(np.clip(h_val,0,1)).astype(float) # ensure sum matches const
    t_val = float(t.value) if t.value is not None else None
    return h_bin, t_val

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def solve_binary_mip(n, time_limit):
    import cvxpy as cp
    const = n / 2.0
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    w = cp.Variable((n, n), boolean=True)
    constraints = [cp.sum(h) == const, t >= 0, t <= 1]
    # product constraints for i<j
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(w[i,j] <= h[i])
            constraints.append(w[i,j] <= h[j])
            constraints.append(w[i,j] >= h[i] + h[j] - 1)
    # shift constraints
    for s in range(1, n):
        # sum of w[i,i+s]
        sum_w = cp.sum([w[i, i+s] for i in range(n - s)])
        constraints.append(sum_w + const * t >= const)
    # objective
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False, **{'seconds': time_limit})
    except Exception:
        return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE, cp.FEASIBLE_INACCURATE]:
        return None, None
    h_val = np.array(h.value).flatten()
    h_bin = np.rint(np.clip(h_val, 0, 1)).astype(float)
    t_val = float(t.value) if t.value is not None else None
    return h_bin, t_val

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

            break
        best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        # iteration budget roughly proportional to remaining time
        iters = int(min(60000, max(10000, time_left() * (2000.0 / n))))
        best_h, best_score = grad_descent(best_h, n_iter=iters, step0=0.5,
                                          beta=0.9, rng=rng, time_left=time_left)
        if best_score <= target:
            break

    # final random restarts at final resolution
    while time_left() > 8.0:
        n = best_h.size
        h_rand = random_binary_balanced(n, rng)
        h_rand = project_onto_box_sum(h_rand, n / 2.0)
        iters = int(min(12000, max(2000, time_left() * 5)))
        h_rand, sc_rand = grad_descent(h_rand, n_iter=iters, step0=0.4,
                                       beta=0.9, rng=rng, time_left=time_left)
        if sc_rand < best_score:
            best_h, best_score = h_rand, sc_rand
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.1, 0.3)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new, sc_new
        if best_score <= target:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
