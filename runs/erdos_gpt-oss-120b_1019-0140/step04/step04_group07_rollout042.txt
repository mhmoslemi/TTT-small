def local_pair_moves_sa(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    n = h.shape[0]
    T0 = 1e-4
    it = 0
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = cur_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            # project not needed, delta respects bounds, and sum unchanged.
            new_score = compute_c5(h_new)
            # Acceptance:
            if new_score < cur_score:
                cur_h, cur_score = h_new, new_score
                if new_score < best_score:
                    best_h, best_score = h_new.copy(), new_score
            else:
                T = T0 * np.exp(-0.001 * it)
                if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                    cur_h, cur_score = h_new, new_score
        # also try reverse direction
        max_delta2 = min(cur_h[j], 1.0 - cur_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = cur_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            new_score2 = compute_c5(h_new2)
            if new_score2 < cur_score:
                cur_h, cur_score = h_new2, new_score2
                if new_score2 < best_score:
                    best_h, best_score = h_new2.copy(), new_score2
            else:
                T = T0 * np.exp(-0.001 * it)
                if rng.random() < np.exp(-(new_score2 - cur_score) / max(T, 1e-12)):
                    cur_h, cur_score = h_new2, new_score2
        it += 1
    return best_h, best_score

def gradient_descent_adaptive_nesterov(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    # momentum variable y? typical NAG: v = 0; h_prev = h
    v = np.zeros_like(h)
    # Also maintain previous h for Nesterov? Actually we use NAG where:
    # y = h + momentum * v
    # compute grad at y
    # v_new = momentum * v - step * grad
    # h_new = h + v_new
    # But we need projection after update.
    # We'll use simple momentum: v = beta*v + (1-beta)*grad.
    # Not strict NAG but similar.
    # We'll keep original structure with momentum.
    momentum = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        # Compute correlation and worst shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # get indices where close to max
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        # Compute gradient for those shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                s_shift = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[s_shift:] = h[:n-s_shift]
                term2[:n-s_shift] = h[s_shift:]
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # Nesterov lookahead (optional): compute h_proposed = h - step * momentum, then project
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-7:
                step = step0 * 0.1
        # Periodic jitter to escape if no improvement in a while:
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand interval until it brackets s
    for _ in range(max_iter):
        if f(lam_low) < s:
            lam_low -= (hi - lo) * 2
        if f(lam_high) > s:
            lam_high += (hi - lo) * 2
        if f(lam_low) <= s <= f(lam_high):
            break

    # Binary search for lam
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation to a larger grid."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def improve_via_max_shift(h, rng, time_left):
    """Transfer mass from the point contributing most to a worst lag."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def gradient_descent_adaptive_nesterov(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum (Nesterov‑style)."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                s_shift = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[s_shift:] = h[:n - s_shift]
                term2[:n - s_shift] = h[s_shift:]
            grad += (1.0 - term1 - term2)
        if shifts.size:
            grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h, cur_score = h_new, new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
        else:
            step *= 0.5
            if step < 1e-7:
                step = step0 * 0.1
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def local_pair_moves_sa(h, rng, time_left):
    """Pairwise mass transfer with simulated‑annealing acceptance."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    n = h.shape[0]
    T0 = 1e-4
    it = 0
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # i -> j
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = cur_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            new_score = compute_c5(h_new)
            if new_score < cur_score:
                cur_h, cur_score = h_new, new_score
                if new_score < best_score:
                    best_h, best_score = h_new.copy(), new_score
            else:
                T = T0 * np.exp(-0.001 * it)
                if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                    cur_h, cur_score = h_new, new_score
        # j -> i
        max_delta2 = min(cur_h[j], 1.0 - cur_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = cur_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            new_score2 = compute_c5(h_new2)
            if new_score2 < cur_score:
                cur_h, cur_score = h_new2, new_score2
                if new_score2 < best_score:
                    best_h, best_score = h_new2.copy(), new_score2

                T = T0 * np.exp(-0.001 * it)
                if rng.random() < np.exp(-(new_score2 - cur_score) / max(T, 1e-12)):
                    cur_h, cur_score = h_new2, new_score2
        it += 1
    return best_h, best_score
