def swap_and_update(h, a, ones_set, i, j):
    # ensure i is 1, j is 0
    # remove i
    ones_set.remove(i)
    # loop over p in ones_set
    for p in ones_set:
        k = p - i
        idx = (len(h)-1) + k
        a[idx] -= 1
        a[2*len(h)-2 - idx] -= 1  # symmetrical counterpart (since a length = 2n-1)
    # add j
    for p in ones_set:
        k = p - j
        idx = (len(h)-1) + k
        a[idx] += 1
        a[2*len(h)-2 - idx] += 1
    # update h
    h[i] = 0
    h[j] = 1
    ones_set.add(j)

def initialize_h(n: int, rng: np.random.Generator) -> np.ndarray:
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        base = upsample_to_n(base, n)
        h = (base > 0.5).astype(np.int8)
        target = n // 2
        cur = int(h.sum())
        if cur > target:
            ones = np.where(h == 1)[0]
            h[rng.choice(ones, size=cur - target, replace=False)] = 0
        elif cur < target:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, size=target - cur, replace=False)] = 1
        return h
    return thue_morse_binary(n)

def setup_structures(h: np.ndarray):
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    ones_list = list(ones_set)
    zeros_list = list(zeros_set)
    pos_in_ones = {i: idx for idx, i in enumerate(ones_list)}
    pos_in_zeros = {i: idx for idx, i in enumerate(zeros_list)}
    return ones_set, zeros_set, ones_list, zeros_list, pos_in_ones, pos_in_zeros

def apply_swap(i: int, j: int, h: np.ndarray, a: np.ndarray,
               ones_set: set, zeros_set: set,
               ones_list: list, zeros_list: list,
               pos_in_ones: dict, pos_in_zeros: dict, n: int):
    mid = n - 1
    len_a = 2 * n - 1
    # remove i from ones
    ones_set.remove(i)
    pi = pos_in_ones[i]
    last_one = ones_list[-1]
    ones_list[pi] = last_one
    pos_in_ones[last_one] = pi
    ones_list.pop()
    del pos_in_ones[i]
    for p in ones_set:
        idx = mid + (p - i)
        a[idx] -= 1
        a[len_a - 1 - idx] -= 1
    # add j to ones (after i removed)
    for p in ones_set:
        idx = mid + (p - j)
        a[idx] += 1
        a[len_a - 1 - idx] += 1
    # remove j from zeros
    zeros_set.remove(j)
    pj = pos_in_zeros[j]
    last_zero = zeros_list[-1]
    zeros_list[pj] = last_zero
    pos_in_zeros[last_zero] = pj
    zeros_list.pop()
    del pos_in_zeros[j]
    zeros_set.add(i)
    pos_in_zeros[i] = len(zeros_list)
    zeros_list.append(i)
    # update vector
    h[i] = 0
    h[j] = 1
    # add j to ones structures
    ones_set.add(j)
    pos_in_ones[j] = len(ones_list)
    ones_list.append(j)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024
    h = initialize_h(n, rng)
    a = compute_initial_a(h)
    cur_score = c5_from_a(a, n)
    best_score = cur_score
    best_h = h.copy()
    ones_set, zeros_set, ones_list, zeros_list, pos_in_ones, pos_in_zeros = setup_structures(h)
    target_bound = 0.3808
    prob_targeted = 0.1
    T0 = 0.02
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.5:
            break
        frac = elapsed / budget_s
        T = T0 * ((1.0 - frac) ** 0.5)
        # decide move type
        if rng.random() < prob_targeted:
            mid = n - 1
            left_min = a[:mid].min()
            right_min = a[mid + 1 :].min()
            if left_min <= right_min:
                min_val = left_min
                idxs = np.where(a[:mid] == min_val)[0]
                idx = rng.choice(idxs)
                shift = idx - mid
            else:
                min_val = right_min
                idxs = np.where(a[mid + 1 :] == min_val)[0]
                idx = rng.choice(idxs) + (mid + 1)
                shift = idx - mid
            i_candidates = [i for i in ones_set if 0 <= i + shift < n and h[i + shift] == 0]
            j_candidates = [j for j in zeros_set if 0 <= j + shift < n and h[j + shift] == 1]
            if i_candidates and j_candidates:
                i = rng.choice(i_candidates)
                j = rng.choice(j_candidates)
            else:
                pi = rng.integers(len(ones_list))
                i = ones_list[pi]
                pj = rng.integers(len(zeros_list))
                j = zeros_list[pj]
        else:
            pi = rng.integers(len(ones_list))
            i = ones_list[pi]
            pj = rng.integers(len(zeros_list))
            j = zeros_list[pj]
        apply_swap(i, j, h, a, ones_set, zeros_set,
                   ones_list, zeros_list, pos_in_ones, pos_in_zeros, n)
        new_score = c5_from_a(a, n)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target_bound:
                    break
        else:
            apply_swap(j, i, h, a, ones_set, zeros_set,
                       ones_list, zeros_list, pos_in_ones, pos_in_zeros, n)
    return best_h.astype(np.float64), float(best_score), n
