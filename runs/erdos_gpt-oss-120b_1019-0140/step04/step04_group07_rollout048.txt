def compute_corr_fft(h):

    rev = (1 - h)[::-1]
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)[:2*n-1]
    return conv

def gradient_worst_shift(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    g = np.zeros_like(h)
    if shift >= 0:
        i = slice(0, n - shift)
        j = slice(shift, n)
        # i corresponds to h[i], j to h[i+shift] = h[j]
        g[i] += 1.0 - h[j]
        g[j] -= h[i]
    else:
        s = -shift
        i = slice(s, n)
        j = slice(0, n - s)
        # Here i = i_original that maps to j = i-shift? Actually shift negative: j = i+shift = i-s.
        # So i in range(s, n) means j = i - s is in range(0, n-s)
        g[i] += 1.0 - h[j]
        g[j] -= h[i]
    return g, shift, idx

def projected_gradient_descent(h0, time_left, max_iters=None, step0=1.0):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    step = step0
    momentum = np.zeros_like(h)
    beta = 0.9

    it = 0
    while time_left() > 0:
        if max_iters is not None and it >= max_iters:
            break
        # compute gradient and shift
        grad, shift, idx = gradient_worst_shift(h)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # step tentative
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= best_score:
            # accept
            h = h_new
            best_h = h.copy()
            best_score = new_score
            # maybe increase step
            step = min(step0, step * 1.05)
        else:
            # reject
            step *= 0.5
            # possibly keep h unchanged
        it += 1

    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # compute max delta that respects bounds
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 1e-12:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h = h_new
                best_score = score
        # also consider reverse direction
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 1e-12:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h = h_new2
                best_score = score2
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_fft(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    rev = (1 - h)[::-1]                # reverse of (1-h)
    L = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n-1
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)[:2 * n - 1]
    return conv


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)


def gradient_worst_shift(h):
    """Gradient of correlation for the shift attaining the maximal value."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.zeros_like(h)

    if shift >= 0:
        i = slice(0, n - shift)
        j = slice(shift, n)
        grad[i] += 1.0 - h[j]
        grad[j] -= h[i]
    else:
        s = -shift
        i = slice(s, n)
        j = slice(0, n - s)
        grad[i] += 1.0 - h[j]
        grad[j] -= h[i]

    return grad, shift, idx


def projected_gradient_descent(h0, time_left, max_iters, step0, rng):
    """Projected gradient descent using worst‑shift subgradient."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    beta = 0.9
    it = 0

    while time_left() > 0 and it < max_iters:
        grad, _, _ = gradient_worst_shift(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h.copy()
                best_score = new_score
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        it += 1

        if it % 2000 == 0:
            # occasional jitter to escape flat regions
            jitter = rng.normal(scale=0.005, size=n)
            h = project_onto_box_sum(h + jitter, target_sum)
            cur_score = compute_c5(h)

    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Hill‑climbing by moving small mass between random positions."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size

    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)

        # move from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score

        # move from j to i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2

    return best_h, best_score

        for _ in range(2):

            h_rand = rng.random(n)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
