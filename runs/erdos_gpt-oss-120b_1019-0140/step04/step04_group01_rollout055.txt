def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def binary_anneal(h0, rng, time_left, max_iters=None, T0=0.2, decay=0.9999, Tmin=1e-6):
    if max_iters is None:
        max_iters = int(1e9)  # effectively unlimited, but break via time.
    h = h0.copy()
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    n = h.size
    while (max_iters is None or it < max_iters) and time_left() > 0:
        # swap a 1 and a 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
        # temperature schedule
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    # Use linear interpolation on unit interval [0, 1)
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1
    return h

def slsqp_optimize_candidate(h0, max_iter=1500):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_scaled_fft(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

def binary_anneal(h0, rng, time_left, max_iters=None, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    cur = compute_c5_fft(h)
    best_h = h.copy()
    best = cur
    T = T0
    it = 0
    n = h.size
    while (max_iters is None or it < max_iters) and time_left() > 0.0:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        new = compute_c5_fft(h)
        delta = new - cur
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best

            best_score, best_h = sc_opt, h_opt.astype(float)

    # Phase 6: light continuous refinement from best (now possibly binary)
    if overall_time_left() > 5.0:
        h_cont = best_h.copy()
        jitter = rng.normal(scale=0.02, size=h_cont.shape)
        h_cont = np.clip(h_cont + jitter, 0.0, 1.0)
        h_cont = project_onto_box_sum(h_cont, n_target / 2.0)
        h_opt_c, sc_opt_c = slsqp_optimize_candidate(h_cont, max_iter=2000)
        if sc_opt_c < best_score:
            best_score, best_h = sc_opt_c, h_opt_c.copy()

    # Final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)

    return best_h, best_score, best_h.size
