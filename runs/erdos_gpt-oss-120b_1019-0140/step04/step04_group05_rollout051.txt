import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_balanced(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n/2.0)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def shift_h(h, k):
    n = h.shape[0]
    if k > 0:
        return np.concatenate((np.zeros(k, dtype=h.dtype), h[:n - k]))
    elif k < 0:
        k = -k
        return np.concatenate((h[k:], np.zeros(k, dtype=h.dtype)))
    else:
        return h.copy()

def compute_c5_and_lag(h):
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, L)
    Hc = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * Hc, L)
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    k = max_idx - (n - 1)
    c5 = max_val * (2.0 / n)
    return c5, k, corr

def attempt_move(h, i, j, max_delta, cur_c5, rng):
    delta = max_delta
    eps = 1e-12
    while delta > eps:
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_c5, _, _ = compute_c5_and_lag(h_new)
        if new_c5 < cur_c5 - 1e-12:
            return True, new_c5, h_new
        delta *= 0.5
    return False, cur_c5, h

def hill_climb_step(h, rng):
    cur_c5, k, _ = compute_c5_and_lag(h)
    grad = 1.0 - shift_h(h, k)
    eps = 1e-12
    src_candidates = np.where((grad > eps) & (h > eps))[0]
    if src_candidates.size == 0:
        return False, cur_c5, h
    i = src_candidates[np.argmax(grad[src_candidates])]
    tgt_candidates = np.where(h < 1.0 - eps)[0]
    if tgt_candidates.size == 0:
        return False, cur_c5, h
    j = tgt_candidates[np.argmin(grad[tgt_candidates])]
    if i == j:
        return False, cur_c5, h
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= eps:
        return False, cur_c5, h
    # random scaling of delta (10% - 50% of max)
    scale = rng.random() * 0.4 + 0.1
    max_delta = max_delta * scale
    return attempt_move(h, i, j, max_delta, cur_c5, rng)

def hill_climb_optimize(h0, budget_s, rng):
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    cur_h = h0.copy()
    cur_c5, _, _ = compute_c5_and_lag(cur_h)
    best_h = cur_h.copy()
    best_score = cur_c5
    no_improve = 0
    while time_left() > 0.02:
        improved, new_score, new_h = hill_climb_step(cur_h, rng)
        if improved:
            cur_h = new_h
            cur_c5 = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
                if best_score <= 0.3808:
                    break
        else:
            no_improve += 1
            if no_improve >= 200:
                # random restart with continuous balanced vector
                n = cur_h.size
                cur_h = random_continuous_balanced(n, rng)
                cur_c5, _, _ = compute_c5_and_lag(cur_h)

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_balanced(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def shift_h(h, k):
    n = h.shape[0]
    if k > 0:
        return np.concatenate((np.zeros(k, dtype=h.dtype), h[:n - k]))
    elif k < 0:
        k = -k
        return np.concatenate((h[k:], np.zeros(k, dtype=h.dtype)))
    else:
        return h.copy()

def compute_c5_and_lag(h):
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, L)
    Hc = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * Hc, L)
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    k = max_idx - (n - 1)
    c5 = max_val * (2.0 / n)
    return c5, k, corr

def attempt_move(h, i, j, max_delta, cur_c5):
    eps = 1e-12
    delta = max_delta
    while delta > eps:
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_c5, _, _ = compute_c5_and_lag(h_new)
        if new_c5 < cur_c5 - 1e-12:
            return True, new_c5, h_new
        delta *= 0.5
    return False, cur_c5, h

def hill_climb_step(h, rng):
    cur_c5, k, _ = compute_c5_and_lag(h)
    grad = 1.0 - shift_h(h, k)
    eps = 1e-12
    src_candidates = np.where((grad > eps) & (h > eps))[0]
    if src_candidates.size == 0:
        return False, cur_c5, h
    i = src_candidates[np.argmax(grad[src_candidates])]
    tgt_candidates = np.where(h < 1.0 - eps)[0]
    if tgt_candidates.size == 0:
        return False, cur_c5, h
    j = tgt_candidates[np.argmin(grad[tgt_candidates])]
    if i == j:
        return False, cur_c5, h
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= eps:
        return False, cur_c5, h
    scale = rng.random() * 0.4 + 0.1   # 10%–50% of max delta
    max_delta = max_delta * scale
    return attempt_move(h, i, j, max_delta, cur_c5)

def hill_climb_optimize(h0, budget_s, rng):
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    cur_h = h0.copy()
    cur_c5, _, _ = compute_c5_and_lag(cur_h)
    best_h = cur_h.copy()
    best_score = cur_c5
    no_improve = 0
    while time_left() > 0.02:
        improved, new_score, new_h = hill_climb_step(cur_h, rng)
        if improved:
            cur_h = new_h
            cur_c5 = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
                if best_score <= 0.3808:
                    break
        else:
            no_improve += 1
            if no_improve >= 200:
                n = cur_h.size
                cur_h = random_continuous_balanced(n, rng)
                cur_c5, _, _ = compute_c5_and_lag(cur_h)
                no_improve = 0
    return best_h, best_score
