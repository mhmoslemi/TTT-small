def ga_optimize(initial_h, n, time_left, rng, max_iters=None):

    pop_size = 70 (maybe)
    # ensure pop size <= n ? doesn't matter.

    # Create initial population:
    pop = []
    pop.append(initial_h.copy())
    while len(pop) < pop_size:
        pop.append(random_binary_balanced(n, rng))
    # compute fitness
    fitness = [compute_c5(ind) for ind in pop]
    # best
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_score = fitness[best_idx]

    # compute number of iterations: if max_iters None, run while time_left > some.

    iter_count = 0
    while time_left() > 0.5:   # small margin
        iter_count += 1

        # tournament select parent1, parent2
        def tournament():
            idxs = rng.choice(pop_size, size=5, replace=False)
            sub_fitness = [fitness[i] for i in idxs]
            best_local = idxs[int(np.argmin(sub_fitness))]
            return pop[best_local]
        p1 = tournament()
        p2 = tournament()

        # Crossover
        if rng.random() < 0.7:
            cut = rng.integers(1, n)  # point in [1,n-1]
            child1 = np.concatenate([p1[:cut], p2[cut:]])
            child2 = np.concatenate([p2[:cut], p1[cut:]])
        else:
            child1 = p1.copy()
            child2 = p2.copy()

        # Mutation: for each child attempt a few swaps (e.g., 1 to 3)
        for child in (child1, child2):
            if rng.random() < 0.2:
                # swap a 1 and a 0
                ones = np.where(child == 1)[0]
                zeros = np.where(child == 0)[0]
                if ones.size > 0 and zeros.size > 0:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    child[i], child[j] = 0, 1

        # Repair (binarize) to ensure exactly n/2 1s
        child1 = binarize(child1)
        child2 = binarize(child2)

        # Evaluate children
        child_fitness = [compute_c5(child1), compute_c5(child2)]

        # Replacement: find worst individuals (largest fitness) to replace
        # we can replace up to 2 worst
        worst_idxs = np.argsort(fitness)[-2:]  # indices of two worst
        # Replace them with children if children better than respective worst?
        # Simpler: always replace worst two with children.
        for idx, new_ind, new_fit in zip(worst_idxs, [child1, child2], child_fitness):
            pop[idx] = new_ind
            fitness[idx] = new_fit

        # Update best
        min_fit_idx = int(np.argmin(fitness))
        if fitness[min_fit_idx] < best_score:

            best_h = pop[min_fit_idx].copy()

def binarize(v):
    n = v.shape[0]
    k = n // 2
    # v may be float; we want binary 0/1.
    # Get indices of top k values.
    idx = np.argpartition(v, -k)[-k:]  # returns unsorted top k
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out

def random_binary_balanced(n, rng):

Now upsample: Use np.interp across [0,1) domain.

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # n must be power of 2? but generate for any n.
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def compute_c5(h):
    n = h.shape[0]
    m = n // 2

def hill_climb(h, time_left, rng):
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while time_left() > 0.5:
        # random swap attempt
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        candidate = best_h.copy()
        candidate[i], candidate[j] = 0, 1
        cand_score = compute_c5(candidate)
        if cand_score < best_score:
            best_h, best_score = candidate, cand_score
            # optional early exit if below target
            if best_score <= 0.3808:

import time
import numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize(v):
    n = v.shape[0]
    k = n // 2
    # Get indices of top k values
    idx = np.argpartition(v, -k)[-k:]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out

def random_binary_balanced(n, rng):
    out = np.zeros(n, dtype=float)
    out[rng.permutation(n)[: n // 2]] = 1.0
    return out

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_c5(h):
    n = h.shape[0]
    m = n // 2
    S = np.correlate(h, h, mode='full')
    return (m - S.min()) * (2.0 / n)

            best_fit = f
    return pop[best_idx]

import time
import numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize(v):
    n = v.shape[0]
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out

def random_binary_balanced(n, rng):
    out = np.zeros(n, dtype=float)
    out[rng.permutation(n)[: n // 2]] = 1.0
    return out

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_c5(h):
    n = h.shape[0]
    m = n // 2
    S = np.correlate(h, h, mode='full')
    return (m - S.min()) * (2.0 / n)

def tournament_select(pop, fitness, k, rng):
    idxs = rng.choice(len(pop), size=k, replace=False)
    best_idx = idxs[0]
    best_fit = fitness[best_idx]
    for idx in idxs[1:]:
        f = fitness[idx]
        if f < best_fit:
            best_idx = idx
            best_fit = f
    return pop[best_idx]

def ga_optimize(init_h, n, stop_time, rng):
    pop_size = 60
    crossover_prob = 0.7
    mutation_prob = 0.2
    tournament_k = 5

    pop = [init_h.copy()]
    while len(pop) < pop_size:
        pop.append(random_binary_balanced(n, rng))
    fitness = [compute_c5(ind) for ind in pop]

    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_score = fitness[best_idx]

    while time.time() < stop_time - 0.5:
        parent1 = tournament_select(pop, fitness, tournament_k, rng)
        parent2 = tournament_select(pop, fitness, tournament_k, rng)

        if rng.random() < crossover_prob:
            cut = rng.integers(1, n)
            child1 = np.concatenate([parent1[:cut], parent2[cut:]])
            child2 = np.concatenate([parent2[:cut], parent1[cut:]])
        else:
            child1 = parent1.copy()
            child2 = parent2.copy()

        if rng.random() < mutation_prob:
            ones = np.where(child1 == 1)[0]
            zeros = np.where(child1 == 0)[0]
            if ones.size > 0 and zeros.size > 0:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                child1[i], child1[j] = 0, 1
        if rng.random() < mutation_prob:
            ones = np.where(child2 == 1)[0]
            zeros = np.where(child2 == 0)[0]
            if ones.size > 0 and zeros.size > 0:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                child2[i], child2[j] = 0, 1

        child1 = binarize(child1)
        child2 = binarize(child2)

        child_fitness1 = compute_c5(child1)
        child_fitness2 = compute_c5(child2)

        worst_idx = int(np.argmax(fitness))
        if child_fitness1 < fitness[worst_idx]:
            pop[worst_idx] = child1
            fitness[worst_idx] = child_fitness1

        worst_idx = int(np.argmax(fitness))
        if child_fitness2 < fitness[worst_idx]:
            pop[worst_idx] = child2
            fitness[worst_idx] = child_fitness2

        current_best_idx = int(np.argmin(fitness))
        if fitness[current_best_idx] < best_score:
            best_score = fitness[current_best_idx]
            best_h = pop[current_best_idx].copy()
            if best_score <= 0.3808:
