def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)
    rng = np.random.default_rng(seed)

    # Helper: sliding_window
    def sliding_window(arr, window):
        shape = (arr.size - window + 1, window)
        strides = (arr.strides[0], arr.strides[0])
        return np.lib.stride_tricks.as_strided(arr, shape=shape, strides=strides)

    # Projection function
    def project_onto_box_sum(v, s):
        lo, hi = 0.0, 1.0
        n = v.size
        if s <= lo * n:
            return np.full_like(v, lo)
        if s >= hi * n:
            return np.full_like(v, hi)
        # binary search for lambda such that sum clip(v - lam) = s
        lam_lo = np.min(v) - hi
        lam_hi = np.max(v) - lo
        for _ in range(80):
            lam_mid = (lam_lo + lam_hi) * 0.5
            clipped = np.clip(v - lam_mid, lo, hi)
            total = clipped.sum()
            if abs(total - s) < 1e-12:
                return clipped
            if total > s:
                lam_lo = lam_mid
            else:
                lam_hi = lam_mid
        return np.clip(v - lam_mid, lo, hi)

    def compute_c5(h):
        n = h.shape[0]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n)

    # gradient of smoothed max using sliding window
    def grad_smooth_max(h, alpha):
        n = h.size
        scale = 2.0 / n
        # correlation (scaled)
        corr = np.correlate(h, 1.0 - h, mode='full') * scale
        # numerically stable weights
        a_corr = alpha * corr
        a_corr_max = np.max(a_corr)
        w = np.exp(a_corr - a_corr_max)
        w_sum = w.sum()
        w_norm = w / w_sum
        # sliding windows
        pad = n - 1
        hp = np.concatenate((np.zeros(pad, dtype=h.dtype), h, np.zeros(pad, dtype=h.dtype)))
        win = sliding_window(hp, n)  # shape (2n-1, n)
        term1 = 1.0 - win
        term2 = - win[::-1, :]  # reversed rows
        grad = np.sum(w_norm[:, None] * (term1 + term2), axis=0) * scale
        return grad, corr

    # Adam optimizer for given resolution
    def optimize_h(h0, max_iters, time_left_func):
        h = h0.copy()
        n = h.shape[0]
        s = n / 2.0
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr0 = 0.2 if n >= 256 else 0.3
        alpha = 0.5  # initial smoothing
        alpha_inc = (20.0 - alpha) / max_iters if max_iters > 0 else 0.0
        best_h = h.copy()
        best_score = compute_c5(h)
        for it in range(1, max_iters + 1):
            if time_left_func() <= 0:
                break
            # optionally update alpha
            alpha = min(20.0, alpha + alpha_inc)
            grad, corr = grad_smooth_max(h, alpha)
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h - step
            h_new = project_onto_box_sum(h_new, s)
            # evaluate exact C5
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                # early exit if good enough
                if best_score <= 0.3808:
                    return best_h, best_score
            h = h_new
        return best_h, best_score

    # main loop over resolutions
    best_h = None
    best_score = np.inf

    scales = [64, 128, 256, 512]
    for n in scales:
        if time_left() <= 5:
            break

        if 'initial_h_values' in globals():
            base = np.asarray(initial_h_values, dtype=float)
        else:

        if base.shape[0] != n:
            # upsample function
            def upsample(arr, new_n):
                m = arr.shape[0]
                x_old = np.linspace(0.0, 1.0, m, endpoint=False)
                x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
                return np.interp(x_new, x_old, arr)
            base = upsample(base, n)
        # enforce feasibility
        base = project_onto_box_sum(base, n / 2.0)
        base_score = compute_c5(base)
        if base_score < best_score:
            best_score = base_score
            best_h = base.copy()
        # determine iterations based on remaining time

        if max_iters > 0:

            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()
        # prepare for next resolution

            next_n = scales[scales.index(n) + 1]

                x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)

            base = upsample(best_h, next_n)

            s = n / 2.0
            t0 = compute_c5(h0)
            x0 = np.concatenate([h0, [t0]])
            bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
            def eq_sum(x):
                return np.sum(x[:n]) - s

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        clipped = np.clip(v - lam_mid, lo, hi)
        total = clipped.sum()
        if abs(total - s) < 1e-12:
            return clipped
        if total > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # Generate Thue-Morse binary sequence (0/1) based on parity of popcount
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def sliding_window(arr, window):
    shape = (arr.size - window + 1, window)
    strides = (arr.strides[0], arr.strides[0])
    return np.lib.stride_tricks.as_strided(arr, shape=shape, strides=strides)

def grad_smooth_max(h, alpha):
    n = h.size
    scale = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * scale
    a_corr = alpha * corr
    a_max = np.max(a_corr)
    w = np.exp(a_corr - a_max)
    w_sum = w.sum()
    w_norm = w / w_sum
    pad = n - 1
    hp = np.concatenate((np.zeros(pad, dtype=h.dtype), h, np.zeros(pad, dtype=h.dtype)))
    win = sliding_window(hp, n)           # shape (2n-1, n)
    term1 = 1.0 - win
    term2 = -win[::-1, :]
    grad = np.sum(w_norm[:, None] * (term1 + term2), axis=0) * scale
    return grad, corr

def optimize_h(h0, max_iters, start, budget_s):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2 if n >= 256 else 0.3
    alpha = 0.5
    alpha_inc = (20.0 - alpha) / max_iters if max_iters > 0 else 0.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters + 1):
        if budget_s - (time.time() - start) <= 0:
            break
        alpha = min(20.0, alpha + alpha_inc)
        grad, _ = grad_smooth_max(h, alpha)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= 0.3808:
                return best_h, best_score
        h = h_new
    return best_h, best_score
