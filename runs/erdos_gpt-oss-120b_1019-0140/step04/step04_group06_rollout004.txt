def grad_descent(h0, n_iter, step0, beta, rng, start, budget_s):
    h = h0.astype(float).copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        now = time.time()
        if now - start >= budget_s - 0.5:
            break
        # compute shifts with max correlation
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        # project onto box and sum
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # widen bounds if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample_binary(arr, n):
    m = arr.shape[0]
    if m == n:
        return arr.copy()
    factor = n // m
    res = np.repeat(arr, factor)
    # if n is not a multiple of m, pad by interpolation (rarely needed here)
    if res.shape[0] < n:
        extra = n - res.shape[0]
        tail = np.interp(np.linspace(0, 1, extra, endpoint=False),
                         np.linspace(0, 1, m, endpoint=False),
                         arr.astype(float))
        res = np.concatenate([res, tail])
    return res

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, start, budget_s):
    h = h0.astype(float).copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= budget_s - 0.5:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # target resolution
    n = 1024
    target_sum = n // 2
    # base initialization
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = upsample_binary(base, n)
        base = project_onto_box_sum(base, target_sum)
    else:
        base = random_binary_balanced(n, rng)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    T0 = 5e-3
    T1 = 1e-6
    early_stop = False

    while not early_stop:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break
        # number of inner iterations scaled to remaining time
        inner_iters = int(min(50000, max(2000, remaining * 3000)))
        # choose start vector for this run
        if rng.random() < 0.5:
            h = base.copy()
        else:
            h = random_binary_balanced(n, rng)
        cur_score = compute_c5(h)
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        for it in range(1, inner_iters + 1):
            now = time.time()
            if now - start >= budget_s - 5.0:
                break
            T = T0 * (T1 / T0) ** (it / inner_iters)
            i = rng.choice(ones)
            j = rng.choice(zeros)
            if i == j:
                continue
            h_new = h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            new_score = compute_c5(h_new)
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True
            if accept:
                h = h_new
                cur_score = new_score
                ones = np.flatnonzero(h)
                zeros = np.flatnonzero(1 - h)
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
                    if best_score <= 0.3808:
                        early_stop = True
                        break
        if early_stop:
            break

    # final continuous refinement if time permits
    remaining = budget_s - (time.time() - start)
    if remaining > 2.0:

        h_gd, sc_gd = grad_descent(best_h.astype(float), n_iter,
                                   step0=0.5, beta=0.9, rng=rng,
                                   start=start, budget_s=budget_s)
        if sc_gd < best_score:
            best_h, best_score = h_gd, sc_gd

    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
