def compute_grad_top_shifts(h, corr, top_k=5, tol=1e-12):
    n = h.size
    # Get indices of top_k largest corr values
    # corr is length 2n-1, shift = idx - (n-1)
    idxes = np.argpartition(corr, -top_k)[-top_k:]
    # but we need to include ties for max? not necessary.
    # compute gradient sum over these shifts
    grad = np.zeros_like(h)
    for idx in idxes:
        k = idx - (n - 1)
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    grad /= top_k
    return grad

    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (zero‑padded)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_top_shifts(h, corr, top_k=5):
    """Gradient averaged over the top_k shifts (subgradient of max)."""
    n = h.size
    if top_k >= corr.size:
        idxes = np.arange(corr.size)
    else:
        idxes = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros_like(h)
    for idx in idxes:
        k = idx - (n - 1)
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    grad /= max(1, len(idxes))
    return grad


def gradient_descent(h0, max_iters, step0, beta, top_k, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = np.clip(h0.copy(), 0.0, 1.0)
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, max_iters + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        grad = compute_grad_top_shifts(h, corr, top_k=top_k)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 50000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def local_search(h, best_score, time_left, rng):
    """Simple hill‑climbing with random mass transfers."""
    n = h.size
    s = n / 2.0
    while time_left() > 0.1:
        if rng.random() < 0.5:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(h[i], 1.0 - h[j])
            if delta <= 0:
                continue
            delta *= rng.uniform(0.2, 1.0)
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            idxs = rng.choice(n, size=rng.integers(1, 4), replace=False)
            h_new = h.copy()
            h_new[idxs] += rng.normal(scale=0.02, size=idxs.shape[0])
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_score, h = sc, h_new
            if best_score <= 0.3808:
                break
    return h, best_score
