import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=50):
    # Project v onto {x: lo <= x <= hi, sum(x) = target}
    n = v.size
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    # lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # Expand bounds if needed
    while f(lam_low) < target:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > target:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target) < 1e-12:
            break
        if fm > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    # Use FFT for correlation
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.empty(n, dtype=h.dtype)
        term1[:n_s] = h[shift:]
        term1[n_s:] = 0.0
        term2 = np.empty(n, dtype=h.dtype)
        term2[:shift] = 0.0
        term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.empty(n, dtype=h.dtype)
        term1[s:] = h[:n_s]
        term1[:s] = 0.0
        term2 = np.empty(n, dtype=h.dtype)
        term2[:n_s] = h[s:]
        term2[n_s:] = 0.0
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    max_val = corr.max()
    idxes = np.where(corr == max_val)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, target, time_left, rng, step0=0.5, beta=0.9,
                              max_iter=200000, noise_interval=5000, noise_scale=0.02):
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        if noise_interval and (it % noise_interval == 0):
            # small random perturbation
            noise = (rng.random(h.shape) - 0.5) * 2.0 * noise_scale
            h_new = project_onto_box_sum(h_new + noise, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def round_to_binary(h):
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    hb = np.zeros_like(h)
    hb[idx] = 1.0
    return hb

def binary_swap_anneal(h0, target, time_left, rng,
                      max_iters=400000, T0=0.2, T_min=1e-5, decay=0.99998):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = T0
    it = 0
    n = h.shape[0]
    while it < max_iters and time_left() > 0.01:
        # find indices of 1 and 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    # ensure feasibility
    best_h = project_onto_box_sum(best_h, target)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    # map to [0,1) intervals
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < target:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > target:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target) < 1e-12:
            break
        if fm > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.empty(n, dtype=h.dtype)
        term1[:n_s] = h[shift:]
        term1[n_s:] = 0.0
        term2 = np.empty(n, dtype=h.dtype)
        term2[:shift] = 0.0
        term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.empty(n, dtype=h.dtype)
        term1[s:] = h[:n_s]
        term1[:s] = 0.0
        term2 = np.empty(n, dtype=h.dtype)
        term2[:n_s] = h[s:]
        term2[n_s:] = 0.0
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    max_val = corr.max()
    shifts = np.where(corr == max_val)[0] - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, target, time_left, rng,
                              step0=0.5, beta=0.9,
                              max_iter=200000,
                              noise_interval=5000, noise_scale=0.02):
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        if noise_interval and (it % noise_interval == 0):
            noise = (rng.random(h.shape) - 0.5) * 2.0 * noise_scale
            h_new = project_onto_box_sum(h_new + noise, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def round_to_binary(h):
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    hb = np.zeros_like(h)
    hb[idx] = 1.0
    return hb

def binary_swap_anneal(h0, target, time_left, rng,
                      max_iters=400000,
                      T0=0.2, T_min=1e-5, decay=0.99998):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    best_h = project_onto_box_sum(best_h, target)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
