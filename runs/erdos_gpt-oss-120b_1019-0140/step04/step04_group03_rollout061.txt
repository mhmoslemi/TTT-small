def upsample_binary(arr, new_n):
    old_n = arr.shape[0]
    if new_n % old_n == 0:
        factor = new_n // old_n
        return np.repeat(arr, factor)
    else:
        # Use interpolation then threshold
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n-1, new_n)
        interp = np.interp(x_new, x_old, arr)
        return (interp > 0.5).astype(int)

def search_and_optimize(h_init, rng, start_time, global_budget, total_budget, time_budget_fraction):
    import time, numpy as np
    h = h_init.copy()
    n = h.shape[0]
    # compute S
    S = np.correlate(h, h, mode='full')
    center = n - 1
    mask_nonzero = np.ones_like(S, dtype=bool)
    mask_nonzero[center] = False
    minS = int(np.min(S[mask_nonzero]))
    # indices array for contributions
    inds = np.arange(-(n-1), n, dtype=int)

    # compute allowed time for this stage
    allowed = total_budget * time_budget_fraction
    # optional: ensure not exceed remaining global time
    # Actually track in while loop.

    # annealing loop
    step = 0
    max_steps = int(1e7)  # maybe large
    T0 = 0.02
    T_final = 1e-6
    while True:
        now = time.time()
        elapsed_global = now - start_time
        if elapsed_global >= total_budget - 2.0:
            break
        # possibly limit steps
        if step >= max_steps:
            break
        # temperature schedule
        progress = step / max_steps
        T = max(T_final, T0 * (1 - progress))

        # random swap
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        a = rng.choice(ones_idx)
        b = rng.choice(zeros_idx)

        # compute contributions for a and b
        # for a
        ip = a + inds
        im = a - inds
        mask_ip = (ip >= 0) & (ip < n)
        mask_im = (im >= 0) & (im < n)
        arr_a = np.zeros(2*n-1, dtype=np.int64)
        arr_a[mask_ip] = h[ip[mask_ip]]
        arr_a[mask_im] += h[im[mask_im]]
        # for b
        ipb = b + inds
        imb = b - inds
        mask_ipb = (ipb >= 0) & (ipb < n)
        mask_imb = (imb >= 0) & (imb < n)
        arr_b = np.zeros(2*n-1, dtype=np.int64)
        arr_b[mask_ipb] = h[ipb[mask_ipb]]
        arr_b[mask_imb] += h[imb[mask_imb]]

        arr_a[center] = 0
        arr_b[center] = 0

        delta = -arr_a + arr_b
        S_new = S + delta
        new_minS = int(np.min(S_new[mask_nonzero]))
        # acceptance
        if new_minS > minS or rng.random() < np.exp((new_minS - minS) / (T + 1e-12)):
            # accept
            h[a] = 0
            h[b] = 1
            S = S_new
            minS = new_minS
        step += 1

    # hill climbing after annealing
    # try deterministic improvements
    no_improve = 0
    max_no_improve = int(5000)
    while True:
        now = time.time()
        if now - start_time >= total_budget - 2.0:
            break
        if no_improve >= max_no_improve:
            break
        # random swap candidate
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        a = rng.choice(ones_idx)
        b = rng.choice(zeros_idx)
        # compute delta and new minS as before
        ip = a + inds
        im = a - inds
        mask_ip = (ip >= 0) & (ip < n)
        mask_im = (im >= 0) & (im < n)
        arr_a = np.zeros(2*n-1, dtype=np.int64)
        arr_a[mask_ip] = h[ip[mask_ip]]
        arr_a[mask_im] += h[im[mask_im]]
        ipb = b + inds
        imb = b - inds
        mask_ipb = (ipb >= 0) & (ipb < n)
        mask_imb = (imb >= 0) & (imb < n)
        arr_b = np.zeros(2*n-1, dtype=np.int64)
        arr_b[mask_ipb] = h[ipb[mask_ipb]]
        arr_b[mask_imb] += h[imb[mask_imb]]
        arr_a[center] = 0
        arr_b[center] = 0
        delta = -arr_a + arr_b
        S_new = S + delta
        new_minS = int(np.min(S_new[mask_nonzero]))
        if new_minS > minS:
            h[a] = 0
            h[b] = 1
            S = S_new
            minS = new_minS
            no_improve = 0
        else:
            no_improve += 1

    return h, minS

import time
import numpy as np

def random_binary_balanced(n, rng):
    """Return a binary array of length n with exactly n//2 ones."""
    arr = np.zeros(n, dtype=np.int64)
    arr[rng.choice(n, size=n // 2, replace=False)] = 1
    return arr

def compute_c5_from_h(h):
    """C5 from a float (0/1) array."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _contrib_array(h, idx, inds, center):
    """Contributions of position idx to all shifts (central entry set to 0)."""
    n = h.shape[0]
    ip = idx + inds
    im = idx - inds
    mask_ip = (ip >= 0) & (ip < n)
    mask_im = (im >= 0) & (im < n)
    arr = np.zeros_like(inds, dtype=np.int64)
    arr[mask_ip] = h[ip[mask_ip]]
    arr[mask_im] += h[im[mask_im]]
    arr[center] = 0
    return arr

def _anneal_and_hill_climb(h, rng, start_time, total_budget, fraction):
    """
    Simulated annealing followed by a greedy hill‑climb.
    Returns the best binary vector found and its minimal self‑overlap (minS).
    """
    n = h.shape[0]
    center = n - 1
    mask_nonzero = np.ones(2 * n - 1, dtype=bool)
    mask_nonzero[center] = False
    inds = np.arange(-(n - 1), n, dtype=int)

    # correlation of h with itself (ordered pairs)
    S = np.correlate(h, h, mode="full")
    minS = int(S[mask_nonzero].min())

    # time budget for this stage
    stage_deadline = start_time + total_budget * fraction

    # --------- simulated annealing ----------
    step = 0
    max_steps = 5_000_000
    T0 = 0.02
    T_end = 1e-6
    while True:
        now = time.time()
        if now >= stage_deadline - 2.0 or step >= max_steps:
            break
        if step % 200 == 0:
            # cheap progress check
            if now >= stage_deadline - 2.0:
                break

        # temperature
        T = max(T_end, T0 * (1.0 - step / max_steps))

        # pick a random 1 and 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        a = rng.choice(ones)
        b = rng.choice(zeros)

        # contributions of a and b
        arr_a = _contrib_array(h, a, inds, center)
        arr_b = _contrib_array(h, b, inds, center)

        delta = -arr_a + arr_b
        S_new = S + delta
        new_minS = int(S_new[mask_nonzero].min())

        # Metropolis acceptance
        if new_minS > minS or rng.random() < np.exp((new_minS - minS) / (T + 1e-12)):
            h[a] = 0
            h[b] = 1
            S = S_new
            minS = new_minS

        step += 1

    # --------- greedy hill‑climb ----------
    no_improve = 0
    max_no_improve = 5_000
    while True:
        now = time.time()
        if now >= stage_deadline - 2.0 or no_improve >= max_no_improve:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        a = rng.choice(ones)
        b = rng.choice(zeros)

        arr_a = _contrib_array(h, a, inds, center)
        arr_b = _contrib_array(h, b, inds, center)
        delta = -arr_a + arr_b
        S_new = S + delta
        new_minS = int(S_new[mask_nonzero].min())

        if new_minS > minS:
            h[a] = 0
            h[b] = 1
            S = S_new
            minS = new_minS
            no_improve = 0
        else:
            no_improve += 1

    return h, minS

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c5 = np.inf

    # ---- initial vector (if provided) ----
    if "initial_h_values" in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        init_bin = (init_arr > 0.5).astype(np.int64)
    else:
        init_bin = None

    # ---- Phase 1: small scale (n=64) ----
    n_small = 64
    if init_bin is not None and init_bin.shape[0] == n_small:
        h_small = init_bin.copy()
    elif init_bin is not None and n_small % init_bin.shape[0] == 0:
        repeat = n_small // init_bin.shape[0]
        h_small = np.repeat(init_bin, repeat)
    else:
        h_small = random_binary_balanced(n_small, rng)

    h_small_opt, minS_small = _anneal_and_hill_climb(
        h_small, rng, start, budget_s, fraction=0.10
    )
    c5_small = 1.0 - (2.0 / n_small) * minS_small
    if c5_small < best_c5:
        best_c5 = c5_small
        best_h = h_small_opt.astype(float)

    # ---- Phase 2: larger scale (n=256) ----
    n_large = 256
    factor = n_large // n_small
    h_large = np.repeat(h_small_opt, factor)

    # Use remaining time (≈90% of budget) for the large scale optimization
    h_large_opt, minS_large = _anneal_and_hill_climb(
        h_large, rng, start, budget_s, fraction=0.90
    )
    c5_large = 1.0 - (2.0 / n_large) * minS_large
    if c5_large < best_c5:
        best_c5 = c5_large
        best_h = h_large_opt.astype(float)

    # ---- fallback (should not be needed) ----
    if best_h is None:
        n_fallback = 64
        best_h = random_binary_balanced(n_fallback, rng).astype(float)
        best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, best_h.size
