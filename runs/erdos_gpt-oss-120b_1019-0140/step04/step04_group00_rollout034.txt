def gradient_smooth_max_corr(h, n, N, dx, beta):
    # compute corr via FFT
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n-1] * dx

    max_c = corr.max()
    # stable weights
    weights = np.exp(beta * (corr - max_c))
    w_sum = weights.sum()
    probs = weights / w_sum

    # gradient via weighted conv
    A = probs  # length 2n-1
    A_rev = A[::-1]

    # compute conv1 = A_rev * (1 - h) convolution
    fft_A_rev = np.fft.rfft(A_rev, n=N)
    fft_one_h = np.fft.rfft(1.0 - h, n=N)
    conv1 = np.fft.irfft(fft_A_rev * fft_one_h, n=N)[:n]

    # conv2 = A * h convolution
    fft_A = np.fft.rfft(A, n=N)
    fft_h = np.fft.rfft(h, n=N)
    conv2 = np.fft.irfft(fft_A * fft_h, n=N)[:n]

    grad = conv1 - conv2
    return grad, corr.max()

def smooth_gradient_descent(h0, time_left, n, target_sum, max_iters=200000, beta_start=5.0, beta_end=80.0, step0=0.5):
    h = project_onto_box_sum(h0, target_sum)
    best_h, best_s = h.copy(), compute_c5_fast(h) # compute_c5_fast defined earlier
    # Precompute N
    N = 1 << ((3*n - 2).bit_length())
    dx = 2.0 / n
    it = 1
    while it <= max_iters and time_left() > 0.0:
        # schedule beta linearly or exponentially: increase
        beta = beta_start + (beta_end - beta_start) * (it - 1) / max_iters
        # schedule step size
        eta = step0 / np.sqrt(it)
        grad, cur_s = gradient_smooth_max_corr(h, n, N, dx, beta)
        # now project after step
        h_new = project_onto_box_sum(h - eta * grad, target_sum)
        # evaluate true C5
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
            if best_s <= 0.3808:
                return best_h, best_s
        h = h_new
        it += 1
    return best_h, best_s

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    conv = np.fft.irfft(np.fft.rfft(h[::-1], n=N) *
                        np.fft.rfft(1.0 - h, n=N), n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_smooth_max_corr(h, n, N, dx, beta):
    # correlation (full) via FFT of size N
    fft_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1] * dx

    max_c = corr.max()
    exp_term = np.exp(beta * (corr - max_c))
    prob = exp_term / exp_term.sum()          # softmax probabilities
    A = prob
    A_rev = A[::-1]

    # conv1 = A_rev * (1 - h)
    fft_A_rev = np.fft.rfft(A_rev, n=N)
    conv1 = np.fft.irfft(fft_A_rev * fft_one, n=N)[:n]

    # conv2 = A * h
    fft_A = np.fft.rfft(A, n=N)
    fft_h = np.fft.rfft(h, n=N)
    conv2 = np.fft.irfft(fft_A * fft_h, n=N)[:n]

    grad = dx * (conv1 - conv2)
    return grad, corr.max()

def smooth_gradient_descent(h0, deadline, n, target_sum,
                           max_iters=200000,
                           beta_start=5.0, beta_end=80.0,
                           step0=0.5):
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    N = 1 << ((3 * n - 2).bit_length())
    dx = 2.0 / n
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        beta = beta_start + (beta_end - beta_start) * (it - 1) / max_iters
        eta = step0 / np.sqrt(it)
        grad, _ = gradient_smooth_max_corr(h, n, N, dx, beta)
        h_new = project_onto_box_sum(h - eta * grad, target_sum)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
            if best_s <= 0.3808:
                return best_h, best_s
        h = h_new
    return best_h, best_s

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(m):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(m)),
                       dtype=float, count=m)

def random_binary_balanced(m, rng):
    h = np.zeros(m, dtype=float)
    h[rng.choice(m, size=m // 2, replace=False)] = 1.0
    return h
