def compute_corr_counts(h):
    n = h.shape[0]
    M = 2*n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2*n-1]
    # rounding to nearest integer
    return np.rint(corr).astype(int)

def gradient_descent_momentum(h0, time_left, max_iter=20000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        # compute corr array
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # shifts achieving max
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient for these shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            # compute term1 and term2 as per compute_gradients_for_shifts
            if shift > 0:
                term1 = np.zeros(n); term1[:n-shift] = h[shift:]
                term2 = np.zeros(n); term2[shift:] = h[:n-shift]
            elif shift < 0:
                s = -shift
                term1 = np.zeros(n); term1[s:] = h[:n-s]
                term2 = np.zeros(n); term2[:n-s] = h[s:]
            else:
                continue
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h = h - eta * momentum
        # project onto box+sum
        h = project_onto_box_sum(h, s)
        score = compute_c5_fft(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def binary_greedy_opt(h, time_left, max_swaps=2000):

    corr = compute_corr_counts(h)
    max_corr = corr.max()
    swaps = 0

        idxes = np.where(corr == max_corr)[0]
        # random order
        np.random.shuffle(idxes)
        improved = False

            i_bad, j_good = shift_masks(h, shift)
            if i_bad.size > 0 and j_good.size > 0:
                # random choice
                i = i_bad[np.random.randint(i_bad.size)]
                j = j_good[np.random.randint(j_good.size)]
                # swap
                h[i], h[j] = 0, 1
                swaps += 1
                corr = compute_corr_counts(h)
                max_corr = corr.max()
                improved = True
                break
        if not improved:
            break
    return h

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT."""
    return np.max(compute_corr_scaled_fft(h))

def compute_corr_counts(h):
    """Integer correlation counts (no scaling)."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.rint(corr).astype(int)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h, dtype=float)
    for shift in shifts:
        if shift > 0:
            term1 = np.zeros(n, dtype=float)
            term2 = np.zeros(n, dtype=float)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        elif shift < 0:
            s = -shift
            term1 = np.zeros(n, dtype=float)
            term2 = np.zeros(n, dtype=float)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        else:
            continue
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def shift_masks(h, shift):
    """Return (i_bad, j_good) indices for a given shift."""
    n = h.size
    if shift > 0:
        i_range = np.arange(n - shift)
        hi = h[i_range]
        hj = h[i_range + shift]
        mask_bad = (hi == 1) & (hj == 0)
        mask_good = (hi == 0) & (hj == 1)
        i_bad = i_range[mask_bad]
        j_good = i_range[mask_good] + shift
    elif shift < 0:
        s = -shift
        i_range = np.arange(s, n)
        hi = h[i_range]
        hj = h[i_range - s]
        mask_bad = (hi == 1) & (hj == 0)
        mask_good = (hi == 0) & (hj == 1)
        i_bad = i_range[mask_bad]
        j_good = i_range[mask_good] - s
    else:
        i_bad = np.empty(0, dtype=int)
        j_good = np.empty(0, dtype=int)
    return i_bad, j_good

def binary_greedy_opt(h, time_left, rng, max_swaps=2000):
    """Greedy local swaps to lower max overlap for binary h."""
    n = h.size
    corr = compute_corr_counts(h)
    max_corr = corr.max()
    swaps = 0
    while swaps < max_swaps and time_left() > 0.05:
        idxes = np.where(corr == max_corr)[0]
        if idxes.size == 0:
            break
        rng.shuffle(idxes)
        improved = False
        for idx in idxes:
            shift = idx - (n - 1)
            i_bad, j_good = shift_masks(h, shift)
            if i_bad.size and j_good.size:
                i = i_bad[rng.integers(i_bad.size)]
                j = j_good[rng.integers(j_good.size)]
                h[i], h[j] = 0, 1
                swaps += 1
                corr = compute_corr_counts(h)
                max_corr = corr.max()
                improved = True
                break
        if not improved:
            break
    return h

def gradient_descent_momentum(h0, time_left, max_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on continuous h."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = corr.max()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h = h - eta * momentum
        h = project_onto_box_sum(h, s)
        score = compute_c5_fft(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        it += 1
    return best_h, best_score

def slsqp_refine(h0, time_left, max_iter=300):
    """SLSQP refinement minimizing max overlap."""
    if time_left() <= 0.0:
        return h0, compute_c5_fft(h0)
    n = h0.size
    s = n / 2.0
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - s

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_scaled_fft(h)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], s)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    rng = np.random.default_rng(seed)
    target = 0.3808
    best_h = None
    best_score = np.inf

    # Determine problem size
    if 'initial_h_values' in globals():

    if sc0 < best_score:
        best_score, best_h = sc0, h0.copy()
    if best_score <= target:
        return best_h, best_score, n

    # Phase 1: binary version of initial and greedy refinement
    binary_h0 = np.zeros_like(h0, dtype=int)
    idx_top = np.argpartition(-h0, n // 2)[:n // 2]
    binary_h0[idx_top] = 1
    binary_h0 = binary_greedy_opt(binary_h0, time_left, rng, max_swaps=2000)
    sc_bin0 = compute_c5_fft(binary_h0.astype(float))
    if sc_bin0 < best_score:
        best_score, best_h = sc_bin0, binary_h0.astype(float).copy()
        if best_score <= target:
            return best_h, best_score, n

    # Phase 2: Thue‑Morse binary seed
    h_tm = thue_morse(n).astype(int)
    h_tm = binary_greedy_opt(h_tm, time_left, rng, max_swaps=2000)
    sc_tm = compute_c5_fft(h_tm.astype(float))
    if sc_tm < best_score:
        best_score, best_h = sc_tm, h_tm.astype(float).copy()
        if best_score <= target:
            return best_h, best_score, n

    # Phase 3: random binary restarts with greedy + continuous refinement
    while time_left() > 30.0:
        # balanced random binary

        sc_rand = compute_c5_fft(h_rand.astype(float))
        if sc_rand < best_score:

            if best_score <= target:

        # continuous refinement
        h_cont, sc_cont = gradient_descent_momentum(h_rand.astype(float), time_left, max_iter=5000)
        if sc_cont < best_score:
            best_score, best_h = sc_cont, h_cont.copy()
            if best_score <= target:

        # occasional SLSQP polish

            h_ref, sc_ref = slsqp_refine(h_cont, time_left, max_iter=200)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()
                if best_score <= target:

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, n
