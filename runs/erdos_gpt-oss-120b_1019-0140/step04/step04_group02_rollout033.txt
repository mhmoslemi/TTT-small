    n = v.shape[0]
    # Quick handling of trivial cases
    if s <= 0:

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # generate thue-morse sequence bits
    # using bit parity
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

    return seq

import time
import numpy as np

def project_onto_box_sum(v, s):
    """Euclidean projection of v onto {0<=x_i<=1, sum x_i = s}."""
    n = v.size
    if s <= 0:
        return np.zeros_like(v)
    if s >= n:
        return np.ones_like(v)
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        tau = cssv[rho] / (rho + 1)
    else:
        tau = 0.0
    w = np.clip(v - tau, 0.0, 1.0)
    diff = s - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0.0) & (w < 1.0)
        if np.any(free):
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n, dtype=float)
    h[idx[:n // 2]] = 1.0
    return h

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def subgrad_worst_shift(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        m = -k
        grad[m:] += 1.0 - h[:n - m]
        grad[:n - m] -= h[m:]
    max_corr = corr[idx] * (2.0 / n)
    return grad, max_corr, k

def gradient_descent_worst(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        grad, cur, _ = subgrad_worst_shift(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        if it % 5000 == 0:
            h += rng.normal(scale=0.02, size=n)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score_new = compute_c5_fft(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
            if best_score <= 0.3808:
                break
        h = h_new
    return best_h, best_score
