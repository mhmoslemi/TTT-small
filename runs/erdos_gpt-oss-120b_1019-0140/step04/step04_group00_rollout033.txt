Implementation details:

compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2*n-1).bit_length())
    # use rfft: real FFT
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)[:2*n-1]
    return conv.max() * (2.0/n)

repair_balance(child):
    # count ones
    count = child.sum()
    target = n//2
    diff = count - target
    if diff > 0:
        # too many ones: need to flip diff ones to zero
        # get indices where child ==1, randomly select diff of them, set to 0
        ones_idx = np.where(child == 1)[0]
        # random select diff indices to set 0
        turn_off = rng.choice(ones_idx, size=diff, replace=False)
        child[turn_off] = 0
    elif diff < 0:
        zeros_idx = np.where(child == 0)[0]
        turn_on = rng.choice(zeros_idx, size=-diff, replace=False)
        child[turn_on] = 1
    # now balanced
    return child

def compute_gradient_max_shift(h):
    n = h.shape[0]
    # compute correlation via FFT to locate max shift(s)
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)[:2 * n - 1]
    mx = conv.max()
    # get index positions where conv == mx
    # Use np.where
    idxes = np.where(conv == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h, dtype=float)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= float(len(shifts))
    return grad

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.shape[0]
    target = n / 2.0
    h = project_onto_box_sum(h0, target)

def binary_swap_anneal(h0, rng, max_iters=400000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s = new
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def resample(arr, new_n):
    old_n = arr.shape[0]

    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def repair_balance(arr, target_sum, rng):
    # arr is numpy array of 0/1
    cur = int(arr.sum())
    diff = cur - target_sum
    if diff > 0:
        # too many ones
        ones_idx = np.where(arr == 1)[0]
        # random select diff of them to switch to 0
        turn_off = rng.choice(ones_idx, size=diff, replace=False)
        arr[turn_off] = 0
    elif diff < 0:
        zeros_idx = np.where(arr == 0)[0]
        turn_on = rng.choice(zeros_idx, size=-diff, replace=False)
        arr[turn_on] = 1
    # else balanced
    return arr

        return 0.0
    return max(0.0, _time_budget - (time.time() - _start_time))

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < target:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > target:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target) < 1e-12:
            break
        if fm > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    hf = h.astype(float, copy=False)
    h_rev = hf[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - hf, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)[:2 * n - 1]
    return conv.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        ns = n - shift
        term1 = np.zeros(n, dtype=h.dtype)
        term1[:ns] = h[shift:]
        term2 = np.zeros(n, dtype=h.dtype)
        term2[shift:] = h[:ns]
    else:
        s = -shift
        ns = n - s
        term1 = np.zeros(n, dtype=h.dtype)
        term1[s:] = h[:ns]
        term2 = np.zeros(n, dtype=h.dtype)
        term2[:ns] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)[:2 * n - 1]
    mx = conv.max()
    idxes = np.where(conv == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h, dtype=float)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= float(len(shifts))
    return grad

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.shape[0]
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def binary_swap_anneal(h0, rng, max_iters=400000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s = new
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def thue_morse_binary(n):
    arr = np.empty(n, dtype=np.int8)
    for i in range(n):
        arr[i] = bin(i).count('1') & 1
    return arr

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=np.int8)
    h[rng.choice(n, size=n // 2, replace=False)] = 1
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def tournament_selection(pop, fitness, tsize, rng):
    idxs = rng.integers(0, len(pop), size=tsize)
    sel_f = fitness[idxs]
    best_local = int(np.argmin(sel_f))
    return pop[idxs[best_local]]

def repair_balance(arr, target, rng):
    cur = int(arr.sum())
    diff = cur - target
    if diff > 0:
        ones = np.where(arr == 1)[0]
        turn_off = rng.choice(ones, size=diff, replace=False)
        arr[turn_off] = 0
    elif diff < 0:
        zeros = np.where(arr == 0)[0]
        turn_on = rng.choice(zeros, size=-diff, replace=False)
        arr[turn_on] = 1
    return arr

def crossover_two_point(p1, p2, rng, target):
    n = p1.shape[0]
    i = rng.integers(0, n - 1)
    j = rng.integers(i + 1, n)
    child = np.concatenate([p1[:i], p2[i:j], p1[j:]])
    child = repair_balance(child, target, rng)
    return child

def swap_mutation(arr, rng):
    ones = np.where(arr == 1)[0]
    zeros = np.where(arr == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return arr
    i = rng.choice(ones)
    j = rng.choice(zeros)
    arr[i] = 0
    arr[j] = 1
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    global _start_time, _time_budget
    _start_time = time.time()
    _time_budget = budget_s
    rng = np.random.default_rng(seed)
    n = 512
    target_sum = n // 2
    pop = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            init = resample(init, n)
        idx = np.argpartition(-init, target_sum)[:target_sum]
        binary = np.zeros(n, dtype=np.int8)
        binary[idx] = 1
        pop.append(binary)
    pop.append(thue_morse_binary(n))
    for _ in range(6):
        pop.append(random_binary_balanced(n, rng))
    pop_size = 40
    while len(pop) < pop_size:
        pop.append(random_binary_balanced(n, rng))
    if len(pop) > pop_size:
        pop = pop[:pop_size]
    fitness = np.array([compute_c5_fast(ind) for ind in pop])
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_s = float(fitness[best_idx])
    target_bound = 0.3808
    tournament_sz = 3
    cx_rate = 0.7
    mut_rate = 0.1
    while time_left() > 1.0:
        new_pop = [best_h.copy()]
        while len(new_pop) < pop_size:
            p1 = tournament_selection(pop, fitness, tournament_sz, rng)
            p2 = tournament_selection(pop, fitness, tournament_sz, rng)
            child = p1.copy()
            if rng.random() < cx_rate:
                child = crossover_two_point(p1, p2, rng, target_sum)
            if rng.random() < mut_rate:
                child = swap_mutation(child, rng)
            new_pop.append(child)
        pop = new_pop
        fitness = np.array([compute_c5_fast(ind) for ind in pop])
        cur_best = int(np.argmin(fitness))
        cur_s = float(fitness[cur_best])
        if cur_s < best_s:
            best_s = cur_s
            best_h = pop[cur_best].copy()
        if best_s <= target_bound:
            break
    best_h_float = best_h.astype(float)
    bh_ref, bs_ref = gradient_descent_momentum(best_h_float, step0=0.5, beta=0.9, max_iter=200000)
    if bs_ref < best_s:
        best_s = bs_ref
        best_h_float = bh_ref.copy()
    if time_left() > 0.5:
        idx = np.argpartition(-best_h_float, target_sum)[:target_sum]
        best_h_bin = np.zeros(n, dtype=np.int8)
        best_h_bin[idx] = 1
        bh_ann, bs_ann = binary_swap_anneal(best_h_bin, rng, max_iters=300000)
        if bs_ann < best_s:
            best_s = bs_ann
            best_h_float = bh_ann.astype(float)
    best_h_float = np.clip(best_h_float, 0.0, 1.0)
    best_h_float = project_onto_box_sum(best_h_float, target_sum)
    final_s = compute_c5_fast(best_h_float)
    if final_s < best_s:
        best_s = final_s
    return best_h_float, best_s, n
