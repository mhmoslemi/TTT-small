def adam_optimize(h0, max_iter, lr, beta1, beta2, eps, rng, max_time):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    target_sum = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter+1):
        if time.time() - start >= max_time:
            break
        # compute gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # occasionally add noise?
        if t % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=h.size)
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def annealing_refine(h0, total_steps, T0, Tf, rng, max_time):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    start = time.time()
    for step in range(total_steps):
        if time.time() - start >= max_time:
            break
        T = T0 * (1 - step/total_steps) + Tf * (step/total_steps) # linear sched
        i, j = rng.choice(n, size=2, replace=False)
        # compute max transferable
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # ensure constraints automatically satisfied
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    # trivial cases
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)

        else:
            lam_high = lam_mid
    # final projection
    return np.clip(v - lam_mid, lo, hi)

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi

    for _ in range(max_iter):

    return np.clip(v - lam_mid, lo, hi)

def adam_optimize(h0, max_iter, lr, beta1, beta2, eps, rng, max_time):
    start = time.time()
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    target_sum = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    b1_pow = beta1
    b2_pow = beta2
    for t in range(1, max_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shift_idxs)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - b1_pow)
        v_hat = v / (1 - b2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        if t % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=h.size)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        b1_pow *= beta1
        b2_pow *= beta2
    return best_h, best_score

def annealing_refine(h0, total_steps, T0, Tf, rng, max_time):
    start = time.time()
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    for step in range(total_steps):
        if time.time() - start >= max_time:
            break
        T = T0 * (1 - step / total_steps) + Tf * (step / total_steps)
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score
