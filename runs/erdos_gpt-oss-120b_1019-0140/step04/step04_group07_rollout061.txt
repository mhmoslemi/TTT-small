def smooth_gradient(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # Weights: w_i = exp(alpha * (corr_i - max_corr))
    # To avoid overflow, subtract max_corr.
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    w = w / w_sum
    # Compute forward sum = sum_s w_s * shifted h by +s
    # Compute conv for backward sum = sum_s w_s * h shifted by -s
    conv_back = np.convolve(h, w, mode='full')
    conv_fwd = np.convolve(h, w[::-1], mode='full')
    idx = n - 1
    backward_sum = conv_back[idx: idx + n]
    forward_sum = conv_fwd[idx: idx + n]
    grad = 1.0 - forward_sum - backward_sum
    return grad

Double-check: Since we use w normalized to sum 1, the gradient considers convex combination of per-shift gradients.

Potential improvement: Use weight only on top shifts to reduce cost; but conv already O(n^2) anyway.

def gradient_descent_smooth(h0, max_iters, step0, beta, rng, time_left, alpha_sched):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    grad_prev = np.zeros_like(h)
    momentum = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        # Determine alpha for this iteration
        # alpha_sched can be a function returning alpha based on iteration or time left
        alpha = alpha_sched(it, max_iters)
        grad = smooth_gradient(h, alpha)
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - step * momentum
        # project
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= best_score:
            # Accept improvement
            h = h_new
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
            # adapt step
            step = min(step0, step * 1.01)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            h = project_onto_box_sum(h + jitter, s)
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=40):
    n = v.size
    if s <= lo * n: return np.full_like(v, lo)
    if s >= hi * n: return np.full_like(v, hi)
    # Initial bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure bounds contain solution

def smooth_gradient(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # Normalized weights
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # Convolution
    conv_back = np.convolve(h, w, mode='full')
    conv_fwd = np.convolve(h, w[::-1], mode='full')
    idx = n - 1
    backward = conv_back[idx: idx + n]
    forward = conv_fwd[idx: idx + n]
    grad = 1.0 - forward - backward
    return grad

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=40):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # initial lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # define sum_func
    def sum_func(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust bounds if necessary (might be extreme)
    # ensure sum_func(lam_low) >= s >= sum_func(lam_high) (since decreasing function)
    # If not, expand
    for _ in range(10):
        if sum_func(lam_low) < s:
            lam_low = lam_low - (hi - lo) * n
        if sum_func(lam_high) > s:
            lam_high = lam_high + (hi - lo) * n
        else:
            break
    # binary search
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        fm = sum_func(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

Now smooth_gradient:

def smooth_gradient(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_back = np.convolve(h, w, mode='full')
    conv_fwd = np.convolve(h, w[::-1], mode='full')
    idx = n - 1
    backward = conv_back[idx: idx + n]
    forward = conv_fwd[idx: idx + n]
    grad = 1.0 - forward - backward
    return grad

def gradient_descent_smooth(h0, max_iters, step0, beta, rng, time_left, alpha_sched):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        alpha = alpha_sched(it, max_iters)
        grad = smooth_gradient(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.01)
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if it % 500 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=40):
    """Project v onto {lo <= x_i <= hi, sum x_i = s} using bisection on λ."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand bounds if needed
    for _ in range(10):
        if f(lam_low) < s:
            lam_low -= (hi - lo) * n
        if f(lam_high) > s:
            lam_high += (hi - lo) * n
        else:
            break

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1 - h(x+k)) dx approximated on grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def smooth_gradient(h, alpha):
    """Weighted gradient of the max‑correlation objective using softmax weights."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_back = np.convolve(h, w, mode='full')
    conv_fwd = np.convolve(h, w[::-1], mode='full')
    idx = n - 1
    backward = conv_back[idx: idx + n]
    forward = conv_fwd[idx: idx + n]
    grad = 1.0 - forward - backward
    return grad


def gradient_descent_smooth(h0, max_iters, step0, beta, rng, time_left, alpha_sched):
    """Projected gradient descent with momentum and a schedule for α."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        alpha = alpha_sched(it, max_iters)
        grad = smooth_gradient(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.01)
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 500 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Transfer small mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    """Move mass away from the worst shift contribution."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[: n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[: n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score
