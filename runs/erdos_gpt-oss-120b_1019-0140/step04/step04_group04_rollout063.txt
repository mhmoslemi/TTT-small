def simulated_annealing_search(h0, max_iter, rng, time_left):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    score = best_score
    # parameters
    init_temp = 0.005
    final_temp = 0.0001
    for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        # linear temperature schedule
        t = init_temp + (final_temp - init_temp) * (it / max_iter)
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        delta_sc = sc_new - score
        if delta_sc < 0 or rng.random() < math.exp(-delta_sc / t):
            h = h_new
            score = sc_new
            if sc_new < best_score:
                best_h, best_score = h_new.copy(), sc_new
    return best_h, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_search(h0, max_iter, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    init_temp = 0.006
    final_temp = 0.0001
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        t = init_temp + (final_temp - init_temp) * (it / max_iter)
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        dsc = sc_new - cur_score
        if dsc < 0 or rng.random() < math.exp(-dsc / t):
            h = h_new
            cur_score = sc_new
            if sc_new < best_score:
                best_h, best_score = h_new.copy(), sc_new
        if best_score <= target:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)
    # resolution loop
    for n in (64, 128, 256, 512):
        if time_left() <= 5:
            break
        if best_h.size != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)
        sc = compute_c5(base)
        if sc < best_score:
            best_h, best_score = base.copy(), sc
        # gradient descent phase
        remaining = time_left()
        if n <= 256:
            max_iter = int(min(150000, remaining * 70))
            step0 = 0.5 if n <= 128 else 0.3
        else:
            max_iter = int(min(80000, remaining * 30))
            step0 = 0.2
        if max_iter > 0:
            h_gd, sc_gd = gradient_descent_momentum_custom(
                base, n_iter=max_iter, step0=step0,
                beta=0.95, rng=rng, time_left=time_left)
            if sc_gd < best_score:
                best_h, best_score = h_gd.copy(), sc_gd
            base = h_gd
        # simulated annealing phase
        remaining = time_left()
        if remaining > 2:
            max_iter = int(min(300000, remaining * 120))
            h_sa, sc_sa = simulated_annealing_search(
                base, max_iter=max_iter, rng=rng,
                time_left=time_left, target=0.3808)
            if sc_sa < best_score:
                best_h, best_score = h_sa.copy(), sc_sa
        if best_score <= 0.3808:
            break
    # final refinement on best size
    if time_left() > 2:
        remaining = time_left()
        max_iter = int(min(400000, remaining * 150))
        h_final, sc_final = simulated_annealing_search(
            best_h, max_iter=max_iter, rng=rng,
            time_left=time_left, target=0.3808)
        if sc_final < best_score:
            best_h, best_score = h_final.copy(), sc_final
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
