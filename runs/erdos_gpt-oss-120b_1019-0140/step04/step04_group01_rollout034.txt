    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)

        if time_left_func() <= 0:
            break
        grad, cur_score = gradient_max_shifts(h)
        # we have cur_score as max correlation (score)
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        # Adam update: scaled by step0
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # project onto feasible set

        h = h_new
    # final best
    return best_h, best_score

    v = np.zeros_like(h)

    # Evaluate final best

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def compute_gradient(h, top_k=5):
    n = h.shape[0]
    corr = compute_corr_scaled_fft(h)
    k = min(top_k, corr.size)
    idxs = np.argpartition(-corr, k - 1)[:k]
    shifts = idxs - (n - 1)
    grad = compute_gradients_for_shifts(h, shifts)
    return grad, np.max(corr)

def adam_optimize(h0, time_left_func, max_iter=5000, step0=0.3, top_k=5):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    for t in range(1, max_iter + 1):
        if time_left_func() <= 0.0:
            break
        grad, cur_score = compute_gradient(h, top_k=top_k)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, h.size / 2.0)
    final_score = compute_c5_fft(best_h)
    if final_score < best_score:
        best_score = final_score
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.arange(old_n) / old_n
    x_new = np.arange(new_n) / new_n
    return np.interp(x_new, x_old, arr)

def binary_swap_anneal(h0, rng, time_left_func,
                      max_iters=200000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
