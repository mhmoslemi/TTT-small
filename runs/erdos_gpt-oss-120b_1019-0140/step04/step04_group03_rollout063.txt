def gradient_descent_custom(h0, n_iter, step0, beta, M, rng, max_time):
   start = time.time()
   h = h0.copy()
   target_sum = h.size / 2.0
   best_h = h.copy()
   best_score = compute_c5(h)
   momentum = np.zeros_like(h)
   for it in range(1, n_iter+1):
       if time.time() - start >= max_time:
           break
       corr = np.correlate(h, 1.0 - h, mode='full')
       # exclude zero shift maybe? Since its value is zero; but max is elsewhere.
       # Take top M indices.
       # Using argpartition may be faster; but for small M we can just use np.argpartition.
       idxes = np.argpartition(corr, -M)[-M:]
       # Ensure we have unique shifts; we can compute shifts = idxes - (h.size - 1)
       shifts = idxes - (h.size - 1)
       grad = compute_gradients_for_shifts(h, shifts)
       momentum = beta * momentum + (1.0 - beta) * grad
       eta = step0 / np.sqrt(it)
       h_new = h - eta * momentum
       # optionally add noise occasionally
       if (it % 5000) == 0:
           h_new += rng.normal(scale=0.02, size=h.size)
       # project
       np.clip(h_new, 0.0, 1.0, out=h_new)
       h_new = project_onto_box_sum(h_new, target_sum)
       score = compute_c5(h_new)
       if score < best_score:
           best_score = score
           best_h = h_new.copy()
       h = h_new
   return best_h, best_score

import time
import numpy as np
import scipy.ndimage as ndimage

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_custom(h0, n_iter, step0, beta, M, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_sc = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -M)[-M:]
        shifts = idx - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        if it % 500 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        np.clip(h_new, 0.0, 1.0, out=h_new)
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_sc
