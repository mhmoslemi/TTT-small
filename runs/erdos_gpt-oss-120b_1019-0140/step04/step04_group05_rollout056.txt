import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    arr[idx] = 1.0
    return arr

    start = time.time()
    step_factor = 0.8
    min_step_factor = 0.01
    dx = 2.0 / n

    while time.time() - start < time_limit:

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = corr.max()
        cur_score = cur_max * dx

        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
            if best_score <= target_c5:
                return best_h, best_score

        # worst lag
        max_idx = corr.argmax()
        k = max_idx - (n - 1)

        # if lag zero, just try a random balanced binary perturbation
        if k == 0:
            cand = random_balanced_binary(n, rng)
            cand_score = compute_c5(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
                if best_score <= target_c5:
                    return best_h, best_score
            continue

        # compute contributions for this lag
        if k > 0:
            hi = h[: n - k]
            hj = h[k:]
            cont = hi * (1.0 - hj)
            i_local = int(np.argmax(cont))
            i = i_local
            i2 = i + k
        else:  # k < 0
            d = -k
            hi = h[d:]
            hj = h[: n - d]
            cont = hi * (1.0 - hj)
            i_local = int(np.argmax(cont))
            i = i_local + d

        max_delta = min(h[i], 1.0 - h[i2])

                h[idx1] -= delta
                h[idx2] += delta

    best_h_big = project_onto_box_sum(best_h_big, n_big / 2.0)

    return best_h_big, final_score, n_big
