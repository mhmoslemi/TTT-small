def weighted_grad(h, w):
    n = h.size
    grad = np.ones_like(h)
    # iterate over shift indices
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        s = idx - (n - 1)
        # term1: h[i+s]
        i_start = max(0, -s)
        i_end = min(n - 1, n - 1 - s)
        if i_end >= i_start:
            grad[i_start:i_end+1] -= weight * h[i_start + s:i_end + s + 1]

        if i_end2 >= i_start2:
            grad[i_start2:i_end2+1] -= weight * h[i_start2 - s:i_end2 - s + 1]

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_array(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    return conv[:2 * n - 1]

def compute_c5_fast(h):
    n = h.shape[0]
    corr = compute_corr_array(h) * (2.0 / n)
    return corr.max()

def softmax_weights(corr, delta):
    m = corr.max()
    exp_vals = np.exp((corr - m) / delta)
    return exp_vals / exp_vals.sum()

def weighted_grad(h, w):
    n = h.size
    grad = np.ones_like(h)
    offset = n - 1
    # iterate over all shifts
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - offset
        # term1: h[i+s]
        i_start = max(0, -s)
        i_end = min(n - 1, n - 1 - s)
        if i_end >= i_start:
            grad[i_start:i_end + 1] -= weight * h[i_start + s:i_end + s + 1]
        # term2: h[i-s]
        i_start2 = max(0, s)
        i_end2 = min(n - 1, n - 1 + s)
        if i_end2 >= i_start2:
            grad[i_start2:i_end2 + 1] -= weight * h[i_start2 - s:i_end2 - s + 1]
    return grad

def adam_optimize(h0, time_left, max_iter=8000, lr=0.05,
                  delta_init=0.02, delta_final=0.001):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_h, best_s = h.copy(), compute_c5_fast(h)
    half = max_iter // 2
    for it in range(1, max_iter + 1):
        if time_left() < 0.01:
            break
        corr = compute_corr_array(h) * (2.0 / n)
        delta = delta_init if it <= half else delta_final
        w = softmax_weights(corr, delta)
        grad = weighted_grad(h, w)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h - step, target)
        s = compute_c5_fast(h)
        if s < best_s:
            best_s, best_h = s, h.copy()
        if best_s <= 0.3808:
            break
    return best_h, best_s

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
