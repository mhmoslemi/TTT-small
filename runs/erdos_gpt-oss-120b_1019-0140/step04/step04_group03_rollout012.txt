Implementation of compute_corr_fft:

def compute_corr(h):
    n = h.shape[0]
    a = 1.0 - h
    # reverse a for correlation: a_rev = a[::-1]
    a_rev = a[::-1]
    L = 1 << ((2*n - 1).bit_length())  # next power of 2 >= 2n-1
    H = np.fft.rfft(np.concatenate([h, np.zeros(L - n)]))
    A = np.fft.rfft(np.concatenate([a_rev, np.zeros(L - n)]))
    corr = np.fft.irfft(H * A, n=L)
    # take first 2n-1 entries
    corr = corr[:2*n-1]
    return corr

def compute_softmax_grad(h, tau=0.005):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    # subtract max for numerical stability:
    exp_weights = np.exp((corr - max_corr) / tau)
    w = exp_weights / np.sum(exp_weights)
    # compute grad from w
    n = h.size
    # w length = 2n-1. offset = n-1 corresponds to shift 0
    # term1: sum_k w_k * (1 - h_{i+k})
    # compute correlation of w with (1 - h)
    a = 1.0 - h
    # correlation of w (len 2n-1) with a (len n)
    # Use np.correlate (w, a)
    # but we need fast method; but n <= 1024, we can use np.correlate directly.

def compute_softmax_grad(h, tau):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    exp_w = np.exp((corr - max_corr) / tau)
    w = exp_w / np.sum(exp_w)
    n = h.shape[0]
    a = 1.0 - h
    # grad1: correlation of w with a
    corr_a = np.correlate(w, a, mode='full')
    grad1 = corr_a[(n-1):(n-1 + n)]

    corr_h = np.correlate(w_rev, h, mode='full')
    grad2 = corr_h[(n-1):(n-1 + n)]
    grad = grad1 - grad2
    return grad, max_corr

import time

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

            if max_move <= 0.0:
                continue
        delta = max_move * rng.random() * min(1.0, T / 0.01 + 0.1)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def local_swap_search(h, cur_score, start, budget_s, rng):
    n = h.shape[0]
    best_h = h.copy()
    best_score = cur_score
    remaining = budget_s - (time.time() - start)
    max_swaps = int(min(200000, remaining * 2000))
    for _ in range(max_swaps):
        if time.time() - start >= budget_s:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if np.isclose(best_h[i], best_h[j]):
            continue
        h_new = best_h.copy()
        h_new[i], h_new[j] = best_h[j], best_h[i]
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_final = 512
    scales = [64, 128, 256, n_final]
    best_h = None
    best_score = np.inf

    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    for idx, n in enumerate(scales):
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # random binary sampling
        if budget_s - (time.time() - start) <= 2.0:
            break
        samples = 200 if n >= 256 else 400
        for _ in range(samples):
            if budget_s - (time.time() - start) <= 1.0:
                break
            h_rand = random_binary_balanced(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score = sc
                best_h = h_rand.copy()

        # gradient descent restarts
        restarts = 2 if n < 256 else 1
        for r in range(restarts):
            if budget_s - (time.time() - start) <= 5.0:
                break
            if r == 0:
                h0 = base.copy()
            else:
                h0 = base.copy()
                h0 += rng.normal(scale=0.02, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, n / 2.0)
            remaining = budget_s - (time.time() - start)
            max_iters = int(min(20000, max(5000, remaining * (20000.0 / max(64, n)))))
            tau_start = 0.02 if n <= 256 else 0.01
            tau_end = 0.001
            step0 = 0.5 if n <= 256 else 0.3
            h_opt, sc_opt = gradient_descent_softmax(
                h0, max_iters, step0, 0.9, rng, tau_start, tau_end, start, budget_s
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()
            base = h_opt

        # targeted annealing refinement
        if best_h is not None and best_h.size == n and budget_s - (time.time() - start) > 5.0:
            h_cur = best_h.copy()
            cur_score = best_score
            h_cur, cur_score = targeted_annealing(
                h_cur, cur_score, start, budget_s, rng
            )
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
            base = best_h

        # prepare for next scale
        if idx < len(scales) - 1:
            base = best_h if best_h is not None else base

    # final refinement
    if best_h is not None and budget_s - (time.time() - start) > 5.0:
        n = best_h.size
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(100000, max(20000, remaining * (20000.0 / max(64, n)))))
        h_ref, sc_ref = gradient_descent_softmax(
            best_h, max_iters, 0.2, 0.9, rng, 0.005, 0.0005, start, budget_s
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()
        if budget_s - (time.time() - start) > 2.0:
            best_h, best_score = local_swap_search(best_h, best_score, start, budget_s, rng)
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    if best_h is None:
        n = n_final
        best_h = np.full(n, 0.5, dtype=float)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
