def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_fourier_balanced(n, rng, K=5):
    x = np.arange(n) * (2.0 / n)
    a = rng.normal(scale=0.1, size=K)
    b = rng.normal(scale=0.1, size=K)
    h = np.full(n, 0.5)
    for k in range(1, K+1):
        h += a[k-1] * np.sin(np.pi * k * x) + b[k-1] * np.cos(np.pi * k * x)
    # clip before projection
    h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # tighten bounds if needed
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def random_fourier_balanced(n, rng, K=5):
    x = np.arange(n) * (2.0 / n)
    a = rng.normal(scale=0.15, size=K)
    b = rng.normal(scale=0.15, size=K)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        h += a[k - 1] * np.sin(np.pi * k * x) + b[k - 1] * np.cos(np.pi * k * x)
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_sc = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idx - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2500 == 0:
            h_new += rng.normal(scale=0.008, size=n)
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_sc

def simulated_annealing_refine(h, rng, budget, start_time):
    n = h.size
    cur = h.copy()
    cur_sc = compute_c5(cur)
    best = cur.copy()
    best_sc = cur_sc
    steps = int(min(80000, max(20000, (budget - (time.time() - start_time)) * 600.0)))
    T0, T1 = 0.02, 1e-6
    for step in range(steps):
        if time.time() - start_time >= budget:
            break
        T = T0 * ((steps - step) / steps) + T1
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(cur[i], 1.0 - cur[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_sc = compute_c5(new)
        if new_sc < cur_sc or rng.random() < np.exp(-(new_sc - cur_sc) / T):
            cur, cur_sc = new, new_sc
            if new_sc < best_sc:
                best, best_sc = new.copy(), new_sc
    return best, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    scales = [64, 128, 256, 512, 1024]
    best_h = None
    best_score = np.inf

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    for idx, n in enumerate(scales):

        base = project_onto_box_sum(base, n / 2.0)

        # Fourier contender
        fourier = random_fourier_balanced(n, rng, K=5)
        fourier = project_onto_box_sum(fourier, n / 2.0)
        if compute_c5(fourier) < compute_c5(base):
            base = fourier

        # random binary sampling
        samp_lim = 300 if n >= 256 else 500
        for _ in range(samp_lim):
            if time.time() - start >= budget_s - 1.0:

            h_rand = random_binary_balanced(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_h, best_score = h_rand.copy(), sc

        # gradient descent restarts
        restarts = 5 if n >= 256 else 3
        for r in range(restarts):
            if time.time() - start >= budget_s - 5.0:
                break
            h0 = base.copy()
            if r > 0:
                h0 += rng.normal(scale=0.04, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, n / 2.0)
            remain = budget_s - (time.time() - start)
            max_iters = int(min(250000, max(20000, remain * (20000.0 / max(64, n)))))
            h_opt, sc_opt = gradient_descent_momentum_custom(
                h0, n_iter=max_iters, step0=0.5, beta=0.9,
                rng=rng, max_time=remain)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
            base = h_opt

        # simulated annealing on current best of this scale
        if best_h is not None and best_h.size == n and time.time() - start < budget_s - 5.0:
            best_h, best_score = simulated_annealing_refine(
                best_h, rng, budget_s, start)

        # prepare for next scale
        if idx < len(scales) - 1 and best_h is not None:
            base = upsample(best_h, scales[idx + 1])

    # final long refinement on the finest scale
    if best_h is not None and time.time() - start < budget_s - 5.0:
        n = best_h.size
        remain = budget_s - (time.time() - start)
        max_iters = int(min(300000, max(50000, remain * (50000.0 / max(64, n)))))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=max_iters, step0=0.2, beta=0.9,
            rng=rng, max_time=remain)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

    if best_h is None:
        n = 512
        best_h = project_onto_box_sum(random_binary_balanced(n, rng), n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
