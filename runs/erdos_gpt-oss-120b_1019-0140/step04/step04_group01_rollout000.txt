import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # widen interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    """Full correlation (h * (1-h)) scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5(h):
    """Maximum overlap integral C5 for vector h."""
    return np.max(compute_corr_scaled_fft(h))

def compute_gradients_for_shifts(h, shifts):
    """Gradient of C5 approximated by averaging over given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.6, beta=0.95, max_iter=200000):
    """Projected gradient descent with momentum, time‑aware."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left, max_iters=500000,
                      T0=0.3, decay=0.99996, Tmin=1e-5):
    """Swap‑annealing for binary vectors."""
    h = h0.copy()
    n = h.size
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5(h)
        delta = new - cur
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_hill_climb(h0, rng, time_left, sigma=0.02, tries_per_sec=1000):
    """Gaussian perturbations with acceptance if better."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best = compute_c5(h)
    best_h = h.copy()
    # approximate budget based on tries_per_sec
    while time_left() > 0.0:
        cand = h + rng.normal(scale=sigma, size=n)
        np.clip(cand, 0.0, 1.0, out=cand)
        cand = project_onto_box_sum(cand, s)
        sc = compute_c5(cand)
        if sc < best:
            best = sc
            best_h = cand.copy()
            h = cand  # move base point
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808

    # Determine base resolution
    if 'initial_h_values' in globals():
        n = int(len(initial_h_values))
        h0 = np.asarray(initial_h_values, dtype=float)
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 512
        h0 = random_binary_balanced(n, rng)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    if best_score <= target:
        return best_h, best_score, best_h.size

    # Phase 1: momentum gradient descent (medium resolution)
    if time_left() > 30.0:
        h1, sc1 = gradient_descent_momentum(best_h, time_left,
                                            step0=0.6, beta=0.95, max_iter=200000)
        if sc1 < best_score:
            best_h, best_score = h1, sc1
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 2: random Gaussian hill climbing
    if time_left() > 20.0:
        h2, sc2 = random_hill_climb(best_h, rng, time_left,
                                    sigma=0.025)
        if sc2 < best_score:
            best_h, best_score = h2, sc2
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 3: binary conversion and annealing
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_h, best_score = h_bin, sc_bin
    if best_score <= target:
        return best_h, best_score, best_h.size

    if time_left() > 10.0:
        h3, sc3 = binary_swap_anneal(best_h, rng, time_left,
                                     max_iters=500000, T0=0.35,
                                     decay=0.99996, Tmin=1e-5)
        if sc3 < best_score:
            best_h, best_score = h3, sc3
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 4: upsample to higher resolution and refine
    if time_left() > 30.0:
        n_high = min(1024, n * 2)
        h_high = upsample(best_h, n_high)
        h_high = project_onto_box_sum(h_high, n_high / 2.0)
        sc_high = compute_c5(h_high)
        if sc_high < best_score:
            best_h, best_score = h_high, sc_high
        if best_score <= target:
            return best_h, best_score, best_h.size
        # gradient descent at high resolution
        h4, sc4 = gradient_descent_momentum(best_h, time_left,
                                            step0=0.5, beta=0.96,
                                            max_iter=150000)
        if sc4 < best_score:
            best_h, best_score = h4, sc4
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 5: final random restarts (continuous)
    n = best_h.size
    while time_left() > 5.0:
        v = rng.random(n)
        v = project_onto_box_sum(v, n / 2.0)
        h_tmp, sc_tmp = gradient_descent_momentum(v, time_left,
                                                  step0=0.5, beta=0.95,
                                                  max_iter=5000)
        if sc_tmp < best_score:
            best_h, best_score = h_tmp, sc_tmp
            if best_score <= target:
                break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
