def run(seed=42,budget_s=1100,**kwargs):

    import time, numpy as np, scipy.optimize as opt

    rng = np.random.default_rng(seed)

    start = time.time()
    def remaining():
        return budget_s - (time.time() - start)

    # Determine parameters
    n_de = 128
    n_final = 512
    K = 8  # number of Fourier terms

    # Build fourier matrix for n_de
    B_de = build_fourier_matrix(n_de, K)

    # define evaluation function
    target_sum_de = n_de / 2.0
    def eval_f(p):
        h = B_de @ p + 0.5
        h = project_onto_box_sum(h, target_sum_de)
        return compute_c5(h)

    # Determine maxiter based on remaining time: allocate say 60% to DE
    time_for_de = min(remaining()*0.6, 500)  # at most 500 sec
    # estimate maxiter: keep detection within time
    # approximate time per eval: we can estimate by a small test: but we can't run now.
    # We'll just pick maxiter = 2000 or until time.
    maxiter = 1000
    # We'll use callback to break early if time low.
    def de_callback(xk, convergence):
        return remaining() < 5.0  # stop if <5 sec left
    # Run DE
    bounds = [(-0.5,0.5)] * (2*K)
    result = opt.differential_evolution(eval_f, bounds, maxiter=maxiter, popsize=15,
                                        polish=False, seed=seed, callback=de_callback,
                                        updating='deferred', workers=1)
    p_best = result.x
    h_de = project_onto_box_sum(B_de @ p_best + 0.5, target_sum_de)
    best_h = h_de.copy()
    best_score = compute_c5(best_h)

    # Upsample to final n
    if remaining() > 5.0:
        B_final = build_fourier_matrix(n_final, K)
        target_sum_final = n_final/2.0
        h_large = project_onto_box_sum(B_final @ p_best + 0.5, target_sum_final)
        # maybe try also upsample from DE h directly (scaling) but we use same p.
        new_score = compute_c5(h_large)
        if new_score < best_score:

    # Further refinement: gradient descent momentum
    if remaining() > 10.0:
        # Determine iterations based on remaining time
        max_iters = int(min(200000, max(50000, remaining() * (50000.0 / max(64, best_h.size)))))

                                 n_iter=max_iters,
                                 step0=0.2,
                                 beta=0.9,
                                 rng=rng,
                                 max_time=remaining())
        best_h = project_onto_box_sum(best_h, best_h.size/2.0)
        best_score = compute_c5(best_h)

    # Simulated annealing final refinement
    if remaining() > 30.0:
        n = best_h.size
        cur_h = best_h.copy()
        cur_score = best_score
        total_steps = int(min(20000, max(5000, remaining()*500.0)))
        T0 = 0.01
        T_final = 1e-6
        for step in range(total_steps):
            if remaining() <= 0.0:
                break
            T = T0 * ((total_steps - step) / total_steps) + T_final
            i,j = rng.choice(n, size=2, replace=False)
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0.0:
                continue
            delta = max_move * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_score = compute_c5(new_h)
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                cur_h = new_h
                cur_score = new_score
                if cur_score < best_score:
                    best_score, best_h = cur_score, cur_h.copy()
        best_h = project_onto_box_sum(best_h, best_h.size/2.0)

        best_score = compute_c5(best_h)

import time
import numpy as np
import scipy.optimize as _opt

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def build_fourier_matrix(n, K):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    cols = []
    for k in range(1, K + 1):
        cols.append(np.cos(np.pi * k * x))
        cols.append(np.sin(np.pi * k * x))
    return np.column_stack(cols)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def remaining():
        return max(0.0, budget_s - (time.time() - start))

    # Parameters
    n_de = 128
    n_final = 512
    K = 8  # Fourier terms

    # Build Fourier matrix for DE stage
    B_de = build_fourier_matrix(n_de, K)
    target_de = n_de / 2.0

    # Evaluation for DE
    def eval_f(p):
        h = B_de @ p + 0.5
        h = project_onto_box_sum(h, target_de)
        return compute_c5(h)

    # DE callback respecting time
    def de_callback(xk, convergence):
        return remaining() < 5.0

    # Run differential evolution
    bounds = [(-0.5, 0.5)] * (2 * K)
    maxiter = 1200  # generous; will stop early via callback
    result = _opt.differential_evolution(
        eval_f,
        bounds,
        maxiter=maxiter,
        popsize=15,
        polish=False,
        seed=seed,
        callback=de_callback,
        updating='deferred',
        workers=1,
    )
    p_best = result.x
    h_de = project_onto_box_sum(B_de @ p_best + 0.5, target_de)
    best_h = h_de.copy()
    best_score = compute_c5(best_h)

    # Upsample to final resolution using same Fourier coeffs
    if remaining() > 5.0:
        B_fin = build_fourier_matrix(n_final, K)
        target_fin = n_final / 2.0
        h_fin = project_onto_box_sum(B_fin @ p_best + 0.5, target_fin)
        sc_fin = compute_c5(h_fin)
        if sc_fin < best_score:
            best_h, best_score = h_fin, sc_fin

    # Momentum gradient descent refinement
    if remaining() > 10.0:
        n_cur = best_h.size
        max_iters = int(min(200000,
                            max(50000, remaining() * (50000.0 / max(64, n_cur)))))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h,
            n_iter=max_iters,
            step0=0.2,
            beta=0.9,
            rng=rng,
            max_time=remaining(),
        )
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    # Simulated annealing final pass
    if remaining() > 30.0:
        n_cur = best_h.size
        cur_h = best_h.copy()
        cur_score = best_score
        total_steps = int(min(20000, max(5000, remaining() * 500.0)))
        T0 = 0.01
        T_final = 1e-6
        for step in range(total_steps):
            if remaining() <= 0.0:
                break
            T = T0 * ((total_steps - step) / total_steps) + T_final
            i, j = rng.choice(n_cur, size=2, replace=False)
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0.0:
                continue
            delta = max_move * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_score = compute_c5(new_h)
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                cur_h, cur_score = new_h, new_score
                if cur_score < best_score:
                    best_score, best_h = cur_score, cur_h.copy()
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    # Fallback if nothing succeeded
    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = np.asarray(initial_h_values, dtype=float)
            best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        else:
            best_h = project_onto_box_sum(thue_morse(n_final), n_final / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
