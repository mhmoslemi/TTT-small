def gradient_maxlag(h):
    n = h.shape[0]
    corr = np.convolve(h, 1 - h, mode='full')
    lmax = np.argmax(corr)
    L = lmax - (n - 1)
    g = np.zeros_like(h)
    if L >= 0:
        # i from 0 to n-1-L => term (1 - h[i+L])
        g[:n-L] += 1 - h[L:]
    # else L < 0 => i from -L to n-1 => term (1 - h[i+L]) where i+L >=0
    else:
        # L negative => -L >0
        g[-L:] += 1 - h[:n+L]  # because L negative, n+L < n
    # second term: -h[i-L]
    if L <= 0:
        # i-L >=0 => i >= L? but L <=0, i-L from 0..n-1+L?
        g[:n+L] -= h[-L:]
    else:
        # L>0, i-L >=0 for i>=L, i up to n-1 -> indices L..n-1
        g[L:] -= h[:n-L]
    return g, corr[lmax] * (2.0/n)

    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initialize base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    # Ensure base feasibly (project)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    # scales and allocation fractions: allocate more time to larger n
    scales = [64, 128, 256, 512]
    # fractions of remaining time for each step (like 0.15, 0.25, 0.35, 0.25)
    alloc_fractions = {64: 0.15, 128: 0.25, 256: 0.35, 512: 0.25}
    best_h = base
    best_score = compute_c5(best_h)

    for n in scales:
        if time_left() <= 0.0:
            break
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n/2.0)
        cur_score = compute_c5(best_h)

            best_h = best_h.copy()
        # allocate time
        alloc = alloc_fractions[n] * budget_s
        # If not enough time left, reduce
        if alloc > time_left():
            alloc = time_left()
        # Run gradient descent on this n
        opt_h, opt_score = pgd_optimize(best_h, alloc, rng)
        # Compare
        if opt_score < best_score:

    # final local random improvements

    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
