def compute_c5(h):
    n = h.shape[0]
    # compute correlation h * (1-h)
    a = h
    b = 1 - h
    # compute full correlation using FFT
    # pad length to next power of two >= 2*n
    L = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(A * B, n=L)[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def gradient_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n-s] = h[s:]
            term2[s:] = h[:n-s]
        else:
            s_abs = -s
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s_abs:] = h[:n-s_abs]
            term2[:n-s_abs] = h[s_abs:]
        grad += (1 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, tol=1e-12, max_iter=100):
    # solves projection onto [lo,hi] with sum = s
    # use binary search on lambda
    n = v.size
    lo_total = lo * n
    hi_total = hi * n
    if s <= lo_total:
        return np.full_like(v, lo)
    if s >= hi_total:
        return np.full_like(v, hi)
    # initial bounds for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # ensure f(lam_low) >= s? Actually f(lambda)=sum clip(v-lambda, lo, hi)
    # As lambda increases, v - lambda decreases, so clip sum decreases.
    # So f is decreasing.
    # So we can binary search to find lambda s.t. f(lambda) = s.
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Expand bounds if needed
    while f(lam_low) < s:
        lam_low = lam_low - (hi - lo) - 1.0 # big decrement
    while f(lam_high) > s:
        lam_high = lam_high + (hi - lo) + 1.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) <= tol:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def gradient_descent_momentum(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    # momentum vector
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0: break
        # compute correlation via FFT
        n_full = 2*n - 1
        L = 1 << ((n_full).bit_length())
        # compute correlation O(n log n)
        a = h
        b = 1 - h
        A = np.fft.rfft(a, n=L)
        B = np.fft.rfft(b, n=L)
        corr = np.fft.irfft(A * B, n=L)[:n_full]
        max_corr = np.max(corr)
        # find shifts achieving max within tolerance
        # define tolerance relative? we use absolute tolerance 1e-12 times maybe scaling (max_corr)
        tol = 1e-12
        idxes = np.where(corr >= max_corr - tol)[0]
        shifts = idxes - (n - 1)
        # compute gradient
        grad = gradient_for_shifts(h, shifts)
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        # step size schedule
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise injection
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # projection
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            # optional early exit threshold
            if best_score <= 0.3808:
                return best_h, best_score
        h = h_new
    return best_h, best_score

import time
import numpy as np

def time_left(start, budget):
    return max(0.0, budget - (time.time() - start))

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, tol=1e-12, max_iter=60):
    """Projection of v onto [lo,hi]^n with sum = s."""
    n = v.size
    lo_total = lo * n
    hi_total = hi * n
    if s <= lo_total:
        return np.full_like(v, lo)
    if s >= hi_total:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure bounds enclose the solution
    while f(lam_low) < s:
        lam_low -= (hi - lo) + 1.0
    while f(lam_high) > s:
        lam_high += (hi - lo) + 1.0

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) <= tol:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def correlate_full(a, b):
    """Linear full correlation a ⋆ b (like np.correlate with mode='full')."""
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    b_rev = b[::-1]
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b_rev, n=L)
    c = np.fft.irfft(A * B, n=L)[:n + m - 1]
    return c

def compute_c5(h):
    """Return C5 for discretized h."""
    n = h.shape[0]
    corr = correlate_full(h, 1.0 - h)
    return np.max(corr) * (2.0 / n)

def gradient_for_shifts(h, shifts):
    """Subgradient for the max-correlation shifts."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - s] = h[s:]
            term2[s:] = h[:n - s]
        else:
            s_abs = -s
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s_abs:] = h[:n - s_abs]
            term2[:n - s_abs] = h[s_abs:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points (periodic domain)."""
    if arr.shape[0] == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, arr.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_descent_momentum(h0, n_iter, step0, beta, rng, start, budget):
    """Projected gradient descent with momentum on the max-correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left(start, budget) <= 0:
            break

        # correlation and shifts achieving the maximum
        corr = correlate_full(h, 1.0 - h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)

        # subgradient for those shifts
        grad = gradient_for_shifts(h, shifts)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        # occasional noise injection to escape plateaus
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        # project back to feasible set
        h_new = project_onto_box_sum(h_new, s)

        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= 0.3808:
                return best_h, best_score
        h = h_new

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

    # initial construction

    best_h = base.copy()
    best_score = compute_c5(best_h)

    for target_n in (256, 512, 1024):
        if time_left(start, budget_s) < 8:

        # upsample current best to target size
        h = upsample(best_h, target_n)
        h = project_onto_box_sum(h, target_n / 2.0)

        sc = compute_c5(h)
        if sc < best_score:
            best_h, best_score = h.copy(), sc

        # allocate a portion of remaining time to gradient descent at this scale
        tl = time_left(start, budget_s)
        max_iters = int(min(150000, tl * 8000))
        if max_iters < 500:

    # final refinement at the current best resolution
    tl = time_left(start, budget_s)
    if tl > 5:
        fin_iters = int(min(80000, tl * 6000))
        best_h, best_score = gradient_descent_momentum(
            best_h, n_iter=fin_iters, step0=0.2,
            beta=0.95, rng=rng, start=start, budget=budget_s)

    # stochastic local search until time runs out
    while time_left(start, budget_s) > 3:
        n = best_h.shape[0]

        # jittered candidate
        cand = best_h + rng.normal(scale=0.02, size=n)
        cand = project_onto_box_sum(cand, n / 2.0)

        tl = time_left(start, budget_s)
        sub_iters = int(min(3000, tl * 2000))
        cand_opt, cand_sc = gradient_descent_momentum(

            beta=0.9, rng=rng, start=start, budget=budget_s)

        if cand_sc < best_score:
            best_h, best_score = cand_opt.copy(), cand_sc
            if best_score <= 0.3808:
                break

        # simple mass transfer move
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.uniform(0.01, 0.5)
            moved = best_h.copy()
            moved[i] -= delta
            moved[j] += delta
            sc_moved = compute_c5(moved)
            if sc_moved < best_score:
                best_h, best_score = moved, sc_moved

    best_h = project_onto_box_sum(best_h, best_h.shape[0] / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.shape[0]
