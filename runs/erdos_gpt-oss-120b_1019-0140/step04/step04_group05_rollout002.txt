def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))
    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.size/2)
    best_h = base.copy()
    best_score = compute_c5(best_h)
    levels = [64, 128, 256, 512, 1024]  # maybe 1024 if time
    # allocate fractions such that sum <= budget
    fractions = {64: 0.05, 128: 0.1, 256: 0.2, 512: 0.4, 1024: 0.25}
    for n in levels:
        if time_left() <= 0:
            break
        if best_h.size != n:
            base = upsample(best_h, n)  # upsample current best
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n/2)
        # allocate time slice
        alloc = budget_s * fractions.get(n, 0.1)
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0:
            continue
        opt_h, opt_score = simulated_annealing_improved(base, alloc, rng)
        if opt_score < best_score:
            best_score = opt_score
            best_h = opt_h.copy()
        else:
            # still update base for next level
            best_h = opt_h.copy()
    # final refinement using remaining budget
    remaining = time_left()
    if remaining > 0.5:
        opt_h, opt_score = simulated_annealing_improved(best_h, remaining, rng)
        if opt_score < best_score:
            best_h = opt_h
            best_score = opt_score
    # final projection
    best_h = project_onto_box_sum(best_h, best_h.size/2)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def simulated_annealing_improved(h0, time_limit, rng,
                                 init_temp=0.02, final_temp=1e-6,
                                 alpha=0.9995):
    """SA that moves mass from high to low values, respecting the sum constraint."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    T = init_temp
    start = time.time()
    while True:
        if time.time() - start >= time_limit:
            break
        if T <= final_temp:
            break

        sum_h = h.sum()                     # = n/2
        sum_one = n - sum_h                 # = n/2 as well
        p_i = h / sum_h
        p_j = (1.0 - h) / sum_one
        i = rng.choice(n, p=p_i)
        j = rng.choice(n, p=p_j)
        if i == j:
            # retry a few times
            for _ in range(2):
                j = rng.choice(n, p=p_j)
                if j != i:
                    break
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            T = max(final_temp, T * alpha)
            continue
        scale = T / init_temp
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            T = max(final_temp, T * alpha)
            continue

        # apply move
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_score = new_score
            else:
                h[i] += delta
                h[j] -= delta

        T = max(final_temp, T * alpha)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initialise base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    levels = [64, 128, 256, 512]
    fractions = {64: 0.05, 128: 0.10, 256: 0.20, 512: 0.40}

    for n in levels:
        if time_left() <= 0.0:
            break
        if best_h.size != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)

        alloc = budget_s * fractions.get(n, 0.0)
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        cand_h, cand_score = simulated_annealing_improved(base, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        else:
            best_h = cand_h.copy()  # propagate for next level

    # final refinement with whatever time remains
    remaining = time_left()
    if remaining > 0.5:
        # random binary search (half of remaining time)
        n = best_h.size
        rand_time = remaining * 0.5
        rand_start = time.time()
        while time.time() - rand_start < rand_time:
            h_rand = random_balanced_binary(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score = sc
                best_h = h_rand.copy()
        remaining = time_left()
        if remaining > 0.5:
            cand_h, cand_score = simulated_annealing_improved(best_h, remaining, rng)
            if cand_score < best_score:
                best_h = cand_h
                best_score = cand_score

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
