def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta*momentum + (1-beta)*grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta*momentum
        # occasional random perturbation
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    cur_h = h_bin.copy()
    while time_left() > 0.5:

def gradient_descent(h0, n_iter, step0, beta, rng, start, budget_s):
    import time
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= budget_s:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_local_search(h_bin, best_score, rng, start, budget_s):
    import time
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_score = best_score
    while time.time() - start < budget_s - 0.5:
        # pick a 1 and a 0
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        sc = compute_c5_fft(cur_h)
        if sc < cur_score:
            cur_score = sc
            # keep swap
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return cur_h, cur_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n as floats."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient of max‑correlation term (averaged over all maximizers)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, start, budget_s):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= budget_s:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_local_search(h_bin, cur_score, rng, start, budget_s):
    """Local swaps on a binary vector, keeping sum fixed."""
    n = h_bin.size
    h = h_bin.copy()
    while time.time() - start < budget_s - 0.5:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        sc = compute_c5_fft(h)
        if sc < cur_score:
            cur_score = sc
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, cur_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Initialise base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128)
    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = None
    best_score = np.inf
    TARGET = 0.3808
    scales = [128, 256, 512]
    for n in scales:
        if time.time() - start >= budget_s - 1.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        # Choose restart parameters per scale
        if n <= 256:
            restarts, n_iter = 12, 150000
        else:
            restarts, n_iter = 6, 200000
        for r in range(restarts):
            if time.time() - start >= budget_s - 0.5:
                break
            h0 = base + rng.normal(scale=0.07, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)
            h_opt, sc_opt = gradient_descent(
                h0, n_iter=n_iter, step0=0.9, beta=0.95,
                rng=rng, start=start, budget_s=budget_s)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_score <= TARGET:
            break
        # Prepare for next finer scale
        if best_h is not None:
            base = upsample(best_h, n) if n < scales[-1] else best_h
        else:
            base = base
        if best_score <= TARGET:
            break

    # Optional final refinement at higher resolution if time permits
    if best_h is not None and time.time() - start < budget_s - 5.0:
        n_fine = 1024
        if best_h.size < n_fine:
            h_fine = upsample(best_h, n_fine)
        else:
            h_fine = best_h.copy()
        h_fine = project_onto_box_sum(h_fine, n_fine / 2.0)
        h_opt, sc_opt = gradient_descent(
            h_fine, n_iter=150000, step0=0.8, beta=0.95,
            rng=rng, start=start, budget_s=budget_s)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Binary rounding and discrete local search
    if best_h is not None and time.time() - start < budget_s - 2.0:
        n = best_h.size
        k = n // 2
        idx_top = np.argpartition(best_h, -k)[-k:]
        h_bin = np.zeros_like(best_h)
        h_bin[idx_top] = 1.0
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
        # binary swaps
        if time.time() - start < budget_s - 1.0:
            h_bin_opt, sc_bin_opt = binary_local_search(best_h, best_score,
                                                        rng, start, budget_s)
            if sc_bin_opt < best_score:
                best_score, best_h = sc_bin_opt, h_bin_opt.copy()

    # Continuous small‑mass random transfers
    if best_h is not None and time.time() - start < budget_s - 3.0:
        n = best_h.size
        h_local = best_h.copy()
        cur_score = best_score
        while time.time() - start < budget_s - 2.0:
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(h_local[i], 1.0 - h_local[j])
            if max_delta <= 1e-12:
                continue
            delta = max_delta * rng.uniform(0.02, 0.3)
            h_new = h_local.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5_fft(h_new)
            if sc_new < cur_score:
                cur_score = sc_new
                h_local = h_new
        best_h, best_score = h_local, cur_score

    # Final projection and evaluation
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
        n_ret = best_h.size
    else:
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, n_ret
