def compute_c5(h):
    n = h.shape[0]
    # use fft based
    n_corr = 2 * n - 1
    # pad to next power of 2 for speed

def targeted_move(h, shift, rng):
    n = h.shape[0]
    # shift integer
    if shift >= 0:
        i_start = 0
        i_end = n - shift  # exclusive
    else:
        i_start = -shift
        i_end = n
    # compute products
    i_range = np.arange(i_start, i_end)
    prod = h[i_range] * h[i_range + shift]
    # select i with max product
    i_idx = i_range[np.argmax(prod)]
    # find candidate j:
    # Choose j randomly among indices where (h[j] < 0.5) or product low
    # compute product for all j in full range (maybe same as above)
    if shift >= 0:
        j_start = 0
        j_end = n - shift

    # Maybe take j = random choice among those with product less than median
    low_mask = prod_j < np.median(prod_j)
    if not np.any(low_mask):

        j_idx = rng.integers(n)

    return h_new

def anneal(h, max_time, rng):
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_h = h.copy()
    cur_score = best_score
    start = time.time()
    # parameters
    T0 = 0.01
    T_final = 1e-6
    # decide total steps based on max_time: assume each step ~ O(n log n) ~ small; set few hundred thousand steps max.
    # We'll compute steps = int(min(5e5, max_time * 1e5)) maybe.
    total_steps = int(min(500000, max_time * 200000))  # 200k steps per second budget (overestimate)
    if total_steps < 1:
        total_steps = 1
    for step in range(total_steps):
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        # temperature
        T = T0 * (T_final / T0) ** (step / total_steps)
        # compute current worst shift
        corr = compute_corr_fft(cur_h)  # compute full correlation array
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        # decide move type
        if rng.random() < 0.7:
            new_h = targeted_move(cur_h, shift, rng)
            if new_h is None:
                # fallback random move
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(cur_h[i], 1.0 - cur_h[j])
                if max_move <= 0:
                    continue
                delta = max_move * rng.random()
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
        else:
            # random move
            i, j = rng.choice(n, size=2, replace=False)
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0:
                continue
            delta = max_move * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        # ensure constraints: due to delta selection, sum unchanged, box satisfied.
        # compute score
        new_score = compute_c5_fft(new_h)
        # acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:

def optimize_at_scale(base_h, n, time_budget, rng):
    # Upsample base_h to size n if needed
    if base_h.shape[0] != n:
        h0 = upsample(base_h, n)
    else:
        h0 = base_h.copy()
    # ensure constraints
    h0 = project_onto_box_sum(h0, n/2.0)
    best_h = h0
    best_score = compute_c5_fft(h0)
    start = time.time()
    # number of restarts: maybe 5
    restarts = 5
    for r in range(restarts):
        elapsed = time.time() - start
        remain = time_budget - elapsed
        if remain < 0.5:
            break
        # perturb start for each restart
        if r == 0:
            h_start = h0
        else:
            # add small noise then project
            h_start = h0 + rng.normal(scale=0.02, size=n)
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = project_onto_box_sum(h_start, n/2.0)
        # allocate time for this restart: equal share of remaining
        sub_time = remain / (restarts - r)
        h_opt, sc_opt = anneal(h_start, sub_time, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_corr_fft(h):
    n = h.shape[0]
    n_corr = 2 * n - 1
    L = 1 << (n_corr - 1).bit_length()  # next power of two
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(np.flip(1.0 - h), n=L)
    corr = np.fft.irfft(H * G, n=L)[:n_corr]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    return np.max(compute_corr_fft(h))

def targeted_move(h, shift, rng):
    n = h.shape[0]
    if shift >= 0:
        i_start, i_end = 0, n - shift
    else:
        i_start, i_end = -shift, n
    i_range = np.arange(i_start, i_end)
    prod = h[i_range] * h[i_range + shift]
    if prod.size == 0:
        return None
    i_idx = i_range[np.argmax(prod)]
    # choose j with low product
    if shift >= 0:
        j_start, j_end = 0, n - shift
    else:
        j_start, j_end = -shift, n
    j_range = np.arange(j_start, j_end)
    prod_j = h[j_range] * h[j_range + shift]
    median = np.median(prod_j) if prod_j.size else 0.0
    low_mask = prod_j <= median
    if low_mask.any():
        candidate_js = j_range[low_mask]
        j_idx = rng.choice(candidate_js)
    else:
        j_idx = rng.integers(n)
    delta_max = min(h[i_idx], 1.0 - h[j_idx])
    if delta_max <= 0:
        return None
    delta = delta_max * rng.random() * 0.5
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i_idx] -= delta
    h_new[j_idx] += delta
    return h_new

def anneal(h_start, max_time, rng):
    n = h_start.shape[0]
    cur_h = h_start.copy()
    cur_score = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    # annealing schedule parameters
    T0 = 0.01
    T_final = 1e-6
    # rough step budget based on time
    total_steps = int(min(500000, max_time * 200000))
    if total_steps < 1:
        total_steps = 1
    for step in range(total_steps):
        if time.time() - start >= max_time:
            break
        if step % 1000 == 0:
            # re‑project to avoid drift
            cur_h = project_onto_box_sum(cur_h, n / 2.0)
        # temperature
        T = T0 * (T_final / T0) ** (step / total_steps)
        # compute worst shift
        corr = compute_corr_fft(cur_h)
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)
        # propose move
        if rng.random() < 0.7:
            new_h = targeted_move(cur_h, shift, rng)
            if new_h is None:
                # fallback random transfer
                i, j = rng.choice(n, size=2, replace=False)
                delta_max = min(cur_h[i], 1.0 - cur_h[j])
                if delta_max <= 0:
                    continue
                delta = delta_max * rng.random()
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
        else:
            i, j = rng.choice(n, size=2, replace=False)
            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max <= 0:
                continue
            delta = delta_max * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        new_score = compute_c5_fft(new_h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
    return best_h, best_score

def optimize_at_scale(base_h, n, time_budget, rng):
    # upsample / adjust size
    h0 = upsample(base_h, n) if base_h.shape[0] != n else base_h.copy()
    h0 = project_onto_box_sum(h0, n / 2.0)
    best_h = h0
    best_score = compute_c5_fft(h0)
    start = time.time()
    restarts = 5
    for r in range(restarts):
        elapsed = time.time() - start
        remain = time_budget - elapsed
        if remain < 0.5:
            break
        if r == 0:
            h_start = h0
        else:
            h_start = h0 + rng.normal(scale=0.02, size=n)
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = project_onto_box_sum(h_start, n / 2.0)
        sub_time = remain / (restarts - r)
        h_opt, sc_opt = anneal(h_start, sub_time, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # base vector (n=64 if available)
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    scales = [128, 256, 512]
    best_h = None
    best_score = np.inf
    for idx, n in enumerate(scales):
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining <= 1.0:
            break
        # allocate equal share of remaining time among pending scales
        alloc = remaining / (len(scales) - idx)
        h_opt, sc_opt = optimize_at_scale(base, n, alloc, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        base = h_opt  # start next scale from this solution
    # final fallback if nothing found
    if best_h is None:
        n = scales[-1]
        best_h = project_onto_box_sum(random_binary_balanced(n, rng), n / 2.0)
        best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
