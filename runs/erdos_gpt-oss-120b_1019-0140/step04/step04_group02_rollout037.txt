def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):

        h_new = h - eta * momentum
        if it % 1500 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    TARGET = 0.3808

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128)

    best_h = None
    best_score = np.inf
    scales = [256, 512, 1024]

    for n in scales:
        if time_left() < 5.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        for r in range(3):
            if time_left() < 4.0:
                break
            h0 = base + rng.normal(scale=0.04, size=n)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = project_onto_box_sum(h0, n / 2.0)

            h_opt, sc_opt = gradient_descent(
                h0,
                n_iter=120000,
                step0=0.7,
                beta=0.9,
                rng=rng,
                time_left=time_left,
                top_k=8,
            )
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_h is not None:
            base = best_h.copy()
        if best_score <= TARGET:
            break

    # Final local search
    if best_h is not None and time_left() > 2.0:
        h = best_h.copy()
        n = h.size
        attempts = 0
        while time_left() > 2.0 and attempts < 50000:
            attempts += 1
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            if h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                continue
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 1e-12:
                continue
            delta = delta_max * rng.uniform(0.01, 0.2)
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5_fft(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                h = h_new
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)

    if best_h is None:
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5_fft(best_h)
    else:
        n_ret = best_h.size

    return best_h, best_score, n_ret
