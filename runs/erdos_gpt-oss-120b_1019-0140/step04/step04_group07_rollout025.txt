def simulated_annealing(h, rng, time_left, total_time):
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    n = cur_h.size
    start = time.time()
    while time_left() > 0:
        elapsed = time.time() - start
        t_frac = elapsed / total_time
        T = init_temp * (1 - t_frac) + final_temp * t_frac
        # pick i, j
        i = rng.integers(n)
        # ensure i has positive mass
        if cur_h[i] <= 0.0:
            continue
        j = rng.integers(n)
        if cur_h[j] >= 1.0:
            continue
        # compute delta
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        delta = max_delta * rng.random() * 0.5
        if delta <= 0:
            continue
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # keep sum constraint holds automatically
        # compute new score
        new_score = compute_c5(h_new)
        diff = new_score - cur_score
        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            cur_h, cur_score = h_new, new_score
    return cur_h, cur_score
```

def simulated_annealing(h, rng, deadline, init_temp=0.01, final_temp=1e-6):
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    n = cur_h.size
    total = max(deadline - time.time(), 1e-12)
    start = time.time()
    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start
        frac = elapsed / total
        T = init_temp * (1 - frac) + final_temp * frac
        # sample i, j
        i, j = rng.choice(n, size=2, replace=False)
        if cur_h[i] <= 0.0 or cur_h[j] >= 1.0:
            continue
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        diff = new_score - cur_score
        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            cur_h, cur_score = new_h, new_score
    return cur_h, cur_score

import time
import numpy as np


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score
    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def simulated_annealing(h, rng, deadline, init_temp=0.01, final_temp=1e-6):
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    n = cur_h.size
    total = max(deadline - time.time(), 1e-12)
    start = time.time()
    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start
        frac = elapsed / total
        T = init_temp * (1.0 - frac) + final_temp * frac
        i, j = rng.choice(n, size=2, replace=False)
        if cur_h[i] <= 0.0 or cur_h[j] >= 1.0:
            continue
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        diff = new_score - cur_score
        if diff <= 0.0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            cur_h, cur_score = new_h, new_score
    return cur_h, cur_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]
    for n in scales:
        if time_left() < 5.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        if time_left() > 5.0:
            max_iters = int(min(20000, max(3000, time_left() * (15000.0 / n))))
            h_gd, sc_gd = gradient_descent_adaptive(
                base, max_iters, step0=0.6, beta=0.9, rng=rng, time_left=time_left
            )
            if sc_gd < best_score:
                best_h, best_score = h_gd.copy(), sc_gd
            base = h_gd

        if time_left() > 5.0:
            dl = time.time() + min(5.0, time_left() * 0.1)
            h_ms, sc_ms = improve_via_max_shift(
                base, rng, lambda: max(0.0, dl - time.time())
            )
            if sc_ms < best_score:
                best_h, best_score = h_ms.copy(), sc_ms
            base = h_ms

        if time_left() > 5.0:
            dl = time.time() + min(5.0, time_left() * 0.1)
            h_pair, sc_pair = local_pair_moves(
                base, rng, lambda: max(0.0, dl - time.time())
            )
            if sc_pair < best_score:
                best_h, best_score = h_pair.copy(), sc_pair
            base = h_pair

        if time_left() > 5.0:
            for _ in range(2):
                if time_left() < 2.0:
                    break
                h_rand = random_binary_balanced(n, rng)
                max_iters_rand = int(min(5000, max(500, time_left() * (3000.0 / n))))
                h_rand_ref, sc_rand = gradient_descent_adaptive(
                    h_rand, max_iters_rand, step0=0.5, beta=0.9,
                    rng=rng, time_left=time_left
                )
                if sc_rand < best_score:
                    best_h, best_score = h_rand_ref.copy(), sc_rand
                if time_left() > 5.0:
                    dl = time.time() + min(3.0, time_left() * 0.1)
                    h_rand_pair, sc_rand_pair = local_pair_moves(
                        h_rand_ref, rng, lambda: max(0.0, dl - time.time())
                    )
                    if sc_rand_pair < best_score:
                        best_h, best_score = h_rand_pair.copy(), sc_rand_pair

    if time_left() > 10.0:
        max_iters_final = int(min(30000, max(5000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(
            best_h, max_iters_final, step0=0.3, beta=0.9, rng=rng, time_left=time_left
        )

    if time_left() > 5.0:
        dl = time.time() + time_left()
        best_h, best_score = simulated_annealing(best_h, rng, dl)

    if time_left() > 3.0:
        dl = time.time() + time_left()
        best_h, best_score = local_pair_moves(best_h, rng, lambda: max(0.0, dl - time.time()))

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
