def binary_swap_anneal(h0, rng, time_left, max_iters=400000, init_temp=0.2, decay=0.99998, min_temp=1e-5):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = init_temp
    it = 0
    n = h.size
    while it < max_iters and time_left() > 0.0:
        # pick random indices to swap
        # choose ones and zeros
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
        T = max(min_temp, T * decay)
        it += 1
    return best_h, best_s
```

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        term1[:n_s] = h[shift:]
        term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        term1[s:] = h[:n_s]
        term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    shifts = np.where(corr == mx)[0] - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    if shifts.size:
        grad /= shifts.size
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_swap_anneal(h0, rng, time_left, max_iters=2000000,
                      init_temp=0.4, decay=0.99995, min_temp=1e-6,
                      target=None):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = init_temp
    n = h.size
    it = 0
    while it < max_iters and time_left() > 0.0:
        # pick two indices with different values
        i = rng.integers(n)
        j = rng.integers(n)
        while h[i] == h[j]:
            i = rng.integers(n)
            j = rng.integers(n)
        # swap
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
                if target is not None and best_s <= target:
                    return best_h, best_s
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
        T = max(min_temp, T * decay)
        it += 1
    return best_h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    n = 512
    target_sum = n / 2.0

    # Build initial candidates
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        candidates.append(project_onto_box_sum(init, target_sum))
    candidates.append(thue_morse_binary(n))
    for _ in range(4):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(4):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))

    best_h = None
    best_s = np.inf

    # Quick continuous refinement on each candidate
    for c in candidates:
        if time_left() < 2.0:
            break
        h_opt, s_opt = gradient_descent_momentum(c, time_left,
                                                 step0=1.0, beta=0.9, max_iter=8000)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= target:
            best_h = project_onto_box_sum(best_h, target_sum)
            return best_h, compute_c5_fast(best_h), n

    # Ensure we have a binary start for annealing
    if best_h is None:
        best_h = random_binary_balanced(n, rng)
    else:
        idx = np.argpartition(-best_h, int(target_sum))[:int(target_sum)]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        best_h = h_bin

    # Simulated annealing on binary patterns
    if time_left() > 1.0:
        h_ann, s_ann = binary_swap_anneal(best_h, rng, time_left,
                                         max_iters=2000000,
                                         init_temp=0.5, decay=0.99996,
                                         min_temp=1e-7, target=target)
        if s_ann < best_s:
            best_s, best_h = s_ann, h_ann.copy()
        if best_s <= target:
            best_h = project_onto_box_sum(best_h, target_sum)
            return best_h, compute_c5_fast(best_h), n

    # Final continuous local search
    if time_left() > 5.0:
        h_ref, s_ref = gradient_descent_momentum(best_h, time_left,
                                                 step0=0.6, beta=0.9,
                                                 max_iter=30000)
        if s_ref < best_s:
            best_s, best_h = s_ref, h_ref.copy()

    best_h = project_onto_box_sum(best_h, target_sum)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s, n
