def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return np.max(corr) * (2.0 / n)

def ga_optimize(n, rng, time_left, init_h=None):
    target = n // 2
    popsize = 50 if n <= 128 else 80
    tournament_sz = 5
    pc = 0.9
    pm = 0.02

    fitness_cache = {}
    def fitness(v):
        key = v.tobytes()
        if key in fitness_cache:
            return fitness_cache[key]
        sc = compute_c5(v)
        fitness_cache[key] = sc
        return sc

    # population init
    pop = []
    if init_h is not None:
        # convert init_h (maybe float) to binary balanced
        v_bin = np.zeros(n, dtype=np.uint8)
        # select top target indices
        idx = np.argsort(-init_h)[:target]
        v_bin[idx] = 1
        pop.append(v_bin)
    while len(pop) < popsize:
        v = np.zeros(n, dtype=np.uint8)
        v[rng.choice(n, size=target, replace=False)] = 1
        pop.append(v)
    # Evaluate
    scores = np.array([fitness(v) for v in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    # main loop
    while time_left() > 0.5:
        new_pop = []
        # elitism: keep best
        new_pop.append(best_h.copy())
        # generate rest
        while len(new_pop) < popsize:
            # tournament selection
            participants = rng.choice(popsize, size=tournament_sz, replace=False)
            p1_idx = participants[np.argmin(scores[participants])]
            participants = rng.choice(popsize, size=tournament_sz, replace=False)
            p2_idx = participants[np.argmin(scores[participants])]
            p1, p2 = pop[p1_idx], pop[p2_idx]
            # crossover
            if rng.random() < pc:
                mask = rng.integers(0, 2, size=n, dtype=bool)
                child = np.where(mask, p1, p2).copy()
            else:
                child = p1.copy()
                # maybe mutate later
            # mutation
            mut_mask = rng.random(n) < pm
            # flip bits at mut_mask
            child[mut_mask] = 1 - child[mut_mask]
            # repair balance
            ones = child.sum()
            if ones > target:
                # too many ones: flip some ones to zero
                excess = int(ones - target)
                ones_idx = np.where(child == 1)[0]
                to_flip = rng.choice(ones_idx, size=excess, replace=False)
                child[to_flip] = 0
            elif ones < target:
                deficit = int(target - ones)
                zeros_idx = np.where(child == 0)[0]
                to_flip = rng.choice(zeros_idx, size=deficit, replace=False)
                child[to_flip] = 1
            new_pop.append(child)
        # evaluate new pop
        new_scores = np.array([fitness(v) for v in new_pop])
        # update best
        cur_best_idx = np.argmin(new_scores)
        if new_scores[cur_best_idx] < best_score:
            best_score = new_scores[cur_best_idx]
            best_h = new_pop[cur_best_idx].copy()
        # replace population
        pop = new_pop
        scores = new_scores

        # occasional local optimization on best
        if rng.random() < 0.2:
            # try a few random swaps
            temp_h = best_h.copy()
            temp_score = best_score
            for _ in range(20):
                # pick 1 and 0
                idx1 = rng.choice(np.where(temp_h == 1)[0])
                idx0 = rng.choice(np.where(temp_h == 0)[0])
                # swap
                temp_h[idx1] = 0
                temp_h[idx0] = 1
                sc = fitness(temp_h)
                if sc < temp_score:
                    temp_score = sc
                    # keep swap
                else:
                    # revert
                    temp_h[idx1] = 1
                    temp_h[idx0] = 0
            if temp_score < best_score:
                best_score = temp_score
                best_h = temp_h.copy()
        # check time and loop
    return best_h, best_score

def anneal(h, score, rng, time_left):
    best_h = h.copy()
    best_score = score
    cur_h = h.copy()
    cur_score = score
    n = h.size
    start = time.time()
    # parameters
    T0 = 0.01
    T1 = 0.0001
    while time_left() > 0:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed)   # approximate (same as previous)
        T = T0 * (T1/T0) ** frac
        # propose swap
        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_score

def ga_optimize(n, rng, time_left, init_h=None):
    target = n // 2
    popsize = 60 if n <= 128 else 80
    tournament_sz = 5
    pc = 0.9
    pm = 0.02

    # initial population
    pop = []
    if init_h is not None:
        pop.append(init_h.copy())
    # ensure binary shape correct
    while len(pop) < popsize:
        v = np.zeros(n, dtype=np.uint8)
        v[rng.choice(n, size=target, replace=False)] = 1
        pop.append(v)
    # evaluate
    scores = np.empty(len(pop))
    for i, v in enumerate(pop):
        scores[i] = compute_c5(v)
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time_left() > 0.5:
        new_pop = [best_h.copy()]  # elitism
        while len(new_pop) < popsize:
            # tournament select parent1
            participants = rng.choice(popsize, size=tournament_sz, replace=False)
            p1_idx = participants[np.argmin(scores[participants])]
            # tournament select parent2
            participants = rng.choice(popsize, size=tournament_sz, replace=False)
            p2_idx = participants[np.argmin(scores[participants])]
            p1, p2 = pop[p1_idx], pop[p2_idx]

            # crossover
            if rng.random() < pc:
                mask = rng.integers(0, 2, size=n, dtype=bool)
                child = np.where(mask, p1, p2).copy()
            else:
                child = p1.copy()

            # mutation: flip bits with probability pm
            mut_mask = rng.random(n) < pm
            child[mut_mask] = 1 - child[mut_mask]

            # repair to maintain exactly target ones
            cur = int(child.sum())
            if cur > target:
                excess = cur - target
                ones_idx = np.where(child == 1)[0]
                to_flip = rng.choice(ones_idx, size=excess, replace=False)
                child[to_flip] = 0
            elif cur < target:
                deficit = target - cur
                zeros_idx = np.where(child == 0)[0]
                to_flip = rng.choice(zeros_idx, size=deficit, replace=False)
                child[to_flip] = 1

            new_pop.append(child)

        # evaluate new population
        new_scores = np.empty(popsize)
        for i, v in enumerate(new_pop):
            new_scores[i] = compute_c5(v)

        # update best
        cur_best_idx = np.argmin(new_scores)
        if new_scores[cur_best_idx] < best_score:
            best_score = new_scores[cur_best_idx]
            best_h = new_pop[cur_best_idx].copy()

        # replace
        pop = new_pop
        scores = new_scores

        # occasional hill climbing on best
        if rng.random() < 0.15:
            # try 10 random swaps
            h_temp = best_h.copy()
            h_temp_score = best_score
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if h_temp[i] == h_temp[j]:
                    continue
                # swap
                h_temp[i], h_temp[j] = h_temp[j], h_temp[i]
                sc = compute_c5(h_temp)
                if sc < h_temp_score:
                    h_temp_score = sc
                else:
                    # revert
                    h_temp[i], h_temp[j] = h_temp[j], h_temp[i]
            if h_temp_score < best_score:
                best_score = h_temp_score
                best_h = h_temp.copy()

        # check time
    return best_h, best_score
```

def anneal(h, score, rng, time_left):
    best_h = h.copy()
    best_score = score
    cur_h = h.copy()
    cur_score = score
    n = h.size
    start = time.time()
    T0 = 0.01
    T1 = 1e-5
    while time_left() > 0.1:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed)
        T = T0 * (T1 / T0) ** frac
        # propose swap
        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_score
```

def gradient_descent(h0, iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, iters+1):
        if time_left() <= 0:
            break
        shifts = max_shifts_indices(h)
        grad = compute_gradient(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        # option: add small noise occasionally
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project to [0,1] and sum constraint
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

import time, math, numpy as np

def compute_c5(h):
    """Compute C5 = max correlation * (2/n) for binary or continuous h."""
    hf = h.astype(float)
    n = hf.shape[0]
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from [0,1) domain."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_balanced(arr, target):
    """Return binary vector with exactly target ones, picking largest entries."""
    idx = np.argpartition(-arr, target - 1)[:target]
    v = np.zeros_like(arr, dtype=np.uint8)
    v[idx] = 1
    return v

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Projection onto [lo,hi] with sum == s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def max_shifts_indices(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idxs = np.where(corr == maxc)[0]
    return (idxs - (n - 1)).tolist()

def compute_gradient(h, shifts):
    n = h.size
    grad = np.zeros_like(h, dtype=float)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=float)
            term2 = np.zeros(n, dtype=float)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=float)
            term2 = np.zeros(n, dtype=float)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, iters, step0, beta, rng, time_left):
    h = h0.astype(float).copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, iters + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts_indices(h)
        grad = compute_gradient(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def ga_optimize(n, rng, time_left, init_h=None):
    target = n // 2
    popsize = 80 if n > 128 else 60
    tournament_sz = 5
    pc = 0.9
    pm = 0.02

    pop = []
    if init_h is not None:
        pop.append(init_h.copy())
    while len(pop) < popsize:
        v = np.zeros(n, dtype=np.uint8)
        v[rng.choice(n, size=target, replace=False)] = 1
        pop.append(v)

    scores = np.empty(len(pop))
    for i, v in enumerate(pop):
        scores[i] = compute_c5(v)
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time_left() > 0.5:
        new_pop = [best_h.copy()]  # elitism
        while len(new_pop) < popsize:
            # parent 1
            participants = rng.choice(popsize, size=tournament_sz, replace=False)
            p1_idx = participants[int(np.argmin(scores[participants]))]
            # parent 2
            participants = rng.choice(popsize, size=tournament_sz, replace=False)
            p2_idx = participants[int(np.argmin(scores[participants]))]
            p1, p2 = pop[p1_idx], pop[p2_idx]

            # crossover
            if rng.random() < pc:
                mask = rng.random(n) < 0.5
                child = np.where(mask, p1, p2).copy()
            else:
                child = p1.copy()

            # mutation (flip bits)
            mut_mask = rng.random(n) < pm
            child[mut_mask] = 1 - child[mut_mask]

            # repair to exact target ones
            cur = int(child.sum())
            if cur > target:
                excess = cur - target
                ones_idx = np.where(child == 1)[0]
                flip = rng.choice(ones_idx, size=excess, replace=False)
                child[flip] = 0
            elif cur < target:
                deficit = target - cur
                zeros_idx = np.where(child == 0)[0]
                flip = rng.choice(zeros_idx, size=deficit, replace=False)
                child[flip] = 1

            new_pop.append(child)

        new_scores = np.empty(popsize)
        for i, v in enumerate(new_pop):
            new_scores[i] = compute_c5(v)

        cur_best_idx = int(np.argmin(new_scores))
        if new_scores[cur_best_idx] < best_score:
            best_score = new_scores[cur_best_idx]
            best_h = new_pop[cur_best_idx].copy()

        pop = new_pop
        scores = new_scores

        # occasional local random swaps on best
        if rng.random() < 0.15:
            h_tmp = best_h.copy()
            sc_tmp = best_score
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if h_tmp[i] == h_tmp[j]:
                    continue
                h_tmp[i], h_tmp[j] = h_tmp[j], h_tmp[i]
                sc = compute_c5(h_tmp)
                if sc < sc_tmp:
                    sc_tmp = sc
                else:
                    h_tmp[i], h_tmp[j] = h_tmp[j], h_tmp[i]
            if sc_tmp < best_score:
                best_score = sc_tmp
                best_h = h_tmp.copy()
    return best_h, best_score

def anneal(h, score, rng, time_left):
    best_h = h.copy()
    best_score = score
    cur_h = h.copy()
    cur_score = score
    n = h.size
    start = time.time()
    T0, T1 = 0.01, 1e-5
    while time_left() > 0.1:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed)
        T = T0 * (T1 / T0) ** frac
        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # --- initial data ---
    init_cont = None
    if 'initial_h_values' in globals():
        init_cont = np.asarray(initial_h_values, dtype=float)

    # --- small scale GA (n=64) ---
    best_small_h = None
    if time_left() > 30.0:
        n_small = 64
        if init_cont is not None and init_cont.shape[0] != n_small:
            init_small = upsample(init_cont, n_small)
        else:
            init_small = init_cont
        init_small_bin = binary_balanced(init_small, n_small // 2) if init_small is not None else None
        small_end = time.time() + min(60.0, time_left() * 0.15)
        def small_time_left():
            return max(0.0, small_end - time.time())
        best_small_h, _ = ga_optimize(n_small, rng, small_time_left, init_small_bin)

    # --- large scale GA (n=256) ---
    n_large = 256
    if best_small_h is not None:
        up_big = upsample(best_small_h.astype(float), n_large)
    elif init_cont is not None:
        up_big = upsample(init_cont, n_large)
    else:
        up_big = None
    init_large_bin = binary_balanced(up_big, n_large // 2) if up_big is not None else None
    best_h, best_score = ga_optimize(n_large, rng, time_left, init_large_bin)

    # --- annealing refinement ---
    if time_left() > 5.0:
        best_h, best_score = anneal(best_h, best_score, rng, time_left)

    # --- gradient descent refinement ---
    if time_left() > 20.0:
        max_iters = int(min(20000, max(5000, time_left() * 1000)))
        best_h, best_score = gradient_descent(best_h.astype(float), max_iters,
                                              step0=0.5, beta=0.9,
                                              rng=rng, time_left=time_left)

    best_score = compute_c5(best_h)
    return best_h.astype(float), best_score, best_h.shape[0]
