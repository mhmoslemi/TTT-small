def simulated_annealing_binary(h_bin, target, total_time, rng, start_time, budget_s):
    n = h_bin.size
    current = h_bin.copy()
    current_score = compute_c5_fft(current)
    best = current.copy()
    best_score = current_score

    # Determine start_T and end_T based on current_score - target
    start_T = max((current_score - target) * 2.0, 0.001)
    end_T = 1e-7

    while True:
        elapsed = time.time() - start_time
        if elapsed >= total_time:
            break
        # compute temperature
        ratio = elapsed / total_time
        T = start_T * (end_T / start_T) ** ratio

        # propose swap
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if current[i] == current[j]:
            continue
        # swap
        new = current.copy()
        new[i], new[j] = new[j], new[i]
        new_score = compute_c5_fft(new)
        delta = new_score - current_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            # accept
            current = new
            current_score = new_score
            if current_score < best_score:
                best = current.copy()
                best_score = current_score
    return best, best_score

def compute_corr_full_fft(h):
    n = h.shape[0]; L = 2*n - 1
    fh = np.fft.fft(h, L); f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise

            h_new += rng.normal(scale=0.02, size=n)

        h = h_new
    return best_h, best_score

import time
import math
import numpy as np
import scipy.optimize as opt

_run_budget = None

def _time_left():
    if _run_start_time is None:
        return 0.0
    return max(0.0, _run_budget - (time.time() - _run_start_time))

_slsqp_n_dim = None
def _slsqp_obj(x):
    return x[-1]

def _slsqp_eq_sum(x):
    return np.sum(x[:_slsqp_n_dim]) - _slsqp_n_dim / 2.0

def _slsqp_ineq_fun(x):
    h = x[:_slsqp_n_dim]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / _slsqp_n_dim)
    return t - corr

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_binary(h_bin, target, rng):
    n = h_bin.size
    current = h_bin.copy()
    current_score = compute_c5_fft(current)
    best = current.copy()
    best_score = current_score
    # allocate a chunk of remaining time (up to 30% of budget)
    alloc = min(120.0, max(5.0, _time_left() * 0.3))
    start = time.time()
    start_T = max((current_score - target) * 2.0, 0.001)
    end_T = 1e-7
    while time.time() - start < alloc:
        elapsed = time.time() - start
        ratio = elapsed / alloc
        T = start_T * (end_T / start_T) ** ratio
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if current[i] == current[j]:
            continue
        new = current.copy()
        new[i], new[j] = new[j], new[i]
        new_score = compute_c5_fft(new)
        delta = new_score - current_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            current = new
            current_score = new_score
            if current_score < best_score:
                best = current.copy()
                best_score = current_score
    return best, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):

        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()

    s = n / 2.0
    best_h = h.copy()

    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
