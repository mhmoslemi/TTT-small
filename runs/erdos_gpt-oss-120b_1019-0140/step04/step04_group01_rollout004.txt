def gradient_descent_momentum(h0, time_left, step0=0.8, beta=0.95, max_iter=200000):
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        # compute gradient
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def local_improve_binary(h, time_left, max_iter=1000000):
    n = h.shape[0]
    best_score = compute_c5_fft(h)
    best_h = h.copy()
    it = 0
    T0 = 0.05
    Tmin = 1e-6
    decay = 0.9997
    T = T0
    while time_left() > 0.0 and it < max_iter:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        if max_corr < best_score - 1e-12:
            best_score = max_corr
            best_h = h.copy()
        # find shift(s) with max
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shift = idxes[0] - (n - 1)
        # attempt beneficial swap
        if shift >= 0:
            i_cand = np.where((h == 1) & (np.arange(n) + shift < n) & (h[np.arange(n) + shift] == 0))[0]
            j_cand = np.where((h == 0) & (np.arange(n) + shift < n) & (h[np.arange(n) + shift] == 1))[0]
        else:
            s = -shift
            i_cand = np.where((h == 1) & (np.arange(n) - s >= 0) & (h[np.arange(n) - s] == 0))[0]
            j_cand = np.where((h == 0) & (np.arange(n) - s >= 0) & (h[np.arange(n) - s] == 1))[0]
        improved = False
        if i_cand.size > 0 and j_cand.size > 0:
            i = i_cand[0]
            j = j_cand[0]
            # swap
            h[i] = 0
            h[j] = 1
            new_score = compute_c5_fft(h)
            if new_score <= best_score + 1e-12:
                best_score = new_score
                best_h = h.copy()
                improved = True
            else:
                # revert
                h[i] = 1
                h[j] = 0
        if not improved:
            # random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # tentative swap
            h[i] = 0
            h[j] = 1
            new_score = compute_c5_fft(h)
            delta = new_score - best_score
            accept = False
            if delta < 0:
                accept = True
            else:
                if T > Tmin and rng.random() < np.exp(-delta / T):
                    accept = True
            if accept:
                best_score = new_score
                best_h = h.copy()
            else:
                # revert
                h[i] = 1
                h[j] = 0
            T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5, using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max correlation w.r.t. h for a set of shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left, step0=0.8, beta=0.95, max_iter=200000):
    """Projected gradient descent with momentum, early‑stop by time."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def random_binary_balanced(n, rng):
    """Generate binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def local_improve_binary(h, rng, time_left, max_iter=500000):
    """Greedy + simulated‑annealing swaps on a binary vector."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    it = 0
    T0 = 0.05
    Tmin = 1e-6
    decay = 0.9997
    T = T0
    while time_left() > 0.0 and it < max_iter:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        if max_corr < best_score - 1e-12:
            best_score = max_corr
            best_h = h.copy()
        # shift with maximal correlation
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shift = int(idxes[0] - (n - 1))
        # attempt a deterministic improving swap for this shift
        made_improvement = False
        ar = np.arange(n)
        if shift >= 0:
            i_cand = np.where((h == 1) & (ar + shift < n) & (h[ar + shift] == 0))[0]
            j_cand = np.where((h == 0) & (ar + shift < n) & (h[ar + shift] == 1))[0]
        else:
            s = -shift
            i_cand = np.where((h == 1) & (ar - s >= 0) & (h[ar - s] == 0))[0]
            j_cand = np.where((h == 0) & (ar - s >= 0) & (h[ar - s] == 1))[0]

            h[i] = 0.0

            if new_score <= best_score + 1e-12:
                best_score = new_score
                best_h = h.copy()
                made_improvement = True
            else:
                h[i] = 1.0

        if not made_improvement:

            else:
                h[i] = 1.0
                h[j] = 0.0
            T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
