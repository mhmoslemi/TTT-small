def simulated_annealing(h0, time_limit, rng, init_temp=0.02, final_temp=0.0, target_move_prob=0.7):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # precompute some shapes?
    while time.time() - start < time_limit:
        # compute max shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        # find shift index of max, convert to shift K (0-index)
        max_idx = np.argmax(corr)
        K = max_idx - (n - 1)  # K can be negative, but we treat modulo n?
        # We'll treat indices modulo n for periodic condition?
        # But correlation for shift K (which can be negative) corresponds to shift K.
        # In moving mass, we want j = (i + K) % n (makes sense for positive or negative K)
        # So compute j accordingly.

        # Choose move type
        move_type = rng.random()
        if move_type < target_move_prob:
            # target move: move mass from i to i+K
            i = rng.integers(n)
            j = (i + K) % n
        else:
            # random move
            i = rng.integers(n)
            # pick j != i
            j = rng.integers(n-1)
            if j >= i:
                j += 1
        # compute max delta
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        # temperature schedule: linear decay based on elapsed proportion of time_limit
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        temp = init_temp * (1.0 - t_frac) + final_temp * t_frac
        # step size scale: maybe linear with temp; if temp > 0 use larger steps
        scale = temp / init_temp if init_temp > 0 else 0.0
        # propose delta: maybe max_delta * rng.random() * (0.5 + scale*0.5) or some factor
        # ensure not too small.
        delta = max_delta * rng.random() * (0.5 + scale * 0.5)
        if delta <= 0:
            continue
        # apply move
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # accept with prob
            if temp > 0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_score = new_score
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta
            else:
                # revert immediately
                h[i] += delta
                h[j] -= delta
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # Starting vector
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
    else:
        h = thue_morse(64)
    h = project_onto_box_sum(h, h.size/2.0)
    
    # define resolution list and time fractions
    res_list = [64, 128, 256, 512]
    frac = {64:0.12, 128:0.18, 256:0.25, 512:0.45}
    
    best_h = h.copy()
    best_score = compute_c5(best_h)

    for n in res_list:
        # upsample to n
        if best_h.size != n:
            best_h = upsample(best_h, n)
        # ensure bounds
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_onto_box_sum(best_h, n/2.0)
        # compute current score
        cur_score = compute_c5(best_h)
        if cur_score < best_score:
            best_score = cur_score
            best_h = best_h.copy()
        # allocate time
        alloc = frac[n] * budget_s
        if alloc > time_left(): alloc = time_left()
        if alloc <= 0: break
        # run SA on this resolution
        best_h, best_score = simulated_annealing(best_h, alloc, rng)
        # maybe run gradient refinement after SA
        if time_left() > 0.5:
            best_h, best_score = gradient_refine(best_h, time_left()*0.5, rng)  # spend half of remaining time
    # final refinement if remaining time
    if time_left() > 1.0:
        # final SA
        best_h, best_score = simulated_annealing(best_h, time_left(), rng)
    # final projection
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

def gradient_refine(h0, time_limit, rng, step0=0.1):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # compute correlation and K
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        K = max_idx - (n - 1)
        # compute gradient
        idx = np.arange(n)
        grad = 1.0 - h[(idx + K) % n] - h[(idx - K) % n]
        # step size decreasing
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        eta = step0 * (1 - t_frac)  # linear decay
        # update

def simulated_annealing(h0, time_limit, rng):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.025
    p_swap = 0.1
    p_target = 0.7
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # compute correlation and max shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        K = max_idx - (n - 1)  # shift may be negative
        # temperature linear decay
        temp = init_temp * (1.0 - elapsed / time_limit)
        # choose move
        r = rng.random()
        if r < p_swap:
            # swap
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            # swap values
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                # accept with prob
                if temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                else:
                    # revert swap
                    h[i], h[j] = h[j], h[i]
        else:
            # transfer move
            i = rng.integers(n)
            if r < p_swap + p_target:
                j = (i + K) % n
            else:
                j = rng.integers(n-1)
                if j >= i:
                    j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            # scale delta: proportion of max_delta
            # make delta larger when temp high
            scale = 0.5 + 0.5 * (temp / init_temp)  # between 0.5 and 1.0
            delta = max_delta * rng.random() * scale
            if delta <= 0.0:
                continue
            # apply
            h[i] -= delta
            h[j] += delta
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                if temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta
    return best_h, best_score
```

def gradient_refine(h0, time_limit, rng, init_step=0.1):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # compute max shift K
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        K = max_idx - (n - 1)
        # compute gradient for shift K
        idx = np.arange(n)
        grad = 1.0 - h[(idx + K) % n] - h[(idx - K) % n]
        # step size linearly decreasing
        step = init_step * (1.0 - elapsed / time_limit)
        if step <= 0:
            break
        h_new = h - step * grad
        # clip and project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, n/2.0)
        new_score = compute_c5(h_new)
        # accept if better, else maybe with small prob
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # small probability to accept uphill
            if rng.random() < 0.001:
                h = h_new
                cur_score = new_score
        # optionally break if best_score is very low (target reached)
        if best_score <= 0.3808:
            break
    return best_h, best_score

        # clip and project
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        # allocate time
        alloc = frac[n] * budget_s

        cand_h, cand_score = simulated_annealing(best_h, alloc, rng)

            best_h = cand_h.copy()

    remaining = time_left()
    if remaining > 1.0:

    best_score = compute_c5(best_h)

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

        else:
            lam_high = lam_mid

    p_swap = 0.1
    p_target = 0.7
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # correlation and shift giving maximum overlap
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        K = max_idx - (n - 1)  # can be negative
        temp = init_temp * (1.0 - elapsed / time_limit)
        r = rng.random()
        if r < p_swap:
            # swap two entries
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                if temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                else:
                    h[i], h[j] = h[j], h[i]  # revert
        else:
            # transfer move
            i = rng.integers(n)
            # target move respecting overlap window
            if r < p_swap + p_target:
                if K >= 0:
                    if n - K <= 0:
                        continue
                    i = rng.integers(n - K)
                    j = i + K
                else:
                    Kabs = -K
                    if n - Kabs <= 0:
                        continue
                    i = rng.integers(Kabs, n)
                    j = i + K  # K negative, j < i
            else:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            scale = 0.5 + 0.5 * (temp / init_temp)  # between 0.5 and 1.0
            delta = max_delta * rng.random() * scale
            if delta <= 0.0:
                continue
            h[i] -= delta
            h[j] += delta
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                if temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                else:
                    h[i] += delta
                    h[j] -= delta
    return best_h, best_score

def gradient_refine(h0, time_limit, rng, init_step=0.12):
    """Deterministic gradient step on the active shift, with projection."""
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        K = max_idx - (n - 1)
        # compute gradient for linear correlation shift K
        grad = np.zeros(n, dtype=float)
        if K >= 0:
            if n - K > 0:
                grad[:n - K] = 1.0 - h[K:]
                grad[K:] -= h[:n - K]
        else:
            Kabs = -K
            if n - Kabs > 0:
                grad[Kabs:] = 1.0 - h[:n - Kabs]
                grad[:n - Kabs] -= h[Kabs:]
        step = init_step * (1.0 - elapsed / time_limit)
        if step <= 0:
            break
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, n / 2.0)
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # tiny probability to accept uphill move
            if rng.random() < 0.001:
                h = h_new
                cur_score = new_score
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial construction
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
    else:
        h = thue_morse(64)
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, h.size / 2.0)

    # resolution schedule
    res_list = [64, 128, 256, 512]
    frac = {64: 0.12, 128: 0.18, 256: 0.25, 512: 0.45}
    best_h = h.copy()
    best_score = compute_c5(best_h)

    for n in res_list:
        if time_left() <= 0.0:
            break
        # upsample if needed
        if best_h.size != n:
            best_h = upsample(best_h, n)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        alloc = frac[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        cand_h, cand_score = simulated_annealing(best_h, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        else:
            best_h = cand_h

        # brief gradient refinement if time permits
        if time_left() > 0.5:
            grad_time = min(0.2 * time_left(), 2.0)
            cand_h2, cand_score2 = gradient_refine(best_h, grad_time, rng)
            if cand_score2 < best_score:
                best_score = cand_score2
                best_h = cand_h2.copy()

    # final refinement with any remaining budget
    remaining = time_left()
    if remaining > 1.0:
        best_h, best_score = simulated_annealing(best_h, remaining, rng)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
