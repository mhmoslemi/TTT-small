import time

import numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

                continue
            i = rng.choice(i_cand)
            j = i + s
        else:
            m = -s
            mask = (best_h[m:] == 1) & (best_h[:n - m] == 0)
            i_cand = np.where(mask)[0] + m
            if i_cand.size == 0:
                continue
            i = rng.choice(i_cand)
            j = i - m
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_h, best_score = h_new, sc_new
        else:
            temp = max_temp * time_left / total_budget
            if temp > 1e-12 and rng.random() < math.exp(-(sc_new - best_score) / temp):
                best_h, best_score = h_new, sc_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # resolution (prefer 256, fallback to 128 if time is very low)
    n = 256
    if total_budget < 300:
        n = 128
    if n % 2:
        n += 1
    half = n // 2

    # obtain a base continuous vector
    if 'initial_h_values' in globals():
        base_cont = np.asarray(initial_h_values, dtype=float)
        if base_cont.shape[0] != n:
            base_cont = upsample(base_cont, n)
    else:
        base_cont = rng.random(n)

    # turn into a balanced binary vector
    idx_top = np.argpartition(base_cont, -half)[-half:]
    base_bin = np.zeros(n, dtype=float)
    base_bin[idx_top] = 1.0

    best_h = base_bin
    best_score = compute_c5(best_h)
    target = 0.3808

    # random binary restarts while we have spare time
    while True:
        elapsed = time.time() - start_time
        time_left = total_budget - elapsed
        if time_left < 5.0 or best_score <= target:
            break
        h_rand = random_binary_balanced(n, rng)
        sc = compute_c5(h_rand)
        if sc < best_score:
            best_h, best_score = h_rand, sc

    # local improvement via hill climbing (simulated annealing)
    best_h, best_score = hill_climb_binary(best_h, start_time, total_budget, rng)

    # final safety projection (should already be binary)
    best_h = project_onto_box_sum(best_h, half)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
