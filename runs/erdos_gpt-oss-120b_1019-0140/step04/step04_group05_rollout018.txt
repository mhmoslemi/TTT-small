def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        s = w.sum()
        if abs(s - total) <= 1e-12:
            return w
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo≤x_i≤hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        s = w.sum()
        if abs(s - total) <= 1e-12:
            return w
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradient(h, k):
    """Gradient of the correlation at lag k w.r.t. h."""
    n = h.size
    if k >= 0:
        shift1 = np.empty_like(h)
        shift1[:n - k] = 1.0 - h[k:]
        shift1[n - k:] = 0.0
        shift2 = np.empty_like(h)
        shift2[:k] = 0.0
        shift2[k:] = h[:n - k]
    else:
        k2 = -k
        shift1 = np.empty_like(h)
        shift1[:k2] = 0.0
        shift1[k2:] = 1.0 - h[:n - k2]
        shift2 = np.empty_like(h)
        shift2[:n - k2] = h[k2:]
        shift2[n - k2:] = 0.0
    return shift1 - shift2

def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_descent_optimize(h, deadline, rng, target=0.3808):
    """Projected sub‑gradient descent on max correlation."""
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    alpha = 0.5            # base step size
    it = 0
    no_improve = 0
    while time.time() < deadline:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx = np.argmax(corr)
        cur_score = corr[idx] * dx
        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h.copy()
            no_improve = 0
            if best_score <= target:
                break
        else:
            no_improve += 1
        if no_improve >= 10000:
            cur_h = best_h + rng.normal(scale=0.02, size=n)
            cur_h = project_onto_box_sum(cur_h, n / 2.0)
            no_improve = 0
            continue
        k = idx - (n - 1)
        grad = compute_gradient(cur_h, k)
        step = alpha / np.sqrt(it + 1.0)
        cur_h = project_onto_box_sum(cur_h - step * grad, n / 2.0)
        it += 1
        if it % 5000 == 0:
            cur_h = cur_h + rng.normal(scale=0.005, size=n)
            cur_h = project_onto_box_sum(cur_h, n / 2.0)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # initial construction
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
    else:
        def thue_morse(n):
            seq = np.empty(n, dtype=float)
            for i in range(n):
                seq[i] = (bin(i).count('1') & 1)
            return seq
        h = thue_morse(64)

    h = project_onto_box_sum(h, h.size / 2.0)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # progressive refinement
    sizes = [128, 256, 512, 1024]
    for n in sizes:
        if time_left() < 5.0:
            break
        if best_h.size != n:
            best_h = np.interp(
                np.linspace(0.0, 1.0, n, endpoint=False),
                np.linspace(0.0, 1.0, best_h.size, endpoint=False),
                best_h)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        # allocate a share of remaining time
        remaining = time_left()
        stages_left = len(sizes) - sizes.index(n)
        alloc = remaining * 0.6 / stages_left
        if alloc < 2.0:
            alloc = remaining * 0.9 / stages_left
        deadline = time.time() + alloc

        opt_h, opt_score = gradient_descent_optimize(best_h, deadline, rng)
        if opt_score < best_score:
            best_score = opt_score
            best_h = opt_h.copy()
        if best_score <= 0.3808:
            break

    # final short random binary search
    remaining = time_left()
    if remaining > 2.0:
        n = best_h.size
        attempts = int(remaining * 5)  # a few attempts per second
        for _ in range(attempts):
            h_rand = random_balanced_binary(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score = sc
                best_h = h_rand.copy()
                if best_score <= 0.3808:
                    break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
