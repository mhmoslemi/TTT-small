def correlation_self(h, N, dx):
    rfft_h = np.fft.rfft(h, n=N)
    corr = np.fft.irfft(rfft_h * np.conj(rfft_h), n=N)[:2*h.size-1] * dx
    return corr

def gradient_weighted_corr(h, w, dx, N):
    # w: length 2n-1
    w_rev = w[::-1]
    rfft_h = np.fft.rfft(h, n=N)
    rfft_w = np.fft.rfft(w, n=N)
    rfft_w_rev = np.fft.rfft(w_rev, n=N)
    conv = np.fft.irfft(rfft_w * rfft_h, n=N)
    conv_rev = np.fft.irfft(rfft_w_rev * rfft_h, n=N)
    n = h.size
    comp1 = conv_rev[(n-1):(n-1)+n]
    comp2 = conv[(n-1):(n-1)+n]
    grad = dx * (comp1 + comp2)
    return grad

def optimize_h_continuous(h0, target_sum, dx, time_left, max_iters=5000):

    mom = np.zeros_like(h)
    best_h, best_c5 = h.copy(), compute_c5_fast(h)

    step0 = 0.5
    for it in range(1, max_iters+1):
        if time_left() <= 0: break
        # compute correlation of h with itself
        corr_self = correlation_self(h, N, dx)
        # beta schedule
        beta = 1.0 + (200.0 - 1.0) * (it / max_iters)
        # compute weights
        max_corr = corr_self.max()
        exp_arg = -beta * (corr_self - max_corr)
        # Clip exp_arg to avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()

def compute_c5_fast(h):

    h_rev = h[::-1]

    corr = conv[:2 * n - 1]

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def correlation_self(h, N, dx):
    r = np.fft.rfft(h, n=N)
    corr = np.fft.irfft(r * np.conj(r), n=N)[:2 * h.size - 1] * dx
    return corr

def gradient_weighted_corr(h, w, dx, N):
    n = h.size
    w_rev = w[::-1]
    rh = np.fft.rfft(h, n=N)
    rw = np.fft.rfft(w, n=N)
    rw_rev = np.fft.rfft(w_rev, n=N)
    conv = np.fft.irfft(rw * rh, n=N)
    conv_rev = np.fft.irfft(rw_rev * rh, n=N)
    start = n - 1
    comp1 = conv_rev[start:start + n]
    comp2 = conv[start:start + n]
    return dx * (comp1 + comp2)

def binarize_h(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    bh = np.zeros_like(h)
    bh[idx] = 1.0
    return bh

def binary_swap_anneal(h, rng, start, budget_s, max_iters=200000):
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    n = h.size
    for i in range(max_iters):
        if budget_s - (time.time() - start) < 0.01:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        a, b = rng.choice(ones), rng.choice(zeros)
        h[a], h[b] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[a], h[b] = 1.0, 0.0
        T = max(T_min, T * decay)
    return best_h, best_s

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5_fast(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x): return np.sum(x[:n]) - n / 2.0
    def ineq(x):
        h, t = x[:n], x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    except Exception:
        h_opt = h0
    return h_opt, compute_c5_fast(h_opt)

def optimize_continuous(h0, target_sum, dx, N_corr, N_grad, budget_s, start, max_iters=5000):
    n = h0.size
    h = project_onto_box_sum(h0, target_sum)
    mom = np.zeros_like(h)
    best_h, best_c5 = h.copy(), compute_c5_fast(h)
    beta_m = 0.9
    step0 = 5.0
    for it in range(1, max_iters + 1):
        if budget_s - (time.time() - start) <= 0:
            break
        corr_self = correlation_self(h, N_corr, dx)
        beta = 1.0 + (200.0 - 1.0) * (it / max_iters)
        maxc = corr_self.max()
        w = np.exp(-beta * (corr_self - maxc))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        grad = gradient_weighted_corr(h, w, dx, N_grad)
        mom = beta_m * mom + (1 - beta_m) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target_sum)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new.copy()
        h = h_new
        if best_c5 <= 0.3808:
            break
    return best_h, best_c5
