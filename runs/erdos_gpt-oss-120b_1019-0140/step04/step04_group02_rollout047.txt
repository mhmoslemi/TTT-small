def subgradient_descent(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0: break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)  # uses correlation array
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # add small noise occasionally
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def binary_local_search(b0, max_iters, rng, time_left):
    b = b0.copy()
    n = b.size
    best_score = compute_c5_fft(b)
    best_b = b.copy()
    for it in range(max_iters):
        if time_left() <= 0: break
        # choose a 1 index and 0 index
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        if len(ones)==0 or len(zeros)==0: break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        b_new = b.copy()
        b_new[i] = 0.0
        b_new[j] = 1.0
        score = compute_c5_fft(b_new)
        # accept if improves
        if score < best_score:
            best_score = score
            best_b = b_new.copy()
            b = b_new
        # optional accept with probability based on temperature
        # not needed for quick search
    return best_b, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets contain the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1 as floats)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Subgradient of max‑correlation term (averaged over all maximal shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts):
        grad /= len(shifts)
    return grad


def subgradient_descent(h0, max_iter, step0, beta, rng, time_left):
    """Projected subgradient descent for the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def round_to_balanced_binary(h):
    """Convert a real vector to a balanced binary (exactly n/2 ones)."""
    n = h.size
    idx = np.argsort(h)[::-1]          # descending
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx[:n // 2]] = 1.0
    return bin_vec


def binary_local_search(b0, max_iters, rng, time_left):
    """Hill‑climbing on balanced binary vectors (swap 1↔0)."""
    b = b0.copy()
    n = b.size
    best_score = compute_c5_fft(b)
    best_b = b.copy()

    for it in range(max_iters):
        if time_left() <= 0.0:
            break
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        b_new = b.copy()
        b_new[i] = 0.0
        b_new[j] = 1.0
        score = compute_c5_fft(b_new)
        if score < best_score:
            best_score = score
            best_b = b_new.copy()
            b = b_new
        else:
            # occasional uphill move (simulated annealing)
            temp = 0.05 * (1 - it / max_iters) + 1e-4
            if rng.random() < np.exp((best_score - score) / temp):
                b = b_new
    return best_b, best_score
