        if current_h[i] == current_h[j]:
            # no swap needed, continue
            continue

        delta = new_score - current_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):

            current_score = new_score
            # update best
            if current_score < best_score:

def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)

    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    # final projection
    best_h = project_onto_box_sum(best_h, s)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score

Now compute_corr_full_fft:

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

Now compute_c5_fft:

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

Now compute_grad_max_shifts:

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def binary_anneal(rng, h_start, time_left, max_iters=200000, T_start=0.01, T_end=1e-5):
    cur = h_start.copy()
    cur_score = compute_c5_fft(cur)
    best_h = cur.copy()
    best_score = cur_score
    n = cur.shape[0]
    for it in range(max_iters):
        if time_left() <= 0.0:
            break
        # exponential cooling
        T = T_start * (T_end / T_start) ** (it / max_iters)
        i = rng.integers(n)
        if cur[i] == 0:
            continue
        j = rng.integers(n)
        if cur[j] == 1:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_score = compute_c5_fft(cur)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # define n_points target
    n = 512

    # get base
    if 'initial_h_values' in globals():
        base_arr = np.asarray(initial_h_values, dtype=float)
        # upsample if needed
        if base_arr.shape[0] != n:
            base_arr = upsample(base_arr, n)
    else:
        base_arr = thue_morse(n).astype(float)
    # project onto sum constraint
    base_arr = project_onto_box_sum(base_arr, n / 2.0)

    best_h = None
    best_score = np.inf
    TARGET = 0.3808

    # Phase 1: gradient descent restarts
    restarts = 3
    for r in range(restarts):
        if time_left() < 5.0:
            break
        h0 = base_arr + rng.normal(scale=0.05, size=n)
        h0 = project_onto_box_sum(h0, n / 2.0)
        # max iterations large; let stop via time
        h_opt, sc_opt = gradient_descent(h0, max_iters=300000, step0=0.8, beta=0.9, rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= TARGET:
            break

    # Ensure we have some solution
    if best_h is None:
        best_h = base_arr.copy()
        best_score = compute_c5_fft(best_h)

    # Phase 2: binary threshold
    # top half as ones
    idx = np.argpartition(best_h, -n//2)[-n//2:]
    binary_h = np.zeros_like(best_h)
    binary_h[idx] = 1.0
    sc_bin = compute_c5_fft(binary_h)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, binary_h.copy()

    # Phase 3: binary hill climbing
    if time_left() > 10.0:
        h_bin_opt, sc_bin_opt = binary_hill_climb(rng, binary_h, time_left, max_iters=200000)
        if sc_bin_opt < best_score:
            best_score, best_h = sc_bin_opt, h_bin_opt.copy()
        binary_h = h_bin_opt

    # Phase 4: binary annealing
    if time_left() > 10.0:
        h_bin_opt2, sc_bin_opt2 = binary_anneal(rng, binary_h, time_left, max_iters=200000)
        if sc_bin_opt2 < best_score:
            best_score, best_h = sc_bin_opt2, h_bin_opt2.copy()

    # Phase 5: random binary restarts with short hill climb
    while time_left() > 15.0:
        # generate random binary
        h_rand = random_binary_balanced(n, rng)
        # short hill climb
        h_rand_opt, sc_rand_opt = binary_hill_climb(rng, h_rand, time_left, max_iters=50000)
        if sc_rand_opt < best_score:
            best_score, best_h = sc_rand_opt, h_rand_opt.copy()
        # optionally break if very good
        if best_score <= TARGET:
            break

    # Final projection to enforce constraints
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)

    return best_h, best_score, n

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient of the max‑correlation term (averaged over all max shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, max_iters + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    best_h = project_onto_box_sum(best_h, s)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score


def binary_hill_climb(rng, h_start, time_left, max_iters=200000):
    """Deterministic hill‑climbing on binary vectors (swap 1↔0 only if improves)."""
    cur = h_start.copy()
    cur_score = compute_c5_fft(cur)
    best_h = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    for it in range(max_iters):
        if time_left() <= 0.0:
            break
        i = rng.integers(n)
        if cur[i] == 0:
            continue
        j = rng.integers(n)
        if cur[j] == 1:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_score = compute_c5_fft(cur)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best_h, best_score


def binary_anneal(rng, h_start, time_left, max_iters=200000,
                 T_start=0.01, T_end=1e-5):
    """Simulated‑annealing on binary vectors (swap 1↔0 with Metropolis acceptance)."""
    cur = h_start.copy()
    cur_score = compute_c5_fft(cur)
    best_h = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    for it in range(max_iters):
        if time_left() <= 0.0:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)
        i = rng.integers(n)
        if cur[i] == 0:
            continue
        j = rng.integers(n)
        if cur[j] == 1:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_score = compute_c5_fft(cur)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best_h, best_score
