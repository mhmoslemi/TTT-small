import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def shift_arr(arr, shift):
    """Shift arr by integer `shift` with zero padding (positive = right)."""
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if shift > 0:
        out[shift:] = arr[:n - shift]
    elif shift < 0:
        out[:n + shift] = arr[-shift:]
    else:
        out[:] = arr
    return out

        if max_delta <= 0.0:
            # fallback to a random feasible pair
            i = rng.integers(n_target)
            j = rng.integers(n_target - 1)
            if j >= i:
                j += 1
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 0.0:
                step_factor = max(step_factor * decay_rate, min_step_factor)
                continue

        delta = max_delta * step_factor
        improved = False
        cur_c5 = compute_c5(h)

        for _ in range(5):
            if delta <= 0:
                break
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            new_c5 = compute_c5(h_new)

            # keep the overall best ever found
            if new_c5 < best_score:
                best_score = new_c5
                best_h = h_new.copy()

            # accept if it improves the current configuration
            if new_c5 < cur_c5 - 1e-12:
                h = h_new
                improved = True
                break
            delta *= 0.5

        if not improved:
            # small random tweak to escape plateaus
            i = rng.integers(n_target)
            j = rng.integers(n_target - 1)
            if j >= i:
                j += 1
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta > 0:
                delta = max_delta * step_factor * 0.3
                h[i] += delta
                h[j] -= delta

        # periodically shrink the step size
        if iteration % 10 == 0:
            step_factor = max(step_factor * decay_rate, min_step_factor)

        # occasional restart around the best solution with tiny noise
        if iteration % 500 == 0 and time_left() > 1.0:
            noise = rng.normal(scale=0.01, size=n_target)
            h = best_h + noise
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_box_sum(h, n_target / 2.0)

    # final safeguard projection
    best_h = project_onto_box_sum(best_h, n_target / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_target
