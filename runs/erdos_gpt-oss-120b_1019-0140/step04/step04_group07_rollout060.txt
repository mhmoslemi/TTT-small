def delta_c_swap(i, j, h):
    n = h.shape[0]
    size = 2*n - 1
    delta = np.zeros(size, dtype=np.float64)
    mid = n - 1
    # delta_h values
    dh_i = -1.0
    dh_j = 1.0
    ks = np.arange(- (n - 1), n)  # size 'size'
    # compute contributions for i
    # term A_i
    i_plus = i + ks
    mask_i_plus = (i_plus >= 0) & (i_plus < n)
    delta[mask_i_plus] += dh_i * (1.0 - h[i_plus[mask_i_plus]])
    # term B_i
    i_minus = i - ks
    mask_i_minus = (i_minus >= 0) & (i_minus < n)
    delta[mask_i_minus] += -dh_i * h[i_minus[mask_i_minus]]
    # contributions for j
    j_plus = j + ks
    mask_j_plus = (j_plus >= 0) & (j_plus < n)
    delta[mask_j_plus] += dh_j * (1.0 - h[j_plus[mask_j_plus]])
    j_minus = j - ks
    mask_j_minus = (j_minus >= 0) & (j_minus < n)
    delta[mask_j_minus] += -dh_j * h[j_minus[mask_j_minus]]
    # term C
    delta[mid] -= 2.0
    diff = j - i
    if diff != 0:
        delta[mid + diff] += 1.0
        delta[mid - diff] += 1.0
    return delta

            cur_max = new_max

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto { lo <= x_i <= hi, sum x_i = s }."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1–h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def thue_morse(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)


def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.01)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 2000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def delta_c_swap(i, j, h):
    """Delta of correlation array when swapping a 1 at i with a 0 at j."""
    n = h.shape[0]
    size = 2 * n - 1
    delta = np.zeros(size, dtype=np.float64)
    mid = n - 1
    dh_i = -1.0
    dh_j = 1.0
    ks = np.arange(-(n - 1), n)  # length size

    # contributions from i
    i_plus = i + ks
    mask = (i_plus >= 0) & (i_plus < n)
    delta[mask] += dh_i * (1.0 - h[i_plus[mask]])
    i_minus = i - ks
    mask = (i_minus >= 0) & (i_minus < n)
    delta[mask] += -dh_i * h[i_minus[mask]]

    # contributions from j
    j_plus = j + ks
    mask = (j_plus >= 0) & (j_plus < n)
    delta[mask] += dh_j * (1.0 - h[j_plus[mask]])
    j_minus = j - ks
    mask = (j_minus >= 0) & (j_minus < n)
    delta[mask] += -dh_j * h[j_minus[mask]]

    # double‑delta term
    delta[mid] -= 2.0
    diff = j - i
    if diff != 0:
        delta[mid + diff] += 1.0
        delta[mid - diff] += 1.0
    return delta


def anneal_binary(h, c, max_swaps, rng, time_left, T0):
    """Simulated annealing on a binary vector."""
    n = h.shape[0]
    mid = n - 1
    cur_max = c.max()
    cur_score = cur_max * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    for s in range(max_swaps):
        if time_left() <= 0:
            break
        # pick random i with h[i]=1 and j with h[j]=0
        while True:
            i = rng.integers(n)
            if h[i] > 0.5:
                break
        while True:
            j = rng.integers(n)
            if h[j] < 0.5:
                break
        delta = delta_c_swap(i, j, h)
        new_c = c + delta
        new_max = new_c.max()
        new_score = new_max * (2.0 / n)
        T = T0 * (1.0 - s / max_swaps)
        if new_score <= cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            h[i] = 0.0
            h[j] = 1.0
            c = new_c
            cur_max = new_max
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    n = 1024  # grid size
    start_vectors = []

    # use parent construction if available
    if 'initial_h_values' in globals():

        base = upsample(base, n)
        idx_top = np.argpartition(base, -n // 2)[-n // 2:]
        h0 = np.zeros(n, dtype=float)
        h0[idx_top] = 1.0
        start_vectors.append(h0)

    # deterministic Thue–Morse start
    tm = thue_morse(n)
    start_vectors.append(tm)  # binary already

    # a few random restarts
    for _ in range(4):
        start_vectors.append(random_binary_balanced(n, rng))

    best_h = None
    best_score = np.inf

    for h0 in start_vectors:
        if time_left() <= 0:
            break

        # continuous refinement
        max_iters = int(min(15000, max(2000, time_left() * 10)))
        h_gd, _ = gradient_descent_adaptive(h0, max_iters, step0=1.0,
                                            beta=0.9, rng=rng, time_left=time_left)

        # binarize
        idx_top = np.argpartition(h_gd, -n // 2)[-n // 2:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx_top] = 1.0

        c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        cur_score = np.max(c) * (2.0 / n)

        # annealing phase
        if time_left() > 0:
            T0 = max(0.01, (cur_score - 0.35) * n)
            max_swaps = int(min(200000, max(5000, time_left() * 200)))
            h_bin, cur_score = anneal_binary(h_bin, c, max_swaps,
                                             rng, time_left, T0)

        if cur_score < best_score:
            best_score = cur_score
            best_h = h_bin.copy()

    # final polish from the best binary solution
    if best_h is not None and time_left() > 5:
        h_float = best_h.astype(float)
        max_iters = int(min(20000, max(5000, time_left() * 5)))
        h_final, _ = gradient_descent_adaptive(h_float, max_iters,
                                               step0=0.5, beta=0.9,
                                               rng=rng, time_left=time_left)
        idx_top = np.argpartition(h_final, -n // 2)[-n // 2:]
        best_h = np.zeros(n, dtype=float)
        best_h[idx_top] = 1.0
        best_score = compute_c5(best_h)

    # safety projection (keeps binary)
    if best_h is None:
        best_h = random_binary_balanced(n, rng)
        best_score = compute_c5(best_h)

    return best_h, best_score, n
