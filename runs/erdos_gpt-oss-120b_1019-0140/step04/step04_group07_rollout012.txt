def compute_c5(h):
    n = h.shape[0]
    g = 1.0 - h
    # reverse g
    grev = g[::-1]
    size = 1 << (2*n-1).bit_length()  # next power of 2 >= 2n-1
    H = np.fft.fft(h, size)
    G = np.fft.fft(grev, size)
    conv = np.fft.ifft(H * G).real
    # conv length is size; the relevant part is first 2n-1 entries.
    max_corr = conv[:2*n-1].max()
    dx = 2.0 / n
    return max_corr * dx

    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Determine base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0]/2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # scales to try
    target_n = 512  # final resolution
    # We'll iteratively double: 64,128,256,512
    scales = [64,128,256,512]

    for n in scales:
        # Upsample base to new size if needed
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n/2.0)

        # Determine number of gradient descent iterations based on time and n
        if time_left() > 5:

        if restart_budget > 20:

def simulated_annealing_swaps(h_bin, rng, time_left):

    cur_h = h_bin.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    temp = 0.02
    final_temp = 1e-6
    cooling = 0.99995
    while time_left() > 0.05 and temp > final_temp:
        # Choose random i with h[i]==1 and j with h[j]==0
        i = rng.integers(n)
        while cur_h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] == 1:
            j = rng.integers(n)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            delta = new_score - cur_score
            if rng.random() < math.exp(-delta / temp):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_h = cur_h.copy()
                best_score = new_score
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        temp *= cooling
    return best_h, best_score

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        corr = compute_full_corr(h)
        max_corr = corr.max()
        # compute shifts where correlation close to max
        # we can treat all equal max within tolerance
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)

            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_full_corr(h):
    """Full correlation of h with (1-h) (non‑periodic) via FFT."""
    n = h.shape[0]
    g = 1.0 - h
    grev = g[::-1]

    G = np.fft.fft(grev, size)
    conv = np.fft.ifft(H * G).real
    return conv[: 2 * n - 1]


def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    return compute_full_corr(h).max() * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0.1:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0.05:
        corr = compute_full_corr(best_h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[: n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[: n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def simulated_annealing_swaps(h_bin, rng, time_left):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    temp = 0.02
    final_temp = 1e-6
    cooling = 0.99995
    while time_left() > 0.05 and temp > final_temp:
        i = rng.integers(n)
        while cur_h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] == 1:
            j = rng.integers(n)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            if rng.random() < math.exp(-(new_score - cur_score) / temp):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_h = cur_h.copy()
                best_score = new_score
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        temp *= cooling
    return best_h, best_score


def round_to_binary(h):
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # base construction
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]

    for n in scales:
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        if time_left() > 5:
            max_iters = int(min(80000, max(20000, time_left() * (20000.0 / n))))
            h_gd, sc_gd = gradient_descent_adaptive(
                base, max_iters, step0=1.0, beta=0.9, rng=rng, time_left=time_left
            )
            if sc_gd < best_score:
                best_h, best_score = h_gd.copy(), sc_gd
            base = h_gd

        if time_left() > 5:
            h_ms, sc_ms = improve_via_max_shift(base, rng, time_left)
            if sc_ms < best_score:
                best_h, best_score = h_ms.copy(), sc_ms
            base = h_ms

        if time_left() > 5:
            h_pair, sc_pair = local_pair_moves(base, rng, time_left)
            if sc_pair < best_score:
                best_h, best_score = h_pair.copy(), sc_pair
            base = h_pair

        # random binary restarts at this scale
        if time_left() > 20:
            restarts = min(5, int(time_left() // 20))
            for _ in range(restarts):
                if time_left() < 5:
                    break
                h_rand = random_binary_balanced(n, rng)
                max_iters_rand = int(min(5000, max(500, time_left() * (1500.0 / n))))
                h_rand_ref, sc_rand = gradient_descent_adaptive(
                    h_rand,
                    max_iters_rand,
                    step0=0.5,
                    beta=0.9,
                    rng=rng,
                    time_left=time_left,
                )
                if sc_rand < best_score:
                    best_h, best_score = h_rand_ref.copy(), sc_rand
                # binary candidate from refined vector
                h_bin = round_to_binary(h_rand_ref)
                if time_left() > 5:
                    h_anneal, sc_anneal = simulated_annealing_swaps(
                        h_bin, rng, time_left
                    )
                    if sc_anneal < best_score:
                        best_h, best_score = h_anneal.copy(), sc_anneal
                if time_left() > 5:
                    h_pair2, sc_pair2 = local_pair_moves(best_h, rng, time_left)
                    if sc_pair2 < best_score:
                        best_h, best_score = h_pair2.copy(), sc_pair2

    # final intensive refinement on the global best
    if time_left() > 5:
        max_iters_final = int(min(100000, max(50000, time_left() * 50)))
        best_h, best_score = gradient_descent_adaptive(
            best_h,
            max_iters_final,
            step0=0.3,
            beta=0.9,
            rng=rng,
            time_left=time_left,
        )

    if time_left() > 2:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    # optional binary rounding if beneficial
    h_bin = round_to_binary(best_h)
    bin_score = compute_c5(h_bin)
    if bin_score < best_score:
        best_h, best_score = h_bin, bin_score

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
