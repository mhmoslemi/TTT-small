    start = time.time()

    # Determine max n

    # Determine scales
    scales = [64, 128, 256, 512, n_final]
    # Ensure unique
    scales = sorted(set(scales))

    best_h = None
    best_score = np.inf

    # Starting base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    # For each scale
    for idx, n in enumerate(scales):
        # Upsample base to n if needed
        if base.shape[0] != n:
            base = upsample(base, n)
        # ensure constraints
        base = project_onto_box_sum(base, n/2.0)

        # Random binary sampling
        # allocate some time for random sampling
        sample_limit = 300 if n >= 256 else 800
        for _ in range(sample_limit):
            if time.time() - start >= budget_s - 1.0:
                break
            h_rand = random_binary_balanced(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score = sc
                best_h = h_rand.copy()
        # Multi-start gradient descent
        # Determine remaining time and allocate for gradient descent
        remaining = budget_s - (time.time() - start)
        if remaining <= 5.0:
            break  # out of time

        # Determine number of restarts based on n and remaining
        # For each restart we run maybe up to max_iters
        # We'll do 3 restarts like before
        restarts = 3
        for r in range(restarts):

            # Update base for next restart

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # Ensure bracket
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def random_continuous_balanced(n, rng):
    h = rng.random(n)
    return project_onto_box_sum(h, n / 2.0)


def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def thue_morse(n):
    # binary Thue-Morse sequence (0/1)
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                      dtype=float, count=n)
    return seq


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_final = 1024
    scales = [64, 128, 256, 512, n_final]
    best_h = None
    best_score = np.inf

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    for idx, n in enumerate(scales):
        # upscale base
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # random sampling (binary and continuous)
        sample_limit = 300 if n >= 256 else 800
        for _ in range(sample_limit):
            if time.time() - start >= budget_s - 1.0:
                break
            if rng.random() < 0.5:
                h_rand = random_binary_balanced(n, rng)
            else:
                h_rand = random_continuous_balanced(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score = sc
                best_h = h_rand.copy()

        # gradient descent restarts
        remaining = budget_s - (time.time() - start)
        if remaining <= 5.0:
            break
        restarts = 3
        for r in range(restarts):
            if time.time() - start >= budget_s - 2.0:
                break
            if r == 0:
                h0 = base.copy()
            else:
                h0 = base.copy()
                h0 += rng.normal(scale=0.03, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, n / 2.0)
            remaining = budget_s - (time.time() - start)
            max_iters = int(min(150000,
                                max(30000, remaining * (30000.0 / max(64, n)))))
            h_opt, sc_opt = gradient_descent_momentum_custom(
                h0, n_iter=max_iters, step0=0.6, beta=0.9,
                rng=rng, max_time=remaining)
            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()
            base = h_opt

        # simulated annealing refinement
        if best_h is not None and best_h.size == n and time.time() - start < budget_s - 5.0:
            h_cur = best_h.copy()
            cur_score = best_score
            remaining = budget_s - (time.time() - start)
            total_steps = int(min(40000,
                                  max(10000, remaining * 400.0)))
            T0 = 0.02
            T_final = 1e-6
            for step in range(total_steps):
                if time.time() - start >= budget_s:
                    break
                t_frac = (total_steps - step) / total_steps
                T = T0 * t_frac + T_final
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(h_cur[i], 1.0 - h_cur[j])
                if max_move <= 0.0:
                    continue
                delta = max_move * rng.random()
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_score = compute_c5(h_new)
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                    h_cur = h_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_h = h_cur.copy()
            base = best_h

        # prepare for next scale
        if idx < len(scales) - 1:
            if best_h is not None:
                base = best_h.copy()
            # else keep current base (already upscaled at top of loop)

    # final refinement at the last scale
    if best_h is not None and time.time() - start < budget_s - 5.0:
        n = best_h.size
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(300000,
                            max(100000, remaining * (100000.0 / max(64, n)))))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=max_iters, step0=0.2, beta=0.9,
            rng=rng, max_time=remaining)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

    # fallback if nothing found
    if best_h is None:
        n = n_final
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
