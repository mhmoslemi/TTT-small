    # Determine steps based on time left / approx per iteration
    # We'll just loop until time runs out - but we need to guard case of infinite loops.
    while time.time() - start < time_budget:
        # compute correlation and max shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shift_idx = np.argmax(corr)
        shift = shift_idx - (h.size - 1)

        # temperature schedule: linear from T0 to T_final based on elapsed fraction
        elapsed = time.time() - start
        fraction = elapsed / time_budget
        T = T0 * (1 - fraction) + T_final * fraction

        # decide move type: probability p_special e.g., 0.7

            high = min(h.size, h.size - shift)  # exclusive

def gradient_descent_max_shift(h0, step0, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    it = 0
    while time.time() - start < max_time:
        it += 1
        # compute correlation and max shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shift_idx = np.argmax(corr)
        shift = shift_idx - (n - 1)
        # compute gradient for this shift
        grad = np.zeros_like(h)
        if shift >= 0:
            s = shift
            # overlapped range i=0..n-s-1
            i_end = n - s
            # grad[i] += 1 - h[i+s]
            grad[:i_end] += 1.0 - h[s:]
            # grad[i+s] -= h[i]
            grad[s:] -= h[:i_end]
        else:
            s = -shift
            # i from s .. n-1 corresponds to original index i, i-s
            i_start = s
            i_end = n
            # grad[i] += 1 - h[i-s]
            grad[i_start:] += 1.0 - h[:n-s]
            # grad[i-s] -= h[i]
            grad[:n-s] -= h[i_start:]
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        clipped = np.clip(v - lam_mid, lo, hi)
        s = clipped.sum()
        if abs(s - target_sum) < 1e-12:
            return clipped
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_descent_max_shift(h0, step0, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    it = 0
    while time.time() - start < max_time:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shift_idx = np.argmax(corr)
        shift = shift_idx - (n - 1)
        grad = np.zeros_like(h)
        if shift >= 0:
            s = shift
            end = n - s
            grad[:end] += 1.0 - h[s:]
            grad[s:] -= h[:end]
        else:
            s = -shift
            start_i = s
            end_i = n
            grad[start_i:] += 1.0 - h[:n - s]
            grad[:n - s] -= h[start_i:]
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def multi_restart_gradient(base, step0, rng, max_time, restarts=3):
    start = time.time()
    best_h = base.copy()
    best_score = compute_c5(best_h)
    time_per = max_time / float(restarts)
    for r in range(restarts):
        if time.time() - start >= max_time:
            break
        h0 = base.copy()
        if r > 0:
            h0 += rng.normal(scale=0.02, size=h0.shape)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = project_onto_box_sum(h0, base.shape[0] / 2.0)
        remaining = max_time - (time.time() - start)
        it_time = min(time_per, remaining)
        h_opt, sc_opt = gradient_descent_max_shift(h0, step0, rng, it_time)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        base = h_opt
    return best_h, best_score

def simulated_annealing(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    T0 = 0.02
    T_final = 1e-6
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget
        T = T0 * (1.0 - frac) + T_final * frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shift_idx = np.argmax(corr)
        shift = shift_idx - (n - 1)
        if rng.random() < 0.8:
            low = max(0, -shift)
            high = min(n, n - shift)
            if high - low > 0:
                i = rng.integers(low, high)
                j = i + shift
            else:
                i, j = rng.choice(n, size=2, replace=False)
        else:
            i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            max_move = min(h[j], 1.0 - h[i])
            if max_move <= 0.0:
                continue
            i, j = j, i
        scale = min(1.0, T / T0 + 0.1)
        delta = max_move * rng.random() * scale
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        if new_score < max_corr or rng.random() < np.exp(-(new_score - max_corr) / (T + 1e-12)):
            h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_final = 512
    # base construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)
    # upscale to mid size
    n_mid = 256
    if base.shape[0] != n_mid:
        base = upsample(base, n_mid)
    base = project_onto_box_sum(base, n_mid / 2.0)
    # gradient descent with restarts on mid size
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 30.0:
        gd_time = min(remaining * 0.2, 250.0)
        base, _ = multi_restart_gradient(base, step0=0.5, rng=rng, max_time=gd_time, restarts=3)
    # upscale to final size
    if base.shape[0] != n_final:
        base = upsample(base, n_final)
    base = project_onto_box_sum(base, n_final / 2.0)
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    # simulated annealing on final size
    if remaining > 10.0:
        best_h, best_score = simulated_annealing(base, rng, remaining)
    else:
        best_h = base
        best_score = compute_c5(best_h)
    # final short gradient refinement if time permits
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        best_h, best_score = gradient_descent_max_shift(best_h, step0=0.2, rng=rng, max_time=remaining)
    best_h = project_onto_box_sum(best_h, n_final / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_final
