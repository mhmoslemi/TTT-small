def compute_c5_fft(h):
    n = h.shape[0]
    # pad to length 2n maybe power of 2
    L = 2**int(np.ceil(np.log2(2*n-1)))
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1-h, n=L)
    corr = np.fft.irfft(H * G, n=L)[:2*n-1]  # Real output length L, then slice.
    # corr now has correlation for lags -(n-1) .. (n-1) if we interpret index offset.
    # In full mode, np.correlate returns array of length 2n-1 where index i corresponds to shift (i - (n-1)).
    # For our purpose, max over all entries.
    max_corr = corr.max()
    return max_corr * (2.0 / n)

def find_swap_pair(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    k = np.argmax(corr) - (n - 1)
    s = abs(k)
    if s == 0:
        return None  # no shift
    if k > 0:
        # mismatched 1->0 at i: h[i]==1 and i+s<n and h[i+s]==0
        i_candidates = np.where((h[:n - s] == 1) & (h[s:] == 0))[0]
        # candidate swaps: find j where h[j]==0 and h[j+s]==1
        j_candidates = np.where((h[:n - s] == 0) & (h[s:] == 1))[0]
    else: # k <0
        # mismatched 0->1 at i: h[i]==0 and i+s<n and h[i+s]==1
        i_candidates = np.where((h[:n - s] == 0) & (h[s:] == 1))[0]
        # complement: j where h[j]==1 and h[j+s]==0
        j_candidates = np.where((h[:n - s] == 1) & (h[s:] == 0))[0]
    if len(i_candidates) == 0 or len(j_candidates) == 0:
        return None
    # select random i and j
    i = i_candidates[np.random.randint(len(i_candidates))]
    j = j_candidates[np.random.randint(len(j_candidates))]
    # swap positions i and j
    return i, j

def compute_c5_fft(h):
    n = h.shape[0]
    # convolution of h and (1-h) using FFT
    L = 2**int(np.ceil(np.log2(2*n-1)))
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * G, n=L)[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

import time
import numpy as np

def compute_c5_fft(h):
    """Maximum overlap integral approximated via linear correlation (FFT)."""
    n = h.shape[0]
    L = 2 ** int(np.ceil(np.log2(2 * n - 1)))
    a = np.fft.fft(h, L)
    b = np.fft.fft((1.0 - h)[::-1], L)
    conv = np.fft.ifft(a * b).real[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def simulated_annealing_binary(h0, time_limit, rng):
    """SA on binary vectors preserving the exact sum."""
    h = h0.copy()
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.006

    while time.time() - start < time_limit:
        # linear temperature decay based on elapsed fraction
        elapsed = time.time() - start
        temp = init_temp * max(0.0, 1.0 - elapsed / time_limit)

        # pick a random 1 and a random 0 to swap
        idx_one = np.where(h == 1)[0]
        idx_zero = np.where(h == 0)[0]
        if idx_one.size == 0 or idx_zero.size == 0:
            break
        i = idx_one[rng.integers(idx_one.size)]
        j = idx_zero[rng.integers(idx_zero.size)]

        # perform swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5_fft(h)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                cur_score = new_score
            else:
                # revert
                h[i], h[j] = h[j], h[i]

        if best_score <= 0.3808:
            break

    return best_h, best_score


def local_optimize_binary(h, time_limit, rng):
    """Greedy improvement: swap to directly reduce the worst lag."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    start = time.time()

    while time.time() - start < time_limit:
        # full correlation (exact, O(n^2) but n is modest)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k = int(np.argmax(corr) - (n - 1))
        s = abs(k)
        if s == 0:
            break

        if k > 0:
            i_cand = np.where((best_h[: n - s] == 1) & (best_h[s:] == 0))[0]
            j_cand = np.where((best_h[: n - s] == 0) & (best_h[s:] == 1))[0]
        else:
            i_cand = np.where((best_h[: n - s] == 0) & (best_h[s:] == 1))[0]
            j_cand = np.where((best_h[: n - s] == 1) & (best_h[s:] == 0))[0]

        if i_cand.size == 0 or j_cand.size == 0:
            break

        i = int(i_cand[rng.integers(i_cand.size)])
        j = int(j_cand[rng.integers(j_cand.size)])

        best_h[i], best_h[j] = best_h[j], best_h[i]
        new_score = compute_c5_fft(best_h)
        if new_score < best_score:
            best_score = new_score
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]
            break

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # ------------------------------------------------------------------
    # Initialise base vector (binary, exact half ones)
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
        n0 = init_h.size
        order = np.argsort(-init_h)  # descending
        base = np.zeros(n0, dtype=float)
        base[order[: n0 // 2]] = 1.0
    else:
        n0 = 64
        base = random_balanced_binary(n0, rng)

    best_h = base
    best_score = compute_c5_fft(best_h)

    # ------------------------------------------------------------------
    # Multi‑scale simulated annealing
    # ------------------------------------------------------------------
    scales = [64, 128, 256, 512]
    fractions = {64: 0.05, 128: 0.15, 256: 0.30, 512: 0.50}
    target = 0.3808

    for n in scales:
        if time_left() < 2.0:
            break
        # upscale if needed (repeat each entry)
        if best_h.size != n:
            factor = n // best_h.size
            best_h = np.repeat(best_h, factor)
        # guarantee exact half ones (repeat keeps it)
        alloc = time_left() * fractions[n]
        if alloc < 2.0:
            alloc = time_left()
        # split allocation: main SA on current best + a random restart
        main_time = alloc * 0.7
        rand_time = alloc - main_time

        cand_h, cand_score = simulated_annealing_binary(best_h, main_time, rng)
        rand_h = random_balanced_binary(n, rng)
        rand_h, rand_score = simulated_annealing_binary(rand_h, rand_time, rng)

        if cand_score <= rand_score:
            cur_h, cur_score = cand_h, cand_score
        else:
            cur_h, cur_score = rand_h, rand_score

        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h

        if best_score <= target:
            break

    # ------------------------------------------------------------------
    # Final local refinement with remaining budget
    # ------------------------------------------------------------------
    if time_left() > 1.0:
        fin_h, fin_score = local_optimize_binary(best_h, time_left(), rng)
        if fin_score < best_score:
            best_h, best_score = fin_h, fin_score

    # final safety projection (binary already meets constraints)
    best_h = best_h.astype(float)
    best_score = compute_c5_fft(best_h)

    return best_h, best_score, best_h.size
