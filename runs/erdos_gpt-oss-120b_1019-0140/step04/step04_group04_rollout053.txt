def gradient_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros(n, dtype=h.dtype)
    for k in shifts:
        # compute contributions
        if k >= 0:
            # term 1: 1 - h_{i+k} for i: 0..n-1-k
            grad[:n-k] += 1.0 - h[k:]
            # term 2: -h_{i-k} for i: k..n-1
            grad[k:] -= h[:n-k]
        else:
            s = -k
            # term 1: 1 - h_{i - s} for i: s..n-1
            grad[s:] += 1.0 - h[:n-s]
            # term 2: -h_{i + s} for i: 0..n-1-s
            grad[:n-s] -= h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def gd_momentum(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        corr = fft_full_corr(h, 1.0 - h)
        max_corr_val = np.max(corr)
        # Find all shifts achieving max_corr within tolerance
        tol = 1e-12
        idxes = np.where(corr >= max_corr_val - tol)[0]
        shifts = idxes - (n - 1)
        grad = gradient_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # optional occasional random perturbation
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

# Global for SLSQP constraints
SL_SQP_N = None

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def fft_full_corr(a, b):
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b[::-1], size)
    conv = np.fft.irfft(A * B, size)
    return conv[:m]

def compute_c5(h):
    n = h.shape[0]
    corr = fft_full_corr(h, 1.0 - h) * (2.0 / n)
    return np.max(corr)

def gradient_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            s = -k
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] -= h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def gd_momentum(h0, max_iter, step0, beta, rng, start, budget_s):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    tol = 1e-12
    for it in range(1, max_iter + 1):
        if budget_s - (time.time() - start) <= 0:
            break
        corr = fft_full_corr(h, 1.0 - h) * (2.0 / n)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - tol)[0]
        shifts = idxes - (n - 1)
        grad = gradient_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def cosine_shape(n):
    i = np.arange(n)
    return (1.0 - np.cos(2.0 * np.pi * i / n)) * 0.5

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def slsqp_obj(x):
    return x[-1]

def slsqp_eq_sum(x):
    n = SL_SQP_N
    return np.sum(x[:n]) - n / 2.0

def slsqp_ineq_fun(x):
    n = SL_SQP_N
    h = x[:n]
    t = x[-1]
    corr = fft_full_corr(h, 1.0 - h) * (2.0 / n)
    return t - corr

def slsqp_optimize_candidate(h0, max_iter, start, budget_s):
    global SL_SQP_N
    n = h0.size
    SL_SQP_N = n
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'eq', 'fun': slsqp_eq_sum},
            {'type': 'ineq', 'fun': slsqp_ineq_fun}]
    opt.minimize(slsqp_obj, x0, method='SLSQP',
                 bounds=bounds, constraints=cons,
                 options={'maxiter': max_iter,
                          'ftol': 1e-12,
                          'disp': False})
    h_opt = x0[:n]  # fallback if optimizer fails
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    SL_SQP_N = None
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- initial candidate(s) ---
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    # add a deterministic cosine shape for diversity
    cos = cosine_shape(base.shape[0])
    cos = project_onto_box_sum(cos, base.shape[0] / 2.0)
    if compute_c5(cos) < compute_c5(base):
        base = cos

    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]
    target_iters = {64: 50000, 128: 40000, 256: 30000, 512: 25000}

    for n in scales:
        if budget_s - (time.time() - start) <= 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        score = compute_c5(base)
        if score < best_score:
            best_h, best_score = base.copy(), score

        max_possible = int((budget_s - (time.time() - start)) * 5000)
        n_iter = min(target_iters[n], max_possible)
        if n_iter > 0:
            step0 = 0.5 if n <= 256 else 0.4
            h_ref, sc_ref = gd_momentum(base, n_iter, step0, 0.95,
                                        rng, start, budget_s)
            if sc_ref < best_score:
                best_h, best_score = h_ref.copy(), sc_ref
            base = h_ref

        # a few SLSQP refinements (lightweight)
        for _ in range(2):
            if budget_s - (time.time() - start) <= 5:
                break
            h_opt, sc_opt = slsqp_optimize_candidate(base, max_iter=200,
                                                     start=start,
                                                     budget_s=budget_s)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
            base = h_opt

    # --- final intensive refinement ---
    if best_h is not None and budget_s - (time.time() - start) > 5:
        n_final = best_h.shape[0]
        final_iters = int(min(200000, (budget_s - (time.time() - start)) * 400))
        if final_iters > 0:
            best_h, best_score = gd_momentum(best_h, final_iters, 0.2, 0.95,
                                             rng, start, budget_s)
        # final SLSQP pass
        if budget_s - (time.time() - start) > 5:
            h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=300,
                                                     start=start,
                                                     budget_s=budget_s)
            if sc_opt < best_score:
                best_h, best_score = h_opt, sc_opt

    # --- random perturbation phase ---
    while budget_s - (time.time() - start) > 2:
        n_cur = best_h.shape[0]
        # binary random restart
        h_rand = random_binary_balanced(n_cur, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_h, best_score = h_rand, sc_rand

        # swap two entries
        i, j = rng.choice(n_cur, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.02, 0.3)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new, sc_new

        # Gaussian perturb
        pert = rng.normal(scale=0.015, size=n_cur)
        h_pert = best_h + pert
        h_pert = project_onto_box_sum(h_pert, n_cur / 2.0)
        sc_pert = compute_c5(h_pert)
        if sc_pert < best_score:
            best_h, best_score = h_pert, sc_pert

        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
