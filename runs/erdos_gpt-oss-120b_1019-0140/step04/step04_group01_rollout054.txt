def find_bad_one_zero(h, s):
    n = h.shape[0]
    i = np.arange(n)
    # BadOnes:
    # condition1: i+s < 0
    # condition2: i+s >= n
    # condition3: h[i+s]==0
    # create mask
    cond1 = i + s < 0
    cond2 = i + s >= n
    cond3 = np.logical_and(~cond1, ~cond2) & (h[i + s] == 0)
    bad_ones_mask = (h == 1) & (cond1 | cond2 | cond3)
    # BadZeros:
    # need 0 <= i+s < n and h[i+s]==1
    cond0 = i + s >= 0
    cond1b = i + s < n
    cond2b = h[i + s] == 1
    bad_zeros_mask = (h == 0) & cond0 & cond1b & cond2b
    return np.where(bad_ones_mask)[0], np.where(bad_zeros_mask)[0]

        # if perfect? not possible, but continue.

                best_h = h_new
                best_score = new_score
                no_improve = 0
            # else maybe accept with probability (annealing)
        else:
            # fallback random swap
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2*n-1]
    return np.max(corr) * (2.0 / n)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # Expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 for vector h (dx = 2/n)."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max_{k in shifts} corr(k) w.r.t h."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    """Projected gradient descent with momentum, stops when time runs out."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones, uniformly random."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def find_bad_one_zero(h, s):
    """Return indices (bad_ones, bad_zeros) for given shift s."""
    n = h.shape[0]
    h_one = h > 0.5
    i = np.arange(n)
    i_shift = i + s
    # Bad ones: h[i]=1 and (out of range or h[i+s]=0)
    cond_oob_low = i_shift < 0
    cond_oob_high = i_shift >= n
    cond_in = ~cond_oob_low & ~cond_oob_high
    bad_one_mask = np.zeros(n, dtype=bool)
    # out‑of‑bounds contribute automatically
    bad_one_mask[cond_oob_low | cond_oob_high] = True
    if np.any(cond_in):
        bad_one_mask[cond_in] = (h[i_shift[cond_in]] == 0)
    bad_one_mask &= h_one
    # Bad zeros: h[j]=0 and 0<=j+s<n and h[j+s]=1
    bad_zero_mask = np.zeros(n, dtype=bool)
    if np.any(cond_in):
        bad_zero_mask[cond_in] = (h[i_shift[cond_in]] == 1)
    bad_zero_mask &= ~h_one
    return np.where(bad_one_mask)[0], np.where(bad_zero_mask)[0]

def binary_targeted_hill_climb(h_bin, time_left, rng, max_no_improve=5000, total_budget=1100):
    """Hill‑climb on binary vector, biased toward reducing the worst overlap."""
    n = h_bin.shape[0]
    best_h = h_bin.copy()
    best_score = compute_c5_fft(best_h)
    no_improve = 0
    init_T = 1e-4
    while time_left() > 0 and no_improve < max_no_improve:
        corr = compute_corr_scaled_fft(best_h)
        max_corr = np.max(corr)
        # Update no‑improve counter
        if max_corr < best_score - 1e-12:
            best_score = max_corr
            no_improve = 0
        else:
            no_improve += 1
        # Choose a shift achieving the maximum
        idxes = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        s = idxes[rng.integers(len(idxes))] - (n - 1)
        bad_ones, bad_zeros = find_bad_one_zero(best_h, s)
        if bad_ones.size and bad_zeros.size:
            i = rng.choice(bad_ones)
            j = rng.choice(bad_zeros)
        else:
            # fallback random swap
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5_fft(h_new)
        if new_score < best_score - 1e-12:
            best_h = h_new
            best_score = new_score
            no_improve = 0
        else:
            # occasional uphill acceptance
            T = init_T * (time_left() / total_budget)
            if T > 0.0 and rng.random() < np.exp(-(new_score - best_score) / T):
                best_h = h_new
                best_score = new_score
                no_improve = 0
    return best_h, best_score
