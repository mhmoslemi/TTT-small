def step_targeted_move(h, shift, rng):
    n = h.size
    if shift == 0:
        return h
    if shift > 0:
        i_vals = np.arange(n - shift)
    else:
        i_vals = np.arange(-shift, n)
    # compute mask of eligible i
    hi = h[i_vals]
    hjs = h[i_vals + shift]
    # eligible where hi > eps and hjs < 1 - eps
    eps = 1e-8
    mask = (hi > eps) & (hjs < 1 - eps)
    if not np.any(mask):
        return h
    # choose random eligible index
    eligible = i_vals[mask]
    idx = rng.choice(eligible)
    hi_idx = h[idx]
    hj_idx = h[idx + shift]
    delta = min(hi_idx, 1 - hj_idx) * rng.uniform(0.01, 0.5)
    if delta <= 0:
        return h
    h_new = h.copy()
    h_new[idx] -= delta
    h_new[idx + shift] += delta
    return h_new

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n=v.size
    if s <= lo*n:
        return np.full_like(v, lo)
    if s >= hi*n:
        return np.full_like(v, hi)
    # Use Efficient algorithm: sort v - lo, hi? Well we can use binary search on lambda.
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        total = w.sum()
        if abs(total - s) < 1e-12:
            return w
        if total > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def adam_optimize(h0, n_iter, rng, step0=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    for t in range(1, n_iter+1):
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n-1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, s)
    score = compute_c5(h)
    return h, score

import time
import numpy as np


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        total = w.sum()
        if abs(total - s) < 1e-12:
            return w
        if total > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[: n - shift] += 1.0 - h[shift:]
            grad[shift:] += 1.0 - h[: n - shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[: n - s]
            grad[: n - s] += 1.0 - h[s:]
    if shifts:
        grad /= len(shifts)
    return grad


def adam_optimize(h0, n_iter, rng, step0=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    for t in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, s)
    return h, compute_c5(h)


def propose_targeted_move(h, shift, rng):
    n = h.size
    if shift == 0:
        return h.copy(), False
    if shift > 0:
        i_vals = np.arange(n - shift)
    else:
        i_vals = np.arange(-shift, n)
    hi = h[i_vals]
    hjs = h[i_vals + shift]
    eps = 1e-12
    mask = (hi > eps) & (hjs < 1.0 - eps)
    if not np.any(mask):
        return h.copy(), False
    eligible = i_vals[mask]
    idx = rng.choice(eligible)
    delta = min(h[idx], 1.0 - h[idx + shift]) * rng.uniform(0.01, 0.5)
    if delta <= 0:
        return h.copy(), False
    h_new = h.copy()
    h_new[idx] -= delta
    h_new[idx + shift] += delta
    return h_new, True


def local_search(h, n, time_limit, rng, target=0.3808):
    cur = h.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        max_corr = np.max(corr)
        shift = int(np.argmax(corr) - (n - 1))
        cand, ok = propose_targeted_move(cur, shift, rng)
        if not ok:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(cur[i], 1.0 - cur[j]) * rng.uniform(0.01, 0.5)
            cand = cur.copy()
            if delta > 0:
                cand[i] -= delta
                cand[j] += delta
        cand_score = compute_c5(cand)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            elapsed = time.time() - start
            T = max(1e-5, (time_limit - elapsed) / time_limit * 0.01)
            if rng.random() < np.exp(-(cand_score - cur_score) / T):
                cur, cur_score = cand, cand_score
        if rng.random() < 0.02:
            pert = rng.normal(scale=0.02, size=n)
            h_pert = cur + pert
            h_pert = project_onto_box_sum(h_pert, n / 2.0)
            sc_pert = compute_c5(h_pert)
            if sc_pert < cur_score:
                cur, cur_score = h_pert, sc_pert
                if sc_pert < best_score:
                    best, best_score = h_pert.copy(), sc_pert
        if it % 2000 == 0 and it > 0:
            remaining = time_limit - (time.time() - start)
            if remaining > 1.0:
                adam_iters = min(100, int(remaining * 5))
                if adam_iters > 0:
                    h_adam, sc_adam = adam_optimize(cur, adam_iters, rng)
                    cur, cur_score = h_adam, sc_adam
                    if sc_adam < best_score:
                        best, best_score = h_adam.copy(), sc_adam
        it += 1
        if best_score <= target:
            break
    return best, best_score
