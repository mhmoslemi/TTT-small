    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    t = 0
    while t < max_iter and time_left() > 0.0:
        t += 1
        g = compute_gradient_max_shift(h)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_s

def compute_gradient_submax(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)

    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    # compute gradient for this shift
    return gradient_of_shift(h, shift)

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.size
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n); term1[:n_s] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n); term1[s:] = h[:n_s]
        term2 = np.zeros(n); term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_submax(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return gradient_of_shift(h, shift)

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def adam_optimize(h0, time_left, max_iter=200000, step0=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    t = 0
    while t < max_iter and time_left() > 0.0:
        t += 1
        g = compute_gradient_submax(h)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_s

def gradient_descent_momentum(h0, time_left, step0=0.2, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h, best_s = h.copy(), compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_submax(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def slsqp_optimize_candidate(h0, max_iter=1000):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5_fast(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x): return np.sum(x[:n]) - n / 2.0
    def ineq(x):
        h, t = x[:n], x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    except Exception:
        h_opt = h0
    return h_opt, compute_c5_fast(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))
    target_bound = 0.3808
    n = 256
    target_sum = n / 2.0
    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        candidates.append(project_onto_box_sum(init, target_sum))

    candidates.append(thue_morse_binary(n))
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(4):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))

    best_h, best_s = None, np.inf

    # Phase 1: quick optimization of initial candidates
    for c in candidates:
        if time_left() < 5.0:
            break
        h_opt, s_opt = adam_optimize(c, time_left, max_iter=20000, step0=0.5)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= target_bound:
            return best_h, best_s, n

    # Phase 2: random restarts with longer runs
    while time_left() > 15.0 and best_s > target_bound:
        v = rng.random(n)
        v = project_onto_box_sum(v, target_sum)
        h_opt, s_opt = adam_optimize(v, time_left, max_iter=60000, step0=0.5)
        if s_opt < best_s:
            best_s, best_h = s_opt, h_opt.copy()
        if best_s <= target_bound:
            return best_h, best_s, n

    # Phase 3: local refinement with gradient descent momentum
    if best_h is not None and time_left() > 5.0:
        h_ref, s_ref = gradient_descent_momentum(best_h, time_left, step0=0.2, max_iter=200000)
        if s_ref < best_s:
            best_s, best_h = s_ref, h_ref.copy()
        if best_s <= target_bound:
            return best_h, best_s, n

    # Phase 4: final SLSQP refinement
    if best_h is not None and time_left() > 2.0:
        h_slsqp, s_slsqp = slsqp_optimize_candidate(best_h, max_iter=1500)
        if s_slsqp < best_s:
            best_s, best_h = s_slsqp, h_slsqp.copy()
        if best_s <= target_bound:
            return best_h, best_s, n

    # Phase 5: upscale to higher resolution if time permits
    if time_left() > 30.0 and best_s > target_bound:
        n2 = 512
        target_sum2 = n2 / 2.0
        h_up = resample(best_h, n2)
        h_up = project_onto_box_sum(h_up, target_sum2)
        h_opt2, s_opt2 = adam_optimize(h_up, time_left, max_iter=120000, step0=0.5)
        if s_opt2 < best_s:
            best_s, best_h, n = s_opt2, h_opt2.copy(), n2
        # final refinement at higher resolution
        if time_left() > 5.0:
            h_ref2, s_ref2 = gradient_descent_momentum(best_h, time_left, step0=0.2, max_iter=200000)
            if s_ref2 < best_s:
                best_s, best_h = s_ref2, h_ref2.copy()
            if time_left() > 2.0:
                h_slsqp2, s_slsqp2 = slsqp_optimize_candidate(best_h, max_iter=1500)
                if s_slsqp2 < best_s:
                    best_s, best_h = s_slsqp2, h_slsqp2.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), target_sum)
        best_s = compute_c5_fast(best_h)
    else:
        best_h = project_onto_box_sum(best_h, n / 2.0 if n == best_h.size else n / 2.0)
        best_s = compute_c5_fast(best_h)

    return best_h, best_s, n
