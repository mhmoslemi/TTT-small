def optimize_h(h, rng, time_left):
    n = h.size
    s = n/2
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    step = 0.5
    
    no_improve = 0
    while time_left() > 2:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # shift indices where corr close to max
        close_shifts = np.where(corr >= max_corr - 1e-8)[0] - (n - 1)
        # compute gradient for these shifts
        grad = compute_gradient(h, close_shifts)
        # momentum
        momentum = 0.9 * momentum + 0.1 * grad
        # step adapt: if no improvement for many steps reduce step
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score < best_score - 1e-7:
            best_h = h_new.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
        # if improvement fine or no_improve large, adjust step
        if no_improve > 200:
            step *= 0.5
            if step < 1e-4:
                step = 0.5
            no_improve = 0
        h = h_new

        # Occasionally perform targeted mass transfer
        if rng.random() < 0.1:
            h, new_score = improve_max_shift_once(h, rng, s)
            if new_score < best_score:
                best_h = h.copy()
                best_score = new_score
                no_improve = 0

    return best_h, best_score

def compute_gradient(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n-shift] += (1.0 - h[shift:]) - h[:n-shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n-s]) - h[s:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)
    rng = np.random.default_rng(seed)

    # Base initialization
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # generate balanced random of size 64 (or thue-morse)
        base = np.zeros(64, dtype=float)
        idx = rng.choice(64, 32, replace=False)
        base[idx] = 1.0

    # Ensure projection
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # scales to try
    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if time_left() < 5:
            break
        # Upsample base to n if needed
        if best_h.size != n:
            base_h = upsample(best_h, n)
        else:
            base_h = best_h.copy()
        base_h = project_onto_box_sum(base_h, n / 2.0)

        # Optimize base_h for given scale
        h_opt, score_opt = optimize_h(base_h, rng, time_left)
        if score_opt < best_score:
            best_h = h_opt.copy()
            best_score = score_opt

        # Perform random restarts at this scale
        # Determine how many restarts we can attempt
        restart_budget = time_left()
        while restart_budget > 5:
            # generate random balanced binary
            h_rand = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h_rand[idx] = 1.0
            # quick refinement (few iterations)
            h_rand_opt, score_rand_opt = optimize_h(h_rand, rng, time_left)
            if score_rand_opt < best_score:
                best_h = h_rand_opt.copy()
                best_score = score_rand_opt
            restart_budget = time_left()
            # limit restarts to reasonable number (e.g., max 5 per scale)
            if restart_budget < 10 or (restart_budget < time_left() and (restart_budget < 0)):
                # But we don't have an iteration count; just rely on time.
                pass

    # final refinement on best_h at max size (maybe upscale to 1024 if not done)
    if best_h.size < 1024 and time_left() > 10:
        best_h = upsample(best_h, 1024)
        best_h = project_onto_box_sum(best_h, 1024/2.0)
        best_h, best_score = optimize_h(best_h, rng, time_left)

    # Ensure final projection and compute final score
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size

def improve_max_shift_once(h, rng, s):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    shifts = np.where(corr >= max_corr - 1e-8)[0] - (n - 1)
    if shifts.size == 0:
        return h, compute_c5(h)
    shift = rng.choice(shifts)
    if shift >= 0:
        contrib = h[:n-shift] * (1 - h[shift:])
        i = np.argmax(contrib)
        j = i + shift
    else:
        sft = -shift
        contrib = h[sft:] * (1 - h[:n - sft])
        i = np.argmax(contrib) + sft
        j = i - sft
    if h[i] <= 1e-12 or h[j] >= 1 - 1e-12:
        return h, compute_c5(h)
    max_delta = min(h[i], 1 - h[j])
    if max_delta <= 0:
        return h, compute_c5(h)
    delta = max_delta * rng.random() * 0.5
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    # ensure projection if numerical error
    h_new = project_onto_box_sum(h_new, s)
    return h_new, compute_c5(h_new)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure brackets contain the root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradient(h, shifts):
    """Gradient of the worst correlation (averaged over given shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:]) - h[:n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s]) - h[s:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def improve_max_shift_once(h, rng, s):
    """Transfer a small amount of mass to reduce the worst overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifts = np.where(corr >= max_corr - 1e-8)[0] - (n - 1)
    if shifts.size == 0:
        return h, max_corr * (2.0 / n)
    shift = rng.choice(shifts)
    if shift >= 0:
        contrib = h[:n - shift] * (1.0 - h[shift:])
        i = np.argmax(contrib)
        j = i + shift
    else:
        sft = -shift
        contrib = h[sft:] * (1.0 - h[:n - sft])
        i = np.argmax(contrib) + sft
        j = i - sft
    if h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
        return h, max_corr * (2.0 / n)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h, max_corr * (2.0 / n)
    delta = max_delta * rng.random() * 0.5
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    h_new = project_onto_box_sum(h_new, s)
    new_corr = np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))
    return h_new, new_corr * (2.0 / n)


def optimize_h(h, rng, time_left):
    """Projected gradient descent + occasional targeted shifts."""
    n = h.shape[0]
    s = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    momentum = np.zeros_like(h)
    beta = 0.9
    step = 0.8 if n <= 256 else 0.5
    it = 0
    max_iters = int(min(20000, max(2000, time_left() * (2000.0 / n))))
    no_improve = 0

    while it < max_iters and time_left() > 1.0:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        close_shifts = np.where(corr >= max_corr - 1e-6)[0] - (n - 1)
        # limit number of shifts to keep gradient cheap
        if close_shifts.size > 12:
            close_shifts = rng.choice(close_shifts, size=12, replace=False)
        grad = compute_gradient(h, close_shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_score = np.max(corr_new) * (2.0 / n)

        if new_score < best_score - 1e-7:
            best_h = h_new.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            step = max(step * 0.5, 1e-4)
            no_improve = 0

        # occasional targeted improvement
        if rng.random() < 0.07:
            h_tmp, sc_tmp = improve_max_shift_once(h_new, rng, s)
            if sc_tmp < best_score:
                best_h = h_tmp.copy()
                best_score = sc_tmp
                h_new = h_tmp
                no_improve = 0

        h = h_new

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)
    rng = np.random.default_rng(seed)

    # Starting vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_balanced_binary(64, rng)

    base = project_onto_box_sum(base, base.shape[0] / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if time_left() < 6.0:
            break

        # Upsample current best to the new resolution
        if best_h.size != n:
            h_cur = upsample(best_h, n)
        else:
            h_cur = best_h.copy()
        h_cur = project_onto_box_sum(h_cur, n / 2.0)

        # Refine the upsampled vector
        h_opt, sc_opt = optimize_h(h_cur, rng, time_left)
        if sc_opt < best_score:
            best_h, best_score = h_opt.copy(), sc_opt

        # A few random balanced restarts at this scale
        for _ in range(3):
            if time_left() < 5.0:
                break
            h_rand = random_balanced_binary(n, rng)
            h_rand_opt, sc_rand = optimize_h(h_rand, rng, time_left)
            if sc_rand < best_score:
                best_h, best_score = h_rand_opt.copy(), sc_rand

    # Final intensive refinement at the largest resolution attained
    if best_h.size < 1024 and time_left() > 12.0:
        best_h = upsample(best_h, 1024)
        best_h = project_onto_box_sum(best_h, 1024 / 2.0)
        best_h, best_score = optimize_h(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
