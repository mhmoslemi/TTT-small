    dh_i = -1.0
    dh_j = 1.0
    # first contributions for i

    k1_start = -i
    k1_end = n - 1 - i  # inclusive?
    # k vector
    k1 = np.arange(k1_start, k1_end + 1)  # inclusive

    delta[k1 + offset] += dh_i * (1.0 - h[idx1])

    best_h, best_score = h.copy(), cur_max
    best_corr = corr.copy()
    # indices sets
    ones = np.flatnonzero(h > 0.5)  # since binary
    zeros = np.flatnonzero(h < 0.5)
    start_time = time.time()
    it = 0
    while time_left() > 0 and it < max_iters:
        it += 1
        # maybe recompute sets occasionally in case of many swaps

                    break

        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_to_balanced(v):
    n = v.size
    k = n // 2
    idx = np.argpartition(-v, k - 1)[:k]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def delta_corr_swap(h, i, j):
    n = h.shape[0]
    off = n - 1
    delta = np.zeros(2 * n - 1, dtype=float)
    dh_i, dh_j = -1.0, 1.0

    # i as first index
    k = np.arange(-i, n - i)
    delta[k + off] += dh_i * (1.0 - h[i + k])

    # i as second index
    k = np.arange(i - (n - 1), i + 1)
    delta[k + off] += -dh_i * h[i - k]

    # j as first index
    k = np.arange(-j, n - j)
    delta[k + off] += dh_j * (1.0 - h[j + k])

    # j as second index
    k = np.arange(j - (n - 1), j + 1)
    delta[k + off] += -dh_j * h[j - k]

    diff = j - i
    delta[diff + off] += 1.0
    delta[-diff + off] += 1.0
    return delta

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def hill_climb_binary(h_start, target_score=0.3808):
    h = h_start.copy()
    n = h.size
    off = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = np.max(corr)
    best_h, best_score = h.copy(), cur_max
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    T0, T1 = 5e-3, 5e-6
    it = 0
    while _time_left() > 0:
        it += 1
        if it % 1000 == 0:
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
            cur_max = np.max(corr)
        i = _rng().choice(ones)
        j = _rng().choice(zeros)
        delta = delta_corr_swap(h, i, j)
        new_corr = corr + delta
        new_max = np.max(new_corr)
        elapsed = time.time() - _global_start_time
        T = T0 * (T1 / T0) ** (elapsed / _global_budget) if _global_budget else 0.0
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0 and _rng().random() < math.exp(-(new_max - cur_max) / T):
                accept = True
        if accept:
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_h = h.copy()
                if best_score <= target_score:
                    break
        if _time_left() < 5.0:
            break
    return best_h, best_score

def projected_gradient_descent(h0, target_score=0.3808):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while _time_left() > 0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        idx = np.where(corr == maxc)[0]
        shifts = (idx - (n - 1)).tolist()
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = 0.9 * momentum + 0.1 * grad
        eta = 0.5 / math.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            if best_score <= target_score:
                break
        h = h_new
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    global _global_start_time, _global_budget, _global_rng
    _global_start_time = time.time()
    _global_budget = budget_s
    _global_rng = np.random.default_rng(seed)

    n = 1024

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # simple Thue‑Morse fallback
        def thue_morse(m):
            return np.fromiter(((bin(i).count('1') & 1) for i in range(m)),
                               dtype=float, count=m)
        base = thue_morse(64)

    h_cont = upsample(base, n)
    h_cont = project_onto_box_sum(h_cont, n / 2.0)
    h_bin = binarize_to_balanced(h_cont)

    best_h_bin, best_score_bin = hill_climb_binary(h_bin)

    # continuous refinement on the binary result
    refined_h, refined_score = projected_gradient_descent(best_h_bin)

    if refined_score < best_score_bin:
        best_h, best_score = refined_h, refined_score
    else:
        best_h, best_score = best_h_bin, best_score_bin

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
