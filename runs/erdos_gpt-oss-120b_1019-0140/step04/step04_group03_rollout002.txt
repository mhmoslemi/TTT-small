def annealing_swap(h, cur_score, steps, rng, time_limit):
    start = time.time()
    n = h.size
    best_h = h.copy()
    best_score = cur_score
    for step in range(steps):
        if time.time() - start > time_limit:
            break
        # temperature schedule: linear cooling
        T = max(1e-6, (steps - step) / steps) * (0.02)  # maybe
        i, j = rng.choice(n, size=2, replace=False)
        # ensure i index has >0 mass to transfer
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # ensure bound ok (they are)
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score
```

def gradient_descent(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time.time() - start >= max_time:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts with max corr within tolerance
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # optionally add random noise every few iterations
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        # project to box and sum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # ensure the bracket contains the solution
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 using FFT based linear convolution."""
    n = h.shape[0]
    H = np.fft.rfft(h, 2 * n)
    G = np.fft.rfft(1.0 - h, 2 * n)
    conv = np.fft.irfft(H * G, 2 * n)
    return conv.max() * (2.0 / n)


def full_corr_fft(h):
    """Return the linear convolution h * (1-h) of length 2n."""
    n = h.shape[0]
    H = np.fft.rfft(h, 2 * n)
    G = np.fft.rfft(1.0 - h, 2 * n)
    return np.fft.irfft(H * G, 2 * n)


def upsample(arr, new_n):
    """Linear interpolation on [0,1) to change size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent(h0, n_iter, step0, beta, rng, max_time):
    """Projected momentum gradient descent on max correlation."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = full_corr_fft(h)
        max_corr = corr.max()
        # shifts achieving max (tolerance tiny)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def annealing_swap(h, cur_score, steps, rng, max_time):
    """Swap small mass between two positions (preserving sum)."""
    start = time.time()
    n = h.size
    best_h = h.copy()
    best_score = cur_score
    for step in range(steps):
        if time.time() - start >= max_time:
            break
        # linear cooling schedule
        T = max(1e-6, (steps - step) / steps) * 0.02
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_final = 512

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    # coarse scales: quick random sampling and short GD
    for n in (64, 128, 256):
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        # few random binary tries
        for _ in range(20):
            if time.time() - start > budget_s - 200:
                break
            cand = random_binary_balanced(n, rng)
            sc = compute_c5(cand)
            if sc < compute_c5(base):
                base = cand
        # short GD
        remaining = budget_s - (time.time() - start)
        if remaining < 5:
            break
        iters = min(15000, int(remaining * 25))
        if iters > 0:
            base, _ = gradient_descent(base, iters, step0=0.6, beta=0.9, rng=rng,
                                       max_time=remaining * 0.9)

    # upscale to final resolution
    base = upsample(base, n_final)
    base = project_onto_box_sum(base, n_final / 2.0)

    best_h = base
    best_score = compute_c5(best_h)

    # main refinement loop: alternate GD and annealing until time runs out
    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining <= 5:
            break

        # allocate roughly half time to GD
        gd_time = remaining * 0.5
        gd_iters = min(60000, int(gd_time * 40))
        if gd_iters > 0:
            best_h, best_score = gradient_descent(best_h, gd_iters, step0=0.4,
                                                  beta=0.9, rng=rng, max_time=gd_time)
            best_h = project_onto_box_sum(best_h, n_final / 2.0)
            best_score = compute_c5(best_h)

        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining <= 5:
            break

        # allocate remaining half to annealing swaps
        ann_time = remaining * 0.5
        steps = min(80000, int(ann_time * 120))
        if steps > 0:
            best_h, best_score = annealing_swap(best_h, best_score, steps, rng, max_time=ann_time)

    return best_h, best_score, n_final
