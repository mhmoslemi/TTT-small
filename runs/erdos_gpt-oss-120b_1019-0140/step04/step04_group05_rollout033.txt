def simulated_annealing_binary(h0, time_limit, rng):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    start = time.time()
    # set temperature parameters
    init_temp = 0.01
    max_iters = int(1e8)
    for it in range(max_iters):
        if time.time() - start >= time_limit:
            break
        temp = init_temp * (1 - it / max_iters)
        # sample 1 and 0
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        improved = new_score < cur_score
        if improved or (temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp)):
            cur_score = new_score
            # update ones/zeros lists
            ones[ones == i] = j
            zeros[zeros == j] = i
            # ensure arrays remain correct type, maybe need to update order?
            # update best
            if new_score < best_score:

                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # optionally reshuffle ones/zeros after certain steps? Not needed.
    return best_h, best_score

def simulated_annealing_continuous(h0, time_limit, rng):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # set temperature schedule
    init_temp = 0.02
    # use max_iters = int(5e7) maybe too high; we'll break on time.
    it = 0
    # pre-generate random draws batch? Not needed.
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # linearly decreasing temperature based on elapsed fraction
        fraction = elapsed / time_limit
        temp = init_temp * (1 - fraction)
        # pick two indices i != j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute max allowed delta
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        # determine delta magnitude: scale by temperature -> larger moves early
        # scale factor in [0,1] maybe sqrt(temp/init_temp)
        scale = temp / init_temp
        # optionally sqrt to vary
        delta = max_delta * rng.random() * scale
        if delta == 0:
            continue
        # apply move
        h[i] -= delta
        h[j] += delta
        # ensure sum unchanged
        new_score = compute_c5(h)
        if new_score < cur_score or (temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        it += 1
    # project final best onto constraints (should already be)
    best_h = project_onto_box_sum(best_h, n/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)

    m = arr.shape[0]

    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    order = np.argsort(base)
    h_bin = np.zeros(target_n, dtype=float)
    h_bin[order[target_n // 2:]] = 1.0
    best_h = h_bin
    best_score = compute_c5(best_h)
    # binary search phase
    while time_left() > 20.0:
        remaining = time_left()
        chunk = min(15.0, remaining - 5.0)
        if chunk <= 5.0:
            break
        if best_score < 0.381:
            h0 = best_h.copy()
        else:
            h0 = random_balanced_binary(target_n, rng)
        cand_h, cand_score = simulated_annealing_binary(h0, chunk, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        if best_score <= 0.3808:
            break
    # continuous refinement
    remaining = time_left()
    if remaining > 5.0:
        cand_h, cand_score = simulated_annealing_continuous(best_h, remaining, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
    best_h = project_onto_box_sum(best_h, target_n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, float(best_score), int(target_n)
