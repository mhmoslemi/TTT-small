def compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n - 1]
    return corr.max() * (2.0 / n)
```

def crossover_and_mutate(p1, p2, target_sum, rng, cross_rate=0.7, mut_rate=0.1):
    n = p1.shape[0]
    if rng.random() < cross_rate:
        mask = rng.random(n) < 0.5
        child = np.where(mask, p1, p2).astype(float)
    else:
        child = p1.copy()
    # Ensure binary? already binary.
    # Adjust sum:
    cur_sum = child.sum()
    diff = int(cur_sum - target_sum)
    if diff > 0:
        # Too many ones: turn diff ones to zero
        ones_idx = np.where(child == 1)[0]
        to_zero = rng.choice(ones_idx, size=diff, replace=False)
        child[to_zero] = 0.0
    elif diff < 0:
        zeros_idx = np.where(child == 0)[0]
        to_one = rng.choice(zeros_idx, size=-diff, replace=False)
        child[to_one] = 1.0
    # Mutation:
    if rng.random() < mut_rate:
        # swap one and zero
        ones_idx = np.where(child == 1)[0]
        zeros_idx = np.where(child == 0)[0]
        if ones_idx.size > 0 and zeros_idx.size > 0:
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            child[i], child[j] = 0.0, 1.0
    return child

import time, numpy as np

def compute_c5_fast(h):
    """Return max overlap C5 for step function h (vector of floats in [0,1])."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[: 2 * n - 1]
    return corr.max() * (2.0 / n)

def random_binary_balanced(n, rng):
    """Binary vector length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def tournament_select(pop, scores, rng, size=3):
    """Select one individual via tournament."""
    idxs = rng.integers(len(pop), size=size)
    best = idxs[0]
    best_score = scores[best]
    for i in idxs[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return pop[best]

def crossover_and_mutate(p1, p2, target_sum, rng,
                         cross_rate=0.7, mut_rate=0.1):
    """Two‑point (actually uniform) crossover + swap mutation, repairing sum."""
    n = p1.shape[0]
    if rng.random() < cross_rate:
        mask = rng.random(n) < 0.5
        child = np.where(mask, p1, p2).astype(float)
    else:
        child = p1.copy()
    cur_sum = child.sum()
    diff = int(cur_sum - target_sum)
    if diff > 0:
        ones = np.where(child == 1)[0]
        child[rng.choice(ones, size=diff, replace=False)] = 0.0
    elif diff < 0:
        zeros = np.where(child == 0)[0]
        child[rng.choice(zeros, size=-diff, replace=False)] = 1.0
    if rng.random() < mut_rate:
        ones = np.where(child == 1)[0]
        zeros = np.where(child == 0)[0]
        if ones.size and zeros.size:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            child[i], child[j] = 0.0, 1.0
    return child

def local_improve(h, target_sum, rng, time_left):
    """Hill‑climbing swaps to lower C5."""
    cur = compute_c5_fast(h)
    no_improve = 0
    while time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        if new < cur:
            cur = new
            no_improve = 0
        else:
            h[i], h[j] = 1.0, 0.0
            no_improve += 1
            if no_improve > 10000:
                break
    return h, cur

def resample_to_n(arr, n, rng):
    """Linear interpolation of arr (assumed on [0,2]) to length n."""
    if arr.size == n:
        return arr.astype(float).copy()
    x_old = np.linspace(0.0, 2.0, arr.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, arr).astype(float)

    # Parameters
    n = 512                     # resolution
    target_sum = n // 2
    target_bound = 0.3808
    pop_size = 40
    max_gens = 20000

    # Initialise population
    population = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample_to_n(init, n, rng)
        # Binary conversion by top‑half threshold
        idx = np.argpartition(-init, target_sum)[:target_sum]
        h0 = np.zeros(n, dtype=float)
        h0[idx] = 1.0
        population.append(h0)
    # Fill remaining slots
    while len(population) < pop_size:
        population.append(random_binary_balanced(n, rng))

    # Evaluate initial population
    scores = np.array([compute_c5_fast(h) for h in population])
    best_idx = scores.argmin()
    best_h = population[best_idx].copy()
    best_s = scores[best_idx]

    # Early exit if already good
    if best_s <= target_bound:
        return best_h, best_s, n

    generation = 0
    while time_left() > 1.0 and generation < max_gens:
        generation += 1
        # Preserve elite
        new_pop = [best_h]
        # Fill rest
        while len(new_pop) < pop_size:
            p1 = tournament_select(population, scores, rng)
            p2 = tournament_select(population, scores, rng)
            child = crossover_and_mutate(p1, p2, target_sum, rng)
            new_pop.append(child)
        population = new_pop
        scores = np.array([compute_c5_fast(h) for h in population])
        cur_best_idx = scores.argmin()
        cur_best_s = scores[cur_best_idx]
        if cur_best_s < best_s:
            best_s = cur_best_s
            best_h = population[cur_best_idx].copy()
        if best_s <= target_bound:
            break

    # Local hill‑climbing on the best individual
    if time_left() > 0.5:
        best_h, best_s = local_improve(best_h, target_sum, rng, time_left)

    # Final safety projection (should already be binary and sum‑correct)
    cur_sum = best_h.sum()
    diff = int(cur_sum - target_sum)
    if diff != 0:
        if diff > 0:
            ones = np.where(best_h == 1)[0]
            best_h[rng.choice(ones, size=diff, replace=False)] = 0.0
        else:
            zeros = np.where(best_h == 0)[0]
            best_h[rng.choice(zeros, size=-diff, replace=False)] = 1.0

    best_s = compute_c5_fast(best_h)
    return best_h, best_s, n
