def gradient_logsumexp(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    # Compute weights
    w = np.exp(alpha * corr - np.max(alpha * corr))  # subtract max for stability
    w_sum = w.sum()
    w_norm = w / w_sum
    # Compute conv1 = convolution of reversed w_norm with (1 - h)
    w_rev = w_norm[::-1]
    # Determine size for FFT
    L = w_norm.size
    m = 1 << ((L + n - 1 - 1).bit_length())  # next power of two >= L+n-1
    # FFT conv
    fft_w_rev = np.fft.rfft(w_rev, n=m)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=m)
    conv1 = np.fft.irfft(fft_w_rev * fft_one_minus_h, n=m)
    # conv2 = convolution of w_norm with h
    fft_w = np.fft.rfft(w_norm, n=m)
    fft_h = np.fft.rfft(h, n=m)
    conv2 = np.fft.irfft(fft_w * fft_h, n=m)
    # Slice central part
    start = n - 1
    term1 = conv1[start:start + n]
    term2 = conv2[start:start + n]
    grad = (2.0 / n) * (term1 - term2)
    return grad, np.max(corr)
```

Note: subtract max from alpha*corr before exponent to avoid overflow. w_norm is normalized anyway.

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # expand bounds if needed
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

Note: This function expects sum(v) == s after clipping.

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

def projected_gradient_descent_logsumexp(h0, max_iter, step0, beta, rng, time_left, alpha_start, alpha_end):

    momentum = np.zeros_like(h)

Now gradient_logsumexp:

```
def gradient_logsumexp(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    # stability shift
    scaled = alpha * corr
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    w_norm = w / w_sum
    # Convolution using FFT
    w_rev = w_norm[::-1]
    L = w_norm.size
    m_len = 1 << ((L + n - 1 - 1).bit_length())
    # fft for term1
    fft_w_rev = np.fft.rfft(w_rev, n=m_len)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=m_len)
    conv1 = np.fft.irfft(fft_w_rev * fft_one_minus_h, n=m_len)
    # term2
    fft_w = np.fft.rfft(w_norm, n=m_len)
    fft_h = np.fft.rfft(h, n=m_len)
    conv2 = np.fft.irfft(fft_w * fft_h, n=m_len)
    start = n - 1
    term1 = conv1[start:start + n]
    term2 = conv2[start:start + n]
    grad = (2.0 / n) * (term1 - term2)
    return grad, np.max(corr)
```

def slsqp_optimize_candidate(h0, max_iter=400):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np
    import scipy.optimize as opt
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # base h
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    best_h = None
    best_score = np.inf

    res_list = [64, 128, 256, 512]
    for n in res_list:
        if time_left() <= 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n/2.0)
        score = compute_c5(base)
        if score < best_score:
            best_h = base.copy()
            best_score = score

        # Determine iteration budget based on remaining time
        max_iters = int(min(5000, time_left() * 8))
        if max_iters <= 0:
            continue
        step0 = 0.6 if n <= 256 else 0.4
        beta = 0.9
        # gradually increase alpha from 0.5 to 15
        alpha_start = 0.5
        alpha_end = 15.0
        h_opt, sc_opt = projected_gradient_descent_logsumexp(
            base, max_iter=max_iters, step0=step0, beta=beta,
            rng=rng, time_left=time_left,
            alpha_start=alpha_start, alpha_end=alpha_end)
        if sc_opt < best_score:
            best_h = h_opt.copy()
            best_score = sc_opt
        base = h_opt  # use refined as starting for next resolution

    # Final refinement at best resolution
    n_final = best_h.shape[0]
    while time_left() > 10:
        # random balanced perturbations
        h_candidate = best_h.copy()
        # perform a few random swaps conserving sum
        for _ in range(3):
            i, j = rng.choice(n_final, size=2, replace=False)
            delta = min(h_candidate[i], 1.0 - h_candidate[j]) * rng.uniform(0.01, 0.5)
            if delta > 0:
                h_candidate[i] -= delta
                h_candidate[j] += delta
        # short gradient descent
        max_iters = int(min(2000, time_left() * 5))
        h_ref, sc_ref = projected_gradient_descent_logsumexp(
            h_candidate, max_iter=max_iters, step0=0.3, beta=0.9,
            rng=rng, time_left=time_left,
            alpha_start=5.0, alpha_end=30.0)
        if sc_ref < best_score:
            best_h = h_ref.copy()
            best_score = sc_ref
        # occasional Gaussian perturb
        if rng.random() < 0.4:
            h_noise = best_h + rng.normal(scale=0.01, size=n_final)
            h_noise = project_onto_box_sum(h_noise, n_final/2.0)
            sc_noise = compute_c5(h_noise)
            if sc_noise < best_score:
                best_h = h_noise.copy()
                best_score = sc_noise
        if best_score <= 0.3808:
            break

    # final SLSQP local refinement if time permits
    if time_left() > 5:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=300)
        if sc_opt < best_score:
            best_h = h_opt
            best_score = sc_opt

    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_logsumexp(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    scaled = alpha * corr
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_norm = w / w.sum()
    w_rev = w_norm[::-1]
    L = w_norm.size
    m_len = 1 << ((3 * n - 2 - 1).bit_length())
    fft_w_rev = np.fft.rfft(w_rev, n=m_len)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=m_len)
    conv1 = np.fft.irfft(fft_w_rev * fft_one_minus_h, n=m_len).real
    fft_w = np.fft.rfft(w_norm, n=m_len)
    fft_h = np.fft.rfft(h, n=m_len)
    conv2 = np.fft.irfft(fft_w * fft_h, n=m_len).real
    start = n - 1
    term1 = conv1[start:start + n]
    term2 = conv2[start:start + n]
    grad = (2.0 / n) * (term1 - term2)
    return grad, np.max(corr)

def projected_gradient_descent_logsumexp(h0, max_iter, step0, beta, rng, time_left, alpha_start, alpha_end):
    h = h0.copy()
    n = h.shape[0]
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * it / max_iter
        grad, _ = gradient_logsumexp(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, n / 2.0)
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        if new_score < cur_score:
            h, cur_score = h_new, new_score
        else:
            if rng.random() < 0.005:
                h, cur_score = h_new, new_score
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
