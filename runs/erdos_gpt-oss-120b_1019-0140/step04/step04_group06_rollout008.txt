def differential_evolution(initial_vec, n, time_budget, rng):
    popsize = 30
    target_sum = n / 2.0
    # Initialize population
    pop = np.empty((popsize, n), dtype=float)
    pop[0] = project_onto_box_sum(initial_vec.copy(), target_sum)
    for i in range(1, popsize):
        pop[i] = project_onto_box_sum(rng.random(n), target_sum)
    # Evaluate
    scores = np.empty(popsize, dtype=float)
    for i in range(popsize):
        scores[i] = compute_c5(pop[i])
    best_idx = np.argmin(scores)
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]
    # start loop
    start = time.time()
    while time.time() - start < time_budget:
        for i in range(popsize):
            # if time out then break
            if time.time() - start >= time_budget:
                break
            # Choose a,b,c distinct from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # Clip to [0,1] before projection? We'll project after crossover.
            # Crossover
            cross = rng.random(n) < CR
            trial = np.where(cross, mutant, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_onto_box_sum(trial, target_sum)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
    return best_vec, best_score
```

import time, math, numpy as np

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds cover target
    if f(lam_low) < target:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target) < 1e-12:
            break
        if fm > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def differential_evolution(initial_vec, n, time_budget, rng, popsize=30, F=0.5, CR=0.9):
    target = n / 2.0
    pop = np.empty((popsize, n), dtype=float)
    pop[0] = project_onto_box_sum(initial_vec.copy(), target)
    for i in range(1, popsize):
        pop[i] = project_onto_box_sum(rng.random(n), target)
    scores = np.empty(popsize, dtype=float)
    for i in range(popsize):
        scores[i] = compute_c5(pop[i])
    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]
    start = time.time()
    while time.time() - start < time_budget:
        for i in range(popsize):
            if time.time() - start >= time_budget:
                break
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            cross_mask = rng.random(n) < CR
            trial = np.where(cross_mask, mutant, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_onto_box_sum(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
    return best_vec, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float).copy()
    else:
        base = thue_morse(64).astype(float)

    best_h = None
    best_score = float('inf')
    scales = [64, 128, 256, 512]
    fractions = [0.15, 0.25, 0.25, 0.35]

    for idx, n in enumerate(scales):
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        remaining = time_left()
        if remaining <= 0:
            break
        stage_budget = min(budget_s * fractions[idx], remaining)
        # Differential Evolution
        h_de, sc_de = differential_evolution(base, n, stage_budget, rng)
        if sc_de < best_score:
            best_score, best_h = sc_de, h_de.copy()
        # Small gradient refinement using remaining time of the stage
        rem = time_left()
        if rem > 2.0:
            gd_budget = min(rem * 0.2, 30.0)
            iters = int(min(20000, max(2000, gd_budget * 20)))
            h_gd, sc_gd = grad_descent(best_h, n_iter=iters, step0=0.5,
                                       beta=0.9, rng=rng, time_left=time_left)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()
        base = best_h.copy()

    # Final refinement if time permits
    rem = time_left()
    if rem > 5.0 and best_h is not None:
        iters = int(min(40000, max(5000, rem * 25)))
        best_h, best_score = grad_descent(best_h, n_iter=iters, step0=0.5,
                                          beta=0.9, rng=rng, time_left=time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
