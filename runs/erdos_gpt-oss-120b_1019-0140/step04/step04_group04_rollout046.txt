def compute_c5(h):
    n = h.shape[0]
    g = 1.0 - h
    # compute convolution using FFT
    size = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, size)
    fg_rev = np.fft.rfft(g[::-1], size)
    conv = np.fft.irfft(fh * fg_rev, size)
    # conv length = size, but valid part is conv[:2*n-1]
    corr = conv[:2*n-1]
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1 = h shifted left (i.e., h[i+shift]) for i up to n-shift-1
            # term2 = h shifted right for i >= shift: h[i-shift]
            # Actually in compute_gradients_for_shifts they computed:
            # term1 [:n-shift] = h[shift:]
            # term2 [shift:] = h[:n-shift]
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_custom(h0, n_iter, time_left, rng, step0=0.3, beta=0.95):
    h = h0.copy()
    n = h.size
    total = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0: break
        # compute full correlation via FFT
        corr = full_corr(h)
        # find max correlation
        max_corr = np.max(corr)
        # find shift indices with max (within tolerance)
        # tolerance relative to max_corr
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # learning rate schedule
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise injection
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project onto box sum
        h_new = project_onto_box_sum(h_new, total)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure the interval brackets the solution
    if f(lam_lo) < total:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > total:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_lo + lam_hi)
        fm = f(lam_mid)
        if abs(fm - total) < 1e-12:
            break
        if fm > total:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def full_corr(h):
    """Full correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    g_rev = (1.0 - h)[::-1]
    size = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, size)
    fg = np.fft.rfft(g_rev, size)
    conv = np.fft.irfft(fh * fg, size)
    return conv[:2 * n - 1]

def compute_c5(h):
    n = h.shape[0]
    corr = full_corr(h)
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_custom(h0, n_iter, time_left, rng, step0=0.3, beta=0.95):
    h = h0.copy()
    n = h.shape[0]
    total = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = full_corr(h)
        max_corr = np.max(corr)
        score = max_corr * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if (it % 500) == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, total)
        h = h_new
    # final evaluation of the returned vector
    final_score = compute_c5(best_h)
    return best_h, final_score

    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):

    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
