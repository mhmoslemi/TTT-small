def solve_milp_binary(n, time_limit):
    import cvxpy as cp
    # define variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    # constraints
    constraints = [cp.sum(h) == n/2]
    # For each shift, create w variables as integer (boolean)
    # but we can avoid explicit w by noting sum_i h_i * h_{i+k} is (h.T * shift_matrix * h) maybe using a quadratic term, but cp doesn't support product of boolean variables directly.
    # So we need w.
    w_vars = []
    for k in range(-(n-1), n):
        # compute indices i where i and i+k in range
        istart = max(0, -k)
        iend = min(n, n - k)
        length = iend - istart
        if length <= 0:
            continue
        # w for this shift: length variables
        w = cp.Variable(length, boolean=True)
        w_vars.append((k, w, istart))
        # constraints linking
        # h_i <= ... define slices
        # We'll refer to h_i rather than an array; we need to build constraints:
        # w <= h_i and w <= h_{i+k}
        # w >= h_i + h_{i+k} - 1
        # in vectorized form: w <= h[istart:iend]; w <= h[istart+k:iend+k]
        # But h slice returns expression; we can do constraints: w <= h_slice; w <= h_slice2; w >= h_slice + h_slice2 - 1
        h_i = h[istart:iend]
        h_ik = h[istart + k:iend + k]  # because k may be negative -> slicing works accordingly.
        constraints += [w <= h_i, w <= h_ik, w >= h_i + h_ik - 1]
        # sum of w >= t
        constraints.append(cp.sum(w) >= t)
    constraints.append(t >= 0)
    # objective
    obj = cp.Maximize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit*1000), verbose=False)
        except Exception:
            return None, None
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
        h_val = np.rint(h.value).astype(float)  # ensure binary
        # compute score
        score = compute_c5(h_val)
        return h_val, score
    else:
        return None, None
```

def binary_local_search(h, rng, max_swaps):

    # Ensure binary
    h_bin = (h > 0.5).astype(float)
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)

    ones_idx = np.flatnonzero(best_h == 1.0)
    zeros_idx = np.flatnonzero(best_h == 0.0)
    for _ in range(max_swaps):
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

def slsqp_optimize_candidate(h0, max_iter=500):
    import scipy.optimize as opt
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    constraints = [{'type':'eq', 'fun':eq_sum},
                   {'type':'ineq', 'fun':ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def solve_milp_binary(n, time_limit):
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n/2]
    # For each shift
    for k in range(-(n-1), n):
        istart = max(0, -k)
        iend = min(n, n - k)
        length = iend - istart
        if length <= 0:
            continue
        w = cp.Variable(length, boolean=True)
        # slices
        h_i = h[istart:iend]
        h_ik = h[istart + k:iend + k]
        constraints += [w <= h_i, w <= h_ik, w >= h_i + h_ik - 1]
        constraints.append(cp.sum(w) >= t)
    constraints.append(t >= 0)
    obj = cp.Maximize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit*1000), verbose=False)
        except Exception:
            return None, None
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
        h_val = h.value
        if h_val is None:
            return None, None
        h_bin = np.rint(h_val).astype(float)
        score = compute_c5(h_bin)
        return h_bin, score
    else:
        return None, None

import time
import numpy as np
import scipy.optimize as opt

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_objective(x):
    return x[-1]

def slsqp_eq_sum(x, n):
    return np.sum(x[:n]) - n / 2.0

def slsqp_ineq_fun(x, n):
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'eq', 'fun': slsqp_eq_sum, 'args': (n,)},
            {'type': 'ineq', 'fun': slsqp_ineq_fun, 'args': (n,)}]
    res = opt.minimize(slsqp_objective, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)

def binary_local_search(h, rng, max_swaps):
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    ones_idx = np.flatnonzero(best_h == 1.0)
    zeros_idx = np.flatnonzero(best_h == 0.0)
    for _ in range(max_swaps):
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        score = compute_c5(h_new)

            best_h = h_new
            best_score = score
            ones_idx = np.flatnonzero(best_h == 1.0)
            zeros_idx = np.flatnonzero(best_h == 0.0)
    return best_h, best_score

def solve_milp_binary(n, time_limit):
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n / 2]
    for k in range(-(n - 1), n):
        istart = max(0, -k)
        iend = min(n, n - k)
        length = iend - istart
        if length <= 0:
            continue
        w = cp.Variable(length, boolean=True)
        h_i = h[istart:iend]
        h_ik = h[istart + k:iend + k]
        constraints += [w <= h_i,
                        w <= h_ik,
                        w >= h_i + h_ik - 1,
                        cp.sum(w) >= t]
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit * 1000), verbose=False)
        except Exception:
            return None, None
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
        h_val = h.value
        if h_val is None:
            return None, None
        h_bin = np.rint(h_val).astype(float)
        score = compute_c5(h_bin)
        return h_bin, score
    return None, None

    return budget - (time.time() - start)

                best_h, best_score = h_bl, sc_bl

        # Gradient descent refinement
        if time_remaining(start, budget_s) > 5:
            remaining_sec = time_remaining(start, budget_s)
            max_iters = min(target_iters[n], int(remaining_sec * 2000))
            step0 = 0.5 if n <= 128 else 0.3
            best_h, best_score = gradient_descent_momentum_custom(
                best_h, n_iter=max_iters, step0=step0, beta=0.95, rng=rng)

        # SLSQP polishing
        if time_remaining(start, budget_s) > 5:
            h_slsqp, sc_slsqp = slsqp_optimize_candidate(best_h, max_iter=200)
            if sc_slsqp < best_score:
                best_h, best_score = h_slsqp, sc_slsqp

        if best_score <= 0.3808:
            break

    # Final binary refinement from rounded solution if time permits
    if time_remaining(start, budget_s) > 5:
        rounded = np.rint(best_h).astype(float)
        diff = int(best_h.size / 2 - rounded.sum())
        if diff > 0:
            zero_idx = np.flatnonzero(rounded == 0.0)
            if diff <= zero_idx.size:
                idx = rng.choice(zero_idx, size=diff, replace=False)
                rounded[idx] = 1.0
        elif diff < 0:
            one_idx = np.flatnonzero(rounded == 1.0)
            if -diff <= one_idx.size:
                idx = rng.choice(one_idx, size=-diff, replace=False)
                rounded[idx] = 0.0
        rounded = project_onto_box_sum(rounded, best_h.size / 2.0)
        max_swaps = min(5000, int(time_remaining(start, budget_s) * 150))
        h_final_bin, sc_final_bin = binary_local_search(rounded, rng, max_swaps)
        if sc_final_bin < best_score:
            best_h, best_score = h_final_bin, sc_final_bin

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
