def projected_adam(h, n_iter, alpha_start, alpha_end, step0, beta1, beta2, eps, rng, time_left):
    h = h0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s = n / 2
    for t in range(1, n_iter+1):
        if time_left() <= 0: break
        # compute C and weights
        C = compute_correlation(h) of shape 2n-1 scaled by dx.
        # compute alpha schedule: maybe increase linearly or exponentially
        alpha = alpha_start + (alpha_end - alpha_start) * (t-1) / (n_iter-1)
        # compute smooth max and weights
        maxC = np.max(C)
        exps = np.exp(alpha * (C - maxC))
        Z = np.sum(exps)
        w = exps / Z
        # gradient via convolution
        L = 2*n - 1
        # compute w_rev
        w_rev = w[::-1]
        # FFT
        fft_w_rev = np.fft.fft(w_rev, L)
        fft_one_minus_2h = np.fft.fft(1.0 - 2.0*h, L)
        grad_full = np.fft.ifft(fft_w_rev * fft_one_minus_2h).real
        grad = grad_full[(n-1):(2*n-1)]  # central n elements
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        eta = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - eta * m_hat
        # project onto box+sum
        h_new = project_onto_box_sum(h_new, s)
        # optionally compute C after projection to keep track of best
        C_new = compute_c5_fft(h_new)
        if C_new < best_score:
            best_score = C_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        eta = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - eta * m_hat
        # projection
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # generate sequence of length n, values 0 or 1.
    # Use vectorized method: compute parity of binary ones count.
    # Efficient: compute via bits? Simpler: use list comprehension.
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

        if time_left() <= 0.0:

        # projection onto box and sum
        h_new = project_onto_box_sum(h_new, s)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_unscaled(h, w):
    """Gradient of ∑_k w_k·C_k (unscaled, i.e. without dx)."""
    n = h.shape[0]
    L = 2 * n - 1
    w_rev = w[::-1]
    fft_w_rev = np.fft.fft(w_rev, L)
    fft_one_minus_2h = np.fft.fft(1.0 - 2.0 * h, L)
    conv_full = np.fft.ifft(fft_w_rev * fft_one_minus_2h).real
    return conv_full[(n - 1):(2 * n - 1)]


def adam_optimize(h0, time_left, max_iter=1000000,
                  step0=0.1, beta1=0.9, beta2=0.999,
                  eps=1e-8, alpha_start=5.0, alpha_end=50.0):
    """Projected Adam minimizing smoothed max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)

    for t in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        # correlation with scaling
        corr = compute_corr_full_fft(h)
        C = corr * (2.0 / n)

        # soft‑max temperature schedule
        if max_iter > 1:
            alpha = alpha_start + (alpha_end - alpha_start) * (t - 1) / (max_iter - 1)
        else:
            alpha = alpha_end

        maxC = np.max(C)
        exps = np.exp(alpha * (C - maxC))
        Z = np.sum(exps)
        w = exps / Z

        # gradient (unscaled)
        grad = compute_grad_unscaled(h, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - eta * m_hat

        # projection onto box and sum constraint
        h_new = project_onto_box_sum(h_new, s)

        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
