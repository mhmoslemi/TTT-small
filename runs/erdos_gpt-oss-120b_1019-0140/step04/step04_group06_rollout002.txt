def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def repair_binary(v, target_sum, rng):
    # v is np.ndarray dtype uint8 with values 0/1
    cur = int(v.sum())
    diff = target_sum - cur
    if diff > 0:
        # set diff zeros to 1
        zeros = np.where(v == 0)[0]
        if zeros.size > 0:
            choose = rng.choice(zeros, size=diff, replace=False)
            v[choose] = 1
    elif diff < 0:
        ones = np.where(v == 1)[0]
        if ones.size > 0:
            choose = rng.choice(ones, size=-diff, replace=False)
            v[choose] = 0
    return v

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project vector v onto [lo,hi] with sum equal to s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 for step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    v = np.zeros(n, dtype=np.uint8)
    idx = rng.choice(n, size=n // 2, replace=False)
    v[idx] = 1
    return v

def repair_binary(v, target_sum, rng):
    """Adjust binary vector v to have sum = target_sum."""
    cur = int(v.sum())
    diff = target_sum - cur
    if diff > 0:
        zeros = np.where(v == 0)[0]
        if zeros.size >= diff:
            choose = rng.choice(zeros, size=diff, replace=False)
            v[choose] = 1
    elif diff < 0:
        ones = np.where(v == 1)[0]
        if ones.size >= -diff:
            choose = rng.choice(ones, size=-diff, replace=False)
            v[choose] = 0
    return v

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, time_left):
    """Simple momentum gradient descent with projection."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # Target resolution (power of two for easy upsampling)
    n = 256
    target_sum = n // 2

    # ---------- Initialize population ----------
    # Try to use provided initial_h_values, upsample if possible
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        init_bin = np.rint(init_arr).astype(np.uint8)
        init_bin = repair_binary(init_bin, init_bin.size // 2, rng)
        if n % init_bin.size == 0:
            factor = n // init_bin.size
            init_bin = np.repeat(init_bin, factor)
            # after repeat sum is automatically target_sum
        else:
            init_bin = random_binary_balanced(n, rng)
    else:
        init_bin = random_binary_balanced(n, rng)

    pop_size = 40
    elite_size = 2
    tournament_size = 3
    mut_rate = 0.02

    pop = [init_bin]
    while len(pop) < pop_size:
        pop.append(random_binary_balanced(n, rng))

    # Fitness cache
    fitness_cache = {}
    fitness = []
    for ind in pop:
        key = ind.tobytes()
        if key in fitness_cache:
            f = fitness_cache[key]
        else:
            f = compute_c5(ind.astype(float))
            fitness_cache[key] = f
        fitness.append(f)

    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].astype(float)
    best_score = fitness[best_idx]

    # ---------- Genetic Algorithm ----------
    while time_left() > 8.0:
        # Elite preservation
        elite_indices = np.argsort(fitness)[:elite_size]
        new_pop = [pop[i].copy() for i in elite_indices]

        # Generate offspring
        while len(new_pop) < pop_size:
            # parent selection by tournament
            cand = rng.choice(pop_size, size=tournament_size, replace=False)
            p1_idx = cand[int(np.argmin([fitness[i] for i in cand]))]
            cand = rng.choice(pop_size, size=tournament_size, replace=False)
            p2_idx = cand[int(np.argmin([fitness[i] for i in cand]))]
            p1, p2 = pop[p1_idx], pop[p2_idx]

            # uniform crossover
            mask = rng.random(n) < 0.5
            child = np.where(mask, p1, p2).astype(np.uint8)

            # mutation
            mut_mask = rng.random(n) < mut_rate
            child = np.where(mut_mask, 1 - child, child)

            # repair to exact sum
            child = repair_binary(child, target_sum, rng)

            new_pop.append(child)

        # Evaluate new population
        pop = new_pop
        fitness = []
        for ind in pop:
            key = ind.tobytes()
            if key in fitness_cache:
                f = fitness_cache[key]
            else:
                f = compute_c5(ind.astype(float))
                fitness_cache[key] = f
            fitness.append(f)

        # Update best
        cur_best_idx = int(np.argmin(fitness))
        cur_best_score = fitness[cur_best_idx]
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_h = pop[cur_best_idx].astype(float)

        if best_score <= 0.3808:
            break

    # ---------- Continuous refinement ----------
    if time_left() > 4.0:
        # allocate iterations proportionally to remaining time
        remaining = time_left()
        n_iter = int(min(20000, max(2000, remaining * (2000.0 / n))))
        best_h, best_score = grad_descent(best_h, n_iter=n_iter,
                                          step0=0.5, beta=0.9,
                                          rng=rng, time_left=time_left)

    # ---------- Final local tweaks ----------
    # Small random Gaussian perturbations
    while time_left() > 2.0:
        h_new = best_h + rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score, best_h = sc_new, h_new
        else:
            # occasional greedy swap of near‑binary entries
            if time_left() > 1.0:
                ones = np.where(best_h > 0.5)[0]
                zeros = np.where(best_h <= 0.5)[0]
                if ones.size and zeros.size:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    h_swap = best_h.copy()
                    h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
                    sc_swap = compute_c5(h_swap)
                    if sc_swap < best_score:
                        best_score, best_h = sc_swap, h_swap

    # Ensure final feasibility
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
