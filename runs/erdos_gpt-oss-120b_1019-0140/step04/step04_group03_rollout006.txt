import time
import numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def balance_binary(h, rng):
    n = h.shape[0]
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.flatnonzero(h == 1)
        flip = rng.choice(idx, cur - target, replace=False)
        h[flip] = 0.0
    elif cur < target:
        idx = np.flatnonzero(h == 0)
        flip = rng.choice(idx, target - cur, replace=False)
        h[flip] = 1.0
    return h

def compute_c5(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a = np.fft.rfft(h, L)
    b = np.fft.rfft((1.0 - h)[::-1], L)
    conv = np.fft.irfft(a * b, L)
    corr = conv[:2 * n - 1][::-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

    best_h = h.copy()

    T_end = 1e-6
