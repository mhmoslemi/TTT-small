```python
def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        # compute correlation via FFT
        conv = compute_corr_fft(h)
        max_corr = conv.max()
        # compute worst shifts
        # tolerance absolute: maybe 1e-12 * max_corr
        tol = 1e-12 * max_corr
        idxs = np.where(conv >= max_corr - tol)[0]
        shifts = (n - 1 - idxs)  # shift = N-1 - idx
        # compute gradient
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        # update
        h_new = h - step * momentum
        # project onto box+sum
        h_new = project_onto_box_sum(h_new, s_target)
        new_score = compute_c5(h_new)
        # accept if not worse
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
            # increase step slightly
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-8:
                step = step0 * 0.1
        # occasional jitter
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s_target)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {lo<=x_i<=hi, sum x_i = s} via bisection on lambda."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    """Full linear correlation of h with (1-h) via FFT (mode='full')."""
    n = h.shape[0]
    L = 2 * n - 1
    b_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(H * B, n=L)
    return conv

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    max_corr = compute_corr_fft(h).max()
    return max_corr * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from current size to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max correlation term w.r.t h for given integer shifts."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += 1.0 - h[s:]
            grad[s:] += 1.0 - h[:n - s]
        else:
            t = -s
            grad[t:] += 1.0 - h[:n - t]
            grad[:n - t] += 1.0 - h[t:]
    if shifts.size > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent on C5 with momentum and adaptive step."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        conv = compute_corr_fft(h)
        max_corr = conv.max()
        tol = 1e-12 * max_corr
        idxs = np.where(conv >= max_corr - tol)[0]
        shifts = (n - 1 - idxs).astype(int)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-8:
                step = step0 * 0.1

        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, target_sum)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    """Small random mass transfers between two positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

    start = time.time()

    scales = [64, 128, 256, 512, 1024, 2048]

    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
