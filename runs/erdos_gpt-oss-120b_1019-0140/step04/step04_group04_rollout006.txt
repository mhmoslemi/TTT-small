def compute_c5(h):
    n = h.shape[0]
    nfft = 2*n
    # compute full linear convolution h * (1 - h)
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * G, nfft)
    # conv length nfft, but linear correlation length 2*n-1 = (2*n-1)
    # take first 2*n-1 entries (0..2*n-2)
    max_val = conv[:2*n-1].max()
    return max_val * (2.0 / n)

def subgradient_descent(h0, n_iter, step0, beta, rng, time_left):

    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter+1):
        if time_left() <= 0: break
        # compute correlation via FFT
        nfft = 2*n
        H = np.fft.rfft(h, nfft)
        G = np.fft.rfft(1.0 - h, nfft)
        conv = np.fft.irfft(H * G, nfft)[:2*n-1]
        max_corr = conv.max()
        # find all shifts achieving max within tolerance small
        # conv index to shift: shift = idx - (n-1); but we only need idxes
        idxes = np.where(conv >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient for each shift and average
        grad = np.zeros_like(h)
        # compute contributions using vectorized approach for each shift
        for shift in shifts:
            if shift >= 0:
                # term1: 1 - h[i+shift] for i in [0,n-shift)
                grad[:n-shift] += (1.0 - h[shift:])
                # term2: -h[i-shift] for i in [shift, n)
                grad[shift:] -= h[:n-shift]
            else:
                s_shift = -shift
                # for negative shift: shift = -s_shift:
                # term1: 1 - h[i+shift] = 1 - h[i-s_shift] for i in [s_shift, n)
                grad[s_shift:] += (1.0 - h[:n-s_shift])
                # term2: -h[i-shift] = -h[i+s_shift] for i in [0,n-s_shift)
                grad[:n-s_shift] -= h[s_shift:]
        # average
        if len(shifts) > 0:
            grad /= len(shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project to box and sum
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_binary(h0, max_iter, T0, Tf, rng, time_left):

    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    score = best_score

    # precompute indices of ones and zeros
    for it in range(1, max_iter+1):
        if time_left() <= 0: break
        # temperature schedule: exponential
        t = T0 * ((Tf/T0) ** (it / max_iter))
        # pick random 1 index and 0 index
        ones = np.where(h >= 0.5)[0]  # since binary but could be fractional if rounding
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        # evaluate
        new_score = compute_c5(h_new)
        delta = new_score - score
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            h = h_new
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection onto [lo,hi]^n intersect {sum = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx, discretized."""
    n = h.shape[0]
    nfft = 2 * n
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * G, nfft)[: 2 * n - 1]
    return conv.max() * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from old size to new_n (both on [0,2) interval)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1 values)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def subgradient_descent(h0, n_iter, step0, beta, rng, time_left):
    """Subgradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        # correlation via FFT
        nfft = 2 * n
        H = np.fft.rfft(h, nfft)
        G = np.fft.rfft(1.0 - h, nfft)
        conv = np.fft.irfft(H * G, nfft)[: 2 * n - 1]
        max_corr = conv.max()
        idxes = np.where(conv >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)

        # compute subgradient for all max shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                grad[: n - shift] += (1.0 - h[shift:])
                grad[shift:] -= h[: n - shift]
            else:
                sft = -shift
                grad[sft:] += (1.0 - h[: n - sft])
                grad[: n - sft] -= h[sft:]
        if shifts.size > 0:
            grad /= shifts.size

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        # occasional random jitter to escape plateaus
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_binary(h0, max_iter, T0, Tf, rng, time_left):
    """Simple SA on binary vectors (values exactly 0/1, sum = n/2)."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    score = best_score
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        t = T0 * ((Tf / T0) ** (it / max_iter))
        # pick a random 1 and a random 0
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - score
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            h = h_new
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    """SLSQP refinement using auxiliary variable t >= all correlations."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        nfft = 2 * n
        H = np.fft.rfft(h, nfft)
        G = np.fft.rfft(1.0 - h, nfft)
        corr = np.fft.irfft(H * G, nfft)[: 2 * n - 1] * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
