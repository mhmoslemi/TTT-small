def projected_gradient_descent(h0, max_iters, step0, beta, temp_schedule, rng, time_left):
    h = h0.copy()
    n = len(h)
    s = n/2
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    step = step0
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        # Possibly update temperature
        temp = temp_schedule(it)
        # compute weighted gradient
        if temp > 0:
            weights = np.exp((corr - max_corr) / temp)
            wsum = weights.sum()
            # Weighted sum of per-shift grads.
            grad = np.zeros_like(h)
            # get shift indices where weight > small threshold to speed.
            # but all weights >0.
            # We'll iterate all shifts.
            # Shifts array:
            shifts = np.arange(-n+1, n)
            # loop
            for w, shift in zip(weights, shifts):
                if w == 0:
                    continue
                # compute grad_of_shift
                if shift >= 0:
                    s = shift
                    # length n-s
                    grad[:n-s] += w * (1 - h[s:])
                    grad[s:] += -w * h[:n-s]
                else:
                    s = -shift
                    grad[s:] += w * (1 - h[:n-s])
                    grad[:n-s] += -w * h[s:]
            grad /= wsum
        else:
            # max only
            max_idx = np.where(corr == max_corr)[0]
            # compute gradient as sum over all worst shifts
            grad = np.zeros_like(h)
            for idx in max_idx:
                shift = idx - (n-1)
                if shift >=0:
                    s = shift
                    grad[:n-s] += (1 - h[s:])
                    grad[s:] += -h[:n-s]
                else:
                    s = -shift
                    grad[s:] += (1 - h[:n-s])
                    grad[:n-s] += -h[s:]
            grad /= len(max_idx)
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        # update
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= best_score:
            best_h, best_score = h_new.copy(), new_score
        # adapt step: if improved, maybe increase, else decrease.
        if new_score < compute_c5(h):
            # improvement
            step = min(step0, step * 1.05)
        else:
            step = step * 0.7
        # assign h for next iteration
        h = h_new
    return best_h, best_score

def gradient_descent_smooth(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    # schedule for temperature: start high and decay.
    # At iteration t, temp = max_temp * (decay ** t)
    # set max_temp based on range of corr.
    # We'll compute max_temp each iteration based on corr of current h.
    # But we need consistent decay: maybe set decay per ~ max_iters/5.
    decay = 0.95  # per iteration, but may decay too fast.
    # Instead we can compute temperature based on iteration number:
    # warmup_iters = int(max_iters * 0.2)
    # after that temperature = 0 then.
    warmup_iters = int(max_iters * 0.2)
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # Determine temperature
        if it < warmup_iters:
            # temperature = fraction of (max_corr - min_corr)
            range_corr = max_corr - corr.min()
            # to avoid zero
            temp = max(1e-6, range_corr * (1.0 - it / warmup_iters) * 0.5)
        else:
            temp = 0.0

        # Compute gradient
        if temp > 0:
            shifts = np.arange(-n+1, n)
            # compute weights
            w = np.exp((corr - max_corr) / temp)
            wsum = w.sum()
            grad = np.zeros_like(h)
            # loop over shifts
            # To speed, we can vectorize using broadcasting? Not easily.
            for wi, shift in zip(w, shifts):
                if wi == 0:
                    continue
                if shift >= 0:
                    sft = shift
                    grad[:n-sft] += wi * (1 - h[sft:])
                    grad[sft:] += wi * (-h[:n-sft])
                else:
                    sft = -shift
                    grad[sft:] += wi * (1 - h[:n-sft])
                    grad[:n-sft] += wi * (-h[sft:])
            grad /= wsum
        else:
            # only max shift(s) gradient
            # find all shifts with max
            max_indices = np.where(corr == max_corr)[0]
            grad = np.zeros_like(h)
            for idx in max_indices:
                shift = idx - (n-1)
                if shift >= 0:
                    sft = shift
                    grad[:n-sft] += (1 - h[sft:])
                    grad[sft:] += -h[:n-sft]
                else:
                    sft = -shift
                    grad[sft:] += (1 - h[:n-sft])
                    grad[:n-sft] += -h[sft:]
            grad /= len(max_indices)

        # momentum update
        momentum = beta * momentum + (1 - beta) * grad

        # update and project
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)

        new_score = compute_c5(h_new)

        # accept if improvement else not? We are using gradient descent not line search.
        # Keep best globally
        if new_score < best_score:
            best_h, best_score = h_new.copy(), new_score

        # adapt step
        if new_score < cur_score:
            # Good direction
            step = min(step0, step * 1.05)
        else:
            step = step * 0.7
            if step < 1e-6:
                step = step0 * 0.1

        # set for next iteration
        h = h_new
        cur_score = new_score

    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation to a larger array."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (float 0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def grad_of_shift(h, shift):
    """Gradient of the correlation term for a given integer shift."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        g[:n - s] = 1.0 - h[s:]
        g[s:] += -h[:n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[:n - s]
        g[:n - s] += -h[s:]
    return g


def gradient_max_corr(h):
    """Average gradient over all shifts attaining the maximal correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    max_idx = np.where(corr == max_corr)[0]
    g = np.zeros_like(h)
    for idx in max_idx:
        shift = idx - (n - 1)
        g += grad_of_shift(h, shift)
    if max_idx.size:
        g /= max_idx.size
    return g


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum and adaptive step."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        grad = gradient_max_corr(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score < best_score:
            best_h, best_score = h_new.copy(), new_score
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)

    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Transfer small mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # i -> j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        # j -> i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    """Move mass from points that most contribute to the worst shift."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def annealing_optimize(h0, max_iters, temp0, temp1, rng, time_left):
    """Simple simulated annealing on the mass-transfer moves."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # exponential temperature schedule
        t = temp0 * ((temp1 / temp0) ** (it / max_iters))
        i, j = rng.choice(n, size=2, replace=False)
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0:
            continue
        delta = delta_max * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(t, 1e-12)):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h, best_score = h.copy(), cur_score
    return best_h, best_score
