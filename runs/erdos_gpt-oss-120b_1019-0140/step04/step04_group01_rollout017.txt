def binary_swap_anneal(h0, rng, time_left_func, T0=0.5, decay=0.9999, Tmin=1e-6, max_iters=1000000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        # find positions of ones and zeros
        ones = np.where(h > 0.5)[0]  # h binary
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
```

            h[j] = 0.0

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    best_h = None
    best_score = np.inf

    def update_best(candidate):
        nonlocal best_h, best_score
        sc = compute_c5_fft(candidate)
        if sc < best_score:
            best_score = sc
            best_h = candidate.copy()

    # ---- Initial candidates (size 512) ----
    n_base = 512
    init_candidates = []

    if 'initial_h_values' in globals():
        arr = np.asarray(initial_h_values, dtype=float)
        if arr.size != n_base:
            arr = upsample(arr, n_base) if arr.size < n_base else downsample(arr, n_base)
        # binary version (top half)
        idx = np.argpartition(-arr, n_base // 2)[:n_base // 2]
        binary_arr = np.zeros(n_base, dtype=float)
        binary_arr[idx] = 1.0
        init_candidates.append(binary_arr)
        # continuous projection
        init_candidates.append(project_onto_box_sum(arr, n_base / 2.0))

    init_candidates.append(thue_morse(n_base))
    for _ in range(3):
        init_candidates.append(random_binary_balanced(n_base, rng))
    for _ in range(2):
        v = rng.random(n_base)
        init_candidates.append(project_onto_box_sum(v, n_base / 2.0))

    for cand in init_candidates:
        if time_left() <= 0.0:
            break
        update_best(cand)
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---- Binary annealing on base size ----
    while time_left() > 5.0:
        if best_h is not None and best_h.size == n_base:
            start_vec = best_h.copy()
        else:
            start_vec = random_binary_balanced(n_base, rng)
        h_ann, sc_ann = binary_swap_anneal(start_vec, rng, time_left,
                                          T0=0.5, decay=0.9999, Tmin=1e-6,
                                          max_iters=500000)
        if sc_ann < best_score:
            best_score = sc_ann
            best_h = h_ann.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---- Upsample to larger size (1024) and anneal ----
    if time_left() > 2.0:
        n_large = 1024
        if best_h is not None:
            # ensure binary before repeat
            if np.any((best_h > 0) & (best_h < 1)):
                idx = np.argpartition(-best_h, n_base // 2)[:n_base // 2]
                binary_best = np.zeros(n_base, dtype=float)
                binary_best[idx] = 1.0
            else:
                binary_best = best_h.copy()
            h_large = np.repeat(binary_best, 2)
        else:
            h_large = np.repeat(random_binary_balanced(n_base, rng), 2)
        update_best(h_large)
        if best_score <= target:
            return best_h, best_score, best_h.size

        while time_left() > 5.0:
            if rng.random() < 0.3 and best_h is not None and best_h.size == n_base:
                start_large = np.repeat(best_h, 2)
            else:
                start_large = random_binary_balanced(n_large, rng)
            h_ann_l, sc_ann_l = binary_swap_anneal(start_large, rng, time_left,
                                                  T0=0.5, decay=0.9999, Tmin=1e-6,
                                                  max_iters=500000)
            if sc_ann_l < best_score:
                best_score = sc_ann_l
                best_h = h_ann_l.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---- Continuous refinement (gradient descent) ----
    if time_left() > 0.5 and best_h is not None:
        h_ref, sc_ref = gradient_descent_momentum_with_time(best_h, time_left,
                                                            step0=0.5, beta=0.9,
                                                            max_iter=300000)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---- Random continuous restarts (gradient) ----
    while time_left() > 2.0:
        n_cur = best_h.size if best_h is not None else n_base
        v = rng.random(n_cur)
        v = project_onto_box_sum(v, n_cur / 2.0)
        h_new, sc_new = gradient_descent_momentum_with_time(v, time_left,
                                                            step0=0.5, beta=0.9,
                                                            max_iter=200000)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        if best_score <= target:
            break

    # ---- Final projection and return ----
    if best_h is None:
        best_h = thue_morse(n_base)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
