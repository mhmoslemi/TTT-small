def gradient_descent_adaptive(h0, step0, beta, rng, time_left, max_iters):
    h = h0.copy()
    n = h.size
    target_sum = n/2
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    step = step0

    for it in range(max_iters):
        if time_left() <= 0.5:
            break
        # compute shift set
        corr = compute_corr_full(h)   # compute using FFT or maybe np.correlate
        max_corr = np.max(corr)
        # Determine all shifts where corr close to max
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        new_score = compute_c5(h_new)

        if new_score <= best_score:
            best_h, best_score = h_new.copy(), new_score
        if new_score <= compute_c5(h): # if improves current
            h = h_new
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

    return best_h, best_score

def local_pair_moves(h, rng, time_left, max_iter=20000):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    it = 0
    while time_left() > 0.1 and it < max_iter:
        i, j = rng.choice(n, size=2, replace=False)
        # i->j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        # j->i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])

def simulated_annealing(h, rng, time_left, init_temp=1e-3, final_temp=1e-6):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    n = h.size
    total_time = time_left()
    start_time = time.time()
    # The temperature schedule based on remaining fraction:
    while time_left() > 0.15:
        # Compute fraction of remaining time
        elapsed = time.time() - start_time
        t_frac = max(0.0, (total_time - elapsed) / total_time)
        T = final_temp + (init_temp - final_temp) * t_frac
        i, j = rng.choice(n, size=2, replace=False)
        # Transfer from i to j
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        # Accept
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score
        # Optionally also try opposite direction rarely
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

# ---------- Helper functions ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_corr_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    a = np.fft.rfft(h, m)
    b = np.fft.rfft((1.0 - h)[::-1], m)
    c = np.fft.irfft(a * b, m)[: 2 * n - 1]
    return c


def c5_from_corr(corr, n):
    """C5 value from a correlation array."""
    return np.max(corr) * (2.0 / n)


def compute_c5(h):
    """C5 for a vector h."""
    n = h.shape[0]
    return c5_from_corr(compute_corr_full(h), n)


def upsample(arr, new_n):
    """Linear interpolation to a larger array."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (as floats)."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_adaptive(h0, step0, beta, rng, time_left, max_iters):
    """Projected gradient descent with simple step‑size adaptation."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    corr = compute_corr_full(h)
    cur_score = c5_from_corr(corr, n)
    best_h = h.copy()
    best_score = cur_score
    momentum = np.zeros_like(h)
    step = step0

    for _ in range(max_iters):
        if time_left() <= 0.2:
            break
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = project_onto_box_sum(h - step * momentum, target_sum)
        corr_new = compute_corr_full(h_new)
        new_score = c5_from_corr(corr_new, n)

        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score

        if new_score <= cur_score:
            h = h_new
            corr = corr_new
            cur_score = new_score
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    """Transfer mass from points that contribute most to the worst lag."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size

    while time_left() > 0.15:
        corr = compute_corr_full(best_h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[: n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[: n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def local_pair_moves(h, rng, time_left, max_iter=20000):
    """Small random mass transfers between pairs, keeping improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    it = 0
    while time_left() > 0.1 and it < max_iter:
        i, j = rng.choice(n, size=2, replace=False)
        # i -> j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        # j -> i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
        it += 1
    return best_h, best_score


def simulated_annealing(h, rng, time_left, init_temp=1e-3, final_temp=1e-6):
    """Simple annealing: probabilistic acceptance of worsening moves."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    n = h.size
    total = time_left()
    start = time.time()

    while time_left() > 0.15:
        elapsed = time.time() - start
        t_frac = max(0.0, (total - elapsed) / total)
        T = final_temp + (init_temp - final_temp) * t_frac
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # initial vector
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # gradient descent on current scale
        if time_left() > 2.0:
            max_iters = int(min(40000, max(4000, time_left() * (15000.0 / n))))
            h_gd, sc_gd = gradient_descent_adaptive(
                base,
                step0=0.8,
                beta=0.9,
                rng=rng,
                time_left=time_left,
                max_iters=max_iters,
            )
            if sc_gd < best_score:
                best_h, best_score = h_gd.copy(), sc_gd
            base = h_gd

        # targeted max‑shift improvement
        if time_left() > 2.0:
            h_ms, sc_ms = improve_via_max_shift(base, rng, time_left)
            if sc_ms < best_score:
                best_h, best_score = h_ms.copy(), sc_ms
            base = h_ms

        # random restarts for larger scales
        if n >= 256 and time_left() > 5.0:
            for _ in range(2):
                if time_left() < 2.0:
                    break
                h_rand = random_balanced_binary(n, rng)
                max_iters_rand = int(min(20000, max(2000, time_left() * (6000.0 / n))))
                h_rand_ref, sc_rand = gradient_descent_adaptive(
                    h_rand,
                    step0=0.5,
                    beta=0.9,
                    rng=rng,
                    time_left=time_left,
                    max_iters=max_iters_rand,
                )
                if sc_rand < best_score:
                    best_h, best_score = h_rand_ref.copy(), sc_rand
                if time_left() > 1.0:
                    h_rand_pair, sc_rand_pair = local_pair_moves(
                        h_rand_ref, rng, time_left, max_iter=5000
                    )
                    if sc_rand_pair < best_score:
                        best_h, best_score = h_rand_pair.copy(), sc_rand_pair

    # final intensive refinement
    if time_left() > 2.0:
        max_iters_final = int(min(100000, max(20000, time_left() * 50)))
        best_h, best_score = gradient_descent_adaptive(
            best_h,
            step0=0.3,
            beta=0.9,
            rng=rng,
            time_left=time_left,
            max_iters=max_iters_final,
        )

    # simulated annealing
    if time_left() > 2.0:
        best_h, best_score = simulated_annealing(best_h, rng, time_left)

    # final local pair moves
    if time_left() > 1.0:
        best_h, best_score = local_pair_moves(best_h, rng, time_left, max_iter=30000)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
