import time
import math
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto the box [lo,hi] with a sum constraint."""
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for the discrete step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    arr = np.zeros(n, dtype=float)
    arr[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return arr

def thue_morse(n):
    """Generate the Thue‑Morse binary sequence of length n (n must be a power of two)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def to_binary_balanced(arr):
    """Convert any real vector to a binary vector with exactly n/2 ones."""
    n = arr.size
    k = n // 2
    idx = np.argpartition(arr, -k)[-k:]
    out = np.zeros_like(arr, dtype=float)
    out[idx] = 1.0
    return out

def simulated_annealing_binary(h, cur_score, time_budget, rng):
    """Binary simulated annealing with swap moves."""
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_score = cur_score
    # temperature schedule
    T_start = 0.02
    T_end = 1e-7
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget
        T = T_start * (1.0 - frac) + T_end
        i = rng.integers(n)
        if h[i] == 0.0:
            continue
        j = rng.integers(n)
        if h[j] == 1.0:
            continue
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target_score = 0.3808
    # resolution schedule
    scales = [128, 256, 512, 1024]
    # initial base at the smallest scale
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = project_onto_box_sum(base, base.shape[0] / 2.0)
        base = to_binary_balanced(base)
    else:
        base = thue_morse(scales[0])
    best_h = None
    best_score = np.inf
    # iterate over scales
    for idx, n in enumerate(scales):
        # upsample by integer repeat if needed
        if base.shape[0] != n:
            factor = n // base.shape[0]
            base = np.repeat(base, factor)
        # ensure binary and correct sum
        base = to_binary_balanced(base)
        cur_score = compute_c5(base)
        if cur_score < best_score:
            best_score, best_h = cur_score, base.copy()
        # allocate time for this scale
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= 5.0:

        if n == scales[-1]:
