def compute_c5_fast(h):
    n = h.shape[0]
    size = 1 << ((3*n-1).bit_length())
    fft_h = np.fft.rfft(h, n=size)  # real FFT maybe faster
    fft_f = np.fft.rfft(1.0 - h, n=size)

def gradient_descent_softmax(h0, max_iter, step0, alpha0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fast(h)  # compute once
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        # compute corr and c
        n = h.shape[0]
        size = 1 << ((3*n - 1).bit_length())
        fft_h = np.fft.rfft(h, n=size)
        fft_f = np.fft.rfft(1.0 - h, n=size)
        corr = np.fft.irfft(fft_h * np.conj(fft_f), n=size)
        start = n - 1
        c = corr[start:start + 2*n - 1] * (2.0 / n)
        # compute weights
        alpha = alpha0 * (1.0 + 0.001 * it)  # maybe increase slowly
        c_max = np.max(c)
        w_raw = np.exp(alpha * (c - c_max))
        w = w_raw / np.sum(w_raw)
        # compute term1 and term2 via convolution
        # w_arr = w; w_rev_arr = w[::-1]
        fft_w = np.fft.rfft(w, n=size)
        conv1 = np.fft.irfft(fft_h * fft_w, n=size)
        fft_w_rev = np.fft.rfft(w[::-1], n=size)
        conv2 = np.fft.irfft(fft_h * fft_w_rev, n=size)
        term1 = conv1[start:start + n]  # since conv1 length size, but we need i from 0 to n-1 -> conv1[i + start]
        term2 = conv2[start:start + n]
        grad = 1.0 - term1 - term2
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection
        h_new = project_onto_box_sum(h_new, n / 2.0)
        # evaluate
        score = compute_c5_fast(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Project vector v onto the box [lo,hi] with sum equal to s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear upsample from len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5_fast(h):
    """Return (max_correlation, full correlation array) for step function h."""
    n = h.shape[0]
    size = 1 << ((3 * n - 1).bit_length())
    fft_h = np.fft.rfft(h, n=size)
    fft_f = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(fft_h * np.conj(fft_f), n=size)
    start = n - 1
    c = corr[start:start + 2 * n - 1] * (2.0 / n)
    return np.max(c), c

def gradient_descent_softmax(h0, max_iter, step0, alpha_start, alpha_max,
                             beta, rng, time_left):
    """Gradient descent using a softmax weighting of all shifts."""
    h = h0.copy()
    n = h.size
    size = 1 << ((3 * n - 1).bit_length())
    start = n - 1
    best_h = h.copy()
    best_score, _ = compute_c5_fast(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break

        # correlation and weighted scores
        fft_h = np.fft.rfft(h, n=size)
        fft_f = np.fft.rfft(1.0 - h, n=size)
        corr = np.fft.irfft(fft_h * np.conj(fft_f), n=size)
        c = corr[start:start + 2 * n - 1] * (2.0 / n)
        score = np.max(c)

        if score < best_score:
            best_score = score
            best_h = h.copy()

        # softmax weights on shifts
        alpha = alpha_start + (alpha_max - alpha_start) * min(1.0, it / max_iter)
        w_raw = np.exp(alpha * (c - score))
        w = w_raw / np.sum(w_raw)

        # convolutions for gradient
        fft_w = np.fft.rfft(w, n=size)
        conv2 = np.fft.irfft(fft_h * fft_w, n=size)          # term2
        fft_w_rev = np.fft.rfft(w[::-1], n=size)
        conv1 = np.fft.irfft(fft_h * fft_w_rev, n=size)     # term1

        term1 = conv1[start:start + n]
        term2 = conv2[start:start + n]
        grad = 1.0 - term1 - term2

        # momentum and step
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        # occasional jitter to escape flats
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.005, size=n)

        h = project_onto_box_sum(h_new, n / 2.0)

    return best_h, best_score

def hill_climb_binary_swaps(h_bin, max_iter, rng, time_left):
    """Simple swap-based hill climbing for binary, sum‑balanced vectors."""
    n = h_bin.size
    best_h = h_bin.copy()
    best_score, _ = compute_c5_fast(best_h)

    for _ in range(max_iter):
        if time_left() <= 0:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        best_h[i], best_h[j] = 0.0, 1.0
        sc, _ = compute_c5_fast(best_h)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            best_h[i], best_h[j] = 1.0, 0.0

        if best_score <= 0.3808:
            break
    return best_h, best_score
