def compute_grad_softmax(h, w):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s_idx, ws in enumerate(w):
        s = s_idx - (n - 1)
        if ws == 0:
            continue
        if s >= 0:
            # i from 0 to n-s-1:
            grad[:n-s] += ws * (1 - h[s:])   # add part
            # i from s to n-1:
            grad[s:] -= ws * h[:n-s]        # subtract part
        else:
            m = -s
            # s negative: i from m to n-1 for add? Actually add term uses i+s index inside.
            # For s <0:
            # add term: ws * (1 - h[i+s]) for i where i+s ∈ [0,n) => i ∈ [-s, n-1] => i ∈ [m, n-1]
            # So add to grad[m:] + ws*(1 - h[:n-m])? Wait i+s = i - m, so we need (1 - h[i - m]).

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,2)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full linear cross‑correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """Exact C5 = max_k ∫ h(x)(1‑h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_from_w(h, w):
    """Gradient of soft‑max weighted correlation w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    center = n - 1
    for idx, ws in enumerate(w):
        if ws == 0.0:
            continue
        s = idx - center
        if s >= 0:
            # add (1 - h[i+s]) for i=0..n-s-1
            grad[:n - s] += ws * (1.0 - h[s:])
            # subtract h[i-s] for i=s..n-1
            grad[s:] -= ws * h[:n - s]
        else:
            m = -s
            # add (1 - h[i-m]) for i=m..n-1
            grad[m:] += ws * (1.0 - h[:n - m])
            # subtract h[i+m] for i=0..n-m-1
            grad[:n - m] -= ws * h[m:]
    return grad


def smooth_max_adam(h0, n, rng, time_left, max_iters):
    """Adam on soft‑max approximation of max‑correlation."""
    h = h0.copy()
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05
    best_h = h.copy()
    best_score = compute_c5_fft(h)

    alpha_start, alpha_end = 2.0, 30.0

    for it in range(1, max_iters + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max_iters
        w = np.exp(alpha * corr - np.max(alpha * corr))
        w /= w.sum()
        grad = compute_grad_from_w(h, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, target_sum)

        # occasional noise to escape shallow basins
        if it % 5000 == 0:
            h += rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h, target_sum)

        # periodic exact evaluation
        if it % 2000 == 0:
            sc = compute_c5_fft(h)
            if sc < best_score:
                best_score, best_h = sc, h.copy()
    # final check
    sc = compute_c5_fft(h)
    if sc < best_score:
        best_score, best_h = sc, h.copy()
    return best_h, best_score


def binary_round(h, rng):
    """Round to a balanced binary vector (exactly n/2 ones)."""
    n = h.shape[0]
    k = n // 2
    # indices of k largest entries
    idx = np.argpartition(-h, k - 1)[:k]
    bin_vec = np.zeros_like(h)
    bin_vec[idx] = 1.0
    return bin_vec


def binary_hill_climb(h_bin, rng, time_left, target):
    """Local search on balanced binary vectors by swapping bits."""
    n = h_bin.shape[0]
    k = n // 2
    best = h_bin.copy()
    best_score = compute_c5_fft(best)
    ones = np.flatnonzero(best == 1.0)
    zeros = np.flatnonzero(best == 0.0)

    while time_left() > 0.1:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        sc = compute_c5_fft(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            ones = np.flatnonzero(best == 1.0)
            zeros = np.flatnonzero(best == 0.0)
            if best_score <= target:
                break
    return best, best_score
