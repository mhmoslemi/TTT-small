def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)

            grad[:end] += 1.0 - h[k:]
            grad[k:] -= h[:end]

            m = -k
            end = n - m
            grad[m:] += 1.0 - h[:end]
            grad[:end] -= h[m:]
    if len(shifts) > 0:

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            end = n - k
            grad[:end] += 1.0 - h[k:]
            grad[k:] -= h[:end]
        else:
            m = -k
            end = n - m
            grad[m:] += 1.0 - h[:end]
            grad[:end] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    TARGET = 0.3808

    # ------------------------------------------------------------
    # Phase 1: medium resolution (256 points)
    # ------------------------------------------------------------
    n = 256
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128)
    base = upsample(base, n)
    base = project_onto_box_sum(base, n / 2.0)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    # Main restart loop
    while time_left() > 5.0 and best_score > TARGET:
        # Random perturbation of the current best
        h0 = best_h + rng.normal(scale=0.05, size=n)
        h0 = project_onto_box_sum(h0, n / 2.0)

        # Subgradient descent with momentum
        h = h0.copy()
        momentum = np.zeros_like(h)
        s = n / 2.0
        step0 = 0.7
        beta = 0.85
        max_iter = 200_000
        it = 0
        while it < max_iter and time_left() > 0.2:
            it += 1
            corr = compute_corr_full_fft(h)
            max_c = np.max(corr)
            grad = compute_grad_max_shifts(h, corr, max_c)
            momentum = beta * momentum + (1.0 - beta) * grad
            eta = step0 / np.sqrt(it)
            h_new = h - eta * momentum
            if it % 4000 == 0:
                h_new += rng.normal(scale=0.02, size=n)
            h_new = project_onto_box_sum(h_new, s)

            score_new = compute_c5_fft(h_new)
            if score_new < best_score:
                best_score = score_new
                best_h = h_new.copy()
                if best_score <= TARGET:
                    break

            # acceptance for the inner state
            cur_score = max_c * (2.0 / n)
            if score_new <= cur_score:
                h = h_new
            else:
                temp = 0.01 * (1.0 - it / max_iter)
                if rng.random() < np.exp(-(score_new - cur_score) / (temp + 1e-12)):
                    h = h_new

        # --------------------------------------------------------
        # Binary hill‑climbing refinement (optional, if time permits)
        # --------------------------------------------------------
        if time_left() > 2.0:
            bin_h = np.zeros_like(best_h)
            top_idx = np.argpartition(best_h, -int(s))[-int(s):]
            bin_h[top_idx] = 1.0
            cur_bin = bin_h.copy()
            cur_score = compute_c5_fft(cur_bin)

            swaps = 0
            max_swaps = 50_000
            while swaps < max_swaps and time_left() > 0.2:
                ones = np.flatnonzero(cur_bin)
                zeros = np.flatnonzero(1.0 - cur_bin)
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                new_bin = cur_bin.copy()
                new_bin[i] = 0.0
                new_bin[j] = 1.0
                new_score = compute_c5_fft(new_bin)
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (0.01 + 1e-12)):
                    cur_bin = new_bin
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_h = cur_bin.copy()
                        if best_score <= TARGET:
                            break
                swaps += 1

    # ------------------------------------------------------------
    # Phase 2: fine resolution (512 points) if time remains
    # ------------------------------------------------------------
    if time_left() > 10.0 and best_score > TARGET:
        n2 = 512
        h_fine = upsample(best_h, n2)
        h_fine = project_onto_box_sum(h_fine, n2 / 2.0)

        h = h_fine.copy()
        momentum = np.zeros_like(h)
        s2 = n2 / 2.0
        step0 = 0.6
        beta = 0.9
        max_iter = 300_000
        it = 0
        while it < max_iter and time_left() > 0.2:
            it += 1
            corr = compute_corr_full_fft(h)
            max_c = np.max(corr)
            grad = compute_grad_max_shifts(h, corr, max_c)
            momentum = beta * momentum + (1.0 - beta) * grad
            eta = step0 / np.sqrt(it)
            h_new = h - eta * momentum
            if it % 4000 == 0:
                h_new += rng.normal(scale=0.01, size=n2)
            h_new = project_onto_box_sum(h_new, s2)

            score_new = compute_c5_fft(h_new)
            if score_new < best_score:
                best_score = score_new
                best_h = h_new.copy()
                if best_score <= TARGET:
                    break

            cur_score = max_c * (2.0 / n2)
            if score_new <= cur_score:
                h = h_new
            else:
                temp = 0.005 * (1.0 - it / max_iter)
                if rng.random() < np.exp(-(score_new - cur_score) / (temp + 1e-12)):
                    h = h_new

        # Binary refinement at fine scale
        if time_left() > 5.0:
            bin_h = np.zeros_like(best_h)
            top_idx = np.argpartition(best_h, -int(s2))[-int(s2):]
            bin_h[top_idx] = 1.0
            cur_bin = bin_h.copy()
            cur_score = compute_c5_fft(cur_bin)

            swaps = 0
            max_swaps = 20_000
            while swaps < max_swaps and time_left() > 0.2:
                ones = np.flatnonzero(cur_bin)
                zeros = np.flatnonzero(1.0 - cur_bin)
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                new_bin = cur_bin.copy()
                new_bin[i] = 0.0
                new_bin[j] = 1.0
                new_score = compute_c5_fft(new_bin)
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (0.005 + 1e-12)):
                    cur_bin = new_bin
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_h = cur_bin.copy()
                        if best_score <= TARGET:
                            break
                swaps += 1

    # Final projection and evaluation
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
        n_ret = best_h.size
    else:
        n_ret = 256
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, n_ret
