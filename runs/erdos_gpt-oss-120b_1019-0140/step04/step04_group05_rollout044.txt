def compute_c5(h):
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1 - h, n=N)
    corr = np.fft.irfft(H * G, n=N)[:2*n-1]  # length 2n-1
    return np.max(corr) * (2.0 / n)
```

def optimize_h(h, time_budget, rng):
    start = time.time()
    n = h.size
    cur_score, cur_lag, _ = max_corr_and_lag(h)
    best_h = h.copy()
    best_score = cur_score
    temp0 = 0.005
    iter_count = 0
    while time.time() - start < time_budget:
        # subgradient via lag
        lag = cur_lag
        grad = np.zeros_like(h)
        if lag >= 0:
            i = np.arange(n - lag)
            grad[i] = (1 - h[i + lag])
            grad[i + lag] -= h[i]
        else:
            lag_abs = -lag
            i = np.arange(lag_abs, n)
            grad[i] = (1 - h[i - lag_abs])
            grad[i - lag_abs] -= h[i]
        # find max pos and min neg
        i_pos = np.argmax(grad)
        i_neg = np.argmin(grad)
        if i_pos == i_neg:
            # fallback random
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i: j+=1
            max_delta = min(h[i], 1 - h[j])
            if max_delta <= 0: continue
            delta = max_delta * rng.random() * 0.5
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            max_delta = min(h[i_pos], 1 - h[i_neg])
            if max_delta <= 0:
                # try random
                i = rng.integers(n)
                j = rng.integers(n-1)
                if j >= i: j+=1
                max_delta = min(h[i], 1 - h[j])
                if max_delta <= 0: continue
                delta = max_delta * rng.random() * 0.5
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
            else:
                delta = max_delta * rng.random() * 0.5
                h_new = h.copy()
                h_new[i_pos] -= delta
                h_new[i_neg] += delta
        # ensure feasible within float tolerance; projection may not be needed but do:
        h_new = project_onto_box_sum(h_new, n/2.0)

        new_score, new_lag, _ = max_corr_and_lag(h_new)

        # acceptance rule
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            cur_lag = new_lag
        else:
            # temperature based
            t = temp0 * (1.0 - (time.time() - start) / time_budget)
            if t > 0 and rng.random() < np.exp(-(new_score - cur_score) / t):
                h = h_new
                cur_score = new_score
                cur_lag = new_lag
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        if best_score <= 0.3808:
            break
        iter_count += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral C5 using FFT convolution."""
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * G, n=N)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def max_corr_and_lag(h):
    """Return (max value, signed lag achieving max, raw correlation array)."""
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * G, n=N)[: 2 * n - 1]
    idx = np.argmax(corr)
    max_val = corr[idx] * (2.0 / n)
    lag = idx - (n - 1)
    return max_val, lag, corr


def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def upsample(arr, new_n):
    """Linear interpolation from old grid to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def optimize_h(h0, time_budget, rng):
    """Gradient‑guided local search with projection."""
    start = time.time()
    h = h0.copy()
    n = h.size
    cur_score, cur_lag, _ = max_corr_and_lag(h)
    best_h = h.copy()
    best_score = cur_score
    temp0 = 0.006

    while time.time() - start < time_budget:
        # subgradient from current lag
        lag = cur_lag
        grad = np.zeros_like(h)
        if lag >= 0:
            i = np.arange(n - lag)
            grad[i] = (1.0 - h[i + lag])
            grad[i + lag] -= h[i]
        else:
            lag_abs = -lag
            i = np.arange(lag_abs, n)
            grad[i] = (1.0 - h[i - lag_abs])
            grad[i - lag_abs] -= h[i]

        i_pos = int(np.argmax(grad))
        i_neg = int(np.argmin(grad))

        if grad[i_pos] <= 0 and grad[i_neg] >= 0:
            # random perturbation
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random() * 0.8
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            max_delta = min(h[i_pos], 1.0 - h[i_neg])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random() * 0.8
            h_new = h.copy()
            h_new[i_pos] -= delta
            h_new[i_neg] += delta

        h_new = project_onto_box_sum(h_new, n / 2.0)
        new_score, new_lag, _ = max_corr_and_lag(h_new)

        # acceptance rule
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            cur_lag = new_lag
        else:
            elapsed = time.time() - start
            t = temp0 * max(0.0, 1.0 - elapsed / time_budget)
            if t > 0 and rng.random() < np.exp(-(new_score - cur_score) / t):
                h = h_new
                cur_score = new_score
                cur_lag = new_lag

        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        if best_score <= 0.3808:
            break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    scales = [64, 128, 256, 512]
    fractions = {64: 0.05, 128: 0.10, 256: 0.25, 512: 0.60}
    best_h = base.copy()
    best_score = compute_c5(best_h)

    for n in scales:
        if time_left() <= 0:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        alloc = min(fractions[n] * budget_s, time_left())
        if alloc <= 0:
            continue
        cand_h, cand_score = optimize_h(best_h, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        # occasional random balanced binary try
        if rng.random() < 0.3 and time_left() > 0.2:
            h_rand = np.zeros(n)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_rand[idx] = 1.0
            sc_rand = compute_c5(h_rand)
            if sc_rand < best_score:
                best_score = sc_rand
                best_h = h_rand.copy()

    # final refinement on current size with remaining budget
    remaining = time_left()
    if remaining > 0.5:
        fine_h, fine_score = optimize_h(best_h, remaining, rng)
        if fine_score < best_score:
            best_h = fine_h
            best_score = fine_score

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
