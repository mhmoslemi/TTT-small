def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    if n >= 256:
        N = 2 * n
        a = h.astype(np.float64)
        b = (1.0 - h).astype(np.float64)
        fa = np.fft.rfft(a, n=N)
        fb = np.fft.rfft(b[::-1], n=N)
        c = np.fft.irfft(fa * fb, n=N)
        # correlation for shifts -n+1...n-1 corresponds to c[n-1:2*n-1]
        corr = c[n-1:2*n-1]
    else:
        corr = np.correlate(h, 1.0 - h, mode='full')
        # length = 2*n-1, shift indices from -(n-1) to n-1, correlation values are in order
        # So max is just max of this array.
    max_corr = np.max(corr)
    return max_corr * dx

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # compute current maximal correlation shifts
        corr = compute_corr_full(h)
        max_corr = np.max(corr)
        # find all shifts achieving max within tolerance
        # tolerance relative?
        eps = 1e-12
        shifts = np.where(corr >= max_corr - eps)[0] - (n-1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            # increase step size modestly
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        # occasional jitter to escape plateau
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_h, best_score = h.copy(), cur_score
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_corr_full(h):
    """Full correlation of h with (1-h) using FFT for speed when large."""
    n = h.shape[0]
    if n >= 256:
        N = 2 * n
        a = h.astype(np.float64)
        b = (1.0 - h).astype(np.float64)
        fa = np.fft.rfft(a, n=N)
        fb = np.fft.rfft(b[::-1], n=N)
        c = np.fft.irfft(fa * fb, n=N)
        return c[:2 * n - 1]
    else:
        return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    return np.max(compute_corr_full(h)) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum and adaptive step."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    eps = 1e-12
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = compute_corr_full(h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_h, best_score = h.copy(), cur_score
    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    """Transfer mass from a point that contributes most to the worst lag."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    eps = 1e-12
    while time_left() > 0:
        corr = compute_corr_full(best_h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Randomly transfer a small mass between two positions, accepting improvements."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # Move from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        # Move from j to i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if time_left() <= 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Gradient descent refinement
        max_iters = int(min(50000, max(5000, time_left() * (4000.0 / n))))
        base, cur_score = gradient_descent_adaptive(
            base, max_iters, step0=0.5, beta=0.9, rng=rng, time_left=time_left)
        if cur_score < best_score:
            best_h, best_score = base.copy(), cur_score

        # Targeted max‑shift improvement
        base, cur_score = improve_via_max_shift(base, rng, time_left)
        if cur_score < best_score:
            best_h, best_score = base.copy(), cur_score

        # Local pairwise moves
        base, cur_score = local_pair_moves(base, rng, time_left)
        if cur_score < best_score:
            best_h, best_score = base.copy(), cur_score

        # Random binary restarts on the final (largest) scale
        if n == scales[-1]:
            for _ in range(5):
                if time_left() < 10:
                    break
                h_rand = random_binary_balanced(n, rng)
                iters = int(min(5000, max(500, time_left() * (2000.0 / n))))
                h_rand, rand_score = gradient_descent_adaptive(
                    h_rand, iters, step0=0.5, beta=0.9, rng=rng, time_left=time_left)
                if rand_score < best_score:
                    best_h, best_score = h_rand.copy(), rand_score
                h_rand, rand_score = local_pair_moves(h_rand, rng, time_left)
                if rand_score < best_score:
                    best_h, best_score = h_rand.copy(), rand_score

        base = best_h.copy()

    # Final intensive refinement
    if time_left() > 5:
        max_iters = int(min(80000, max(20000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(
            best_h, max_iters, step0=0.3, beta=0.9, rng=rng, time_left=time_left)

    if time_left() > 2:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
