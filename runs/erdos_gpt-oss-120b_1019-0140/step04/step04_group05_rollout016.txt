import time

def project_onto_box_sum(v,total, lo=0.0, hi=1.0, max_iter=60):
    n=v.size
    if total <= lo*n:
        return np.full_like(v, lo)
    if total >= hi*n:
        return np.full_like(v, hi)
    # binary search lam
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = np.clip(v - lam_mid, lo, hi).sum()
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def smoothed_c5_and_grad(h, alpha):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    a_corr = alpha * corr
    max_a = np.max(a_corr)
    exp_vals = np.exp(a_corr - max_a)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    conv = np.convolve(weights, h, mode='full')
    conv_rev = np.convolve(weights[::-1], h, mode='full')
    start = n - 1
    grad = dx * (1.0 - (conv[start:start + n] + conv_rev[start:start + n]))
    obj = (np.log(sum_exp) + max_a) / alpha
    return obj, grad

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = np.clip(v - lam_mid, lo, hi).sum()
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    """Exact C5 for discrete h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def smoothed_c5_and_grad(h, alpha):
    """Softmax‑approximated C5 and its gradient."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    a = alpha * corr
    a_max = np.max(a)
    exp_vals = np.exp(a - a_max)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp                     # length 2n‑1
    conv = np.convolve(w, h, mode='full')
    conv_rev = np.convolve(w[::-1], h, mode='full')
    start = n - 1
    grad = dx * (1.0 - (conv[start:start + n] + conv_rev[start:start + n]))
    obj = (np.log(sum_exp) + a_max) / alpha
    return obj, grad

def random_balanced_binary(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h
