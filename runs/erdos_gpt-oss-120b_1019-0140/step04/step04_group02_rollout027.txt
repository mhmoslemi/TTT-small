def binary_hill_climb(h, best_score, time_left, rng, target):
    n = h.size
    # Ensure binary
    h = np.where(h >= 0.5, 1.0, 0.0)
    # Precompute indices
    zeros = np.flatnonzero(h == 0.0).tolist()
    ones = np.flatnonzero(h == 1.0).tolist()
    # Temperature schedule for simulated annealing (optional)
    while time_left() > 0.0:
        if not zeros or not ones:
            break
        # Random single swap
        i = rng.choice(zeros)
        j = rng.choice(ones)
        h[i] = 1.0
        h[j] = 0.0

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):

    s = n/2.0
    best_h = h.copy()

        if time_left() <= 0:
            break

        max_c = np.max(corr)
        # gradient for max correlation term - we compute gradient of max corr (equal to sum of contributions for max shifts)
        # The gradient of max_c term w.r.t. h. Derive:
        # corr[k] = sum_i h[i] * (1 - h[i+k]) where shift k index as described.
        # For each shift attaining max, we add gradient contributions:
        # For shift k >=0: contribution to grad[i] = (1 - h[i+k]) (if i in domain) - h[i-k] ?
        # derived earlier: For shift k >=0:
        #   For i from 0..n-k-1: term h[i] * (1 - h[i+k])
        #   gradient w.r.t. h[i] includes derivative of term: (1 - h[i+k]) + (-h[i]) * d/dh[i] of (1 - h[i+k])? Actually (1 - h[i+k]) depends on h[i+k].
        # This gradient was implemented earlier using loops.
        # Let's reuse compute_grad_max_shifts from previous code.
        idxes = np.where(corr >= max_c - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for k in shifts:
            if k >= 0:
                grad[:n - k] += 1.0 - h[k:]
                grad[k:] -= h[:n - k]
            else:
                m = -k
                grad[m:] += 1.0 - h[:n - m]
                grad[:n - m] -= h[m:]
        if len(shifts) > 0:
            grad /= len(shifts)

        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad

        eta = step0 / np.sqrt(it)

        # occasional random perturb

            h_new += rng.normal(scale=0.02, size=n)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2*n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        idxes = np.where(corr >= max_c - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for k in shifts:
            if k >= 0:
                grad[: n - k] += 1.0 - h[k:]
                grad[k:] -= h[: n - k]
            else:
                m = -k
                grad[m:] += 1.0 - h[: n - m]
                grad[: n - m] -= h[m:]
        if len(shifts) > 0:
            grad /= len(shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def hill_climb_binary(h, score, time_left, rng, target):
    """Greedy binary hill‑climbing with swap moves."""
    n = h.shape[0]
    zeros = np.flatnonzero(h == 0.0).tolist()
    ones = np.flatnonzero(h == 1.0).tolist()
    current_score = score
    best_h = h.copy()
    best_score = score
    swaps = 0
    max_swaps = int(min(20000, max(1000, time_left() * 5000)))
    no_improve = 0
    while swaps < max_swaps and time_left() > 0.0:
        if not zeros or not ones:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        h[i] = 1.0
        h[j] = 0.0
        sc_new = compute_c5_fft(h)
        swaps += 1
        if sc_new < current_score:
            current_score = sc_new
            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)
            ones.append(i)
            if sc_new < best_score:

                best_h = h.copy()
            no_improve = 0
            if best_score <= target:
                break
        else:
            h[i] = 0.0
            h[j] = 1.0
            no_improve += 1
            if no_improve >= 500:
                break
    return best_h, best_score
