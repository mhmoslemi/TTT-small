import time
import numpy as np
import cvxpy as cp

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    a = np.zeros(n, dtype=float)
    a[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return a


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            a = np.zeros(n, dtype=h.dtype)
            b = np.zeros(n, dtype=h.dtype)
            a[:n - shift] = h[shift:]
            b[shift:] = h[:n - shift]
        else:
            s = -shift
            a = np.zeros(n, dtype=h.dtype)
            b = np.zeros(n, dtype=h.dtype)
            a[s:] = h[:n - s]
            b[:n - s] = h[s:]
        grad += (1.0 - a - b)
    if len(shifts):
        grad /= len(shifts)
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with adaptive step size."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 3000 == 0 and time_left() > 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Transfer a small mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc = compute_c5(h_new)
            if sc < best_score:
                best_h, best_score = h_new, sc
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            sc2 = compute_c5(h_new2)
            if sc2 < best_score:
                best_h, best_score = h_new2, sc2
    return best_h, best_score


def lp_refine_once(h_old):
    """Single LP refinement step using linearized constraints."""
    n = h_old.size
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2]
    ho = np.asarray(h_old, dtype=float)

    for s in range(n):
        if s == 0:
            coeff = 1.0 - 2.0 * ho
            const = np.dot(ho, ho)
        else:
            coeff = np.ones(n)
            coeff[:n - s] -= ho[s:]
            coeff[s:] -= ho[:n - s]
            const = np.dot(ho[:n - s], ho[s:])
        constraints.append(coeff @ h <= t - const)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=200, warm_start=False)
    except Exception:
        prob.solve(solver=cp.ECOS, warm_start=False)

    if h.value is None:
        return h_old.copy()
    return np.array(h.value).flatten()


def lp_iterative_refine(h0, n, max_iters, time_left):
    """Iterate LP refinement several times."""
    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_sc = compute_c5(best_h)
    for _ in range(max_iters):
        if time_left() <= 0:
            break
        h_new = lp_refine_once(h_cur)
        h_new = project_onto_box_sum(h_new, n / 2.0)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_sc


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    for n in (128, 256, 512):
        if time_left() <= 0:
            break
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        # LP refinement
        lp_iters = int(min(150, max(30, time_left() // 5)))
        best_h, best_score = lp_iterative_refine(best_h, n, lp_iters, time_left)

        # Gradient descent fine‑tuning
        if time_left() > 5:
            gd_iters = int(min(15000, max(5000, time_left() * (1200.0 / n))))
            best_h, best_score = gradient_descent_adaptive(best_h, gd_iters,
                                                          step0=0.5, beta=0.9,
                                                          rng=rng, time_left=time_left)

        # Small local moves
        if time_left() > 2:
            best_h, best_score = local_pair_moves(best_h, rng, time_left)

        # Random balanced restarts (light)
        for _ in range(2):
            if time_left() < 5:
                break
            h_rand = random_balanced_binary(n, rng)
            h_rand = project_onto_box_sum(h_rand, n / 2.0)
            # quick LP then GD
            h_rand, sc_rand = lp_iterative_refine(h_rand, n, 20, time_left)
            if sc_rand < best_score:
                best_h, best_score = h_rand.copy(), sc_rand
            if time_left() > 5:
                gd_iters = int(min(5000, time_left() * 2))
                h_rand, sc_rand = gradient_descent_adaptive(h_rand, gd_iters,
                                                           step0=0.5, beta=0.9,
                                                           rng=rng, time_left=time_left)
                if sc_rand < best_score:
                    best_h, best_score = h_rand.copy(), sc_rand

    # Final intensive gradient refinement
    if time_left() > 5:
        final_iters = int(min(50000, max(20000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(best_h, final_iters,
                                                      step0=0.3, beta=0.9,
                                                      rng=rng, time_left=time_left)

    # One last round of local improvements
    if time_left() > 2:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
