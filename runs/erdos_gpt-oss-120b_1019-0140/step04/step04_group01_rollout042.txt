def binary_anneal(h_bin, rng, time_left, total_time, max_iters=None):
    n = len(h_bin)
    ones = list(np.flatnonzero(h_bin))
    zeros = list(np.flatnonzero(1 - h_bin))
    # initial correlation f using FFT:
    f = np.convolve(h_bin, h_bin[::-1], mode='full')[n-1:]
    # ensure f is int
    cur_min_f = f.min()
    cur_score = 1 - (2.0/n) * cur_min_f
    best_score = cur_score
    best_h = h_bin.copy()

    it = 0
    while time_left() > 0.05 and (max_iters is None or it < max_iters):
        # choose random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute distances for removal i
        # using np.abs(np.array(ones) - i)
        ones_arr = np.array(ones, dtype=int)
        distances_i = np.abs(ones_arr - i)
        # compute f after removal i
        f_temp = f.copy()
        # For each d in distances_i: decrement
        np.subtract.at(f_temp, distances_i, 1)  # decrement by 1
        # Also need to remove i from ones_arr for later distances; but we already decrement self-pair.
        # Now compute distances for addition j using ones after removal:
        # Remove i from ones list temporarily
        idx_i = ones_arr == i
        # to get ones without i:
        ones_without_i_arr = ones_arr[~idx_i]
        distances_j = np.abs(ones_without_i_arr - j)
        np.add.at(f_temp, distances_j, 1)
        # increment self-pair for j:
        f_temp[0] += 1
        new_min_f = f_temp.min()
        new_score = 1 - (2.0/n) * new_min_f
        delta = new_score - cur_score
        # temperature
        fraction = time_left() / total_time
        T = max(1e-8, 0.5 * fraction)  # simple linear cooling
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta/T):
                accept = True
        if accept:
            # commit changes
            f = f_temp
            cur_min_f = new_min_f
            cur_score = new_score
            # update ones and zeros lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            # update h_bin
            h_bin[i] = 0
            h_bin[j] = 1
            if new_score < best_score:
                best_score = new_score
                best_h = h_bin.copy()
        # else reject, do nothing (f, ones unchanged)
        it += 1
    return best_h, best_score

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        # compute correlation and max
        corr = compute_corr_scaled_fft(h)  # already scaled by dx
        max_corr = np.max(corr)
        idxes = np.where(np.abs(corr - max_corr) <= 1e-12)[0]
        # convert idxes to shifts
        shifts = idxes - (n - 1)
        # compute correct gradient
        grad = np.zeros_like(h)
        for sft in shifts:
            if sft >= 0:
                # forward term
                grad[:n - sft] += (1 - h[sft:])
                # backward term
                grad[sft:] -= h[:n - sft]

    n = len(h_bin)
    # f compute via convolution

    f = f.astype(np.int64)  # ensure int
    cur_min_f = f.min()

    best_h = h_bin.copy()

    ones = list(np.flatnonzero(h_bin))
    zeros = list(np.flatnonzero(1 - h_bin))
    init_temp = 0.5
    min_temp = 1e-6
    total_time_remaining = time_left()  # capture at start
    # ensure not zero
    while time_left() > 0.01:
        # random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute temporary f using copy
        f_temp = f.copy()
        # distances for removal i

        distances_i = np.abs(ones_arr - i)

        # add self-pair for j

        accept = False

            accept = True

        if accept:

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure the bracket contains the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def thue_morse(n):
    """Return Thue‑Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    """Projected momentum gradient descent using correct subgradient."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(np.abs(corr - max_corr) <= 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for sh in shifts:
            if sh >= 0:
                grad[:n - sh] += (1 - h[sh:])
                grad[sh:]    -= h[:n - sh]
            else:
                k = -sh
                grad[k:]    += (1 - h[:n - k])
                grad[:n - k] -= h[k:]
        if len(shifts) > 0:
            grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def c5_from_binary(b):
    """C5 for a binary vector using exact integer autocorrelation."""
    n = b.size
    f = np.convolve(b, b[::-1], mode='full')[n - 1:].astype(np.int64)
    min_f = f.min()
    return 1.0 - (2.0 / n) * min_f

def binary_anneal(h_bin, rng, time_left):
    """Simulated annealing on binary vectors with O(n) updates."""
    n = h_bin.size
    f = np.convolve(h_bin, h_bin[::-1], mode='full')[n - 1:].astype(np.int64)
    cur_min_f = f.min()
    cur_score = 1.0 - (2.0 / n) * cur_min_f
    best_h = h_bin.copy()
    best_score = cur_score
    ones = list(np.flatnonzero(h_bin))
    zeros = list(np.flatnonzero(1 - h_bin))
    init_temp = 0.5
    min_temp = 1e-6
    total_rem = time_left()
    while time_left() > 0.01:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # temporary copy of f
        f_tmp = f.copy()
        ones_arr = np.array(ones, dtype=np.int64)
        # remove i
        d_i = np.abs(ones_arr - i)
        np.subtract.at(f_tmp, d_i, 1)
        # add j (use ones without i)
        mask = ones_arr != i
        ones_wo_i = ones_arr[mask]
        d_j = np.abs(ones_wo_i - j)
        np.add.at(f_tmp, d_j, 1)
        f_tmp[0] += 1  # self‑pair for j
        new_min_f = f_tmp.min()
        new_score = 1.0 - (2.0 / n) * new_min_f
        delta = new_score - cur_score
        fraction = time_left() / total_rem
        T = max(min_temp, init_temp * fraction)
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):

        if accept:
            f = f_tmp
            cur_min_f, cur_score = new_min_f, new_score
            h_bin[i] = 0
            h_bin[j] = 1
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if new_score < best_score:

    return best_h, best_score

        return max(0.0, budget_s - (time.time() - start))

    # Determine problem size

        h0 = project_onto_box_sum(init_arr, n / 2.0)
    else:
        n = 512
        h0 = random_binary_balanced(n, rng)

    # ---------- Phase 1: continuous refinement ----------
    phase1_budget = budget_s * 0.35
    phase1_start = time.time()
    def time_left_phase1():
        return max(0.0, phase1_budget - (time.time() - phase1_start))

    candidates = [h0,
                  project_onto_box_sum(thue_morse(n).astype(float), n / 2.0),
                  project_onto_box_sum(random_binary_balanced(n, rng), n / 2.0)]

    best_h = None
    best_score = np.inf
    target = 0.3808

    for cand in candidates:
        if time_left_phase1() <= 0.0:
            break
        cand = project_onto_box_sum(cand, n / 2.0)
        h_opt, sc_opt = gradient_descent_momentum(cand, time_left_phase1,
                                                  step0=0.5, beta=0.9,
                                                  max_iter=200000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, n

    # If continuous phase failed to produce a candidate, fall back to h0
    if best_h is None:
        best_h = h0.copy()
        best_score = compute_c5_fft(best_h)

    # ---------- Phase 2: binary conversion ----------
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros(n, dtype=int)
    h_bin[idx] = 1
    bin_score = c5_from_binary(h_bin)
    if bin_score < best_score:
        best_score, best_h = bin_score, h_bin.astype(float).copy()
    if best_score <= target:
        return best_h, best_score, n

    # ---------- Phase 3: binary annealing ----------
    remaining_time = lambda: max(0.0, budget_s - (time.time() - start))
    if remaining_time() > 0.2:
        h_bin_best, sc_bin_best = binary_anneal(h_bin.copy(), rng, remaining_time)
        if sc_bin_best < best_score:
            best_score, best_h = sc_bin_best, h_bin_best.astype(float).copy()
        if best_score <= target:
            return best_h, best_score, n

    # ---------- Phase 4: final projection (safety) ----------
    best_h = project_onto_box_sum(best_h, n / 2.0)
    final_score = compute_c5_fft(best_h) if best_h.dtype != np.int64 else c5_from_binary(best_h.astype(int))
    if final_score < best_score:
        best_score = final_score
    return best_h, best_score, n
