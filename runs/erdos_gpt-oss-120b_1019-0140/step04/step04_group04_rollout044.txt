def adam_optimize(h0, max_iter, lr, beta1, beta2, eps, time_left):

    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        # compute gradient for max shifts
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # project onto box+sum
        h_new = project_onto_box_sum(h_new, h.size/2.)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        # optionally occasional random jumps:
        if it % 2000 == 0:
            # add noise
            noise = np.random.default_rng().normal(scale=0.02, size=h.size)

            sc_noise = compute_c5(h_noise)
            if sc_noise < best_score:
                best_score, best_h = sc_noise, h_noise.copy()
        h = h_new

    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_swap_local_search(h, max_iter, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = h.size
    # Precompute indices of ones and zeros
    for _ in range(max_iter):
        if time_left() <= 0:
            break
        # get indices of ones and zeros
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h_new = best_h.copy()
        h_new[i] = 0.0

        sc = compute_c5(h_new)
        if sc < best_score:
            best_h = h_new
            best_score = sc
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=30):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # initial bounds for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # f(lam) = sum clip(v - lam, lo, hi)
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        clipped = v - lam_mid
        clipped = np.clip(clipped, lo, hi)
        total = clipped.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    # final
    return np.clip(v - lam_mid, lo, hi)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto [lo,hi]^n with sum equal to s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        clipped = np.clip(v - lam_mid, lo, hi)
        total = clipped.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx, discretized."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Generate Thue-Morse sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    arr[idx] = 1.0
    return arr

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max correlation w.r.t h, averaged over given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def adam_optimize(h0, max_iter, lr, beta1, beta2, eps, rng, time_left):
    """Adam-style optimization of C5."""
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        # occasional noise to escape plateaus
        if it % 2000 == 0:
            noise = rng.normal(scale=0.02, size=n)
            h_noise = h_new + noise
            h_noise = project_onto_box_sum(h_noise, s_target)
            sc_noise = compute_c5(h_noise)
            if sc_noise < best_score:
                best_score, best_h = sc_noise, h_noise.copy()
        h = h_new
    return best_h, best_score

def perturb_and_project(h, scale, rng, s):
    """Add Gaussian noise and re‑project."""
    h_pert = h + rng.normal(scale=scale, size=h.shape)
    return project_onto_box_sum(h_pert, s)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # --- initialization ----------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    best_h = None
    best_score = np.inf

    # --- multi‑resolution schedule ----------------------------------------
    target_sizes = [64, 128, 256, 512, 1024]
    lr_map = {64: 0.5, 128: 0.4, 256: 0.3, 512: 0.2, 1024: 0.15}
    iters_map = {64: 5000, 128: 6000, 256: 8000, 512: 12000, 1024: 20000}

    for n in target_sizes:
        if time_left() <= 5.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        sc = compute_c5(base)
        if sc < best_score:
            best_h, best_score = base.copy(), sc

        # allocate iterations respecting remaining time
        max_it = iters_map[n]
        possible_it = int(min(max_it, time_left() * 30))
        if possible_it > 0:
            h_opt, sc_opt = adam_optimize(base,
                                          max_iter=possible_it,
                                          lr=lr_map[n],
                                          beta1=0.9,
                                          beta2=0.999,
                                          eps=1e-8,
                                          rng=rng,
                                          time_left=time_left)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
            base = h_opt

        # extra random restarts at the finest level
        if n == target_sizes[-1] and time_left() > 5.0:
            for _ in range(3):
                if time_left() <= 5.0:
                    break
                h_rand = random_binary_balanced(n, rng)
                it_rand = int(min(2000, time_left() * 10))
                h_rand_opt, sc_rand_opt = adam_optimize(h_rand,
                                                        max_iter=it_rand,
                                                        lr=lr_map[n],
                                                        beta1=0.9,
                                                        beta2=0.999,
                                                        eps=1e-8,
                                                        rng=rng,
                                                        time_left=time_left)
                if sc_rand_opt < best_score:
                    best_h, best_score = h_rand_opt.copy(), sc_rand_opt

    # --- final refinement --------------------------------------------------
    if best_h is not None and time_left() > 2.0:
        n_final = best_h.size
        it_final = int(min(50000, time_left() * 30))
        best_h, best_score = adam_optimize(best_h,
                                           max_iter=it_final,
                                           lr=0.05,
                                           beta1=0.9,
                                           beta2=0.999,
                                           eps=1e-8,
                                           rng=rng,
                                           time_left=time_left)

    # --- optional light perturbations if time remains ----------------------
    while time_left() > 1.5:
        n_cur = best_h.size
        h_pert = perturb_and_project(best_h, scale=0.04, rng=rng, s=n_cur / 2.0)
        sc_pert = compute_c5(h_pert)
        if sc_pert < best_score:
            best_h, best_score = h_pert.copy(), sc_pert
        else:
            break

    # --- final projection & return -----------------------------------------
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, float(best_score), int(best_h.size)
