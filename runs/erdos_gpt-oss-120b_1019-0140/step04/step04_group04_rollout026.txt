def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    # v: np array
    # s: desired sum
    # Return projection onto [lo, hi] with sum s.
    n = v.size
    # Clip extremes?
    # If s <= lo*n, return all lo; if s >= hi*n, return all hi.
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # lambda binary search.
    # The function f(lam) = sum(clip(v - lam, lo, hi)).
    # Since f is decreasing in lam.
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    # Ensure f(lam_lo) >= s >= f(lam_hi)
    # Actually f(lam_lo) = sum(clip(v - lam_lo, lo, hi)) = sum(clip(v - min(v)+hi, lo, hi)) = sum(hi) = hi*n -> >= s
    # f(lam_hi) = sum(lo) = lo*n -> <= s
    # So set lam_lo = np.min(v)-hi, lam_hi = np.max(v)-lo.
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fmid = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fmid - s) < 1e-12:
            break
        if fmid > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import numpy as np
import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fmid = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fmid - s) < 1e-12:
            break
        if fmid > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # generate 0/1 pattern of length n
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def improve_binary(h_bin, rng, time_left):

            best_score = sc
            best = h_new
        # limit number of attempts to avoid infinite loop
        # break if no improvement after several attempts
        # we rely on time limit
    return best, best_score

import numpy as np
import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fmid = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fmid - s) < 1e-12:
            break
        if fmid > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def improve_binary(h_bin, rng, time_left):
    best = h_bin.copy()
    best_score = compute_c5(best)
    while time_left() > 0.2:
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best = h_new
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    n = 512  # final resolution

    # base construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = upsample(base, n)
    base = project_onto_box_sum(base, n / 2.0)

    # seed with a few random binary vectors
    best_h = base.copy()
    best_score = compute_c5(best_h)
    for _ in range(10):
        if time_left() <= 0:
            break
        h_rand = random_binary_balanced(n, rng)
        sc = compute_c5(h_rand)
        if sc < best_score:
            best_score = sc
            best_h = h_rand.copy()

    # smooth‑max gradient descent
    max_iters = int(min(60000, time_left() * 50))
    beta = 0.9
    momentum = np.zeros_like(best_h)
    step0 = 0.5
    alpha_start = 5.0
    alpha_end = 200.0
    top_k = 50

    h = best_h.copy()
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max_iters
        max_corr = np.max(corr)
        shift_corr = corr - max_corr
        exps = np.exp(alpha * shift_corr)
        w = exps / np.sum(exps)

        # only the most influential shifts
        if top_k < len(w):
            idx_top = np.argpartition(w, -top_k)[-top_k:]
        else:
            idx_top = np.arange(len(w))

        grad = np.zeros_like(h)
        for idx in idx_top:
            ws = w[idx]
            if ws == 0:
                continue
            s = idx - (n - 1)
            if s >= 0:
                grad[:n - s] -= ws * h[s:]
                grad[s:] -= ws * h[:n - s]
            else:
                sneg = -s
                grad[sneg:] -= ws * h[:n - sneg]
                grad[:n - sneg] -= ws * h[sneg:]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / np.sqrt(it)
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, n / 2.0)

        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            if best_score <= 0.3808:
                h = h_new
                break
        if it % 5000 == 0:
            pert = rng.normal(scale=0.02, size=n)
            h_pert = project_onto_box_sum(h_new + pert, n / 2.0)
            sc_pert = compute_c5(h_pert)
            if sc_pert < best_score:
                best_score = sc_pert
                best_h = h_pert.copy()
                h_new = h_pert
        h = h_new

    # final binary rounding and local swap improvement
    if time_left() > 0:
        idx_top = np.argpartition(best_h, -int(n / 2))[-int(n / 2):]
        h_bin = np.zeros_like(best_h)
        h_bin[idx_top] = 1.0
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_h = h_bin
            best_score = sc_bin
        if time_left() > 2:
            h_bin_imp, sc_imp = improve_binary(best_h, rng, time_left)
            if sc_imp < best_score:
                best_h = h_bin_imp
                best_score = sc_imp

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
