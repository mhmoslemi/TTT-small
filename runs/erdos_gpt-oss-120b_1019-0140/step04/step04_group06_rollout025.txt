def smoothed_gradient(h, alpha):
   n = h.size
   corr = np.correlate(h, 1.0 - h, mode='full')
   max_corr = np.max(corr)
   # compute weights normalized
   # to avoid overflow: w = np.exp(alpha * (corr - max_corr))
   w = np.exp(alpha * (corr - max_corr))
   w_sum = w.sum()
   w /= w_sum
   sum_shift = np.zeros(n, dtype=h.dtype)
   # fill contributions
   # shift index from -(n-1) to (n-1)
   offset = n - 1
   for idx, weight in enumerate(w):
       if weight == 0.0:
           continue
       k = idx - offset
       if k >= 0:
           # shift_plus: h[k:] added to sum_shift[:n-k]
           sum_shift[:n - k] += weight * h[k:]
           # shift_minus: h[:n - k] added to sum_shift[k:]

           sum_shift[:n - s] += weight * h[s:]

def smooth_grad_descent(h0, n_iter, step0, beta, rng, alpha_start, alpha_end, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        frac = it / n_iter
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        grad = smoothed_gradient(h, alpha)
        momentum = beta * momentum + (1-beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def smoothed_gradient(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    w = np.exp(alpha * (corr - maxc))
    w /= w.sum()
    # weighted sum via convolution (zero padding)
    weighted = np.convolve(w[::-1], h, mode='full')[n - 1:n - 1 + n]
    grad = 1.0 - 2.0 * weighted
    return grad

def smooth_grad_descent(h0, n_iter, step0, beta, rng, alpha_start, alpha_end, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        frac = it / n_iter
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        grad = smoothed_gradient(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def hard_grad_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_anneal(h0, score0, rng, time_left):
    h = h0.copy()
    best_h, best_score = h.copy(), score0
    cur_score = score0
    start = time.time()
    T0, T1 = 1e-2, 1e-5
    while time_left() > 0:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed)
        T = T0 * (T1 / T0) ** frac
        i, j = rng.choice(h.size, size=2, replace=False)
        delta = min(h[i], 1.0 - h[j]) * rng.uniform(0.05, 0.4)
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            h, cur_score = h_new, sc_new
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
        if rng.random() < 0.05:
            grad = compute_gradients_for_shifts(h, max_shifts(h))
            step = 0.02
            h_tmp = h - step * grad
            h_tmp = project_onto_box_sum(h_tmp, h.size / 2.0)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < cur_score:
                h, cur_score = h_tmp, sc_tmp
                if cur_score < best_score:
                    best_score, best_h = cur_score, h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    scales = [64, 128, 256, 512]
    best_h, best_score = None, float('inf')

    for n in scales:
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        if time_left() > 5:
            max_iter = int(min(50000, max(8000, time_left() * (2000.0 / n))))
            h_s, sc_s = smooth_grad_descent(base, max_iter, step0=0.8,
                                            beta=0.9, rng=rng,
                                            alpha_start=5.0, alpha_end=200.0,
                                            time_left=time_left)
            if sc_s < best_score:
                best_score, best_h = sc_s, h_s.copy()
            base = h_s

        if time_left() > 5:
            hard_iter = max_iter // 2
            h_h, sc_h = hard_grad_descent(base, hard_iter, step0=0.5,
                                          beta=0.9, rng=rng, time_left=time_left)
            if sc_h < best_score:
                best_score, best_h = sc_h, h_h.copy()
            base = h_h

        if time_left() > 5:
            h_a, sc_a = simulated_anneal(base, compute_c5(base), rng, time_left)
            if sc_a < best_score:
                best_score, best_h = sc_a, h_a.copy()
            base = h_a

    # Final random restarts at the finest scale
    n_final = scales[-1]
    while time_left() > 8:
        h_rand = random_binary_balanced(n_final, rng)
        h_rand = project_onto_box_sum(h_rand, n_final / 2.0)
        iters = int(min(12000, max(2000, time_left() * 5)))
        h_rand, sc_rand = smooth_grad_descent(h_rand, iters, step0=0.6,
                                              beta=0.9, rng=rng,
                                              alpha_start=10.0, alpha_end=300.0,
                                              time_left=time_left)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()

        h_h2, sc_h2 = hard_grad_descent(best_h, 3000, step0=0.4,
                                        beta=0.9, rng=rng, time_left=time_left)
        if sc_h2 < best_score:
            best_score, best_h = sc_h2, h_h2.copy()

        if best_score <= 0.3808:
            break

    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(scales[-1]), scales[-1] / 2.0)
        best_score = compute_c5(best_h)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
