def project_onto_box_sum(v, s, lo=0., hi=1., max_iter=100):
    # Ensure s within feasible range
    n = v.shape[0]
    lo_sum = lo * n
    hi_sum = hi * n
    if s <= lo_sum:
        return np.full_like(v, lo)
    if s >= hi_sum:
        return np.full_like(v, hi)
    # lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for i in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        x = np.clip(v - lam_mid, lo, hi)
        cur = x.sum()
        if abs(cur - s) < 1e-12:
            return x
        if cur > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    # final
    return np.clip(v - (lam_low+lam_high)/2.0, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2*n-1).bit_length())
    # compute correlation of h and (1 - h) via FFT
    # linear correlation: conv = irfft(rfft(h_rev, N) * rfft(1-h, N), N)[:2*n-1]
    # but we need correlation, not convolution? correlation of h and (1-h) = conv of h_rev and (1-h).
    # So choose h_rev = h[::-1]
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n-1]
    # Multiply by dx = 2.0 / n
    dx = 2.0 / n
    return np.max(corr) * dx

   m = np.zeros_like(h)
   v = np.zeros_like(h)
   last_improve_iter = 0

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    lo_sum = lo * n
    hi_sum = hi * n
    if s <= lo_sum:
        return np.full_like(v, lo)
    if s >= hi_sum:
        return np.full_like(v, hi)
    # get bounds for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # ensure f(lam_low) >= s and f(lam_high) <= s
    # using monotonic decreasing f(l)
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam_mid, lo, hi)
        cur = x.sum()
        if abs(cur - s) < 1e-12:
            return x
        if cur > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - (lam_low + lam_high) * 0.5, lo, hi)

def compute_corr_vals(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    h_rev = h[::-1]

    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    return conv[:2 * n - 1]

Compute_c5_fast:

def compute_c5_fast(h):
    n = h.shape[0]
    corr = compute_corr_vals(h)
    return np.max(corr) * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    grad = np.ones_like(h)
    if shift >= 0:
        sz = n - shift
        grad[:sz] -= h[shift:]
        grad[shift:] -= h[:sz]
    else:
        s = -shift
        sz = n - s
        grad[s:] -= h[:sz]
        grad[:sz] -= h[s:]
    return grad

Compute_gradient_max_shift:

def compute_gradient_max_shift(h):
    n = h.shape[0]
    corr = compute_corr_vals(h)
    max_val = corr.max()
    idxes = np.where(corr == max_val)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

Compute_gradient_softmax:

def compute_gradient_softmax(h, alpha):
    n = h.shape[0]
    corr = compute_corr_vals(h)
    max_val = corr.max()
    exp_weights = np.exp(alpha * (corr - max_val))
    w = exp_weights / exp_weights.sum()
    shifts = np.arange(len(corr)) - (n - 1)
    grad = np.zeros_like(h)
    for wi, s in zip(w, shifts):
        if wi > 0:
            grad += wi * gradient_of_shift(h, s)
    return grad

def binary_swap_anneal(h0, rng, time_left, max_iters=200000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    # Precompute indices of ones and zeros
    it = 0
    while it < max_iters and time_left() > 0.0:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def adam_optimize(h0, target_sum, time_left, rng, max_iters=5000, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8, alpha_soft=15.0, stagnation_patience=200, noise_scale=0.02):
    n = h0.size
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    last_improve = 0
    for t in range(1, max_iters + 1):
        if time_left() < 0.5:
            break
        grad = compute_gradient_max_shift(h)
        if np.linalg.norm(grad) < 1e-12:
            grad = compute_gradient_softmax(h, alpha_soft)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h - step, target_sum)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new.copy()
            last_improve = t
            if best_s <= 0.3808:
                return best_h, best_s
        if t - last_improve > stagnation_patience:
            # add small random perturbation
            h_new = project_onto_box_sum(h_new + rng.normal(scale=noise_scale, size=n), target_sum)
            last_improve = t
        h = h_new
    return best_h, best_s

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    lo_sum = lo * n
    hi_sum = hi * n
    if s <= lo_sum:
        return np.full_like(v, lo)
    if s >= hi_sum:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam_mid, lo, hi)
        cur = x.sum()
        if abs(cur - s) < 1e-12:
            return x
        if cur > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - (lam_low + lam_high) * 0.5, lo, hi)

def compute_corr_vals(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    return conv[:2 * n - 1]

def compute_c5_fast(h):
    n = h.shape[0]
    corr = compute_corr_vals(h)
    return np.max(corr) * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    grad = np.ones_like(h)
    if shift >= 0:
        sz = n - shift
        grad[:sz] -= h[shift:]
        grad[shift:] -= h[:sz]
    else:
        s = -shift
        sz = n - s
        grad[s:] -= h[:sz]
        grad[:sz] -= h[s:]
    return grad

def compute_gradient_max_shift(h):
    n = h.shape[0]
    corr = compute_corr_vals(h)
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def compute_gradient_softmax(h, alpha):
    n = h.shape[0]
    corr = compute_corr_vals(h)
    mx = corr.max()
    w = np.exp(alpha * (corr - mx))
    w /= w.sum()
    shifts = np.arange(len(corr)) - (n - 1)
    grad = np.zeros_like(h)
    for wi, s in zip(w, shifts):
        if wi:
            grad += wi * gradient_of_shift(h, s)
    return grad

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_swap_anneal(h0, rng, time_left, max_iters=150000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    while it < max_iters and time_left() > 0.0:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

def adam_optimize(h0, target_sum, time_left, rng,
                  max_iters=5000, lr=0.05, beta1=0.9, beta2=0.999,
                  eps=1e-8, alpha_soft=15.0,
                  stagnation_patience=200, noise_scale=0.02):
    n = h0.size
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    last_improve = 0
    for t in range(1, max_iters + 1):
        if time_left() < 0.5:
            break
        grad = compute_gradient_max_shift(h)
        if np.linalg.norm(grad) < 1e-12:
            grad = compute_gradient_softmax(h, alpha_soft)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h - step, target_sum)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
            last_improve = t
            if best_s <= 0.3808:
                return best_h, best_s
        if t - last_improve > stagnation_patience:
            h_new = project_onto_box_sum(
                h_new + rng.normal(scale=noise_scale, size=n), target_sum)
            last_improve = t
        h = h_new
    return best_h, best_s
