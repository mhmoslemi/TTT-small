def binary_swap_anneal_fast(h0, rng, time_left_func, max_iters=None, T0=0.2, decay=0.9999, Tmin=1e-8):
    n = h0.shape[0]
    h = h0.copy()
    # compute current corr
    corr = compute_corr_scaled_fft(h)
    score = np.max(corr)
    best_h = h.copy()
    best_score = score

    # precompute arrays for indices
    k = np.arange(-(n-1), n, dtype=np.int32)
    zero_idx = n - 1
    # track indices of ones and zeros
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]

    T = T0
    it = 0
    if max_iters is None:
        max_iters = int(1e9)  # effectively unlimited, stop by time

    # preallocate delta array
    delta = np.empty(2*n-1, dtype=np.float64)

    while it < max_iters and time_left_func() > 0:
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2*n-1]
    corr *= 2.0 / n
    return corr

def compute_c5_fft(h):
    return np.max(compute_corr_scaled_fft(h))

def thue_morse(n):

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def gradient_descent_momentum_time_limited(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # get shifts where corr is max (including near ties)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient for these shifts
        grad = np.zeros_like(h)
        # compute contributions for each shift efficiently?
        # Use the previous compute_gradients_for_shifts function but reimplement here to avoid dependency.
        for shift in shifts:
            if shift >= 0:
                # term1: h[i+shift] from i=0..n-1-shift
                term1 = np.zeros(n, dtype=h.dtype)
                term1[:n - shift] = h[shift:]
                term2 = np.zeros(n, dtype=h.dtype)
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros(n, dtype=h.dtype)
                term1[sft:] = h[:n - sft]
                term2 = np.zeros(n, dtype=h.dtype)
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal_fast(h0, rng, time_left_func, max_iters=None, T0=0.2, decay=0.9999, Tmin=1e-8):
    n = h0.shape[0]
    h = h0.astype(float).copy()
    corr = compute_corr_scaled_fft(h)
    score = np.max(corr)
    best_h = h.copy()
    best_score = score
    zero_idx = n - 1
    # maintain lists of positions
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    T = T0
    it = 0
    if max_iters is None:
        max_iters = 10**12
    # preallocate delta
    delta = np.empty(2 * n - 1, dtype=np.float64)
    while it < max_iters and time_left_func() > 0.0:
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        delta.fill(0.0)
        # precompute reversed h for term2
        h_rev = h[::-1]
        # apply for i: remove 1 => delta_h = -1
        # term1 slice
        start1 = zero_idx - i
        end1 = zero_idx + (n - 1 - i)
        if n - i > 0:
            delta[start1:end1+1] += -1.0 * (1.0 - h[0:n - i])
        # term2 slice
        start2 = i
        end2 = n - 1 + i
        delta[start2:end2+1] += +1.0 * h_rev[0:n]  # note -delta_h = +1
        # apply for j: add 1 => delta_h = +1
        # term1 slice
        start1 = zero_idx - j
        end1 = zero_idx + (n - 1 - j)
        if n - j > 0:
            delta[start1:end1+1] += 1.0 * (1.0 - h[0:n - j])
        # term2 slice
        start2 = j
        end2 = n - 1 + j
        delta[start2:end2+1] += -1.0 * h_rev[0:n]  # -delta_h = -1
        # zero out middle
        delta[zero_idx] = 0.0
        new_score = np.max(corr + delta)
        accept = False
        if new_score < score:

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    """Full correlation h * (1-h) scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    corr *= 2.0 / n
    return corr

def compute_c5_fft(h):
    """Maximum overlap integral."""
    return np.max(compute_corr_scaled_fft(h))

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=np.float64, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    return grad

def gradient_descent_momentum_time_limited(h0, time_left_func,
                                          step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        if shifts.size:
            grad /= shifts.size
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:

        h = h_new
        it += 1

def binary_swap_anneal_fast(h0, rng, time_left_func,
                           max_iters=None, T0=0.2, decay=0.9999, Tmin=1e-8):
    n = h0.shape[0]
    h = h0.astype(float).copy()
    corr = compute_corr_scaled_fft(h)
    score = np.max(corr)
    best_h = h.copy()
    best_score = score
    zero_idx = n - 1
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    T = T0
    it = 0
    if max_iters is None:
        max_iters = 10**12
    delta = np.empty(2 * n - 1, dtype=np.float64)
    while it < max_iters and time_left_func() > 0.0:
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        delta.fill(0.0)
        h_rev = h[::-1]

        # i -> 0  (Δh = -1)
        start1 = zero_idx - i
        end1 = zero_idx + (n - 1 - i)
        delta[start1:end1 + 1] += -1.0 * (1.0 - h[0:n - i])
        start2 = i
        end2 = n - 1 + i
        delta[start2:end2 + 1] += +1.0 * h_rev[0:n]

        # j -> 1  (Δh = +1)
        start1 = zero_idx - j
        end1 = zero_idx + (n - 1 - j)
        delta[start1:end1 + 1] += +1.0 * (1.0 - h[0:n - j])
        start2 = j
        end2 = n - 1 + j
        delta[start2:end2 + 1] += -1.0 * h_rev[0:n]

        delta[zero_idx] = 0.0
        new_score = np.max(corr + delta)

        accept = False
        if new_score < score:
            accept = True
        elif T > Tmin and rng.random() < np.exp(-(new_score - score) / T):
            accept = True

        if accept:
            h[i] = 0.0
            h[j] = 1.0
            ones[i_idx] = j
            zeros[j_idx] = i
            corr += delta
            score = new_score
            if score < best_score:
                best_score = score
                best_h = h.copy()
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
