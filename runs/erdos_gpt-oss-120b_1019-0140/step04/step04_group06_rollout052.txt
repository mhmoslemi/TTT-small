    n = h.size
    cur_score = compute_c5(h)
    # maintain lists
    ones = np.flatnonzero(h == 1)

        i = rng.choice(ones)

        h[i], h[j] = 0.0, 1.0

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

def hill_climb_binary(h, rng, time_left, max_iter=5000):
    n = h.size
    cur_score = compute_c5(h)
    # maintain list of indices
    ones = [int(i) for i in np.flatnonzero(h)]
    zeros = [int(i) for i in np.flatnonzero(1 - h)]
    it = 0
    while it < max_iter and time_left() > 0:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            # update lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
        it += 1
        # occasional random shuffle to diversify
        if it % 1000 == 0 and rng.random() < 0.05:

            for _ in range(5):
                i = rng.choice(ones)
                j = rng.choice(zeros)

                    zeros.append(i)
                    zeros.remove(j)
                    ones.append(j)

def continuous_optimize_slsqp(h0, rng, time_left, max_iter=1000):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
    # options
    options = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    res = opt.minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options=options)
    h_opt = project_onto_box_sum(res.x, n/2)
    return h_opt, compute_c5(h_opt)

def grad_descent_subgrad(h0, rng, time_left, max_iter=5000):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        # compute gradient of max correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        idxs = np.where(corr == max_c)[0]
        shifts = idxs - (n-1)
        grad = np.zeros_like(h)
        for k in shifts:
            # for each index i
            # compute contributions vectorized
            # shift positive
            if k >= 0:
                # handle i where i+k < n
                i_range = np.arange(n - k)
                grad[i_range] += 1.0 - h[i_range + k] - np.concatenate([np.zeros(k), np.zeros(0)]) # need to also subtract h[i-k] but i-k negative for i < k
                # Actually for k>=0, i-k >=0 when i>=k
                # So handle separately
                # Subtract h[i+k] for i in range(0,n-k)
                # Subtract h[i-k] for i in range(k,n)
                grad[:n - k] -= h[k:]   # subtract h[i+k]
                grad[k:] -= h[:n - k]   # subtract h[i-k]
                # add 1 for all i
                grad += 1.0
            else:
                s = -k
                # k negative => i+k = i - s
                # For i >= s: i - s in range
                grad[s:] -= h[:n - s]    # subtract h[i+k] = h[i-s]
                # i - k = i + s for i < n - s
                grad[:n - s] -= h[s:]    # subtract h[i - k] = h[i + s]
                grad += 1.0
        grad /= len(shifts)
        # step size schedule
        step = 0.5 / np.sqrt(it)
        h_new = h - step * grad
        # project onto box-sum
        h_new = project_onto_box_sum(h_new, target_sum)

        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()

import time, math, numpy as np, scipy.optimize as opt

# ---------- helper functions ----------
def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Clamp v to [lo,hi] while fixing sum(v)=s (s must be feasible)."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def hill_climb_binary(h, rng, time_left, max_iter=5000):
    """Swap a 1 with a 0 if it reduces C5."""
    n = h.size
    cur_score = compute_c5(h)
    ones = [int(i) for i in np.flatnonzero(h)]
    zeros = [int(i) for i in np.flatnonzero(1.0 - h)]
    it = 0
    while it < max_iter and time_left() > 0:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
        else:
            h[i], h[j] = 1.0, 0.0
        it += 1
        if it % 1000 == 0 and rng.random() < 0.05:
            # occasional tiny diversification
            for _ in range(5):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i], h[j] = 0.0, 1.0
                tmp = compute_c5(h)
                if tmp < cur_score:
                    cur_score = tmp
                    ones.remove(i)
                    zeros.append(i)
                    zeros.remove(j)
                    ones.append(j)
                else:
                    h[i], h[j] = 1.0, 0.0
    return h, cur_score


def grad_descent_subgrad(h0, rng, time_left, max_iter=3000):
    """Projected subgradient descent on max_{k} C5 using the subgradient of the active shifts."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        idxs = np.where(corr == max_c)[0]
        shifts = idxs - (n - 1)
        grad = np.zeros_like(h)
        for k in shifts:
            grad += 1.0
            if k >= 0:
                grad[: n - k] -= h[k:]
                grad[k:] -= h[: n - k]
            else:
                s = -k
                grad[s:] -= h[: n - s]
                grad[: n - s] -= h[s:]
        grad /= len(shifts)
        step = 0.5 / math.sqrt(it)
        h = project_onto_box_sum(h - step * grad, target)
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h.copy()
    return best_h, best_score


def continuous_optimize_slsqp(h0, rng, time_left, max_iter=400):
    """SLSQP refinement (finite‑difference gradient)."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}
    opts = {'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    res = opt.minimize(compute_c5, h0, method='SLSQP',
                       bounds=bounds, constraints=cons, options=opts)
    h_opt = project_onto_box_sum(res.x, n / 2.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # initial global best (if a parent construction exists)
    best_h = None
    best_score = float('inf')
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        h0 = project_onto_box_sum(h0, h0.size / 2.0)
        sc0 = compute_c5(h0)
        best_h, best_score = h0, sc0

    # scale schedule and per‑scale restart limits
    scales = [64, 128, 256]
    restart_limits = {64: 10, 128: 5, 256: 3}
    hill_iters = {64: 6000, 128: 4000, 256: 3000}
    for n in scales:
        if time_left() < 5:
            break
        max_restarts = restart_limits.get(n, 2)
        hill_iter = hill_iters.get(n, 3000)
        best_h_n = None
        best_score_n = float('inf')
        for _ in range(max_restarts):
            if time_left() < 2:
                break
            h = random_balanced_binary(n, rng)
            h, sc = hill_climb_binary(h, rng, time_left, max_iter=hill_iter)
            if sc < best_score_n:
                best_h_n, best_score_n = h.copy(), sc
            # a quick subgradient refinement on the local optimum
            if time_left() > 1.0:
                h_ref, sc_ref = grad_descent_subgrad(best_h_n, rng, time_left, max_iter=1500)
                if sc_ref < best_score_n:
                    best_h_n, best_score_n = h_ref.copy(), sc_ref
        # after all restarts, a final SLSQP polish
        if best_h_n is not None and time_left() > 5.0:
            h_opt, sc_opt = continuous_optimize_slsqp(best_h_n, rng, time_left, max_iter=400)
            if sc_opt < best_score_n:
                best_h_n, best_score_n = h_opt, sc_opt
        # update global best
        if best_score_n < best_score:
            best_h, best_score = best_h_n.copy(), best_score_n
        # stop early if we already beat the target
        if best_score <= 0.3808:
            break
        # prepare for next scale by up‑sampling the current best
        if n != scales[-1]:
            nxt = scales[scales.index(n) + 1]
            repeat = nxt // n
            rem = nxt % n
            up = np.repeat(best_h, repeat)
            if rem:
                up = np.concatenate([up, best_h[:rem]])
            best_h = project_onto_box_sum(up, nxt / 2.0)

    # final refinement on the best candidate found
    if best_h is not None and time_left() > 3.0:
        best_h, best_score = grad_descent_subgrad(best_h, rng, time_left, max_iter=4000)
        if time_left() > 1.0:
            best_h, best_score = continuous_optimize_slsqp(best_h, rng, time_left, max_iter=800)

    # ensure feasibility and compute final score
    if best_h is None:
        best_h = random_balanced_binary(64, rng)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
