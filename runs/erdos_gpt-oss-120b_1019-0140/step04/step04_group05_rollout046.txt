    best_h = h.copy()

    start = time.time()

    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))
    # base: if initial_h_values exist, use it, else generate random.
    if 'initial_h_values' in globals():

    base = project_onto_box_sum(base, base.shape[0]/2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)
    # Multi-scale: sizes = [64, 128, 256, 512]
    for n in [64,128,256,512]:
        if time_left() <= 0.0: break

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral C5 for discrete h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def neighbor_sum(h, shift):
    """Compute h_{i+shift}+h_{i-shift} with zero outside."""
    n = h.size
    res = np.zeros_like(h)
    if shift >= 0:
        m = n - shift
        if m > 0:
            res[:m] += h[shift:]
            res[shift:] += h[:m]
    else:
        s = -shift
        m = n - s
        if m > 0:
            res[s:] += h[:m]
            res[:m] += h[s:]
    return res


def subgradient_descent(h0, time_limit, rng, step0=0.2):
    """Projected subgradient descent on max_{k} corr_k."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        # neighbor sum = gradient of -sum_i h_i h_{i+shift}
        nb = neighbor_sum(h, shift)
        # step decreasing linearly
        step = step0 * (1.0 - elapsed / time_limit)
        if step <= 0:
            break
        h = h + step * nb
        h = project_onto_box_sum(h, n / 2.0)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # occasional random shake to escape plateaus
        if rng.random() < 0.001:
            h = h + rng.normal(scale=0.01, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_box_sum(h, n / 2.0)
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = rng.random(64)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # multi‑resolution loop
    for n in (64, 128, 256, 512):
        if time_left() <= 0.0:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        # allocate ~30 % of remaining time to this size
        alloc = min(time_left() * 0.3, time_left())
        if alloc <= 0.0:
            break

        # two runs with different step scales
        for stp in (0.3, 0.07):
            sub_alloc = alloc / 2.0
            cand_h, cand_score = subgradient_descent(best_h, sub_alloc, rng, step0=stp)
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand_h.copy()
            if time_left() <= 0.0:
                break
        # keep the latest best as starting point for finer grid
        best_h = project_onto_box_sum(best_h, n / 2.0)

    # final random refinements with leftover budget
    while time_left() > 0.6:
        n = best_h.size
        # random restart
        h_rand = rng.random(n)
        h_rand = project_onto_box_sum(h_rand, n / 2.0)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score = sc_rand
            best_h = h_rand.copy()
        # single gradient step from current best
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        nb = neighbor_sum(best_h, shift)
        step = 0.05 * (2.0 / n)
        h_new = best_h + step * nb
        h_new = project_onto_box_sum(h_new, n / 2.0)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
