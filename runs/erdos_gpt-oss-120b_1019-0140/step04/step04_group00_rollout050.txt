    target = n/2.0
    h = project_onto_box_sum(h0, target)

    beta2 = 0.999

def compute_corr_and_gradient_weighted(h, alpha=10.0):
    n = h.shape[0]
    # raw correlation via FFT
    L = 2*n - 1
    N = 1 << ((L).bit_length())
    fft_h_rev = np.fft.rfft(h[::-1], n=N)
    fft_f = np.fft.rfft(1.0 - h, n=N)
    conv_full = np.fft.irfft(fft_h_rev * fft_f, n=N)
    corr_raw = conv_full[:L]  # length L
    max_corr = np.max(corr_raw)
    # softmax weights
    w = np.exp(alpha * (corr_raw - max_corr))
    w_sum = np.sum(w)
    w = w / w_sum
    # convolution for gradient
    M = 3*n - 2
    N2 = 1 << ((M).bit_length())
    # compute fft_h for size N2
    fft_h2 = np.fft.rfft(h, n=N2)
    fft_w_rev = np.fft.rfft(w[::-1], n=N2)
    conv1_full = np.fft.irfft(fft_w_rev * fft_h2, n=N2)
    conv1 = conv1_full[(n-1):(n-1)+n]
    fft_w = np.fft.rfft(w, n=N2)
    conv2_full = np.fft.irfft(fft_w * fft_h2, n=N2)
    conv2 = conv2_full[(n-1):(n-1)+n]
    grad = 1.0 - conv1 - conv2
    return max_corr, grad

def compute_gradient_max_shift(h):
    n = h.shape[0]
    # compute correlation via FFT to find max shift(s)
    N = 1 << ((2*n - 1).bit_length())
    fft_h_rev = np.fft.rfft(h[::-1], n=N)
    fft_f = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_h_rev * fft_f, n=N)
    corr = conv[:2*n - 1]
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            n_s = n - s
            term1 = np.zeros(n); term1[:n_s] = h[s:]
            term2 = np.zeros(n); term2[s:] = h[:n_s]
        else:
            s2 = -s
            n_s = n - s2
            term1 = np.zeros(n); term1[s2:] = h[:n_s]
            term2 = np.zeros(n); term2[:n_s] = h[s2:]
        grad += 1.0 - term1 - term2
    grad /= len(shifts)
    return grad

    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            n_s = n - s
            term1 = np.zeros(n); term1[:n_s] = h[s:]
            term2 = np.zeros(n); term2[s:] = h[:n_s]
        else:
            s2 = -s
            n_s = n - s2
            term1 = np.zeros(n); term1[s2:] = h[:n_s]
            term2 = np.zeros(n); term2[:n_s] = h[s2:]
        grad += 1.0 - term1 - term2
    grad /= len(shifts)
    return grad

def refine_max_shift(h0, time_left, step0=0.5, beta=0.9, max_iter=20000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.02:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c5

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # base resolution
    if 'initial_h_values' in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
        base_n = base_h.size
        base_h = project_onto_box_sum(base_h, base_n / 2.0)
    else:
        base_n = 256
        base_h = random_binary_balanced(base_n, rng)

    candidates = [
        base_h,
        thue_morse_binary(base_n),
        random_binary_balanced(base_n, rng),
        project_onto_box_sum(rng.random(base_n), base_n / 2.0)
    ]

    best_h = None
    best_c5 = np.inf

    for h0 in candidates:
        if time_left() < 2.0:
            break
        h_opt, c5_opt = adam_optimize(h0, time_left, max_iters=5000, alpha=12.0)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
        if best_c5 <= 0.3808:
            return best_h, best_c5, best_h.size

    if best_h is None:
        best_h = project_onto_box_sum(np.full(base_n, 0.5), base_n / 2.0)
        best_c5 = compute_c5_fast(best_h)

    # refinement with true max‑shift subgradient
    if time_left() > 1.0:
        h_ref, c5_ref = refine_max_shift(best_h, time_left, step0=0.5, max_iter=8000)
        if c5_ref < best_c5:
            best_c5, best_h = c5_ref, h_ref.copy()
        if best_c5 <= 0.3808:
            return best_h, best_c5, best_h.size

    # higher resolutions
    for n in (512, 1024):
        if time_left() < 5.0:
            break
        h_up = resample(best_h, n)
        h_up = project_onto_box_sum(h_up, n / 2.0)
        cand_n = [
            h_up,
            random_binary_balanced(n, rng),
            project_onto_box_sum(rng.random(n), n / 2.0)
        ]
        for h0 in cand_n:
            if time_left() < 2.0:
                break
            h_opt, c5_opt = adam_optimize(h0, time_left, max_iters=3000, alpha=12.0)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
            if best_c5 <= 0.3808:
                return best_h, best_c5, n
        if time_left() > 1.0:
            h_ref, c5_ref = refine_max_shift(best_h, time_left, step0=0.4, max_iter=6000)
            if c5_ref < best_c5:
                best_c5, best_h = c5_ref, h_ref.copy()
            if best_c5 <= 0.3808:
                return best_h, best_c5, best_h.size

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_c5 = compute_c5_fast(best_h)
    return best_h, best_c5, best_h.size
