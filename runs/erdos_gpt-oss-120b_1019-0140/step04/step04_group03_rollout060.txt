import time
import numpy as np

def compute_c5(h):
   n = h.shape[0]
   corr = np.correlate(h, 1.0 - h, mode='full')
   return np.max(corr) * (2.0 / n)

def thue_morse(n):
   return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=np.int8, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 256
    # initial h
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=np.int8)
        m = base.shape[0]
        if n % m == 0:
            cur_h = np.tile(base, n // m)
        else:
            cur_h = thue_morse(n)
    else:
        cur_h = thue_morse(n)
    cur_h = cur_h.astype(np.int8)
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    target = 0.3808
    T0 = 0.01
    T_final = 1e-6
    step = 0
    # streak counter for no improvement
    no_improve = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.5:
            break
        # compute correlation
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_overlap = corr.max()
        cur_score = max_overlap * (2.0 / n)
        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h.copy()
            no_improve = 0
            if best_score <= target:
                break
        else:
            no_improve += 1

        rng.shuffle(worst_idxs)
        swapped = False
        i = j = None

            w = np.ones(n, dtype=np.int8)

                sz = n - L
                w[:sz] = 1 - cur_h[L:]
            else:

                w[s:] = 1 - cur_h[:sz]

                break

            j = int(rng.choice(zeros))
            cur_h[i] = 0
            cur_h[j] = 1
        # compute new score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=np.int8, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h, dtype=float)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    idx = np.argpartition(-arr, k - 1)[:k]
    b = np.zeros_like(arr, dtype=np.int8)

        if not swapped:
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            cur_h[i] = 0
            cur_h[j] = 1
        new_score = compute_c5(cur_h)
        elapsed = time.time() - start
        T = T0 * max(0.0, (max_time - elapsed) / max_time) + T_final
        if new_score <= cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        if no_improve >= 50000:
            cur_h = random_binary_balanced(n, rng)
            cur_score = compute_c5(cur_h)
            no_improve = 0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_final = 512
    scales = [64, 128, 256, n_final]
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng).astype(float)
    best_h = None
    best_score = np.inf
    for sidx, n in enumerate(scales):
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        sample_limit = 200 if n >= 256 else 500
        for _ in range(sample_limit):
            if budget_s - (time.time() - start) <= 1.0:
                break
            h_rand = random_binary_balanced(n, rng).astype(float)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score, best_h = sc, h_rand.copy()
        for r in range(4):
            if budget_s - (time.time() - start) <= 5.0:
                break
            h0 = base.copy()
            if r > 0:
                h0 += rng.normal(scale=0.03, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, n / 2.0)
            remaining = budget_s - (time.time() - start)
            max_iters = int(min(200000, max(20000, remaining * (20000.0 / max(64, n)))))
            h_opt, sc_opt = gradient_descent_momentum_custom(
                h0, n_iter=max_iters, step0=0.5, beta=0.9,
                rng=rng, max_time=remaining)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt
        if best_h is not None and best_h.size == n and budget_s - (time.time() - start) > 5.0:
            h_cur = best_h.copy()
            cur_score = best_score
            total_steps = int(min(40000, max(10000,
                (budget_s - (time.time() - start)) * 500.0)))
            T0 = 0.01
            T_final = 1e-6
            for step in range(total_steps):
                if budget_s - (time.time() - start) <= 0.0:
                    break
                T = T0 * max(0.0, (total_steps - step) / total_steps) + T_final
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(h_cur[i], 1.0 - h_cur[j])
                if max_move <= 0.0:
                    continue
                delta = max_move * rng.random()
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_score = compute_c5(h_new)
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                    h_cur = h_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score, best_h = cur_score, h_cur.copy()
        if sidx < len(scales) - 1:
            base = best_h if best_h is not None else base
            base = upsample(base, scales[sidx + 1])
    if best_h is not None and budget_s - (time.time() - start) > 2.0:
        n = best_h.shape[0]
        best_bin = to_binary_balanced(best_h)
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            max_iters = int(min(300000, max(50000,
                remaining * (50000.0 / max(64, n)))))
            best_bin_f, _ = gradient_descent_momentum_custom(
                best_bin.astype(float), n_iter=max_iters,
                step0=0.2, beta=0.9, rng=rng,
                max_time=remaining)
            best_bin = to_binary_balanced(best_bin_f)
        remaining = budget_s - (time.time() - start)
        if remaining > 1.0:
            best_bin, best_score = greedy_swap_fine_tune(best_bin, rng, remaining, target=0.3808)
        else:
            best_score = compute_c5(best_bin)
        best_h = best_bin.astype(float)
    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(n_final), n_final / 2.0).astype(float)
        best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
