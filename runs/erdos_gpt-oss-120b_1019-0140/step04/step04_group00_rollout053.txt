```
def softmin_lagrangian(h, beta):
    a = np.correlate(h, h, mode='full')
    exp_term = np.exp(-beta * a)
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp
    L = np.log(sum_exp) / beta
    # gradient:
    n = h.size
    grad = np.zeros_like(h)
    # shift index loop:
    offset = n - 1
    for idx, weight in enumerate(w):
        shift = idx - offset
        if shift >= 0:
            # positive shift
            if shift < n:
                grad[:n-shift] -= weight * h[shift:]
                grad[shift:] -= weight * h[:n-shift]
        else:
            s = -shift
            if s < n:
                grad[s:] -= weight * h[:n-s]
                grad[:n-s] -= weight * h[s:]
    return L, grad
```

```python
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # Expand bounds if needed
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

        grad += 1.0 - term1 - term2
    grad /= len(shifts)

            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def thue_morse_binary(N):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(N)), dtype=float, count=N)

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target_c5 = 0.3808
    n = 256
    target_sum = n / 2.0
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        candidates.append(project_onto_box_sum(init, target_sum))
    candidates.append(project_onto_box_sum(thue_morse_binary(n), target_sum))
    for _ in range(6):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))
    best_h = None
    best_c5 = np.inf
    for init_h in candidates:
        if time_left() < 5.0:
            break
        h = init_h.copy()
        m_adam = np.zeros_like(h)
        v_adam = np.zeros_like(h)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        step0 = 0.05
        max_iters = 3000
        for it in range(1, max_iters + 1):
            if time_left() < 0.1:
                break
            beta = 2.0 + (30.0 - 2.0) * (it - 1) / (max_iters - 1)
            a = np.correlate(h, h, mode='full')
            amin = a.min()
            exp_shift = np.exp(-beta * (a - amin))
            w = exp_shift / exp_shift.sum()
            grad = np.zeros_like(h)
            offset = n - 1
            for idx, weight in enumerate(w):
                shift = idx - offset
                if shift >= 0:
                    if shift < n:
                        grad[:n-shift] -= weight * h[shift:]
                        grad[shift:] -= weight * h[:n-shift]
                else:
                    s = -shift
                    if s < n:
                        grad[s:] -= weight * h[:n-s]
                        grad[:n-s] -= weight * h[s:]
            m_adam = beta1 * m_adam + (1 - beta1) * grad
            v_adam = beta2 * v_adam + (1 - beta2) * (grad * grad)
            m_hat = m_adam / (1 - beta1**it)
            v_hat = v_adam / (1 - beta2**it)
            step = step0 / np.sqrt(it)
            h_new = h + step * (m_hat / (np.sqrt(v_hat) + eps))
            h_new = project_onto_box_sum(h_new, target_sum)
            h = h_new
            if it % 200 == 0 or it == max_iters:
                c5 = compute_c5_fast(h)
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
                    if best_c5 <= target_c5:
                        return best_h, best_c5, n
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), target_sum)
        best_c5 = compute_c5_fast(best_h)
    if time_left() > 2.0:
        h_ref, c_ref = gradient_descent_momentum(best_h, time_left, step0=0.4, beta=0.9, max_iter=100000)
        if c_ref < best_c5:
            best_c5, best_h = c_ref, h_ref.copy()
            if best_c5 <= target_c5:
                return best_h, best_c5, n
    if time_left() > 1.0:
        h = best_h.copy()
        cur = compute_c5_fast(h)
        best_local_h, best_local_c = h.copy(), cur
        T, T_min, decay = 0.2, 1e-5, 0.99997
        while time_left() > 0.1:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = (rng.random() - 0.5) * 0.2
            hi, hj = h[i], h[j]
            new_i = np.clip(hi + delta, 0.0, 1.0)
            new_j = np.clip(hj - delta, 0.0, 1.0)
            if abs((new_i + new_j) - (hi + hj)) > 1e-12:
                new_i, new_j = hi, hj
            h[i], h[j] = new_i, new_j
            new_c = compute_c5_fast(h)
            d = new_c - cur
            if d < 0 or rng.random() < np.exp(-d / T):
                cur = new_c
                if new_c < best_local_c:
                    best_local_c, best_local_h = new_c, h.copy()
                    if best_local_c <= target_c5:
                        return best_local_h, best_local_c, n
            else:
                h[i], h[j] = hi, hj
            T = max(T_min, T * decay)
        if best_local_c < best_c5:
            best_c5, best_h = best_local_c, best_local_h
    best_h = project_onto_box_sum(best_h, target_sum)
    best_c5 = compute_c5_fast(best_h)
    return best_h, best_c5, n
```
