def gradient_descent_adaptive_fft(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        # determine shifts within tolerance: using absolute tolerance maybe 1e-12
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        # compute gradient for these
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                end = n - shift
                grad[:end] += (1.0 - h[shift:])
                grad[shift:] += (1.0 - h[:end])
            else:
                sft = -shift
                end = n - sft
                grad[sft:] += (1.0 - h[:end])
                grad[:end] += (1.0 - h[sft:])
        if shifts.size > 0:
            grad /= shifts.size
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fft(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            # increase step slightly
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        # occasional jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5_fft(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def local_pair_moves(h, rng, time_left, max_moves=1000000):
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size
    # may set max_moves based on time left crudely; but we just loop while time left.
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # move from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            new_score = compute_c5_fft(h_new)
            if new_score < best_score:
                best_h, best_score = h_new, new_score
        # move from j to i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            new_score2 = compute_c5_fft(h_new2)
            if new_score2 < best_score:
                best_h, best_score = h_new2, new_score2
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = compute_corr_fft(best_h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            end = n - shift
            contrib = best_h[:end] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            sft = -shift
            end = n - sft
            contrib = best_h[sft:] * (1.0 - best_h[:end])
            i = np.argmax(contrib) + sft
            j = i - sft
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)
        if new_score < best_score:
            best_h, best_score = h_new, new_score
    return best_h, best_score

Now simulated_annealing:

def simulated_annealing(h, rng, time_left, budget_s):
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = h.shape[0]
    max_temp = 0.01
    while time_left() > 0:
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j]) * 0.5
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)
        if new_score < cur_score:
            cur_h, cur_score = h_new, new_score
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
        else:
            # temperature decays linearly with remaining time
            temp = max_temp * (time_left() / budget_s)
            # avoid zero division
            if temp > 0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_h, cur_score = h_new, new_score
    return best_h, best_score

    else:
        # generate thue-morse of length 64
        def thue_morse(n):
            return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
        base = thue_morse(64)
    # upsample to target size
    target_n = 512
    def upsample(arr, new_n):
        m = arr.shape[0]
        if m == new_n:
            return arr.copy()
        x_old = np.linspace(0.0, 1.0, m, endpoint=False)
        x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
        return np.interp(x_new, x_old, arr)
    base = upsample(base, target_n)
    # project to constraints
    def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
        n = v.size
        if s <= lo * n:
            return np.full_like(v, lo)
        if s >= hi * n:
            return np.full_like(v, hi)
        # bounds for lambda
        lam_low = np.min(v) - hi
        lam_high = np.max(v) - lo
        def f(lam):
            return np.clip(v - lam, lo, hi).sum()
        # ensure brackets
        if f(lam_low) < s:
            lam_low = np.min(v) - 10 * hi
        if f(lam_high) > s:
            lam_high = np.max(v) - 10 * lo
        for _ in range(max_iter):
            lam_mid = (lam_low + lam_high) * 0.5
            fm = f(lam_mid)
            if abs(fm - s) < 1e-12:
                break
            if fm > s:
                lam_low = lam_mid
            else:
                lam_high = lam_mid
        return np.clip(v - lam_mid, lo, hi)

    best_h = base.copy()

import time
import numpy as np

# ---------- Basic utilities ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo≤x_i≤hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from old size to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


# ---------- Correlation & objective ----------
def compute_corr_fft(h):
    """Full correlation of h with (1‑h) using FFT (linear, no circular wrap)."""
    n = h.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1.0 - h, n=nfft)
    C = H * np.conj(G)
    corr = np.fft.irfft(C, n=nfft)
    return corr[: 2 * n - 1]


def compute_c5_fft(h):
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    return compute_corr_fft(h).max() * (2.0 / n)


# ---------- Optimization primitives ----------
def gradient_descent_adaptive_fft(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent on C₅ using FFT for correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)

        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                end = n - shift
                grad[:end] += (1.0 - h[shift:])
                grad[shift:] += (1.0 - h[:end])
            else:
                sft = -shift
                end = n - sft
                grad[sft:] += (1.0 - h[:end])
                grad[:end] += (1.0 - h[sft:])
        if shifts.size > 0:
            grad /= shifts.size

        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fft(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5_fft(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    """Transfer mass from points contributing most to the worst shift."""
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size

    while time_left() > 0:
        corr = compute_corr_fft(best_h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            end = n - shift
            contrib = best_h[:end] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            sft = -shift
            end = n - sft
            contrib = best_h[sft:] * (1.0 - best_h[:end])
            i = np.argmax(contrib) + sft
            j = i - sft
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)
        if new_score < best_score:
            best_h, best_score = h_new, new_score
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Small random re‑distributions between pairs of points."""
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    n = best_h.size

    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5_fft(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        # reverse direction
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5_fft(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def simulated_annealing(h, rng, time_left, budget_s):
    """Metropolis‑style SA with linearly decaying temperature."""
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = h.shape[0]
    max_temp = 0.01

    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j]) * 0.5
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)

        if new_score < cur_score:
            cur_h, cur_score = h_new, new_score
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
        else:
            temp = max_temp * (time_left() / budget_s)
            if temp > 0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_h, cur_score = h_new, new_score
    return best_h, best_score

    if time_left() > 10:
