import time
import numpy as np

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    arr = np.array([0], dtype=np.uint8)
    while arr.size < n:
        arr = np.concatenate([arr, 1 - arr])
    return arr[:n].astype(float)

def compute_c5_fft(h):
    n = h.shape[0]
    g = 1.0 - h
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, n=L)
    fg = np.fft.rfft(g, n=L)
    corr = np.fft.irfft(fh * np.conj(fg), n=L)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024  # number of sample points, even
    # Build initial binary step function
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.size < n:
            repeats = (n // base.size) + 1
            base = np.tile(base, repeats)[:n]
        else:
            base = base[:n]
        # select top n/2 positions as ones
        idx = np.argpartition(base, -n // 2)[-n // 2:]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
    else:
        h = random_binary_balanced(n, rng)
    # Ensure exactly n/2 ones (in case of rounding)
    cur_ones = int(h.sum())
    target_ones = n // 2
    if cur_ones > target_ones:
        excess = cur_ones - target_ones
        ones_idx = np.where(h > 0.5)[0]
        flip = rng.choice(ones_idx, size=excess, replace=False)
        h[flip] = 0.0
    elif cur_ones < target_ones:
        deficit = target_ones - cur_ones
        zero_idx = np.where(h < 0.5)[0]
        flip = rng.choice(zero_idx, size=deficit, replace=False)
        h[flip] = 1.0
    # Initial score
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score
    # Simulated annealing parameters
    T0 = 0.12
    T_end = 1e-7
    # Main loop
    while True:
        now = time.time()
        if now - start >= budget_s - 0.2:
            break
        # pick a random 1 and a random 0 to swap
        # simple rejection loops
        i = rng.integers(0, n)
        while h[i] < 0.5:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] > 0.5:
            j = rng.integers(0, n)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        elapsed_frac = (now - start) / budget_s
        T = T0 * (1 - elapsed_frac) + T_end * elapsed_frac
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    # Return best solution found
    return best_h, best_score, n

def random_binary_balanced(n, rng):
    """Return a binary vector of length n with exactly n//2 ones."""
    h = np.zeros(n, dtype=float)

    h[idx] = 1.0

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (n should be a power of two)."""
    arr = np.array([0], dtype=np.uint8)
    while arr.size < n:
        arr = np.concatenate([arr, 1 - arr])
    return arr[:n].astype(float)

def correlation_fft(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    g = 1.0 - h
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.rfft(h, n=L)
    fg = np.fft.rfft(g, n=L)
    corr = np.fft.irfft(fh * np.conj(fg), n=L)[:2 * n - 1]
    return corr
