def adam_projected(h0, max_iters, step_size, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters+1):
        if time_left() <= 0: break
        # compute gradient of max correlation term
        # need gradient for each shift that attains max correlation
        # Similar to compute_gradients_for_shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts) # re-use function
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)

def anneal_pairwise(h, rng, time_left, init_temp=0.01, cooling=0.999):

        if max_delta <= 0:
            temperature *= cooling
            continue
        delta = max_delta * rng.random() * 0.5
        new_h = current_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # ensure still sum unchanged; fine
        new_score = compute_c5(new_h)
        delta_score = new_score - current_score

            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
        temperature = max(temperature * cooling, 1e-7)
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad


def adam_projected(h0, max_iters, step_size, beta1, beta2, eps, rng, time_left):
    """Adam optimizer with projection onto box+sum constraints."""
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - step_size * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def anneal_pairwise(h0, rng, time_left, init_temp=0.01, cooling=0.9995):
    """Simulated annealing of pairwise mass transfers."""
    cur = h0.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    temp = init_temp
    n = cur.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            new_score = compute_c5(new)
            diff = new_score - cur_score
            if diff < 0 or rng.random() < np.exp(-diff / max(temp, 1e-12)):
                cur, cur_score = new, new_score
                if new_score < best_score:
                    best, best_score = new.copy(), new_score
        temp = max(temp * cooling, 1e-8)
    return best, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if time_left() <= 0:
            break
        # Upsample / resize
        if best_h.size != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)

        # Adam refinement
        max_iters = int(min(80000, max(10000, time_left() * (20000.0 / n))))
        h_adam, sc_adam = adam_projected(
            base, max_iters, step_size=0.5,
            beta1=0.9, beta2=0.999, eps=1e-8,
            rng=rng, time_left=time_left
        )
        if sc_adam < best_score:
            best_h, best_score = h_adam.copy(), sc_adam

        # Short annealing phase
        if time_left() > 2:
            t_end = time.time() + min(30.0, time_left() * 0.2)
            anneal_tl = lambda: max(0.0, t_end - time.time())
            h_ann, sc_ann = anneal_pairwise(best_h, rng, anneal_tl,
                                            init_temp=0.01, cooling=0.9995)
            if sc_ann < best_score:
                best_h, best_score = h_ann.copy(), sc_ann

        # Random balanced binary restarts
        if time_left() > 5:
            for _ in range(2):
                if time_left() < 3:
                    break
                h_rand = random_binary_balanced(n, rng).astype(float)
                h_rand = project_onto_box_sum(h_rand, n / 2.0)
                it_rand = int(min(30000, max(5000, time_left() * (15000.0 / n))))
                h_rand_opt, sc_rand = adam_projected(
                    h_rand, it_rand, step_size=0.5,
                    beta1=0.9, beta2=0.999, eps=1e-8,
                    rng=rng, time_left=time_left
                )
                if sc_rand < best_score:
                    best_h, best_score = h_rand_opt.copy(), sc_rand

    # Final intensive Adam refinement
    if time_left() > 5:
        final_iters = int(min(120000, max(50000, time_left() * 50)))
        best_h, best_score = adam_projected(
            best_h, final_iters, step_size=0.3,
            beta1=0.9, beta2=0.999, eps=1e-8,
            rng=rng, time_left=time_left
        )

    # Final annealing fine‑tuning
    if time_left() > 2:
        final_ann_tl = lambda: max(0.0, budget_s - (time.time() - start))
        best_h, best_score = anneal_pairwise(best_h, rng, final_ann_tl,
                                            init_temp=0.005, cooling=0.9995)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
