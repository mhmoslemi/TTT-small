import time

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid

    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""

        else:
            no_improve += 1

        if best_score <= 0.3808:
            break

        p = np.argmax(corr)
        lag = p - (n - 1)

        # Gradient of the worst correlation
        g = np.zeros_like(h)
        i_start = max(0, -lag)
        i_end = min(n - 1, n - 1 - lag)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = i + lag
        g[i] += 1.0 - h[j]
        g[j] -= h[i]

        step = alpha0 * (1.0 - elapsed() / budget_s)
        if step < 1e-5:
            step = 1e-5

        h_candidate = h - step * g
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_candidate = project_onto_box_sum(h_candidate, n / 2.0)

        corr_new = compute_corr_full(h_candidate)
        c5_new = np.max(corr_new) * (2.0 / n)

        if c5_new < c5 or rng.random() < 0.005:
            h = h_candidate

        if no_improve >= 3000:
            # Mix best with a fresh random binary vector to escape stagnation
            rand_vec = random_balanced_binary(n, rng)
            h = project_onto_box_sum(0.9 * best_h + 0.1 * rand_vec, n / 2.0)
            no_improve = 0

    # Final local tweaks if time permits
    while time_left() > 0.2:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.3
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < best_score:
            best_score = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            if rng.random() < 0.001:
                h = h_new
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
