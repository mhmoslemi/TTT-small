import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, max_iter, step0, beta,
                     noise_interval, noise_scale, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if noise_interval and (it % noise_interval) == 0:
            h_new += rng.normal(scale=noise_scale, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # base construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128)
    n_target = 512
    if base.shape[0] != n_target:
        base = upsample(base, n_target)
    base = project_onto_box_sum(base, n_target / 2.0)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    # Gradient descent stage
    restarts = 10
    for r in range(restarts):
        if time_left() <= 5.0:
            break
        h0 = base + rng.normal(scale=0.05, size=n_target)
        h0 = project_onto_box_sum(h0, n_target / 2.0)
        h_opt, sc_opt = gradient_descent(
            h0,
            max_iter=250000,
            step0=0.7,
            beta=0.9,
            noise_interval=2000,
            noise_scale=0.02,
            rng=rng,
            time_left=time_left
        )
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        # occasionally update base to a good solution to diversify restarts
        if r % 3 == 0 and sc_opt < best_score:
            base = h_opt.copy()

    # Simulated annealing refinement
    T0 = 5e-4  # initial temperature
    h = best_h.copy()
    score = best_score
    n = n_target
    while time_left() > 0.6:
        i, j = rng.choice(n, size=2, replace=False)
        # move mass from i to j
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 1e-12:
            continue
        delta = delta_max * rng.uniform(0.05, 0.2)  # up to 20% of feasible
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)
        d = new_score - score
        elapsed = time.time() - start
        T = T0 * max(0.0, (budget_s - elapsed) / budget_s)
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            h = h_new
            score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
        # occasional larger jumps
        if rng.random() < 0.001:
            idxs = rng.choice(n, size=4, replace=False)
            for a, b in zip(idxs[::2], idxs[1::2]):
                dm = min(h[a], 1.0 - h[b])
                if dm > 1e-12:
                    h[a] -= dm * 0.5
                    h[b] += dm * 0.5
    # final projection
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)

    return best_h, float(best_score), n
