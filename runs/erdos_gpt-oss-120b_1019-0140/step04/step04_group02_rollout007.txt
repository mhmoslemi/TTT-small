import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

    return grad

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            max_transfer = min(h[j], 1.0 - h[i])
            if max_transfer <= 0:
                continue
            i, j = j, i
        delta = rng.random() * max_transfer * 0.5
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            h = h_new
            cur_score = new_score
            if it % 2000 == 0:
                h += rng.normal(scale=0.005, size=n)
                h = project_onto_box_sum(h, n / 2.0)
                cur_score = compute_c5_fft(h)

        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
            if best_score <= TARGET:
                break
        it += 1

    # Gradient descent refinement
    if time_left() > 5.0:
        hd, sc = gradient_descent(best_h, n_iter=500000, step0=0.3, beta=0.9,
                                  rng=rng, time_left=time_left)
        if sc < best_score:
            best_score, best_h = sc, hd.copy()

    # Final random binary / local mass‑transfer search
    while time_left() > 2.0:
        # random balanced binary candidate
        h_bin = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h_bin[idx] = 1.0
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
            if best_score <= TARGET:
                break

        # small mass transfer on current best
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.2)
        if delta > 1e-12:
            h_tmp = best_h.copy()
            h_tmp[i] -= delta
            h_tmp[j] += delta
            sc_tmp = compute_c5_fft(h_tmp)
            if sc_tmp < best_score:
                best_score, best_h = sc_tmp, h_tmp.copy()
                if best_score <= TARGET:
                    break

    # Optional SLSQP polishing
    if time_left() > 5.0:
        try:
            n_dim = best_h.size
            x0 = np.concatenate([best_h, [best_score]])
            bounds = [(0.0, 1.0)] * n_dim + [(0.0, 1.0)]

            def eq_sum(x):
                return np.sum(x[:n_dim]) - n_dim / 2.0

            def ineq_fun(x):
                h = x[:n_dim]
                t = x[-1]
                corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n_dim)
                return t - corr

            cons = [{'type': 'eq', 'fun': eq_sum},
                    {'type': 'ineq', 'fun': ineq_fun}]
            res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-12,
                                        'disp': False})
            if res.success:
                h_sls = res.x[:n_dim]
                h_sls = project_onto_box_sum(h_sls, n_dim / 2.0)
                sc_sls = compute_c5_fft(h_sls)
                if sc_sls < best_score:
                    best_score, best_h = sc_sls, h_sls.copy()
        except Exception:
            pass

    # Final projection and evaluation
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, n
