import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as floats."""
    return np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # --- initialization -------------------------------------------------
    n_target = 1024                     # final resolution
    # base from provided construction or Thue‑Morse
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = upsample(base, n_target)
    else:
        base = thue_morse(64)
        base = upsample(base, n_target)
    base = project_onto_box_sum(base, n_target / 2.0)

    # random binary alternative
    rand_h = random_balanced_binary(n_target, rng)
    rand_score = compute_c5(rand_h)
    base_score = compute_c5(base)

    if rand_score < base_score:
        h_cur = rand_h.copy()
        cur_score = rand_score
        best_h = rand_h.copy()
        best_score = rand_score
    else:
        h_cur = base.copy()
        cur_score = base_score
        best_h = base.copy()
        best_score = base_score

    # --- simulated annealing -------------------------------------------
    init_temp = 0.06
    max_iters = 10_000_000
    it = 0
    no_improve = 0
    target_score = 0.3808
    n = n_target

    while time_left() > 0.1 and it < max_iters:

        i = rng.integers(n)

        if delta <= 0.0:

        # apply move

        new_score = compute_c5(h_cur)

    best_score = compute_c5(best_h)
    return best_h, best_score, n
