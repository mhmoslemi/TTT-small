import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (balanced for powers of two)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    half_n = n // 2

    # --------- construct initial candidates ----------
    if 'initial_h_values' in globals():
        base_cont = np.asarray(initial_h_values, dtype=float)
        base_cont = upsample(base_cont, n)
        base_cont = np.clip(base_cont, 0.0, 1.0)
        base = project_onto_box_sum(base_cont, half_n)
    else:
        base = thue_morse(n)

    thue = thue_morse(n)
    rand_bal = np.zeros(n, dtype=float)
    rand_bal[rng.choice(n, half_n, replace=False)] = 1.0

    candidates = [base, thue, rand_bal]
    scores = [compute_c5(c) for c in candidates]
    best_idx = int(np.argmin(scores))
    best_h = candidates[best_idx].copy()
    best_score = scores[best_idx]

    # start simulated annealing from the best candidate
    h = best_h.copy()
    cur_score = best_score

    init_temp = 0.04  # higher initial temperature for broader exploration
    target = 0.3808

    # ---------- targeted simulated annealing ----------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.6:
            break
        # linear cooling
        temp = init_temp * max(0.0, 1.0 - elapsed / budget_s)

        # correlation to locate the worst shift
        full_corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(full_corr))
        s = k_max - (n - 1)  # shift index (can be negative)

        # prepare masks for guided swaps
        idx = np.arange(n)
        shifted = idx + s
        valid = (shifted >= 0) & (shifted < n)

        low_mask = np.zeros(n, dtype=bool)
        high_mask = np.zeros(n, dtype=bool)
        low_mask[valid] = (h[valid] < 0.2) & (h[shifted[valid]] > 0.8)
        high_mask[valid] = (h[valid] > 0.8) & (h[shifted[valid]] < 0.2)

        low_idxs = np.where(low_mask)[0]
        high_idxs = np.where(high_mask)[0]

        if low_idxs.size and high_idxs.size:
            i = rng.choice(low_idxs)
            j = rng.choice(high_idxs)
        else:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0.0:
            continue

        scale = max(0.1, temp / init_temp)
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            continue

        # apply tentative move
        old_i, old_j = h[i], h[j]
        h[i] = old_i + delta
        h[j] = old_j - delta
        new_score = compute_c5(h)
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
        else:
            # revert
            h[i] = old_i
            h[j] = old_j

    # ---------- final local refinement ----------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.2:
            break
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.1
        old_i, old_j = best_h[i], best_h[j]
        best_h[i] = old_i - delta
        best_h[j] = old_j + delta
        new_score = compute_c5(best_h)
        if new_score < best_score:
            best_score = new_score
        else:
            best_h[i] = old_i
            best_h[j] = old_j

    # final evaluation with the provided evaluator
    final_score = evaluate_erdos_solution(best_h.astype(float))
    return best_h.astype(float), float(final_score), n
