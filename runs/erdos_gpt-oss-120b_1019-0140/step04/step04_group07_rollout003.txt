def compute_c5(h):
    n = h.shape[0]
    # compute linear correlation of h with 1-h
    # The correlation at lag k (0-indexed) is sum_i h[i] * (1-h[i + k]) for i where i+k < n.
    # Using FFT:
    N = 2 * n
    # zero pad to length N for linear convolution
    hpad = np.r_[h, np.zeros(n)]
    gpad = np.r_[1 - h, np.zeros(n)]
    # compute convolution
    conv = np.fft.ifft(np.fft.fft(hpad) * np.fft.fft(gpad)).real
    # correlation result (full) length 2n-1: conv[:2*n-1]? Actually convolution length 2n-1. Need full correlation.
    corr = conv[:2*n-1]
    return np.max(corr) * (2.0 / n)

def simulated_annealing(h0, time_left, rng):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start_sa = time.time()
    total_sa_time = time_left() # run until time runs out in outer loop's context?
    while time_left() > 0.5:
        # propose a transfer
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        # temperature schedule based on elapsed time
        elapsed = time.time() - start_sa
        T0 = 0.01  # initial temperature?
        T1 = 0.0001
        # linear cooling
        if total_sa_time > 0:
            temp = T0 + (T1 - T0) * (elapsed / total_sa_time)
        else:
            temp = T1
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score
        # occasionally reheat?
    return best_h, best_score

def improve_max_shift(h0, time_left, rng):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    while time_left() > 0.1:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        # compute contribution array for that shift
        if shift >= 0:
            contrib = h[:n - shift] * (1.0 - h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = h[s:] * (1.0 - h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h = h_new.copy()
                best_score = cur_score
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # h[i+shift] contributes: -h[i] to gradient at i+shift? Actually gradient of h[i] * (1 - h[i+shift]) w.r.t h[i+shift] is -h[i]
            # For product h[i] * (1 - h[i+shift]), derivative wrt h[i] = (1 - h[i+shift]), derivative wrt h[i+shift] = -h[i]
            # So gradient contributions across i.
            # term1 = (1 - h[shift:]) for positions i with i from 0..n-shift-1 => contributes to grad[i]
            # term2 = -h[:n-shift] for positions i+shift
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    # avg gradient
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

import time
import numpy as np

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure bounds encompass solution
    if f(lam_low) < target:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target) < 1e-12:
            break
        if fm > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            t = -s
            grad[t:] += (1.0 - h[:n - t])
            grad[:n - t] -= h[t:]
    if shifts.size:
        grad /= shifts.size
    return grad

def gradient_descent(h0, time_left, rng):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    step = 0.8
    beta = 0.9
    momentum = np.zeros_like(h)
    while time_left() > 0.5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        shift_idxs = np.where(corr >= maxc - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shift_idxs)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = project_onto_box_sum(h - step * momentum, target)
        new = compute_c5(h_new)
        if new < cur:
            h, cur = h_new, new
            if new < best:
                best_h, best = h_new.copy(), new
            step = min(step * 1.02, 1.0)
        else:
            step *= 0.7
            if rng.random() < 0.01:
                h = project_onto_box_sum(h + rng.normal(scale=0.02, size=n), target)
                cur = compute_c5(h)
        if step < 1e-6:
            step = 0.5
    return best_h, best

def simulated_annealing(h0, time_left, rng):
    h = h0.copy()
    n = h.size
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    start = time.time()
    total = time_left()
    T0, T1 = 0.01, 1e-4
    while time_left() > 0.5:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new = compute_c5(h_new)
        elapsed = time.time() - start
        temp = T0 + (T1 - T0) * (elapsed / total) if total > 0 else T1
        if new < cur or rng.random() < np.exp(-(new - cur) / max(temp, 1e-12)):
            h, cur = h_new, new
            if new < best:
                best_h, best = h_new.copy(), new
    return best_h, best

    best_h, best = h.copy(), cur

    while time_left() > 0.3:

        max_delta2 = min(h[j], 1.0 - h[i])

            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = h.copy()

def improve_max_shift(h0, time_left, rng):
    h = h0.copy()
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    n = h.size
    while time_left() > 0.3:
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        shifts = np.where(corr >= maxc - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = h[:n - shift] * (1.0 - h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = h[s:] * (1.0 - h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5

        h_new[i] -= delta
        h_new[j] += delta
        new = compute_c5(h_new)
        if new < cur:
            h, cur = h_new, new
            if new < best:
                best_h, best = h_new.copy(), new
    return best_h, best

    rng = np.random.default_rng(seed)
