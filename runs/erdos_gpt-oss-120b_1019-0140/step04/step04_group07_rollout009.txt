```
def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Approximate C5 on the grid."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    jitter_interval = max(2000, n // 2)
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def annealing_optimize(h, rng, time_left, total_time):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    n = h.size
    T0 = best_score * 0.02 + 1e-12
    while time_left() > 0.2:
        T = T0 * (time_left() / total_time)
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        scale = max(0.1, T / T0)
        delta = max_delta * rng.random() * scale
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        dE = new_score - cur_score
        if dE <= 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # multi‑scale refinement (quick passes)
    scales = [64, 128, 256, 512, 1024]
    for n in scales:
        if time_left() <= 30:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
            base = base * ((n / 2.0) / np.sum(base))
            base = project_onto_box_sum(base, n / 2.0)
        max_iters = int(min(20000, max(2000, time_left() * (5000.0 / n))))
        base, _ = gradient_descent_adaptive(base, max_iters,
                                            step0=0.8, beta=0.9,
                                            rng=rng, time_left=time_left)
        if compute_c5(base) < best_score:
            best_h, best_score = base.copy(), compute_c5(base)

    # final intensive refinement on best (high‑resolution) solution
    if time_left() > 10:
        n = best_h.size
        max_iters = int(min(40000, max(20000, time_left() * (3000.0 / n))))
        best_h, best_score = gradient_descent_adaptive(best_h, max_iters,
                                                       step0=0.3, beta=0.9,
                                                       rng=rng, time_left=time_left)
    if time_left() > 5:
        total = time_left()
        best_h, best_score = annealing_optimize(best_h, rng, time_left, total)
    if time_left() > 5:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    # occasional random balanced restarts with brief refinement
    while time_left() > 30:
        h_rand = random_binary_balanced(best_h.size, rng).astype(float)
        # short GD on random start
        max_iters_rand = int(min(8000, time_left() * 0.5))
        h_rand, sc_rand = gradient_descent_adaptive(h_rand, max_iters_rand,
                                                    step0=0.5, beta=0.9,
                                                    rng=rng, time_left=time_left)
        if sc_rand < best_score:
            best_h, best_score = h_rand.copy(), sc_rand
        if time_left() > 5:
            best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
