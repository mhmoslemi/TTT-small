def population_optimize(pop, rng, time_left, max_iters=None):
    pop_size = len(pop)
    scores = [compute_c5(h) for h in pop]
    # sort
    while time_left() > 0:
        # sort indices by score
        order = np.argsort(scores)
        pop = [pop[i] for i in order]
        scores = [scores[i] for i in order]
        # best is pop[0]
        # generate children
        children = []
        # simple convex combos

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Starting base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # Use Thue-Morse pattern
        base = thue_morse(64).astype(float)

    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # define scales
    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if best_h.size != n:
            # upsample from previous best
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n/2.0)

        # initial population list: best_h + some randoms
        pop = [best_h]
        # add a few random balanced binaries
        for _ in range(3):
            if time_left() <= 0:
                break
            h_rand = random_balanced_binary(n, rng)
            pop.append(h_rand)

        # Optimize population
        if time_left() > 0:
            pop_best_h, pop_best_score = population_optimize(pop, rng, time_left, max_iters=None)

            # after optimization make sure best_h is updated
            if pop_best_score < best_score:
                best_h, best_score = pop_best_h, pop_best_score

        # Optional final refinement using gradient descent and max shift
        if time_left() > 0:
            # gradient descent
            best_h, best_score = gradient_descent_adaptive(best_h, max_iters=5000,
                                                          step0=0.4, beta=0.9,
                                                          rng=rng, time_left=time_left)
        if time_left() > 0:
            best_h, best_score = improve_via_max_shift(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

def pairwise_move(h, rng, max_delta=0.5):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    # ensure we move from higher to lower to allow delta
    max_possible = min(h[i], 1 - h[j])
    if max_possible <= 0:
        # try other direction
        max_possible = min(h[j], 1 - h[i])
        if max_possible <= 0:
            return h
        # swap i/j
        i, j = j, i
    delta = max_possible * rng.random() * max_delta
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # compute correlation and find worst shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = int(rng.choice(shifts))
        if shift >= 0:
            # indices where contributions exist: i from 0 to n-shift-1
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = int(np.argmax(contrib))
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = int(np.argmax(contrib)) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1: h[i+shift] appears as h(x+k) term: gradient w.r.t h[i] includes - (1 - h[i+shift])? Wait.
            # Actually gradient of h[i] * (1 - h[i+shift]) with respect to h[i] is (1 - h[i+shift]).
            # gradient w.r.t h[i+shift] is -h[i].
            # So we add contributions accordingly.
            # Using vectorized:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] += (-h[:n - shift])
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] += (-h[s:])
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    arr[idx] = 1.0
    return arr


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] += (-h[:n - shift])
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] += (-h[s:])
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with adaptive step size."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
    return best_h, best_score


def improve_via_max_shift(h, rng, time_left):
    """Transfer mass from a point that contributes most to the worst lag."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = int(rng.choice(shifts))
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = int(np.argmax(contrib))
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = int(np.argmax(contrib)) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def pairwise_move(h, rng, max_delta=0.5):
    """Transfer a small amount of mass from i to j respecting box constraints."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    max_possible = min(h[i], 1.0 - h[j])
    if max_possible <= 0:
        max_possible = min(h[j], 1.0 - h[i])
        if max_possible <= 0:
            return h
        i, j = j, i
    delta = max_possible * rng.random() * max_delta
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def population_optimize(pop, rng, time_left, max_iters=None):
    """Simple evolutionary loop on a small population."""
    pop_size = len(pop)
    scores = [compute_c5(h) for h in pop]
    it = 0
    while time_left() > 0.1:
        # sort by score
        order = np.argsort(scores)
        pop = [pop[i] for i in order]
        scores = [scores[i] for i in order]

        # generate a few children
        children = []

        # convex combinations of the two best individuals
        if pop_size >= 2:
            a, b = pop[0], pop[1]
            for _ in range(3):
                if time_left() <= 0:
                    break
                alpha = rng.random()
                child = alpha * a + (1 - alpha) * b
                # apply a few random pairwise moves
                for _ in range(rng.integers(1, 4)):
                    child = pairwise_move(child, rng, max_delta=0.5)
                child = project_onto_box_sum(child, pop[0].size / 2.0)
                children.append(child)

        # add some random balanced binaries
        for _ in range(2):
            if time_left() <= 0:
                break
            child = random_balanced_binary(pop[0].size, rng)
            child = project_onto_box_sum(child, pop[0].size / 2.0)
            children.append(child)

        # evaluate children
        if children:
            child_scores = [compute_c5(c) for c in children]
            pop.extend(children)
            scores.extend(child_scores)

        # retain the top pop_size individuals
        if len(pop) > pop_size:
            order = np.argsort(scores)[:pop_size]
            pop = [pop[i] for i in order]
            scores = [scores[i] for i in order]

        # occasional refinement of the best individual
        if rng.random() < 0.02 and time_left() > 0:
            refined_h, refined_sc = gradient_descent_adaptive(
                pop[0], max_iters=2000, step0=0.6, beta=0.9,
                rng=rng, time_left=time_left)
            pop[0] = refined_h
            scores[0] = refined_sc

        it += 1
        if max_iters is not None and it >= max_iters:
            break

    best_idx = int(np.argmin(scores))
    return pop[best_idx], scores[best_idx]


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Base construction
    if 'initial_h_values' in globals():

    scales = [64, 128, 256, 512, 1024]

    for n in scales:

        # initial population: current best + a few random binaries
        pop = [best_h]
        for _ in range(3):
            if time_left() <= 0:

            pop.append(random_balanced_binary(n, rng))

        # evolutionary optimization at this scale
        if time_left() > 0:
            best_h, best_score = population_optimize(pop, rng, time_left)

        # local refinements
        if time_left() > 0:
            best_h, best_score = gradient_descent_adaptive(

                rng=rng, time_left=time_left)
        if time_left() > 0:

        # keep globally best
        if best_score < compute_c5(best_h):
            best_h, best_score = best_h, best_score

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
