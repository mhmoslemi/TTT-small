def smoothed_descent(h0, time_left, max_iters=None, step0=0.2, tau_decay=0.99):
    h = h0.copy()
    n = h.size
    total = n/2.0
    # compute initial corr
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_score = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    step = step0
    # initial tau:
    max_corr = np.max(corr)
    tau = max_corr * 0.2 + 1e-6
    if max_iters is None:
        max_iters = int(min(5000, max(500, time_left() * (2000.0 / n))))
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # compute weights
        # subtract max_corr for numerical stability
        cmax = np.max(corr)
        # compute exponential weights
        exp_w = np.exp((corr - cmax) / tau)
        sum_w = np.sum(exp_w)
        w = exp_w / sum_w  # shape (2n-1)
        # compute gradient via convolutions
        offset = n - 1
        conv1 = np.convolve(1.0 - h, w, mode='full')
        conv2 = np.convolve(h, w[::-1], mode='full')
        grad = conv1[offset:offset + n] - conv2[offset:offset + n]
        # update
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, total)
        # compute new corr and score
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_score = np.max(corr_new) * (2.0 / n)
        # accept if improvement
        if new_score <= cur_score:
            h = h_new
            corr = corr_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            # reduce step
            step *= 0.5
        # decay tau
        tau *= tau_decay
        # ensure tau min
        if tau < 1e-12:
            tau = 1e-12
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    n = h.shape[0]
    # while time left, try random moves
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # compute max delta for moving mass from i to j
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        # choose delta randomly up to max_delta
        delta = max_delta * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < cur_score:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
        # also try opposite direction j -> i
        max_delta2 = min(cur_h[j], 1.0 - cur_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5

                cur_score = new_score2

    return best_h, best_score

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda such that sum(clamp(v - lam, lo, hi)) = total
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure f(lam_low) >= total >= f(lam_high) ??? Wait f decreases as lam increases? 
    # Let's think: v - lam => as lam increases, v - lam decreases, so after clipping, sum decreases. So f is decreasing in lam. 
    # For lam_low = min(v)-hi, v - lam_low >= hi => after clipping at hi, sum = hi * n. So f(lam_low) = hi * n >= total.
    # For lam_high = max(v)-lo, v - lam_high <= lo => clipped to lo, sum = lo * n <= total.
    # So we can binary search for lam such that f(lam) = total.

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - total) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

            h, corr, cur_score = h_new, corr_new, new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
        else:
            step *= 0.5
        tau = max(tau * tau_decay, 1e-12)
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h, cur_score = best_h.copy(), best_score
    n = h.shape[0]
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_score = compute_c5(new_h)
            if new_score < cur_score:
                cur_h, cur_score = new_h, new_score
                if new_score < best_score:
                    best_h, best_score = new_h.copy(), new_score
        max_delta2 = min(cur_h[j], 1.0 - cur_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            new_h2 = cur_h.copy()
            new_h2[j] -= delta2
            new_h2[i] += delta2
            new_score2 = compute_c5(new_h2)
            if new_score2 < cur_score:
                cur_h, cur_score = new_h2, new_score2
                if new_score2 < best_score:
                    best_h, best_score = new_h2.copy(), new_score2
    return best_h, best_score


def refine_scale(base_h, n, time_left, rng):
    if base_h.shape[0] != n:
        base_h = upsample(base_h, n)
    base_h = project_onto_box_sum(base_h, n / 2.0)
    best_h, best_score = base_h.copy(), compute_c5(base_h)
    restarts = 4
    for r in range(restarts):
        if time_left() <= 0:
            break
        if r == 0:
            start_h = base_h
        elif r == 1:
            start_h = random_binary_balanced(n, rng)
        elif r == 2:
            start_h = rng.random(n)
            start_h = project_onto_box_sum(start_h, n / 2.0)
        else:
            pert = rng.normal(scale=0.05, size=n)
            start_h = np.clip(base_h + pert, 0.0, 1.0)
            start_h = project_onto_box_sum(start_h, n / 2.0)
        t_rem = time_left()
        max_iters = int(min(5000, max(500, t_rem * (2000.0 / n))))
        cand_h, cand_score = smoothed_descent(start_h, time_left, max_iters=max_iters, step0=0.2)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.full(64, 0.5, dtype=float)
    base = project_onto_box_sum(base, 64 / 2.0)

    best_h, best_score = base.copy(), compute_c5(base)

    scales = [64, 128, 256, 512]
    for n in scales:
        if time_left() <= 2:
            break
        if best_h.shape[0] != n:
            cur_base = upsample(best_h, n)
        else:
            cur_base = best_h.copy()
        cur_base = project_onto_box_sum(cur_base, n / 2.0)
        cand_h, cand_score = refine_scale(cur_base, n, time_left, rng)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
        if time_left() > 2:
            best_h, best_score = local_pair_moves(best_h, rng, time_left)

    if time_left() > 2:
        best_h, best_score = smoothed_descent(best_h, time_left, step0=0.1)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
