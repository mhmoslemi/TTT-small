def adam_optimize(h0, num_iters, lr, rng, max_time):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    target_sum = h.size / 2
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    for t in range(1, num_iters+1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr / (np.sqrt(v_hat) + eps) * m_hat
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_targeted(h, cur_score, steps, rng, max_time, n):
    start = time.time()
    best_h = h.copy()
    best_score = cur_score
    for step in range(steps):
        if time.time() - start >= max_time:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        k = np.argmax(corr) - (n - 1)
        # compute product array for shift k
        if k >= 0:
            prod = h[:n-k] * h[k:]
            idx_range = np.arange(n-k)
        else:
            k_abs = -k
            prod = h[k_abs:] * h[:n-k_abs]
            idx_range = np.arange(k_abs, n)
        i_local = np.argmax(prod)
        i = idx_range[i_local]
        # choose j with low product; sample candidates
        # sample up to 20 random indices
        cand = rng.choice(n, size=20, replace=False)
        # compute product for each candidate
        # handle out-of-range (product = 0)
        if k >= 0:
            # product for cand = h[cand] * h[cand+k] if cand+k < n else 0
            cand_k = cand + k
            valid = cand_k < n
            prod_cand = np.zeros_like(cand, dtype=float)
            prod_cand[valid] = h[cand[valid]] * h[cand_k[valid]]
        else:
            cand_k = cand + k  # since k negative
            valid = cand_k >= 0
            prod_cand = np.zeros_like(cand, dtype=float)
            prod_cand[valid] = h[cand[valid]] * h[cand_k[valid]]
        j = cand[np.argmin(prod_cand)]

        # transfer delta
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0:
            continue
        # adjust delta magnitude based on temperature (larger moves early)
        # e.g., delta = delta_max * rng.random() * (T_factor)
        # compute temperature factor: start from 1, linearly decrease
        T_factor = 1.0 - step / steps

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    v = np.asarray(v, dtype=float)
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def adam_optimize(h0, max_iters, lr, rng, max_time):
    start = time.time()
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    target = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        h_new = h - lr / (np.sqrt(v_hat) + eps) * m_hat
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal(h, cur_score, steps, rng, max_time, n):
    start = time.time()
    best_h = h.copy()
    best_score = cur_score
    T0 = 0.02
    T_final = 1e-6
    for step in range(steps):
        if time.time() - start >= max_time:
            break
        T = T0 * ((steps - step) / steps) + T_final
        # compute current maximal shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        k = np.argmax(corr) - (n - 1)
        # product for shift k
        if k >= 0:
            prod = h[:n - k] * h[k:]
            idx_range = np.arange(n - k)
        else:
            k_abs = -k
            prod = h[k_abs:] * h[:n - k_abs]
            idx_range = np.arange(k_abs, n)
        i = idx_range[np.argmax(prod)]
        # candidate j with low product
        cand = rng.choice(n, size=20, replace=False)
        if k >= 0:
            cand_k = cand + k
            valid = cand_k < n
            prod_cand = np.zeros_like(cand, dtype=float)
            prod_cand[valid] = h[cand[valid]] * h[cand_k[valid]]
        else:
            cand_k = cand + k
            valid = cand_k >= 0
            prod_cand = np.zeros_like(cand, dtype=float)
            prod_cand[valid] = h[cand[valid]] * h[cand_k[valid]]
        j = cand[np.argmin(prod_cand)]
        # propose transfer from i to j
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            continue
        delta = delta_max * rng.random() * (T / T0)
        if delta <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        # occasional random move for diversification
        if rng.random() < 0.05:
            i2, j2 = rng.choice(n, size=2, replace=False)
            dmax = min(h[i2], 1.0 - h[j2])
            if dmax > 0.0:
                d = dmax * rng.random() * (T / T0)
                h_rand = h.copy()
                h_rand[i2] -= d
                h_rand[j2] += d
                s_rand = compute_c5(h_rand)
                if s_rand < cur_score or rng.random() < np.exp(-(s_rand - cur_score) / T):
                    h = h_rand
                    cur_score = s_rand
                    if cur_score < best_score:
                        best_score = cur_score
                        best_h = h.copy()
    return best_h, best_score

def optimize_once(h_start, time_budget, rng):
    start = time.time()
    # Phase 1: Adam
    t_adam = time_budget * 0.3
    h_opt, sc_opt = adam_optimize(h_start, max_iters=50000, lr=0.5, rng=rng, max_time=t_adam)
    # Phase 2: Targeted annealing
    elapsed = time.time() - start
    remaining = time_budget - elapsed
    if remaining > 1.0:
        steps = int(min(200000, max(50000, remaining * 200.0)))
        h_opt, sc_opt = anneal(h_opt, sc_opt, steps, rng, remaining, h_opt.size)
    # Phase 3: short Adam fine‑tune
    elapsed = time.time() - start
    remaining = time_budget - elapsed
    if remaining > 1.0:
        h_opt, sc_opt = adam_optimize(h_opt, max_iters=20000, lr=0.2, rng=rng, max_time=remaining)
    # final projection
    h_opt = project_onto_box_sum(h_opt, h_opt.size / 2.0)
    sc_opt = compute_c5(h_opt)
    return h_opt, sc_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512  # number of samples (dx = 2/n)
