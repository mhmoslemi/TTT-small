def grad_for_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        # term1: 1 - h[shift:] for indices 0..n-1-shift; then zeros for last shift.
        term1 = np.empty(n, dtype=h.dtype)
        term1[:n-shift] = 1.0 - h[shift:]
        term1[n-shift:] = 0.0
        # term2: -h[:n-shift] for indices shift..n-1
        term2 = np.empty(n, dtype=h.dtype)
        term2[shift:] = -h[:n-shift]
        term2[:shift] = 0.0
    else:
        s = -shift
        term1 = np.empty(n, dtype=h.dtype)
        term1[s:] = 1.0 - h[:n-s]
        term1[:s] = 0.0
        term2 = np.empty(n, dtype=h.dtype)
        term2[:n-s] = -h[s:]
        term2[n-s:] = 0.0
    return term1 + term2

import time, numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    # degenerate case
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if s > target_sum:
            lam_low = lam
        else:
            lam_high = lam
        if abs(s - target_sum) < 1e-12:
            break
    return np.clip(v - lam, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_continuous_balanced(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n/2.0)

def corr_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    Lfft = 1 << (L - 1).bit_length()  # next power of two
    H = np.fft.rfft(h, Lfft)
    C = np.fft.rfft(1.0 - h, Lfft)
    prod = H * np.conj(C)
    corr = np.fft.irfft(prod, Lfft)[:L]
    return corr

def compute_c5(h):
    n = h.shape[0]
    return np.max(corr_fft(h)) * (2.0 / n)

def grad_for_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        sz = n - shift
        term1 = np.empty(n, dtype=h.dtype)
        term1[:sz] = 1.0 - h[shift:]
        term1[sz:] = 0.0
        term2 = np.empty(n, dtype=h.dtype)
        term2[shift:] = -h[:sz]
        term2[:shift] = 0.0
    else:
        s = -shift
        sz = n - s
        term1 = np.empty(n, dtype=h.dtype)
        term1[s:] = 1.0 - h[:sz]
        term1[:s] = 0.0
        term2 = np.empty(n, dtype=h.dtype)
        term2[:sz] = -h[s:]
        term2[sz:] = 0.0
    return term1 + term2

def gradient_subgrad(h, corr):
    n = h.shape[0]
    max_corr = np.max(corr)
    eps = 1e-12
    active = np.where(corr >= max_corr - eps)[0]
    shifts = active - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += grad_for_shift(h, s)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)

    return best_h, best_sc

def annealing_refine(h, cur_sc, start, budget_s, anneal_time, rng):
    n = h.shape[0]
    steps = int(anneal_time * 800)
    if steps < 2000:
        steps = 2000
    T0, Tf = 0.02, 1e-6
    for s in range(steps):
        if time.time() - start > budget_s - 0.1:
            break
        frac = s / steps
        T = T0 * (1 - frac) + Tf * frac
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_sc = compute_c5(h_new)
        if new_sc < cur_sc or rng.random() < np.exp((cur_sc - new_sc) / T):
            h, cur_sc = h_new, new_sc
    return h, cur_sc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    scales = [64, 128, 256, 512]

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = project_onto_box_sum(base, 64 / 2.0)
    else:
        base = random_continuous_balanced(64, rng)

    best_h, best_sc = base, compute_c5(base)

    for idx, n in enumerate(scales):
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)
        # random sampling
        remaining = budget_s - (time.time() - start)
        if remaining < 5:
            break
        n_rand = max(200, int(2000 / (2 ** idx)))
        for _ in range(n_rand):
            if time.time() - start > budget_s - 5:
                break
            cand = random_continuous_balanced(n, rng)
            sc = compute_c5(cand)
            if sc < best_sc:
                best_sc, best_h = sc, cand.copy()
        # gradient descent (Adam)
        remaining = budget_s - (time.time() - start)
        if remaining < 5:
            break
        grad_time = remaining * 0.3
        max_iters = int(grad_time * 1200)
        if max_iters < 2000:
            max_iters = 2000
        h_opt, sc_opt = adam_optimize(best_h, max_iters, lr=0.12,
                                      beta1=0.9, beta2=0.999, eps=1e-8,
                                      start=start, budget_s=budget_s, rng=rng)
        if sc_opt < best_sc:
            best_sc, best_h = sc_opt, h_opt.copy()
        # brief annealing
        remaining = budget_s - (time.time() - start)
        if remaining > 3:
            anneal_time = min(remaining * 0.2, 30)
            best_h, best_sc = annealing_refine(best_h, best_sc,
                                               start, budget_s,
                                               anneal_time, rng)

    # final intensive Adam on the current best size
    remaining = budget_s - (time.time() - start)
    if remaining > 5:
        max_iters = int(remaining * 1300)
        max_iters = min(max_iters, 600000)
        h_opt, sc_opt = adam_optimize(best_h, max_iters, lr=0.06,
                                      beta1=0.9, beta2=0.999, eps=1e-8,
                                      start=start, budget_s=budget_s, rng=rng)
        if sc_opt < best_sc:
            best_sc, best_h = sc_opt, h_opt.copy()
        remaining = budget_s - (time.time() - start)
        if remaining > 2:
            best_h, best_sc = annealing_refine(best_h, best_sc,
                                               start, budget_s,
                                               remaining, rng)

    n = best_h.shape[0]
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_sc = compute_c5(best_h)
    return best_h, best_sc, n
