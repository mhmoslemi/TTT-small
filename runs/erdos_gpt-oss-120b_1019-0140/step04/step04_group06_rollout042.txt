def smooth_max_corr_and_grad(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # stabilizing
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    w = w / w_sum
    grad = np.zeros_like(h)
    t0 = n - 1
    for t in range(len(w)):
        weight = w[t]
        if weight == 0.0:
            continue
        k = t - t0
        kk = abs(k)
        if kk == 0:
            grad += weight * (1.0 - 2.0 * h)
        else:
            # term1: shift_right
            grad[:n-kk] += weight * (1.0 - h[kk:])
            # term2: shift_left
            grad[kk:] += weight * (1.0 - h[:n-kk])
            if n > 2*kk:
                # subtract double-counted constant term
                grad[kk:n-kk] -= weight
    return corr, grad

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def smooth_max_corr_and_grad(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    w = w / w_sum
    grad = np.zeros_like(h)
    t0 = n - 1
    for t, weight in enumerate(w):
        if weight == 0.0:
            continue
        k = t - t0
        kk = abs(k)
        if kk == 0:
            grad += weight * (1.0 - 2.0 * h)
        else:
            grad[:n - kk] += weight * (1.0 - h[kk:])
            grad[kk:] += weight * (1.0 - h[:n - kk])
            if n > 2 * kk:
                grad[kk:n - kk] -= weight
    return corr, grad

def smooth_max_optimize(h0, time_left_func, n):
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    h = h0.copy()
    momentum = np.zeros_like(h)
    beta = 0.9
    step0 = 0.5
    alphas = [1.0, 5.0, 20.0, 50.0, 100.0, 200.0]
    for alpha in alphas:
        it = 0
        # allocate iterations based on remaining time
        max_iters = int(min(5000, max(800, time_left_func() * 8)))
        while it < max_iters:
            if time_left_func() < 0.5:
                return best_h, best_score, n
            corr, grad = smooth_max_corr_and_grad(h, alpha)
            momentum = beta * momentum + (1.0 - beta) * grad
            it += 1
            eta = step0 / math.sqrt(it)
            h_new = h - eta * momentum
            h_new = project_onto_box_sum(h_new, n / 2.0)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score, best_h = sc, h_new.copy()
                if best_score <= 0.3808:
                    return best_h, best_score, n
            h = h_new
    return best_h, best_score, n

def grad_descent_fine(h0, time_left_func, rng):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()

        h_new = project_onto_box_sum(h_new, s)

        if sc < best_score:

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    rng = np.random.default_rng(seed)

    final_n = 256
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != final_n:
            base = upsample(base, final_n)
    else:
        if 'thue_morse' in globals():
            base = thue_morse(final_n)
        else:
            base = random_binary_balanced(final_n, rng)
    base = project_onto_box_sum(base, final_n / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # Prepare a few starting points
    init_vectors = [best_h]
    for _ in range(3):
        if time_left() < 5:
            break
        h_rand = random_binary_balanced(final_n, rng)
        h_rand = project_onto_box_sum(h_rand, final_n / 2.0)
        init_vectors.append(h_rand)

    # Optimize each start
    for h0 in init_vectors:
        if time_left() < 5:
            break
        h_opt, sc_opt, _ = smooth_max_optimize(h0, time_left, final_n)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        # short fine‑tune
        if time_left() > 2:
            h_fine, sc_fine = grad_descent_fine(best_h, time_left, rng)
            if sc_fine < best_score:
                best_score, best_h = sc_fine, h_fine.copy()
        if best_score <= 0.3808:
            break

    # optional upscale to 512 if time permits
    if time_left() > 200:
        n2 = 512
        h2 = upsample(best_h, n2)
        h2 = project_onto_box_sum(h2, n2 / 2.0)
        h2_opt, sc2, _ = smooth_max_optimize(h2, time_left, n2)
        if sc2 < best_score:
            best_score, best_h, final_n = sc2, h2_opt.copy(), n2

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
