```python
def compute_c5(h):
    n = h.shape[0]
    # correlation uses mode='full', returns length 2*n-1
    # Multiply by dx = 2.0/n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)
```

```python
def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()
```

```python
def anneal_stage(h0, time_left, rng):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    T0, T1 = 1e-3, 1e-7
    while time_left() > 0.5:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed)  # approximate progress
        T = T0 * (T1 / T0) ** frac
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.uniform(0.05, 0.5)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / max(T, 1e-12)):
            h, cur_score = h_new, sc_new
            if sc_new < best_score:

                if best_score <= 0.3808:
```

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[shift:] += 1.0 - h[:n - shift]
            grad[:n - shift] -= h[shift:]
        else:
            s = -shift
            grad[:n - s] += 1.0 - h[s:]
            grad[s:] -= h[:n - s]
    if shifts:
        grad /= len(shifts)
    return grad

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_balanced(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def grad_descent_stage(h0, time_left, max_iters, rng):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    momentum = np.zeros_like(h)
    beta, step0 = 0.9, 0.6
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
            if best_score <= 0.3808:
                break
        if sc <= cur_score or rng.random() < 0.001:
            h, cur_score = h_new, sc
    return best_h, best_score

def anneal_stage(h0, time_left, rng):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    start = time.time()
    T0, T1 = 1e-3, 1e-7
    while time_left() > 0.5:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed)
        T = T0 * (T1 / T0) ** frac
        i, j = rng.choice(n, size=2, replace=False)
        mx = min(h[i], 1.0 - h[j])
        if mx <= 0:
            continue
        delta = mx * rng.uniform(0.05, 0.5)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / max(T, 1e-12)):
            h, cur_score = h_new, sc_new
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                if best_score <= 0.3808:
                    break
        if rng.random() < 0.02:
            a, b = rng.integers(n), rng.integers(n)
            mx2 = min(h[a], 1.0 - h[b])
            if mx2 > 0:
                d2 = mx2 * rng.uniform(0.05, 0.3)
                h2 = h.copy()
                h2[a] -= d2
                h2[b] += d2
                sc2 = compute_c5(h2)
                if sc2 < cur_score or rng.random() < math.exp(-(sc2 - cur_score) / max(T, 1e-12)):
                    h, cur_score = h2, sc2
                    if sc2 < best_score:
                        best_score, best_h = sc2, h2.copy()
                        if best_score <= 0.3808:
                            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.fromiter(((bin(i).count('1') & 1) for i in range(64)), dtype=float)
    base = project_onto_box_sum(base, base.size / 2.0)

    scales = [128, 256, 512, 1024]
    best_h, best_score = None, float('inf')
    target = 0.3808

    for n in scales:
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        if time_left() > 2:
            max_iters = int(min(200000, max(20000, time_left() * (5000.0 / n))))
            h_gd, sc_gd = grad_descent_stage(base, time_left, max_iters, rng)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()
                if best_score <= target:
                    break
            base = h_gd

        if time_left() > 2:
            h_ann, sc_ann = anneal_stage(base, time_left, rng)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()
                if best_score <= target:
                    break
            base = h_ann

        if best_score <= target:
            break

    if best_h is None:
        best_h = base.copy()
        best_score = compute_c5(best_h)

    # final refinement at finest scale
    if time_left() > 5:
        n = best_h.size
        max_iters = int(min(200000, max(20000, time_left() * (5000.0 / n))))
        h_ref, sc_ref = grad_descent_stage(best_h, time_left, max_iters, rng)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # random restarts if still time
    while time_left() > 6:
        n = best_h.size
        h_rand = random_continuous_balanced(n, rng)
        max_iters = int(min(10000, max(2000, time_left() * 3)))
        h_rand_opt, sc_rand = grad_descent_stage(h_rand, time_left, max_iters, rng)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand_opt.copy()
        if best_score <= target:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
