def adam_descent(h0, max_iter, time_left):

   h = h0.copy(); n = h.size; s = n/2

   best_h = h.copy(); best_score = compute_c5(h)

   m = np.zeros_like(h); v = np.zeros_like(h); beta1=0.9; beta2=0.999; eps=1e-8; lr = 0.1

   for t in range(1, max_iter+1):

       if time_left() <= 0: break

       # compute gradient for max shifts

       corr = np.correlate(h, 1.0 - h, mode='full')

       max_corr = np.max(corr)

       idxes = np.where(corr >= max_corr - 1e-12)[0]

       shifts = idxes - (n - 1)

       grad = compute_gradients_for_shifts(h, shifts)

       m = beta1*m + (1 - beta1)*grad

       v = beta2*v + (1 - beta2)*(grad**2)

       m_hat = m / (1 - beta1**t)

       v_hat = v / (1 - beta2**t)

       step = lr * m_hat / (np.sqrt(v_hat) + eps)

       h_new = h - step

       # optionally add Gaussian noise occasionally

       if t % 2000 == 0 and time_left() > 1:

           h_new += np.random.normal(scale=0.01, size=n)

       h_new = project_onto_box_sum(h_new, s)

       score = compute_c5(h_new)

       if score < best_score:

           best_score = score; best_h = h_new.copy()

       h = h_new

   return best_h, best_score

def solve_mip_binary(n, time_limit):

    import cvxpy as cp

    # Build pair list

    pairs = []

    shift_idx = [[] for _ in range(n)]  # shift zero not needed

    for k in range(1, n):

        for i in range(n - k):

            j = i + k

            pairs.append((i, j))

            shift_idx[k].append(len(pairs) - 1)

    num_pairs = len(pairs)

    h = cp.Variable(n, boolean=True)

    y = cp.Variable(num_pairs, nonneg=True)

    t = cp.Variable()

    constraints = [cp.sum(h) == n/2]

    # linear constraints for each pair

    for idx, (i,j) in enumerate(pairs):

        constraints.append(y[idx] <= h[i])

        constraints.append(y[idx] <= h[j])

        constraints.append(y[idx] >= h[i] + h[j] - 1)

    # shift constraints

    for k in range(1, n):

        idxs = shift_idx[k]

        if idxs:

            constraints.append(cp.sum(cp.hstack([y[i] for i in idxs])) >= t)

    constraints.append(t >= 0) # correlation can't be negative

    # Objective maximize t

    prob = cp.Problem(cp.Maximize(t), constraints)

    # Solve with time limit

    try:

        prob.solve(solver=cp.CBC, verbose=False, max_seconds=time_limit)

    except Exception as e:

        # fallback: no solution

        return None

    if h.value is None:

        return None

    h_bin = np.rint(np.array(h.value)).astype(float)  # ensure 0/1

    # Actually rounding may be needed due to numerical tolerance

    return h_bin

def solve_mip_binary(n, time_limit):
    import cvxpy as cp
    # Build pairs and shift->indices
    pairs = []
    shift_idx = [[] for _ in range(n)]
    for k in range(1, n):
        for i in range(n - k):
            j = i + k
            pairs.append((i, j))
            shift_idx[k].append(len(pairs)-1)
    num_pairs = len(pairs)
    h = cp.Variable(n, boolean=True)
    y = cp.Variable(num_pairs, nonneg=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n/2]
    # pair constraints
    for idx, (i, j) in enumerate(pairs):
        constraints.append(y[idx] <= h[i])
        constraints.append(y[idx] <= h[j])
        constraints.append(y[idx] >= h[i] + h[j] - 1)
    # shift constraints
    for k in range(1, n):
        idxs = shift_idx[k]
        if idxs:
            constraints.append(cp.sum(cp.hstack([y[i] for i in idxs])) >= t)
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        return None
    if h.value is None:
        return None
    h_bin = np.array(h.value).astype(float)
    # Due to numerical tolerance, round small values
    h_bin = np.rint(h_bin)
    return h_bin
```

def adam_descent(h0, max_iter, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2
    for t in range(1, max_iter+1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # occasional perturbation
        if t % 5000 == 0 and time_left() > 1:
            h_new += np.random.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad
```

import time
import numpy as np
import cvxpy as cp

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def resample_arr(arr, new_n):
    old_n = arr.size
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def solve_mip_binary(n, time_limit):
    # Build pair list and shift mapping
    pairs = []
    shift_idx = [[] for _ in range(n)]
    for k in range(1, n):
        for i in range(n - k):
            j = i + k
            pairs.append((i, j))
            shift_idx[k].append(len(pairs) - 1)
    m = len(pairs)
    h = cp.Variable(n, boolean=True)
    y = cp.Variable(m, nonneg=True)
    t = cp.Variable()
    cons = [cp.sum(h) == n / 2]
    for idx, (i, j) in enumerate(pairs):
        cons.append(y[idx] <= h[i])
        cons.append(y[idx] <= h[j])
        cons.append(y[idx] >= h[i] + h[j] - 1)
    for k in range(1, n):
        idxs = shift_idx[k]
        if idxs:
            cons.append(cp.sum(cp.hstack([y[i] for i in idxs])) >= t)
    cons.append(t >= 0)
    prob = cp.Problem(cp.Maximize(t), cons)
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        return None
    if h.value is None:
        return None
    hb = np.rint(np.array(h.value)).astype(float)
    return hb

def adam_descent(h0, max_iter, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.2
    for t in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        if t % 5000 == 0 and time_left() > 1:
            h_new += np.random.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 64
        base = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        base[idx] = 1.0

    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # MIP phase (binary starting point)
    if time_left() > 100:
        h_bin = solve_mip_binary(64, max(5.0, time_left() - 5.0))
        if h_bin is not None:
            h_bin = project_onto_box_sum(h_bin, 64 / 2.0)
            sc_bin = compute_c5(h_bin)
            if sc_bin < best_score:
                best_h, best_score = h_bin.copy(), sc_bin

    # Hierarchical upsampling + local refinement
    for target_n in (128, 256, 512):
        if time_left() < 5:
            break
        if best_h.size >= target_n:
            continue
        h_up = resample_arr(best_h, target_n)
        h_up = project_onto_box_sum(h_up, target_n / 2.0)
        max_iter = int(min(50000, max(1, time_left() * 40)))
        h_ref, sc_ref = adam_descent(h_up, max_iter, time_left)
        if sc_ref < best_score:
            best_h, best_score = h_ref.copy(), sc_ref

    # Final refinement on current resolution
    if time_left() > 2:
        max_iter = int(min(200000, max(1, time_left() * 80)))
        best_h, best_score = adam_descent(best_h, max_iter, time_left)

    # Quick random restarts for any remaining time
    while time_left() > 1.5:
        n = best_h.size
        h_rand = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h_rand[idx] = 1.0
        h_rand = project_onto_box_sum(h_rand, n / 2.0)
        sc = compute_c5(h_rand)
        if sc < best_score:
            best_h, best_score = h_rand, sc
        # short local tweak
        max_iter = int(min(5000, max(1, time_left() * 20)))
        best_h, best_score = adam_descent(best_h, max_iter, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
