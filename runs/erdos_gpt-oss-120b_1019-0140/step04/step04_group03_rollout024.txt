def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            sz = n - s
            grad += 1
            grad[:sz] -= h[s:]
            grad[s:] -= h[:sz]

    if len(shifts) > 0:

    return grad

def gradient_descent(h0, iters, step0, beta, rng, time_budget):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, iters+1):
        if time.time() - start > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # optional random jitter at intervals

            h_new += rng.normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h0, total_steps, rng):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T0 = 0.05
    T_final = 1e-5
    for step in range(total_steps):
        t = T0 * ((total_steps - step) / total_steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        # move mass from i to j
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        # optionally also try opposite direction (j->i) with some probability
        if rng.random() < 0.5:
            # move i->j as above
            pass
        else:
            # move j->i
            max_move2 = min(h[j], 1.0 - h[i])
            if max_move2 <= 0:
                continue
            delta = max_move2 * rng.random()
            i, j = j, i  # swap roles
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / t):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def refine_with_line_search(h, max_iter, rng, time_budget):
    start = time.time()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(max_iter):
        if time.time() - start > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        direction = -grad
        # compute step size via backtracking
        step = 0.5
        improved = False
        for _ in range(5):
            h_new = project_onto_box_sum(h + step * direction, target_sum)
            new_score = compute_c5(h_new)
            if new_score < best_score - 1e-12:
                h = h_new
                best_h = h_new.copy()
                best_score = new_score
                improved = True
                break
            step *= 0.5
        if not improved:
            # small random perturbation to escape
            h = h + rng.normal(scale=0.001, size=n)
            h = project_onto_box_sum(h, target_sum)
    return best_h, best_score

import time, numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low, lam_high = np.min(v) - hi, np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            sz = n - s
            grad += 1.0
            grad[:sz] -= h[s:]
            grad[s:] -= h[:sz]
        else:
            s = -s
            sz = n - s
            grad += 1.0
            grad[:sz] -= h[s:]
            grad[s:] -= h[:sz]
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, iters, step0, beta, rng, time_budget):
    start = time.time()
    iters = min(iters, 200000)
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h, best_sc = h.copy(), compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, iters + 1):
        if time.time() - start > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        idxes = np.where(np.abs(corr - maxc) < 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_sc

def simulated_annealing(h0, steps, rng):
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    cur_sc = compute_c5(h)
    best_h, best_sc = h.copy(), cur_sc
    T0, Tf = 0.05, 1e-5
    for k in range(steps):
        t = T0 * ((steps - k) / steps) + Tf
        i, j = rng.choice(n, 2, replace=False)
        # direction i->j
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0 or rng.random() > 0.5:
            # try opposite direction
            max_move = min(h[j], 1.0 - h[i])
            if max_move <= 0:
                continue
            i, j = j, i
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc = compute_c5(h_new)
        if sc < cur_sc or rng.random() < np.exp(-(sc - cur_sc) / t):
            h, cur_sc = h_new, sc
            if sc < best_sc:
                best_sc, best_h = sc, h_new.copy()
    return best_h, best_sc

def refine_with_line_search(h0, max_iter, rng, time_budget):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h, best_sc = h.copy(), compute_c5(h)
    for _ in range(max_iter):
        if time.time() - start > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        idxes = np.where(np.abs(corr - maxc) < 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        direction = -grad
        step = 0.5
        improved = False
        for _ in range(5):
            h_new = project_onto_box_sum(h + step * direction, target)
            sc = compute_c5(h_new)
            if sc < best_sc - 1e-12:
                h, best_sc, best_h = h_new, sc, h_new.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            # small jitter to escape flat regions
            h = project_onto_box_sum(h + rng.normal(scale=0.001, size=n), target)
    return best_h, best_sc
