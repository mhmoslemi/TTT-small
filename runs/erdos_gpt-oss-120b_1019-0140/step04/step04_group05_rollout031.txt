def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

        n0 = 64
        h0 = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        h0[idx] = 1.0
    h0 = project_onto_box_sum(h0, h0.shape[0] / 2.0)

    # ----- upsample to fine grid -----------------------------------------
    target_n = 512
    h = upsample(h0, target_n) if h0.shape[0] != target_n else h0.copy()
    h = project_onto_box_sum(h, target_n / 2.0)

    best_h = h.copy()
    best_score = compute_c5(best_h)
    target_score = 0.3808

    n = target_n
    dx = 2.0 / n

    # ----- gradient descent (≈70 % of budget) ----------------------------
    grad_budget = budget_s * 0.70
    grad_deadline = start + grad_budget
    lr = 1.0
    min_lr = 0.01
    decay = 0.9995
    iter_count = 0
    cur_score = best_score

    while time.time() < grad_deadline:
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(np.argmax(corr))
        k = idx_max - (n - 1)

        if k >= 0:
            forward = np.concatenate([h[k:], np.zeros(k)])
            backward = np.concatenate([np.zeros(k), h[:n - k]])
        else:
            kk = -k
            forward = np.concatenate([np.zeros(kk), h[:n - kk]])
            backward = np.concatenate([h[kk:], np.zeros(kk)])

        grad = dx * (1.0 - forward - backward)
        h_new = h - lr * grad
        h_new = project_onto_box_sum(h_new, n / 2.0)
        new_score = compute_c5(h_new)

        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
            if best_score <= target_score:
                break

        if new_score < cur_score or rng.random() < 0.01:
            h = h_new
            cur_score = new_score

        lr = max(min_lr, lr * decay)
        iter_count += 1

        # occasional large random perturbation
        if iter_count % 2000 == 0:
            h_rand = np.zeros(n, dtype=float)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_rand[idx] = 1.0
            h = project_onto_box_sum(0.5 * h + 0.5 * h_rand, n / 2.0)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                if best_score <= target_score:
                    break

    # ----- final simulated‑annealing refinement (≈30 % of budget) ------
    remaining = max(0.0, budget_s - elapsed())
    if remaining > 0.5 and best_score > target_score:
        h_sa = best_h.copy()
        cur_score = best_score
        init_temp = 0.005
        sa_start = time.time()
        while time.time() - sa_start < remaining:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h_sa[i], 1.0 - h_sa[j])
            if max_delta <= 0:
                continue
            # linear temperature decay
            t = time.time() - sa_start
            temp = init_temp * (1.0 - t / remaining)
            delta = max_delta * rng.random() * (temp / init_temp) * 0.5
            if delta <= 0:
                continue
            h_sa[i] -= delta
            h_sa[j] += delta
            new_sc = compute_c5(h_sa)
            if new_sc < cur_score:
                cur_score = new_sc
                if new_sc < best_score:
                    best_score = new_sc
                    best_h = h_sa.copy()
                    if best_score <= target_score:
                        break
            else:
                if temp > 0 and rng.random() < np.exp(-(new_sc - cur_score) / temp):
                    cur_score = new_sc
                else:
                    h_sa[i] += delta
                    h_sa[j] -= delta

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
