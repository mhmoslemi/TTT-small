def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_binary(v, target_ones, rng):
    """Adjust binary vector v to have exactly target_ones ones."""
    current = int(v.sum())
    if current == target_ones:
        return v
    diff = target_ones - current
    if diff > 0:
        zeros = np.where(v == 0)[0]
        if diff > zeros.size:
            diff = zeros.size
        choose = rng.choice(zeros, size=diff, replace=False)
        v[choose] = 1
    else:
        ones = np.where(v == 1)[0]
        d = -diff
        if d > ones.size:
            d = ones.size
        choose = rng.choice(ones, size=d, replace=False)
        v[choose] = 0
    return v

def evaluate_fitness(v, cache):
    """Return C5 for vector v, using cache."""
    key = v.tobytes()
    if key in cache:
        return cache[key]
    f = compute_c5(v.astype(float))
    cache[key] = f
    return f

def ga_optimize(v, time_left, rng, cache, target_score):
    """Genetic algorithm on binary vector v."""
    n = v.shape[0]
    target_ones = n // 2
    pop_size = 80 if n <= 256 else 120
    tournament = 3
    p_cross = 0.9
    p_mut = 0.02
    # initial population
    pop = []
    pop_fit = []
    fv = evaluate_fitness(v, cache)
    pop.append(v.copy())
    pop_fit.append(fv)
    while len(pop) < pop_size:
        cand = np.zeros(n, dtype=np.int8)
        cand[:target_ones] = 1
        rng.shuffle(cand)
        cand = repair_binary(cand, target_ones, rng)
        pop.append(cand)
        pop_fit.append(evaluate_fitness(cand, cache))
    idx = np.argsort(pop_fit)
    pop = [pop[i] for i in idx]
    pop_fit = [pop_fit[i] for i in idx]
    best_h = pop[0].copy()
    best_score = pop_fit[0]
    max_iters = 500
    it = 0
    while time_left() > 0.2 and it < max_iters:
        it += 1
        # tournament selection for parent1
        cand_idx = rng.choice(len(pop), size=tournament, replace=False)
        p1_idx = cand_idx[np.argmin([pop_fit[i] for i in cand_idx])]
        p1 = pop[p1_idx]
        # tournament selection for parent2
        cand_idx = rng.choice(len(pop), size=tournament, replace=False)
        p2_idx = cand_idx[np.argmin([pop_fit[i] for i in cand_idx])]
        p2 = pop[p2_idx]
        # crossover
        child = p1.copy()
        if rng.random() < p_cross:
            mask = rng.random(n) < 0.5
            child = np.where(mask, p1, p2)
        # mutation
        mut_mask = rng.random(n) < p_mut
        if mut_mask.any():
            child[mut_mask] = 1 - child[mut_mask]
        # repair
        child = repair_binary(child, target_ones, rng)
        # fitness
        f_child = evaluate_fitness(child, cache)
        # insert if better than worst
        if f_child < pop_fit[-1]:
            pop.append(child)
            pop_fit.append(f_child)
            idx = np.argsort(pop_fit)[:pop_size]
            pop = [pop[i] for i in idx]
            pop_fit = [pop_fit[i] for i in idx]
            if f_child < best_score:
                best_score = f_child
                best_h = child.copy()
        if best_score <= target_score:
            break
    return best_h, best_score

def thue_morse_binary(n):
    """Generate Thue-Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=np.int8, count=n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # initial vector
    if 'initial_h_values' in globals():
        base = np.rint(np.asarray(initial_h_values, dtype=float)).astype(np.int8)
    else:
        base = thue_morse_binary(64)
    base = repair_binary(base, base.size // 2, rng)

    # scaling schedule
    max_n = 512
    scales = [base.shape[0]]
    while scales[-1] < max_n:
        scales.append(scales[-1] * 2)

    best_h = base.copy()
    best_score = compute_c5(best_h.astype(float))
    cache = {}
    cache[best_h.tobytes()] = best_score
    target_score = 0.3808

    for n in scales:
        if best_h.size != n:
            factor = n // best_h.size
            best_h = np.repeat(best_h, factor)
            best_h = repair_binary(best_h, n // 2, rng)
        # run GA at current scale
        local_h, local_score = ga_optimize(best_h, time_left, rng, cache, target_score)
        if local_score < best_score:
            best_score = local_score
            best_h = local_h.copy()
        if best_score <= target_score or time_left() <= 0:
            break

    # final random swap refinement
    while time_left() > 0.1:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        f_cand = evaluate_fitness(cand, cache)
        if f_cand < best_score:
            best_score = f_cand
            best_h = cand.copy()
        if best_score <= target_score:
            break

    best_h_float = best_h.astype(float)
    best_score = compute_c5(best_h_float)
    return best_h_float, best_score, best_h_float.size
