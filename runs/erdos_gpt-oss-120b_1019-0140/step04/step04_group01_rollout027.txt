import time
import numpy as np

# Ensure a projection onto the box [0,1] with a fixed sum is available
if 'project_onto_box_sum' not in globals():
    def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
        """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
        n = v.size
        if s <= lo * n:
            return np.full_like(v, lo)
        if s >= hi * n:
            return np.full_like(v, hi)
        lam_low = np.min(v) - hi
        lam_high = np.max(v) - lo
        def f(l):
            return np.clip(v - l, lo, hi).sum()
        # adjust bounds if needed
        if f(lam_low) < s:
            lam_low = np.min(v) - 10 * hi
        if f(lam_high) > s:
            lam_high = np.max(v) - 10 * lo
        for _ in range(max_iter):
            lam_mid = (lam_low + lam_high) * 0.5
            fm = f(lam_mid)
            if abs(fm - s) < 1e-12:
                break
            if fm > s:
                lam_low = lam_mid
            else:
                lam_high = lam_mid
        return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT (dx = 2/n)."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def _obj_and_grad(h, alpha):
    """Smooth approximation of -min autocorrelation and its gradient."""
    n = h.size
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    # autocorrelation A_k
    A = np.fft.irfft(H * np.conj(H), n=M)[:2 * n - 1]
    # stable soft-min weights
    neg_alpha_A = -alpha * A
    max_neg = np.max(neg_alpha_A)
    w = np.exp(neg_alpha_A - max_neg)
    w_sum = w.sum()
    w /= w_sum
    # split weights for positive and negative shifts
    w_pos = w[n - 1:]                 # shifts 0..n-1
    w_neg = w[:n][::-1]               # shifts 0..-(n-1) (reversed)
    # convolution via FFT
    w_pos_rev = w_pos[::-1]
    Wp = np.fft.rfft(w_pos_rev, n=M)
    Wn = np.fft.rfft(w_neg, n=M)
    conv_rev = np.fft.irfft(Wp * H, n=M)
    conv1 = conv_rev[n - 1:n - 1 + n]         # forward contribution
    conv2 = np.fft.irfft(Wn * H, n=M)[:n]    # backward contribution
    grad = -(conv1 + conv2)
    # objective = (1/alpha) * log sum exp(-alpha * A)
    obj = (np.log(w_sum) + max_neg) / alpha
    return obj, grad

def _binary_swap_anneal(h0, rng, time_left, max_iters=200000,
                       T0=0.3, decay=0.9998, Tmin=1e-5):
    """Swap a 1 and a 0, accepting according to a temperature schedule."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_fft(h)
    best_h, best_score = h.copy(), cur_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def _up_interpolate(arr, new_n):
    """Linear interpolation of arr (assumed on [0,2]) to new_n points."""
    x_old = np.linspace(0.0, 2.0, arr.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # target resolution
    n = 1024
    target_sum = n / 2.0

    # initialise h
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.size != n:
            h = _up_interpolate(init_arr, n)
        else:
            h = init_arr.copy()
    else:
        h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, target_sum)

    best_h = h.copy()
    best_score = compute_c5_fft(best_h)

    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2
    m = np.zeros_like(h)
    v = np.zeros_like(h)

    alpha = 0.5
    alpha_max = 20.0
    alpha_factor = 1.05
    alpha_step = 2000

    it = 1
    # Gradient descent phase (leave ~200 s for later refinements)
    while time_left() > 200.0:
        obj, grad = _obj_and_grad(h, alpha)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, target_sum)

        # periodic best check
        if it % 500 == 0:
            score = compute_c5_fft(h_new)
            if score < best_score:
                best_score, best_h = score, h_new.copy()
                if best_score <= 0.3808:
                    return best_h, best_score, best_h.size

        h = h_new

        # increase smoothing parameter gradually
        if it % alpha_step == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_factor, alpha_max)

        it += 1

    # Post‑gradient binary conversion and annealing
    # Convert to a balanced binary vector (choose top half)
    idx = np.argpartition(-best_h, n // 2)[:n // 2]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    score_bin = compute_c5_fft(h_bin)
    if score_bin < best_score:
        best_score, best_h = score_bin, h_bin.copy()
        if best_score <= 0.3808:
            return best_h, best_score, best_h.size

    # Binary annealing if time permits
    if time_left() > 5.0:
        h_anneal, sc_anneal = _binary_swap_anneal(best_h, rng, time_left,
                                                  max_iters=200000,
                                                  T0=0.3, decay=0.9998, Tmin=1e-5)
        if sc_anneal < best_score:
            best_score, best_h = sc_anneal, h_anneal.copy()
            if best_score <= 0.3808:
                return best_h, best_score, best_h.size

    # Final safety projection and evaluation
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
