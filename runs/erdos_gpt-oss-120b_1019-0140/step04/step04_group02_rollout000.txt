import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse 0/1 sequence of length n (as float)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient of the max‑correlation term (average over all max shifts)."""
    n = h.shape[0]
    # include near‑max shifts for smoother gradient
    idxes = np.where(corr >= max_c - max(tol, max_c * 1e-8))[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if shifts.size:
        grad /= shifts.size
    return grad


def gradient_descent(h0, max_iters, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    # initial score from correlation (avoids extra FFT)
    corr = compute_corr_full_fft(h)
    best_score = np.max(corr) * (2.0 / n)
    momentum = np.zeros_like(h)

    for it in range(1, max_iters + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        score = max_c * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
    return best_h, best_score

    start = time.time()

        return max(0.0, budget_s - (time.time() - start))

    TARGET = 0.3808

    # base construction

    else:
        base = thue_morse(128)

    # upscale to target size

    base = project_onto_box_sum(base, n_target / 2.0)

    best_h = None

    restart_count = 0

    # main multi‑restart gradient descent loop
    while time_left() > 20.0 and restart_count < 12:
        restart_count += 1
        # choose a random start type

            h0 = random_binary_balanced(n_target, rng)
        else:

    while time_left() > 2.0 and best_h is not None:

        corr_bin = compute_corr_full_fft(h_bin)

        if sc_bin < best_score:

            if best_score <= TARGET:

        # small random transfer between two positions
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.2)

            sc_new = np.max(corr_new) * (2.0 / n)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                if best_score <= TARGET:
                    break

    # final safety projection and evaluation
    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(n_target), n_target / 2.0)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_corr = compute_corr_full_fft(best_h)
    best_score = np.max(final_corr) * (2.0 / best_h.size)

    return best_h, best_score, best_h.size
