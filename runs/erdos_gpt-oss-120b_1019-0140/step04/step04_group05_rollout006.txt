```python
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # initial construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # balance random binary
        n0 = 64
        base = np.zeros(n0, dtype=float)
        base[rng.choice(n0, size=n0//2, replace=False)] = 1.0
    # ensure constraints (projection)
    base = project_onto_box_sum(base, base.size/2)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    scales = [64, 128, 256, 512]
    fractions = {64:0.15, 128:0.25, 256:0.3, 512:0.3}
    for n in scales:
        if time_left() <= 0:
            break
        # upsample if needed
        if best_h.size != n:
            # linear interpolation preserving integral? use upsample
            best_h = upsample(best_h, n)
        # project to sum constraint
        best_h = project_onto_box_sum(best_h, n/2)
        # allocate time budget for this scale
        alloc = fractions[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0:
            continue

        # perform simulated annealing from best_h
        cand_h, cand_score = simulated_annealing(best_h, alloc, rng)
        # update global best
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        # set best_h as starting point for next scale
    # final refinement on best_h with any remaining time
    while time_left() > 0.5:
        h_new, sc_new = simulated_annealing(best_h, min(5.0, time_left()), rng)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        else:
            # maybe try local pair move
            n = best_h.size
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i: j+=1
            max_delta = min(best_h[i], 1 - best_h[j])
            if max_delta>0:
                delta = max_delta * rng.random() * 0.5
                h_candidate = best_h.copy()
                h_candidate[i] -= delta
                h_candidate[j] += delta
                sc_candidate = compute_c5_fft(h_candidate)
                if sc_candidate < best_score:

                    best_h = h_candidate.copy()
```

```python
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

```python
def compute_c5_fft(h):
    n = h.shape[0]
    # compute linear correlation using FFT
    N = 2 * n
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)
    corr = np.real(np.fft.ifft(H * np.conj(G)))
    # corr length N, corresponds to shifts -n+1..n-1
    # we take max
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)
```

```python
def simulated_annealing(h0, time_limit, rng):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # parameters
    T0 = 0.01  # initial temperature
    T_end = 1e-6
    max_iters = int(5e6)  # large upper bound
    # probability of global perturb
    p_global = 0.001
    sigma0 = 0.5  # scale for Gaussian perturb
    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        t_frac = it / max_iters
        T = T0 * (T_end / T0) ** t_frac
        if rng.random() < p_global:
            sigma = sigma0 * (T / T0)
            # add noise and project
            h_candidate = project_onto_box_sum(h + sigma * rng.standard_normal(n), n/2)
            sc = compute_c5_fft(h_candidate)
        else:
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            # step size scaled by temperature
            delta = max_delta * rng.random() * (T / T0)
            if delta <= 0:
                continue
            h_candidate = h.copy()
            h_candidate[i] -= delta
            h_candidate[j] += delta
            sc = compute_c5_fft(h_candidate)
        if sc < cur_score:
            cur_score = sc
            h = h_candidate
            if sc < best_score:
                best_score = sc
                best_h = h_candidate.copy()
        else:
            if T > 0:
                accept_prob = np.exp(-(sc - cur_score) / T)
                if rng.random() < accept_prob:
                    cur_score = sc
                    h = h_candidate
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_c5_fft(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx via FFT (full linear correlation)."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)
    corr = np.real(np.fft.ifft(H * np.conj(G)))
    return np.max(corr) * (2.0 / n)


def simulated_annealing(h0, time_limit, rng):
    """SA that preserves sum and box constraints, temperature decays over real time."""
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    T0 = 0.02           # initial temperature
    T_end = 1e-6        # final temperature
    p_global = 0.003    # probability of a global Gaussian perturbation
    sigma0 = 0.3        # base stddev for global moves

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        T = T0 * (T_end / T0) ** frac  # exponential cooling in real time

        if rng.random() < p_global:
            sigma = sigma0 * (T / T0)
            h_candidate = project_onto_box_sum(h + sigma * rng.standard_normal(n), n / 2.0)
        else:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            delta = max_delta * rng.random() * (T / T0)
            if delta <= 0.0:
                continue
            h_candidate = h.copy()
            h_candidate[i] -= delta
            h_candidate[j] += delta

        sc = compute_c5_fft(h_candidate)

        if sc < cur_score:
            cur_score = sc
            h = h_candidate
            if sc < best_score:
                best_score = sc
                best_h = h_candidate.copy()
        else:
            if T > 0.0 and rng.random() < np.exp(-(sc - cur_score) / T):
                cur_score = sc
                h = h_candidate

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # --- initial construction -------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 64
        base = np.zeros(n0, dtype=float)
        base[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    # --- multi‑resolution SA --------------------------------------------------
    scales = [64, 128, 256, 512]
    fractions = {64: 0.15, 128: 0.25, 256: 0.30, 512: 0.30}
    for n in scales:
        if time_left() <= 0.0:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        alloc = fractions[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        cand_h, cand_score = simulated_annealing(best_h, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        # carry forward the improved vector
        best_h = cand_h

    # --- final refinement on current resolution -------------------------------
    if time_left() > 0.5:
        cand_h, cand_score = simulated_annealing(best_h, time_left(), rng)
        if cand_score < best_score:
            best_h = cand_h
            best_score = cand_score

        best_h = upsample(best_h, 1024)
        best_h = project_onto_box_sum(best_h, 1024 / 2.0)
        cand_h, cand_score = simulated_annealing(best_h, time_left(), rng)
        if cand_score < best_score:
            best_h = cand_h
            best_score = cand_score

    # --- final projection and score recomputation ----------------------------
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
