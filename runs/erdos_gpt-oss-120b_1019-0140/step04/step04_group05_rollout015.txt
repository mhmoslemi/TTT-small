import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def anneal_on_grid(h0, time_budget, rng):
    """Simulated annealing on a fixed grid size."""
    TARGET = 0.3808
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.008
    final_temp = 1e-5

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        t_frac = elapsed / time_budget
        temp = init_temp * (1 - t_frac) + final_temp * t_frac
        if best_score <= TARGET:
            break

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue

        # Full transfer occasionally
        if rng.random() < 0.05:
            delta = max_delta
        else:
            delta = max_delta * rng.random() * (temp / init_temp)
            if delta <= 0.0:
                continue

        old_i, old_j = cur_h[i], cur_h[j]
        cur_h[i] = old_i - delta
        cur_h[j] = old_j + delta
        new_score = compute_c5(cur_h)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            if temp > 0.0:
                prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
                if rng.random() < prob:
                    cur_score = new_score
                else:
                    cur_h[i] = old_i
                    cur_h[j] = old_j
            else:
                cur_h[i] = old_i
                cur_h[j] = old_j
    return best_h, best_score

    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # Initial base vector
    if 'initial_h_values' in globals():

    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    # Grid schedule
    grid_sizes = [64, 128, 256, 512, 1024]
    fractions = {64: 0.04, 128: 0.08, 256: 0.12, 512: 0.24, 1024: 0.52}

    while time_left() > 0.5:

        # Random balanced binary trial
        h_rand = random_balanced_binary(n, rng)
        sc_rand = compute_c5(h_rand)

        # Small mass transfer
        i = rng.integers(n)

        if j >= i:

        max_delta = min(best_h[i], 1.0 - best_h[j])

            h_new[i] -= delta

            sc_new = compute_c5(h_new)

                best_h, best_score = h_new, sc_new
                continue

        # Targeted move to reduce the maximal correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift >= 0:
            valid_i = np.arange(0, n - shift)

        if contrib.size > 0:
            i_sel = valid_i[np.argmax(contrib)]
            j_sel = rng.integers(n - 1)

                sc_new = compute_c5(h_new)
                if sc_new < best_score:

                    continue

        if best_score <= TARGET:
            break

    # Final projection to ensure feasibility
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
