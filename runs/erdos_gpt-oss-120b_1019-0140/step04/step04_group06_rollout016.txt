def grad_descent(h0, iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, iters+1):
        if time_left() <= 0:
            break
        # compute active shifts
        raw_corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = raw_corr.max()
        # threshold at 0.99 of max
        thr = max_corr * 0.99
        # get indices where >= thr
        idx = np.where(raw_corr >= thr)[0]
        # limit number
        if idx.size > 15:
            # sort top values
            topk = np.argpartition(raw_corr[idx], -15)[-15:]
            idx = idx[topk]
        active_shifts = (idx - (n - 1)).tolist()
        grad = np.zeros_like(h)
        for shift in active_shifts:
            if shift >= 0:
                # forward shift
                term1 = np.empty_like(h)
                term1[:n-shift] = h[shift:]
                term1[n-shift:] = 0.0
                # backward shift
                term2 = np.empty_like(h)
                term2[shift:] = h[:n-shift]
                term2[:shift] = 0.0
            else:
                sft = -shift
                term1 = np.empty_like(h)
                term1[sft:] = h[:n-sft]
                term1[:sft] = 0.0
                term2 = np.empty_like(h)
                term2[:n-sft] = h[sft:]
                term2[n-sft:] = 0.0
            grad += (1.0 - term1 - term2)
        if active_shifts:
            grad /= len(active_shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        # maybe add small noise occasionally
        if it % 4000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    # ensure projection
    best_h = project_onto_box_sum(best_h, n/2.0)
    best_score = compute_c5(best_h)

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project vector v onto the hyperplane sum(v)=s with box bounds [lo,hi]."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # generate first n values of Thue–Morse sequence (0/1)
    return np.fromiter(((bin(int(v)).count('1') & 1) for v in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_subgradient(h, active_shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in active_shifts:
        term1 = np.zeros_like(h)
        term2 = np.zeros_like(h)
        if shift >= 0:
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if active_shifts:
        grad /= len(active_shifts)
    return grad

def get_active_shifts(h, max_shifts=15, rel_thr=0.99):
    n = h.size
    raw_corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = raw_corr.max()
    thr = max_corr * rel_thr
    idx = np.where(raw_corr >= thr)[0]
    if idx.size > max_shifts:
        # keep the top max_shifts by correlation value
        top = np.argpartition(raw_corr[idx], -max_shifts)[-max_shifts:]
        idx = idx[top]
    return (idx - (n - 1)).tolist()

def grad_descent(h0, iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, iters + 1):
        if time_left() <= 0:
            break
        active = get_active_shifts(h)
        grad = compute_subgradient(h, active)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 4000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_anneal(h0, score0, rng, time_left, budget_s):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), score0
    cur_score = score0
    T0, T1 = 5e-3, 1e-6
    while time_left() > 0:
        frac = 1.0 - time_left() / budget_s
        T = T0 * (T1 / T0) ** frac
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(h[i], 1.0 - h[j]) * rng.uniform(0.1, 0.5)
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            h, cur_score = h_new, sc_new
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        # occasional gradient nudge on the current worst shift
        if rng.random() < 0.03:
            raw_corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = raw_corr.max()
            idx = np.where(raw_corr == max_corr)[0]
            shift = int(rng.choice(idx) - (n - 1))
            grad = compute_subgradient(h, [shift])
            h_tmp = h - 0.02 * grad
            h_tmp = project_onto_box_sum(h_tmp, s)
            sc_tmp = compute_c5(h_tmp)
            if sc_tmp < cur_score:
                h, cur_score = h_tmp, sc_tmp
                if sc_tmp < best_score:
                    best_score, best_h = sc_tmp, h_tmp.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # target resolution (increase if time allows)
    n = 1024
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != n:
            base = upsample(base, n)
    else:
        base = upsample(thue_morse(64), n)
    base = project_onto_box_sum(base, n / 2.0)

    best_h = None
    best_score = float('inf')

    # generate a small pool of starting points
    candidates = [base]
    for _ in range(6):
        if time_left() < 5:
            break
        cand = random_binary_balanced(n, rng)
        cand = project_onto_box_sum(cand, n / 2.0)
        candidates.append(cand)

    # process each candidate
    for h0 in candidates:
        if time_left() < 1:
            break
        iters = int(min(15000, max(3000, time_left() * 6)))
        h_gd, sc_gd = grad_descent(h0, iters, step0=0.6, beta=0.9,
                                   rng=rng, time_left=time_left)
        if sc_gd < best_score:
            best_score, best_h = sc_gd, h_gd.copy()
        if time_left() > 4:
            h_ann, sc_ann = simulated_anneal(best_h, best_score, rng,
                                            time_left, budget_s)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()

    # final refinement loop
    while time_left() > 8:
        # small random mass transfer
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.2)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new, sc_new
        # occasional short gradient descent
        if rng.random() < 0.2:
            iters = int(min(3000, max(1000, time_left() * 2)))
            h_gd, sc_gd = grad_descent(best_h, iters, step0=0.4,
                                       beta=0.9, rng=rng, time_left=time_left)
            if sc_gd < best_score:
                best_h, best_score = h_gd, sc_gd
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
