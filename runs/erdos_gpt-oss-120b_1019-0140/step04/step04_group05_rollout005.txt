def local_search(h, time_limit, rng):
    cur = h.copy()
    cur_score = compute_c5(cur)
    best_local = cur.copy()
    best_local_score = cur_score
    start = time.time()
    init_temp = 0.01
    # define schedule: exponential decay
    while time.time() - start < time_limit:
        # temperature decreasing linearly with elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = init_temp * (1 - frac)
        # propose move: pick i!=j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i: j += 1
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        # delta drawn from uniform scaled by temp
        delta = max_delta * rng.random() * (temp / init_temp + 0.1)
        if delta <= 0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = compute_c5(cur)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local = cur.copy()
        else:
            # accept with probability
            if temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                cur_score = new_score
            else:
                # revert
                cur[i] += delta
                cur[j] -= delta
    return best_local, best_local_score

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def local_search(h, time_limit, rng):
    """Simple SA on a continuous vector, preserving sum and box constraints."""
    n = h.size
    target_sum = n / 2.0
    cur = h.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.01

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = init_temp * (1.0 - frac)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0.0:
            continue
        # delta scaled by temperature (higher temp -> larger moves)
        delta = max_delta * rng.random() * (0.1 + 0.9 * (temp / init_temp))
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = compute_c5(cur)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                cur_score = new_score
            else:
                # revert
                cur[i] += delta
                cur[j] -= delta
    best = project_onto_box_sum(best, target_sum)
    best_score = compute_c5(best)
    return best, best_score


def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    total_budget = budget_s

    def time_left():
        return max(0.0, total_budget - (time.time() - start_time))

    target_sum = n / 2.0

    # ---- initial population ----

    population = []

        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != n:

        base = project_onto_box_sum(base, target_sum)
        population.append(base)

        population.append(h)

    best_score = scores[best_idx]

    # Main DE loop, limited only by time budget

            mutant = project_onto_box_sum(mutant, target_sum)

            # crossover
            rand_vec = rng.random(n)
            trial = np.where(rand_vec < CR, mutant, population[i])
            # ensure at least one mutant component
            j_rand = rng.integers(n)
            trial[j_rand] = mutant[j_rand]
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_onto_box_sum(trial, target_sum)

            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
                    if best_score <= 0.3808:
                        break

            # occasional random reinitialisation to keep diversity
            if rng.random() < 0.02:
                rand_vec = random_balanced_binary(n, rng)
                rand_vec = project_onto_box_sum(rand_vec, target_sum)
                rand_score = compute_c5(rand_vec)
                if rand_score < scores[i]:
                    population[i] = rand_vec
                    scores[i] = rand_score
                    if rand_score < best_score:
                        best_score = rand_score
                        best_h = rand_vec.copy()
                        if best_score <= 0.3808:
                            break
        if best_score <= 0.3808:
            break

    # ---- final local refinement ----
    remaining = time_left()
    if remaining > 0.6:
        refined_h, refined_score = local_search(best_h, remaining, rng)
        if refined_score < best_score:
            best_h, best_score = refined_h, refined_score

    # final projection and score
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
