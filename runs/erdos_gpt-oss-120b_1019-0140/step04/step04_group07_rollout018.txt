def full_corr_fft(h):
    n = h.size
    n2 = 2*n
    # compute FFT of length n2
    H = np.fft.fft(h, n2)
    G = np.fft.fft(1 - h, n2)
    corr = np.fft.ifft(H * np.conj(G)).real
    # Now corr has length n2, but the circular correlation includes wrap; the linear full corr is first 2*n-1 entries of corr, but the zero padding ensures that wrap contributions are zero.
    # However, due to zero padding, the linear correlation of length 2*n-1 is corr[:2*n-1]
    return corr[:2*n-1]

def simulated_annealing(h_start, cur_score, time_left, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_score = cur_score
    temp = 0.02  # initial temperature
    # schedule: temp *= 0.9995 per step, and sigma = temp
    while time_left() > 0.5:  # ensure not too close to end
        sigma = temp
        h_candidate = h + rng.normal(scale=sigma, size=h.size)
        h_candidate = project_onto_box_sum(h_candidate, h_candidate.size / 2.0)
        cand_score = compute_c5_fast(h_candidate)  # fast
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            h = h_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_h, best_score = h_candidate.copy(), cand_score
        temp *= 0.9995
        # maybe stop cooling early if temp becomes too small.
        if temp < 1e-5:
            temp = 1e-5
    return best_h, best_score

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)  # fast
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(max_iters):
        if time_left() <= 0:
            break
        # compute correlation
        corr = full_corr_fft(h)
        max_corr = np.max(corr)
        # find worst shifts indices: shift = idx - (n-1)
        # shift positions: idx from 0 to 2n-2
        # So shift = idx - (n-1)
        # worst idx where corr close to max_corr
        # tolerance maybe 1e-12. But use exact equality via np.isclose
        # Could use np.where(corr >= max_corr - 1e-12)[0] - (n-1)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        # compute gradient
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        # project onto constraint
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fast(h_new)
        # accept if improves
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            # increase step mildly
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            # reduce step
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        # optionally occasional random jitter
        if (it + 1) % 5000 == 0:
            # jitter
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5_fast(h)
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score
    # return best_h, best_score (fast)
    return best_h, best_score

import time
import numpy as np

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as float."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def full_corr_fft(h):
    """Linear full correlation of h with (1-h) via FFT."""
    n = h.size
    n2 = 2 * n
    H = np.fft.fft(h, n2)
    G = np.fft.fft(1.0 - h, n2)
    corr = np.fft.ifft(H * np.conj(G)).real
    return corr[:2 * n - 1]

def compute_c5_fast(h):
    """Fast C5 using FFT (returns max correlation * dx)."""
    n = h.size
    return np.max(full_corr_fft(h)) * (2.0 / n)

def compute_c5(h):
    """Accurate C5 using np.correlate (used for final reporting)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with adaptive step size and momentum."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = full_corr_fft(h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fast(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if (it + 1) % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5_fast(h)
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    """Transfer a small mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5_fast(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # move from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5_fast(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        # move from j to i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5_fast(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def simulated_annealing(h_start, cur_score, time_left, rng):
    """Simple SA: Gaussian perturbations with decreasing temperature."""
    h = h_start.copy()
    best_h = h.copy()
    best_score = cur_score
    temp = 0.02
    decay = 0.9995
    min_temp = 1e-5
    n = h.size
    s = n / 2.0

    while time_left() > 0.5:
        sigma = max(temp, 1e-8)
        cand = h + rng.normal(scale=sigma, size=n)
        cand = project_onto_box_sum(cand, s)
        cand_score = compute_c5_fast(cand)

        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            h = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score
        temp = max(min_temp, temp * decay)
    return best_h, best_score

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h
