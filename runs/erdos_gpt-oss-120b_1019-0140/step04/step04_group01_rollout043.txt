def binary_anneal_fast(rng, h0, time_left_func, dx, required_min_s):
    n = h0.shape[0]
    mid = n - 1
    s = np.correlate(h0, h0, mode='full')
    min_s = s.min()
    best_h = h0.copy()
    best_min_s = min_s
    best_score = 1 - dx * min_s
    # Parameter for temperature schedule
    T = 0.5
    decay = 0.9999
    Tmin = 1e-6
    iterations = 0
    # Use while loop
    while time_left_func() > 0.0:
        # Choose random i with h[i]==1
        # Randomly sample until find.
        while True:
            i = rng.integers(n)
            if h0[i] == 1:
                break
        while True:
            j = rng.integers(n)
            if h0[j] == 0:
                break
        s_old = s.copy()
        rev_h = h0[::-1]
        ti = n - 1 - i
        tj = n - 1 - j

        # Apply updates
        s[ti:ti+n] += h0
        s[tj:tj+n] -= h0
        s[i:i+n] -= rev_h
        s[j:j+n] += rev_h
        s[mid] += 2
        off1 = mid + (i - j)
        off2 = mid + (j - i)
        s[off1] -= 1
        s[off2] -= 1

        new_min_s = s.min()
        delta = new_min_s - min_s

        accept = False
        if delta >= 0:
            accept = True
        else:
            # acceptance probability
            # avoid overflow in exp
            prob = np.exp(delta * dx / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            # perform swap
            h0[i] = 0
            h0[j] = 1
            min_s = new_min_s
            # update best
            if min_s > best_min_s:
                best_min_s = min_s
                best_h = h0.copy()
                best_score = 1 - dx * min_s
            # early exit if target reached
            if best_min_s >= required_min_s:
                return best_h, 1 - dx * best_min_s, best_min_s
        else:
            s[:] = s_old  # revert
        # temperature decay
        if T > Tmin:
            T = max(Tmin, T * decay)
        iterations += 1
        # optionally break after huge iterations? Not needed.

    return best_h, 1 - dx * best_min_s, best_min_s
