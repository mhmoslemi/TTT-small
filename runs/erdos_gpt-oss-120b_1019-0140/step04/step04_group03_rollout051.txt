def compute_c5(h):
    n = h.shape[0]
    g = 1.0 - h
    L = 1 << ((2*n - 1).bit_length())  # next power of two
    # zero pad
    h_padded = np.zeros(L, dtype=np.float64)
    g_padded = np.zeros(L, dtype=np.float64)
    h_padded[:n] = h
    g_padded[:n] = g
    H = np.fft.rfft(h_padded)
    G = np.fft.rfft(g_padded)
    conv = np.fft.irfft(H * G, n=L)
    conv = conv[:2*n - 1]
    max_val = conv.max()
    return max_val * (2.0 / n)

def compute_grads(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            sz = n - s
            term1 = np.empty_like(h)
            term1[:sz] = h[s:]
            term1[sz:] = 0.0
            term2 = np.empty_like(h)
            term2[s:] = h[:sz]
            term2[:s] = 0.0
        else:
            sneg = -s
            sz = n - sneg
            term1 = np.empty_like(h)
            term1[sneg:] = h[:sz]
            term1[:sneg] = 0.0
            term2 = np.empty_like(h)
            term2[:sz] = h[sneg:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def simulated_annealing(h_start, score_start, rng, max_time, n_steps_factor=500.0):
    start = time.time()
    h_cur = h_start.copy()
    cur_score = score_start
    n = h_cur.size
    total_steps = int(min(20000, max(5000, (max_time) * n_steps_factor)))
    T0 = 0.02  # initial temperature can be larger for larger n
    T_final = 1e-6
    for step in range(total_steps):
        if time.time() - start > max_time:
            break
        T = T0 * ((total_steps - step) / total_steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h_cur[i], 1.0 - h_cur[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h_cur.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h_cur = h_new
            cur_score = new_score
    return h_cur, cur_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        s = w.sum()
        if abs(s - target_sum) < 1e-12:
            return w
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    n = h.shape[0]
    g = 1.0 - h
    L = 1 << ((2 * n - 1).bit_length())
    hp = np.zeros(L, dtype=np.float64)
    gp = np.zeros(L, dtype=np.float64)
    hp[:n] = h
    gp[:n] = g
    H = np.fft.rfft(hp)
    G = np.fft.rfft(gp)
    conv = np.fft.irfft(H * G, n=L)
    return conv[: 2 * n - 1]

def compute_c5(h):
    n = h.shape[0]
    conv = compute_corr_fft(h)
    return conv.max() * (2.0 / n)

def compute_grads(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            sz = n - s
            term1 = np.empty_like(h)
            term1[:sz] = h[s:]
            term1[sz:] = 0.0
            term2 = np.empty_like(h)
            term2[s:] = h[:sz]
            term2[:s] = 0.0
        else:
            sneg = -s
            sz = n - sneg
            term1 = np.empty_like(h)
            term1[sneg:] = h[:sz]
            term1[:sneg] = 0.0
            term2 = np.empty_like(h)
            term2[:sz] = h[sneg:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start > max_time:
            break
        conv = compute_corr_fft(h)
        max_val = conv.max()
        shifts = np.where(conv >= max_val - 1e-12)[0] - (n - 1)
        grad = compute_grads(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h_start, score_start, rng, max_time, steps_per_sec=500.0):
    start = time.time()
    h_cur = h_start.copy()
    cur_score = score_start
    n = h_cur.size
    total_steps = int(min(20000, max(5000, (max_time) * steps_per_sec)))
    T0 = 0.02
    T_final = 1e-6
    for step in range(total_steps):
        if time.time() - start > max_time:
            break
        T = T0 * ((total_steps - step) / total_steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(h_cur[i], 1.0 - h_cur[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        h_new = h_cur.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h_cur = h_new
            cur_score = new_score
    return h_cur, cur_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=np.float64, count=n)
