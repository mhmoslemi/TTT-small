import time, math, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def random_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def greedy_improve(h, rng, max_trials=2000):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    for _ in range(max_trials):
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        s = compute_c5(new_h)
        if s < best_score:
            best_h, best_score = new_h, s
    return best_h, best_score

def anneal(start_h, rng, N, time_limit):
    T0 = 0.1
    T1 = 1e-6
    current = start_h.copy()
    current_score = compute_c5(current)
    best_h = current.copy()
    best_score = current_score
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        T = T0 * (T1 / T0) ** frac

        zeros = np.where(current == 0)[0]
