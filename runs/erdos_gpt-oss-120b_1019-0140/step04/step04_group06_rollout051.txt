def local_search(h, start_time, budget_s, rng):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    best_h = h.copy()
    best_corr = max_corr
    T0, T1 = 5e-3, 1e-5
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # identify max shift(s)
        max_inds = np.where(corr == max_corr)[0]
        idx = rng.choice(max_inds)
        k = idx - (n - 1)
        zeros, ones = get_pairs_for_shift(h, k)
        if zeros.size > 0 and ones.size > 0:
            i = rng.choice(zeros)
            j = rng.choice(ones)
        else:
            # random swap
            ones_pos = np.flatnonzero(h)
            zeros_pos = np.flatnonzero(1 - h)
            if ones_pos.size == 0 or zeros_pos.size == 0:
                break
            i = rng.choice(ones_pos)
            j = rng.choice(zeros_pos)
        delta = swap_delta_corr(h, i, j)
        new_corr = corr + delta
        new_max_corr = new_corr.max()
        if new_max_corr <= max_corr:
            accept = True
        else:
            T = T0 * (T1 / T0) ** (elapsed / budget_s)
            accept = rng.random() < math.exp(-(new_max_corr - max_corr) / T)
        if accept:
            # swap entries in h
            h[i], h[j] = h[j], h[i]
            corr = new_corr
            max_corr = new_max_corr
            if max_corr < best_corr:
                best_corr = max_corr
                best_h = h.copy()
        # optional early exit if best_corr already good enough
        if best_corr * (2.0 / n) <= 0.3808:
            break
    return best_h, best_corr

    n = h.shape[0]
    size = 2*n - 1
    delta = np.zeros(size, dtype=np.int16)
    # i contributions:
    ks_i = np.arange(-i, n - i, dtype=np.int32)
    idx_i = ks_i + n - 1
    delta[idx_i] += h[i + ks_i]
    ks_i2 = np.arange(i - (n - 1), i + 1, dtype=np.int32)
    idx_i2 = ks_i2 + n - 1
    delta[idx_i2] += h[i - ks_i2]
    # j contributions with minus sign:
    ks_j = np.arange(-j, n - j, dtype=np.int32)
    idx_j = ks_j + n - 1
    delta[idx_j] -= h[j + ks_j]
    ks_j2 = np.arange(j - (n - 1), j + 1, dtype=np.int32)
    idx_j2 = ks_j2 + n - 1
    delta[idx_j2] -= h[j - ks_j2]
    # extras:
    delta[n - 1] -= 2
    shift = j - i
    delta[shift + n - 1] += 1
    delta[-shift + n - 1] += 1

import time, math, numpy as np

def thue_morse_binary(n):
    seq = np.array([0], dtype=np.int8)
    while seq.size < n:
        seq = np.concatenate([seq, 1 - seq])
    return seq[:n]

def upsample_to_n(arr, n):
    m = arr.shape[0]
    if m == n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_top(arr):
    n = arr.shape[0]
    k = n // 2
    idx = np.argpartition(arr, -k)[-k:]
    out = np.zeros(n, dtype=np.int8)
    out[idx] = 1
    return out

def random_binary_balanced(n, rng):
    out = np.zeros(n, dtype=np.int8)
    out[rng.choice(n, size=n // 2, replace=False)] = 1
    return out

def get_pairs_for_shift(h, k):
    n = h.shape[0]
    if k >= 0:
        rng = np.arange(0, n - k, dtype=np.int32)
        zeros = rng[(h[rng] == 0) & (h[rng + k] == 0)]
        ones = rng[(h[rng] == 1) & (h[rng + k] == 1)]
    else:
        k2 = -k
        rng = np.arange(k2, n, dtype=np.int32)
        zeros = rng[(h[rng] == 0) & (h[rng - k2] == 0)]
        ones = rng[(h[rng] == 1) & (h[rng - k2] == 1)]
    return zeros, ones

def swap_delta_corr(h, i, j):
    n = h.shape[0]
    size = 2 * n - 1
    d = np.zeros(size, dtype=np.int16)

    # i contributions (+)
    ks = np.arange(-i, n - i, dtype=np.int32)
    d[ks + n - 1] += h[i + ks]
    ks2 = np.arange(i - (n - 1), i + 1, dtype=np.int32)
    d[ks2 + n - 1] += h[i - ks2]

    # j contributions (-)
    ks = np.arange(-j, n - j, dtype=np.int32)
    d[ks + n - 1] -= h[j + ks]
    ks2 = np.arange(j - (n - 1), j + 1, dtype=np.int32)
    d[ks2 + n - 1] -= h[j - ks2]

    # extra terms
    d[n - 1] -= 2                      # k=0
    shift = j - i
    d[shift + n - 1] += 1               # k=j-i
    d[-shift + n - 1] += 1              # k=i-j
    return d

def local_search(h, start_time, budget_s, rng):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    best_h = h.copy()
    best_corr = max_corr
    T0, T1 = 5e-3, 1e-6
    no_improve = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # choose a max shift
        max_idxs = np.where(corr == max_corr)[0]
        k_idx = rng.choice(max_idxs)
        k = k_idx - (n - 1)

        zeros, ones = get_pairs_for_shift(h, k)
        if zeros.size and ones.size:
            i = rng.choice(zeros)
            j = rng.choice(ones)
        else:
            ones_pos = np.flatnonzero(h)
            zeros_pos = np.flatnonzero(1 - h)
            if ones_pos.size == 0 or zeros_pos.size == 0:
                break
            i = rng.choice(ones_pos)
            j = rng.choice(zeros_pos)

        delta = swap_delta_corr(h, i, j)
        new_corr = corr + delta
        new_max = new_corr.max()

        if new_max <= max_corr:
            accept = True
        else:
            T = T0 * (T1 / T0) ** (elapsed / budget_s)
            accept = rng.random() < math.exp(-(new_max - max_corr) / T)

        if accept:
            h[i], h[j] = h[j], h[i]
            corr = new_corr
            max_corr = new_max
            no_improve = 0
            if max_corr < best_corr:
                best_corr = max_corr
                best_h = h.copy()
                # early exit if record beat
                if best_corr * (2.0 / n) <= 0.3808:
                    break
        else:
            no_improve += 1

        # occasional random restart to escape traps
        if no_improve >= 5000 and rng.random() < 0.1:
            h = random_binary_balanced(n, rng)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = corr.max()
            no_improve = 0

    return best_h, best_corr

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- initial resolution ---
    n = 512
    if 'initial_h_values' in globals():
        base = upsample_to_n(np.asarray(initial_h_values, dtype=float), n)
        h0 = binarize_top(base)
    else:
        h0 = thue_morse_binary(n)

    # --- first search ---
    best_h, best_corr = local_search(h0, start, budget_s, rng)
    best_score = best_corr * (2.0 / n)

    # --- upscale if time permits ---
    elapsed = time.time() - start
    if budget_s - elapsed > 30:
        n2 = 1024
        h_float = best_h.astype(float)
        up = upsample_to_n(h_float, n2)
        h2 = binarize_top(up)
        best_h2, best_corr2 = local_search(h2, start, budget_s, rng)
        best_score2 = best_corr2 * (2.0 / n2)
        if best_score2 < best_score:
            best_h, best_corr, n, best_score = best_h2, best_corr2, n2, best_score2

    # final projection (already binary)
    h_vals = best_h.astype(float)
    return h_vals, best_score, n
