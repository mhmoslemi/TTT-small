def compute_c5_fft(h, n=None):
    if n is None:

def smooth_gradient_descent(h0, n, rng, time_left, max_iter=200000):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_fft(h, n)
    momentum = np.zeros_like(h)
    beta = 0.9
    step0 = 0.8
    L = 2*n - 1
    L_fft = 1 << ((L-1).bit_length())
    alpha_start = 0.3
    alpha_end = 80.0
    for it in range(1, max_iter+1):
        if time_left() <= 0.0:
            break
        f_h = np.fft.rfft(h, n=L_fft)
        f_a = np.fft.rfft(1 - h, n=L_fft)
        corr_raw = np.fft.irfft(f_h * np.conj(f_a), n=L_fft).real[:L]
        max_corr_raw = corr_raw.max()
        # schedule alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)
        # compute weights
        w = np.exp(alpha * (corr_raw - max_corr_raw))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # weight vectors
        w_pos = w[n-1:]            # length n
        w_rev = w_pos[::-1]        # reversed
        w_pos_fft = np.fft.rfft(w_pos, n=L_fft)
        w_rev_fft = np.fft.rfft(w_rev, n=L_fft)
        # term1: conv of (1-h) with w_rev
        conv1 = np.fft.irfft(f_a * w_rev_fft, n=L_fft).real
        term1 = conv1[n-1 : n-1+n]
        # term2: conv of h with w_pos
        conv2 = np.fft.irfft(f_h * w_pos_fft, n=L_fft).real
        term2 = conv2[:n]
        grad = term1 - term2
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        h_new = project_onto_box_sum(h_new, n/2.0)

        if c5 < best_score:
            best_score = c5
            best_h = h_new.copy()
        h = h_new

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search on lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    step0 = 0.8
    L = 2 * n - 1
    L_fft = 1 << ((L - 1).bit_length())
    alpha_start = 0.3
    alpha_end = 80.0
    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        f_h = np.fft.rfft(h, n=L_fft)
        f_a = np.fft.rfft(1.0 - h, n=L_fft)
        corr = np.fft.irfft(f_h * np.conj(f_a), n=L_fft).real[:L]
        max_corr = corr.max()
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)
        w = np.exp(alpha * (corr - max_corr))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_pos = w[n - 1:]          # non‑negative shifts
        w_rev = w_pos[::-1]
        w_pos_fft = np.fft.rfft(w_pos, n=L_fft)
        w_rev_fft = np.fft.rfft(w_rev, n=L_fft)
        conv1 = np.fft.irfft(f_a * w_rev_fft, n=L_fft).real
        term1 = conv1[n - 1 : n - 1 + n]
        conv2 = np.fft.irfft(f_h * w_pos_fft, n=L_fft).real
        term2 = conv2[:n]
        grad = term1 - term2
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def max_gradient_descent(h0, n, rng, time_left, max_iter=10000):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    step0 = 0.6
    L = 2 * n - 1
    L_fft = 1 << ((L - 1).bit_length())
    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        f_h = np.fft.rfft(h, n=L_fft)
        f_a = np.fft.rfft(1.0 - h, n=L_fft)
        corr = np.fft.irfft(f_h * np.conj(f_a), n=L_fft).real[:L]
        max_c = corr.max()
        grad = compute_grad_max_shifts(h, corr, max_c)

        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    base = globals().get('initial_h_values')
    if base is not None:
        base = np.asarray(base, dtype=float)
    else:
        base = thue_morse(128)
    if base.shape[0] != 128:
        base = upsample(base, 128)
    base = project_onto_box_sum(base, 128 / 2.0)

    best_h = None
    best_score = np.inf
    target = 0.3808

    for n in (128, 256, 512):
        if time_left() <= 2.0:
            break
        cur = upsample(best_h, n) if best_h is not None else (base if base.shape[0] == n else upsample(base, n))
        cur = project_onto_box_sum(cur, n / 2.0)

        restarts = 2 if n < 512 else 1
        for _ in range(restarts):
            if time_left() <= 1.0:
                break
            h0 = cur + rng.normal(scale=0.05, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)

            max_iter_s = 80000 if n == 128 else (120000 if n == 256 else 150000)
            h_opt, sc_opt = smooth_gradient_descent(h0, n, rng, time_left, max_iter=max_iter_s)
            h_opt, sc_opt = max_gradient_descent(h_opt, n, rng, time_left, max_iter=8000)

            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= target:
                    break
        if best_score <= target:
            break

        while time_left() > 4.0:
            if best_h is None:
                break
            h_bin = random_binary_balanced(n, rng)
            sc_bin = compute_c5_fft(h_bin)
            if sc_bin < best_score:
                best_score, best_h = sc_bin, h_bin.copy()
                if best_score <= target:
                    break
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.2)
            if delta > 1e-12:
                h_new = best_h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                sc_new = compute_c5_fft(h_new)
                if sc_new < best_score:
                    best_score, best_h = sc_new, h_new.copy()
                    if best_score <= target:
                        break
        if best_score <= target:
            break

    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(128), 128 / 2.0)
        best_score = compute_c5_fft(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, best_h.size
