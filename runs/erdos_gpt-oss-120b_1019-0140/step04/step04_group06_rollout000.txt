def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smoothed_gradient(h, tau):
    n = h.size
    corr = compute_corr(h)  # length 2n-1
    max_corr = corr.max()
    # shift indexing: shift = idx - (n-1)
    # compute weights
    # To avoid overflow, subtract max_corr.
    w = np.exp((corr - max_corr) / tau)  # shape 2n-1
    # If tau is very small, w may be near zero except max; fine.
    # sum weights
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback to gradient of max shifts
        # find indices where corr == max_corr
        idxs = np.where(corr == max_corr)[0]
        # compute equal weight per max shift
        w = np.zeros_like(corr)
        w[idxs] = 1.0 / len(idxs)
        w_sum = 1.0
    # compute weighted gradient
    grad = np.zeros_like(h)
    # iterate over shift indices
    # For each shift index k, shift = k - (n-1)
    # weight = w[k] / w_sum
    weight_factor = w / w_sum
    # loop
    for idx_shift, weight in enumerate(weight_factor):
        if weight == 0.0:
            continue
        shift = idx_shift - (n - 1)
        if shift >= 0:
            length = n - shift
            # grad[0:length] += weight * (1 - h[shift:])
            grad[:length] += weight * (1.0 - h[shift:])
            # grad[shift:] += weight * (- h[:length])
            grad[shift:] += weight * (-h[:length])
        else:
            t = -shift
            length = n - t
            # grad[t:] += weight * (1 - h[:length])
            grad[t:] += weight * (1.0 - h[:length])
            # grad[:length] += weight * (-h[t:])
            grad[:length] += weight * (-h[t:])
    return grad

    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # helper functions defined above

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # thue_morse balanced
        def thue_morse(n):
            # generate binary sequence via parity of popcount
            return np.fromfunction(lambda i: (np.unpackbits(i.astype(np.uint64).view(np.uint8))[::-1][:].sum() % 2), (n,), dtype=int)
        # Actually simpler: iterative method
        base = np.array([bin(i).count('1') % 2 for i in range(64)], dtype=float)

    # scales: 64->128->256->512

    best_h = None

        m = arr.shape[0]

        x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
        return np.interp(x_new, x_old, arr)

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def smoothed_gradient(h, tau):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    w = np.exp((corr - max_corr) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(corr)
        w[corr == max_corr] = 1.0
        w_sum = 1.0
    w /= w_sum
    grad = np.zeros_like(h)
    for idx_shift, weight in enumerate(w):
        if weight == 0.0:
            continue
        shift = idx_shift - (n - 1)
        if shift >= 0:
            length = n - shift
            grad[:length] += weight * (1.0 - h[shift:])
            grad[shift:] += weight * (-h[:length])
        else:
            t = -shift
            length = n - t
            grad[t:] += weight * (1.0 - h[:length])
            grad[:length] += weight * (-h[t:])
    return grad

def best_local_search(h0, max_iter, tau0, tau_end, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        tau = tau0 * ((tau_end / tau0) ** (it / max_iter))
        grad = smoothed_gradient(h, tau)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
            if best_score <= 0.3808:
                break
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_seq(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse_seq(64)

    best_h = None
    best_score = float('inf')

    for n in (64, 128, 256, 512):
        if time_left() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        max_iter = int(min(15000, max(2000, time_left() * (3000.0 / n))))
        if max_iter < 100:
            max_iter = 100
        h_opt, sc_opt = best_local_search(
            base, max_iter,
            tau0=0.05, tau_end=0.001,
            step0=0.5, beta=0.9,
            rng=rng, time_left=time_left
        )
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        base = h_opt

    if best_h is None:
        best_h = project_onto_box_sum(thue_morse_seq(512), 512 / 2.0)
        best_score = compute_c5(best_h)

    n_final = best_h.shape[0]
    if n_final < 512:
        best_h = upsample(best_h, 512)
        n_final = 512
        best_h = project_onto_box_sum(best_h, n_final / 2.0)

    while time_left() > 5:
        # random binary restart
        h_rand = np.zeros(n_final, dtype=float)
        idx = rng.choice(n_final, size=n_final // 2, replace=False)
        h_rand[idx] = 1.0
        max_iter = int(min(5000, max(1000, time_left() * 1.5)))
        h_rand_opt, sc_rand_opt = best_local_search(
            h_rand, max_iter,
            tau0=0.05, tau_end=0.001,
            step0=0.4, beta=0.9,
            rng=rng, time_left=time_left
        )
        if sc_rand_opt < best_score:
            best_score, best_h = sc_rand_opt, h_rand_opt.copy()
            if best_score <= 0.3808:
                break

        # small Gaussian perturbation around current best
        h_tmp = best_h + rng.normal(scale=0.005, size=n_final)
        h_tmp = project_onto_box_sum(h_tmp, n_final / 2.0)
        sc_tmp = compute_c5(h_tmp)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
            if best_score <= 0.3808:
                break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
