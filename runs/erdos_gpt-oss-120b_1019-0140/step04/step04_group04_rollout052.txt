def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n: return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.ones(n, dtype=h.dtype)
    for s in shifts:
        if s >= 0:

            grad[s:]   -= h[:n-s]
        else:

            grad[sneg:] -= h[:n-sneg]

def descent_with_momentum(h0, max_iters, init_step, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    total = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        corr = linear_corr_fft(h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n-1)
        grad = gradient_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = init_step / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random jitter every 5000 iterations
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, total)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

Incoming parameter s = desired sum.

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo*n:
        return np.full_like(v, lo)
    if s >= hi*n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # Ensure f(lam_lo) >= s and f(lam_hi) <= s (since f decreasing in lambda)
    # Actually f(l) is decreasing as lam increases because v - lam shifts down.
    # So if f(lam_lo) < s, need to decrease lam_lo more.
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 1e5  # large negative
    if f(lam_hi) > s:
        lam_hi = np.max(v) + 1e5
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) <= 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

Now gradient_for_shifts:

def gradient_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.ones(n, dtype=h.dtype)
    for s in shifts:
        if s >= 0:
            grad[:n-s] -= h[s:]
            grad[s:]   -= h[:n-s]
        else:
            sneg = -s
            grad[sneg:] -= h[:n-sneg]
            grad[:n-sneg] -= h[sneg:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def descent_momentum(h0, max_iters, init_step, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    total = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        corr = linear_corr_fft(h)
        max_corr = corr.max()
        # find active shifts
        tol = 1e-12
        shifts = np.where(corr >= max_corr - tol)[0] - (n-1)
        grad = gradient_for_shifts(h, shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = init_step / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, total)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 1e5
    if f(lam_hi) > s:
        lam_hi = np.max(v) + 1e5
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) <= 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def linear_corr_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    return np.fft.irfft(H * np.conj(G), n=L)

def compute_c5_fft(h):
    n = h.shape[0]
    max_corr = linear_corr_fft(h).max()
    return max_corr * (2.0 / n)

def gradient_for_shifts(h, shifts):
    n = h.size
    grad = np.ones(n, dtype=h.dtype)
    for s in shifts:
        if s >= 0:
            grad[:n - s] -= h[s:]
            grad[s:] -= h[:n - s]
        else:
            sneg = -s
            grad[sneg:] -= h[:n - sneg]
            grad[:n - sneg] -= h[sneg:]
    if shifts.size:
        grad /= shifts.size
    return grad

def descent_momentum(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    total = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = linear_corr_fft(h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = gradient_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, total)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binary_hill_climb(h_bin, max_swaps, rng, time_left):
    best_h = h_bin.copy()
    best_score = compute_c5_fft(best_h)
    for _ in range(max_swaps):
        if time_left() <= 0:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_h, best_score = h_new, sc
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h, best_score = None, np.inf
    resolutions = [64, 128, 256, 512]
    target_iters = {64: 20000, 128: 25000, 256: 30000, 512: 50000}
    steps = {64: 0.5, 128: 0.4, 256: 0.35, 512: 0.3}
    beta = 0.95

    for n in resolutions:
        if time_left() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        cur = compute_c5_fft(base)
        if cur < best_score:
            best_h, best_score = base.copy(), cur
        max_it = int(min(target_iters[n], time_left() * 40))
        if max_it > 0:
            h_opt, sc_opt = descent_momentum(base, max_it, steps[n], beta, rng, time_left)
            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
            base = h_opt

    # ensure final size is 512
    n_final = best_h.size if best_h is not None else 512
    if n_final != 512:
        best_h = upsample(best_h, 512) if best_h is not None else random_binary_balanced(512, rng)
        best_h = project_onto_box_sum(best_h, 256.0)
        n_final = 512

    # a few random binary restarts
    for _ in range(5):
        if time_left() < 5:
            break
        it = int(min(20000, time_left() * 20))
        if it <= 0:
            break
        h_rand = random_binary_balanced(n_final, rng)
        h_opt, sc_opt = descent_momentum(h_rand, it, steps.get(n_final, 0.3), beta, rng, time_left)
        if sc_opt < best_score:
            best_h, best_score = h_opt.copy(), sc_opt

    # final momentum refinement
    if time_left() > 5:
        it = int(min(80000, time_left() * 30))
        if it > 0:
            best_h, best_score = descent_momentum(best_h, it, steps.get(best_h.size, 0.3), beta, rng, time_left)

    # binary rounding & hill climbing
    if time_left() > 5:
        h_bin = (best_h >= 0.5).astype(float)
        cur_sum = h_bin.sum()
        target_sum = best_h.size / 2
        diff = int(cur_sum - target_sum)
        if diff > 0:
            ones = np.where(best_h >= 0.5)[0]
            order = ones[np.argsort(best_h[ones])]
            h_bin[order[:diff]] = 0.0
        elif diff < 0:
            diff = -diff
            zeros = np.where(best_h < 0.5)[0]
            order = zeros[np.argsort(-best_h[zeros])]
            h_bin[order[:diff]] = 1.0
        h_bin = project_onto_box_sum(h_bin, target_sum)
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_h, best_score = h_bin, sc_bin
        swaps = int(min(60000, time_left() * 40))
        if swaps > 0:
            best_h, best_score = binary_hill_climb(best_h, swaps, rng, time_left)

    # last projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
