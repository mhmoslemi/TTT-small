def compute_c5_and_shift(h):
    n = h.size
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    c5 = np.max(corr) * (2.0 / n)
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    return c5, shift, corr

def binary_anneal(h, rng, time_left):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_h = h.copy()
    cur_score = best_score
    T0 = 0.5
    decay = 0.99995
    Tmin = 1e-7
    it = 0
    while time_left()>0.01:
        # compute T
        T = max(Tmin, T0 * (decay ** it))
        # pick random swap
        while True:
            i = rng.integers(n)
            if cur_h[i] == 1:
                break
        while True:
            j = rng.integers(n)
            if cur_h[j] == 0:
                break
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_score = compute_c5_fft(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
    return best_h, best_score

def gradient_descent_momentum_with_time(h0, time_left, step0=0.1, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left()>0:
        # compute correlation and max shift(s)
        c5, shift, _ = compute_c5_and_shift(h)
        # compute gradient for this shift
        grad = compute_gradients_for_shifts(h, np.array([shift]))
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        # step size decays
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project back onto feasible set
        h_new = project_onto_box_sum(h_new, target_sum)
        # compute new score
        new_score = compute_c5_fft(h_new)
        # accept if better, else optionally accept worse? maybe just accept if better.
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # Find lam such that sum clip(v - lam) = s
    # Use bisection with bounds based on v.
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure bounds bracket solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def binary_anneal(h0, rng, time_left_func, max_iters=None, T0=0.5, decay=0.99995, Tmin=1e-7):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    it = 0
    while time_left_func() > 0.0:
        # compute temperature
        T = max(Tmin, T0 * (decay ** it))
        # select random ones and zeros
        # pick i with h[i]==1
        while True:
            i = rng.integers(n)
            if h[i] == 1:
                break
        while True:
            j = rng.integers(n)
            if h[j] == 0:
                break
        # swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # acceptance probability
            prob = math.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
        it += 1
        if max_iters is not None and it >= max_iters:
            break
    return best_h, best_score

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.2, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        # compute c5 and active shifts
        c5, shifts, _ = compute_c5_and_shifts(h)
        # compute gradient for these shifts
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # step size
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        new_score = compute_c5_fft(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

        h_up_bin[idx_up] = 1.0
        best_h_up = h_up_bin.copy()
        best_score_up = compute_c5_fft(best_h_up)

        if time_left() > 150.0:
            h_up_ann, sc_up_ann = binary_anneal(best_h_up, rng, time_left)
            if sc_up_ann < best_score_up:
                best_h_up, best_score_up = h_up_ann, sc_up_ann

        if time_left() > 100.0:
            h_up_gd, sc_up_gd = gradient_descent_momentum_with_time(best_h_up, time_left)
            if sc_up_gd < best_score_up:
                best_h_up, best_score_up = h_up_gd, sc_up_gd

        if best_score_up < best_score:
            best_h, best_score = best_h_up, best_score_up

    # Phase E: random binary restarts (short anneals)
    while time_left() > 30.0:
        rand_h = np.zeros(best_h.shape[0], dtype=float)
        ones = rng.choice(best_h.shape[0], size=best_h.shape[0] // 2, replace=False)
        rand_h[ones] = 1.0
        sub_time = min(5.0, time_left())
        sub_start = time.time()
        def sub_time_left():
            return max(0.0, sub_time - (time.time() - sub_start))
        h_rand, sc_rand = binary_anneal(rand_h, rng, sub_time_left)
        if sc_rand < best_score:
            best_h, best_score = h_rand, sc_rand
        if time_left() < 5.0:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
