                    accept = True

import time
import numpy as np
from scipy import optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure the interval brackets s
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def slsqp_optimize_candidate(h0, max_iter=1500):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_scaled_fft(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    if not res.success:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

def binary_swap_anneal_multi(h0, rng, time_left_func,
                             max_total_iters=500000, restarts=3):
    n = h0.size
    # Ensure binary start
    if not np.all(np.isin(h0, [0.0, 1.0])):
        k = n // 2
        idx = np.argpartition(-h0, k)[:k]
        h_bin = np.zeros_like(h0)
        h_bin[idx] = 1.0
    else:
        h_bin = h0.copy()
    best_h = h_bin.copy()
    best_score = compute_c5_fft(best_h)
    for r in range(restarts):
        if time_left_func() < 0.1:
            break
        if r == 0:
            h = h_bin.copy()
        else:
            h = np.zeros(n, dtype=float)
            idx = rng.choice(n, size=n // 2, replace=False)
            h[idx] = 1.0
        current_score = compute_c5_fft(h)
        T = 0.5
        Tmin = 1e-5
        decay = 0.99997
        it = 0
        while time_left_func() > 0.02 and it < max_total_iters:
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            new_score = compute_c5_fft(h)
            delta = new_score - current_score
            accept = False
            if delta < 0:
                accept = True
            else:
                if T > Tmin and rng.random() < np.exp(-delta / T):
                    accept = True
            if accept:
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
            T = max(Tmin, T * decay)
            it += 1
    return best_h, best_score

def hill_climb_swap_improve(h, rng, time_left_func, max_attempts=20000):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    attempts = 0
    no_improve = 0
    while time_left_func() > 0.0 and attempts < max_attempts and no_improve < 2000:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best_h[i] = 0.0
        best_h[j] = 1.0
        new_score = compute_c5_fft(best_h)
        if new_score < best_score:
            best_score = new_score
            no_improve = 0
        else:
            best_h[i] = 1.0
            best_h[j] = 0.0
            no_improve += 1
        attempts += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    n_large = 512

    # Initial vector
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.size != n_large:
            init_arr = downsample(init_arr, n_large)
        h_current = project_onto_box_sum(init_arr, n_large / 2.0)
    else:
        h_current = np.zeros(n_large, dtype=float)
        idx = rng.choice(n_large, size=n_large // 2, replace=False)
        h_current[idx] = 1.0

    best_h = h_current.copy()
    best_score = compute_c5_fft(best_h)
    if best_score <= target:
        return best_h, best_score, best_h.size

    # Phase 1: small exploration (n_small = 128)
    n_small = 128
    h_small = downsample(best_h, n_small)
    h_small = project_onto_box_sum(h_small, n_small / 2.0)
    sc_small = compute_c5_fft(h_small)
    if sc_small < best_score:
        best_score, best_h = sc_small, upsample(h_small, n_large)
        best_h = project_onto_box_sum(best_h, n_large / 2.0)

    for _ in range(6):
        if time_left() < 5.0:
            break
        h_rand = random_binary_balanced(n_small, rng)
        h_opt, sc_opt = gradient_descent_momentum_multi(
            h_rand, rng, time_left,
            max_iters=50000, restarts=1,
            step0=0.5, beta=0.9,
            epsilon_rel=0.01, max_shifts=5)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = upsample(h_opt, n_large)
            best_h = project_onto_box_sum(best_h, n_large / 2.0)
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 2: medium resolution (n_mid = 256)
    n_mid = 256
    h_mid = downsample(best_h, n_mid)
    h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
    sc_mid = compute_c5_fft(h_mid)
    if sc_mid < best_score:
        best_score, best_h = sc_mid, upsample(h_mid, n_large)
        best_h = project_onto_box_sum(best_h, n_large / 2.0)
    if best_score <= target:
        return best_h, best_score, best_h.size

    if time_left() > 5.0:
        h_mid_opt, sc_mid_opt = slsqp_optimize_candidate(h_mid, max_iter=2000)
        if sc_mid_opt < best_score:
            best_score = sc_mid_opt
            best_h = upsample(h_mid_opt, n_large)
            best_h = project_onto_box_sum(best_h, n_large / 2.0)
        if best_score <= target:
            return best_h, best_score, best_h.size

    if time_left() > 2.0:
        h_ref, sc_ref = gradient_descent_momentum_multi(
            h_mid, rng, time_left,
            max_iters=100000, restarts=2,
            step0=0.5, beta=0.9,
            epsilon_rel=0.01, max_shifts=5)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = upsample(h_ref, n_large)
            best_h = project_onto_box_sum(best_h, n_large / 2.0)
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 3: large resolution
    h_large = downsample(best_h, n_large) if best_h.size != n_large else best_h.copy()
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    sc_large = compute_c5_fft(h_large)
    if sc_large < best_score:
        best_score, best_h = sc_large, h_large.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    if time_left() > 5.0:
        h_large_opt, sc_large_opt = slsqp_optimize_candidate(h_large, max_iter=2000)
        if sc_large_opt < best_score:
            best_score, best_h = sc_large_opt, h_large_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    if time_left() > 2.0:
        h_large_ref, sc_large_ref = gradient_descent_momentum_multi(
            h_large, rng, time_left,
            max_iters=100000, restarts=2,
            step0=0.5, beta=0.9,
            epsilon_rel=0.01, max_shifts=5)
        if sc_large_ref < best_score:
            best_score, best_h = sc_large_ref, h_large_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Binary annealing passes
    if time_left() > 1.0:
        h_bin_ref, sc_bin_ref = binary_swap_anneal_multi(best_h, rng, time_left,
                                                         max_total_iters=500000,
                                                         restarts=3)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Local hill climbing on binary solution
    if time_left() > 1.0:
        h_hc, sc_hc = hill_climb_swap_improve(best_h, rng, time_left, max_attempts=20000)
        if sc_hc < best_score:
            best_score, best_h = sc_hc, h_hc.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Final brief SLSQP polish
    if time_left() > 0.5:
        h_final, sc_final = slsqp_optimize_candidate(best_h, max_iter=500)
        if sc_final < best_score:
            best_score, best_h = sc_final, h_final.copy()

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
