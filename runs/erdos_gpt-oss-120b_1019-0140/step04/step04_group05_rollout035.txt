def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n: return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    interpolated = np.interp(x_new, x_old, arr)
    # Ensure within [0,1] and project sum
    target_sum = new_n / 2
    return project_onto_box_sum(interpolated, target_sum)
```

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample(arr, new_n):
    """Linear interpolation to a larger size, then enforce constraints."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    interp = np.interp(x_new, x_old, arr)
    return project_onto_box_sum(interp, new_n / 2.0)


def worst_shift_step(h):
    """Perform a single mass‑transfer move that reduces the worst shift."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)

    # shift = 0: try to equalise values
    if shift == 0:
        i_cand = np.where(h > 1e-12)[0]
        j_cand = np.where(h < 1 - 1e-12)[0]
        if i_cand.size == 0 or j_cand.size == 0:
            return h, np.max(corr) * dx
        i = i_cand[np.argmax(h[i_cand])]
        j = j_cand[np.argmin(h[j_cand])]
        delta = min(h[i], 1.0 - h[j])
        if delta <= 0.0:
            return h, np.max(corr) * dx
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        return h_new, np.max(new_corr) * dx

    # non‑zero shift
    if shift > 0:

def deterministic_improve(h, time_budget):
    """Iteratively apply worst‑shift moves while they improve the score."""
    start = time.time()
    cur_h = h
    cur_score = compute_c5(cur_h)

            cur_score = new_score
        else:

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial vector
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = thue_morse(64)
    h0 = project_onto_box_sum(h0, h0.shape[0] / 2.0)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    grids = [64, 128, 256, 512]

    for g in grids:
        if time_left() < 0.5:
            break
        if best_h.size != g:
            best_h = upsample(best_h, g)
        best_h = project_onto_box_sum(best_h, g / 2.0)

        # focused deterministic refinement
        alloc = min(30.0, time_left() * 0.4)
        if alloc > 0.1:
            refined_h, refined_score = deterministic_improve(best_h, alloc)
            if refined_score < best_score:
                best_h, best_score = refined_h, refined_score
            best_h = refined_h  # continue from refined state

        # a few random restarts at this resolution
        for _ in range(3):
            if time_left() < 0.2:
                break
            if rng.random() < 0.5:
                cand = random_balanced_binary(g, rng)
            else:
                cand = project_onto_box_sum(rng.random(g), g / 2.0)
            local_budget = min(5.0, time_left() * 0.3)
            cand, cand_score = deterministic_improve(cand, local_budget)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score

    # final exploration using remaining time
    while time_left() > 0.6:
        n = best_h.size
        if rng.random() < 0.5:
            trial = random_balanced_binary(n, rng)
        else:
            trial = project_onto_box_sum(rng.random(n), n / 2.0)
        trial, trial_score = deterministic_improve(trial, min(2.0, time_left() * 0.5))
        if trial_score < best_score:
            best_h, best_score = trial, trial_score
        # try to improve the current best a bit more
        if time_left() > 0.2:
            best_h, best_score = deterministic_improve(best_h, min(1.0, time_left() * 0.2))

    # final projection and exact scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
