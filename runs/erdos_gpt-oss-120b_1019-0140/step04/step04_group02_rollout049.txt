def adam_descent(h0, n_iter, step0, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for t in range(1, n_iter+1):
        if time_left() <= 0: break
        corr = compute_corr_full_fft(h)  # need also compute gradient maybe
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - step0 * m_hat / (np.sqrt(v_hat) + eps)

        h = h_new
    return best_h, best_score

def binary_hill_climb(h_cont, n_iter, rng, time_left):
    # Create binary vector with same sum (n/2 ones)
    n = h_cont.size
    # choose top half indices
    idx = np.argsort(h_cont)[-n//2:]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    best_h = b.copy()
    best_score = compute_c5_fft(b)
    for it in range(n_iter):
        if time_left() <= 0: break
        # random swap
        i = rng.choice(np.where(best_h == 1)[0])
        j = rng.choice(np.where(best_h == 0)[0])
        # swap
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        sc = compute_c5_fft(new_h)
        if sc < best_score:
            best_h = new_h
            best_score = sc
    return best_h, best_score

    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    rng = np.random.default_rng(seed)

    # base initialization
    if 'initial_h_values' in globals():
        base0 = np.asarray(initial_h_values, dtype=float)
    else:
        base0 = thue_morse(128)

    # scales
    scales = [128, 256, 512, 768]  # maybe up to 768, less than 1000.
    best_h = None
    best_score = np.inf
    TARGET = 0.3808

    for n in scales:
        if time_left() <= 0.5:
            break
        # upsample base
        if best_h is not None:
            base = upsample(best_h, n)

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    return np.fft.ifft(fh * np.conj(f1mh)).real

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.size
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts):
        grad /= len(shifts)
    return grad

def adam_descent(h0, n_iter, step0, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    pow1 = beta1
    pow2 = beta2
    for t in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - pow1)
        v_hat = v / (1.0 - pow2)
        step = step0 / np.sqrt(t)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        if t % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=h.size)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        pow1 *= beta1
        pow2 *= beta2
    return best_h, best_score

def binary_hill_climb(h_cont, n_iter, rng, time_left):
    n = h_cont.size
    half = n // 2
    top_idx = np.argpartition(h_cont, -half)[-half:]
    b = np.zeros(n, dtype=float)
    b[top_idx] = 1.0
    best_h = b
    best_score = compute_c5_fft(best_h)
    for _ in range(n_iter):
        if time_left() <= 0.0:
            break
        ones = np.where(best_h == 1.0)[0]
        zeros = np.where(best_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        sc = compute_c5_fft(new_h)
