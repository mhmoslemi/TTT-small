import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5_bin(h):
    """C5 for binary step function h."""
    n = h.size
    # use int16 to avoid overflow
    corr = np.correlate(h.astype(np.int16), h.astype(np.int16), mode='full')
    min_ov = corr.min()
    return 1.0 - 2.0 * min_ov / n

def shift_array(arr, k):
    """Shift arr by k (positive = right, negative = left) with zero padding."""
    n = arr.size
    if k == 0:
        return arr
    if k > 0:
        if k >= n:
            return np.zeros_like(arr)
        return np.concatenate((np.zeros(k, dtype=arr.dtype), arr[: n - k]))
    else:
        k = -k
        if k >= n:
            return np.zeros_like(arr)
        return np.concatenate((arr[k:], np.zeros(k, dtype=arr.dtype)))

def binary_from_initial(init):
    """Convert any vector to binary with exactly half ones."""
    n = init.shape[0]
    target = n // 2
    idx = np.argpartition(-init, target)[:target]
    h = np.zeros(n, dtype=np.uint8)
    h[idx] = 1
    return h

def thue_morse_seq(n):
    """Thue–Morse binary sequence of length n (n must be a power of two)."""
    seq = np.empty(n, dtype=np.uint8)
    for i in range(n):
        seq[i] = (i.bit_count() & 1)
    return seq

def upsample_binary(arr, new_n):
    """Upsample binary array by integer factor (repeat each entry)."""
    factor = new_n // arr.size
    if factor * arr.size != new_n:
        raise ValueError("new_n must be a multiple of arr.size")
    return np.repeat(arr, factor)

# ---------- Main entry point ----------

    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # resolution schedule

    # ----- initial vector -----

            init = np.interp(x_new, x_old, init)
        h = binary_from_initial(init)
    else:
        h = thue_morse_seq(scales[0])

    best_h = h.copy()
    best_c5 = compute_c5_bin(best_h)

    # ----- iterative refinement over scales -----
    for n in scales:
        if time_left() < 2.0:
            break
        if best_h.shape[0] != n:
            best_h = upsample_binary(best_h, n)
        h = best_h.copy()

        while time_left() > 0.5:

            min_ov = corr.min()
