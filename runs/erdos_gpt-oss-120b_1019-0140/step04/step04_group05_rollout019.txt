def binary_sa(h, time_limit, rng, init_temp, final_temp=1e-5):
    current = h.copy()
    cur_score = compute_c5(current)
    best = current.copy()
    best_score = cur_score
    start = time.time()
    # exponential schedule: temp = init_temp * (final_temp/init_temp)^(elapsed/total)
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # compute temperature
        frac = elapsed / time_limit
        temp = init_temp * (final_temp / init_temp) ** frac
        # propose swap
        i = rng.integers(n)
        if current[i] == 0:
            continue
        j = rng.integers(n)
        if current[j] == 1:
            continue
        # make swap
        current[i] = 0.0
        current[j] = 1.0
        new_score = compute_c5(current)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if temp > 0:
                prob = np.exp(-delta / temp)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = current.copy()
        else:
            # revert swap
            current[i] = 1.0
            current[j] = 0.0
    return best, best_score

def continuous_local_search(h, time_limit, rng):
    n = h.shape[0]
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    # parameters
    init_alpha = 0.1
    min_alpha = 1e-5
    alpha = init_alpha
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # compute gradient at max shift
        grad = gradient_of_max_corr(cur_h)
        # try step
        step = -alpha * grad
        # ensure step not exceed bounds: compute tentative h
        h_new = cur_h + step
        # project onto box and sum
        h_new = project_onto_box_sum(h_new, n/2.0)
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            cur_h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score
            # possibly increase alpha slightly for faster convergence?
            alpha = min(alpha * 1.1, init_alpha)
        else:
            # reduce alpha
            alpha *= 0.5
            if alpha < min_alpha:
                # random perturbation to escape

                h_new = project_onto_box_sum(cur_h + noise, n/2.0)
                cur_h = h_new
                cur_score = compute_c5(cur_h)
                alpha = init_alpha
        # continue loop
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - total) < 1e-12:
            return proj
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq

def gradient_of_max_corr(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    k = lag
    term1 = np.zeros(n, dtype=float)
    term2 = np.zeros(n, dtype=float)
    if k >= 0:
        if k < n:
            i = np.arange(n - k)
            term1[i] = 1.0 - h[i + k]
            i2 = np.arange(k, n)
            term2[i2] = h[i2 - k]
    else:
        kp = -k
        if kp < n:
            i = np.arange(kp, n)
            term1[i] = 1.0 - h[i - kp]
            i2 = np.arange(0, n - kp)
            term2[i2] = h[i2 + kp]
    return term1 - term2

def binary_sa(h_bin, time_limit, rng, init_temp, final_temp=1e-6):
    n = h_bin.shape[0]
    cur = h_bin.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        temp = init_temp * (final_temp / init_temp) ** frac
        i = rng.integers(n)
        if cur[i] == 0.0:
            continue
        j = rng.integers(n)
        if cur[j] == 1.0:
            continue
        cur[i] = 0.0
        cur[j] = 1.0
        new_score = compute_c5(cur)
        delta = new_score - cur_score
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if temp > 0.0:
                if rng.random() < np.exp(-delta / temp):
                    accept = True
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
    return best, best_score

def continuous_local_search(h_start, time_limit, rng):
    n = h_start.shape[0]
    cur = h_start.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    init_alpha = 0.07
    min_alpha = 1e-5
    alpha = init_alpha
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        grad = gradient_of_max_corr(cur)
        step = -alpha * grad
        h_new = project_onto_box_sum(cur + step, n / 2.0)
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            cur = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
            alpha = min(alpha * 1.1, init_alpha)
        else:
            alpha *= 0.5
            if alpha < min_alpha:
                sigma = 0.015
                noise = rng.normal(scale=sigma, size=n)
                cur = project_onto_box_sum(cur + noise, n / 2.0)
                cur_score = compute_c5(cur)
                alpha = init_alpha
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    scales = [64, 128, 256, 512]
    fractions = {64: 0.10, 128: 0.15, 256: 0.25, 512: 0.50}
    total_budget = budget_s
    best_h = base.copy()
    best_score = compute_c5(best_h)

    for n in scales:
        if time_left() <= 0.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        alloc = fractions[n] * total_budget
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        # binary start from top half of base
        bin_h = np.zeros(n, dtype=float)
        idx = np.argpartition(base, -n // 2)[-n // 2:]
        bin_h[idx] = 1.0

        # split time between binary SA and continuous refinement
        sa_time = alloc * 0.6
        cont_time = alloc - sa_time

        best_bin, best_bin_score = binary_sa(bin_h, sa_time, rng, init_temp=0.05)
        if best_bin_score < best_score:
            best_h = best_bin.copy()
            best_score = best_bin_score

        h_cont, cont_score = continuous_local_search(best_h, cont_time, rng)
        if cont_score < best_score:
            best_h = h_cont.copy()
            best_score = cont_score

        base = best_h.copy()

    # final exploitation with any remaining budget
    remaining = time_left()
    if remaining > 0.5:
        # binary SA on current best (thresholded)
        n = best_h.shape[0]
        bin_h = np.zeros(n, dtype=float)
        idx = np.argpartition(best_h, -n // 2)[-n // 2:]
        bin_h[idx] = 1.0
        best_bin, best_bin_score = binary_sa(bin_h, remaining, rng, init_temp=0.04)
        if best_bin_score < best_score:
            best_h = best_bin
            best_score = best_bin_score

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
