def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    # compute initial score
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        # time check
        if time.time() - start >= max_time:
            break
        # correlation
        corr = correlation_raw(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # step size
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional noise injection every 5000 it
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # projection
        h_new = project_onto_box_sum(h_new, target_sum)
        # compute score
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_refine(h, rng, max_time):
    start = time.time()
    n = h.shape[0]
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    # Determine total steps based on max_time
    # approximate step_time as correlation: say 0.00005 sec, safe allocate steps = int(max_time/0.00006)
    # But we can also set total_steps = int(min(200000, max(5000, max_time * 2000))) which yields ~2000 steps per sec
    total_steps = int(min(200000, max(5000, max_time * 2000)))
    T0 = 0.02
    T_final = 1e-6
    for step in range(total_steps):
        if time.time() - start >= max_time:
            break
        T = T0 * (1 - step / total_steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def correlation_raw(h):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    return corr[:2 * n - 1]

def compute_c5(h):
    n = h.shape[0]
    max_corr = correlation_raw(h).max()
    return max_corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = correlation_raw(h)
        max_corr = corr.max()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        if len(shifts) > 20:
            shifts = rng.choice(shifts, size=20, replace=False)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = correlation_raw(h_new).max() * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_refine(h, rng, max_time):
    start = time.time()
    n = h.shape[0]
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    total_steps = int(min(200000, max(5000, max_time * 2000)))
    T0 = 0.02
    T_final = 1e-6
    for step in range(total_steps):
        if time.time() - start >= max_time:
            break
        T = T0 * (1 - step / total_steps) + T_final
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    final_n = 1024
    scales = [64, 128, 256, 512, final_n]

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    best_h = None
    best_score = np.inf

    for idx_scale, n in enumerate(scales):
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # random binary sampling
        sample_limit = 200 if n <= 256 else 80
        for _ in range(sample_limit):
            if time.time() - start >= budget_s - 1.0:
                break
            h_rand = random_binary_balanced(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score, best_h = sc, h_rand.copy()

        # gradient descent restarts
        rem = budget_s - (time.time() - start)
        if rem <= 0:
            break
        remaining_scales = len(scales) - idx_scale
        time_for_gd = rem / remaining_scales * 0.6
        max_iters = int(min(300000, max(20000, time_for_gd * 2000)))
        for r in range(3):
            if time.time() - start >= budget_s - 5.0:
                break
            h0 = base.copy()
            if r > 0:
                h0 += rng.normal(scale=0.05, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, n / 2.0)
            gd_time = min(time_for_gd / 3, budget_s - (time.time() - start))
            h_opt, sc_opt = gradient_descent_momentum_custom(
                h0, n_iter=max_iters,
                step0=0.8, beta=0.95,
                rng=rng, max_time=gd_time
            )
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt

        # simulated annealing refinement
        if best_h is not None and best_h.size == n and time.time() - start < budget_s - 5.0:
            anneal_time = budget_s - (time.time() - start)
            h_ref, sc_ref = anneal_refine(best_h, rng, anneal_time)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()
            base = best_h

    # final refinement if time remains
    if best_h is not None and time.time() - start < budget_s - 5.0:
        n = best_h.size
        remaining = budget_s - (time.time() - start)
        gd_time = remaining * 0.5
        max_iters = int(min(500000, max(50000, gd_time * 3000)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=max_iters,
            step0=0.4, beta=0.95,
            rng=rng, max_time=gd_time
        )
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            h_ref, sc_ref = anneal_refine(best_h, rng, remaining)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref

    if best_h is None:
        n = final_n
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
