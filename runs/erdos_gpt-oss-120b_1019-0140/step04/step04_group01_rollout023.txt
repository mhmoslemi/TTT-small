def adam_optimize(h0, time_left_func, max_iter=100000, alpha=0.05):
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    t = 0
    while t < max_iter and time_left_func() > 0.0:
        t += 1
        grad, cur_score = compute_worst_grad_and_score(h)   # get gradient and max_corr
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        # update Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h = project_onto_box_sum(h_new, n/2.0)
    # final best score recomputed
    final_score = compute_c5_fft(best_h)
    return best_h, final_score
```

def compute_worst_grad_and_score(h):
    n = h.size
    corr = compute_corr_scaled_fft(h)
    max_corr = np.max(corr)
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[:n-shift] = 1 - h[shift:]
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        grad[s:] = 1 - h[:n-s]
        grad[:n-s] -= h[s:]
    return grad, max_corr

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto { lo <= x_i <= hi, sum x_i = s }."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * (hi - lo)
    if f(lam_high) > s:
        lam_high = np.max(v) + 10 * (hi - lo)
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_worst_grad_and_score(h):
    """Return gradient (subgradient of max C5) and current C5 value."""
    n = h.size
    corr = compute_corr_scaled_fft(h)
    max_corr = np.max(corr)
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.ones(n, dtype=float)  # constant term from sum_i h_i
    if shift >= 0:
        # term1: h_{i+shift}
        grad[:n - shift] -= h[shift:]
        # term2: h_{i-shift}
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] -= h[:n - s]          # term1: h_{i-s}
        grad[:n - s] -= h[s:]          # term2: h_{i+s}
    return grad, max_corr

def adam_optimize(h0, time_left_func, max_iter=200000, alpha=0.05):
    """Adam optimizer on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = beta1
    beta2_pow = beta2
    t = 0
    while t < max_iter and time_left_func() > 0.0:
        t += 1
        grad, cur_score = compute_worst_grad_and_score(h)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h - step, n / 2.0)
        beta1_pow *= beta1
        beta2_pow *= beta2
    final_score = compute_c5_fft(best_h)
    return best_h, final_score

def binary_swap_anneal(h0, rng, time_left_func,
                      max_iters=400000, T0=0.2, decay=0.99995, Tmin=1e-5):
    """Swap‑based simulated annealing on binary (threshold‑0.5) representation."""
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    final_score = compute_c5_fft(best_h)
    return best_h, final_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    n_fine = 1024

    # Initialize
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float).ravel()
        h0 = upsample(init_arr, n_fine)
    else:
        h0 = random_binary_balanced(n_fine, rng)
    h0 = project_onto_box_sum(h0, n_fine / 2.0)
    best_h = h0.copy()
    best_score = compute_c5_fft(best_h)

    # Phase 1: Adam refinement
    if time_left() > 5.0:
        h_opt, sc_opt = adam_optimize(best_h, time_left,
                                      max_iter=200000, alpha=0.05)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 2: Deterministic binarization
    k = best_h.size // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    c5_bin = compute_c5_fft(h_bin)
    if c5_bin < best_score:
        best_h, best_score = h_bin, c5_bin
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 3: Binary annealing
    if time_left() > 5.0:
        h_bin_opt, sc_bin_opt = binary_swap_anneal(best_h, rng, time_left,
                                                   max_iters=500000,
                                                   T0=0.3, decay=0.99996,
                                                   Tmin=1e-6)
        if sc_bin_opt < best_score:
            best_h, best_score = h_bin_opt, sc_bin_opt
            if best_score <= target:
                return best_h, best_score, best_h.size

    # Phase 4: Further Adam refinement
    if time_left() > 5.0:
        h_opt2, sc_opt2 = adam_optimize(best_h, time_left,
                                        max_iter=200000, alpha=0.02)
        if sc_opt2 < best_score:
            best_h, best_score = h_opt2, sc_opt2
            if best_score <= target:
                return best_h, best_score, best_h.size

    # Phase 5: Random restarts
    while time_left() > 15.0:
        h_rand = rng.random(n_fine)
        h_rand = project_onto_box_sum(h_rand, n_fine / 2.0)
        h_rand_opt, sc_rand_opt = adam_optimize(h_rand, time_left,
                                                max_iter=5000, alpha=0.05)
        if sc_rand_opt < best_score:
            best_h, best_score = h_rand_opt, sc_rand_opt
            if best_score <= target:
                return best_h, best_score, best_h.size
        if time_left() > 5.0:
            h_rand_bin, sc_rand_bin = binary_swap_anneal(h_rand_opt, rng,
                                                         time_left,
                                                         max_iters=50000,
                                                         T0=0.2, decay=0.9999,
                                                         Tmin=1e-5)
            if sc_rand_bin < best_score:
                best_h, best_score = h_rand_bin, sc_rand_bin
                if best_score <= target:
                    return best_h, best_score, best_h.size

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
