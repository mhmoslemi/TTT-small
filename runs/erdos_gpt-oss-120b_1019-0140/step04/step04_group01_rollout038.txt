def binary_anneal(h, rng, time_left_func, T0=0.5, decay=0.99999, Tmin=1e-7, max_iters=10**7):
    n = h.size
    cur_c5 = compute_c5_fft(h)
    best_local_h = h.copy()
    best_local_c5 = cur_c5
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.1:
        # random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5_fft(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_local_c5:
                best_local_c5 = cur_c5
                best_local_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_local_h, best_local_c5

def gradient_descent_momentum(h0, rng, time_left_func, step0=0.5, beta=0.9, max_iter=500000):
    h = h0.copy()
    n = h.size
    s = n/2.0
    cur_c5 = compute_c5_fft(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        # compute corr
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # get shift indices with max corr
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        shifts = shift_idxs - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                # term1: h shifted left by shift? they did term1[:n-shift] = h[shift:]
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sgl = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[sgl:] = h[:n - sgl]
                term2[:n - sgl] = h[sgl:]
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        # update momentum
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure brackets contain solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_scaled_fft(h):
    """Full correlation of h with (1-h), scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT."""
    corr = compute_corr_scaled_fft(h)
    return np.max(corr)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def binary_anneal(h0, rng, time_left_func, T0=0.5, decay=0.99999,
                 Tmin=1e-7, max_iters=10_000_000):
    """Simulated annealing on binary vectors (swap a 1 with a 0)."""
    h = h0.copy()
    n = h.size
    cur_c5 = compute_c5_fft(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.1:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5_fft(h)
        delta = new_c5 - cur_c5
        accept = delta <= 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_c5

def targeted_local_search(h0, rng, time_left_func, max_no_improve=2000):
    """Swap a mismatching 1 with a matching 0 for the worst shift."""
    h = h0.copy()
    n = h.size
    cur_c5 = compute_c5_fft(h)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    while time_left_func() > 0.0 and no_improve < max_no_improve:
        corr = compute_corr_scaled_fft(h)
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        if shift < 0:
            shift += n
        if shift == 0:
            break
        rolled = np.roll(h, -shift)
        mism_one = np.where((h == 1) & (rolled == 0))[0]
        good_zero = np.where((h == 0) & (rolled == 1))[0]
        if mism_one.size > 0 and good_zero.size > 0:
            i = rng.choice(mism_one)
            j = rng.choice(good_zero)
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5_fft(h)
            if new_c5 < cur_c5:
                cur_c5 = new_c5
                no_improve = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                continue
            else:
                # revert
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
                continue
        # fallback: try a random improving swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5_fft(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            no_improve = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return best_h, best_c5

def gradient_descent_momentum(h0, rng, time_left_func,
                              step0=0.5, beta=0.9, max_iter=500_000):
    """Projected gradient descent using the gradient of the worst shift(s)."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    cur_c5 = compute_c5_fft(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        shifts = shift_idxs - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[sft:] = h[:n - sft]
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        if shifts.size > 0:
            grad /= shifts.size
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_c5 = compute_c5_fft(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new
        cur_c5 = new_c5
        it += 1
    return best_h, best_c5
