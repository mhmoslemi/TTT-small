def refine_h(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    # We will also compute gradient for worst shift periodically
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        # compute correlation and max shift
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr_full)
        idxes = np.where(corr_full >= max_corr - 1e-12)[0]
        shift = idxes[0] - (n - 1)  # choose first
        # compute gradient for this shift
        grad = np.zeros_like(h)
        if shift >= 0:
            # term1: h[shift:] placed at positions 0..n-shift-1
            grad[:n - shift] += 1.0 - h[shift:]
            # term2: h[:n-shift] placed at positions shift..
            grad[shift:] += -h[:n - shift]
        else:
            sft = -shift
            grad[sft:] += 1.0 - h[:n - sft]
            grad[:n - sft] += -h[sft:]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # keep within box and sum via projection
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

        # occasional pairwise guided move and random perturb
        if it % 500 == 0:
            # pairwise guided move
            # using gradient for worst shift again
            # compute gradient vectors term1 term2 as above (we have grad)
            # choose i where grad[i] < 0 and h[i] < 1
            # choose j where grad[j] > 0 and h[j] > 0
            neg_mask = (grad < -1e-12) & (h < 1 - 1e-12)
            pos_mask = (grad > 1e-12) & (h > 1e-12)
            if np.any(neg_mask) and np.any(pos_mask):
                i = np.argmax(-grad * neg_mask)  # largest magnitude negative
                j = np.argmax(grad * pos_mask)   # largest positive
                delta = min(1.0 - h[i], h[j], 0.1 * eta)  # small
                if delta > 1e-12:
                    h_pair = h.copy()
                    h_pair[i] += delta
                    h_pair[j] -= delta
                    # ensure feasibility via projection (in case of rounding)
                    h_pair = project_onto_box_sum(h_pair, s)
                    sc_pair = compute_c5(h_pair)
                    if sc_pair < best_score:
                        best_score = sc_pair
                        best_h = h_pair.copy()
                        h = h_pair
            # random perturbation
            if rng.random() < 0.3:
                pert = rng.normal(scale=0.02, size=n)
                h_pert = h + pert
                h_pert = project_onto_box_sum(h_pert, s)
                sc_pert = compute_c5(h_pert)
                if sc_pert < best_score:
                    best_score = sc_pert
                    best_h = h_pert.copy()
                    h = h_pert

        # early exit if best_score already below target
        if best_score <= 0.3808:
            break

    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def refine_h(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    target = 0.3808
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr_full)
        idxes = np.where(corr_full >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        if it % 500 == 0:
            # pairwise guided move
            neg_mask = (grad < -1e-12) & (h < 1.0 - 1e-12)
            pos_mask = (grad > 1e-12) & (h > 1e-12)
            if np.any(neg_mask) and np.any(pos_mask):
                i = np.argmax(-grad * neg_mask)
                j = np.argmax(grad * pos_mask)
                delta = min(1.0 - h[i], h[j], 0.1 * eta)
                if delta > 1e-12:
                    h_pair = h.copy()
                    h_pair[i] += delta
                    h_pair[j] -= delta
                    h_pair = project_onto_box_sum(h_pair, s)
                    sc_pair = compute_c5(h_pair)
                    if sc_pair < best_score:
                        best_score, best_h = sc_pair, h_pair.copy()
                        h = h_pair
            # random Gaussian perturbation
            if rng.random() < 0.3:
                pert = rng.normal(scale=0.02, size=n)
                h_pert = h + pert
                h_pert = project_onto_box_sum(h_pert, s)
                sc_pert = compute_c5(h_pert)
                if sc_pert < best_score:
                    best_score, best_h = sc_pert, h_pert.copy()
                    h = h_pert
        if best_score <= target:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # base construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # try upsampled deterministic starts
    for method in ('interp', 'repeat'):
        if time_left() < 5:
            break
        cand = upsample(base, 512) if method == 'interp' else np.repeat(base, 8)[:512]
        cand = project_onto_box_sum(cand, 256.0)
        cand_h, cand_sc = refine_h(cand, max_iters=50000, step0=0.3, beta=0.95,
                                   rng=rng, time_left=time_left)
        if cand_sc < best_score:
            best_h, best_score = cand_h, cand_sc
        if best_score <= 0.3808:
            break

    # random binary restarts at full resolution
    restarts = 5
    for _ in range(restarts):
        if time_left() < 5:
            break
        cand = random_binary_balanced(512, rng)
        cand_h, cand_sc = refine_h(cand, max_iters=50000, step0=0.3, beta=0.95,
                                   rng=rng, time_left=time_left)
        if cand_sc < best_score:
            best_h, best_score = cand_h, cand_sc
        if best_score <= 0.3808:
            break

    # final fine‑tuning on the current best
    if time_left() > 5:
        best_h, best_score = refine_h(best_h, max_iters=80000, step0=0.2,
                                      beta=0.95, rng=rng, time_left=time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
