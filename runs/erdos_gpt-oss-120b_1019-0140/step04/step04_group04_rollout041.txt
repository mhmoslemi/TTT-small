def update_P_k_swap(P, h, i, j):
    n = h.shape[0]
    # assume hi=1, hj=0 before swap
    delta = np.zeros(n, dtype=int)  # int for binary; new P may be int
    # forward from i
    k_max = n - i - 1
    if k_max > 0:
        delta[1:k_max+1] -= h[i+1:i+1+k_max]
    # backward from i
    if i > 0:
        delta[1:i+1] -= h[i-1::-1]   # reversed slice
    # forward from j (positive contribution)
    k_max = n - j - 1
    if k_max > 0:
        delta[1:k_max+1] += h[j+1:j+1+k_max]
    # backward from j
    if j > 0:
        delta[1:j+1] += h[j-1::-1]
    # Note: delta[0] remains 0
    P += delta
    # swap values in h
    h[i] = 0.0

def hill_climb_binary(h0, time_left, rng, n):
    # h0 is binary balanced vector
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    score = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_score = score
    no_improve = 0
    # preallocate delta array
    delta = np.empty(2*n-1, dtype=int)
    while time_left() > 1.0:
        # select i with h[i]=1, j with h[j]=0
        # using while loops to avoid creating large arrays
        # find i:
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)
        # compute delta using current h
        h_rev = h[::-1]
        # slice updates
        start_i_first = n-1 - i
        delta[start_i_first:start_i_first+n] = h - 1  # (h - 1)
        start_j_first = n-1 - j
        delta[start_j_first:start_j_first+n] = 1 - h  # (1 - h)
        # second slices
        delta[i:i+n] += h_rev
        delta[j:j+n] -= h_rev
        # apply delta
        cand_corr = corr + delta
        cand_score = np.max(cand_corr) * (2.0 / n)
        # accept if better or with probability
        if cand_score < score:
            # accept
            h[i] = 0.0
            h[j] = 1.0
            corr = cand_corr
            score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_h = h.copy()
        else:
            no_improve += 1
            # maybe accept with small probability based on temperature
            # compute temperature schedule
            remaining = time_left()
            T = max(1e-6, 0.01 * remaining / budget_s)
            if rng.random() < np.exp((score - cand_score) / T):
                h[i] = 0.0
                h[j] = 1.0
                corr = cand_corr
                score = cand_score
                no_improve = 0
        # occasional random restart if stagnation
        if no_improve >= 50000:
            # restart
            idx = rng.choice(n, size=n//2, replace=False)
            h = np.zeros(n, dtype=float)
            h[idx] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            score = np.max(corr) * (2.0 / n)
            no_improve = 0
        # loop continues
    return best_h, best_score

def hill_climb_binary(h0, time_left, rng, n, budget_s):
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    score = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_score = score
    no_improve = 0
    # Pre-allocate arrays
    while time_left() > 1.0:
        k_idx = np.argmax(corr)
        k = k_idx - (n-1)
        i_start = max(0, -k)
        i_end = min(n-1, n-1-k)
        # compute mismatches
        i_range = np.arange(i_start, i_end+1)
        i_k = i_range + k
        h_i = h[i_range]
        h_ik = h[i_k]
        mism_i = i_range[(h_i == 1) & (h_ik == 0)]
        mism_j = i_range[(h_i == 0) & (h_ik == 1)]
        if mism_i.size > 0 and mism_j.size > 0:
            i_swap = int(rng.choice(mism_i))
            j_swap = int(rng.choice(mism_j))
        else:
            # fallback random swap
            # pick random i where h[i]==1
            i_swap = int(rng.choice(np.flatnonzero(h)))
            j_swap = int(rng.choice(np.flatnonzero(1 - h)))
        # compute delta
        delta = np.zeros(2*n-1, dtype=int)
        # compute h_rev
        h_rev = h[::-1]
        start_i_first = n-1 - i_swap
        delta[start_i_first:start_i_first+n] += (h - 1)
        start_j_first = n-1 - j_swap
        delta[start_j_first:start_j_first+n] += (1 - h)
        delta[i_swap:i_swap+n] += h_rev
        delta[j_swap:j_swap+n] -= h_rev
        cand_corr = corr + delta
        cand_score = np.max(cand_corr) * (2.0 / n)
        if cand_score < score:
            # accept
            h[i_swap] = 0.0
            h[j_swap] = 1.0
            corr = cand_corr
            score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_h = h.copy()
        else:
            no_improve += 1
            remaining = time_left()
            T = max(1e-6, 0.01 * remaining / budget_s)
            if rng.random() < np.exp((score - cand_score) / T):
                h[i_swap] = 0.0
                h[j_swap] = 1.0
                corr = cand_corr
                score = cand_score
                no_improve = 0
        if no_improve >= 50000:
            # restart
            idx = rng.choice(n, size=n//2, replace=False)
            h = np.zeros(n, dtype=float)
            h[idx] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            score = np.max(corr) * (2.0 / n)
            no_improve = 0
    return best_h, best_score
