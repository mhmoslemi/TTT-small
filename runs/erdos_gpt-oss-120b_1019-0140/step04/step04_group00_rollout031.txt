def compute_c5_fft(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2*n-1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1-h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n-1]
    return corr.max() * (2.0 / n)
```

def binary_swap_anneal(h0, rng, time_left, max_iters=200000):
    h = h0.copy()
    cur = compute_c5_fft(h)
    best_h, best_s = h.copy(), cur
    T0 = 0.4
    T_min = 1e-6
    decay = 0.9999
    T = T0
    n = h.size
    # precompute indices of ones/zeros each iteration
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fft(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def smooth_min_grad(h, lam):
    n = h.shape[0]
    s_raw = np.correlate(h, h, mode='full')
    min_s = s_raw.min()
    w = np.exp(-lam * (s_raw - min_s))
    w /= w.sum()
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            if shift < n:
                grad[:n - shift] += wk * h[shift:]
                grad[shift:] += wk * h[:n - shift]
        else:
            s = -shift
            if s < n:
                grad[s:] += wk * h[:n - s]
                grad[:n - s] += wk * h[s:]
    return grad

    cur = compute_c5_fft(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.4, 1e-6, 0.9999
    n = h.size
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fft(h)
        delta = new - cur
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:

        else:

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        candidates.append(project_onto_box_sum(init, target_sum))
    candidates.append(thue_morse_binary(n))
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))
    for cand in candidates:
        if time_left() < 5.0:
            break
        h = cand.copy()
        max_iter = 3000
        step0 = 0.5
        lam_start, lam_end = 0.5, 20.0
        for it in range(1, max_iter + 1):
            if time_left() <= 0.0:
                break
            lam = lam_start + (lam_end - lam_start) * it / max_iter
            grad = smooth_min_grad(h, lam)
            eta = step0 / np.sqrt(it)
            h = project_onto_box_sum(h + eta * grad, target_sum)
        c5 = compute_c5_fft(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
            if best_c5 <= 0.3808:
                return best_h, best_c5
    return best_h, best_c5

def round_to_binary(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    hb = np.zeros_like(h)
    hb[idx] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))
    # First stage: n = 256
    n = 256
    target_sum = n / 2.0
    best_h, best_c5 = continuous_optimize(n, rng, time_left)
    if best_h is not None:
        # binary rounding and local anneal
        hb = round_to_binary(best_h)
        c5b = compute_c5_fft(hb)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb.copy()
        if time_left() > 5.0:
            hb2, c5b2 = binary_swap_anneal(best_h, rng, time_left, max_iters=300000)
            if c5b2 < best_c5:
                best_c5, best_h = c5b2, hb2.copy()
    # Second stage: larger n if time permits
    if time_left() > 200.0:
        n2 = 512
        target_sum2 = n2 / 2.0
        # start from simple binary pattern
        h0 = random_binary_balanced(n2, rng)
        # quick anneal
        hb2, c5b2 = binary_swap_anneal(h0, rng, time_left, max_iters=300000)
        if c5b2 < best_c5:
            best_c5, best_h, n = c5b2, hb2.copy(), n2
    # Ensure final projection and compute final c5
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_c5 = compute_c5_fft(best_h)
    else:
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_c5 = compute_c5_fft(best_h)
    return best_h, best_c5, n
