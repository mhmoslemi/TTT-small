def compute_corr_full(h):
    n = h.shape[0]
    # compute correlation (full) of h and 1-h:
    h_rev = h[::-1]
    N = 1 << ((2*n-1).bit_length())
    fft_h_rev = np.fft.rfft(h_rev, n=N)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_h_rev * fft_one_minus_h, n=N)
    return conv[:2*n-1]
```

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

        if s_ref < best_s:

        if best_s <= 0.3808:
            return best_h, best_s, n
    # Phase 4: targeted hill climbing
    if time_left() > 1.0:
        h_hc, s_hc = targeted_hill_climb(best_h, rng, time_left)
        if s_hc < best_s:
            best_s, best_h = s_hc, h_hc.copy()
        if best_s <= 0.3808:
            return best_h, best_s, n
    # Final projection and evaluation
    best_h = project_onto_box_sum(best_h, target_sum)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s, n
```
