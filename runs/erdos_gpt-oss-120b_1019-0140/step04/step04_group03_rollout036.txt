        s = np.arange(-(n-1), n)
        # compute terms
        i_pos = i + s
        i_neg = i - s
        mask_pos_i = (i_pos >= 0) & (i_pos < n)
        mask_neg_i = (i_neg >= 0) & (i_neg < n)
        val_pos_i = np.zeros_like(s, dtype=h.dtype)
        val_neg_i = np.zeros_like(s, dtype=h.dtype)
        val_pos_i[mask_pos_i] = h[i_pos[mask_pos_i]]
        val_neg_i[mask_neg_i] = h[i_neg[mask_neg_i]]
        term_i = d_i * (1 - val_pos_i - val_neg_i)

        # similarly for j
        j_pos = j + s
        j_neg = j - s
        mask_pos_j = (j_pos >= 0) & (j_pos < n)
        mask_neg_j = (j_neg >= 0) & (j_neg < n)
        val_pos_j = np.zeros_like(s, dtype=h.dtype)
        val_neg_j = np.zeros_like(s, dtype=h.dtype)
        val_pos_j[mask_pos_j] = h[j_pos[mask_pos_j]]
        val_neg_j[mask_neg_j] = h[j_neg[mask_neg_j]]
        term_j = d_j * (1 - val_pos_j - val_neg_j)

        # second order
        second = np.zeros_like(s, dtype=h.dtype)
        idx0 = n - 1
        second[idx0] += -(d_i**2 + d_j**2)
        delta_shift = j - i
        idx1 = delta_shift + n - 1
        if 0 <= idx1 < second.size:
            second[idx1] += -d_i * d_j
        idx2 = -delta_shift + n - 1
        if 0 <= idx2 < second.size:
            second[idx2] += -d_i * d_j

        new_corr = corr + term_i + term_j + second
        new_max_corr = new_corr.max()
        new_score = new_max_corr * dx

        # acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            # accept
            h[i] += d_i
            h[j] += d_j
            corr = new_corr
            cur_score = new_score
            # best?

                best_score = cur_score
                best_h = h.copy()

                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def anneal_h(h, time_budget, rng):

    n = h.shape[0]
    dx = 2.0 / n
    s_arr = np.arange(-(n-1), n)
    idx0 = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max_corr = corr.max()
    cur_score = cur_max_corr * dx

    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    iter_count = 0
    recompute_interval = max(1000, n)  # we can recompute every n steps

    # Temperature schedule based on linear decaying from T0 to T_final
    T0 = cur_score * 0.02 + 1e-8  # avoid zero
    T_final = cur_score * 1e-6 + 1e-10

    while True:
        if time.time() - start >= time_budget:
            break

        # compute temperature
        elapsed = time.time() - start
        t_frac = 1 - (elapsed / time_budget)
        T = T_final + (T0 - T_final) * t_frac

        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        if delta <= 0.0:
            continue
        d_i = -delta
        d_j = delta

        # compute updates
        # compute arrays for term_i/j
        # We'll use vectorized approach as described
        # pre-alloc maybe inside loop: but okay.

        # i + s
        i_pos = i + s_arr
        i_neg = i - s_arr
        # val_pos_i
        val_pos_i = np.zeros_like(s_arr, dtype=h.dtype)
        mask = (i_pos >= 0) & (i_pos < n)
        val_pos_i[mask] = h[i_pos[mask]]
        val_neg_i = np.zeros_like(s_arr, dtype=h.dtype)
        mask = (i_neg >= 0) & (i_neg < n)
        val_neg_i[mask] = h[i_neg[mask]]
        term_i = d_i * (1.0 - val_pos_i - val_neg_i)

        # j
        j_pos = j + s_arr
        j_neg = j - s_arr
        val_pos_j = np.zeros_like(s_arr, dtype=h.dtype)
        mask = (j_pos >= 0) & (j_pos < n)
        val_pos_j[mask] = h[j_pos[mask]]
        val_neg_j = np.zeros_like(s_arr, dtype=h.dtype)
        mask = (j_neg >= 0) & (j_neg < n)
        val_neg_j[mask] = h[j_neg[mask]]
        term_j = d_j * (1.0 - val_pos_j - val_neg_j)

        second = np.zeros_like(s_arr, dtype=h.dtype)
        second[idx0] += -(d_i**2 + d_j**2)
        delta_shift = j - i
        idx1 = delta_shift + n - 1
        if 0 <= idx1 < second.size:
            second[idx1] += -d_i * d_j
        idx2 = -delta_shift + n - 1
        if 0 <= idx2 < second.size:
            second[idx2] += -d_i * d_j

        new_corr = corr + term_i + term_j + second
        new_max_corr = new_corr.max()
        new_score = new_max_corr * dx

        # acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            # accept
            h[i] = h[i] + d_i
            h[j] = h[j] + d_j

            cur_score = corr.max() * dx
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if new_n % old_n == 0:
        return upsample_repeat(arr, new_n)
    else:
        x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
        return np.interp(x_new, x_old, arr)

import time, numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        s = w.sum()
        if abs(s - target_sum) < 1e-12:
            return w
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if new_n % old_n == 0:
        factor = new_n // old_n
        if factor == 1:
            return arr.copy()
        return np.repeat(arr, factor)
    else:
        x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
        return np.interp(x_new, x_old, arr)

def anneal_h(h0, time_budget, rng):
    n = h0.shape[0]
    dx = 2.0 / n
    h = h0.copy()
    # padded version for fast shift access
    pad = n - 1
    h_pad = np.concatenate([np.zeros(pad, dtype=h.dtype), h, np.zeros(pad, dtype=h.dtype)])
    # correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max_corr = corr.max()
    cur_score = cur_max_corr * dx
    best_h = h.copy()
    best_score = cur_score
    s_arr = np.arange(-(n-1), n)
    idx0 = n - 1
    start = time.time()
    recompute_every = max(500, n)  # recompute from scratch periodically
    it = 0
    # temperature parameters
    T0 = max(cur_score * 0.05, 1e-8)
    T_final = max(cur_score * 1e-6, 1e-12)
    while True:
        if time.time() - start >= time_budget:
            break
        # linear temperature schedule
        elapsed = time.time() - start
        t_frac = 1.0 - elapsed / time_budget
        T = T_final + (T0 - T_final) * t_frac
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        if delta <= 0.0:
            continue
        d_i = -delta
        d_j = delta
        # term_i
        val_i_pos = h_pad[pad + i + s_arr]
        val_i_neg = h_pad[pad + i - s_arr]
        term_i = d_i * (1.0 - val_i_pos - val_i_neg)
        # term_j
        val_j_pos = h_pad[pad + j + s_arr]
        val_j_neg = h_pad[pad + j - s_arr]
        term_j = d_j * (1.0 - val_j_pos - val_j_neg)
        # new correlation
        new_corr = corr + term_i + term_j
        # second-order adjustments
        delta_shift = j - i
        idx1 = delta_shift + n - 1
        idx2 = -delta_shift + n - 1
        new_corr[idx0] -= (d_i*d_i + d_j*d_j)
        if 0 <= idx1 < new_corr.size:
            new_corr[idx1] -= d_i * d_j
        if 0 <= idx2 < new_corr.size:
            new_corr[idx2] -= d_i * d_j
        new_score = new_corr.max() * dx
        # acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            # accept
            h[i] += d_i
            h[j] += d_j
            # update padded
            h_pad[pad + i] = h[i]
            h_pad[pad + j] = h[j]
            corr = new_corr
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        it += 1
        if it % recompute_every == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_score = corr.max() * dx
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        s = w.sum()
        if abs(s - target_sum) < 1e-12:
            return w
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if new_n % old_n == 0:
        factor = new_n // old_n
        if factor == 1:
            return arr.copy()
        return np.repeat(arr, factor)
    else:
        x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
        return np.interp(x_new, x_old, arr)

def anneal_h(h0, time_budget, rng):
    n = h0.shape[0]
    dx = 2.0 / n
    h = h0.copy()
    pad = n - 1
    h_pad = np.concatenate([np.zeros(pad, dtype=h.dtype), h, np.zeros(pad, dtype=h.dtype)])
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_score = corr.max() * dx
    best_h = h.copy()
    best_score = cur_score
    s_arr = np.arange(-(n - 1), n)
    idx0 = n - 1
    start = time.time()
    recompute_every = max(500, n)
    it = 0
    T0 = max(cur_score * 0.05, 1e-8)
    T_final = max(cur_score * 1e-6, 1e-12)
    while True:
        if time.time() - start >= time_budget:
            break
        elapsed = time.time() - start
        t_frac = 1.0 - elapsed / time_budget
        T = T_final + (T0 - T_final) * t_frac
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random()
        if delta <= 0.0:
            continue
        d_i = -delta
        d_j = delta
        val_i_pos = h_pad[pad + i + s_arr]
        val_i_neg = h_pad[pad + i - s_arr]
        term_i = d_i * (1.0 - val_i_pos - val_i_neg)
        val_j_pos = h_pad[pad + j + s_arr]
        val_j_neg = h_pad[pad + j - s_arr]
        term_j = d_j * (1.0 - val_j_pos - val_j_neg)
        new_corr = corr + term_i + term_j
        new_corr[idx0] -= (d_i * d_i + d_j * d_j)
        delta_shift = j - i
        idx1 = delta_shift + n - 1
        idx2 = -delta_shift + n - 1
        if 0 <= idx1 < new_corr.size:
            new_corr[idx1] -= d_i * d_j
        if 0 <= idx2 < new_corr.size:
            new_corr[idx2] -= d_i * d_j
        new_score = new_corr.max() * dx
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h[i] += d_i
            h[j] += d_j
            h_pad[pad + i] = h[i]
            h_pad[pad + j] = h[j]
            corr = new_corr
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        it += 1
        if it % recompute_every == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_score = corr.max() * dx
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n0 = 64
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.size != n0:
            base = upsample(base, n0)
    else:
        base = random_binary_balanced(n0, rng)
    base = project_onto_box_sum(base, n0 / 2.0)
    if rng.random() < 0.5:
        pert = rng.normal(scale=0.02, size=base.shape)
        base = np.clip(base + pert, 0.0, 1.0)
        base = project_onto_box_sum(base, n0 / 2.0)
    n_final = 512
    base = upsample(base, n_final)
    base = project_onto_box_sum(base, n_final / 2.0)
    time_left = budget_s - (time.time() - start)
    if time_left > 5.0:
        best_h, best_score = anneal_h(base, time_left, rng)
    else:
        best_h = base
        best_score = compute_c5(best_h)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
