def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):

def compute_c5(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
def thue_morse(n):
    # generate binary pattern
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1 = h[shift:]  (size n-shift)
            # term2 = h[:n-shift] (size n-shift)
            grad[:n-shift] += (1.0 - h[shift:])  # (1 - h[i+shift])
            grad[shift:] += -h[:n-shift]        # -h[i-shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n-s])         # (1 - h[i-s])
            grad[:n-s] += -h[s:]               # -h[i+s]
    return grad / len(shifts) if shifts else grad

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def gradient_descent_float(h0, max_iters, step0, rng, time_left):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        # select a few top shifts for gradient
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = 0.9 * momentum + 0.1 * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        # small random perturbation occasionally
        if it % 200 == 0 and rng.random() < 0.2:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h, rng, time_left, max_no_improve=500):
    # Input binary h (0,1)
    n = h.size
    target_sum = n/2.0
    score = compute_c5(h)
    no_improve = 0
    while time_left() > 0 and no_improve < max_no_improve:
        # select random a (1) and b (0)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        j = rng.choice(zeros)

        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        if new_score < score:
            h = h_new
            score = new_score
            no_improve = 0
            # update ones/zeros arrays? will recompute each loop anyway.
        else:
            no_improve += 1
        # occasional temperature acceptance
        if no_improve >= max_no_improve//2 and rng.random() < 0.05:

            T = 0.001 * (max_no_improve - no_improve) / max_no_improve
            if rng.random() < math.exp(-(new_score - score) / T):
                h = h_new
                score = new_score
                no_improve = 0
    return h, score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] += -h[:n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] += -h[s:]
    return grad / len(shifts) if shifts else grad

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def gradient_descent_float(h0, max_iters, step0, rng, time_left):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = 0.9 * momentum + 0.1 * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 200 == 0 and rng.random() < 0.2:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h, rng, time_left, max_no_improve=800):
    n = h.size
    score = compute_c5(h)
    no_improve = 0
    while time_left() > 0 and no_improve < max_no_improve:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        if new_score < score:
            h, score = h_new, new_score
            no_improve = 0
        else:
            no_improve += 1
    return h, score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    n_final = 256
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = upsample(base, n_final)
    base = project_onto_box_sum(base, n_final / 2.0)
    best_h, best_score = base.copy(), compute_c5(base)
    if time_left() > 10:
        iters = int(min(20000, max(5000, time_left() * 5)))
        best_h, best_score = gradient_descent_float(best_h, iters, 0.5, rng, time_left)
    while time_left() > 5:
        h = random_binary_balanced(n_final, rng)
        h, sc = binary_hill_climb(h, rng, time_left, max_no_improve=800)
        if time_left() > 2:
            iters = int(min(5000, max(1000, time_left() * 2)))
            h, sc = gradient_descent_float(h, iters, 0.4, rng, time_left)
        if sc < best_score:
            best_score, best_h = sc, h.copy()
        if best_score <= 0.3808:
            break
    best_h = project_onto_box_sum(best_h, n_final / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_final
