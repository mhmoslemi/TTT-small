def compute_smoothed_gradient(h, alpha):
    n = h.shape[0]
    corr = compute_corr_unscaled_fft(h)   # shape (2n-1,)
    max_corr = np.max(corr)
    # shift corr for numeric stability
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)   # shape (2n-1,)

    # compute conv_neg = sum_k w_k * h[t - k]
    # conv_pos = sum_k w_k * h[t + k]
    L = 2*n-1
    conv_len = 3*n-2
    M = 1 << ((conv_len).bit_length())  # next pow2 >= conv_len
    # FFT of w padded
    W = np.fft.rfft(w, n=M)
    H = np.fft.rfft(h, n=M)
    conv_wh = np.fft.irfft(W * H, n=M)[:conv_len]
    # conv_wrev_h: use w_rev
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=M)
    conv_wrev = np.fft.irfft(W_rev * H, n=M)[:conv_len]

    # extract central n entries
    start = n-1
    conv_neg = conv_wh[start:start+n]
    conv_pos = conv_wrev[start:start+n]

    grad = 1.0 - (conv_neg + conv_pos)
    return grad
```

def smoothed_gradient_descent(h0, time_left_func, step0=0.2, alpha0=0.5, alpha_inc=0.001, beta=0.9, max_iter=10**7):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    alpha = alpha0
    while it <= max_iter and time_left_func() > 0.0:
        grad = compute_smoothed_gradient(h, alpha)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        # maybe adjust alpha gradually
        alpha = min(alpha + alpha_inc, 5.0)  # cap
        h = h_new
        it += 1
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_unscaled_fft(h):
    """Full correlation array ∑ h[i]·(1‑h[i+k]) (no scaling)."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr

def compute_corr_scaled_fft(h):
    """Full correlation scaled by dx."""
    n = h.shape[0]
    return compute_corr_unscaled_fft(h) * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binary_swap_anneal(h0, rng, time_left_func, max_iters=400000,
                      T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=1500):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5_fft(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_scaled_fft(h)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5_fft(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5_fft(h_opt)
    return h_opt, t_opt

def compute_smoothed_gradient(h, alpha):
    """Gradient of (1/α)·log‑sum‑exp(α·corr_k) w.r.t. h."""
    n = h.shape[0]
    corr = compute_corr_unscaled_fft(h)          # length 2n‑1
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / exp_vals.sum()                # softmax weights
    # convolution lengths
    conv_len = 3 * n - 2
    M = 1 << ((conv_len).bit_length())
    W = np.fft.rfft(w, n=M)
    H = np.fft.rfft(h, n=M)
    conv_wh = np.fft.irfft(W * H, n=M)[:conv_len]    # w * h
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=M)
    conv_wrev = np.fft.irfft(W_rev * H, n=M)[:conv_len]  # w_rev * h
    start = n - 1
    conv_neg = conv_wh[start:start + n]          # sum_k w_k·h[t‑k]
    conv_pos = conv_wrev[start:start + n]        # sum_k w_k·h[t+k]
    grad = 1.0 - (conv_neg + conv_pos)
    return grad

def smoothed_gradient_descent(h0, time_left_func,
                              step0=0.3, alpha0=0.5, alpha_inc=0.002,
                              beta=0.9, max_iter=10**7):
    """Projected gradient descent on smooth max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    alpha = alpha0
    while it <= max_iter and time_left_func() > 0.0:
        grad = compute_smoothed_gradient(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        alpha = min(alpha + alpha_inc, 5.0)
        h = h_new
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # initial construction
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        n0 = init_arr.size
        init_arr = project_onto_box_sum(init_arr, n0 / 2.0)
    else:
        n0 = 512
        init_arr = np.full(n0, 0.5)
        init_arr = project_onto_box_sum(init_arr, n0 / 2.0)

    best_h = init_arr.copy()
    best_score = compute_c5_fft(best_h)

    # Phase 1: modest SLSQP refinement at medium resolution
    if time_left() > 12.0:
        n_mid = 256
        h_mid = downsample(best_h, n_mid)

        h_opt, sc_opt = slsqp_optimize_candidate(h_mid, max_iter=2500)

            best_h = upsample(h_opt, n0)
            best_h = project_onto_box_sum(best_h, n0 / 2.0)

    # Phase 2: smoothed gradient descent on full resolution
    if time_left() > 6.0:
        h_ref, sc_ref = smoothed_gradient_descent(best_h, time_left,
                                                  step0=0.3, alpha0=0.5,
                                                  alpha_inc=0.003, beta=0.9)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # Phase 3: random restarts with brief smoothed descent
    while time_left() > 20.0:
        v = rng.random(n0)
        v = project_onto_box_sum(v, n0 / 2.0)
        h_tmp, sc_tmp = smoothed_gradient_descent(v, time_left,
                                                  step0=0.3, alpha0=0.5,
                                                  alpha_inc=0.003, beta=0.9,
                                                  max_iter=30000)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
            if best_score <= 0.3808:

    if time_left() > 2.0:
        n = best_h.size
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:

        if time_left() > 0.5:

                                                       time_left,
                                                       max_iters=300000,
                                                       T0=0.2, decay=0.9999,
                                                       Tmin=1e-6)
            if sc_bin_ref < best_score:
                best_score, best_h = sc_bin_ref, h_bin_ref.copy()

    # Final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
