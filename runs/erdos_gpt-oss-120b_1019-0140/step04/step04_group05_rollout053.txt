run():
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return budget_s - (time.time()-start)

    # Determine target n = 512
    if 'initial_h_values' in globals(): base = np.asarray(initial_h_values, dtype=float)
    else: base = thue_morse(64).astype(float)
    # upsample to n
    n_target = 512
    base = upsample(base, n_target) if base.shape[0] != n_target else base.copy()
    base = project_onto_box_sum(base, n_target/2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # Phase 1: gradient pair descent
    if time_left() > 10:
        gp_h, gp_score = gradient_pair_descent(best_h.copy(), time_left(), rng)
        if gp_score < best_score:
            best_h, best_score = gp_h, gp_score

    # Phase 2: simulated annealing
    if time_left() > 10:
        sa_h, sa_score = simulated_annealing(best_h.copy(), time_left(), rng)
        if sa_score < best_score:
            best_h, best_score = sa_h, sa_score

    # Phase 3: random perturbation & local search
    while time_left() > 0.5:
        # random balanced binary perturbation
        n = best_h.size
        h_bin = random_balanced_binary(n, rng)
        sc = compute_c5(h_bin)
        if sc < best_score:
            best_h = h_bin
            best_score = sc
        # small move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i: j += 1
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc = compute_c5(h_new)
            if sc < best_score:
                best_h = h_new
                best_score = sc

    # final projection
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_vector(h):
    """Linear correlation of h with (1-h) using FFT (full mode)."""
    n = h.shape[0]
    a = h.astype(np.float64)
    b_rev = (1.0 - a)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b_rev, L)
    conv = np.fft.irfft(fa * fb, L)
    return conv[: 2 * n - 1]

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_vector(h)
    return corr.max() * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger one."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq

def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_pair_descent(h0, time_limit, rng):
    """Deterministic descent moving mass from positive to negative gradient indices."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    step = 0.02  # initial delta

        if k >= 0:

            grad[k:] += -h[: n - k]
        else:

            grad[k2:] += 1.0 - h[: n - k2]
            grad[: n - k2] += -h[k2:]
        i_down = int(np.argmax(grad))   # positive gradient → should decrease
