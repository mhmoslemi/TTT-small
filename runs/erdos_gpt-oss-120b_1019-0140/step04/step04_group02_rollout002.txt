def compute_grad_max_corr(h, corr, delta=0.005):
    n = h.size
    max_c = np.max(corr)
    thresh = max_c - max_c * delta
    idxes = np.where(corr >= thresh)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n-k] += 1.0 - h[k:]
            grad[k:] -= h[:n-k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n-m]
            grad[:n-m] -= h[m:]
    if idxes.size > 0:
        grad /= idxes.size
    return grad
```

def adam_descent(h0, max_iter, lr, beta1, beta2, rng, time_left, n):
    h = h0.copy()
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for t in range(1, max_iter+1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        grad = compute_grad_max_corr(h, corr, delta=0.005)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # occasional noise injection to escape
        if t % 50000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_round(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]  # get top k indices
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec

def binary_hill_climb(h_bin, max_iter, rng, time_left):
    n = h_bin.size
    best_h = h_bin.copy()
    best_score = compute_c5_fft(best_h)
    # precompute indices of ones and zeros
    for it in range(max_iter):
        if time_left() <= 0.0:
            break
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = compute_c5_fft(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {x: lo<=x_i<=hi, sum x_i = s} using bisection on lambda."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # enlarge interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation of arr (size m) to size new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_corr_full_fft(h):
    """Full linear cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    """Approximate C5 = max_k ∫ h(x)(1‑h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_corr(h, corr, delta=0.005):
    """
    Gradient of the maximal correlation term.
    Includes all shifts whose correlation is within (1‑delta) of the max.
    """
    n = h.size
    max_c = np.max(corr)
    thresh = max_c - max_c * delta
    idxes = np.where(corr >= thresh)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if idxes.size > 0:
        grad /= idxes.size
    return grad

def adam_descent(h0, max_iter, lr, beta1, beta2, rng, time_left, n):
    """Projected Adam optimiser for the max‑correlation objective."""
    h = h0.copy()
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for t in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        grad = compute_grad_max_corr(h, corr, delta=0.005)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        if t % 50000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_round(h):
    """Return a balanced binary vector with the same sum constraint."""
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec

def random_binary_balanced(n, rng):
    """Balanced random binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binary_hill_climb(h_bin, max_iter, rng, time_left):
    """Local search on binary vectors by swapping a 1 and a 0."""
    n = h_bin.size
    best_h = h_bin.copy()
    best_score = compute_c5_fft(best_h)
    for _ in range(max_iter):
        if time_left() <= 0.0:
            break
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = compute_c5_fft(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
    return best_h, best_score

    # initialise base vector
