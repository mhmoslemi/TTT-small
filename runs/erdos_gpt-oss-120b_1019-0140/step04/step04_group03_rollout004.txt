def compute_c5(h):

def simulated_annealing_binary(h0, init_score, rng, start_time, max_time):
    cur_h = h0.copy()
    cur_score = init_score
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.shape[0]
    # schedule: T0 start near maybe 0.005? Actually typical delta of score maybe 0.001? We'll set T0 = 0.001.
    # We'll adapt T = T0 * (remaining / max_time)
    # set exponent factor maybe 0.5.
    while True:
        elapsed = time.time() - start_time
        remaining = max_time - elapsed
        if remaining <= 0:
            break
        # temperature
        frac = remaining / max_time
        T = 0.005 * (frac**0.7) + 1e-8
        # random pick i,j
        i = rng.integers(n)
        # ensure i has some mass to move
        if cur_h[i] <= 0:

        j = rng.integers(n)
        if cur_h[j] >= 1:

                best_h = cur_h.copy()

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """Projection of v onto [lo,hi] box with exact linear sum constraint."""
    # Clip to bounds first
    v = np.clip(v, lo, hi)
    # If already satisfied, return
    s = v.sum()
    if abs(s - target_sum) < 1e-12:
        return v
    # Binary search for lambda such that sum(clip(v - lam, lo, hi)) == target_sum
    lam_lo = -1.0
    lam_hi = 1.0
    # Expand bounds if needed
    while np.sum(np.clip(v - lam_lo, lo, hi)) < target_sum:
        lam_lo *= 2
    while np.sum(np.clip(v - lam_hi, lo, hi)) > target_sum:
        lam_hi *= 2
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        s_mid = np.sum(np.clip(v - lam_mid, lo, hi))
        if s_mid > target_sum:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
        if abs(s_mid - target_sum) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Compute max_k ∫ h(x)(1-h(x+k))dx via FFT convolution."""
    n = h.shape[0]
    L = 2 * n
    fh = np.fft.rfft(h, n=L)
    f1mh = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(fh * f1mh, n=L)
    # linear convolution length 2*n-1 occupies first 2*n-1 entries
    max_val = np.max(conv[:2 * n - 1])
    return max_val * (2.0 / n)

def upsample(arr, new_n):
    if arr.shape[0] == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, arr.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                       dtype=float, count=n)

def simulated_annealing_binary(h0, init_score, rng, start_time, max_time):
    cur_h = h0.copy()
    cur_score = init_score
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.shape[0]
    # parameters for temperature schedule
    while True:
        elapsed = time.time() - start_time
        remaining = max_time - elapsed
        if remaining <= 0:
            break
        frac = remaining / max_time
        T = 0.005 * (frac ** 0.7) + 1e-9
        # choose a pair (i,j) with mass to move
        i = rng.integers(n)
        if cur_h[i] <= 0.0:
            continue
        j = rng.integers(n)
        if cur_h[j] >= 1.0:
            continue
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 0.0:
            continue
        # occasional full swap
        if rng.random() < 0.1:
            delta = max_move
        else:
            delta = rng.random() * max_move
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / T):
            cur_h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # define target scales; larger scale gives finer control
    scales = [256, 512, 1024]
    # base vector
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(scales[0], rng)
    best_h = None
    best_score = np.inf
    for idx, n in enumerate(scales):
        # upsample / resize
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        cur_score = compute_c5(base)
        if cur_score < best_score:
            best_score, best_h = cur_score, base.copy()
        # allocate time for this scale
        elapsed = time.time() - start
        remaining_total = budget_s - elapsed
        if remaining_total <= 5.0:
            break
        scale_time = remaining_total * (1.0 / (len(scales) - idx)) * 0.9
        # perform several SA runs while time permits
        while True:
            elapsed = time.time() - start
            remaining_total = budget_s - elapsed
            if remaining_total <= 5.0:
                break
            max_time = min(scale_time * 0.3, remaining_total - 2.0)
            if max_time <= 0.5:
                break
            h_sa, sc_sa = simulated_annealing_binary(base, cur_score, rng,
                                                     start, max_time)
            if sc_sa < best_score:
                best_score, best_h = sc_sa, h_sa.copy()
            # use result as new base for next SA run
            base = h_sa
            cur_score = sc_sa
        # prepare for next finer scale
        if idx < len(scales) - 1:
            base = best_h if best_h is not None else base
    # final verification / projection
    if best_h is None:
        n = scales[-1]
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)
    return best_h, float(best_score), int(best_h.size)
