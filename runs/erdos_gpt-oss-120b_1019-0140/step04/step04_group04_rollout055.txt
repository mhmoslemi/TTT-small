    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

            best_score = score
            best_h = h.copy()
        # early stop if below desired threshold
        if best_score <= 0.3808:
            break
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        return t - corr  # want t >= corr => t - corr >= 0
    cons = [{'type':'eq','fun':eq_sum},
            {'type':'ineq','fun':ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

            base = best_h.copy()
        max_possible = int(time_left() * 150)
        max_iters = min(target_iters[n], max_possible)
        if max_iters <= 0:
            break

        cur_best_h = base.copy()
        cur_best_score = compute_c5(cur_best_h)

        # perform a couple of restarts
        for r in range(2):
            if time_left() < 2:
                break
            if r == 0:
                h0 = cur_best_h + rng.normal(scale=0.02, size=n)
            else:
                h0 = random_balanced(n, rng).astype(float)
            h0 = project_onto_box_sum(h0, n / 2.0)
            lr = 0.5 if n <= 256 else 0.3
            h_opt, sc_opt = adam_optimize(
                h0, max_iters, lr=lr,
                beta1=0.9, beta2=0.999, eps=1e-8,
                rng=rng, time_left=time_left)
            if sc_opt < cur_best_score:
                cur_best_h, cur_best_score = h_opt, sc_opt

        if time_left() > 2:
            h_ref, sc_ref = slsqp_optimize_candidate(cur_best_h, max_iter=200)
            if sc_ref < cur_best_score:
                cur_best_h, cur_best_score = h_ref, sc_ref

        if cur_best_score < best_score:
            best_h, best_score = cur_best_h.copy(), cur_best_score
        best_h = cur_best_h.copy()

    # final refinement at the highest attained resolution
    if time_left() > 5:
        n_final = best_h.size
        iters = int(min(80000, time_left() * 30))
        if iters > 0:
            best_h, best_score = adam_optimize(
                best_h, iters, lr=0.2,
                beta1=0.9, beta2=0.999, eps=1e-8,
                rng=rng, time_left=time_left)

    if time_left() > 2:
        h_final, sc_final = slsqp_optimize_candidate(best_h, max_iter=300)
        if sc_final < best_score:
            best_h, best_score = h_final, sc_final

    # lightweight random local search while time permits
    while time_left() > 2:
        # double‑swap move preserving the sum
        i, j = rng.choice(best_h.size, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.01, 0.5)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_onto_box_sum(h_new, best_h.size / 2.0)
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_h, best_score = h_new, sc_new
        if rng.random() < 0.3:
            h_pert = best_h + rng.normal(scale=0.02, size=best_h.size)
            h_pert = project_onto_box_sum(h_pert, best_h.size / 2.0)
            sc_pert = compute_c5(h_pert)
            if sc_pert < best_score:
                best_h, best_score = h_pert, sc_pert
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
