import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}."""
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def gradient_descent(h0, time_limit, rng):
    """Projected gradient descent on a smooth approximation of max overlap."""
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_score = np.max(corr) * dx
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    max_iter = 2000000
    for it in range(1, max_iter + 1):
        if time.time() - start >= time_limit:
            break
        # gradually increase smoothing parameter
        alpha = 5.0 + 25.0 * (it / max_iter)

        # correlation & smooth weights
        corr = np.correlate(h, 1.0 - h, mode='full')
        A = corr * dx
        cur_score = np.max(A)
        A_max = cur_score
        exp_arg = alpha * (A - A_max)
        w = np.exp(exp_arg)
        w /= w.sum()

        # gradient via convolutions
        w_rev = w[::-1]
        D2_full = np.convolve(w, h, mode='full')
        D2 = D2_full[n - 1:n - 1 + n]
        D1_full = np.convolve(w_rev, 1.0 - h, mode='full')
        D1 = D1_full[n - 1:n - 1 + n]
        grad = dx * (D1 - D2)

        # step size schedule
        step0 = 0.5
        step = step0 * (1.0 - it / max_iter)
        gnorm = np.linalg.norm(grad)
        delta = step * grad / gnorm if gnorm > 0 else np.zeros_like(h)

        h_new = h - delta
        h_new = project_onto_box_sum(h_new, n / 2.0)

        # evaluate new point
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_score = np.max(corr_new) * dx

        # accept if improvement; occasional uphill move
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # tiny probabilistic uphill step to escape plateaus
            temp = 0.001 * (time_limit - (time.time() - start)) / time_limit
            if temp > 0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                h = h_new
                cur_score = new_score

        if best_score <= 0.3808:
            break
    return best_h, best_score


def binary_refine(h_start, time_limit, rng):
    """Hill‑climbing on binary vectors (half ones) starting from top half of h_start."""
    n = h_start.size
    idx = np.argsort(-h_start)
    binary_h = np.zeros(n, dtype=float)
    binary_h[idx[: n // 2]] = 1.0
    best_h = binary_h
    best_score = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        sc = compute_c5(cand)
        if sc < best_score:
            best_h = cand
            best_score = sc
            if best_score <= 0.3808:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # base construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 64
        base = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        base[idx] = 1.0

    n_target = 512
    base = upsample(base, n_target)
    base = project_onto_box_sum(base, n_target / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # time allocation
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining <= 0:
        return best_h, best_score, n_target

    # gradient descent phase (≈85% of remaining time)
    gd_time = remaining * 0.85
    h_gd, sc_gd = gradient_descent(best_h, gd_time, rng)
    if sc_gd < best_score:
        best_h, best_score = h_gd, sc_gd

    # update remaining budget
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining <= 0.5:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)
        return best_h, best_score, best_h.size

    # binary refinement from continuous solution (≈10% of remaining)
    bin_time = remaining * 0.1
    h_bin, sc_bin = binary_refine(best_h, bin_time, rng)
    if sc_bin < best_score:
        best_h, best_score = h_bin, sc_bin

    # final stochastic binary search with remaining time
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    n = best_h.size
    while remaining > 0.2:
        # random balanced binary candidate
        idx = rng.choice(n, size=n // 2, replace=False)
        h_rand = np.zeros(n, dtype=float)
        h_rand[idx] = 1.0
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_h, best_score = h_rand, sc_rand
            # short local hill‑climb
            sub_time = min(0.5, remaining - 0.2)
            h_hc, sc_hc = binary_refine(best_h, sub_time, rng)
            if sc_hc < best_score:
                best_h, best_score = h_hc, sc_hc
        if best_score <= 0.3808:
            break
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
