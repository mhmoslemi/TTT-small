def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0., budget_s - (time.time() - start))

    # Determine n
    target_n = 256
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64) # as before
    # Upsample to target_n
    if base.shape[0] != target_n:
        base = upsample(base, target_n)
    # Ensure sum constraint
    base = project_onto_box_sum(base, target_n/2)
    h = base.copy()
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    current_score = best_score

    # parameters
    step_max = 0.5   # fraction of max possible delta per iteration
    decay_rate = 0.9995 # step decay per iteration
    min_step = 1e-4
    it = 0

    while time_left() > 0.5:
        it += 1
        # compute gradient of max correlation
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)  # shape n
        # Find indices
        i_pos = np.argmax(grad) # positive gradient => decreasing h reduces obj
        i_neg = np.argmin(grad) # negative gradient => increasing h reduces obj
        # Determine max feasible delta
        delta_max = min(h[i_pos], 1.0 - h[i_neg])
        if delta_max < 1e-12:
            # can't move, break
            break
        # step size
        step = step_max * delta_max
        # ensure step not too small
        if step < min_step:
            step = min_step
        # propose update
        h_new = h.copy()
        h_new[i_pos] -= step
        h_new[i_neg] += step
        # evaluate
        new_score = compute_c5_fft(h_new)
        # accept if improvement
        if new_score <= current_score:
            h = h_new
            current_score = new_score
            # update best
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # reject and reduce step_max
            step_max *= 0.9
        # decay step_max slowly
        step_max = max(min_step, step_max * decay_rate)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient of the max‑correlation term (averaged over all max shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

            if best_h_local is not None:

            current_score = compute_c5_fft(h)

                    break

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, n
