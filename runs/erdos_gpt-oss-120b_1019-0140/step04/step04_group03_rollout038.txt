def smoothed_gradient_descent(h0, max_iters, step0, beta, gamma, rng, max_time):
    n = h0.shape[0]; dx = 2.0 / n; target_sum = n/2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time.time() - start >= max_time:
            break
        # conv_hh
        conv_hh = np.correlate(h, h, mode='full')
        c = 1.0 - dx * conv_hh
        max_c = c.max()
        # compute p
        # stable exponent
        exp_shift = np.exp(gamma * (c - max_c))
        p = exp_shift / exp_shift.sum()
        # compute conv of p with h
        conv_ph = np.convolve(p, h, mode='full')
        conv_ph_left = conv_ph[n-1:2*n-1]
        conv_ph_rev = np.convolve(p[::-1], h, mode='full')
        conv_ph_right = conv_ph_rev[n-1:2*n-1]
        grad = -dx * (conv_ph_left + conv_ph_right)
        momentum = beta * momentum + (1-beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random noise
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    # Clip trivial cases
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # binary search on lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) <= 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) <= 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def linear_conv_fft(a, b):
    size = a.size + b.size - 1
    nfft = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.ifft(A * B).real
    return c[:size]

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def smoothed_gradient_descent(h0, max_iters, step0, beta, gamma, rng, max_time):
    start = time.time()
    n = h0.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    sqrt = np.sqrt
    for it in range(1, max_iters + 1):
        if time.time() - start >= max_time:
            break
        conv_hh = linear_conv_fft(h[::-1], h)               # length 2n-1
        c = 1.0 - dx * conv_hh
        max_c = c.max()
        exp_shift = np.exp(gamma * (c - max_c))
        p = exp_shift / exp_shift.sum()
        conv_ph = linear_conv_fft(p, h)                     # length 3n-2
        left = conv_ph[n - 1:2 * n - 1]
        conv_ph_rev = linear_conv_fft(p[::-1], h)
        right = conv_ph_rev[n - 1:2 * n - 1]
        grad = -dx * (left + right)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    scales = [64, 128, 256, 512, 1024]
    best_h = None
    best_score = np.inf

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    base = upsample(base, 64)
    base = project_onto_box_sum(base, 64 / 2.0)

    for n in scales:
        if time.time() - start > budget_s - 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        trials = 50 if n <= 256 else 20
        for _ in range(trials):
            if time.time() - start > budget_s - 5:
                break
            h_rand = random_binary_balanced(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score, best_h = sc, h_rand.copy()

        remaining = budget_s - (time.time() - start)
        if remaining < 5:
            break
        max_iters = int(min(20000, max(5000, remaining * (2000.0 / max(64, n)))))
        h_opt, sc_opt = smoothed_gradient_descent(
            h0=base,
            max_iters=max_iters,
            step0=0.5,
            beta=0.9,
            gamma=20.0,
            rng=rng,
            max_time=remaining
        )
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        base = h_opt

        # short simulated annealing refinement
        remaining = budget_s - (time.time() - start)
        if remaining > 5:
            h_cur = best_h.copy()
            cur_sc = best_score
            steps = int(min(20000, max(5000, remaining * 500.0)))
            T0, T_final = 0.01, 1e-6
            for step in range(steps):
                if time.time() - start > budget_s:
                    break
                T = T0 * ((steps - step) / steps) + T_final
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(h_cur[i], 1.0 - h_cur[j])
                if max_move <= 0.0:
                    continue
                delta = max_move * rng.random()
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_sc = compute_c5(h_new)
                if new_sc < cur_sc or rng.random() < np.exp(-(new_sc - cur_sc) / T):
                    h_cur, cur_sc = h_new, new_sc
                    if cur_sc < best_score:
                        best_score, best_h = cur_sc, h_cur.copy()
            base = best_h

    # final refinement at highest resolution
    if best_h is not None and time.time() - start < budget_s - 5:
        n = best_h.size
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(50000, max(10000, remaining * (10000.0 / max(64, n)))))
        best_h, best_score = smoothed_gradient_descent(
            h0=best_h,
            max_iters=max_iters,
            step0=0.2,
            beta=0.9,
            gamma=30.0,
            rng=rng,
            max_time=remaining
        )
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    if best_h is None:
        n = scales[-1]
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
