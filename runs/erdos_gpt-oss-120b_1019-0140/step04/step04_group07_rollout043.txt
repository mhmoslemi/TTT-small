def annealing(h, rng, time_left, init_temp=0.05, final_temp=1e-5):

    n = h.size
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    total = time_left()
    # If time left is zero, just return
    if total <= 0:
        return cur_h, cur_score
    while True:
        elapsed = time.time() - start
        if elapsed >= total:
            break
        # fraction of total time elapsed
        frac = elapsed / total
        T = init_temp * ((final_temp / init_temp) ** frac)  # exponential decay
        # propose move: pick two indices and move delta from i to j
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        # delta scale maybe proportional to T
        delta = max_delta * rng.random() * min(1.0, T * 10)  # if T small, small moves
        if delta <= 0:
            continue
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        dE = new_score - cur_score
        accept = False
        if dE < 0:
            accept = True
        else:
            # accept with prob exp(-dE/T)
            # to avoid overflow, if T very small not accept
            if T > 0:
                accept = (rng.random() < np.exp(-dE / T))
        if accept:
            cur_h, cur_score = new_h, new_score
            if cur_score < best_score:
                best_h, best_score = cur_h.copy(), cur_score
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            part = h[shift:]
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = part
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            part = h[:n - s]
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = part
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if it % 3000 == 0 and time_left() > 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def improve_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        score = compute_c5(new_h)
        if score < best_score:
            best_h, best_score = new_h, score
    return best_h, best_score

def pair_gradient_move(h, rng, time_left):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    while time_left() > 0.2:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(cur_h, shifts)
        pos = np.where(grad > 1e-8)[0]
        neg = np.where(grad < -1e-8)[0]
        if pos.size == 0 or neg.size == 0:
            i, j = rng.choice(n, size=2, replace=False)
        else:
            i = pos[np.argmax(grad[pos])]
            j = neg[np.argmin(grad[neg])]
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < cur_score:
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
    return best_h, best_score

def annealing(h, rng, time_left, init_temp=0.05, final_temp=1e-5):
    n = h.size
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    block_total = time_left()
    if block_total <= 0:
        return cur_h, cur_score
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= block_total:
            break
        frac = elapsed / block_total
        T = init_temp * ((final_temp / init_temp) ** frac)
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * min(1.0, T * 10)
        if delta <= 0:
            continue
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        dE = new_score - cur_score
        if dE < 0 or (T > 0 and rng.random() < np.exp(-dE / T)):
            cur_h, cur_score = new_h, new_score
            if cur_score < best_score:
                best_h, best_score = cur_h.copy(), cur_score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if time_left() <= 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Gradient descent
        if time_left() > 5:
            factor = 2000.0 if n <= 256 else 8000.0
            max_iters = int(min(40000, max(2000, time_left() * (factor / n))))
            step0 = 0.8 if n <= 256 else 0.3
            base, sc_gd = gradient_descent_adaptive(base, max_iters, step0=step0,
                                                    beta=0.9, rng=rng, time_left=time_left)
            if sc_gd < best_score:
                best_h, best_score = base.copy(), sc_gd

        # Targeted max‑shift improvement
        if time_left() > 5:
            base, sc_ms = improve_max_shift(base, rng, time_left)
            if sc_ms < best_score:
                best_h, best_score = base.copy(), sc_ms

        # Gradient‑guided pair moves
        if time_left() > 5:
            base, sc_pair = pair_gradient_move(base, rng, time_left)
            if sc_pair < best_score:
                best_h, best_score = base.copy(), sc_pair

        # Simulated annealing
        if time_left() > 5:
            base, sc_ann = annealing(base, rng, time_left, init_temp=0.05, final_temp=1e-5)
            if sc_ann < best_score:
                best_h, best_score = base.copy(), sc_ann

        # Random fractional restarts (few)
        if time_left() > 5:
            for _ in range(2):
                if time_left() < 2:
                    break
                h_rand = rng.random(n)
                h_rand = project_onto_box_sum(h_rand, n / 2.0)
                max_iters_rand = int(min(3000, max(500, time_left() * (1500.0 / n))))
                h_rand, sc_rand = gradient_descent_adaptive(h_rand, max_iters_rand,
                                                            step0=0.5, beta=0.9,
                                                            rng=rng, time_left=time_left)
                if sc_rand < best_score:
                    best_h, best_score = h_rand.copy(), sc_rand
                if time_left() > 3:
                    h_rand, sc_rand = improve_max_shift(h_rand, rng, time_left)
                    if sc_rand < best_score:
                        best_h, best_score = h_rand.copy(), sc_rand

    # Final intensive refinement on global best
    if time_left() > 5:
        max_iters_final = int(min(80000, max(20000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(best_h, max_iters_final,
                                                      step0=0.2, beta=0.9,
                                                      rng=rng, time_left=time_left)
    if time_left() > 2:
        best_h, best_score = improve_max_shift(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = pair_gradient_move(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = annealing(best_h, rng, time_left, init_temp=0.02, final_temp=1e-6)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
