def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=50):
    if total <= lo * v.size:
        return np.full_like(v, lo)
    if total >= hi * v.size:
        return np.full_like(v, hi)
    lo_lam = np.min(v) - hi
    hi_lam = np.max(v) - lo
    for _ in range(max_iter):
        mid = (lo_lam + hi_lam) * 0.5
        proj = np.clip(v - mid, lo, hi)
        s = proj.sum()
        if abs(s - total) < 1e-12:
            return proj
        if s > total:
            lo_lam = mid
        else:
            hi_lam = mid
    return np.clip(v - mid, lo, hi)
```

def gradient_for_shift(h, shift):
    n = h.size
    # compute neighbors
    # note: out-of-bounds neighbor considered 0
    # we compute arr1 = h shifted by +shift (i+shift)
    # arr2 = h shifted by -shift (i-shift)
    # where shift may be positive or negative
    # we can use np.roll but need to zero out wrap-around contributions.
    # Simpler: create zero-padded copy.
    # Use np.take with mode='clip'? maybe easiest: create neighbor arrays of shape n filled with 0.
    neighbor_pos = np.zeros_like(h)
    neighbor_neg = np.zeros_like(h)
    if shift >= 0:
        # positive shift: i+shift within range for i < n-shift
        neighbor_pos[:n-shift] = h[shift:]
        # negative shift: i-shift corresponds to i >= shift
        neighbor_neg[shift:] = h[:n-shift]
    else:
        s = -shift
        # shift negative => i+shift = i-s
        neighbor_pos[s:] = h[:n-s]   # i-s >=0
        neighbor_neg[:n-s] = h[s:]   # i - (-shift) = i+s
    grad = 1.0 - (neighbor_pos + neighbor_neg)
    return grad

def simulated_annealing_continuous(h0, time_budget, rng):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    # define parameters
    T0 = 0.02
    # number of steps approximated: we will adjust T each iteration
    it = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1 - frac)  # linear decay to 0
        # random i,j
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        # scale delta
        delta = max_delta * rng.random() * (T / T0)  # T/T0 in [0,1]
        # apply move
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # accept with probability exp(-(new_score-cur_score)/T)
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                cur_score = new_score
            else:
                # revert
                h[i] += delta
                h[j] -= delta
        it += 1
        # optionally occasional discrete swap if h[i] and h[j] close to 0 or 1
        if rng.random() < 0.001:
            # try binary swap
            i2 = rng.integers(n)
            j2 = rng.integers(n - 1)
            if j2 >= i2:
                j2 += 1
            # swap values if one is near 0 and other near 1
            if (h[i2] < 0.1 and h[j2] > 0.9) or (h[i2] > 0.9 and h[j2] < 0.1):
                h[i2], h[j2] = h[j2], h[i2]
                new_score2 = compute_c5(h)
                if new_score2 < cur_score:
                    cur_score = new_score2
                    if new_score2 < best_score:
                        best_score = new_score2
                        best_h = h.copy()
                else:
                    # revert
                    if new_score2 > cur_score:
                        # revert swap
                        h[i2], h[j2] = h[j2], h[i2]
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=50):
    if total <= lo * v.size:
        return np.full_like(v, lo)
    if total >= hi * v.size:
        return np.full_like(v, hi)
    low = np.min(v) - hi
    high = np.max(v) - lo
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, lo, hi)
        s = proj.sum()
        if abs(s - total) < 1e-12:
            return proj
        if s > total:
            low = mid
        else:
            high = mid
    return np.clip(v - mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def max_shift_and_corr(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    max_corr = corr[idx]
    return shift, max_corr, corr

def gradient_for_shift(h, shift):
    n = h.size
    neighbor_pos = np.zeros_like(h)
    neighbor_neg = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            neighbor_pos[:n-shift] = h[shift:]
        # neighbor_neg
        if shift < n:
            neighbor_neg[shift:] = h[:n-shift]
    else:
        s = -shift
        if s < n:
            neighbor_pos[s:] = h[:n-s]
        if s < n:
            neighbor_neg[:n-s] = h[s:]
    return 1.0 - (neighbor_pos + neighbor_neg)

def gradient_descent(h0, time_budget, rng):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    base_alpha = 0.5
    it = 0
    # simple loop with time check
    while time.time() - start < time_budget:
        shift, cur_score, _ = max_shift_and_corr(h)
        grad = gradient_for_shift(h, shift)
        alpha = base_alpha / np.sqrt(it + 1)
        h_new = project_onto_box_sum(h - alpha * grad, n / 2.0)
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # occasional random jump
            if rng.random() < 0.001:
                h = h_new
                cur_score = new_score
        it += 1
        if best_score <= 0.3808:
            break
    return best_h, best_score

import time

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=50):
    if total <= lo * v.size:
        return np.full_like(v, lo)
    if total >= hi * v.size:
        return np.full_like(v, hi)
    low = np.min(v) - hi
    high = np.max(v) - lo
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, lo, hi)
        s = proj.sum()
        if abs(s - total) < 1e-12:
            return proj
        if s > total:
            low = mid
        else:
            high = mid
    return np.clip(v - mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq

def random_balanced_binary(n, rng):

    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def max_shift_and_corr(h):

    idx = np.argmax(corr)
    shift = idx - (n - 1)
    max_corr = corr[idx]
    return shift, max_corr, corr

def gradient_for_shift(h, shift):
    n = h.size
    neighbor_pos = np.zeros_like(h)
    neighbor_neg = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            neighbor_pos[:n - shift] = h[shift:]
            neighbor_neg[shift:] = h[:n - shift]
    else:
        s = -shift
        if s < n:
            neighbor_pos[s:] = h[:n - s]
            neighbor_neg[:n - s] = h[s:]
    return 1.0 - (neighbor_pos + neighbor_neg)

def gradient_descent(h0, time_budget, rng):
    n = h0.size

    best_h = h.copy()

    start = time.time()

    it = 0
    while time.time() - start < time_budget:
        shift, cur_score, _ = max_shift_and_corr(h)
        grad = gradient_for_shift(h, shift)
        alpha = base_alpha / np.sqrt(it + 1)
        h_new = project_onto_box_sum(h - alpha * grad, n / 2.0)
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h = h_new
            cur_score = new_score

                best_h = h_new.copy()
        else:
