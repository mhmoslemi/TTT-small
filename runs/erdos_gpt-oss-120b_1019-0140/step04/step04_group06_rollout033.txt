def neighbor_sum_for_shift(h, k_abs):
    n = h.size
    if k_abs == 0:
        # neighbor sum = 2*h
        return 2 * h
    n_k = n - k_abs
    # neighbor_sum = zeros
    ns = np.zeros(n, dtype=int)
    # Add h shifted left (i.e., h[k_abs:] goes to indices 0..n_k-1)
    ns[:n_k] += h[k_abs:]
    # Add h shifted right (h[:n_k] goes to indices k_abs..n-1)
    ns[k_abs:] += h[:n_k]
    return ns

import time, math, numpy as np

def compute_c5_from_h(h):
    """Return C5 = 1 - 2*min_autocorr / n for binary h."""
    n = h.size
    auto = np.correlate(h, h, mode='full')
    min_auto = auto.min()
    return 1.0 - (2.0 * min_auto) / n

def neighbor_sum_for_shift(h, k_abs):
    """Sum of h at distance k_abs on both sides (zero padding)."""
    n = h.size
    if k_abs == 0:
        return 2 * h
    n_k = n - k_abs
    ns = np.zeros(n, dtype=int)
    ns[:n_k] += h[k_abs:]
    ns[k_abs:] += h[:n_k]
    return ns

def hill_climb(h0, time_left, budget_s, rng):
    """Greedy + annealed hill climbing on a balanced binary vector."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_from_h(h)
    best_h = h.copy()
    best_score = cur_score
    iter_without_improve = 0
    max_no_improve = 3000
    center_idx = n - 1

    while time_left() > 0:
        # current worst shift
        auto = np.correlate(h, h, mode='full')
        min_auto = auto.min()
        min_idxs = np.where(auto == min_auto)[0]
        if min_idxs.size > 1 and center_idx in min_idxs:
            min_idxs = min_idxs[min_idxs != center_idx]
        min_idx = rng.choice(min_idxs)
        k_abs = abs(min_idx - center_idx)

        ns = neighbor_sum_for_shift(h, k_abs)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        a_candidate = ones[np.argmin(ns[ones])]
        b_candidate = zeros[np.argmax(ns[zeros])]
        delta = ns[b_candidate] - ns[a_candidate]

        performed = False
        if delta > 0:
            # deterministic improvement
            h[a_candidate] = 0
            h[b_candidate] = 1
            cur_score = compute_c5_from_h(h)
            performed = True
        else:
            # random perturbation with annealing
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = 0, 1
            new_score = compute_c5_from_h(h)
            elapsed = budget_s - time_left()
            prog = elapsed / budget_s
            T = 0.01 * (1e-5 / 0.01) ** prog
            if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
                cur_score = new_score
                performed = True
            else:
                h[i], h[j] = 1, 0  # revert

        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
            iter_without_improve = 0
        else:
            iter_without_improve += 1

        if iter_without_improve >= max_no_improve:
            break

    return best_h, best_score
