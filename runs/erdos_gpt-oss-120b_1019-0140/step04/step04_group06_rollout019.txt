def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    # choose size
    n = 256  # try largest feasible
    # get base h
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != n:
            # upscale to n using upsample
            base = upsample(base, n)
    else:
        # start with Thue-Morse upsampled?
        base = thue_morse(n)  # maybe random binary? Eh Thue-Morse yields balanced binary? Let's check: thue_morse produce 0,1 pattern with equal? For n power of 2, it's balanced (since sum = n/2). So fine.
    # ensure constraints
    base = project_onto_box_sum(base, n/2.0)

    # maybe start with binary pattern for annealing
    bin_h = (base > 0.5).astype(float)
    # adjust to exact sum n/2
    diff = int(bin_h.sum() - n/2)
    if diff > 0:
        # too many ones; turn diff ones to zero
        ones_idx = np.where(bin_h == 1)[0]
        flip = rng.choice(ones_idx, size=diff, replace=False)
        bin_h[flip] = 0.0
    elif diff < 0:
        zeros_idx = np.where(bin_h == 0)[0]
        flip = rng.choice(zeros_idx, size=-diff, replace=False)
        bin_h[flip] = 1.0
    # maintain binary.

    # compute best score
    best_h = bin_h.copy()
    best_score = compute_c5(best_h)

    # simulated annealing binary swaps
    # set schedule: T_start = 0.1 maybe, T_end = 1e-4
    T0 = 0.02  # adjust based on typical c5 value ~0.38, difference maybe 0.05
    T1 = 1e-5
    # we will compute total iterations based on time left; while loop
    while time_left() > 2.0:
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = T0 * (T1 / T0) ** frac
        # propose swap
        # choose random i where h[i]=1 and j where h[j]=0
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # apply swap
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc_new = compute_c5(h_new)
        # acceptance
        if sc_new < best_score or rng.random() < math.exp(-(sc_new - best_score) / T):
            best_h = h_new
            best_score = sc_new
        # optionally periodically try gradient refinement
        if rng.random() < 0.01 and time_left()>1.0:
            # small gradient descent
            # Use gradient descent from current best_h (float)
            # Set step count based on remaining time maybe 2000 steps
            steps = int(min(2000, max(200, time_left()*50)))
            h_cont, sc_cont = grad_descent(best_h, steps, step0=0.3, beta=0.9, rng=rng, time_left=time_left)
            if sc_cont < best_score:
                best_h = h_cont
                best_score = sc_cont
    # final continuous refinement
    if time_left() > 5.0:
        # start from best_h (float)
        steps = int(min(50000, max(5000, time_left()*30)))
        h_ref, sc_ref = grad_descent(best_h, steps, step0=0.4, beta=0.9, rng=rng, time_left=time_left)
        if sc_ref < best_score:
            best_h = h_ref
            best_score = sc_ref

    best_h = project_onto_box_sum(best_h, n/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1 = h shifted left by shift (i.e., h[i+shift] corresponds to i)
            term1 = np.empty(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term1[n-shift:] = 0.0
            # term2 = h shifted right by shift (i.e., h[i-shift] corresponds to i)
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:n-shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:n-s] = h[s:]
            term2[n-s:] = 0.0
        grad += 1.0 - term1 - term2
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        # occasionally add noise to escape
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project onto box and sum
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()

    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # Target size
    n = 256
    # Prepare base h
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != n:
            base = upsample(base, n)
    else:
        base = thue_morse(n)
    base = project_onto_box_sum(base, n/2.0)

    # Convert to binary start
    bin_h = (base > 0.5).astype(float)
    # Adjust to exact sum
    diff = int(bin_h.sum() - n/2)
    if diff > 0:
        ones_idx = np.where(bin_h == 1)[0]
        flip = rng.choice(ones_idx, size=diff, replace=False)
        bin_h[flip] = 0.0
    elif diff < 0:
        zeros_idx = np.where(bin_h == 0)[0]
        flip = rng.choice(zeros_idx, size=-diff, replace=False)
        bin_h[flip] = 1.0

    cur_h = bin_h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # Simulated annealing parameters
    T0 = best_score * 0.2 + 1e-4   # start temperature
    T1 = 1e-6

    while time_left() > 2.0:
        # temperature schedule based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = T0 * (T1 / T0) ** frac

        # propose swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = cur_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc_new = compute_c5(h_new)

        # acceptance
        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            cur_h = h_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()

        # occasional gradient refinement
        if rng.random() < 0.02 and time_left() > 5.0:
            steps = int(min(3000, max(500, time_left() * 20)))
            h_tmp, sc_tmp = grad_descent(cur_h, steps, step0=0.35, beta=0.9, rng=rng, time_left=time_left)
            if sc_tmp < best_score:
                best_score = sc_tmp
                best_h = h_tmp.copy()
                cur_h = h_tmp.copy()
                cur_score = sc_tmp

    # final continuous refinement from best
    if time_left() > 5.0:
        steps = int(min(40000, max(5000, time_left() * 30)))
        h_ref, sc_ref = grad_descent(best_h, steps, step0=0.4, beta=0.9, rng=rng, time_left=time_left)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    best_h = project_onto_box_sum(best_h, n/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    n = 256
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != n:
            base = upsample(base, n)
    else:
        base = thue_morse(n)
    base = project_onto_box_sum(base, n / 2.0)

    bin_h = (base > 0.5).astype(float)
    diff = int(bin_h.sum() - n / 2)
    if diff > 0:
        ones = np.where(bin_h == 1)[0]
        flip = rng.choice(ones, size=diff, replace=False)
        bin_h[flip] = 0.0
    elif diff < 0:
        zeros = np.where(bin_h == 0)[0]
        flip = rng.choice(zeros, size=-diff, replace=False)
        bin_h[flip] = 1.0

    cur_h = bin_h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    last_improve = time.time()

    T0 = best_score * 0.2 + 1e-4
    T1 = 1e-6

    while time_left() > 2.0:
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = T0 * (T1 / T0) ** frac

        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = cur_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc_new = compute_c5(h_new)

        if sc_new < cur_score or rng.random() < math.exp(-(sc_new - cur_score) / T):
            cur_h, cur_score = h_new, sc_new
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                last_improve = time.time()

        if rng.random() < 0.02 and time_left() > 5.0:
            steps = int(min(3000, max(500, time_left() * 20)))
            h_tmp, sc_tmp = grad_descent(cur_h, steps, step0=0.35,
                                         beta=0.9, rng=rng, time_left=time_left)
            if sc_tmp < best_score:
                best_score, best_h = sc_tmp, h_tmp.copy()
                cur_h, cur_score = h_tmp.copy(), sc_tmp
                last_improve = time.time()

        if time.time() - last_improve > 30 and time_left() > 60:
            cur_h = random_binary_balanced(n, rng)
            cur_score = compute_c5(cur_h)
            last_improve = time.time()

    if time_left() > 5.0:
        steps = int(min(40000, max(5000, time_left() * 30)))
        h_ref, sc_ref = grad_descent(best_h, steps, step0=0.4,
                                     beta=0.9, rng=rng, time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
