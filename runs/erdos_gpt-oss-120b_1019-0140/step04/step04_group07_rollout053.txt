def gradient_descent(h0, time_left, max_iter=100000, init_step=0.8, beta=0.9, tol_shift=1e-12, jitter_interval=10000):
    h = h0.copy()
    n = h.size
    s = n/2.0
    step = init_step
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score

    it = 0
    while it < max_iter and time_left() > 0:
        it += 1
        # compute gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = corr.max()
        # identify top shifts: use up to top_shifts = min(5, number of shifts with corr near max)
        # approach: choose top 5 shifts
        k = min(5, len(corr))
        top_idx = np.argpartition(-corr, k-1)[:k]  # unsorted indices
        top_shifts = top_idx - (n-1)
        # compute gradient using these shifts (similar to compute_gradients_for_shifts)
        grad = np.zeros_like(h)
        for shift in top_shifts:
            if shift >= 0:
                term1 = np.zeros(n, dtype=h.dtype)
                term2 = np.zeros(n, dtype=h.dtype)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                sshift = -shift
                term1 = np.zeros(n, dtype=h.dtype)
                term2 = np.zeros(n, dtype=h.dtype)
                term1[sshift:] = h[:n-sshift]
                term2[:n-sshift] = h[sshift:]
            grad += (1.0 - term1 - term2)
        grad /= len(top_shifts)

        # momentum and update
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        # project
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(init_step, step * 1.02)  # increase
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = init_step * 0.1

        if it % jitter_interval == 0:
            jitter = np.random.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def run(seed=42,budget_s=1100, **kwargs):
    import time, numpy as ...

    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)
    rng = np.random.default_rng(seed)

    # Acquire base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0]/2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # progression of scales
    scales = [128, 256, 512, 1024]
    for n in scales:
        if time_left() < 5:
            break
        # upsample base to n
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n/2.0)

        # gradient descent
        max_iters = int(min(50000, time_left()*10)) # allocate many iters; but time_left *10 may exceed enough
        # compute step based on n: larger n may need smaller step? maybe initial step = 0.5
        step0 = 0.5 if n >= 512 else 0.8
        gd_h, gd_score = gradient_descent(base, time_left, max_iter=max_iters, init_step=step0, beta=0.9, jitter_interval=2000)
        if gd_score < best_score:
            best_h, best_score = gd_h.copy(), gd_score
        base = gd_h

        # targeted shift improvement
        if time_left() > 2:
            ms_h, ms_score = improve_via_max_shift(base, rng, time_left)
            if ms_score < best_score:
                best_h, best_score = ms_h.copy(), ms_score
            base = ms_h

        # local pair moves
        if time_left() > 2:
            pair_h, pair_score = local_pair_moves(base, rng, time_left)
            if pair_score < best_score:
                best_h, best_score = pair_h.copy(), pair_score
            base = pair_h

        # random binary restarts (maybe only for small n)
        if n <= 256 and time_left() > 5:
            for _ in range(2):
                # random balanced binary
                h_rand = random_binary_balanced(n, rng)
                # refine a few steps
                rand_it = int(min(8000, time_left()*5))
                rand_h, rand_score = gradient_descent(h_rand, time_left, max_iter=rand_it, init_step=0.6, beta=0.9, jitter_interval=2000)
                if rand_score < best_score:
                    best_h, best_score = rand_h.copy(), rand_score
                # pair moves
                if time_left() > 2:
                    p_h, p_score = local_pair_moves(rand_h, rng, time_left)
                    if p_score < best_score:
                        best_h, best_score = p_h.copy(), p_score

    # final refinement on global best
    if time_left() > 10:
        final_iters = int(min(150000, time_left()*30))
        final_h, final_score = gradient_descent(best_h, time_left, max_iter=final_iters, init_step=0.3, beta=0.9, jitter_interval=5000)
        best_h, best_score = final_h, final_score

    # final targeted and pair moves
    if time_left() > 5:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 5:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = corr.max()
        # find shift with max
        shift_idxs = np.where(corr == max_corr)[0]
        # randomly choose one if multiple
        shift_idx = rng.choice(shift_idxs)
        shift = shift_idx - (n-1)
        # compute contributions
        if shift >= 0:
            contrib = best_h[:n-shift] * (1.0 - best_h[shift:])
            if contrib.size == 0:
                continue
            i_local = np.argmax(contrib)
            i = i_local
            j = i + shift
        else:
            sshift = -shift
            contrib = best_h[sshift:] * (1.0 - best_h[:n-sshift])
            if contrib.size == 0:
                continue
            i_local = np.argmax(contrib)
            i = i_local + sshift
            j = i - sshift
        # transfer mass from i to j (i currently high, j low)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            # maybe try opposite direction
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # try moving from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        # try moving from j to i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure the root bracket includes s
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # Return binary Thue-Morse sequence as floats 0/1
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def gradient_descent(h0, time_left, max_iter=100000, init_step=0.8, beta=0.9, rng=None, jitter_interval=5000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    step = init_step
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    it = 0
    while it < max_iter and time_left() > 0:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # choose up to 5 top shifts
        k_shifts = 5
        if k_shifts < len(corr):
            idxs = np.argpartition(-corr, k_shifts-1)[:k_shifts]
        else:
            idxs = np.arange(len(corr))
        shifts = idxs - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.empty(n, dtype=h.dtype)
                term2 = np.empty(n, dtype=h.dtype)
                term1[:n-shift] = h[shift:]
                term1[n-shift:] = 0.0
                term2[shift:] = h[:n-shift]
                term2[:shift] = 0.0
            else:
                sshift = -shift
                term1 = np.empty(n, dtype=h.dtype)
                term2 = np.empty(n, dtype=h.dtype)
                term1[sshift:] = h[:n-sshift]
                term1[:sshift] = 0.0
                term2[:n-sshift] = h[sshift:]
                term2[n-sshift:] = 0.0
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(init_step, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = init_step * 0.1
        if jitter_interval and it % jitter_interval == 0 and rng is not None:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = corr.max()
        shift_idxs = np.where(corr == max_corr)[0]
        if shift_idxs.size == 0:
            break
        shift_idx = rng.choice(shift_idxs)
        shift = int(shift_idx) - (n - 1)
        if shift >= 0:
            contrib = best_h[:n-shift] * (1.0 - best_h[shift:])
            if contrib.size == 0:
                continue
            i_local = np.argmax(contrib)
            i = i_local
            j = i + shift
        else:
            sshift = -shift
            contrib = best_h[sshift:] * (1.0 - best_h[:n-sshift])
            if contrib.size == 0:

            i_local = np.argmax(contrib)
            i = i_local + sshift
            j = i - sshift
        # try moving mass i->j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
                continue
        # try moving mass j->i
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta

            contrib = best_h[sshift:] * (1.0 - best_h[:n - sshift])
            if contrib.size == 0:
                continue
            i_local = np.argmax(contrib)
            i = i_local + sshift
            j = i - sshift
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
                continue
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [128, 256, 512, 1024]
    for n in scales:
        if time_left() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        max_iters = int(min(50000, time_left() * 10))
        step0 = 0.5 if n >= 512 else 0.8
        gd_h, gd_score = gradient_descent(
            base, time_left, max_iter=max_iters,
            init_step=step0, beta=0.9, rng=rng, jitter_interval=2000)
        if gd_score < best_score:
            best_h, best_score = gd_h.copy(), gd_score
        base = gd_h

        if time_left() > 2:
            ms_h, ms_score = improve_via_max_shift(base, rng, time_left)
            if ms_score < best_score:
                best_h, best_score = ms_h.copy(), ms_score
            base = ms_h

        if time_left() > 2:
            pair_h, pair_score = local_pair_moves(base, rng, time_left)
            if pair_score < best_score:
                best_h, best_score = pair_h.copy(), pair_score
            base = pair_h

        if n <= 256 and time_left() > 5:
            for _ in range(2):
                if time_left() < 2:
                    break
                h_rand = random_binary_balanced(n, rng)
                rand_iters = int(min(8000, time_left() * 5))
                rand_h, rand_score = gradient_descent(
                    h_rand, time_left, max_iter=rand_iters,
                    init_step=0.6, beta=0.9, rng=rng, jitter_interval=2000)
                if rand_score < best_score:
                    best_h, best_score = rand_h.copy(), rand_score
                if time_left() > 2:
                    p_h, p_score = local_pair_moves(rand_h, rng, time_left)
                    if p_score < best_score:
                        best_h, best_score = p_h.copy(), p_score

    if time_left() > 10:
        final_iters = int(min(150000, time_left() * 30))
        best_h, best_score = gradient_descent(
            best_h, time_left, max_iter=final_iters,
            init_step=0.3, beta=0.9, rng=rng, jitter_interval=5000)

    while time_left() > 5:
        h_jit = best_h + rng.normal(scale=0.02, size=best_h.size)
        h_jit = project_onto_box_sum(h_jit, best_h.size / 2.0)
        short_iters = int(min(5000, time_left() * 5))
        h_ref, sc_ref = gradient_descent(
            h_jit, time_left, max_iter=short_iters,
            init_step=0.4, beta=0.9, rng=rng, jitter_interval=1000)
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref
        else:
            break

    if time_left() > 5:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 5:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
