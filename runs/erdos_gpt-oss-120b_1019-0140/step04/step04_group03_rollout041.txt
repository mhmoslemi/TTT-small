def gradient_descent_custom(h0, max_iters, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time.time() - start >= max_time:
            break
        # compute correlation to get worst shift(s)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # identify shifts achieving max_corr (worst)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)   # step schedule
        h_new = h - eta * momentum
        # optional random noise injection few times
        if it % 10000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def local_swap_improvement(h, max_steps, rng):
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for step in range(max_steps):
        # compute correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shift = np.argmax(corr) - (n - 1)
        # Determine loss and gain positions
        # loss: positions i where h[i] > 0 and (i+shift >= n or h[i+shift] == 0)
        # gain: positions j where h[j] < 1 and j+shift < n and h[j+shift] > 0
        # Use boolean masks
        i_range = np.arange(n)
        if shift >= 0:
            # compute mask for overlap region existence
            mask_i = i_range + shift < n
            overlap_i = mask_i & (h[i_range + shift] >= 1e-12)
            # loss mask
            loss_mask = (h > 1e-12) & (~overlap_i | ~mask_i)  # if i+shift out-of-range or h[i+shift]==0
            # gain mask: i where h[i] < 1 and overlap_i and h[i+shift] > small
            gain_mask = (h < 1 - 1e-12) & overlap_i & (h[i_range + shift] > 1e-12)
        else:
            s = -shift
            mask_i = i_range >= s
            overlap_i = mask_i & (h[i_range - s] >= 1e-12)
            loss_mask = (h > 1e-12) & (~overlap_i | ~mask_i)
            gain_mask = (h < 1 - 1e-12) & overlap_i & (h[i_range - s] > 1e-12)
        loss_idx = np.where(loss_mask)[0]
        gain_idx = np.where(gain_mask)[0]
        if loss_idx.size == 0 or gain_idx.size == 0:
            # try random perturbation if no swap possible
            perturb = rng.normal(scale=0.01, size=n)
            h_new = np.clip(h + perturb, 0.0, 1.0)
            h_new = project_onto_box_sum(h_new, target_sum)
        else:
            i = rng.choice(loss_idx)
            j = rng.choice(gain_idx)
            # compute max possible delta from i to j (ensuring bounds)
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 0:
                continue
            delta = max_move * rng.random()
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score or rng.random() < np.exp(-(score - best_score) * 10):  # accept with some probability
            h = h_new
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(s - target_sum) < 1e-12:
            break
    return proj

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_custom(h0, max_iters, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time.time() - start >= max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 20000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

    x = np.linspace(0.0, 2.0, n_final, endpoint=False)
    h_cos = 0.5 * (1.0 + np.cos(np.pi * (x - 1.0)))
    h_cos = np.clip(h_cos, 0.0, 1.0)
    h_cos = project_onto_box_sum(h_cos, target_sum)
    candidates.append(h_cos)

    # triangular bump
    h_tri = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    h_tri = project_onto_box_sum(h_tri, target_sum)
    candidates.append(h_tri)

    # use parent construction if provided
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = upsample(base, n_final)
        base = project_onto_box_sum(base, target_sum)
        candidates.append(base)

    # random balanced binary
    h_rand = np.zeros(n_final, dtype=float)
    idx = rng.choice(n_final, size=n_final // 2, replace=False)
    h_rand[idx] = 1.0
    candidates.append(h_rand)

    # main optimization loop over candidates
    for h0 in candidates:
        remaining = budget_s - (time.time() - start)
        if remaining <= 5.0:
            break
        max_iters = int(min(200000, max(50000, remaining * (50000.0 / n_final))))
        h_opt, sc_opt = gradient_descent_custom(
            h0, max_iters=max_iters, step0=0.6, beta=0.9, rng=rng,
            max_time=remaining)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

        # simulated‑annealing refinement on the current best
        remaining = budget_s - (time.time() - start)
        if best_h is not None and remaining > 5.0:
            total_steps = int(min(50000, max(10000, remaining * 500.0)))
            T0 = 0.02
            T_final = 1e-6
            h_cur = best_h.copy()
            cur_score = best_score
            for step in range(total_steps):
                if time.time() - start >= budget_s:
                    break
                T = T0 * ((total_steps - step) / total_steps) + T_final
                i, j = rng.choice(n_final, size=2, replace=False)
                max_move = min(h_cur[i], 1.0 - h_cur[j])
                if max_move <= 0.0:
                    continue
                delta = max_move * rng.random()
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_score = compute_c5(h_new)
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                    h_cur = h_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score, best_h = cur_score, h_cur.copy()
    # final long refinement on the global best
    if best_h is not None:
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            max_iters = int(min(300000, max(80000, remaining * (80000.0 / n_final))))
            best_h, best_score = gradient_descent_custom(
                best_h, max_iters=max_iters, step0=0.3, beta=0.9,
                rng=rng, max_time=remaining)
    else:
        # fallback construction
        best_h = project_onto_box_sum(thue_morse(n_final), target_sum)
        best_score = compute_c5(best_h)

    # ensure feasibility and recompute final score
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_final
