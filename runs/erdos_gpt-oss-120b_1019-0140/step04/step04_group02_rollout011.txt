def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random perturbation
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    # base init
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64) # length 64
        base = project_onto_box_sum(base, 64/2.0)
    # target scales
    scales = [128, 256, 512]
    best_h = None
    best_score = np.inf
    for n in scales:
        if time_left() < 5.0:
            break
        # adjust base size
        if base.shape[0] != n:
            base = upsample(base, n)
            base = project_onto_box_sum(base, n/2.0)
        # perform restarts
        n_restarts = 4
        for r in range(n_restarts):
            if time_left() < 5.0:
                break
            # choose initialization
            if r % 2 == 0:
                h0 = random_binary_balanced(n, rng)
            else:
                h0 = base + rng.normal(scale=0.05, size=n)
                h0 = project_onto_box_sum(h0, n/2.0)
            # gradient descent
            iter_count = 150000 if n <= 256 else 300000
            h_opt, sc_opt = gradient_descent(h0, n_iter=iter_count,
                                             step0=0.6, beta=0.9,
                                             rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()
        # after restarts, refine best at this scale with more iterations
        if best_h is not None and best_h.shape[0] == n:
            # long refinement
            extra_iters = 400000 if n == 512 else 200000
            h_opt, sc_opt = gradient_descent(best_h, n_iter=extra_iters,
                                             step0=0.3, beta=0.9,
                                             rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()
        # local search
        if best_h is not None and best_h.shape[0] == n:
            best_h, best_score = local_search(best_h, best_score, rng, time_left)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation of arr to new_n points (domain [0,1))."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector of length n (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) via FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient of the max‑correlation term (averaged over all max shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent(h0, n_iter, step0, beta, rng, start_time, budget_s):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    corr = compute_corr_full_fft(h)
    best_score = np.max(corr) * (2.0 / n)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if budget_s - (time.time() - start_time) <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        corr_new = compute_corr_full_fft(h_new)
        score = np.max(corr_new) * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def local_search(h, score, rng, start_time, budget_s):
    """Simple random mass‑transfer local search."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = score
    while budget_s - (time.time() - start_time) > 1.0:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.uniform(0.01, 0.2)
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
        base = project_onto_box_sum(base, 64 / 2.0)

    scales = [128, 256, 512]
    best_h = None
    best_score = np.inf

    for n in scales:
        # Ensure we have enough time for this scale
        if budget_s - (time.time() - start_time) < 5.0:
            break

        # Resize base to current resolution
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Multiple restarts with simple initializations
        for r in range(4):
            if budget_s - (time.time() - start_time) < 5.0:
                break
            if r % 2 == 0:
                h0 = random_binary_balanced(n, rng)
            else:
                h0 = base + rng.normal(scale=0.05, size=n)
                h0 = project_onto_box_sum(h0, n / 2.0)

            iters = 150000 if n <= 256 else 300000
            h_opt, sc_opt = gradient_descent(
                h0, n_iter=iters, step0=0.6, beta=0.9,
                rng=rng, start_time=start_time, budget_s=budget_s)

            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()

        # Refine the best of this resolution with extra iterations
        if best_h is not None and best_h.shape[0] == n:
            extra_iters = 400000 if n == 512 else 200000
            h_opt, sc_opt = gradient_descent(
                best_h, n_iter=extra_iters, step0=0.3, beta=0.9,
                rng=rng, start_time=start_time, budget_s=budget_s)
            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()

        # Local random mass‑transfer search
        if best_h is not None and best_h.shape[0] == n:
            best_h, best_score = local_search(
                best_h, best_score, rng, start_time, budget_s)

        # Prepare for next finer scale
        base = best_h if best_h is not None else base

    # Final random binary explorations if time permits
    while best_h is not None and budget_s - (time.time() - start_time) > 2.0:
        h_bin = random_binary_balanced(best_h.shape[0], rng)
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

    # Fallback in case nothing was found
    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(512), 512 / 2.0)
        best_score = compute_c5_fft(best_h)

    # Final projection and exact evaluation
    best_h = project_onto_box_sum(best_h, best_h.shape[0] / 2.0)
    best_score = compute_c5_fft(best_h)
    n_ret = best_h.shape[0]

    return best_h, best_score, n_ret
