import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def gradient_for_lag(h, lag):
    """Subgradient of the active overlap term for given lag."""
    n = h.size
    g = np.ones(n, dtype=float)
    if lag >= 0:
        if lag < n:
            g[:n - lag] -= h[lag:]
            g[lag:] -= h[:n - lag]
    else:
        L = -lag
        if L < n:
            g[L:] -= h[:n - L]
            g[:n - L] -= h[L:]
    return g


def subgradient_descent(h0, time_limit, rng, target=0.3808, alpha0=0.5):
    """Projected subgradient descent on max overlap objective."""
    h = h0.copy()
    n = h.size
    full = np.correlate(h, 1.0 - h, mode='full')
    cur_score = full.max() * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        max_idx = full.argmax()
        lag = max_idx - (n - 1)
        g = gradient_for_lag(h, lag)
        step = alpha0 / np.sqrt(it + 1)
        cand = project_onto_box_sum(h - step * g, n / 2.0)
        cand_full = np.correlate(cand, 1.0 - cand, mode='full')
        cand_score = cand_full.max() * (2.0 / n)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand.copy()
            if best_score <= target:
                break
        if cand_score <= cur_score:
            h = cand
            full = cand_full
            cur_score = cand_score
        it += 1
        # occasional random jitter to escape stagnation
        if it % 1000 == 0 and rng.random() < 0.05:
            i, j = rng.integers(n), rng.integers(n - 1)
            if j >= i:
                j += 1
            delta = min(h[i], 1.0 - h[j]) * rng.random() * 0.3
            if delta > 0:
                h[i] -= delta
                h[j] += delta
                h = project_onto_box_sum(h, n / 2.0)
                full = np.correlate(h, 1.0 - h, mode='full')
                cur_score = full.max() * (2.0 / n)
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # base initialization

    scales = [128, 256, 512, 1024]

        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        base = cand_h

            break

    # final local stochastic improvements

        # random balanced binary perturbation

        sc_rand = compute_c5(h_rand)

        if best_score <= 0.3808:

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
