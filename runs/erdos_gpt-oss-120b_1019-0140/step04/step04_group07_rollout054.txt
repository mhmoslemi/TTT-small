def compute_subgradient(h):

    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
    grad = np.zeros_like(h)

def gradient_descent(h0, step0, beta, max_iters, time_left):
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    for it in range(max_iters):
        if time_left() <= 0: break
        grad = compute_subgradient(h)
        momentum = beta * momentum + (1 - beta) * grad
        h_new = project_onto_box_sum(h - step * momentum, s)
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0 * 10, step * 1.05)  # increase up to some max
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            step = step * 0.5
            # maybe keep h unchanged
        # occasional random jitter
        if it % 500 == 0 and it > 0:
            jitter = np.random.default_rng().normal(scale=0.01, size=h.shape)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
    return best_h, best_score

def binary_anneal(h_one_hot, rng, time_left, start_temp=0.05, end_temp=0.0005):
    # h is binary array with sum = n/2
    n = h_one_hot.size
    s = n // 2
    cur_h = h_one_hot.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    total_time = time_left()
    # We'll compute schedule based on elapsed proportion.
    start_time = time.time()
    while time_left() > 0:
        elapsed = time.time() - start_time
        proportion = elapsed / total_time if total_time > 0 else 1.0
        # Temperature linearly decreasing
        T = start_temp + (end_temp - start_temp) * proportion
        # Choose random pair (i with 1, j with 0)
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_h = cur_h.copy()
                best_score = new_score
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Determine target resolution: let's try n = 256 initially, then maybe 512 if time allows.
    # Base solution from initial_h_values or Thue-Morse
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)   # need to define thue_morse

    # Ensure projection onto box sum for base
    n0 = base.shape[0]
    # if base not sum to n0/2, project
    base = project_onto_box_sum(base, n0 / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # List of scales to try
    scales = [128, 256, 512]
    for n in scales:
        if time_left() <= 1:
            break
        # upsample base to size n
        if best_h.shape[0] != n:
            # linear interpolation
            x_old = np.linspace(0.0, 1.0, best_h.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            h_upsampled = np.interp(x_new, x_old, best_h)
        else:
            h_upsampled = best_h.copy()
        # project
        h_upsampled = project_onto_box_sum(h_upsampled, n / 2.0)
        # gradient descent
        max_iters = int(min(20000, max(2000, time_left() * (5000.0 / n))))
        h_gd, sc_gd = gradient_descent(h_upsampled, step0=0.8, beta=0.9,
                                        max_iters=max_iters,
                                        time_left=time_left)
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd
        # binary annealing on rounded version
        if time_left() > 2:
            # round to nearest 0/1 while preserving sum
            # we can get binary by sorting values and set top n/2 to 1
            idx = np.argsort(-h_gd)  # descending
            binary = np.zeros(n, dtype=int)
            binary[idx[:n//2]] = 1
            h_bin, sc_bin = binary_anneal(binary, rng, time_left)
            if sc_bin < best_score:
                best_h, best_score = h_bin.astype(float), sc_bin
            # Further gradient descent from binary (as float)
            max_iters2 = int(min(10000, max(1000, time_left() * (3000.0 / n))))
            h_gd2, sc_gd2 = gradient_descent(best_h, step0=0.5, beta=0.9,
                                             max_iters=max_iters2,
                                             time_left=time_left)
            if sc_gd2 < best_score:
                best_h, best_score = h_gd2.copy(), sc_gd2
    # final very fine gradient descent on best solution at its current size
    if time_left() > 2:
        max_iters_final = int(min(50000, max(20000, time_left() * 20)))
        best_h, best_score = gradient_descent(best_h, step0=0.3, beta=0.9,
                                              max_iters=max_iters_final,
                                              time_left=time_left)
    # ensure constraints
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the interval brackets the target sum
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_subgradient(h, tol=1e-12):
    """Subgradient of the max‑over‑shift objective (averaged over worst shifts)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifts = np.where(corr >= max_corr - tol)[0] - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            i = np.arange(n - shift)
            j = i + shift
        else:
            s = -shift
            i = np.arange(s, n)
            j = i - s
        grad[i] += (1.0 - h[j])
        grad[j] += -h[i]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent(h0, step0, beta, max_iters, rng, time_left):
    """Projected gradient descent with momentum and occasional jitter."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(max_iters):
        if time_left() <= 0:
            break
        grad = compute_subgradient(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = project_onto_box_sum(h - step * momentum, target_sum)
        new_score = compute_c5(h_new)

        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0 * 10, step * 1.05)
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            step *= 0.5
            if step < 1e-6:
                # small jitter to possibly escape
                jitter = rng.normal(scale=0.01, size=h.shape)
                h = project_onto_box_sum(h + jitter, target_sum)
                cur_score = compute_c5(h)

        if (it + 1) % 1000 == 0:
            jitter = rng.normal(scale=0.02, size=h.shape)
            h = project_onto_box_sum(h + jitter, target_sum)
            cur_score = compute_c5(h)

    return best_h, best_score


def binary_anneal(bin_h, rng, time_left,
                 start_temp=0.05, end_temp=0.0005):
    """Simulated annealing on a balanced binary vector."""
    n = bin_h.size
    cur_h = bin_h.copy()
    cur_score = compute_c5(cur_h.astype(float))
    best_h = cur_h.copy()
    best_score = cur_score
    total = max(time_left(), 1e-6)
    start = time.time()

    while time_left() > 0:
        elapsed = time.time() - start
        prop = elapsed / total
        T = start_temp + (end_temp - start_temp) * prop

        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_score = compute_c5(cur_h.astype(float))
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_h = cur_h.copy()
                best_score = new_score
        else:
            cur_h[i] = 1
            cur_h[j] = 0

    return best_h.astype(float), best_score


def local_pair_moves(h, rng, time_left, max_moves=None):
    """Transfer a small amount of mass between two positions if it improves."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    moves = 0
    while time_left() > 0:
        if max_moves is not None and moves >= max_moves:
            break
        i, j = rng.choice(n, size=2, replace=False)
        # try moving from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            # try opposite direction
            max_delta = min(best_h[j], 1.0 - best_h[i])
            if max_delta <= 0:
                moves += 1
                continue
            i, j = j, i
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
        moves += 1
    return best_h, best_score


def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    seq = np.arange(n, dtype=np.uint64)
    bits = np.unpackbits(seq[:, None].view(np.uint8), axis=1)
    parity = bits.sum(axis=1) & 1
    return parity.astype(float)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
