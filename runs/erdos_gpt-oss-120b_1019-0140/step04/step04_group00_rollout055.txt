def smoothed_gradient(h, p, K):
    n = h.shape[0]
    # compute conv via FFT
    h_rev = h[::-1]
    N = 1 << ((2*n - 1).bit_length())
    conv = np.fft.irfft(np.fft.rfft(h_rev, n=N) * np.fft.rfft(1.0 - h, n=N), n=N)[:2*n-1]
    # find top K indices
    if K >= conv.size:
        top_idx = np.arange(conv.size)
    else:
        top_idx = np.argpartition(-conv, K-1)[:K]
    maxc = conv.max()
    # compute weights
    exp_vals = np.exp(p * (conv[top_idx] - maxc))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, weight in zip(top_idx, w):
        shift = idx - offset
        if shift >= 0:
            d = shift
            if n - d > 0:
                grad[:n-d] += weight * (1.0 - h[d:])
            if d < n:
                grad[d:] -= weight * h[:n-d]
        else:
            d = -shift
            if n - d > 0:
                grad[d:] += weight * (1.0 - h[:n-d])
            if d < n:
                grad[:n-d] -= weight * h[d:]
    return grad, conv.max()

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

    return conv.max() * (2.0 / n)

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def smoothed_gradient(h, p, K):
    n = h.size
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    conv = np.fft.irfft(np.fft.rfft(h_rev, n=N) * np.fft.rfft(1.0 - h, n=N), n=N)[:2 * n - 1]
    maxc = conv.max()
    if K >= conv.size:
        idx_top = np.arange(conv.size)
    else:
        idx_top = np.argpartition(-conv, K - 1)[:K]
    exp_vals = np.exp(p * (conv[idx_top] - maxc))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, weight in zip(idx_top, w):
        shift = idx - offset
        if shift >= 0:
            d = shift
            if n - d > 0:
                grad[:n - d] += weight * (1.0 - h[d:])
            if d < n:
                grad[d:] -= weight * h[:n - d]
        else:
            d = -shift
            if n - d > 0:
                grad[d:] += weight * (1.0 - h[:n - d])
            if d < n:
                grad[:n - d] -= weight * h[d:]
    return grad, maxc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return max(0.0, budget_s - (time.time() - start))

    n = 1024
    target_sum = n / 2.0
    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        candidates.append(project_onto_box_sum(init, target_sum))

    candidates.append(thue_morse_binary(n))

    for _ in range(3):
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0
        candidates.append(h)

    for _ in range(3):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))

    best_h = None
    best_s = np.inf
    K = 30
    step0 = 0.5
    beta = 0.9

    for h0 in candidates:
        if time_left() < 1.0:
            break
        h = project_onto_box_sum(h0, target_sum)
        mom = np.zeros_like(h)
        p = 10.0
        it = 1
        max_iter = 200000
        while it <= max_iter and time_left() > 0.05:
            grad, _ = smoothed_gradient(h, p, K)
            mom = beta * mom + (1.0 - beta) * grad
            eta = step0 / np.sqrt(it)
            h_new = project_onto_box_sum(h - eta * mom, target_sum)
            s_new = compute_c5_fast(h_new)
            if s_new < best_s:
                best_s, best_h = s_new, h_new.copy()
                if best_s <= 0.3808:
                    return best_h, best_s, n
            h = h_new
            if it % 2000 == 0:
                p = min(200.0, p * 1.2)
            it += 1

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), target_sum)
        best_s = compute_c5_fast(best_h)

    # final refinement with a short binary swap anneal (optional)
    if time_left() > 0.5:
        h_bin = best_h.copy()
        cur = compute_c5_fast(h_bin)
        best_h2, best_s2 = h_bin.copy(), cur
        T, T_min, decay = 0.2, 1e-5, 0.99998
        while time_left() > 0.01:
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_bin[i], h_bin[j] = 0.0, 1.0
            new = compute_c5_fast(h_bin)
            delta = new - cur
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur = new
                if new < best_s2:
                    best_s2, best_h2 = new, h_bin.copy()
            else:
                h_bin[i], h_bin[j] = 1.0, 0.0
            T = max(T_min, T * decay)
        if best_s2 < best_s:
            best_s, best_h = best_s2, best_h2.copy()

    best_h = project_onto_box_sum(best_h, target_sum)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s, n
