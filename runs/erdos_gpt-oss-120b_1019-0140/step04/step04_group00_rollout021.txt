        new_pop = []

            # tournament selection
            def tournament():

            parent2 = tournament()
            if rng.random() < crossover_rate:

def two_point_crossover(p1, p2, rng, target):
    n = p1.size
    c1 = rng.integers(1, n-1)
    c2 = rng.integers(1, n-1)
    if c2 < c1:

    offspring = np.concatenate((p1[:c1], p2[c1:c2], p1[c2:]))
    # repair
    ones = np.flatnonzero(offspring)
    count = ones.size
    if count > target:
        # too many ones
        to_zero = rng.choice(ones, count - target, replace=False)
        offspring[to_zero] = 0.0
    elif count < target:
        zeros = np.flatnonzero(offspring == 0)
        to_one = rng.choice(zeros, target - count, replace=False)
        offspring[to_one] = 1.0
    # ensure float dtype
    return offspring.astype(float)

import time, numpy as np

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2*n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n - 1]
    return corr.max() * (2.0 / n)

def thue_morse_binary(n):
    # returns array of 0/1 floats
    # compute parity of number of 1 bits for each index
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def two_point_crossover(p1, p2, rng, target):
    n = p1.size
    c1 = rng.integers(1, n-1)
    c2 = rng.integers(1, n-1)
    if c2 < c1:
        c1, c2 = c2, c1
    offspring = np.concatenate((p1[:c1], p2[c1:c2], p1[c2:]))
    # repair to target sum
    ones = np.flatnonzero(offspring)
    cnt = ones.size
    if cnt > target:
        to_zero = rng.choice(ones, cnt - target, replace=False)
        offspring[to_zero] = 0.0
    elif cnt < target:
        zeros = np.flatnonzero(offspring == 0)
        to_one = rng.choice(zeros, target - cnt, replace=False)
        offspring[to_one] = 1.0
    return offspring.astype(float)

def swap_mutation(ind, rng):
    ones = np.flatnonzero(ind == 1)
    zeros = np.flatnonzero(ind == 0)
    if ones.size == 0 or zeros.size == 0:
        return
    i = rng.choice(ones)
    j = rng.choice(zeros)
    ind[i], ind[j] = 0.0, 1.0

def local_search(h, rng, time_left):
    best = h.copy()
    best_s = compute_c5_fast(best)
    while time_left() > 0.1:
        ones = np.flatnonzero(best == 1)
        zeros = np.flatnonzero(best == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i], cand[j] = 0.0, 1.0
        s = compute_c5_fast(cand)
        if s < best_s:
            best, best_s = cand, s
    return best, best_s

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():

    n = h.shape[0]
    h_rev = h[::-1]

    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)
