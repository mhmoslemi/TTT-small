        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure lam_low gives f > s and lam_high gives f < s
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr(h):
    """Correlation vector corr[k] = sum_i h[i] * (1 - h[i+k]) for k = -(n-1)...(n-1)"""
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    return conv[:2 * n - 1]

def compute_c5_fast(h):
    n = h.shape[0]
    corr = compute_corr(h)
    return corr.max() * (2.0 / n)

def compute_gradient_softmax(h, alpha):
    """Gradient of (1/alpha) log-sum-exp(alpha * corr * dx) w.r.t. h."""
    n = h.shape[0]
    L = 2 * n - 1
    dx = 2.0 / n

    # correlation (without dx)
    corr = compute_corr(h)

    # softmax weights over shifts
    max_term = np.max(alpha * corr)
    exp_vals = np.exp(alpha * corr - max_term)
    w = exp_vals / np.sum(exp_vals)                # shape (L,)

    # reverse weights for convolution
    w_rev = w[::-1]                                # shape (L,)

    # convolution size
    Nfft = 1 << ((L + n - 1).bit_length())

    # FFT of w_rev
    fft_w = np.fft.rfft(w_rev, n=Nfft)

    # conv1 = w_rev * (1 - h)
    fft_oneh = np.fft.rfft(1.0 - h, n=Nfft)
    conv1_full = np.fft.irfft(fft_w * fft_oneh, n=Nfft)

    # conv2 = w_rev * h
    fft_h = np.fft.rfft(h, n=Nfft)
    conv2_full = np.fft.irfft(fft_w * fft_h, n=Nfft)

    offset = L - 1
    conv1 = conv1_full[offset:offset + n]
    conv2 = conv2_full[offset:offset + n]

    grad = (conv1 - conv2) * dx
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    n = 256
    target_sum = n / 2.0
    # initial vector
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            x_old = np.linspace(0, 1, init.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        h = project_onto_box_sum(init, target_sum)
    else:
        h = project_onto_box_sum(rng.random(n), target_sum)

    best_h = h.copy()
    best_c5 = compute_c5_fast(best_h)

    # hyper‑parameters
    it = 0
    alpha = 5.0
    max_alpha = 200.0
    alpha_inc_every = 200
    init_lr = 0.5

    while time_left() > 0.5 and it < 200000:
        lr = init_lr / np.sqrt(it + 1)
        grad = compute_gradient_softmax(h, alpha)
        h_new = project_onto_box_sum(h - lr * grad, target_sum)
        c5 = compute_c5_fast(h_new)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            if best_c5 <= 0.3808:
                break

        h = h_new
        it += 1

        if it % alpha_inc_every == 0:
            alpha = min(max_alpha, alpha * 1.05)

        if it % 500 == 0:
            # small random jitter to help escape plateaus
            eps = 0.02
            h = project_onto_box_sum(h + eps * (rng.random(n) - 0.5), target_sum)

    # try binary rounding of the best real solution
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    c5_bin = compute_c5_fast(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, target_sum)
    best_c5 = compute_c5_fast(best_h)

    return best_h, float(best_c5), n
