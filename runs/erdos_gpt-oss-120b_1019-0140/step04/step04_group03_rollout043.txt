def compute_gradient(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty_like(h)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty_like(h)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            # Should not occur, shifts are non-negative
            continue
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

def adam_optimize(h0, max_iters, lr0, beta1, beta2, eps, start_time, time_limit):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_from_h(h)
    for t in range(1, max_iters+1):
        if time.time() - start_time >= time_limit:
            break
        # compute correlation via FFT
        h_fft = np.fft.rfft(h, n=2*n)
        corr_full = np.fft.irfft(h_fft * np.conj(h_fft), n=2*n)
        corr_hh = corr_full[:n]
        min_corr = corr_hh.min()
        shifts = np.where(corr_hh <= min_corr + 1e-12)[0]
        grad = compute_gradient(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr0 * m_hat / (np.sqrt(v_hat) + eps)
        # Project onto box sum
        h_new = project_onto_box_sum(h_new, target_sum)
        # Evaluate score
        score = 1 - min_corr * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if new_n == m:
        return arr.copy()
    if new_n % m == 0:
        factor = new_n // m
        return np.repeat(arr, factor)
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
