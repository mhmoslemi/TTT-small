Now compute_c5_fft:

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2*n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2*n-1]
    return np.max(corr) * (2.0 / n)

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        # shift indices where corr is near max (tolerance small)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left, max_iters=400000, T0=0.2, decay=0.99995, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    # Ensure binary: values 0/1, sum equals n/2.
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    # define ensure sums
    # This h may not be exactly binary? We'll convert before calling.
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    for it in range(max_iters):
        if time_left() <= 0.01:
            break
        # random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            # update ones/zeros sets
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
    return best_h, best_score

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift

            term1[s:] = h[:n - s]

            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def binary_swap_anneal(h0, rng, time_left, max_iters=300000, T0=0.3, decay=0.99997, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    # ensure binary
    is_one = h > 0.5
    ones = np.flatnonzero(is_one).tolist()
    zeros = np.flatnonzero(~is_one).tolist()
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.01:
        # choose random indices in list
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            # swap entries in lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure brackets contain solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx = 2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_c5_fft(h):
    """Maximum overlap integral C5."""
    corr = compute_corr_scaled_fft(h)
    return np.max(corr)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max shift(s) correlation w.r.t. h."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    """Projected gradient descent with momentum on the C5 objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left, max_iters=400000, T0=0.3, decay=0.99997, Tmin=1e-6):
    """Simulated annealing on binary (0/1) vectors by swapping one 1 with one 0."""
    h = h0.copy()
    n = h.size
    is_one = h > 0.5
    ones = np.flatnonzero(is_one).tolist()
    zeros = np.flatnonzero(~is_one).tolist()
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.01:
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            # update the index lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def thue_morse(n):
    """Binary Thue–Morse sequence (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def rudin_shapiro(n):
    """Rudin–Shapiro binary sequence (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        x = i
        cnt = 0
        while x:
            if (x & 3) == 3:  # last two bits are '11'
                cnt += 1
            x >>= 1
        seq[i] = 0.0 if (cnt & 1) else 1.0
    return seq

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    rng = np.random.default_rng(seed)
    target = 0.3808
    best_h = None
    best_score = np.inf

    N = 1024  # target resolution

    # ---------- Phase 0: generate candidate pool ----------
    candidates = []

    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.size != N:
            init_arr = upsample(init_arr, N)
        init_arr = project_onto_box_sum(init_arr, N / 2.0)
        candidates.append(init_arr)

    # deterministic structured candidates
    candidates.append(project_onto_box_sum(thue_morse(N), N / 2.0))
    candidates.append(project_onto_box_sum(rudin_shapiro(N), N / 2.0))

    # sinusoidal patterns
    x = np.arange(N) * (2.0 / N)
    for freq in (0.5, 1.0, 2.0, 3.0):
        h = 0.5 + 0.5 * np.sin(np.pi * freq * x)
        candidates.append(project_onto_box_sum(h, N / 2.0))

    # random seeds
    for _ in range(2):
        candidates.append(random_binary_balanced(N, rng))
    for _ in range(2):
        v = rng.random(N)
        candidates.append(project_onto_box_sum(v, N / 2.0))

    # ---------- Phase 1: continuous refinement of each candidate ----------
    for cand in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = gradient_descent_momentum(cand, time_left, step0=0.5, beta=0.9, max_iter=150000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 2: additional random continuous restarts ----------
    while time_left() > 20.0:
        v = rng.random(N)
        v = project_onto_box_sum(v, N / 2.0)
        h_opt, sc_opt = gradient_descent_momentum(v, time_left, step0=0.5, beta=0.9, max_iter=50000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 3: binary conversion ----------
    if best_h is not None:
        n = best_h.size
        k = n // 2
        h_bin = np.zeros_like(best_h)
        idx_top = np.argpartition(-best_h, k)[:k]
        h_bin[idx_top] = 1.0
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 4: binary annealing ----------
    if best_h is not None and time_left() > 5.0:
        # ensure binary start
        if not np.all(np.isin(best_h, [0.0, 1.0])):
            n = best_h.size
            k = n // 2
            tmp = np.zeros_like(best_h)
            idx_top = np.argpartition(-best_h, k)[:k]
            tmp[idx_top] = 1.0
            best_h = tmp
        h_bin_opt, sc_bin_opt = binary_swap_anneal(best_h, rng, time_left,
                                                   max_iters=400000,
                                                   T0=0.3, decay=0.99997, Tmin=1e-6)
        if sc_bin_opt < best_score:
            best_score, best_h = sc_bin_opt, h_bin_opt.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 5: random binary annealing restarts ----------
    while time_left() > 10.0:
        h_rand = np.zeros(N, dtype=float)
        idx = rng.choice(N, size=N // 2, replace=False)
        h_rand[idx] = 1.0
        h_ann, sc_ann = binary_swap_anneal(h_rand, rng, time_left,
                                           max_iters=200000,
                                           T0=0.3, decay=0.99997, Tmin=1e-6)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()
            if best_score <= target:
                break

    # ---------- Final projection and return ----------
    if best_h is None:
        # fallback: uniform half-half
        best_h = np.zeros(N, dtype=float)
        best_h[:N // 2] = 1.0
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
