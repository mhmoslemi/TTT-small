def adam_projected(h0, step0, beta1, beta2, epsilon, max_iter, rng, max_time, target_sum):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    for t in range(1, max_iter+1):
        if time.time() - start >= max_time:
            break
        # compute gradient for max overlap shift(s)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # get all shifts achieving max_corr (within tolerance)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # gradient
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                sz = n - shift
                term1 = np.empty(n, dtype=h.dtype)
                term1[:sz] = h[shift:]
                term1[sz:] = 0.0
                term2 = np.empty(n, dtype=h.dtype)
                term2[shift:] = h[:sz]
                term2[:shift] = 0.0
            else:
                s = -shift
                sz = n - s
                term1 = np.empty(n, dtype=h.dtype)
                term1[s:] = h[:sz]
                term1[:s] = 0.0
                term2 = np.empty(n, dtype=h.dtype)
                term2[:sz] = h[s:]
                term2[sz:] = 0.0
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 / (np.sqrt(v_hat) + epsilon)
        h_new = h - step * m_hat
        # projection
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

import time, numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lower = np.min(v) - hi
    upper = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lower + upper) / 2.0
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lower = lam
        else:
            upper = lam
    return np.clip(v - (lower + upper) / 2.0, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def gradient_subgrad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    idxes = np.where(corr >= max_corr - 1e-12)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def adam_projected(h0, step0, beta1, beta2, epsilon, max_iter, rng, max_time, target_sum):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    for t in range(1, max_iter + 1):
        if time.time() - start >= max_time:
            break
        grad = gradient_subgrad(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 / (np.sqrt(v_hat) + epsilon)
        h_new = h - step * m_hat
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
