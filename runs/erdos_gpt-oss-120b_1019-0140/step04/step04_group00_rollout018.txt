def compute_gradient_max_shift(h):
    n = h.size
    # FFT size N
    N = 1 << ((2*n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n - 1]
    mx = corr.max()
    # Identify shifts (zero-centered)
    idxes = np.flatnonzero(corr == mx)
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad
```

def gradient_of_shift(h, shift):
    n = h.size
    if shift >= 0:
        # term1: h_{i+shift} for i where i+shift < n
        term1 = np.zeros_like(h)
        term1[:n-shift] = h[shift:]
        # term2: h_{i-shift} for i where i-shift >= 0
        term2 = np.zeros_like(h)
        term2[shift:] = h[:n-shift]

        term1 = np.zeros_like(h)
        term1[s:] = h[:n-s]
        term2 = np.zeros_like(h)
        term2[:n-s] = h[s:]
    return 1.0 - term1 - term2

def compute_c5_fast(h):
    n = h.size
    N = 1 << ((2*n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n - 1]
    return corr.max() * (2.0 / n)
```

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_swap_anneal(h0, rng, time_left, max_iters=200000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = 0.2
    T_min = 1e-5
    decay = 0.99995  # adjust
    it = 0
    # Pre-compute index sets for ones and zeros. We'll update each iteration.
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0  # revert
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s
```

import time
import numpy as np

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target) < 1e-12:
            break
        if fm > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.size
    if shift >= 0:
        term1 = np.zeros_like(h)
        term1[:n - shift] = h[shift:]
        term2 = np.zeros_like(h)
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1 = np.zeros_like(h)
        term1[s:] = h[:n - s]
        term2 = np.zeros_like(h)
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    h_rev = h[::-1]
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.flatnonzero(corr == mx)
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_s = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def thue_morse_binary(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = bin(i).count('1') & 1
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_swap_anneal(h0, rng, time_left, max_iters=200000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99995
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s
