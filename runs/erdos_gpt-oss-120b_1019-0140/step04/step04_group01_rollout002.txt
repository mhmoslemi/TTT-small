def binary_anneal_optimized(h0, rng, time_left, max_total_iters=2000000, T0=1.0, decay=0.99995, Tmin=1e-7):
    h = h0.astype(np.float64).copy()
    n = h.shape[0]
    current_score = compute_c5_fft(h)
    best_score = current_score
    best_h = h.copy()
    T = T0
    it = 0
    steps_since_imp = 0
    while time_left() > 0.0 and it < max_total_iters:
        # pick random 1 and 0 indices
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            # acceptance probability
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            current_score = new_score
            steps_since_imp = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            steps_since_imp += 1
        # update temperature
        T = max(Tmin, T * decay)
        # reheat if stagnated
        if steps_since_imp >= 20000:
            T = T0
            steps_since_imp = 0
        it += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx = 2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Return binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func,
                                        step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_anneal_optimized(h0, rng, time_left, max_total_iters=500000,
                            T0=1.0, decay=0.99995, Tmin=1e-7,
                            stagnation_iters=20000):
    h = h0.astype(float).copy()
    n = h.shape[0]
    current_score = compute_c5_fft(h)
    best_score = current_score
    best_h = h.copy()
    T = T0
    it = 0
    since_improve = 0
    while time_left() > 0.0 and it < max_total_iters:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            current_score = new_score
            since_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            since_improve += 1
        T = max(Tmin, T * decay)
        if since_improve >= stagnation_iters:
            T = T0
            since_improve = 0
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    best_h = None
    best_score = np.inf

    # ----- Base binary initialization (n0 = 512) -----
    n0 = 512
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.size != n0:
            init_arr = upsample(init_arr, n0)
        # Binarize by top half entries
        idx = np.argpartition(-init_arr, n0 // 2)[:n0 // 2]
        base_h = np.zeros(n0, dtype=float)
        base_h[idx] = 1.0
    else:
        base_h = random_binary_balanced(n0, rng)

    # ----- Phase 1: binary annealing at n0 -----
    if time_left() > 5.0:
        h_opt, sc_opt = binary_anneal_optimized(
            base_h, rng, time_left,
            max_total_iters=400000, T0=1.0, decay=0.9999, Tmin=1e-7)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 2: continuous gradient descent at n0 -----
    if time_left() > 5.0:
        h_gd, sc_gd = gradient_descent_momentum_with_time(
            best_h if best_h is not None else base_h,
            time_left, step0=0.5, beta=0.9, max_iter=200000)
        if sc_gd < best_score:
            best_score, best_h = sc_gd, h_gd.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Upsample to higher resolution (n1 = 1024) -----
    n1 = 1024
    if best_h is None:
        best_h = base_h.copy()
        best_score = compute_c5_fft(best_h)
    h_up = upsample(best_h, n1)
    # Binarize the upsampled vector
    idx_up = np.argpartition(-h_up, n1 // 2)[:n1 // 2]
    h_up_bin = np.zeros(n1, dtype=float)
    h_up_bin[idx_up] = 1.0

    # ----- Phase 3: binary annealing at n1 -----
    if time_left() > 5.0:
        h_up_opt, sc_up_opt = binary_anneal_optimized(
            h_up_bin, rng, time_left,
            max_total_iters=600000, T0=1.0, decay=0.9999, Tmin=1e-7)
        if sc_up_opt < best_score:
            best_score, best_h = sc_up_opt, h_up_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 4: continuous gradient descent at n1 -----
    if time_left() > 5.0:
        h_up_gd, sc_up_gd = gradient_descent_momentum_with_time(
            best_h if best_h.size == n1 else h_up_opt,
            time_left, step0=0.5, beta=0.9, max_iter=200000)
        if sc_up_gd < best_score:
            best_score, best_h = sc_up_gd, h_up_gd.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 5: random binary restarts at n1 -----
    while time_left() > 10.0:
        v = random_binary_balanced(n1, rng)
        h_tmp, sc_tmp = binary_anneal_optimized(
            v, rng, time_left,
            max_total_iters=200000, T0=1.0, decay=0.9999, Tmin=1e-7)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
            if best_score <= target:
                break

    # Final projection to satisfy exact sum (should already hold)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
