import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, h, mode='full')
    min_corr = np.min(corr)
    return 1.0 - (2.0 / n) * min_corr

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    seq = np.arange(n, dtype=np.int64)
    bits = ((seq[:, None] & (1 << np.arange(seq.max().bit_length()))) > 0).sum(axis=1)
    return (bits & 1).astype(float)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

        grad = compute_gradients_for_shifts(h, shifts)

    best_score = compute_c5(best_h)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_final = 512
    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(n_final)
    if base.shape[0] != n_final:
        base = upsample(base, n_final)
    base = project_onto_box_sum(base, n_final / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # random binary sampling
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 1.0:
            break
        h_rand = random_binary_balanced(n_final, rng)
        sc = compute_c5(h_rand)
        if sc < best_score:
            best_score, best_h = sc, h_rand.copy()

    # gradient descent restarts
    for r in range(5):
        elapsed = time.time() - start
        if elapsed >= budget_s - 5.0:
            break
        h0 = best_h.copy()
        if r > 0:
            h0 += rng.normal(scale=0.03, size=n_final)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = project_onto_box_sum(h0, n_final / 2.0)
        remaining = budget_s - (time.time() - start)
        h_opt, sc_opt = gradient_descent_momentum(h0, remaining, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # simulated annealing
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        best_h, best_score = simulated_annealing(best_h, remaining, rng)

    # final binary rounding
    n = best_h.size
    binary_h = np.zeros_like(best_h)
    binary_h[np.argsort(best_h)[-n // 2:]] = 1.0
    binary_score = compute_c5(binary_h)
    if binary_score < best_score:
        best_h, best_score = binary_h, binary_score

    return best_h, best_score, n
