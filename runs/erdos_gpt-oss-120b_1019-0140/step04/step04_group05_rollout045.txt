def upsample(arr, new_n):
    # arr shape (m,)
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def local_anneal(h0, time_limit, rng):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    # use exponential cooling: temp0 = 0.01, tau = time_limit / log(temp0/min_temp)
    temp0 = 0.02
    min_temp = 1e-5
    # compute tau such that at time_limit, temp = min_temp: temp = temp0 * exp(-t/tau)
    if time_limit > 0:
        tau = time_limit / max(1e-12, np.log(temp0 / min_temp))
    else:
        tau = 1.0

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        temp = max(min_temp, temp0 * np.exp(-elapsed / tau))
        # pick i != j random
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5  # limit step size to 0.5 of max_delta
        if delta <= 0:
            continue
        # apply move
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temp):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best_score

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def de_optimize(pop, n, time_left_func, rng):
    """Simple Differential Evolution with projection onto constraints."""
    scores = [compute_c5(v) for v in pop]
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    F = 0.5
    CR = 0.7
    pop_size = len(pop)

    while time_left_func() > 1.0:
        for t in range(pop_size):
            if time_left_func() <= 1.0:
                break
            idxs = list(range(pop_size))
            idxs.remove(t)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, n / 2.0)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, pop[t])
            trial = project_onto_box_sum(trial, n / 2.0)

            trial_score = compute_c5(trial)
            if trial_score < scores[t]:
                pop[t] = trial
                scores[t] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
    return best_h, best_score


def local_anneal(h0, time_limit, rng):
    """Simulated annealing for fine-tuning."""
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    temp0 = 0.02
    min_temp = 1e-5
    if time_limit > 0:
        tau = time_limit / max(1e-12, np.log(temp0 / min_temp))
    else:
        tau = 1.0

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        temp = max(min_temp, temp0 * np.exp(-elapsed / tau))

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.5
        if delta <= 0.0:
            continue

        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temp):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # Base vector
    if 'initial_h_values' in globals():
        base0 = np.asarray(initial_h_values, dtype=float)
    else:
        base0 = thue_morse(64)
    base0 = project_onto_box_sum(base0, base0.size / 2.0)

    n_target = 512
    if base0.size != n_target:
        base = upsample(base0, n_target)
        base = project_onto_box_sum(base, n_target / 2.0)
    else:
        base = base0

    # Initialize population
    pop_size = 8
    pop = [base]
    for _ in range(pop_size - 1):
        rand_vec = rng.random(n_target)
        rand_vec = project_onto_box_sum(rand_vec, n_target / 2.0)
        pop.append(rand_vec)

    # Global differential evolution
    best_h, best_score = de_optimize(pop, n_target, time_left, rng)

    # Local refinement with remaining budget
    remaining = time_left()
    if remaining > 0.6:
        best_h, best_score = local_anneal(best_h, remaining, rng)

    # Final projection and scoring
    best_h = project_onto_box_sum(best_h, n_target / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, n_target
