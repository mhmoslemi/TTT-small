def adam_optimize(h0, n_iter, lr, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0: break
        # compute gradient
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c, tol=1e-9)
        # Adam update
        m = beta1*m + (1 - beta1)*grad
        v = beta2*v + (1 - beta2)*(grad*grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        # add small noise occasionally (e.g., every 5000 steps)
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=h_new.shape)
        # project
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2*n - 1
    # Using real FFT? Use np.fft.rfft maybe more efficient. But keep as prior.
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-9):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            # for i from 0 to n-k-1: grad[i] += (1 - h[i+k])
            grad[:n - k] += 1.0 - h[k:]
            # for i from k to n-1: grad[i] -= h[i-k]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

import time
import numpy as np
import scipy.optimize as opt

# ---------- utilities ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full circular correlation of h with (1‑h) using power‑of‑two FFT."""
    n = h.shape[0]
    L = 2 * n                # power of two for speed
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr[: 2 * n - 1]


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-9):
    """Sub‑gradient of max‑correlation (average over all max shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[: n - k] += 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[: n - m]
            grad[: n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def adam_optimize(h0, n_iter, lr, beta1, beta2, eps, rng, time_left):
    """Adam optimizer with Euclidean projection onto box+sum."""
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    b1t = beta1
    b2t = beta2
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break

        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - b1t)
        v_hat = v / (1.0 - b2t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step

        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=h.shape)

        h_new = project_onto_box_sum(h_new, s)

        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()

        h = h_new
        b1t *= beta1
        b2t *= beta2

    return best_h, best_score
