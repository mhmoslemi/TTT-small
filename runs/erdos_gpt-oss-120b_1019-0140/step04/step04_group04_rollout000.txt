def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.shape[0]
    # clamp to box if sum out of feasible
    if s <= lo*n:
        return np.full_like(v, lo)
    if s >= hi*n:
        return np.full_like(v, hi)
    # binary search on lambda
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    # function f(lam) = sum clamp(v - lam, lo, hi)
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure bounds
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10*hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10*lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

Now upsample:

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_balanced(n, rng):
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    arr[idx] = 1.0
    return arr

def gd_momentum(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts achieving max (within eps)
        shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
        # gradient sum
        grad = np.zeros_like(h)
        # average over shifts
        inv = 1.0 / max(len(shifts), 1)
        for s in shifts:
            grad += (1 - np.roll(h, s) - np.roll(h, -s))
        grad *= inv
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random Gaussian noise
        if it % 5000 == 0:
            scale = 0.02 * (0.9 ** (it // 5000))
            h_new += rng.normal(scale=scale, size=n)
        # projection
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # early exit if below target

def slsqp_refine(h0, rng, time_left):
    n = h0.shape[0]
    target_sum = n / 2.0
    # variable includes h vector, we treat t as dummy not needed? We can minimize C5 directly using constraints.
    # We'll define t as objective variable to maximize correlation constraints.
    # Use same formulation as before.
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - target_sum
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        # constraint: t >= corr for all shifts => t - corr >= 0 -> min(t - corr) >= 0 => t - max(corr) >= 0?
        # Actually we need t >= all corr values; enforce t - corr_i >= 0 for each i. SciPy only supports scalar constraint per function. 
        # We'll enforce t >= max(corr) via t - max(corr) >= 0.
        return t - np.max(corr)
    cons = [{'type':'eq', 'fun':eq_sum},
            {'type':'ineq','fun':ineq_fun}]
    # limit time: set maxiter based on remaining time
    max_iter = int(min(500, max(50, time_left() * 5)))
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, target_sum)
    score = compute_c5(h_opt)
    return h_opt, score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.shape[0]
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_balanced(n, rng):
    a = np.zeros(n, dtype=float)
    a[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return a

def gd_momentum(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
        grad = np.zeros_like(h)
        for s in shifts:
            grad += 1 - np.roll(h, s) - np.roll(h, -s)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 5000 == 0:
            scale = 0.02 * (0.9 ** (it // 5000))
            h_new += rng.normal(scale=scale, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_refine(h0, rng, time_left):
    n = h0.shape[0]
    target_sum = n / 2.0
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - target_sum
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    max_iter = int(min(500, max(50, time_left() * 5)))
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, target_sum)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.fromiter(((bin(i).count('1') & 1) for i in range(64)), dtype=float, count=64)

    if base.shape[0] != 64:
        base = upsample(base, 64)
    base = project_onto_box_sum(base, 32.0)
    best_h, best_score = base.copy(), compute_c5(base)

    # Stage 1: 256 points
    if time_left() > 5:
        n1 = 256
        h1 = upsample(best_h, n1)
        h1 = project_onto_box_sum(h1, n1 / 2.0)
        max_iters = int(min(100000, time_left() * 200))
        h1, sc1 = gd_momentum(h1, max_iters, 0.8, 0.9, rng, time_left)
        if sc1 < best_score:
            best_h, best_score = h1, sc1
        for _ in range(3):
            if time_left() <= 5:
                break
            h_rand = random_balanced(n1, rng)
            max_iters = int(min(60000, time_left() * 200))
            h_rand, sc_rand = gd_momentum(h_rand, max_iters, 0.8, 0.9, rng, time_left)
            if sc_rand < best_score:
                best_h, best_score = h_rand, sc_rand

    # Stage 2: 512 points
    if time_left() > 5:
        n2 = 512
        h2 = upsample(best_h, n2)
        h2 = project_onto_box_sum(h2, n2 / 2.0)
        max_iters = int(min(250000, time_left() * 200))
        h2, sc2 = gd_momentum(h2, max_iters, 0.4, 0.95, rng, time_left)
        if sc2 < best_score:
            best_h, best_score = h2, sc2
        if time_left() > 10:
            h_ref, sc_ref = slsqp_refine(best_h, rng, time_left)
            if sc_ref < best_score:
                best_h, best_score = h_ref, sc_ref

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
