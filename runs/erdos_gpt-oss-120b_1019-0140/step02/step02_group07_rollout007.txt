def continuous_anneal(start_h, time_end, rng):
    h = start_h.copy()
    n = len(h)
    target_sum = n/2.0
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # parameters
    init_temp = 0.01
    final_temp = 1e-6
    total_iter_est = 200000

    it = 0
    while time.time() < time_end:
        # Choose two indices i, j such that h[i] > 0 and h[j] < 1
        # We'll compute eligible idx each time? That may be heavy. Use random sampling and rejection.
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Ensure we move delta from i to j: we can increase j, decrease i.
        if h[i] <= 1e-12 or h[j] >= 1 - 1e-12:
            continue
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            continue
        # propose delta
        # scale with temperature via rng.random()
        # temperature linear decreasing with iteration over max_iter_est
        T = init_temp * (1 - it / total_iter_est) + final_temp * (it / total_iter_est)
        # random delta fraction
        delta = max_delta * rng.random() * 0.5  # max 50% of max_delta
        # apply
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-12)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
        # optional restart in case of stagnation: after certain iterations without improvement, re-randomize small fraction
        if it % 50000 == 0:
            # small perturbation: random swap of bits
            # add random noise
            idx = rng.choice(n, size=5, replace=False)
            h[idx] = rng.random(5)  # random values in [0,1)
            # re-normalize sum
            diff = target_sum - np.sum(h)
            h += diff / n
            h = np.clip(h,0.0,1.0)
            # recompute cur_c5
            cur_c5 = compute_c5(h)
    # after loop, final fix sum if drift
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:

    n = arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * (2.0 / n))


def milp_binary_optimal(n: int):
    """Solve the exact binary min‑max problem for given n using MILP.
    Returns (h, c5) where h is a binary np.ndarray of length n, or (None, None)
    if the solver is unavailable or fails."""
    if cp is None:
        return None, None
    # variable for the step function (binary)
    x = cp.Variable(n, boolean=True)
    # auxiliary variable for the max correlation value
    t = cp.Variable()
    constraints = [cp.sum(x) == n / 2.0]

    # build product variables for each shift
    for k in range(1, n):
        length = n - k
        y = cp.Variable(length, boolean=True)  # y_i = x_i * x_{i+k}
        constraints += [
            y <= x[:length],
            y <= x[k:],
            y >= x[:length] + x[k:] - 1,
        ]
        # O_k = sum_{i< n-k} x_i - sum y_i   (positive shift)
        constraints.append(cp.sum(x[:length]) - cp.sum(y) <= t)
        # O_-k = sum_{i>=k} x_i - sum y_i   (negative shift)
        constraints.append(cp.sum(x[k:]) - cp.sum(y) <= t)
    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK_MI, msg=False)  # GLPK_MI works for modest sizes
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.CBC, msg=False)
        except Exception:
            return None, None

    if prob.status not in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}:
        return None, None

    x_val = np.rint(np.array(x.value)).astype(float)  # force exact 0/1
    # sanity fix for sum (rare rounding issues)
    target_sum = n / 2.0
    cur_sum = np.sum(x_val)
    diff = int(round(target_sum - cur_sum))
    if diff != 0:
        if diff > 0:
            zeros = np.where(x_val == 0)[0]
            if zeros.size >= diff:
                x_val[zeros[:diff]] = 1.0
        else:
            ones = np.where(x_val == 1)[0]
            if ones.size >= -diff:
                x_val[ones[: -diff]] = 0.0
    c5 = compute_c5(x_val)
    return x_val, c5


def continuous_anneal(start_h: np.ndarray, time_end: float, rng: np.random.Generator):
    """Simulated annealing on continuous values preserving the total sum."""
    h = start_h.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # annealing schedule parameters
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter_est = 300000  # rough estimate for scaling temperature
    it = 0

    while time.time() < time_end:
        # pick two distinct indices, i (decrease) and j (increase)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
            # try again quickly
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue

        # temperature based on iteration count (linear decay)
        T = init_temp * (1.0 - it / max_iter_est) + final_temp * (it / max_iter_est)

        # propose a delta ( up to half of the allowed range )
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        if dE <= 0.0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1

        # occasional large perturbation to avoid stagnation
        if it % 50000 == 0:
            # add small random noise to a few coordinates
            idx = rng.choice(n, size=5, replace=False)
            h[idx] = rng.random(5)
            # re‑normalize sum
            diff = target_sum - np.sum(h)
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            cur_c5 = compute_c5(h)

    # final exactness fix
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1.0 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        init_h = None

    target_sum = n / 2.0

    best_h = None
    best_c5 = float("inf")

    # 1. Try exact binary optimisation for modest n (≤100)
    if n <= 100:
        h_bin, c5_bin = milp_binary_optimal(n)
        if h_bin is not None and c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin

    # 2. Consider the provided initial construction (if any)
    if init_h is not None:
        c0 = compute_c5(init_h)
        if c0 < best_c5:
            best_h = init_h.copy()
            best_c5 = c0

    # 3. If we still have time, perform continuous annealing starting from best found.
    time_left = budget_s - (time.time() - start_time)
    if time_left > 6.0:
        # allocate remaining time (minus a small safety margin)
        anneal_deadline = time.time() + time_left - 0.5

        if best_h is None:
            # generate a random feasible binary start
            start_h = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            start_h[idx] = 1.0
        else:
            start_h = best_h.copy()

        # enforce exact feasibility (tiny numeric errors)
        start_h = np.clip(start_h, 0.0, 1.0)
        cur_sum = np.sum(start_h)
        if abs(cur_sum - target_sum) > 1e-12:
            start_h += (target_sum - cur_sum) / n
            start_h = np.clip(start_h, 0.0, 1.0)
            residual = target_sum - np.sum(start_h)
            for i in range(n):
                if abs(residual) < 1e-12:
                    break
                adj = np.clip(residual, -start_h[i], 1.0 - start_h[i])
                start_h[i] += adj
                residual = target_sum - np.sum(start_h)

        cand_h, cand_c5 = continuous_anneal(start_h, anneal_deadline, rng)
        if cand_c5 < best_c5:
            best_h = cand_h
            best_c5 = cand_c5

    # final safety clamp and exact sum enforcement
    if best_h is None:  # extremely unlikely fallback
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1.0 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
