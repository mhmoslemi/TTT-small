def random_feasible_vector(n, target_sum, rng):
    v = np.zeros(n)
    remaining = target_sum
    indices = rng.permutation(n)
    for i in indices:
        if remaining >= 1.0:
            v[i] = 1.0
            remaining -= 1.0
        else:
            v[i] = remaining
            remaining = 0.0
            break
    # the rest remain 0
    return v

def random_feasible_vector(n, target_sum, rng):
    v = np.zeros(n, dtype=float)

        if remaining >= 1.0:
            v[i] = 1.0
            remaining -= 1.0
        else:
            v[i] = remaining
            remaining = 0.0
            break
    return v

def continuous_annealing(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n/2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    init_temp = 5e-3
    final_temp = 1e-7
    max_iters = 500_000

    it = 0

    while time.time() < time_end and it < max_iters:
        # choose two indices to adjust
        # We'll pick any two distinct indices (i,j)
        i, j = rng.choice(n, size=2, replace=False)
        # compute max delta for increase at i (subject to <=1) and decrease at j (>=0)
        # For keep sum constant, we will increase one and decrease another.
        # To avoid zero delta, ensure i can be increased, j can be decreased.
        # Determine allowed increase and decrease.
        # We can also randomly decide direction: either increase i, decrease j or opposite.
        # Simpler: ensure i has spare capacity and j has positive value.
        if h[i] >= 1 - 1e-12 or h[j] <= 1e-12:
            # try swap indices
            continue
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        max_delta = min(max_inc, max_dec)
        if max_delta <= 1e-12:
            it += 1
            continue
        # propose delta scaled by temperature? Could just random fraction * max_delta
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        # temperature schedule

                best_c5 = new_c5

    # final correction enforce sum exact.
    diff = target_sum - np.sum(best_h)

def continuous_annealing(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    init_temp = 5e-3
    final_temp = 1e-7

    while time.time() < time_end:
        # select i,j
        i, j = rng.choice(n, size=2, replace=False)
        # ensure i can increase, j can decrease. We'll decide direction to ensure move.
        # We'll decide to increase i if h[i] < 1 and h[j] > 0; else swap roles.
        if h[i] >= 1 - 1e-12 or h[j] <= 1e-12:
            # try opposite direction
            if h[j] < 1 - 1e-12 and h[i] > 1e-12:
                i, j = j, i
            else:
                continue
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        # temperature based on elapsed fraction:
        frac = (time.time() - start) / (time_end - start)
        T = init_temp * (1 - frac) + final_temp * frac
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    # final sum adjust
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_c5

import time
import math
import numpy as np

def compute_c5(arr):
    """Approximate C5 = max_k ∫ h(x)(1−h(x+k))dx via discrete correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def random_feasible_vector(n, target_sum, rng):
    """Create a feasible vector with entries in [0,1] and exact sum=target_sum."""
    v = np.zeros(n, dtype=float)
    remaining = target_sum
    order = rng.permutation(n)
    for i in order:
        if remaining >= 1.0:
            v[i] = 1.0
            remaining -= 1.0
        else:
            v[i] = remaining
            remaining = 0.0
            break
    return v

def binary_simulated_annealing(start_bin, time_end, rng):
    """SA on binary vectors (swap moves) with temperature schedule."""
    n = len(start_bin)
    h = start_bin.astype(float)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    ones = list(np.flatnonzero(h >= 0.5))
    zeros = list(np.flatnonzero(h < 0.5))

    it = 0
    max_iters = 10_000_000
    T0, Tf = 5e-2, 1e-6

    while time.time() < time_end and it < max_iters:
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        cand = h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5

        if delta <= 0.0 or rng.random() < math.exp(-delta / (T + 1e-18)):

            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

    # enforce exact sum
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1.0 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_c5

def binary_hill_climb(start_bin, time_end, rng):
    """Deterministic hill‑climbing on binary vectors (swap moves)."""
    n = len(start_bin)
    h = start_bin.astype(float)
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    ones = list(np.flatnonzero(h >= 0.5))
    zeros = list(np.flatnonzero(h < 0.5))
    no_improve = 0
    while time.time() < time_end and no_improve < 2000:
        if not ones or not zeros:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        cand = h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
            i_idx = ones.index(i)
            j_idx = zeros.index(j)
            ones[i_idx] = j
            zeros[j_idx] = i
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # initialise h0
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = random_feasible_vector(n, n / 2.0, rng)

    target_sum = n / 2.0
    # enforce feasibility exactly
    cur_sum = np.sum(h0)
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        h0 += diff / n
        h0 = np.clip(h0, 0.0, 1.0)
        residual = target_sum - np.sum(h0)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -h0[i], 1.0 - h0[i])
            h0[i] += adj
            residual = target_sum - np.sum(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    target_threshold = 0.3808
    time_limit = start_time + budget_s

    # main alternating search loop
    while True:
        now = time.time()
        remaining = time_limit - now
        if remaining < 0.5:
            break
        # allocate a short chunk (up to 5 s or 30 % of remaining)
        chunk = min(5.0, remaining * 0.3)
        end_chunk = now + chunk

        if rng.random() < 0.5:
            # binary simulated annealing
            start_bin = (best_h >= 0.5).astype(int)
            cand_h, cand_c5 = binary_simulated_annealing(start_bin, end_chunk, rng)
        else:
            # continuous annealing
            cand_h, cand_c5 = continuous_annealing(best_h, end_chunk, rng)

        if cand_c5 < best_c5:
            best_h = cand_h.copy()
            best_c5 = cand_c5
            if best_c5 <= target_threshold:
                break

    # final deterministic refinement (binary hill climb)
    if time.time() < time_limit - 0.2:
        bin_start = (best_h >= 0.5).astype(int)
        final_h, final_c5 = binary_hill_climb(bin_start, time_limit - 0.1, rng)
        if final_c5 < best_c5:
            best_h = final_h.astype(float)
            best_c5 = final_c5

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1.0 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
