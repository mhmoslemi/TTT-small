def gradient_step(h, target_sum, step):
    n = h.shape[0]
    # compute worst shift
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Overlap values already already dx factor later; for gradient we ignore dx factor because step can absorb.
    idx_max = np.argmax(corr)
    k = idx_max - (n - 1)
    # compute neighbor sum
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]

    grad = (1.0 - neigh)  # ignoring dx factor; incorporate into step
    cand = h - step * grad
    cand = project_feasible(cand, target_sum)
    return cand

def slsqp_refine(h, target_sum, budget_remaining):

        return np.sum(x[:n]) - target_sum

import time
import math
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral C5 for discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum, eps=1e-12, max_iter=1000):
    """
    Clip to [0,1] and adjust to satisfy sum(h) == target_sum.
    Simple water‑filling on free variables.
    """
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    it = 0
    while abs(diff) > eps and it < max_iter:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # no interior variables – distribute uniformly (should be rare)
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
            it += 1
            continue
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        it += 1
    return h

def gradient_step(h, target_sum, step):
    """
    Single projected gradient step for the shift attaining the maximal overlap.
    """
    n = h.shape[0]
    # worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)  # shift index
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    # gradient of the worst overlap (dx omitted, folded into step)
    grad = 1.0 - neigh
    cand = h - step * grad
    return project_feasible(cand, target_sum)

    """One Gaussian‑type smoothing pass with feasibility projection."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    return sm

                else:

    hb[rng.choice(n, n // 2, replace=False)] = 1.0
    hb = project_feasible(hb, target_sum)
    init_candidates.append(hb)

    # random continuous start
    hc = rng.random(n)
    hc = project_feasible(hc, target_sum)
    init_candidates.append(hc)

    # ----- select the best of the initial batch --------------------
    best_h = None
    best_val = np.inf
    for cand in init_candidates:
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()

    cur_h = best_h.copy()
    cur_val = best_val

    # ----- parameters for the stochastic search --------------------
    T0 = 0.01                       # SA temperature scale
    step = 0.2                      # initial gradient step magnitude
    it = 0
    max_noimp = 500                # after this many iterations without improvement we may restart

    # main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.90:
            break

        # temperature decays linearly
        T = max(1e-9, T0 * max(0.0, 1.0 - elapsed / budget_s))
        step = max(0.01, step * 0.998)   # gentle step‑size decay

        # ----- propose a move ---------------------------------------
        if rng.random() < 0.7:
            # gradient‑driven move
            cand = gradient_step(cur_h, target_sum, step)
        else:
            # simple transfer move between two coordinates
            i, j = rng.choice(n, size=2, replace=False)
            # try i -> j
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr > 0 and rng.random() < 0.5:
                delta = rng.random() * max_tr
                cand = cur_h.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                max_tr = min(cur_h[j], 1.0 - cur_h[i])
                if max_tr <= 0:
                    continue
                delta = rng.random() * max_tr
                cand = cur_h.copy()
                cand[j] -= delta
                cand[i] += delta
            cand = project_feasible(cand, target_sum)

        new_val = compute_c5(cand)

        # ----- acceptance test ---------------------------------------
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if rng.random() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            cur_h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    # success – exit early
                    break

        # ----- occasional smoothing -----------------------------------
        if it % 250 == 0 and it > 0:
            sm = smooth_h(cur_h)
            cur_h = project_feasible(sm, target_sum)
            cur_val = compute_c5(cur_h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()

        # ----- random restart if stagnating ---------------------------
        if it % max_noimp == 0 and it > 0:
            if rng.random() < 0.2:
                # pick a fresh start and add a tiny perturbation
                idx = rng.integers(len(init_candidates))
                cand0 = init_candidates[idx].copy()
                cand0 += rng.normal(scale=0.04, size=n)
                cur_h = project_feasible(cand0, target_sum)
                cur_val = compute_c5(cur_h)

        it += 1

    # ----- final refinement (if time permits) -----------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        refined_h, refined_val = slsqp_refine(best_h, target_sum, remaining)
        if refined_val + 1e-12 < best_val:
            best_val, best_h = refined_val, refined_h

    # ----- final safety fix -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(np.sum(best_h) - target_sum) > 1e-9:
        best_h = project_feasible(best_h, target_sum)

    return best_h, best_val, n
