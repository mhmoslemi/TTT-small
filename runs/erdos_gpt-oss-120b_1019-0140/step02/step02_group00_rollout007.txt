def compute_c5_fft(h, n_fft=None):
    n = h.shape[0]
    if n_fft is None:
        L = 2 * n - 1
        n_fft = 1 << (L-1).bit_length()
    g_rev = (1.0 - h)[::-1]
    fft_h = np.fft.rfft(h, n=n_fft)
    fft_g = np.fft.rfft(g_rev, n=n_fft)
    conv = np.fft.irfft(fft_h * fft_g, n=n_fft)[:2*n-1]
    return np.max(conv) * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    n = h.shape[0]
    # Determine FFT length: next power of 2 >= 2*n-1
    L = 2 * n - 1
    n_fft = 1 << (L-1).bit_length()
    g_rev = (1.0 - h)[::-1]
    # Compute real FFT to improve speed
    fft_h = np.fft.rfft(h, n=n_fft)
    fft_g = np.fft.rfft(g_rev, n=n_fft)
    conv = np.fft.irfft(fft_h * fft_g, n=n_fft)[:L]
    return np.max(conv) * (2.0 / n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral for step function h (using FFT for speed)."""
    n = h.shape[0]
    L = 2 * n - 1
    n_fft = 1 << (L - 1).bit_length()  # next power of two
    g_rev = (1.0 - h)[::-1]
    fft_h = np.fft.rfft(h, n=n_fft)
    fft_g = np.fft.rfft(g_rev, n=n_fft)
    conv = np.fft.irfft(fft_h * fft_g, n=n_fft)[:L]
    return np.max(conv) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def binary_from_continuous(v):
    """Convert a real vector into a binary vector with exactly half ones."""
    n = v.shape[0]
    idx = np.argpartition(-v, n // 2)[: n // 2]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max-correlation term averaged over the given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=None):
    """Projected gradient descent with momentum on the max-correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left is not None and time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def hill_climb_swap(h, max_swaps, time_left_fun, rng):
    """Swap a 1 and a 0 to reduce C5; stop after max_swaps or time runs out."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    ones = np.flatnonzero(best_h == 1.0)
    zeros = np.flatnonzero(best_h == 0.0)
    swaps = 0
    while swaps < max_swaps and time_left_fun() > 0.1:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = compute_c5(new_h)
        if new_score < best_score - 1e-12:
            best_h = new_h
            best_score = new_score
            ones = np.flatnonzero(best_h == 1.0)
            zeros = np.flatnonzero(best_h == 0.0)
        swaps += 1
    return best_h, best_score
