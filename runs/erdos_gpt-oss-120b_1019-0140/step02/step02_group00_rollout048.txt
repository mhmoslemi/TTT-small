import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_from_continuous(vals, n_ones):
    # choose n_ones indices with largest values
    idx = np.argpartition(vals, -n_ones)[-n_ones:]
    h = np.zeros_like(vals, dtype=float)
    h[idx] = 1.0
    return h

def hill_climb_sa(h_init, max_iter, time_budget):
    # returns best h and best score.
    h = h_init.copy()
    n = h.shape[0]
    n_ones = n // 2
    # maintain indices groups
    ones = np.flatnonzero(h > 0.5)  # indices of ones
    zeros = np.flatnonzero(h <= 0.5)  # zeros
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    start = time.time()
    # schedule
    T0 = 0.01
    T1 = 1e-5
    for it in range(1, max_iter+1):
        if time.time() - start > time_budget:
            break
        # temperature schedule
        T = T0 * ((T1 / T0) ** (it / max_iter))
        # random swap
        i_one = ones[rng.integers(len(ones))]
        i_zero = zeros[rng.integers(len(zeros))]
        # swap
        h[i_one], h[i_zero] = h[i_zero], h[i_one]
        # evaluate
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # accept swap, update indices and current_score
            # swap indices in lists
            # replace i_one in ones with i_zero, and i_zero in zeros with i_one
            # Efficient update by find positions
            # We can maintain sets or just recompute ones, zeros each iteration? That costs O(N).
            # But we can keep update simple: after swap, ones and zeros indices need to be updated.
            # We'll just recompute ones and zeros each iteration for simplicity (cost O(N) ~512).
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h <= 0.5)
            current_score = new_score

                best_score = new_score
                best_h = h.copy()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1: h at positions shift: h[shift:] -> length n-shift
            # term2: h at positions 0:n-shift placed at shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)

def gradient_descent_momentum_cont(h0, n_iter, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_binary(h_init, max_iter, time_budget, rng):
    h = h_init.copy()
    n = h.shape[0]
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    start = time.time()
    # linear schedule
    T0 = 0.02
    T1 = 1e-5
    for it in range(1, max_iter+1):
        if time.time() - start > time_budget:
            break
        # schedule temperature
        T = T0 * ((T1 / T0) ** (it / max_iter))
        # pick random one and zero indices
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h <= 0.5)
        i_one = ones[rng.integers(len(ones))]
        i_zero = zeros[rng.integers(len(zeros))]
        # swap
        h[i_one], h[i_zero] = h[i_zero], h[i_one]
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_from_continuous(vals, n_ones):
    """Select n_ones largest entries of vals as 1, others 0."""
    idx = np.argpartition(vals, -n_ones)[-n_ones:]
    h = np.zeros_like(vals, dtype=float)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    """Balanced Thue–Morse binary sequence of length n (n power of 2)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    # Ensure exact balance (should hold for powers of two)
    excess = int(seq.sum() - n // 2)
    if excess > 0:
        # turn some 1s to 0s
        idx = np.flatnonzero(seq)[:excess]
        seq[idx] = 0.0
    elif excess < 0:
        idx = np.flatnonzero(1.0 - seq)[: -excess]
        seq[idx] = 1.0
    return seq

def random_binary_balanced(n, rng):
    """Generate a binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term w.r.t. h (averaged over active shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for sh in shifts:
        if sh >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[: n - sh] = h[sh:]
            term2[sh:] = h[: n - sh]
        else:
            s = -sh
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum_cont(h0, n_iter, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing_binary(h_init, max_iter, time_budget, rng):
    """Binary SA preserving exact sum constraint via swaps."""
    h = h_init.copy()
    n = h.shape[0]
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.12
    T1 = 1e-5
    for it in range(1, max_iter + 1):
        if time.time() - start > time_budget:
            break
        T = T0 * ((T1 / T0) ** (it / max_iter))
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h <= 0.5)
        i_one = ones[rng.integers(len(ones))]
        i_zero = zeros[rng.integers(len(zeros))]
        h[i_one], h[i_zero] = h[i_zero], h[i_one]  # swap
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i_one], h[i_zero] = h[i_zero], h[i_one]  # revert
    return best_h, best_score

        # allocate a modest slice of remaining time for this attempt
        attempt_time = min(30.0, time_left() * 0.3)
        max_iter = int(min(200000, attempt_time * 1800))  # ~1800 it/s estimate
        # start from a random perturbation of the current best
        cand = best_h.copy()
        # randomize a small fraction of bits (5% of pairs)
        n_swaps = max(1, N // 100)
        for _ in range(n_swaps):
            ones = np.flatnonzero(cand > 0.5)
            zeros = np.flatnonzero(cand <= 0.5)
            i1 = ones[rng.integers(len(ones))]
            i0 = zeros[rng.integers(len(zeros))]
            cand[i1], cand[i0] = cand[i0], cand[i1]
        h_sa, sc_sa = simulated_annealing_binary(cand, max_iter, attempt_time, rng)
        if sc_sa < best_score:
            best_h, best_score = h_sa, sc_sa
        if best_score <= target:
            break

    # ----- Phase 2: gradient refinement on continuous domain -----
    if time_left() > 8.0:
        gd_time = time_left() * 0.9
        n_iter = int(min(25000, gd_time * 900))  # ~900 it/s estimate
        h_gd, sc_gd = gradient_descent_momentum_cont(best_h, n_iter, step0=0.5, beta=0.9)
        if sc_gd < best_score:
            best_h, best_score = h_gd, sc_gd

    # final projection to guarantee exact constraints
    best_h = project_onto_box_sum(best_h, half)
    best_score = compute_c5(best_h)

    return best_h, best_score, N
