def compute_grad_from_corr_weights(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    comp = 1.0 - h
    # loop over shifts k from -(n-1) to n-1 inclusive
    for idx, wk in enumerate(w):

            continue
        k = idx - (n - 1)
        if k >= 0:
            # i in 0..n-1-k: grad[i] += wk * comp[i+k]
            grad[: n - k] += wk * comp[k:]
            # i in k..n-1: grad[i] -= wk * h[i - k]
            grad[k:] -= wk * h[: n - k]
        else:
            kk = -k
            # i in kk..n-1: grad[i] += wk * comp[i-kk]
            grad[kk:] += wk * comp[: n - kk]
            # i in 0..n-1-kk: grad[i] -= wk * h[i + kk]
            grad[: n - kk] -= wk * h[kk:]

    return grad

def smooth_obj_and_grad(h, alpha, lambda_eq, target, dx):
    n = len(h)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    corr_dx = corr * dx
    # use logsumexp for stability
    # compute lse = logsumexp(alpha * corr_dx)
    a_corr = alpha * corr_dx

    # weights
    w = np.exp(a_corr - lse)  # sum to 1
    # gradient of smooth part
    grad = np.zeros(n)
    # compute contributions via loops
    # optionally vectorize loops with two branches

            grad[kk:] += wk * comp[: n - kk]

    # penalty term
    sum_diff = np.sum(h) - target
    f = f_smooth + lambda_eq * sum_diff * sum_diff
    grad += 2.0 * lambda_eq * sum_diff
    return f, grad

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj_and_grad(h, alpha, lam_eq, target, dx):
    n = h.shape[0]
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    corr_dx = corr * dx
    a_corr = alpha * corr_dx
    lse = logsumexp(a_corr)
    f_smooth = lse / alpha
    w = np.exp(a_corr - lse)  # sum to 1
    grad = np.zeros(n)
    # loop over shifts
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # i from 0 to n-1-k
            grad[: n - k] += wk * comp[k:]
            # i from k to n-1
            grad[k:] -= wk * h[: n - k]
        else:
            kk = -k
            grad[kk:] += wk * comp[: n - kk]
            grad[: n - kk] -= wk * h[kk:]
    grad *= dx
    sum_diff = np.sum(h) - target
    f = f_smooth + lam_eq * sum_diff * sum_diff
    grad += 2.0 * lam_eq * sum_diff
    return f, grad

    target = n / 2.0

        return f, g

                   options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})

    diff = target - cur

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _smooth_obj_grad(h, alpha, lam_eq, target, dx):
    """Smooth approximation to max_k ∫ h(1‑h∘k) and its gradient."""
    n = h.shape[0]
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    corr_dx = corr * dx                       # overlap values for each shift
    a_corr = alpha * corr_dx
    lse = logsumexp(a_corr)                   # stable log‑sum‑exp
    f_smooth = lse / alpha
    w = np.exp(a_corr - lse)                  # softmax weights, sum to 1

    # gradient of the smooth max term
    grad = np.zeros(n)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[: n - k] += wk * comp[k:]
            grad[k:]      -= wk * h[: n - k]
        else:
            kk = -k
            grad[kk:]     += wk * comp[: n - kk]
            grad[: n - kk] -= wk * h[kk:]
    grad *= dx

    # equality‑sum penalty
    sum_diff = np.sum(h) - target
    f = f_smooth + lam_eq * sum_diff * sum_diff
    grad += 2.0 * lam_eq * sum_diff
    return f, grad

def _local_optimize(h0, alpha=80.0, lam_eq=500.0, maxiter=250):
    """L‑BFGS‑B refinement of a candidate."""
    n = h0.shape[0]
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        return _smooth_obj_grad(x, alpha, lam_eq, target, dx)

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method="L-BFGS-B", jac=True,
                   bounds=bounds, options={"maxiter": maxiter,
                                           "ftol": 1e-12,
                                           "disp": False})
    h = np.clip(res.x, 0.0, 1.0)

    # enforce the integral exactly
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        cur2 = np.sum(h)
        diff2 = target - cur2
        if abs(diff2) > 1e-12:
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size:
                inc = diff2 / free.size
                h[free] += inc
                h = np.clip(h, 0.0, 1.0)
    return h
