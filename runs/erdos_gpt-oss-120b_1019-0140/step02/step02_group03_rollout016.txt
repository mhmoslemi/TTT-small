class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
        self.shift = k
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        n = self.n
        if k >= 0:
            # overlapping parts
            overlap = np.dot(h[:n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
        return x[-1] - overlap * self.dx

from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = hi - np.min(v) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k > 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        elif k < 0:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        else:
            # shift 0 not used in constraints normally
            overlap = np.dot(h, 1.0 - h)
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, n, maxiter=500):
    # Build constraints
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res

    h = h_init.copy()
    best_local_h = h.copy()
    best_local_c5 = compute_c5(h)
    step0 = 0.8
    no_improve_counter = 0
    for it in range(20000):
        if time.time() > time_deadline:
            break

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = hi - np.min(v) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

        k = self.k
        if k > 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        elif k < 0:
            kk = -k

        else:  # shift 0 (rarely needed)

        return x[-1] - overlap * self.dx
