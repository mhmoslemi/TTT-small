    n = len(h)
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]
    improved = True
    while improved and time.time() < deadline - 0.01:
        improved = False
        best_swap = None
        best_val = cur_val
        # Evaluate swaps
        for i in ones:
            # early break if time near
            if time.time() > deadline - 0.01:
                break
            for j in zeros:
                # swap and compute
                cur_h[i], cur_h[j] = 0, 1  # temporary swap
                val = compute_c5(cur_h)
                # revert
                cur_h[i], cur_h[j] = 1, 0
                if val < best_val - 1e-12:
                    best_val = val
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            cur_h[i], cur_h[j] = 0, 1
            cur_val = best_val
            # update ones and zeros arrays
            # Instead recompute each iteration for simplicity
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            improved = True
    return cur_h, cur_val

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlate h with 1-h
    # use np.correlate mode='full' on float arrays.
    # produce array length 2*n-1
    c = np.correlate(h, 1.0 - h, mode='full')
    # max is max across all k
    return np.max(c) * dx

def hill_climb_binary(h, deadline, rng):
    # Do hill climbing with best swap moves
    n = len(h)
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    # Precompute list of positions for ones and zeros each iteration
    while time.time() < deadline - 0.01:
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        best_swap = None
        best_val = cur_val
        # loop over all swaps
        for i in ones:
            # maybe break early if time up
            if time.time() > deadline - 0.01:
                break
            # for each zero j
            # Random order to maybe find better early? Not needed.
            for j in zeros:
                # swapping
                cur_h[i] = 0
                cur_h[j] = 1
                new_val = compute_c5(cur_h)
                # revert
                cur_h[i] = 1
                cur_h[j] = 0
                if new_val < best_val - 1e-12:
                    best_val = new_val
                    best_swap = (i, j)
            # optional early break if best_val already equals 0? Not necessary.
        if best_swap is not None:
            i, j = best_swap
            cur_h[i] = 0
            cur_h[j] = 1
            cur_val = best_val
            # continue loop
        else:
            break
    return cur_h, cur_val

def make_overlap_constr(k, n, dx):
    def con(x, k=k):
        h = x[:n]
        if k >= 0:
            overl = np.dot(h[: n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: n - kk])
        # overl * dx <= t (which is x[-1])
        return x[-1] - overl * dx
    return con

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated by full discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_binary(n, target, rng):
    """Uniform random binary vector with exactly `target` ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, target, replace=False)
    h[idx] = 1.0
    return h

def hill_climb_binary(h_start, deadline, rng):
    """Best‑swap hill climbing on binary vectors (0/1)."""
    n = h_start.shape[0]
    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h)

    while time.time() < deadline - 0.01:
        ones = np.where(cur_h == 1.0)[0]
        zeros = np.where(cur_h == 0.0)[0]
        best_swap = None
        best_val = cur_val

        # evaluate every possible 1↔0 swap
        for i in ones:
            if time.time() > deadline - 0.01:
                break
            cur_h[i] = 0.0
            for j in zeros:
                cur_h[j] = 1.0
                val = compute_c5(cur_h)
                cur_h[j] = 0.0
                if val < best_val - 1e-12:
                    best_val = val
                    best_swap = (i, j)
            cur_h[i] = 1.0

        if best_swap is not None:
            i, j = best_swap
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            cur_val = best_val
        else:
            break
    return cur_h, cur_val

def perturb_binary(h, rng, swaps=2):
    """Perform `swaps` random 1↔0 exchanges on a binary vector."""
    n = h.shape[0]
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    h_new = h.copy()
    rng.shuffle(ones)
    rng.shuffle(zeros)
    for s in range(min(swaps, len(ones), len(zeros))):
        i = ones[s]
        j = zeros[s]
        h_new[i] = 0.0
        h_new[j] = 1.0
    return h_new

def slsqp_optimize(h_start, t_start, deadline, rng):
    """Run a single SLSQP optimisation (continuous) starting from a binary point."""
    n = h_start.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # equality for integral
    def sum_con(x):
        return np.sum(x[:n]) - target_sum

    constraints = [{"type": "eq", "fun": sum_con}]

    # inequality for each shift k (exclude 0)
    def make_ineq(k):
        def con(x, k=k):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])
            return x[-1] - overl * dx
        return con

    for k in range(-n + 1, n):
        if k == 0:
            continue
        constraints.append({"type": "ineq", "fun": make_ineq(k)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    x0 = np.concatenate([h_start, [t_start]])

    maxiter = 800
    # shrink maxiter if very little time left
    remaining = deadline - time.time()
    if remaining < 2.0:
        return None, None
    options = {"maxiter": maxiter, "ftol": 1e-8, "disp": False}
    try:
        res = minimize(
            fun=lambda x: x[-1],
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options=options,
        )
    except Exception:
        return None, None

    if not res.success:
        return None, None

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = h_opt.sum()
    if abs(cur - target_sum) > 1e-9:
        h_opt += (target_sum - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    val = compute_c5(h_opt)
    return h_opt, val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n = base.shape[0]
        # force binary start
        h0 = (base >= 0.5).astype(float)
        target = n // 2
        cur = int(h0.sum())
        if cur != target:
            diff = target - cur
            if diff > 0:
                zeros = np.where(h0 == 0)[0]
                rng.shuffle(zeros)
                h0[zeros[:diff]] = 1.0
            else:
                ones = np.where(h0 == 1)[0]
                rng.shuffle(ones)
                h0[ones[: -diff]] = 0.0
    else:
        n = 80
        target = n // 2
        h0 = random_binary(n, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- hill‑climb from the (possibly binary) start ----------------
    if time.time() < deadline - 2.0:
        hc_h, hc_val = hill_climb_binary(best_h.copy(), deadline, rng)
        if hc_val < best_val:
            best_h, best_val = hc_h, hc_val

    # ----- repeated random restarts with hill‑climb ------------------
    while time.time() < deadline - 0.5:
        # occasionally perturb the current best to explore neighbourhood
        if rng.random() < 0.2:
            start_vec = perturb_binary(best_h, rng, swaps=2)
        else:
            start_vec = random_binary(n, target, rng)
        hc_h, hc_val = hill_climb_binary(start_vec, deadline, rng)
        if hc_val < best_val:
            best_h, best_val = hc_h, hc_val
            if best_val < 0.3808:
                break

    # ----- optional continuous refinement via SLSQP -------------------
    if time.time() < deadline - 1.0:
        opt_h, opt_val = slsqp_optimize(best_h, best_val, deadline, rng)
        if opt_h is not None and opt_val < best_val:
            best_h, best_val = opt_h, opt_val

    # ----- final safety clip -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    target_sum = n / 2.0
    if abs(cur_sum - target_sum) > 1e-9:
        diff = target_sum - cur_sum
        # tiny redistribution
        for i in range(n):
            if diff > 0 and best_h[i] < 1.0:
                inc = min(diff, 1.0 - best_h[i])
                best_h[i] += inc
                diff -= inc
            elif diff < 0 and best_h[i] > 0.0:
                dec = min(-diff, best_h[i])
                best_h[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
