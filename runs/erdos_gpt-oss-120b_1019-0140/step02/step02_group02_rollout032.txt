def enforce_sum(h):
    target = n / 2
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12: return h
    # adjust by small increments using random indices
    # while diff !=0
    idxs = np.arange(n)
    rng.shuffle(idxs)
    for i in idxs:
        if abs(diff) < 1e-12: break
        if diff > 0 and h[i] < 1 - 1e-12:
            inc = min(diff, 1 - h[i])
            h[i] += inc
            diff -= inc
        elif diff < 0 and h[i] > 1e-12:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h

            # fallback: random feasible pair
            donor = rng.choice(donor_idxs)
            recipient = rng.choice(rec_idxs)
            max_transfer = min(cur_h[donor], 1.0 - cur_h[recipient])
            if max_transfer <= 1e-12:
                break

        # take a random fraction of the feasible transfer (occasionally full)
        delta = max_transfer * rng.random()
        if delta < 1e-12:
            delta = max_transfer * 0.5

        # perform transfer
        cur_h[donor] -= delta
        cur_h[recipient] += delta
        cur_h = np.clip(cur_h, 0.0, 1.0)

        new_val = compute_c5(cur_h)

        if new_val < cur_val - 1e-12:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = cur_h.copy()
                if best_val <= 0.3808:
                    break
        else:
            # Metropolis acceptance for occasional uphill moves
            remaining = deadline - time.time()
            T = max(1e-12, (remaining / budget_s) * 0.001)
            if np.exp(-(new_val - cur_val) / T) > rng.random():
                cur_val = new_val
                no_improve = 0
            else:
                # revert the move
                cur_h[donor] += delta
                cur_h[recipient] -= delta
                no_improve += 1

        if no_improve >= max_no_improve:
            # restart from a fresh random point
            cur_h = random_start_vector(n, rng)
            cur_h = enforce_sum(cur_h, target, rng)
            cur_val = compute_c5(cur_h)
            no_improve = 0

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
