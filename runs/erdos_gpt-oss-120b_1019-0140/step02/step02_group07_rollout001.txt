def binary_anneal(start_bin, time_end, rng, target_sum):
    # start_bin is array of 0/1 floats, sum = target_sum.
    n = len(start_bin)
    h = start_bin.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # schedule: temperature from init to final based on iteration progress.
    # We'll use max_iter based on time: each iteration we estimate time as small constant, we just loop until time_end.
    it = 0
    # We need to track something to compute T: for simplicity we can use exponential cooling based on time fraction.
    # We'll compute fraction = (time.time() - start_time) / total_time where total_time = time_end - start_time

    # We'll need start_time; but we can compute fraction as (current - time_start) / (time_end - time_start). But not passed.
    # Instead compute T as init_temp * (time_end - now) / (time_end - start_time). But not have start_time.
    # Let's approximate: we can keep a local iteration counter and gradually decay T by factor each iteration.

    init_temp = 5e-3
    final_temp = 1e-7
    # We'll update T as exponential: T = init_temp * (final_temp/init_temp)^(it / max_iter) approximating schedule where max_iter guessed.
    # Determine max_iter as (time_end - now) / average iteration time. Let's just set a large max_iter = 200000.
    max_iter = 200000

    while time.time() < time_end and it < max_iter:
        # Select random pair (i with 1, j with 0)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        dE = new_c5 - cur_c5
        # compute temperature
        T = init_temp * (final_temp / init_temp) ** (it / max_iter)
        # acceptance
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
        it += 1
    return best_h, best_c5

        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:

        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        # temperature schedule
        T = init_temp * (final_temp / init_temp) ** (it / max_iter)
        # propose delta: possibly scaled with temperature to adapt step size
        # Use delta proportional to max_delta, scaled by random^(1/T?) not needed.
        # Use a basic delta = max_delta * rng.random() * 0.5
        delta = max_delta * rng.random() * 0.5

        # if temperature is low, reduce delta
        delta = delta * (T / init_temp)  # scaling down as temperature decreases

        # ensure delta > 0
        if delta <= 0:
            it += 1
            continue

        old_i = h[i]
        old_j = h[j]
        h[i] += delta
        h[j] -= delta

        # project sum? Since we moved mass from j to i, total sum preserved exactly.

            # revert
            h[i] = old_i
            h[j] = old_j
        it += 1

import time, math, numpy as np

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_sum(v, target, iters=50):
    low, high = -1.0, 1.0
    for _ in range(iters):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    x = np.clip(v + (low + high) * 0.5, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt:
            x[free] += residual / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=int, count=n)

def rudin_shapiro_seq(n):
    a = np.array([1], dtype=int)
    while a.size < n:
        b = a * ((-1) ** np.arange(a.size))
        a = np.concatenate([a, b])
    a = a[:n]
    return ((a + 1) // 2).astype(float)

def adjust_binary(vec, target):
    cur = int(vec.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(vec == 0)[:diff]
        vec[zeros] = 1.0
    elif diff < 0:
        ones = np.flatnonzero(vec == 1)[: -diff]
        vec[ones] = 0.0
    return vec

def continuous_anneal(start, end_time, rng, target):
    n = len(start)
    h = project_sum(start, target)
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    max_iters = 200000
    init_T, final_T = 5e-3, 1e-7
    it = 0
    while time.time() < end_time and it < max_iters:
        inc = np.where(h < 1 - 1e-12)[0]

        if inc.size == 0 or dec.size == 0:
            break
