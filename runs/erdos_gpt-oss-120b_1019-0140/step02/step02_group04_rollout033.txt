def compute_c5_bin(h):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, h, mode='full')
    # c index shift: shift = k = i - j?
    # For shift k = - (n-1) ... n-1, the overlap count = c[n-1 + k]
    # remove k=0
    min_overlap = np.min(np.concatenate([c[:n-1], c[n:]]))
    C5 = 1.0 - min_overlap * dx
    return C5, min_overlap

def simulated_annealing_binary(n, rng, time_limit):
    # initially random binary with n/2 ones
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    rng.shuffle(h)
    C5, min_ov = compute_c5_bin(h)
    best_h = h.copy()
    best_val = C5
    best_min_ov = min_ov
    
    T0 = 0.5
    # schedule: linear cooling to near zero at end
    start = time.time()
    while time.time() < time_limit:
        # compute remaining fraction
        frac = (time.time() - start) / (time_limit - start)
        T = T0 * (1 - frac)  # linear cooling
        # propose swap
        idx1 = rng.integers(0, n)
        while h[idx1] == 0:
            idx1 = rng.integers(0, n)
        idx0 = rng.integers(0, n)
        while h[idx0] == 1:
            idx0 = rng.integers(0, n)
        # swap
        h[idx1] = 0
        h[idx0] = 1
        Ccand, min_cand = compute_c5_bin(h)
        delta = Ccand - C5  # if negative, better.
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            C5 = Ccand
            min_ov = min_cand
            if C5 < best_val:
                best_val = C5
                best_h = h.copy()
                best_min_ov = min_ov
        else:
            # revert swap
            h[idx1] = 1
            h[idx0] = 0
    return best_h, best_val
```

def greedy_improve_binary(h):
    n = len(h)
    best_h = h.copy()
    best_val, _ = compute_c5_bin(best_h)
    improved = True
    while improved:
        improved = False
        # for each 1-0 pair
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        for i in ones:
            for j in zeros:
                best_h[i] = 0
                best_h[j] = 1
                val, _ = compute_c5_bin(best_h)
                if val < best_val - 1e-12:
                    best_val = val
                    improved = True
                    # keep new config, break inner loops
                    break
                else:
                    # revert
                    best_h[i] = 1
                    best_h[j] = 0
            if improved:
                break
    return best_h, best_val

        # compute fraction of time used
        frac = elapsed / (time_limit - start)
        # schedule alpha linearly increase
        alpha = 5 + 45 * frac  # from 5 to 50
        # learning rate schedule: start 0.1, dec to 0.001
        eta = 0.1 * (0.01 ** frac)  # exponential decay

import time
import numpy as np

def compute_c5_vec(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_capped_simplex(v, target_sum):
    n = v.shape[0]
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        idx = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if idx.size > 0:
            w[idx] += diff / idx.size
            w = np.clip(w, 0.0, 1.0)
    return w

def generate_random_feasible(n, target_sum, rng):
    x = rng.random(n)
    x = x * (target_sum / x.sum())
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    # distribute diff greedily
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

def gradient_descent_smooth(h_init, dx, target_sum, end_time, rng):
    h = h_init.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5_vec(best_h, dx)
    start = time.time()
    while time.time() < end_time:
        elapsed = time.time() - start
        frac = elapsed / (end_time - start + 1e-12)
        alpha = 5.0 + 45.0 * frac
        eta = 0.5 * (1.0 - frac) + 0.001
        O = np.correlate(h, 1.0 - h, mode='full') * dx
        maxO = np.max(O)
        expO = np.exp(alpha * (O - maxO))
        w = expO / np.sum(expO)
        g = np.full_like(h, dx)
        for k in range(-n + 1, n):
            wk = w[k + n - 1]
            if wk == 0.0:
                continue
            if k >= 0:
                g[: n - k] -= wk * h[k:] * dx
                g[k:] -= wk * h[: n - k] * dx
            else:
                kk = -k
                g[kk:] -= wk * h[: n - kk] * dx
                g[: n - kk] -= wk * h[kk:] * dx
        h = h - eta * g
        h = np.clip(h, 0.0, 1.0)
        h = project_capped_simplex(h, target_sum)
        val = compute_c5_vec(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

            step = min(step * 1.2, 0.5)

        else:
            step *= 0.9
    cur_h = project_capped_simplex(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():

        n = len(h0)

    else:

    best_val = compute_c5_vec(best_h, dx)

import time
import numpy as np

def compute_c5_vec(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_capped_simplex(v, target_sum):
    n = v.shape[0]
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        idx = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if idx.size:
            w[idx] += diff / idx.size
            w = np.clip(w, 0.0, 1.0)
    return w

def generate_random_feasible(n, target_sum, rng):
    x = rng.random(n)
    x = x * (target_sum / x.sum())
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

def gradient_descent_smooth(h_init, dx, target_sum, end_time, rng):
    h = h_init.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5_vec(best_h, dx)
    start = time.time()
    while time.time() < end_time:
        elapsed = time.time() - start
        frac = elapsed / (end_time - start + 1e-12)
        alpha = 5.0 + 45.0 * frac      # from 5 to 50
        eta = 0.5 * (1.0 - frac) + 0.001
        O = np.correlate(h, 1.0 - h, mode="full") * dx
        maxO = np.max(O)
        expO = np.exp(alpha * (O - maxO))
        w = expO / np.sum(expO)        # softmax over shifts
        g = np.full_like(h, dx)        # constant part from sum_i h_i term
        for k in range(-n + 1, n):
            wk = w[k + n - 1]
            if wk == 0.0:
                continue
            if k >= 0:
                g[: n - k] -= wk * h[k:] * dx
                g[k:]      -= wk * h[: n - k] * dx
            else:
                kk = -k
                g[kk:]      -= wk * h[: n - kk] * dx
                g[: n - kk] -= wk * h[kk:] * dx
        h = h - eta * g
        h = np.clip(h, 0.0, 1.0)
        h = project_capped_simplex(h, target_sum)
        val = compute_c5_vec(h, dx)
        if val < best_val:
            best_val, best_h = val, h.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

def refine_continuous(h, dx, target_sum, end_time, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.1
    n = h.shape[0]
    while time.time() < end_time:
        donors = np.where(cur_h > 1e-9)[0]
        recpts = np.where(cur_h < 1.0 - 1e-9)[0]
        if donors.size == 0 or recpts.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recpts)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_capped_simplex(cand, target_sum)
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h, cur_val = cand, cand_val
            step = min(step * 1.2, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.9
    cur_h = project_capped_simplex(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 100
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    if h0 is not None:
        best_h = project_capped_simplex(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        best_h = generate_random_feasible(n, target_sum, rng)
    best_val = compute_c5_vec(best_h, dx)

    # multi‑start smooth gradient descent
    while time.time() - start_time < budget_s * 0.85:
        time_left = budget_s - (time.time() - start_time)
        if time_left < 5.0:
            break
        h_start = generate_random_feasible(n, target_sum, rng)
        gd_time = min(12.0, time_left * 0.5)
        h_opt, val_opt = gradient_descent_smooth(
            h_start, dx, target_sum, time.time() + gd_time, rng
        )
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt
            if best_val < 0.3808:
                break

    # final local refinement with remaining budget
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        best_h, best_val = refine_continuous(
            best_h, dx, target_sum, time.time() + remaining, rng
        )

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_capped_simplex(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)
    return best_h, best_val, n
