def simulated_annealing(h0, time_limit, rng):

    cur_h = h.copy()
    best_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_val = cur_val
    n = len(h)
    target_sum = n/2
    start = time.time()
    # parameters
    T0 = 0.05  # maybe higher
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / time_limit)
        # choose move
        r = rng.random()
        if r < 0.5:
            # swap move
            i,j = rng.choice(n, size=2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.8:
            # transfer move (continuous)
            i,j = rng.choice(n, size=2, replace=False)
            # compute max transfer amount such that both stay in [0,1]
            max_tr = min(cur_h[i], 1 - cur_h[j])
            if max_tr <= 0:
                # reverse direction
                max_tr = min(cur_h[j], 1 - cur_h[i])
                donor, recv = j, i
            else:
                donor, recv = i, j
            if max_tr <= 0:
                continue
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        else:
            # gradient move
            O = np.correlate(cur_h, 1 - cur_h, mode='full')
            k = np.argmax(O) - (n - 1)
            grad = np.zeros_like(cur_h)
            if k >= 0:
                grad[:n - k] += (1 - cur_h[k:])
                grad[k:] -= cur_h[:n - k]
            else:
                kk = -k
                grad[kk:] += (1 - cur_h[:n - kk])
                grad[:n - kk] -= cur_h[kk:]
            # take step opposite to gradient
            step = rng.random() * 0.1  # step size scale
            cand = cur_h - step * grad
            # clip and fix sum
            cand = np.clip(cand, 0.0, 1.0)
            cur_sum = np.sum(cand)
            diff = target_sum - cur_sum
            if abs(diff) > 1e-12:
                # distribute diff uniformly among free variables
                free = np.where((cand > 0) & (cand < 1))[0]
                if free.size:
                    inc = diff / free.size
                    cand[free] += inc
                    cand = np.clip(cand, 0.0, 1.0)
                # then fix any leftover
                diff2 = target_sum - np.sum(cand)
                for idx in range(n):
                    if abs(diff2) < 1e-12:
                        break
                    if diff2 > 0 and cand[idx] < 1:
                        add = min(diff2, 1 - cand[idx])
                        cand[idx] += add
                        diff2 -= add
                    elif diff2 < 0 and cand[idx] > 0:
                        sub = min(-diff2, cand[idx])
                        cand[idx] -= sub
                        diff2 += sub
                cand = np.clip(cand, 0.0, 1.0)
        # evaluate:
        new_val = compute_c5(cand)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))

                best_val = new_val
                best_h = cand.copy()
                # early exit if below target
                if best_val < 0.3808:
                    break
    return best_h, best_val

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_and_fix(h, target_sum):
    """Gaussian smoothing with exact sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    cur = np.sum(sm)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            inc = diff / free.size
            sm[free] += inc
            sm = np.clip(sm, 0.0, 1.0)
        diff = target_sum - np.sum(sm)
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
        sm = np.clip(sm, 0.0, 1.0)
    return sm

def fix_sum_exact(h, target_sum):
    """Project onto box [0,1] and adjust to satisfy the sum exactly."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        # residual correction (linear sweep)
        diff = target_sum - np.sum(h)
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def simulated_annealing(h_start, time_limit, rng):
    """Simple SA that respects the integral constraint."""
    n = len(h_start)
    target_sum = n / 2.0
    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    T0 = 0.02  # initial temperature
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)

        # move selection
        r = rng.random()
        if r < 0.45:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.85:  # transfer (continuous)
            i, j = rng.choice(n, size=2, replace=False)
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr <= 0.0:
                max_tr = min(cur_h[j], 1.0 - cur_h[i])
                if max_tr <= 0.0:
                    continue
                donor, recv = j, i
            else:
                donor, recv = i, j
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        else:  # gradient step
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            k = int(np.argmax(corr)) - (n - 1)
            grad = np.zeros_like(cur_h)
            if k >= 0:
                grad[: n - k] += (1.0 - cur_h[k:])
                grad[k:] -= cur_h[: n - k]
            else:
                kk = -k
                grad[kk:] += (1.0 - cur_h[: n - kk])
                grad[: n - kk] -= cur_h[kk:]
            step = 0.08 * rng.random()
            cand = cur_h - step * grad
            cand = np.clip(cand, 0.0, 1.0)
            cand = fix_sum_exact(cand, target_sum)

        new_val = compute_c5(cand)
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            cur_h, cur_val = cand, new_val
            if new_val < best_val:
                best_h, best_val = cand.copy(), new_val
                if best_val < 0.3808:
                    break
    return best_h, best_val

def lbfgsb_refine(h0, target_sum, time_limit):
    """L‑BFGS‑B minimizing a smooth approximation of max‑overlap."""
    n = len(h0)
    dx = 2.0 / n
    BETA = 30.0          # smoothing parameter
    PENALTY = 500.0      # sum‑constraint penalty weight

    def fun_and_grad(x):
        # correlation vector (full mode)
        O = np.correlate(x, 1.0 - x, mode="full") * dx
        max_O = np.max(O)
        # smooth max via log‑sum‑exp
        shift = BETA * (O - max_O)
        exp_shift = np.exp(shift)
        sumexp = np.sum(exp_shift)          # includes the max term (exp(0)=1)
        smooth = max_O + np.log(sumexp) / BETA

        diff = np.sum(x) - target_sum
        f = smooth + PENALTY * diff * diff

        w = exp_shift / sumexp               # softmax weights
        grad = np.zeros_like(x)
        idx = 0
        for k in range(-n + 1, n):
            wk = w[idx]
            if k >= 0:
                grad[: n - k] += wk * (1.0 - x[k:])
                grad[k:]      -= wk * x[: n - k]
            else:
                kk = -k
                grad[kk:]     += wk * (1.0 - x[: n - kk])
                grad[: n - kk] -= wk * x[kk:]
            idx += 1
        grad += 2.0 * PENALTY * diff
        return f, grad

    bounds = [(0.0, 1.0)] * n
    # limit iterations loosely; optimizer is fast for n≈80
    res = opt.minimize(fun_and_grad, h0, method="L-BFGS-B", jac=True,
                       bounds=bounds, options={"maxiter": 500, "disp": False})
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = fix_sum_exact(h_opt, target_sum)
    val_opt = compute_c5(h_opt)
    return h_opt, val_opt

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():

        n = len(h0)
    else:

        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target_sum = n / 2.0
    h0 = fix_sum_exact(h0, target_sum)

    # ---- phase 1 : simulated annealing -----------------------------
    elapsed = time.time() - start
    sa_budget = max(0.0, budget_s * 0.6 - elapsed - 5.0)
    if sa_budget > 2.0:
        best_h, best_val = simulated_annealing(h0, sa_budget, rng)
    else:
        best_h = h0.copy()
        best_val = compute_c5(best_h)

    # early exit if already good enough
    if best_val < 0.3808:
        best_h = fix_sum_exact(best_h, target_sum)
        return best_h, best_val, n

    # ---- phase 2 : L‑BFGS‑B refinement -----------------------------
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed - 3.0)
    if remaining > 0.5:
        try:
            refined_h, refined_val = lbfgsb_refine(best_h, target_sum, remaining)
            if refined_val < best_val:
                best_h, best_val = refined_h, refined_val
        except Exception:
            pass

    # ---- optional smoothing passes ---------------------------------
    elapsed = time.time() - start
    if elapsed < budget_s - 1.0:
        for _ in range(6):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_h, best_val = sm, sm_val
            else:
                break

    # ---- final safety correction -----------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum_exact(best_h, target_sum)

    return best_h, best_val, n
