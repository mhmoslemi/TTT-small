def worst_shift_info(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    # compute shift: shift = max_idx - (n - 1)
    shift = max_idx - (n - 1)
    # also compute neighbor_sums for each i: sum of h[i+shift] and h[i-shift] if indices in bounds.
    # create array neighbor_sum of length n.
    ks = shift
    neighbor_sum = np.zeros_like(h)
    if ks >= 0:
        # indices i from 0 to n-ks-1 have h[i+ks]
        neighbor_sum[:n - ks] += h[ks:]
        # for i >= ks: h[i-ks] defined for i=ks..n-1
        neighbor_sum[ks:] += h[:n - ks]
    else:
        k = -ks
        neighbor_sum[k:] += h[:n - k]
        neighbor_sum[:n - k] += h[k:]
    return shift, neighbor_sum, corr[max_idx] * dx
```

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- Utility Functions ------------------- #
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """
    Project vector v onto the hypercube [lo,hi]^n while fixing the sum to target.
    Uses binary search for the Lagrange multiplier lambda.
    """
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# ------------------- Constraint Classes ------------------- #
class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

# ------------------- SLSQP Optimisation ------------------- #
def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    """Run a single SLSQP optimisation from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

# ------------------- Main Entry Point ------------------- #
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- Initialise ----- #
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----- Constraints ----- #
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # ----- Phase 1: Multi‑restart SLSQP ----- #
    phase1_deadline = start_time + budget_s * 0.6
    while time.time() < phase1_deadline:
        h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, target_sum)
        t_start = compute_c5(h_start)

        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800)
        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        if abs(h_opt.sum() - target_sum) > 1e-9:
            h_opt = project_to_box_sum(h_opt, target_sum)

        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            if best_c5 < 0.3808:
                break

    # ----- Phase 2: Targeted pairwise & stochastic hill‑climbing ----- #
    cur_h = best_h.copy()
    cur_c5 = best_c5
    base_eps = 0.2  # initial step scale

    while time.time() - start_time < budget_s - 0.5:
        # Identify the shift giving the current worst overlap
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # Build neighbor‑sum vector for this shift
        neighbor = np.zeros_like(cur_h)
        k = shift
        if k >= 0:
            neighbor[:n - k] += cur_h[k:]
            neighbor[k:]   += cur_h[:n - k]
        else:
            kk = -k
            neighbor[kk:]   += cur_h[:n - kk]
            neighbor[:n - kk] += cur_h[kk:]

        # Choose indices to transfer mass from low‑to‑high neighbor sum
        i_max = int(np.argmax(neighbor))
        i_min = int(np.argmin(neighbor))
        eps = base_eps * (rng.random() ** 2)
        delta = min(eps, 1.0 - cur_h[i_max], cur_h[i_min])

        improved = False
        if delta > 0:
            cand_h = cur_h.copy()
            cand_h[i_max] += delta
            cand_h[i_min] -= delta
            cand_c5 = compute_c5(cand_h)
            if cand_c5 + 1e-12 < cur_c5:
                cur_h = cand_h
                cur_c5 = cand_c5
                improved = True
        else:
            # Random perturbation as a fallback
            d = rng.normal(size=n)
            d -= d.mean()
            step = base_eps * (rng.random() ** 2)
            cand = cur_h + step * d
            cand = project_to_box_sum(cand, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 + 1e-12 < cur_c5:
                cur_h = cand
                cur_c5 = cand_c5
                improved = True

        if improved and cur_c5 + 1e-12 < best_c5:
            best_h = cur_h.copy()
            best_c5 = cur_c5
            if best_c5 < 0.3808:
                break

        # Occasionally shrink the exploratory step size
        if rng.random() < 0.001:
            base_eps *= 0.9

    # ----- Final projection & evaluation ----- #
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
