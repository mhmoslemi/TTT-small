def gradient_move(h, k, step=0.1):
    # compute grad = (1 - h_{i+k}) - h_{i-k}
    # where indices out of bounds? For shift k, only indices with both terms defined contributed to overlap (i from 0..n-1-k if k>=0).
    # For gradient, we can set missing term to 0 (since those terms not in sum).
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # for i in 0..n-1-k: term (1 - h[i+k])
        grad[:n-k] += 1 - h[k:]
        # for i in k..n-1: term -h[i-k]
        grad[k:] -= h[:n-k]
    else:
        kk = -k
        # for i in kk..n-1: term (1 - h[i-kk])? Actually using formula (1 - h_{i+k}) where k negative => i+k = i-kk
        grad[kk:] += 1 - h[:n-kk]
        # for i in 0..n-1-kk: term -h_{i+kk}
        grad[:n-kk] -= h[kk:]
    # We want to move opposite gradient (i.e., reduce the max overlap)
    delta = -grad
    # Normalize step
    max_abs = np.max(np.abs(delta))

    cand = h + delta
    cand = np.clip(cand, 0.0, 1.0)

import time, numpy as np, cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # distribute diff uniformly first
    h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    # small sequential fix
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0.0, 1.0)

def solve_lp(n):
    target_sum = n / 2.0
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    w = cp.Variable((n, n))
    constraints = [
        h >= 0, h <= 1,
        cp.sum(h) == target_sum,
        t >= 0,
        w >= 0,
        w >= cp.reshape(h, (n,1)) + cp.reshape(h, (1,n)) - 1,
        w <= cp.reshape(h, (n,1)),
        w <= cp.reshape(h, (1,n))
    ]
    for k in range(-n+1, n):
        if k == 0:
            continue
        if k > 0:
            sum_w = cp.sum([w[i, i+k] for i in range(n - k)])
        else:
            kk = -k
            sum_w = cp.sum([w[i+kk, i] for i in range(n - kk)])
        constraints.append(t >= dx * (target_sum - sum_w))
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-8, reltol=1e-8, feastol=1e-8, verbose=False)
        if h.value is None:
            return None
        h_opt = np.clip(np.array(h.value).astype(float), 0.0, 1.0)
        h_opt = fix_sum(h_opt, target_sum)
        return h_opt
    except Exception:
        return None

import time
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for discretized h on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def fix_sum(h, target):
    """Clip to [0,1] and enforce exact sum==target."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        # uniform correction
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        # sequential fix for any residual
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)

def solve_lp(n):
    """Linear relaxation with McCormick envelopes; returns a feasible h."""
    target = n / 2.0
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    w = cp.Variable((n, n))
    constraints = [
        h >= 0, h <= 1,
        cp.sum(h) == target,
        t >= 0,
        w >= 0,
        w >= cp.reshape(h, (n, 1)) + cp.reshape(h, (1, n)) - 1,
        w <= cp.reshape(h, (n, 1)),
        w <= cp.reshape(h, (1, n)),
    ]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        if k > 0:
            sum_w = cp.sum([w[i, i + k] for i in range(n - k)])
        else:
            kk = -k
            sum_w = cp.sum([w[i + kk, i] for i in range(n - kk)])
        constraints.append(t >= dx * (target - sum_w))
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-8,
                   reltol=1e-8, feastol=1e-8, verbose=False)
        if h.value is None:
            return None
        h0 = np.clip(np.array(h.value, dtype=float), 0.0, 1.0)
        return fix_sum(h0, target)
    except Exception:
        return None

def overlap_shift(h, k):
    """Overlap for a given integer shift k (no scaling)."""
    n = len(h)
    if k >= 0:
        return np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        return np.dot(h[kk:], 1.0 - h[: n - kk])

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h_cur = np.array(initial_h_values, dtype=float)
        n = len(h_cur)
    else:
        n = 80
        h_cur = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h_cur[idx] = 1.0

    target = n / 2.0
    h_cur = fix_sum(h_cur, target)

    # ---- linear relaxation -----------------------------------------
    h_lp = solve_lp(n)
    if h_lp is not None:
        h_cur = h_lp.copy()

    cur_val = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_val = cur_val

    # compare with raw initial if available
    if "initial_h_values" in globals():
        init_val = compute_c5(np.array(initial_h_values, dtype=float))
        if init_val < best_val:
            best_h = np.array(initial_h_values, dtype=float)
            best_val = init_val
            h_cur = best_h.copy()
            cur_val = best_val

    # ---- simulated annealing ---------------------------------------
    p_swap, p_transfer, p_gradient, p_smooth, p_noise = 0.30, 0.30, 0.20, 0.10, 0.10
    T0 = 0.008  # starting temperature
    kernel = np.array([0.25, 0.5, 0.25])

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.88:
            break
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)
        r = rng.random()
        cand = None

        # ---- swap --------------------------------------------------
        if r < p_swap:
            i, j = rng.choice(n, size=2, replace=False)
            cand = h_cur.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # ---- transfer -----------------------------------------------
        elif r < p_swap + p_transfer:

            donor, recv = (i, j) if h_cur[i] > 0 and h_cur[j] < 1 else (j, i)
            max_tr = min(h_cur[donor], 1.0 - h_cur[recv])

        cand = fix_sum(cand, target)
        new_val = compute_c5(cand)

        accept = False
        if new_val < cur_val:

        elif T > 0:
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            h_cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break

    # ---- post‑SA smoothing -----------------------------------------
    if time.time() - start < budget_s - 5.0:
        for _ in range(5):
            sm = np.convolve(best_h, kernel, mode="same")
            sm = np.clip(sm, 0.0, 1.0)
            sm = fix_sum(sm, target)
            val = compute_c5(sm)
            if val + 1e-12 < best_val:
                best_val = val
                best_h = sm
            else:
                break

    # ---- local refinement with SLSQP -------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 6.0:
        n = len(best_h)
        dx = 2.0 / n
        target = n / 2.0

        def sum_eq(x):
            return np.sum(x[:n]) - target

        constraints = [{"type": "eq", "fun": sum_eq}]
        for kk in range(-n + 1, n):
            if kk == 0:
                continue

            def ineq(x, shift=kk):
                h = x[:n]
                return x[-1] - overlap_shift(h, shift) * dx

            constraints.append({"type": "ineq", "fun": ineq})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},

            if res.success:
                h_opt = fix_sum(np.clip(res.x[:n], 0.0, 1.0), target)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt.copy()
        except Exception:
            pass

    # final safety
    best_h = fix_sum(np.clip(best_h, 0.0, 1.0), n / 2.0)
    best_val = compute_c5(best_h)
    return best_h, best_val, n
