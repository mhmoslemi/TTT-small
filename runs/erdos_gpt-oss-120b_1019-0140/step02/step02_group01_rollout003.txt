def shift_array(arr, k):
    if k > 0:
        # shift left by k: arr[k:] at positions 0:n-k, zeros at end
        return np.concatenate([arr[k:], np.zeros(k)])
    elif k < 0:

        # shift right by kp: zeros at start, arr[:n-kp] at end
        return np.concatenate([np.zeros(kp), arr[:n - kp]])
    else:
        return arr.copy()

import time
import numpy as np

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given discretized h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def shift_array(arr, k):
    """Shift arr by k positions with zero padding (positive k → left shift)."""
    n = arr.shape[0]
    if k > 0:
        out = np.empty_like(arr)
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
    elif k < 0:
        kp = -k
        out = np.empty_like(arr)
        out[kp:] = arr[: n - kp]
        out[:kp] = 0.0
    else:
        out = arr.copy()
    return out


def project_to_sum_bound(v, target):
    """
    Euclidean projection of v onto the hyperplane sum(v)=target
    intersected with box [0,1]^n.
    Solves for lambda in min ||v - (proj)|| s.t. proj_i = clip(v_i - lambda,0,1).
    """
    n = v.shape[0]
    # lambda bounds: if lambda is very negative, all entries become 1; if very positive, all 0
    lo = -np.max(np.abs(v)) - 10.0
    hi = np.max(v) + 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lambda_star = (lo + hi) * 0.5
    return np.clip(v - lambda_star, 0.0, 1.0)


# ---------- Core optimization ----------
def gradient_descent(h0, target_sum, max_iter, step0, rng):
    """Projected gradient descent on the max-overlap objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)

    for it in range(1, max_iter + 1):
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        k = idx_max - (n - 1)

        # gradient of the active overlap term
        grad = 1.0 - shift_array(h, k) - shift_array(h, -k)

        # step size schedule
        step = step0 * (1.0 - it / max_iter)

        # gradient step + projection
        h_new = h - step * grad
        h_new = project_to_sum_bound(h_new, target_sum)

        # evaluate
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            if best_c5 < 0.3808:
                break

        h = h_new

    return best_h, best_c5


def local_refine(h_start, target_sum, iters, rng):
    """Simple pairwise mass‑transfer hill climbing."""
    h = h_start.copy()
    best_c5 = compute_c5(h)
    n = len(h)

    for _ in range(iters):
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            h = h_new
            best_c5 = c5_new
    return h, best_c5


def generate_initial_candidates(n, target_sum, rng):
    """Return a modest list of feasible starting vectors."""
    candidates = []

    # parent if available
    if "initial_h_values" in globals():
        if len(initial_h_values) == n:
            candidates.append(np.array(initial_h_values, dtype=float))

    # random uniform
    for _ in range(2):
        u = rng.random(n)
        u = project_to_sum_bound(u, target_sum)
        candidates.append(u)

    # random binary
    for _ in range(2):
        b = np.zeros(n)
        idx = rng.choice(n, target_sum, replace=False)
        b[idx] = 1.0
        candidates.append(b)

    # linear decreasing shape (as used previously)
    lin = np.linspace(1.0, 0.0, n)
    scaling = target_sum / np.sum(lin)
    lin = np.clip(lin * scaling, 0.0, 1.0)
    lin = project_to_sum_bound(lin, target_sum)
    candidates.append(lin)

    return candidates


def search_n(n, time_budget, rng):
    """Search for a good h at resolution n within allocated time."""
    target_sum = n // 2
    best_h = None
    best_c5 = np.inf
    start = time.time()

    candidates = generate_initial_candidates(n, target_sum, rng)

    for h0 in candidates:
        if time.time() - start > time_budget - 0.2:
            break

        # allocate iterations based on remaining budget
        remaining = time_budget - (time.time() - start)
        max_iter = int(min(8000, max(200, remaining * 30)))  # crude estimate

        h_opt, c5_opt = gradient_descent(h0, target_sum, max_iter, step0=0.5, rng=rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                return best_h, best_c5

        # short local refinement
        if time.time() - start < time_budget - 0.2:
            refine_iters = int(min(4000, (time_budget - (time.time() - start)) * 2000))
            h_ref, c5_ref = local_refine(best_h, target_sum, refine_iters, rng)
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref
                if best_c5 < 0.3808:
                    return best_h, best_c5

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) with a step function
    achieving a low upper bound for the Erdős C5 constant.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # candidate discretisations (even numbers, modest size)
    candidate_ns = [80, 96, 112, 128]

    # ensure parent size is considered first
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break

        remaining = budget_s - (time.time() - start_time)
        alloc = min(250.0, remaining * 0.5)  # cap per resolution
        h_opt, c5_opt = search_n(n, alloc, rng)

        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global = h_opt
            best_c5_global = c5_opt
            best_n_global = n
            if best_c5_global < 0.3808:
                break

    # fallback if nothing succeeded
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n

    # final feasibility correction
    n = best_n_global
    target_sum = n // 2
    best_h_global = np.clip(best_h_global, 0.0, 1.0)
    cur = best_h_global.sum()
    if abs(cur - target_sum) > 1e-9:
        best_h_global = project_to_sum_bound(best_h_global, target_sum)
        best_h_global = np.clip(best_h_global, 0.0, 1.0)

    return best_h_global, best_c5_global, n
