def gradient_descent_h(h_start, deadline, rng, n):
    dx = 2.0 / n
    target = n/2.0
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    max_iter = 3000
    # we could compute big now: allocate for loops
    for it in range(max_iter):
        if time.time() > deadline - 0.1:
            break
        # schedule for alpha and step
        t = it / max_iter
        # use alpha from 5 to 30
        alpha = 5.0 + t * 25.0
        # step from 0.3 to 0.03 linearly decreasing
        step = 0.3 - t * 0.27  # final step 0.03
        # compute overl using np.correlate
        w = 1.0 - h
        overl = np.correlate(h, w, mode='full') * dx  # length 2n-1
        # smooth max weights
        max_overl = np.max(overl)
        exp_vals = np.exp(alpha * (overl - max_overl))
        sum_exp = np.sum(exp_vals)
        weight = exp_vals / sum_exp

        # gradient
        grad = np.zeros_like(h)
        # vectorized loop across shift indexes
        # Predefine n_1 = n - 1
        n_1 = n - 1
        for idx, wk in enumerate(weight):
            shift = idx - n_1
            if wk == 0.0:
                continue
            if shift >= 0:
                # w term
                grad[:n-shift] += wk * w[shift:]
                # h term
                grad[shift:] -= wk * h[:n-shift]
            else:
                s = -shift
                grad[s:] += wk * w[:n-s]
                grad[:n-s] -= wk * h[s:]
        grad = grad * dx  # overall factor dx
        # gradient descent step
        v = h - step * grad
        # projection onto capped simplex
        h = project_onto_capped_simplex(v, target)
        # evaluate
        cur_val = compute_c5(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

def project_onto_capped_simplex(v, target):
    # v is a numpy array.
    # Solve x = argmin ||x - v||^2 s.t. 0 <= x_i <= 1, sum x_i = target
    # Use bisection on lambda.
    low = np.min(v) - 1.0  # ensures x_i = 1 for all i at low
    high = np.max(v)       # ensures x_i = 0 for all i at high
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    # final projection
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # due to numeric error may have sum slightly off; adjust by distributing residual
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff to entries not at bounds
        mask = (x > 0.0 + 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            # distribute equally
            x[mask] += diff / mask.sum()
            # clip again
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust all entries proportionally
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np

def compute_c5(h):
    """Discrete approximation of C5 = max_k ∫ h(x)(1-h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_onto_capped_simplex(v, target):
    """Euclidean projection of v onto {0<=x_i<=1, sum x_i = target}."""
    low = np.min(v) - 1.0      # ensures all entries become 1 at low
    high = np.max(v)           # ensures all entries become 0 at high
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def gradient_descent_h(h_start, deadline, rng, n):
    """Projected gradient descent on a smooth approximation of max overlap."""
    dx = 2.0 / n
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    max_iter = 2000
    n_1 = n - 1
    for it in range(max_iter):
        if time.time() > deadline - 0.1:
            break
        # annealing schedule for smoothness and step size
        t = it / max_iter
        alpha = 5.0 + 25.0 * t            # 5 → 30
        step = 0.4 - 0.38 * t             # 0.4 → 0.02
        w = 1.0 - h
        overl = np.correlate(h, w, mode="full") * dx
        max_ol = np.max(overl)
        exp_vals = np.exp(alpha * (overl - max_ol))
        weight = exp_vals / exp_vals.sum()
        grad = np.zeros_like(h)
        for idx, wk in enumerate(weight):
            if wk == 0.0:
                continue
            shift = idx - n_1
            if shift >= 0:
                grad[: n - shift] += wk * w[shift:]
                grad[shift:] -= wk * h[: n - shift]
            else:
                s = -shift
                grad[s:] += wk * w[: n - s]
                grad[: n - s] -= wk * h[s:]
        grad *= dx
        v = h - step * grad
        h = project_onto_capped_simplex(v, target)
        cur_val = compute_c5(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

def refine_hill_climb(h_init, cur_val, deadline, rng):
    """Simple hill‑climbing: random mass transfers that improve C5."""
    h = h_init.copy()
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        # choose source with mass and destination with room
        src_idxs = np.where(h > 1e-12)[0]
        dst_idxs = np.where(h < 1 - 1e-12)[0]
        if src_idxs.size == 0 or dst_idxs.size == 0:
            break
        i = rng.choice(src_idxs)
        j = rng.choice(dst_idxs)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < best_val:
            best_val = new_val
            best_h = h.copy()
            if best_val < 0.3808:
                break
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # ---- initialise h ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target = n / 2.0
    h0 = project_onto_capped_simplex(h0, target)
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    if best_val < 0.3808:
        return best_h.astype(float), best_val, n

    attempts = 0
    # main multistart loop
    while time.time() < deadline - 2.0:
        # generate a fresh start
        if attempts % 3 == 0:
            # binary start
            h_start = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h_start[idx] = 1.0
        else:
            # continuous random start projected onto the simplex
            h_start = rng.random(n)
            h_start = project_onto_capped_simplex(h_start, target)

        # gradient descent refinement
        h_opt, val_opt = gradient_descent_h(h_start, deadline, rng, n)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break
        attempts += 1

    # optional final hill‑climbing refinement if time permits
    if time.time() < deadline - 0.5:
        best_h, best_val = refine_hill_climb(best_h, best_val, deadline, rng)

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, target)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
