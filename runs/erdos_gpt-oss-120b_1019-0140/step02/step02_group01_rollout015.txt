import time
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation includes shift 0 (which yields 0 for binary/continuous)
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * dx

def _adjust_sum_exact(h, target):
    """Clip to [0,1] and then adjust to hit the exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) <= 1e-12:
        return h
    # distribute the remaining diff by adding to smallest entries or removing from largest
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        idx = np.argsort(-h)
        for i in idx:
            dec = min(diff, h[i])
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    return h

def _random_feasible_h(n, rng):
    """Generate a feasible h with values in [0,1] and sum = n/2."""
    M = n // 2
    # start from a binary vector with exactly M ones
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, M, replace=False)
    h[ones] = 1.0
    # add modest random noise and re‑project to feasibility
    noise_amp = 0.2
    h += (rng.random(n) - 0.5) * 2.0 * noise_amp
    h = _adjust_sum_exact(h, M)
    return h

def optimise_sa(n, time_budget, rng, init_h=None):
    """
    Simulated‑annealing / hill‑climbing for a single resolution `n`.
    Returns the best (h, c5) found within `time_budget` seconds.
    """
    M = n // 2
    # initialise
    if init_h is not None:
        h = np.array(init_h, dtype=np.float64)
        h = _adjust_sum_exact(h, M)
    else:
        h = _random_feasible_h(n, rng)

    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    start = time.time()
    iter_cnt = 0
    T0 = 0.005  # initial temperature
    max_no_improve = 5000
    no_improve = 0

    while True:
        # time guard
        if time.time() - start > time_budget - 0.01:
            break

        # pick two distinct indices i (receiver) and j (donor)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # ensure a positive transfer is possible
        if h[j] <= 0.0 or h[i] >= 1.0:
            continue
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0.0:
            continue

        # occasionally make a full transfer
        if rng.random() < 0.1:
            delta = max_delta
        else:
            delta = max_delta * rng.random() ** 2

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        new_c5 = compute_c5(h_new)
        delta_obj = new_c5 - cur_c5

        # temperature schedule (geometric decay)
        T = T0 * (0.999 ** iter_cnt)

        accept = False
        if delta_obj < -1e-12:
            accept = True
        elif T > 1e-12:
            prob = np.exp(-delta_obj / T)
            if rng.random() < prob:
                accept = True

        if accept:
            h = h_new
            cur_c5 = new_c5
            no_improve = 0
            if cur_c5 + 1e-12 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                if best_c5 < 0.3808:
                    break
        else:
            no_improve += 1

        iter_cnt += 1

        # restart if stuck
        if no_improve >= max_no_improve:
            h = _random_feasible_h(n, rng)
            cur_c5 = compute_c5(h)
            no_improve = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # candidate sample sizes (even only)
    candidate_ns = [80, 96, 112, 128, 144, 160, 192, 224, 256]
    candidate_ns = [n for n in candidate_ns if n <= 500]

        parent_n = len(initial_h_values)

    for n in candidate_ns:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5.0:
            break
        remaining = budget_s - elapsed
        # allocate at most 200 s per resolution, otherwise a fraction of the remainder
        alloc = min(200.0, remaining * 0.4)

        init_vec = None
        if parent_n is not None and n == parent_n:

        if c5_opt + 1e-12 < best_c5_global:

            best_h_global = h_opt

            if best_c5_global < 0.3808:

    if best_h_global is None:
        n = 80
        best_n_global = n
        best_h_global = np.full(n, 0.5, dtype=np.float64)

        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        M = n // 2
        cur_sum = np.sum(best_h_global)
        if abs(cur_sum - M) > 1e-9:
            best_h_global = _adjust_sum_exact(best_h_global, M)
