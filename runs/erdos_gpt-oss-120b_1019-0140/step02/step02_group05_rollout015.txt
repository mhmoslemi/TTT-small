def project_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # free indices where adjusting while staying inside bounds
    free = np.where((h > 0) & (h < 1))[0]
    # if diff small relative to free capacity
    for _ in range(2): # two passes
        if free.size > 0:
            inc = diff / free.size
            # Clip inc to not violate bounds for each free entry (ensuring after add still within [0,1]).
            # Might need per-element limiting.

            h[free] += inc
            # Clip again, then recompute diff and free
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            free = np.where((h > 0) & (h < 1))[0]

        if abs(diff) < 1e-12:

        # Add diff to first index that can accommodate

                delta = min(abs(diff), cap)
                h[i] += delta if diff > 0 else -delta
                diff += -delta if diff > 0 else delta
                if abs(diff) < 1e-12:
                    break
    return h

def swap_move(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    cand = h.copy()
    cand[i], cand[j] = cand[j], cand[i]
    # No sum change sign; stays same sum.
    return cand

def transfer_move(h, target, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    donor, recv = i, j
    max_tr = min(h[donor], 1.0 - h[recv])
    if max_tr <= 0:
        # try opposite direction
        max_tr = min(h[recv], 1.0 - h[donor])
        if max_tr <= 0:
            return h.copy()
        donor, recv = recv, donor
    delta = rng.random() * max_tr
    cand = h.copy()
    cand[donor] -= delta
    cand[recv] += delta
    # sum unchanged
    return cand

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum(h, target):
    """Clip to [0,1] and adjust to meet exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # distribute uniformly over free entries
    free = np.where((h > 0) & (h < 1))[0]
    if free.size:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
    # residual greedy fix
    for i in range(h.size):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def smooth_and_fix(h, target):
    """Gaussian‑type smoothing (kernel [0.25,0.5,0.25]) and projection."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    sm = project_sum(sm, target)
    return sm

def swap_move(h, rng):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    cand = h.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand

def transfer_move(h, rng):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    donor, recv = i, j
    max_tr = min(h[donor], 1.0 - h[recv])
    if max_tr <= 0:
        max_tr = min(h[recv], 1.0 - h[donor])
        if max_tr <= 0:
            return h.copy()
        donor, recv = recv, donor
    delta = rng.random() * max_tr
    cand = h.copy()
    cand[donor] -= delta
    cand[recv] += delta
    return cand

def gradient_move(h, target, lr):
    """Descend the overlap for the currently worst shift."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[: n - k] += 1.0 - h[k:]
        g[k:] -= h[: n - k]
    else:
        kk = -k
        g[kk:] += 1.0 - h[: n - kk]
        g[: n - kk] -= h[kk:]
    delta = -lr * g * dx
    max_abs = np.max(np.abs(delta))
    if max_abs > 0.2:
        delta = delta * (0.2 / max_abs)
    cand = h + delta
    cand = np.clip(cand, 0.0, 1.0)
    cand = project_sum(cand, target)
    return cand
