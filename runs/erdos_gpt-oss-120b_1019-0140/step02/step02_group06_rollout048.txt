def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    T0 = 0.01
    no_improve = 0
    iteration = 0

    while time.time() - start_time < budget_s:
        elapsed = time.time() - start_time
        T = max(1e-9, T0 * (1 - elapsed / budget_s))
        # compute offset at current solution to target the worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        offset = max_idx - (n - 1)

        # propose move
        if rng.random() < 0.7 and n > 1:  # targeted move probability
            # generate alignable list
            if offset >= 0:
                i_candidates = np.arange(0, n - offset)
            else:
                i_candidates = np.arange(-offset, n)
            if len(i_candidates) > 0:
                i = rng.choice(i_candidates)
                inc_i = i
                inc_j = i + offset
                # choose dec index randomly not in inc_i/j
                dec = rng.integers(0, n)
                # ensure dec not inc_i/j
                while dec == inc_i or dec == inc_j:
                    dec = rng.integers(0, n)
                max_delta = min(1.0 - cur_h[inc_i], 1.0 - cur_h[inc_j], cur_h[dec] / 2.0)
                if max_delta > 0:
                    delta = max_delta * rng.random() * 0.5
                    new_h = cur_h.copy()
                    new_h[inc_i] += delta
                    new_h[inc_j] += delta
                    new_h[dec] -= 2.0 * delta
                else:
                    # fallback to random move
                    new_h = None
            else:
                new_h = None
        else:
            new_h = None

        if new_h is None:
            # random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)

import time
import math
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum_and_bounds(x, target):
    # clip and adjust sum to target
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute diff uniformly
        x += diff / len(x)
        # clip again
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # if still imbalance, fix iteratively
        if abs(diff) > 1e-12:
            # adjust elementwise
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    # Determine initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
        target = n / 2.0
        h0 = project_sum_and_bounds(h0, target)
    else:
        n = 200
        target = n / 2.0
        h0 = rng.random(n)
        h0 = project_sum_and_bounds(h0, target)
    # Initialize state
    cur_h = h0.copy()
    cur_corr = compute_corr(cur_h)
    cur_c5 = np.max(cur_corr) * (2.0 / n)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    no_improve = 0
    T0 = 0.01
    iteration = 0
    # Main loop
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s:
            break
        # Temperature schedule
        T = max(1e-9, T0 * (1.0 - elapsed / budget_s))
        # Determine worst shift offset
        max_idx = np.argmax(cur_corr)
        offset = max_idx - (n - 1)
        # Propose move
        new_h = None
        # Targeted move with probability 0.7
        if rng.random() < 0.7:
            # Alignable indices for offset
            if offset >= 0:
                i_candidates = np.arange(0, n - offset)
            else:
                i_candidates = np.arange(-offset, n)
            if i_candidates.size > 0:
                i = int(rng.choice(i_candidates))
                inc_i = i
                inc_j = i + offset
                # choose a dec index distinct
                dec = int(rng.integers(0, n))
                while dec == inc_i or dec == inc_j:
                    dec = int(rng.integers(0, n))
                max_delta = min(1.0 - cur_h[inc_i], 1.0 - cur_h[inc_j], cur_h[dec] / 2.0)
                if max_delta > 0:
                    delta = max_delta * rng.random() * 0.5
                    new_h = cur_h.copy()
                    new_h[inc_i] += delta
                    new_h[inc_j] += delta
                    new_h[dec] -= 2.0 * delta
        if new_h is None:
            # Random pair move
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            while j == i:
                j = int(rng.integers(0, n))
            max_delta = min(1.0 - cur_h[i], cur_h[j])
            if max_delta <= 0:
                # cannot move, skip this iteration
                continue
            delta = max_delta * rng.random() * 0.5
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        # Ensure feasibility (clip and project if needed)
        # Since delta respects bounds, minimal adjustment needed; still clip
        new_h = np.clip(new_h, 0.0, 1.0)
        # Minor sum correction
        cur_sum = new_h.sum()
        diff = target - cur_sum
        if abs(diff) > 1e-12:
            new_h = project_sum_and_bounds(new_h, target)
        new_corr = compute_corr(new_h)
        new_c5 = np.max(new_corr) * (2.0 / n)
        # Acceptance
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur_h = new_h
            cur_corr = new_corr
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = new_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # Stagnation handling
        if no_improve > 5000:
            # Reinitialize with perturbation around best
            cur_h = best_h.copy()
            perturb = rng.normal(scale=0.02, size=n)
            cur_h = cur_h + perturb
            cur_h = project_sum_and_bounds(cur_h, target)
            cur_corr = compute_corr(cur_h)
            cur_c5 = np.max(cur_corr) * (2.0 / n)
            no_improve = 0
        # Periodic re-projection for numeric drift
        if iteration % 2000 == 0:
            cur_h = project_sum_and_bounds(cur_h, target)
            cur_corr = compute_corr(cur_h)
            cur_c5 = np.max(cur_corr) * (2.0 / n)
        iteration += 1
    # Final projection and compute final bound
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n

import time
import math
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def compute_corr(h):
    """Full correlation vector without dx scaling."""
    return np.correlate(h, 1.0 - h, mode="full")

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and adjust to hit exact sum `target`."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute uniformly
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # fix remaining imbalance elementwise
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x
