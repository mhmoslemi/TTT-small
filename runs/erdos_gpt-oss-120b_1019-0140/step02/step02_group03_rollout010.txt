def run(...):
    import time, numpy as np, scipy.optimize as opt

    def compute_c5, project_to_box_sum, SumConstraint, OverlapConstraint as before.

    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 0: Random sampling
    max_phase0_time = budget_s * 0.1  # 10%
    while time.time() - start_time < max_phase0_time:
        # generate candidate
        cand = rng.random(n)
        cand = project_to_box_sum(cand, target_sum)
        c5 = compute_c5(cand)
        if c5 + 1e-12 < best_c5:
            best_h = cand
            best_c5 = c5

    # Phase 1: SLSQP restarts
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    phase1_deadline = start_time + budget_s * 0.6
    while time.time() < phase1_deadline:
        # start from random feasible
        h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, target_sum)
        t_start = compute_c5(h_start)
        # run optimization
        res = opt.minimize(fun=lambda x: x[-1],
                           x0=np.concatenate([h_start, [t_start]]),
                           method='SLSQP',
                           bounds=[(0.,1.)]*n+[(0.,None)],
                           constraints=[{'type':'eq','fun':sum_con}] + [{'type':'ineq','fun':oc} for oc in overlap_cons],
                           options={'maxiter':400, 'ftol':1e-9, 'disp':False})
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0., 1.)
        # enforce sum exactly
        h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            if best_c5 < 0.3806:
                break

    # Phase 2: Simulated annealing
    cur_h = best_h.copy()
    cur_c5 = best_c5
    # SA parameters:
    T0 = 0.005
    Tf = 1e-6
    total_sa_time = budget_s - (time.time() - start_time) - 0.5
    # adapt while time left.
    while time.time() - start_time < budget_s - 0.5:
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        T = (1 - frac) * T0 + frac * Tf
        # generate candidate: random direction with zero mean
        d = rng.normal(size=n)
        d -= d.mean()
        step_scale = 0.5 * (rng.random() ** 2) # may be adjustable
        cand = cur_h + step_scale * d
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)
        diff = cand_c5 - cur_c5
        if diff < 0 or rng.random() < np.exp(-diff / T):
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 + 1e-12 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
        # occasional low step shrink?
        # we could also occasionally reset to best_h and add random perturbation
        # If cur_c5 is far above best, we can restart.
        if rng.random() < 0.001:
            # restart from best with random perturbation
            cur_h = best_h.copy()
            cur_c5 = best_c5
    # Final polish local optimization
    # Run a final SLSQP from best_h with more iterations
    final_res = opt.minimize(fun=lambda x: x[-1],
                             x0=np.concatenate([best_h, [best_c5]]),
                             method='SLSQP',
                             bounds=[(0.,1.)]*n+[(0.,None)],
                             constraints=[{'type':'eq','fun':sum_con}] + [{'type':'ineq','fun':oc} for oc in overlap_cons],
                             options={'maxiter':800, 'ftol':1e-9, 'disp':False})
    if final_res.success:
        h_final = np.clip(final_res.x[:n], 0., 1.)
        h_final = project_to_box_sum(h_final, target_sum)
        c5_final = compute_c5(h_final)
        if c5_final + 1e-12 < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Ensure final feasiblity
    best_h = np.clip(best_h, 0., 1.)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo, hi]^n while enforcing sum(v)=target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
