def simulated_annealing_binary(h_bin, dx, start_temp, min_temp, alpha, max_iter, time_limit, rng):
    cur_h = h_bin.copy()
    cur_val = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    temp = start_temp
    iter = 0
    while time.time() < time_limit and iter < max_iter:
        # randomly choose one swap
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cand_h = cur_h.copy()
        cand_h[i] = 0
        cand_h[j] = 1
        cand_val = compute_c5(cand_h, dx)
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_h = cand_h
            cur_val = cand_val
            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val
        # update temperature
        temp = max(min_temp, temp * alpha)
        iter += 1
    return best_h, best_val
```

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum(vec, target_sum):
    # Clip and adjust sum to target
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    # fix any remaining diff greedily
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec

def simulated_annealing_binary(h_bin, dx, start_temp, min_temp, alpha, max_iter, time_limit, rng):
    cur_h = h_bin.copy()
    cur_val = compute_c5(cur_h.astype(float), dx)
    best_h = cur_h.copy()
    best_val = cur_val
    temp = start_temp
    for _ in range(max_iter):
        if time.time() >= time_limit:
            break
        # select random swap
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cand_h = cur_h.copy()
        cand_h[i] = 0
        cand_h[j] = 1
        cand_val = compute_c5(cand_h.astype(float), dx)
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_h = cand_h
            cur_val = cand_val
            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val
        temp = max(min_temp, temp * alpha)
    return best_h, best_val

def refine_continuous(h, dx, target_sum, time_limit, start_time, rng):

    cur_val = compute_c5(cur_h, dx)
    step = 0.05
    n = len(h)

        else:

    cur_val = compute_c5(cur_h, dx)
    return cur_h, cur_val

import time
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral C5 for vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target_sum):
    """Clip to [0,1] and adjust sum to target_sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec

def simulated_annealing_binary(h_bin, dx, start_temp, min_temp, alpha, max_iter, time_limit, rng):
    """Binary simulated annealing with swaps preserving weight."""
    cur_h = h_bin.copy()
    cur_val = compute_c5(cur_h.astype(float), dx)
    best_h = cur_h.copy()
    best_val = cur_val
    temp = start_temp
    for _ in range(max_iter):
        if time.time() >= time_limit:
            break
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand.astype(float), dx)
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_h = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val
        temp = max(min_temp, temp * alpha)
    return best_h, best_val

def refine_continuous(h, dx, target_sum, time_limit, start_time, rng):
    """Local continuous refinement moving mass between entries."""
    cur_h = h.copy()
    cur_val = compute_c5(cur_h, dx)
    step = 0.05
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5(cur_h, dx)
    return cur_h, cur_val
