def project_sum_and_bounds(x, target, tol=1e-12):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) <= tol:
        return x
    if diff > 0:
        for i in range(len(x)):
            cap = 1.0 - x[i]
            if cap <= 0:
                continue
            inc = min(diff, cap)
            x[i] += inc
            diff -= inc
            if diff <= tol:
                break
    else:
        diff_abs = -diff
        for i in range(len(x)):
            cap = x[i]
            if cap <= 0:
                continue
            dec = min(diff_abs, cap)
            x[i] -= dec
            diff_abs -= dec
            if diff_abs <= tol:
                break
        diff = -diff_abs
    # after this diff should be <= tol; else we may need to adjust but improbable.
    return x
```

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral multiplied by the grid spacing."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_sum_and_bounds(x, target, tol=1e-12):
    """Clip to [0,1] and adjust so that sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) <= tol:
        return x
    n = x.shape[0]

    if diff > 0:                # need to increase total mass
        for i in range(n):
            cap = 1.0 - x[i]
            if cap <= 0.0:
                continue
            inc = diff if diff < cap else cap
            x[i] += inc
            diff -= inc
            if diff <= tol:
                break
    else:                       # need to decrease total mass
        diff_abs = -diff
        for i in range(n):
            cap = x[i]
            if cap <= 0.0:
                continue
            dec = diff_abs if diff_abs < cap else cap
            x[i] -= dec
            diff_abs -= dec
            if diff_abs <= tol:
                break
        diff = -diff_abs

    # tiny leftover adjustment (should be negligible)
    if abs(diff) > tol:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x


def de_optimize(initial_h, rng, time_budget):
    """Differential Evolution to minimize C5."""
    n = len(initial_h)
    target = n / 2.0
    pop_size = 40
    F = 0.6
    CR = 0.9

    # ----- population init -----
    pop = np.empty((pop_size, n), dtype=float)
    pop[0] = project_sum_and_bounds(initial_h.copy(), target)
    for i in range(1, pop_size):
        cand = rng.random(n)
        pop[i] = project_sum_and_bounds(cand, target)

    fitness = np.array([compute_c5(ind) for ind in pop])
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c5 = float(fitness[best_idx])

    start = time.time()
    gen = 0
    last_improve_gen = 0

    # ----- main DE loop -----
    while time.time() - start < time_budget:
        gen += 1
        for i in range(pop_size):
            # pick three distinct indices not equal to i
            idxs = rng.choice(pop_size, size=3, replace=False)
            while i in idxs:
                idxs = rng.choice(pop_size, size=3, replace=False)
            a, b, c = idxs
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = project_sum_and_bounds(mutant, target)

            # binomial crossover
            jrand = rng.integers(0, n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_sum_and_bounds(trial, target)

            f = compute_c5(trial)
            if f <= fitness[i]:
                pop[i] = trial
                fitness[i] = f
                if f < best_c5 - 1e-12:
                    best_c5 = f
                    best_h = trial.copy()
                    last_improve_gen = gen

        # stop if stagnated for many generations
        if gen - last_improve_gen > 300:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Return a feasible h, its C5 bound, and the number of samples."""
    start = time.time()
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 80
        h0 = rng.random(n)

    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    remaining = max(0.0, budget_s - (time.time() - start))
    best_h, best_c5 = de_optimize(h0, rng, remaining)

    best_h = project_sum_and_bounds(np.clip(best_h, 0.0, 1.0), target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
