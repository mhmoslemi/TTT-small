    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        # add diff uniformly
        x = x + diff / len(x)
        # clip again
        x = np.clip(x, 0.0, 1.0)
        # redo residual
        diff2 = target_sum - np.sum(x)
        if abs(diff2) > 1e-12:
            # adjust only indices not at bound
            # get mask of not at bound
            not_at_bound = (x > 1e-12) & (x < 1-1e-12)
            if np.any(not_at_bound):
                # distribute residual equally among free indices
                idx = np.where(not_at_bound)[0]

                # clip again

def repair(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    diff = target_sum - s
    if abs(diff) > 1e-12:
        # distribute diff uniformly
        x = x + diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        # update residual
        residual = target_sum - np.sum(x)
        if abs(residual) > 1e-12:
            # adjust free indices
            free = (x > 1e-12) & (x < 1-1e-12)
            if np.any(free):
                x[free] = x[free] + residual / np.count_nonzero(free)
                x = np.clip(x, 0.0, 1.0)
            else:
                # fall back: adjust one index
                i = np.argmin(np.abs(x - 0.5))
                low = -x[i]
                high = 1 - x[i]
                adj = max(min(residual, high), low)
                x[i] += adj
    return x

def continuous_anneal(start_h, end_time, rng, target_sum):
    n = len(start_h)
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 1e-2
    final_temp = 1e-6
    max_iter = 200000
    it = 0
    while time.time() < end_time and it < max_iter:
        # pick two indices to adjust
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # propose delta
        max_inc = min(1 - h[i], h[j])
        if max_inc <= 0:
            continue
        # scale by temperature maybe
        delta = max_inc * rng.random() * (init_temp * (1 - it/max_iter) + final_temp * (it/max_iter))
        # alternative: random delta independent of temperature
        # We'll use random scaling 0..max_inc.
        delta = max_inc * rng.random()
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # repair sum (though sum remains unchanged)
        # Actually we maintain sum exactly because we add delta to i, subtract from j.
        # So sum unchanged.
        # Ensure bounds:
        if h_new[i] > 1 or h_new[j] < 0:
            continue
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        # Temperature decays linearly.
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < np.exp(-dE / (T + 1e-12)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # final safety repair: enforce sum
    best_h = repair(best_h, target_sum)
    return best_h, best_c5

    """Approximate C5 via discrete correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_sum(x, target):
    """Project x onto [0,1] box and enforce sum == target (within tolerance)."""
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    diff = target - s
    if abs(diff) > 1e-12:
        x = x + diff / x.size
        x = np.clip(x, 0.0, 1.0)
        residual = target - np.sum(x)
        if abs(residual) > 1e-12:
            free = (x > 1e-12) & (x < 1.0 - 1e-12)
            if np.any(free):
                x[free] = x[free] + residual / np.count_nonzero(free)
                x = np.clip(x, 0.0, 1.0)
            else:
                i = np.argmin(np.abs(x - 0.5))
                low = -x[i]
                high = 1.0 - x[i]
                adj = max(min(residual, high), low)
                x[i] += adj
    return x

def continuous_anneal(start_h, end_time, rng, target):
    """Simple annealing on continuous variables preserving total sum."""
    n = start_h.size
    h = start_h.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 200000
    it = 0
    while time.time() < end_time and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0.0:
            it += 1
            continue
        # propose delta
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # sum unchanged, bounds respected by construction
        new = compute_c5(h_new)
        dE = new - cur
        T = init_temp * (1.0 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0.0 or rng.random() < np.exp(-dE / (T + 1e-18)):
            h = h_new
            cur = new
            if new < best:
                best = new
                best_h = h_new.copy()
        it += 1
    best_h = repair_sum(best_h, target)
    return best_h, best

def binary_hill_climb(start_bin, end_time, rng):
    """Swap‑based hill climbing on binary vectors."""
    n = start_bin.size
    h = start_bin.astype(float)
    best = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    while time.time() < end_time and no_improve < 3000:
        ones = np.flatnonzero(start_bin)
        zeros = np.flatnonzero(1 - start_bin)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = start_bin.copy()
        cand[i] = 0
        cand[j] = 1
        cand_h = cand.astype(float)
        cand_val = compute_c5(cand_h)
        if cand_val + 1e-12 < best:
            best = cand_val
            best_h = cand_h.copy()
            start_bin = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.uniform(size=n)
    target_sum = n / 2.0
    h0 = repair_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # --------------------------------------------------------------
    # Global search with differential evolution
    # --------------------------------------------------------------
    bounds = [(0.0, 1.0)] * n
    popsize = 5  # population = popsize * n = 5*n individuals
    # build initial population (first individual = current best)
    pop = np.empty((popsize * n, n))
    pop[0] = best_h
    for i in range(1, popsize * n):
        pop[i] = rng.uniform(size=n)

    # time‑aware callback
    target_val = 0.3808
    cb_state = {'best_val': best_c5, 'best_x': best_h.copy()}

    def de_callback(xk, convergence):
        val = compute_c5(repair_sum(xk, target_sum))
        if val < cb_state['best_val']:
            cb_state['best_val'] = val
            cb_state['best_x'] = xk.copy()
        if val <= target_val:
            return True
        if time.time() - start_time > budget_s - 0.5:
            return True
        return False

    # run DE (allow early stop via callback)
    try:
        result = differential_evolution(
            lambda x: compute_c5(repair_sum(x, target_sum)),
            bounds,
            strategy='best1bin',
            maxiter=200,
            popsize=popsize,
            init=pop,
            updating='deferred',
            polish=False,
            seed=seed,
            callback=de_callback,
            disp=False,
        )
    except Exception:
        result = None

    # pick the best solution found by DE or previous best
    if cb_state['best_val'] < best_c5:
        best_h = repair_sum(cb_state['best_x'], target_sum)
        best_c5 = cb_state['best_val']

    # --------------------------------------------------------------
    # Local continuous refinement (simulated annealing)
    # --------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time) - 0.3
    if remaining > 5.0:
        end = time.time() + remaining * 0.6
        refined_h, refined_c5 = continuous_anneal(best_h, end, rng, target_sum)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # --------------------------------------------------------------
    # Final binary hill‑climb from rounded version
    # --------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time) - 0.2
    if remaining > 1.0:
        end = time.time() + remaining
        bin_start = (best_h >= 0.5).astype(int)
        final_h, final_c5 = binary_hill_climb(bin_start, end, rng)
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5

    # final safety repair (ensure exact sum)
    best_h = repair_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
