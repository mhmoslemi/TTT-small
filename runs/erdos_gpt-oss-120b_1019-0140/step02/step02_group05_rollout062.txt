def enforce_sum_and_bounds(h, target_sum, rng):

    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # free indices
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        # residual
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) > 1e-12:
            # adjust iteratively
            # for positive diff, increase from smallest values
            # for negative diff, decrease from largest values
            order = np.argsort(h) if diff > 0 else np.argsort(-h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[idx] < 1.0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                elif diff < 0 and h[idx] > 0.0:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
        h = np.clip(h, 0.0, 1.0)

def compute_c5(h):
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum_and_bounds(h, target, rng):
    """Clip to [0,1] and adjust to hit the exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # first spread uniformly over free variables
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # residual fix – greedy over sorted indices
    if abs(diff) > 1e-12:
        order = np.argsort(h) if diff > 0 else np.argsort(-h)
        for i in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub

    cand = h.copy()
    cand[idx] += delta
    cand = enforce_sum_and_bounds(cand, target, rng)
    return cand

def worst_shifts_gradient(h, n, target, rng, num=3):
    """Gradient of the worst (largest) overlap, summed over `num` worst shifts."""
    dx = 2.0 / n
    O = np.correlate(h, 1.0 - h, mode="full")   # raw overlaps, no dx factor
    # indices of the largest overlaps
    if num >= O.size:
        idxs = np.arange(O.size)
    else:
        idxs = np.argpartition(O, -num)[-num:]
    grad = np.zeros_like(h)
    for s in idxs:
        k = s - (n - 1)   # shift corresponding to index s
        if k >= 0:
            grad[: n - k] += dx * (1.0 - h[k:])
            grad[k:]     += dx * (1.0 - h[: n - k])
        else:
            kk = -k
            grad[kk:]    += dx * (1.0 - h[: n - kk])
            grad[: n - kk] += dx * (1.0 - h[kk:])
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.1
    rng = np.random.default_rng(seed)

    # ----- initial solution -----------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = enforce_sum_and_bounds(h, n / 2.0, rng)
    else:
        n = 80
        h = rng.random(n)
        h = enforce_sum_and_bounds(h, n / 2.0, rng)

    target_sum = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    # ----- simulated annealing --------------------------------------
    T0 = 0.2
    probs = np.cumsum([0.25, 0.25, 0.20, 0.30])   # swap, transfer, perturb, gradient
    while True:
        now = time.time()
        if now >= deadline * 0.85:
            break
        elapsed = now - start
        T = T0 * max(0.0, 1.0 - elapsed / (budget_s * 0.85))
        r = rng.random()
        if r < probs[0]:
            cand = move_swap(cur_h, rng)
        elif r < probs[1]:
            cand = move_transfer(cur_h, rng)
        elif r < probs[2]:
            cand = move_perturb(cur_h, rng, target_sum)
        else:
            grad = worst_shifts_gradient(cur_h, n, target_sum, rng, num=3)
            m = np.max(np.abs(grad))
            if m < 1e-12:
                continue
            step = -grad / m * 0.2
            cand = cur_h + step
            cand = enforce_sum_and_bounds(cand, target_sum, rng)

        cand_val = compute_c5(cand)
        accept = False
        if cand_val < cur_val:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(cand_val - cur_val) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_h = cand_val, cand.copy()
                if best_val < 0.3808:
                    break

    # ----- optional smoothing passes ---------------------------------
    if time.time() < deadline - 5.0:
        for _ in range(8):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val, best_h = sm_val, sm
            else:
                break

    # ----- focused gradient descent ---------------------------------
    if time.time() < deadline - 5.0:
        cur_h, cur_val = best_h.copy(), best_val
        step_max = 0.2
        for _ in range(3000):
            if time.time() >= deadline - 5.0:
                break
            grad = worst_shifts_gradient(cur_h, n, target_sum, rng, num=3)
            m = np.max(np.abs(grad))
            if m < 1e-12:
                break
            step = -grad / m * step_max
            cand = cur_h + step
            cand = enforce_sum_and_bounds(cand, target_sum, rng)
            cand_val = compute_c5(cand)
            if cand_val < cur_val:
                cur_h, cur_val = cand, cand_val
                if cand_val < best_val:
                    best_val, best_h = cand_val, cand.copy()
            step_max *= 0.9995

    # ----- final SLSQP refinement ------------------------------------
    if time.time() < deadline - 2.0:
        dx = 2.0 / n
        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

        def make_ineq(k):
            def fun(x):
                hh = x[:n]
                if k >= 0:
                    over = np.dot(hh[: n - k], 1.0 - hh[k:]) * dx
                else:
                    kk = -k
                    over = np.dot(hh[kk:], 1.0 - hh[: n - kk]) * dx
                return x[-1] - over
            return fun

        cons = [{"type": "eq", "fun": sum_eq}]
        for kk in range(-n + 1, n):
            if kk == 0:
                continue
            cons.append({"type": "ineq", "fun": make_ineq(kk)})
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 600, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = enforce_sum_and_bounds(h_opt, target_sum, rng)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val, best_h = val_opt, h_opt
        except Exception:
            pass

    # final safety correction
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_and_bounds(best_h, target_sum, rng)

    return best_h, best_val, n
