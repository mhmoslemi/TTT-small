def project_onto_box_and_sum(v, target):
    # Find lambda s.t. sum(clip(v + lambda,0,1)) = target
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    # ensure sum at lo <= target, sum at hi >= target
    # compute sum at lo: all clipped to 0 -> sum=0
    # compute sum at hi: all clipped to 1 -> sum=len(v)
    # target between 0 and n
    # Bisection
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    return proj
```

def compute_c5(h):

    n = h.shape[0]

    """
    lo = -np.max(v) - 1.0  # ensures all v+lo <= 0  -> sum=0
    hi = 1.0 - np.min(v) + 1.0  # ensures all v+hi >= 1 -> sum=n
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)

def overlap_gradient(h, k):
    """
    Gradient of f_k(h) = sum_i h_i (1 - h_{i+k}) w.r.t. h.
    Includes the constant term 1; projection later removes the sum component.
    """
    n = h.shape[0]
    grad = np.ones(n, dtype=float)
    if k >= 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] -= h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        if abs(h0.sum() - target) > 1e-12:
            h0 = project_onto_box_and_sum(h0, target)
    else:
        n = 80
        target = n / 2.0
        v = rng.random(n)
        v = v / v.sum() * target
        v = np.clip(v, 0.0, 1.0)
        h0 = project_onto_box_and_sum(v, target)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # parameters for subgradient descent
    step0 = 0.6
    max_iters_per_run = 12000

    while time.time() < deadline - 0.5:
        # choose start point: either refine best or random
        if rng.random() < 0.35:
            h = best_h.copy()
        else:
            v = rng.random(n)
            v = v / v.sum() * target
            v = np.clip(v, 0.0, 1.0)
            h = project_onto_box_and_sum(v, target)

        for it in range(max_iters_per_run):
            # time check
            if time.time() > deadline - 0.2:
                break

            corr = np.correlate(h, 1.0 - h, mode="full")
            max_idx = np.argmax(corr)
            cur_val = corr[max_idx] * (2.0 / n)

            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break

            k = max_idx - (n - 1)          # shift corresponding to max overlap
            grad = overlap_gradient(h, k)   # gradient of the active overlap
            # normalize direction
            norm = np.linalg.norm(grad)
            if norm == 0:
                break
            step = step0 / np.sqrt(it + 1)
            h = h - step * grad / norm
            h = project_onto_box_and_sum(h, target)

            # occasional small random perturbation to escape flat regions
            if (it + 1) % 2500 == 0 and rng.random() < 0.4:
                noise = rng.normal(scale=0.015, size=n)
                h = project_onto_box_and_sum(h + noise, target)

        if best_val < 0.3808:
            break

    # final safety projection and evaluation
    best_h = project_onto_box_and_sum(best_h, target)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
