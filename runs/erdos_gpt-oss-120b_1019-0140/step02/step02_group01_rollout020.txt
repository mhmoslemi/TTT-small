def project_to_sum_bounds(h, M):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur < M:
        # need to add mass
        deficit = M - cur
        # distribute to indices with room (h < 1) proportionally or simply add to smallest first
        idx = np.argsort(h)
        for i in idx:
            inc = min(deficit, 1.0 - h[i])
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    elif cur > M:
        excess = cur - M
        idx = np.argsort(-h)
        for i in idx:
            dec = min(excess, h[i])
            h[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    return h
```

import time, math, numpy as np

def _project_to_sum_bounds(h, M):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur < M:
        deficit = M - cur
        idx = np.argsort(h)
        for i in idx:
            inc = min(deficit, 1.0 - h[i])
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    elif cur > M:
        excess = cur - M
        idx = np.argsort(-h)
        for i in idx:
            dec = min(excess, h[i])
            h[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    return h

def _random_h(n, rng):
    h = rng.random(n)
    return _project_to_sum_bounds(h, n / 2.0)

def _c5_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # decide base resolution
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = len(init)

            n0 += 1
            init = np.resize(init, n0)
        n_list = [n0]
    else:
        n_list = [80]

    # possibly try a larger resolution after the first one

    n_list.extend(extra_res)

    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in n_list:
        M = n / 2.0
        # initial h for this resolution
        if 'initial_h_values' in globals() and n == len(initial_h_values):
            h = np.array(initial_h_values, dtype=float)

        elif best_h is not None and n == best_n:

            h = best_h.copy()
        else:

            if best_h is not None and best_n is not None and best_n != n:

                xs_new = np.linspace(0, 2, n, endpoint=False)
                h = np.interp(xs_new, xs_old, best_h)
                h = _project_to_sum_bounds(h, M)
            else:

        cur_c5 = _c5_of(h)
        if cur_c5 < best_c5:

        # simulated annealing parameters

        max_no_imp = 5000

        while True:

            if elapsed > budget_s - 0.5:
