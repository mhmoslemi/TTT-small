import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target via a scalar shift."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# ---------- Constraint classes for SLSQP ----------
class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_refine(h, t_start=None, maxiter=800):
    """One-shot SLSQP refinement of a feasible vector."""
    n = h.shape[0]
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    if t_start is None:
        t_start = compute_c5(h)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': sum_con}] + [{'type': 'ineq', 'fun': oc} for oc in overlap_cons],
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, n / 2.0)
        return h_opt, compute_c5(h_opt)
    return h, compute_c5(h)

# ---------- Gradient descent on the worst overlap ----------
def gradient_step(h, step, target_sum):
    """Take a projected gradient step reducing the worst overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation to find worst shift
    corr = np.correlate(h, 1 - h, mode='full')
    O = corr * dx
    max_idx = np.argmax(O)
    shift = max_idx - (n - 1)
    if shift == 0:
        # pick next worst if zero shift dominates (should not happen)
        sorted_idx = np.argsort(O)[::-1]
        for idx in sorted_idx:
            if idx != n - 1:
                max_idx = idx
                shift = idx - (n - 1)
                break
    k = abs(shift)
    if k == 0:
        return h  # nothing to do
    # gradient of O_k w.r.t h
    forward = np.concatenate([h[k:], np.zeros(k)])
    backward = np.concatenate([np.zeros(k), h[:n - k]])
    grad = -dx * (forward + backward)
    cand = h - step * grad
    cand = project_to_box_sum(cand, target_sum)
    return cand

# ---------- Main entry point ----------
