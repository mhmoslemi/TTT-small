def hill_climb_binary(h, max_iters, time_left):

    best_score = compute_c5(best_h)
    n = len(h)
    rng = np.random.default_rng()

def simulated_anneal_binary(h, max_iters, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = len(h)
    rng = np.random.default_rng()
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # temperature schedule
        T = max(0.001, (max_iters - it) / max_iters) * 0.02  # adjust
        # swap
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = cur_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_score = h_new, new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, cur_h.copy()
    return best_h, best_score
```

def anneal_improve(h, max_iters, time_left):
    best_h = h
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = h.shape[0]
    rng = np.random.default_rng()
    # Precompute indices of ones and zeros each iter for speed
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # temperature schedule: linear from T_start to T_end
        T_start = 0.02
        T_end = 1e-4
        T = T_start + (T_end - T_start) * (it / max_iters)
        # pick random swap
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = cur_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_score = h_new, new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, cur_h.copy()
    return best_h, best_score

    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # set resolution (choose 512 or 256 based on time)
    n = 512

    # Prepare candidates list
    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # Upsample using repeat if factor integer else interpolation
        if n % init.size == 0:
            cand = np.repeat(init, n // init.size)
        else:
            cand = upsample(init, n)
        # Convert to binary by picking top n/2 values
        # Use Argpartition for speed
        idx_sel = np.argpartition(-cand, n // 2)[:n // 2]
        cand_bin = np.zeros_like(cand)
        cand_bin[idx_sel] = 1.0
        candidates.append(cand_bin)

    n = h.shape[0]

    ones_set = set(np.flatnonzero(h > 0.5).tolist())
    zeros_set = set(np.flatnonzero(h < 0.5).tolist())
    # Temperature schedule parameters
    T_start = 0.02
    T_end = 1e-5
    for it in range(1, max_iters + 1):
        if time_left_func() <= 0:
            break
        # linear temperature decay
        T = T_start + (T_end - T_start) * (it / max_iters)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Exact C5 using np.correlate as required by the evaluation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def anneal_improve(h0, max_iters, time_left_func):
    """Simulated‑annealing hill‑climb on binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    ones_set = set(np.flatnonzero(h > 0.5).tolist())
    zeros_set = set(np.flatnonzero(h < 0.5).tolist())

    T_start = 0.02
    T_end = 1e-5

    for it in range(1, max_iters + 1):
        if time_left_func() <= 0:
            break
        # linear temperature decay
        T = T_start + (T_end - T_start) * (it / max_iters)

        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        ones_set.remove(i)
        zeros_set.add(i)
        zeros_set.remove(j)
        ones_set.add(j)

        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
            if best_score <= 0.3808:
                break
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones_set.add(i)
            zeros_set.remove(i)
            zeros_set.add(j)
            ones_set.remove(j)

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # target resolution (feel free to adjust if time runs low)
    n = 512
    half_n = n // 2

    candidates = []

    # 1) upsampled parent if available
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if n % init.size == 0:

        else:
            cand = upsample(init, n)
        # binarise by keeping the half_n largest entries
        idx_sel = np.argpartition(-cand, half_n)[:half_n]
        bin_cand = np.zeros_like(cand)
        bin_cand[idx_sel] = 1.0
        candidates.append(bin_cand)

    # 2) Thue‑Morse binary
    candidates.append(thue_morse(n))

    # 3) a few random balanced starts
    for _ in range(3):
        if time_left() < 1.0:

        candidates.append(random_binary_balanced(n, rng))

    # evaluate initial candidates
    best_h = None
    best_score = np.inf
    for cand in candidates:
        sc = compute_c5(cand)
        if sc < best_score:
            best_score, best_h = sc, cand.copy()
        if time_left() < 0.5:
            break

    # -----------------------------------------------------------
    # Main annealing phase on the best candidate so far
    # -----------------------------------------------------------
    remaining = time_left()
    if best_h is not None and remaining > 5.0:
        # roughly 800 swaps per second on this hardware
        est_iters = int(remaining * 800)
        if est_iters < 2000:
            est_iters = int(remaining * 2000)
        best_h, best_score = anneal_improve(best_h, est_iters, time_left)

    # -----------------------------------------------------------
    # Additional random restarts if time permits
    # -----------------------------------------------------------
    while time_left() > 5.0:
        cand = random_binary_balanced(n, rng)
        # quick anneal with modest budget
        iters = min(3000, int(time_left() * 500))
        cand_h, cand_sc = anneal_improve(cand, iters, time_left)
        if cand_sc < best_score:
            best_score, best_h = cand_sc, cand_h.copy()
        if best_score <= 0.3808:
            break

    # final feasibility guard
    best_h = project_onto_box_sum(best_h, half_n)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
