import time
import numpy as np

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def enforce_sum(h, target, rng):
    """Adjust h in [0,1] so that sum(h) == target (exact up to floating tolerance)."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # uniform shift then clipping
    h = np.clip(h + diff / len(h), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idxs = np.arange(len(h))
        rng.shuffle(idxs)
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
    return h

def worst_shift_info(h):
    """Return (i,k) where k is the shift attaining max overlap and i contributes maximally."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    if k >= 0:
        idx1 = np.arange(0, n - k)
        idx2 = idx1 + k
    else:
        idx1 = np.arange(-k, n)
        idx2 = idx1 + k
    contrib = h[idx1] * (1.0 - h[idx2])
    i_local = np.argmax(contrib)
    i = idx1[i_local]
    return i, k

def swap_move(h, rng):
    """Swap a near‑1 entry with a near‑0 entry."""
    high = np.where(h > 0.9)[0]
    low = np.where(h < 0.1)[0]
    if high.size == 0 or low.size == 0:
        return None
    i = rng.choice(high)
    j = rng.choice(low)
    h2 = h.copy()
    h2[i], h2[j] = h2[j], h2[i]
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 80
        h = random_binary_vector(n, rng)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = enforce_sum(h, target, rng)

    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    indices = np.arange(n)
    no_improve = 0
    initial_T = 0.05  # initial temperature for annealing

    # ----- main stochastic search -----
    while time.time() < deadline - 0.1:
        # decide move type
        move_type = rng.random()
        if move_type < 0.6:  # try a swap
            h_new = swap_move(h, rng)
            if h_new is None:
                # fallback to a transfer move
                i, _ = worst_shift_info(h)
                feasible = np.where((h < 1.0 - 1e-12) & (indices != i))[0]
                if feasible.size == 0:
                    continue
                j = rng.choice(feasible)
                max_transfer = min(h[i], 1.0 - h[j])
                if max_transfer <= 0:
                    continue
                delta = max_transfer if rng.random() < 0.2 else max_transfer * rng.random() * 0.5
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
        else:  # transfer move guided by worst shift
            i, _ = worst_shift_info(h)
            feasible = np.where((h < 1.0 - 1e-12) & (indices != i))[0]
            if feasible.size == 0:
                continue
            j = rng.choice(feasible)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                continue
            delta = max_transfer if rng.random() < 0.2 else max_transfer * rng.random() * 0.5
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta

        new_val = compute_c5(h_new)

        # temperature schedule (linear cooling)
        elapsed = time.time() - start
        t_frac = elapsed / (budget_s - 0.1)
        T = initial_T * max(0.0, 1.0 - t_frac) + 1e-12

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / T):
            h = h_new
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h_new.copy()
                no_improve = 0
                if best_val < 0.3808:
                    break
            else:
                # accepted but not a new global best
                pass
        else:
            # reject move; keep current h
            pass

        # stagnation handling
        if cur_val >= best_val - 1e-12:
            no_improve += 1
        else:
            no_improve = 0

        if no_improve >= 5000:
            # random restart to escape flat region
            h = random_binary_vector(n, rng)
            h = enforce_sum(h, target, rng)
            cur_val = compute_c5(h)
            no_improve = 0

    # ----- final projection onto feasible set -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
