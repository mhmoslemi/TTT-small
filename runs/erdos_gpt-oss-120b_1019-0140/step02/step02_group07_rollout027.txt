def run(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np

    start = time.time()
    rng = np.random.default_rng(seed)

    # determine n
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 120  # choose even

    target_sum = n // 2

    # define compute_c5

    # define helper to fix binary sum
    def repair_binary(bin_arr):
        # bin_arr is np.ndarray dtype uint8 (0/1)
        s = bin_arr.sum()
        if s == target_sum:
            return bin_arr
        diff = target_sum - s
        # if diff>0 need to turn diff zeros to ones
        if diff > 0:
            zeros = np.where(bin_arr == 0)[0]
            if zeros.size < diff:
                # just set all zeros to 1; should not happen
                bin_arr[zeros] = 1
            else:
                idx = rng.choice(zeros, diff, replace=False)
                bin_arr[idx] = 1
        elif diff < 0:
            ones = np.where(bin_arr == 1)[0]
            if ones.size < -diff:
                bin_arr[ones] = 0
            else:
                idx = rng.choice(ones, -diff, replace=False)
                bin_arr[idx] = 0
        return bin_arr

    # define generator functions: thue_morse, rudin_shapiro
    def thue_morse_seq(length):
        # generate using binary parity of bits count
        seq = np.fromiter(((bin(i).count('1') & 1) for i in range(length)), dtype=np.uint8, count=length)
        return repair_binary(seq)

    def rudin_shapiro_seq(length):
        # produce +-1 using recursion - then map to binary
        # Use iterative building as before
        a = np.array([1], dtype=np.int8)
        while a.size < length:
            # a_{2k} = a_k; a_{2k+1} = a_k * (-1)^k
            # compute (-1)^k as 1 where k even else -1
            k = np.arange(a.size, dtype=np.int64)
            sign = np.where(k % 2 == 0, 1, -1)
            nxt = np.empty(a.size * 2, dtype=np.int8)
            nxt[0::2] = a
            nxt[1::2] = a * sign
            a = nxt
        seq = a[:length]
        # map +1 -> 1, -1 -> 0
        bin_seq = ((seq + 1) // 2).astype(np.uint8)
        return repair_binary(bin_seq)

    # Population
    pop_size = 100
    elite_size = 5
    pop = []
    # deterministic seeds
    deterministic_seeds = []
    deterministic_seeds.append(thue_morse_seq(n))
    deterministic_seeds.append(rudin_shapiro_seq(n))
    # also include random balanced
    for _ in range(pop_size - len(deterministic_seeds)):
        # generate random binary with target sum
        arr = np.zeros(n, dtype=np.uint8)
        arr[:target_sum] = 1
        rng.shuffle(arr)
        pop.append(arr)
    # add deterministic seeds to pop
    pop.extend(deterministic_seeds)
    # If we have extra beyond pop_size, truncate
    if len(pop) > pop_size:
        pop = pop[:pop_size]

    # compute fitness
    fitness = np.empty(len(pop), dtype=np.float64)
    for i, ind in enumerate(pop):
        fitness[i] = compute_c5(ind.astype(float))

    # track best
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].astype(float)
    best_c5 = fitness[best_idx]

    # main GA loop
    generation = 0
    # parameters
    mutation_rate = 0.02  # proportion of bits to swap per child
    # Use tournament size 2 and elitism
    while time.time() - start < budget_s - 0.5:
        generation += 1
        new_pop = []
        new_fitness = []
        # elitism: copy best elite_size individuals
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_pop.append(pop[idx].copy())
            new_fitness.append(fitness[idx])
        # fill rest
        while len(new_pop) < pop_size:
            # tournament selection
            contestants = rng.choice(len(pop), 2, replace=False)
            parent1 = pop[contestants[0]] if fitness[contestants[0]] <= fitness[contestants[1]] else pop[contestants[1]]
            # second parent selection
            contestants2 = rng.choice(len(pop), 2, replace=False)
            parent2 = pop[contestants2[0]] if fitness[contestants2[0]] <= fitness[contestants2[1]] else pop[contestants2[1]]
            # uniform crossover
            mask = rng.random(n) < 0.5
            child = np.where(mask, parent1, parent2).astype(np.uint8)
            # repair sum
            child = repair_binary(child)
            # mutation: perform swaps
            if rng.random() < 0.7:  # mutation probability
                # determine number of swaps
                num_swaps = max(1, int(mutation_rate * n))
                for _ in range(num_swaps):
                    # pick a random 1 and a random 0
                    ones = np.where(child == 1)[0]
                    zeros = np.where(child == 0)[0]
                    if ones.size == 0 or zeros.size == 0:
                        break
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    child[i] = 0
                    child[j] = 1
                # after swaps still sum constant, no repair needed.
            # evaluate
            c5_val = compute_c5(child.astype(float))
            new_pop.append(child)
            new_fitness.append(c5_val)

        pop = new_pop
        fitness = np.array(new_fitness, dtype=np.float64)

        # track best
        cur_best_idx = np.argmin(fitness)
        cur_best_c5 = fitness[cur_best_idx]
        if cur_best_c5 < best_c5:
            best_c5 = cur_best_c5
            best_h = pop[cur_best_idx].astype(float)

        # occasional local hill climbing on best
        if generation % 10 == 0:
            # perform a few random swap improvements
            h_local = best_h.copy()
            # We will try to improve via swapping randomly several times with acceptance if improves.
            # Use temperature schedule decreasing over generations
            # Simplify: perform simple hill climb for limited steps
            for _ in range(200):
                # pick one 1 and one 0
                ones_idx = np.where(h_local == 1)[0]
                zeros_idx = np.where(h_local == 0)[0]
                if ones_idx.size == 0 or zeros_idx.size == 0:
                    break
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                # propose swap
                h_candidate = h_local.copy()
                h_candidate[i] = 0
                h_candidate[j] = 1
                cand_c5 = compute_c5(h_candidate)
                if cand_c5 + 1e-12 < best_c5:
                    best_c5 = cand_c5
                    best_h = h_candidate.copy()
                    h_local = h_candidate
            # after local search, ensure best_h updated

        # early exit if target reached
        if best_c5 <= 0.3808:
            break

    # final check and return
    # ensure best_h array is float
    best_h = np.clip(best_h, 0.0, 1.0)
    # ensure exact sum
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # adjust by adding small amount to each element proportionally
        diff = target_sum - cur_sum
        best_h += diff / n
        # clip and correct residual if any
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - best_h.sum()
        if abs(residual) > 1e-12:
            # distribute residual arbitrarily
            # find indices where we can add/subtract
            if residual > 0:
                inc_idx = np.where(best_h < 1-1e-12)[0]
                if inc_idx.size:
                    amount = residual / inc_idx.size
                    best_h[inc_idx] += amount
            else:
                dec_idx = np.where(best_h > 1e-12)[0]
                if dec_idx.size:
                    amount = residual / dec_idx.size
                    best_h[dec_idx] += amount
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n

def improve_by_targeted_swaps(h, max_iters, rng):
    # h: binary np.ndarray uint8 shape n
    # returns improved h, final C5
    n = len(h)
    current_c5 = compute_c5(h.astype(float))
    target_sum = n // 2
    for it in range(max_iters):
        # compute correlation to find worst shift
        corr = np.correlate(h.astype(float), 1 - h.astype(float), mode='full')
        max_idx = np.argmax(corr)
        max_c5 = corr[max_idx] * (2.0 / n)
        if max_c5 < current_c5 - 1e-12:
            current_c5 = max_c5
        # shift amount:
        shift = max_idx - (n - 1)
        # compute candidate indices
        # define range for i where i+shift in bounds
        if shift >= 0:
            i_range = np.arange(0, n - shift)
        else:
            i_range = np.arange(-shift, n)
        # indices where h[i]==1 and h[i+shift]==0
        # i_range may be large; use vectorized check
        hi = h[i_range]
        h_shift = h[i_range + shift]
        mismatched1 = i_range[(hi == 1) & (h_shift == 0)]
        # indices where h[i]==0 and h[i+shift]==1
        mismatched0 = i_range[(hi == 0) & (h_shift == 1)]
        if mismatched1.size == 0 or mismatched0.size == 0:
            # can't improve with this shift, perhaps break
            break
        i = rng.choice(mismatched1)
        j = rng.choice(mismatched0)
        # swap bits
        h[i] = 0
        h[j] = 1
        # recompute c5 for early termination
        new_c5 = compute_c5(h.astype(float))
        if new_c5 <= current_c5 + 1e-12:
            current_c5 = new_c5
        else:
            # if not improvement revert? But swap should improve theoretically.
            # revert perhaps to maintain monotonic improvement.
            # Let's revert if worsened
            h[i] = 1
            h[j] = 0
            # break out if can't improve
            break
    return h, current_c5

import time
import math
import numpy as np

def compute_c5(arr):
    """Maximum overlap integral approximated by discrete correlation."""
    n = arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_binary(bin_arr, target_sum, rng):
    """Flip bits to enforce exactly target_sum ones."""
    cur = int(bin_arr.sum())
    if cur == target_sum:
        return bin_arr
    diff = target_sum - cur
    if diff > 0:
        zeros = np.where(bin_arr == 0)[0]
        if zeros.size >= diff:
            idx = rng.choice(zeros, diff, replace=False)
            bin_arr[idx] = 1
    else:
        ones = np.where(bin_arr == 1)[0]
        if ones.size >= -diff:
            idx = rng.choice(ones, -diff, replace=False)
            bin_arr[idx] = 0
    return bin_arr

def thue_morse_seq(length, target_sum, rng):
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(length)),
                      dtype=np.uint8, count=length)
    return repair_binary(seq, target_sum, rng)

def rudin_shapiro_seq(length, target_sum, rng):
    a = np.array([1], dtype=np.int8)
    while a.size < length:
        k = np.arange(a.size, dtype=np.int64)
        sign = np.where(k % 2 == 0, 1, -1)
        nxt = np.empty(a.size * 2, dtype=np.int8)
        nxt[0::2] = a
        nxt[1::2] = a * sign
        a = nxt
    seq = ((a[:length] + 1) // 2).astype(np.uint8)
    return repair_binary(seq, target_sum, rng)

def improve_by_targeted_swaps(h, max_iters, time_end, rng):
    """Local search: repeatedly fix the worst shift by a beneficial swap."""
    n = h.shape[0]
    for _ in range(max_iters):
        if time.time() >= time_end:
            break
        corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        # index range where i and i+shift are both valid
        if shift >= 0:
            i_range = np.arange(0, n - shift)
        else:
            i_range = np.arange(-shift, n)
        hi = h[i_range]
        hsh = h[i_range + shift]
        mism1 = i_range[(hi == 1) & (hsh == 0)]
        mism0 = i_range[(hi == 0) & (hsh == 1)]
        if mism1.size == 0 or mism0.size == 0:
            break
        i = rng.choice(mism1)
        j = rng.choice(mism0)
        h[i] = 0
        h[j] = 1
    return h, compute_c5(h.astype(float))

def continuous_anneal(start_h, time_end, rng):
    """Simulated annealing on continuous values preserving the total sum."""
    n = start_h.shape[0]
    target = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    max_iter = 300000
    init_temp = 5e-3
    final_temp = 1e-7

    for it in range(max_iter):
        if time.time() >= time_end:
            break
        inc = np.where(h < 1.0 - 1e-12)[0]
        dec = np.where(h > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            continue
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    # exact sum correction
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target - np.sum(best_h)
        if abs(residual) > 1e-12:

                best_h[idx] += adj
                residual = target - np.sum(best_h)
    return best_h, best_c5
