def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    # Use convolution: corr = np.convolve(h, 1-h[::-1], mode='full')
    # But use FFT for speed:
    N = 1 << (2*n-1).bit_length()
    # Actually compute next power of two >= 2*n.
    N = 1 << ((2*n-1).bit_length())
    # compute h_pad = np.fft.rfft, but easier: use np.fft.fft.
    # Using real FFT:
    f_h = np.fft.rfft(h, n=N)
    f_rev = np.fft.rfft((1-h)[::-1], n=N)
    conv = np.fft.irfft(f_h * f_rev, n=N)
    # conv length N; linear convolution results in length n+m-1 = 2n-1; we need first 2n-1 entries.
    corr = conv[:2*n-1]  # this is correlation (with reversed second argument)

    # Since conv = convolution of h with rev(1-h) = sum_i h[i] * (1-h)[j-i] where j index . This matches correlation.
    c5 = np.max(corr) * dx
    return c5

def anneal_h(h_init, time_budget, start_time, rng):

    best_h = h_current = h_init.copy()
    best_c5 = current = compute_c5_fft(h_current)
    n = len(h_current)
    M = n//2
    total = time_budget
    # schedule: initial temperature perhaps 0.01 or based on best tolerance.
    T0 = 0.01
    while time.time() - start_time < total:
        # compute remaining fraction:
        elapsed = time.time() - start_time
        frac = (elapsed) / total
        temp = T0 * (1 - frac)
        # Do iteration
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - h_current[i], h_current[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        # apply
        h_current[i] += delta
        h_current[j] -= delta
        new_c5 = compute_c5_fft(h_current)
        if new_c5 < current:
            # accept improvement
            current = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_current.copy()
        else:
            # accept uphill with probability
            if temp > 0:
                prob = np.exp(-(new_c5 - current) / temp)
                if rng.random() < prob:
                    current = new_c5
                else:
                    # revert changes
                    h_current[i] -= delta
                    h_current[j] += delta
            else:
                # revert
                h_current[i] -= delta
                h_current[j] += delta
        # maybe periodically reset to best if too many rejects? Not needed.

    return best_h, best_c5

def compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    N = 1 << ((2*n-1).bit_length())
    # using real FFT to reduce time: use np.fft.rfft for both.
    f_h = np.fft.rfft(h, n=N)
    f_rev = np.fft.rfft((1.0 - h)[::-1], n=N)
    conv = np.fft.irfft(f_h * f_rev, n=N)
    # conv array length N, but linear convolution result length 2*n-1 at start.
    # Only need first 2*n-1 entries.
    max_val = np.max(conv[:2*n-1])
    return max_val * dx

import time

# ---------- Helper for feasible random h ----------
def generate_initial_h(n, M, rng):
    """Generate a feasible h with sum exactly M (M = n/2)."""
    method = rng.integers(3)
    if method == 0:  # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:  # uniform random then project
        h = rng.random(n)
        h = h / np.sum(h) * M
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    else:  # linear decreasing shape with correction
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    return h

# ---------- C5 evaluation via FFT ----------
def compute_c5_fft(h):
    """Maximum overlap integral for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # Linear convolution length = 2*n-1
    N = 1 << ((2 * n - 1).bit_length())  # next power of two
    f_h = np.fft.rfft(h, n=N)
    f_rev = np.fft.rfft((1.0 - h)[::-1], n=N)
    conv = np.fft.irfft(f_h * f_rev, n=N)
    max_val = np.max(conv[:2 * n - 1])
    return max_val * dx

# ---------- Simulated annealing for a single run ----------
def anneal_h(h0, max_seconds, rng):
    """Return a (possibly) improved h and its C5 after annealing."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    current = compute_c5_fft(h)
    best_h = h.copy()
    best_c5 = current
    T0 = 0.01                # initial temperature
    p_swap = 0.02            # probability to attempt a full swap

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        frac = elapsed / max_seconds
        temp = T0 * (1.0 - frac)

        i, j = rng.choice(n, 2, replace=False)

        # occasional full swap of values
        if rng.random() < p_swap:
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5_fft(h)
            accept = (new_c5 < current) or (temp > 0 and rng.random() < np.exp(-(new_c5 - current) / temp))
            if accept:
                current = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]
            if best_c5 < 0.3808:
                break
            continue

        # mass transfer from j to i
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] += delta
        h[j] -= delta
        new_c5 = compute_c5_fft(h)

        if new_c5 < current:
            current = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            if temp > 0 and rng.random() < np.exp(-(new_c5 - current) / temp):
                current = new_c5
            else:
                h[i] -= delta
                h[j] += delta

        if best_c5 < 0.3808:
            break

    return best_h, best_c5

# ---------- Optimization for a fixed discretisation ----------
def optimise_n(n, time_budget, rng):
    """Run several annealing restarts for a given n within time_budget."""
    start = time.time()
    best_h = None
    best_c5 = np.inf
    M = n // 2

    while True:

        remaining = time_budget - elapsed
        if remaining < 12.0:       # need a few seconds for a meaningful run
            break

        # initial point

            h0 = np.array(initial_h_values, dtype=float)
            cur = h0.sum()
            if abs(cur - M) > 1e-12:
                h0 += (M - cur) / n

        else:
            h0 = generate_initial_h(n, M, rng)

        # devote a fraction of the remaining budget to this restart
        sub_budget = min(remaining * 0.5, 60.0)
        h_opt, c5_opt = anneal_h(h0, sub_budget, rng)

            best_c5 = c5_opt
            best_h = h_opt

        if best_c5 < 0.3808:
            break

    return best_h, best_c5
