import time, math, numpy as np

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # Clip and binary search lambda to satisfy sum constraint
    low = -np.max(v) - 5.0
    high = hi - np.min(v) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def compute_overlap_array(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def gradient_smoothed_max(h, alpha):
    n = h.size
    overlaps = compute_overlap_array(h)  # length 2n-1
    max_val = overlaps.max()
    # weights
    w = np.exp(alpha * (overlaps - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        # fallback uniform
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum

    # neighbor weighted sum
    # w indexed from -(n-1) to n-1
    neighbor = np.zeros(n)
    # loop over offsets
    # For each offset k from -(n-1) to (n-1):
    # weight = w[k + n - 1]
    # apply contributions
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if wk == 0.0:
            continue
        if k >= 0:
            # +k shift: h[i+k] aligns with i (i from 0..n-k-1)
            neighbor[:n - k] += wk * h[k:]
            # -k shift: h[i-k] aligns with i (i from k..n-1)
            neighbor[k:] += wk * h[:n - k]
        else:
            kk = -k
            # +k negative: shift left, i.e., h[i-k] aligns with i
            neighbor[kk:] += wk * h[:n - kk]
            neighbor[:n - kk] += wk * h[kk:]
    # gradient = 1 - neighbor
    return 1.0 - neighbor

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine n and initial h

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Parameters
    alpha = 30.0    # smoothing parameter
    step0 = 0.5     # initial step size for gradient moves
    noise_step = 0.2  # initial noise magnitude for random perturbations
    iter_count = 0
    # Simulated annealing temperature schedule
    T0 = 0.01
    # acceptable improvement threshold
    improve_tol = 1e-7

    while True:
        now = time.time()
        if now - start_time >= budget_s - 0.5:
            break

        iter_count += 1

        # Decide move type
        if rng.random() < 0.6:
            # Gradient move
            grad = gradient_smoothed_max(cur_h, alpha)
            # step size decays

            cur_c5 = cand_c5

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

import time
import math
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the box [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = hi - np.min(v) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def compute_overlap_array(h):
    """Full array of overlaps (including all integer shifts)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def gradient_smoothed_max(h, alpha):
    """Gradient of the smoothed max overlap (using softmax with parameter alpha)."""
    n = h.shape[0]
    dx = 2.0 / n
    overlaps = compute_overlap_array(h)                     # length 2n-1
    max_val = overlaps.max()
    w = np.exp(alpha * (overlaps - max_val))               # stability via subtraction
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                                          # softmax weights, sum to 1

    # weighted neighbor sum: Σ_k w_k (h_{i+k}+h_{i-k})
    neighbor = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if wk == 0.0:
            continue
        if k >= 0:
            # +k shift
            neighbor[: n - k] += wk * h[k:]
            # -k shift
            neighbor[k:] += wk * h[: n - k]
        else:
            kk = -k
            neighbor[kk:] += wk * h[: n - kk]
            neighbor[: n - kk] += wk * h[kk:]
    # gradient of the smoothed objective
    return dx * (1.0 - neighbor)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Initialise h -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.size
    else:
        n = 80
        # smooth cosine start (integrates to 1 automatically)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h0 = 0.5 * (1.0 + np.cos(np.pi * x))
    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ---- Parameters ---------------------------------------------------
    alpha = 30.0            # smoothing for softmax
    step0 = 0.5             # base step size for gradient moves
    noise_step = 0.2        # base magnitude for random perturbations
    T0 = 0.02               # initial temperature for simulated annealing
    iter_count = 0
    no_improve = 0

    # ---- Main optimisation loop ----------------------------------------
    while True:
        now = time.time()
        if now - start_time >= budget_s - 0.5:
            break
        iter_count += 1

        # Choose move type
        if rng.random() < 0.6:
            # Gradient descent on smoothed max
            grad = gradient_smoothed_max(cur_h, alpha)
            step = step0 / math.sqrt(iter_count)
            cand = cur_h - step * grad
        else:
            # Random perturbation
            scale = noise_step * (rng.random() ** 2)
            cand = cur_h + scale * rng.normal(size=n)

        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)

        if cand_c5 < cur_c5 - 1e-12:
            cur_h, cur_c5 = cand, cand_c5
            no_improve = 0
            if cand_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cand.copy(), cand_c5
                if best_c5 < 0.3805:
                    break
        else:
            # Simulated‑annealing acceptance
            T = T0 * max(0.0, 1.0 - (now - start_time) / budget_s)
            if T > 0.0:
                delta = cand_c5 - cur_c5
                if rng.random() < math.exp(-delta / T):
                    cur_h, cur_c5 = cand, cand_c5
            no_improve += 1

        # Adapt smoothing parameter
        if iter_count % 500 == 0:
            alpha = min(alpha * 1.5, 200.0)

        # Restart if stuck for a while
        if no_improve >= 4000:
            cur_h = rng.random(n)
            cur_h = project_to_box_sum(cur_h, target_sum)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0

    # ---- Final safety projection ---------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
