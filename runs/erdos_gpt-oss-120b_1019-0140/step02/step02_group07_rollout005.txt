def gradient_max_shift(h):
    n = len(h)
    # compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    s_idx = np.argmax(corr)
    s = s_idx - (n - 1)
    g = np.zeros(n)
    # indices where i + s valid
    if s >= 0:
        # i from 0 to n-1-s
        i_valid = np.arange(n - s)
        g[i_valid] += (1 - h[i_valid + s])
        # i - s valid for i from s to n-1
        i_valid2 = np.arange(s, n)
        g[i_valid2] -= h[i_valid2 - s]
    else:
        # s negative
        s_neg = -s
        # i + s valid => i from s_neg to n-1
        i_valid = np.arange(s_neg, n)
        g[i_valid] += (1 - h[i_valid + s])  # i + s = i - s_neg
        # i - s valid => i from 0 to n-1 - s_neg
        i_valid2 = np.arange(n - s_neg)

def enforce_sum_and_bounds(h, target_sum):
    # ensure within [0,1]
    h = np.clip(h, 0.0, 1.0)
    # enforce sum via iterative water-filling
    # compute current sum
    current = np.sum(h)
    diff = target_sum - current
    if abs(diff) < 1e-12:
        return h
    # if diff > 0: need to increase some entries not at 1
    # if diff < 0: need to decrease some entries not at 0
    # allocate equally among free entries
    # loop may need to adjust as some entries hit bound
    free = np.ones_like(h, dtype=bool)
    # Do iterative water-filling
    while abs(diff) > 1e-12:
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        # amount per free entry
        delta = diff / free_idx.size
        # limit adjusting each free entry to maintain bounds
        for i in free_idx:
            if diff > 0:
                max_inc = 1.0 - h[i]
                inc = min(delta, max_inc)
                h[i] += inc
                diff -= inc
                if h[i] >= 1.0 - 1e-12:
                    free[i] = False
            else:
                max_dec = h[i]
                dec = min(-delta, max_dec)
                h[i] -= dec
                diff += dec
                if h[i] <= 1e-12:
                    free[i] = False

        # If diff small enough break

            break

    current = np.sum(h)
    diff = target_sum - current
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        total = cap.sum()
        if total > 0:
            h += diff * cap / total
    else:
        cap = h
        total = cap.sum()
        if total > 0:
            h -= (-diff) * cap / total
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # Due to numerical, might be off a tiny amount; adjust by adding residual uniformly then clip again
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        h += residual / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_of_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    s = np.argmax(corr) - (n - 1)
    idx = np.arange(n)
    i_plus = idx + s
    mask_plus = (i_plus >= 0) & (i_plus < n)
    i_minus = idx - s
    mask_minus = (i_minus >= 0) & (i_minus < n)
    g = np.zeros(n)
    g[mask_plus] += (1 - h[i_plus[mask_plus]])
    g[mask_minus] -= h[i_minus[mask_minus]]
    return g, s, corr

    it = 0
    while time.time() < time_end and it < max_iter:
        # choose inc and dec indices
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break

        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        # delta scaling with temperature maybe
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # preserve sum exactly (should already)
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)

                best_c5 = new_c5

    diff = target_sum - np.sum(best_h)

                break

import time, math, numpy as np, scipy.optimize as opt

def compute_c5(arr):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    n = arr.shape[0]
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def enforce_sum_and_bounds(h, target_sum):
    """Clip to [0,1] and project onto the hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        total = cap.sum()
        if total > 0:
            h += diff * cap / total
    else:
        cap = h
        total = cap.sum()
        if total > 0:
            h -= (-diff) * cap / total
    # final tiny correction
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        h += residual / h.size
    return np.clip(h, 0.0, 1.0)

def gradient_of_max_corr(h):
    """Gradient of the shift giving the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    s = np.argmax(corr) - (n - 1)
    idx = np.arange(n)
    i_plus = idx + s
    mask_plus = (i_plus >= 0) & (i_plus < n)
    i_minus = idx - s
    mask_minus = (i_minus >= 0) & (i_minus < n)
    g = np.zeros(n)
    g[mask_plus] += (1 - h[i_plus[mask_plus]])
    g[mask_minus] -= h[i_minus[mask_minus]]
    return g, s, corr

def gradient_descent(start_h, target_sum, start_time, budget_s, time_end, rng):
    """Projected gradient descent on the max‑overlap shift."""
    n = start_h.size
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    init_step, final_step = 0.2, 1e-5
    while time.time() < time_end:
        frac = (time.time() - start_time) / budget_s
        step = init_step * (1 - frac) + final_step * frac
        g, _, _ = gradient_of_max_corr(h)
        if np.linalg.norm(g) == 0:
            i = rng.integers(n)
            delta = (rng.random() - 0.5) * step
            h_new = h.copy()
            h_new[i] += delta
        else:
            h_new = h - step * g
        h_new = enforce_sum_and_bounds(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h, best_c5 = h_new.copy(), new_c5
        T = max(0.01 * (1 - frac), 0.0001)
        if new_c5 <= cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            h, cur_c5 = h_new, new_c5
    return best_h, best_c5

def continuous_anneal(start_h, time_end, rng):
    """Simple simulated annealing preserving the exact sum."""
    n = start_h.size
    target = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    init_temp, final_temp = 1e-2, 1e-7
    max_iter = 500000
    it = 0
    while time.time() < time_end and it < max_iter:
        inc = np.where(h < 1 - 1e-12)[0]
        dec = np.where(h > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h, cur_c5 = h_new, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h_new.copy(), new_c5
        it += 1
    # exact‑sum correction
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target - np.sum(best_h)
    return best_h, best_c5

def binary_hill_climb(start_bin, time_end, rng):
    """Swap‑based hill climbing on binary vectors."""
    n = start_bin.size
    cur = start_bin.copy()
    cur_c5 = compute_c5(cur.astype(float))
    best = cur.copy()
    best_c5 = cur_c5
    no_improve = 0
    while time.time() < time_end and no_improve < 3000:
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best, best_c5 = cand.copy(), cand_c5
            cur = cand
            no_improve = 0
        else:
            no_improve += 1
    return best.astype(float), best_c5

def _de_phase(start_h, target_sum, time_end, seed):
    """Quick DE run limited to a few generations."""
    n = start_h.size
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        h = enforce_sum_and_bounds(np.array(x, dtype=float), target_sum)
        return compute_c5(h)
    maxiter = max(1, int((time_end - time.time()) // 0.2))
    try:
        res = opt.differential_evolution(obj, bounds, maxiter=maxiter,
                                         popsize=5, polish=False,
                                         seed=seed, updating='deferred',
                                         workers=1)
        h = enforce_sum_and_bounds(res.x, target_sum)
        return h, compute_c5(h)
    except Exception:
        return start_h, compute_c5(start_h)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:

        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
    target_sum = n / 2.0
    h0 = enforce_sum_and_bounds(h0, target_sum)

    best_h, best_c5 = h0.copy(), compute_c5(h0)

    # main iterative search
    while time.time() - start_time < budget_s - 0.5:
        remaining = budget_s - (time.time() - start_time)
        chunk = min(4.0, remaining * 0.4)
        end = time.time() + chunk
        r = rng.random()
        if r < 0.3:
            cand_h, cand_c5 = gradient_descent(best_h, target_sum,
                                               start_time, budget_s, end, rng)
        elif r < 0.6:
            cand_h, cand_c5 = continuous_anneal(best_h, end, rng)
        elif r < 0.85:
            bin_start = (best_h >= 0.5).astype(int)
            cand_h, cand_c5 = binary_hill_climb(bin_start, end, rng)
        else:
            cand_h, cand_c5 = _de_phase(best_h, target_sum, end, seed)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h.copy(), cand_c5
        if best_c5 <= 0.3808:
            break

    # final binary refinement if time permits
    if time.time() - start_time < budget_s - 0.2:
        bin_start = (best_h >= 0.5).astype(int)
        final_end = start_time + budget_s - 0.1
        fin_h, fin_c5 = binary_hill_climb(bin_start, final_end, rng)
        if fin_c5 < best_c5:
            best_h, best_c5 = fin_h, fin_c5

    best_h = enforce_sum_and_bounds(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
