def min_overlap_binary(h):
    conv = np.correlate(h, h, mode='full')
    # Positive shifts are conv[n:] (exclude zero shift)
    return int(conv[len(h):].min())

def c5_from_min_overlap(min_overlap, n):
    return 1.0 - (2.0 * min_overlap) / n

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)

def rudin_shapiro_seq(n):
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def adjust_weight(bin_arr, target_sum, rng):
    arr = bin_arr.astype(int).copy()
    cur = arr.sum()
    diff = target_sum - cur
    if diff == 0:
        return arr
    idxs = np.arange(arr.size)
    rng.shuffle(idxs)
    for idx in idxs:
        if diff == 0:
            break
        if diff > 0 and arr[idx] == 0:
            arr[idx] = 1
            diff -= 1
        elif diff < 0 and arr[idx] == 1:
            arr[idx] = 0
            diff += 1
    return arr

import time
import numpy as np

def min_overlap_binary(h):
    # autocorrelation of binary vector, positive shifts only
    conv = np.correlate(h, h, mode='full')
    return int(conv[len(h):].min())

def c5_from_min_overlap(min_overlap, n):
    return 1.0 - (2.0 * min_overlap) / n

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=int, count=n)

def rudin_shapiro_seq(n):
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def adjust_weight(bin_arr, target_sum, rng):
    arr = bin_arr.astype(int).copy()
    cur = arr.sum()
    diff = target_sum - cur
    if diff == 0:
        return arr
    idxs = np.arange(arr.size)
    rng.shuffle(idxs)
    for idx in idxs:
        if diff == 0:
            break
        if diff > 0 and arr[idx] == 0:
            arr[idx] = 1
            diff -= 1
        elif diff < 0 and arr[idx] == 1:
            arr[idx] = 0
            diff += 1
    return arr
