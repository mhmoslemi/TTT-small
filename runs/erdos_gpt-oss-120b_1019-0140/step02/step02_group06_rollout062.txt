def adam_step(h, g, m, v, t, alpha=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * g
    v = beta2 * v + (1 - beta2) * (g * g)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = alpha * m_hat / (np.sqrt(v_hat) + eps)
    h_new = h - step
    return h_new, m, v
