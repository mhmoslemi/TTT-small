import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        v = np.clip(v + diff / n, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    inc = min(diff, 1.0 - v[i])
                    v[i] += inc
                    diff -= inc
                elif diff < 0 and v[i] > 0.0:
                    dec = min(-diff, v[i])
                    v[i] -= dec
                    diff += dec
    return v

def anneal(h_start, best_val, start_time, deadline, rng):
    """Simulated annealing on the mass vector to reduce max overlap."""
    h = h_start.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best = cur_val
    n = h.shape[0]
    total_budget = deadline - start_time
    while time.time() < deadline - 0.1:
        # find worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # contribution of each index to this shift
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # choose destination j that does not increase this shift if possible
        cand_j = np.where(h < 1.0 - 1e-12)[0]
        if cand_j.size == 0:
            break
        if k >= 0:
            out_of_range = cand_j + k >= n
        else:
            out_of_range = cand_j + k < 0
        if np.any(out_of_range):
            j = rng.choice(cand_j[out_of_range])
        else:
            # pick j with largest h[j+k] (most reduction when receiving mass)
            if k >= 0:
                shifted_vals = h[cand_j + k]
            else:
                shifted_vals = h[cand_j + k]
            j = cand_j[np.argmax(shifted_vals)]
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
                if best < 0.3808:
                    return best_h, best
        else:
            remaining = max(deadline - time.time(), 0.0)
            T = max(1e-12, 0.001 * remaining / total_budget)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best
