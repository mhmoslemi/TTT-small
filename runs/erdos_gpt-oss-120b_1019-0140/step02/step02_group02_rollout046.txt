import time
import math
import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_c5(h, dx):
    """Maximum overlap integral approximated by discrete correlation."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        inc = diff / n
        v = np.clip(v + inc, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    delta = min(diff, 1.0 - v[i])
                    v[i] += delta
                    diff -= delta
                elif diff < 0 and v[i] > 0.0:
                    delta = min(-diff, v[i])
                    v[i] -= delta
                    diff += delta
    return v

def project_to_sum(vec, target_sum, rng):
    """Adjust vector (still within [0,1]) so that its sum equals target_sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    diff = target_sum - cur
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 2000:
        idx = rng.integers(0, vec.size)
        max_inc = (1.0 - vec[idx]) if diff > 0 else -vec[idx]
        if max_inc == 0:
            attempts += 1
            continue
        inc = np.clip(diff, -max_inc, max_inc)
        vec[idx] += inc
        diff = target_sum - vec.sum()
        attempts += 1
    return np.clip(vec, 0.0, 1.0)

def de_objective(x, target_sum, dx, deadline):
    """Objective for differential evolution: C5 + penalty for sum deviation."""
    # Early abort if time is almost exhausted
    if time.time() > deadline - 0.5:
        return 1e9
    s = x.sum()
    penalty = 1e4 * abs(s - target_sum)
    return compute_c5(x, dx) + penalty

def anneal_h(h_start, target_sum, dx, deadline, rng):
    """Simple simulated annealing that preserves the total sum."""
    h = h_start.copy()
    cur_val = compute_c5(h, dx)
    best_h = h.copy()
    best_val = cur_val
    T0 = 0.01
    start_time = time.time()
    max_steps = 5000
    steps = 0
    while time.time() < deadline - 0.2 and steps < max_steps:
        steps += 1
        # pick source (must have some mass) and destination (capacity left)
        src_idx = rng.choice(np.where(h > 1e-12)[0])
        dst_idx = rng.choice(np.where(h < 1.0 - 1e-12)[0])
        if src_idx == dst_idx:
            continue
        max_transfer = min(h[src_idx], 1.0 - h[dst_idx])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[src_idx] -= delta
        h[dst_idx] += delta
        new_val = compute_c5(h, dx)
        # acceptance criterion
        if new_val < cur_val or rng.random() < math.exp((cur_val - new_val) / max(T0, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert move
            h[src_idx] += delta
            h[dst_idx] -= delta
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / max(1e-9, (deadline - start_time - 0.2))
        T0 = max(1e-6, 0.01 * (1.0 - frac))
        if best_val < 0.3808:
            break
    return best_h, best_val
