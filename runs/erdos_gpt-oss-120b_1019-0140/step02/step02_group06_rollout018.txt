    h = h0.copy()

    start = time.time()

    max_beta = 20.0
    alpha0 = 0.2
    no_improve = 0
    while time.time() - start < time_budget:
        # compute correlation vector
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # increase smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
        # compute softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w = w / w_sum
        # gradient
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-15:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)

                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        g -= np.mean(g)

import time

def compute_c5(h):
    """Maximum overlap integral scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum_and_bounds(x, target, tol=1e-12):
    """Clip to [0,1] and force sum(x) == target (exact up to tolerance)."""
    x = np.clip(x.astype(float), 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        # uniform adjustment
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > tol:
            if diff > 0:
                need = diff
                idx = np.argsort(x)  # smallest first
                for i in idx:
                    if need <= 0:
                        break
                    space = 1.0 - x[i]
                    if space > 0:
                        inc = min(space, need)
                        x[i] += inc
                        need -= inc
            else:
                need = -diff
                idx = np.argsort(-x)  # largest first
                for i in idx:
                    if need <= 0:
                        break
                    space = x[i]
                    if space > 0:
                        dec = min(space, need)
                        x[i] -= dec
                        need -= dec
    return x

        init_h = np.array(initial_h_values, dtype=float)
        n = len(init_h)
    else:
        n = 100
        init_h = np.zeros(n, dtype=float)
        init_h[:n // 2] = 1.0
        rng.shuffle(init_h)

    target_sum = n / 2.0
    use_binary = (n % 2 == 0)

    # Prepare binary start if applicable
    if use_binary:
        order = np.argsort(init_h)
        bin_start = np.zeros(n, dtype=np.int8)
        bin_start[order[-int(target_sum):]] = 1
    else:
        bin_start = None

    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)

    # Phase 1: binary combinatorial search
    if use_binary and remaining > budget_s * 0.2:
        bin_budget = budget_s * 0.6
        if bin_budget > remaining:
            bin_budget = remaining * 0.9
        best_bin, best_bin_c5 = binary_search(bin_start, bin_budget, rng)
        elapsed = time.time() - start
        remaining = max(0.0, budget_s - elapsed)
    else:
        best_bin = None
        best_bin_c5 = np.inf

    # Phase 2: continuous refinement if time permits
    final_h = None
    final_c5 = np.inf

    if remaining > budget_s * 0.1:
        if best_bin is not None:
            h0 = best_bin.astype(float)
        else:
            h0 = init_h.copy()
        h0 = project_sum_and_bounds(h0, target_sum)
        refined_h, refined_c5 = optimize_h(h0, remaining, rng.integers(1_000_000_000))
        final_h = refined_h
        final_c5 = refined_c5

    # Compare with pure binary result
    if best_bin is not None and best_bin_c5 < final_c5:
        final_h = best_bin.astype(float)
        final_c5 = best_bin_c5

    # Safety projection
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = project_sum_and_bounds(final_h, target_sum)
    final_c5 = compute_c5(final_h)

    return final_h, final_c5, n
