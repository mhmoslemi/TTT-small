def continuous_anneal(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    # ensure sum, clip
    h = np.clip(h, 0.0, 1.0)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    it = 0
    max_iter = 500000  # large
    # schedule: temperature linear from init to final based on iteration count
    init_temp = 1e-2
    final_temp = 1e-7
    while time.time() < time_end and it < max_iter:
        # pick indices i (increase) and j (decrease)
        # to preserve sum (increase some and decrease others)
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        # propose delta as random fraction of max_delta
        delta = max_delta * rng.random() * 0.5
        # new h
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < np.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # final correction
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        # residual fix
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_c5

import time
import math
import numpy as np

def compute_c5(arr):
    """Maximum overlap integral approximated by discrete correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Thue‑Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=int, count=n)

def rudin_shapiro_seq(n):
    """Rudin‑Shapiro binary sequence (+1/‑1 → 1/0) of length n."""
    a = np.array([1], dtype=int)
    while a.size < n:
        a = np.concatenate([a, a * ((-1) ** np.arange(a.size) % 2)])
    a = a[:n]
    return ((a + 1) // 2).astype(int)

def adjust_binary_to_target(vec, target, rng):
    """Flip bits to achieve exactly target ones."""
    cur = int(vec.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(vec == 0)
        if zeros.size:
            choose = rng.choice(zeros, diff, replace=False)
            vec[choose] = 1
    elif diff < 0:
        ones = np.flatnonzero(vec == 1)
        if ones.size:
            choose = rng.choice(ones, -diff, replace=False)
            vec[choose] = 0
    return vec

def random_binary(n, target, rng):
    """Random binary vector with exactly target ones."""
    vec = np.zeros(n, dtype=int)
    idx = rng.choice(n, target, replace=False)
    vec[idx] = 1
    return vec

def local_optimize_binary(h_bin, time_end):
    """Exhaustive swap hill‑climb on a binary vector."""
    cur_f = compute_c5(h_bin.astype(float))
    n = len(h_bin)
    while True:
        if time.time() > time_end:
            break
        ones = np.flatnonzero(h_bin)
        zeros = np.flatnonzero(1 - h_bin)
        best_f = cur_f
        best_swap = None
        for i in ones:
            for j in zeros:
                # apply tentative swap
                h_bin[i] = 0
                h_bin[j] = 1
                cand_f = compute_c5(h_bin.astype(float))
                if cand_f < best_f - 1e-12:
                    best_f = cand_f
                    best_swap = (i, j)
                # revert
                h_bin[i] = 1
                h_bin[j] = 0
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0
            h_bin[j] = 1
            cur_f = best_f
        else:
            break
    return h_bin, cur_f

def continuous_anneal(start_h, time_end, rng):
    """Simulated annealing on continuous values preserving the total sum."""
    n = len(start_h)
    target_sum = n / 2.0
    h = np.clip(start_h.copy(), 0.0, 1.0)
    cur_f = compute_c5(h)
    best_h = h.copy()
    best_f = cur_f
    max_iter = 400000
    init_temp = 5e-3
    final_temp = 1e-7
    it = 0
    while time.time() < time_end and it < max_iter:
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_f = compute_c5(h_new)
        dE = new_f - cur_f
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_f = new_f
            if new_f < best_f:
                best_f = new_f
                best_h = h_new.copy()
        it += 1
    # final exact‑sum correction
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        init_h = np.zeros(n, dtype=float)

    target = n // 2

    # Initial best from the provided construction (if any)
    best_h = init_h.copy()
    best_c5 = compute_c5(best_h)

    # Prepare binary seeds
    seeds_bin = []

    # Seed from the provided construction (rounded)
    bin0 = (best_h >= 0.5).astype(int)
    bin0 = adjust_binary_to_target(bin0, target, rng)
    seeds_bin.append(bin0)

    # Thue‑Morse
    tm = thue_morse_seq(n)
    tm = adjust_binary_to_target(tm, target, rng)
    seeds_bin.append(tm)

    # Rudin‑Shapiro
    rs = rudin_shapiro_seq(n)
    rs = adjust_binary_to_target(rs, target, rng)
    seeds_bin.append(rs)

    # Simple alternating pattern
    alt = np.tile([1, 0], n // 2)
    alt = adjust_binary_to_target(alt, target, rng)
    seeds_bin.append(alt)

    # Allocate time for binary combinatorial search
    binary_deadline = start_time + budget_s * 0.75

    # Exhaustive hill‑climb each deterministic seed
    for seed_bin in seeds_bin:
        if time.time() >= binary_deadline:
            break
        opt_h, opt_c5 = local_optimize_binary(seed_bin.copy(), binary_deadline)
        if opt_c5 < best_c5:
            best_h = opt_h.astype(float)
            best_c5 = opt_c5
        if best_c5 <= 0.3808:
            break

    # Random restarts with exhaustive hill‑climb
    while time.time() < binary_deadline - 0.5:
        rand_bin = random_binary(n, target, rng)
        opt_h, opt_c5 = local_optimize_binary(rand_bin, binary_deadline)
        if opt_c5 < best_c5:
            best_h = opt_h.astype(float)
            best_c5 = opt_c5
        if best_c5 <= 0.3808:
            break

    # Continuous refinement with remaining budget
    remaining = start_time + budget_s - time.time()
    if remaining > 0.6:
        cont_deadline = time.time() + remaining
        # Ensure exact sum before annealing
        diff = target - np.sum(best_h)
        if abs(diff) > 1e-12:
            best_h += diff / n
            best_h = np.clip(best_h, 0.0, 1.0)
        cand_h, cand_c5 = continuous_anneal(best_h, cont_deadline, rng)
        if cand_c5 < best_c5:
            best_h = cand_h
            best_c5 = cand_c5

    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target - np.sum(best_h)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
