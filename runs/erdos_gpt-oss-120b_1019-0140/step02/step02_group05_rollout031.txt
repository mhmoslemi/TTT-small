def enforce_sum(h, target):
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # try to distribute uniformly to free entries
    # get mask of elements not at bound
    for _ in range(2):
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-12:
                return h
        # if still diff, handle saturations
        if diff > 0:
            # add to any <1
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        if idx.size == 0:
            break
        # add/subtract as needed from first few entries
        for i in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)

def gradient_step(h, target, dx, max_step):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = dx * (1.0 - neigh)
    grad_centered = grad - np.mean(grad)
    delta = -grad_centered
    max_abs = np.max(np.abs(delta))
    if max_abs > max_step:
        delta = delta * (max_step / max_abs)
    cand = h + delta
    cand = np.clip(cand, 0.0, 1.0)
    cand = enforce_sum(cand, target)
    return cand
```

Define random_transfer:

```
def random_transfer(h, target, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # move from i to j
    max_tr = min(h[i], 1.0 - h[j])
    if max_tr <= 0:
        # try reverse
        max_tr = min(h[j], 1.0 - h[i])
        if max_tr <= 0:
            return h.copy()
        donor, recv = j, i
    else:
        donor, recv = i, j
    delta = rng.random() * max_tr
    cand = h.copy()
    cand[donor] -= delta
    cand[recv] += delta
    # sum stays same, no need enforce (except clipping)
    cand = np.clip(cand, 0.0, 1.0)
    return cand
```

def random_swap(h):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    cand = h.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum(h, target):
    """Project h onto [0,1]^n while fixing the total sum = target."""
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # First try uniform distribution over free entries
    for _ in range(2):
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-12:
                return h
        # If still not fixed, adjust remaining capacity greedily
        if diff > 0:
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        if idx.size == 0:
            break
        for i in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)

def smooth_and_fix(h, target, kernel):
    """Convolve with kernel, clip and re‑project onto the sum constraint."""
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    sm = enforce_sum(sm, target)
    return sm

def gradient_step(h, target, dx, max_step):
    """One projected gradient step aimed at reducing the worst overlap."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = dx * (1.0 - neigh)
    grad_centered = grad - np.mean(grad)        # enforce zero‑sum direction
    delta = -grad_centered
    m = np.max(np.abs(delta))
    if m > max_step:
        delta = delta * (max_step / m)
    cand = h + delta
    cand = np.clip(cand, 0.0, 1.0)
    cand = enforce_sum(cand, target)
    return cand

def random_transfer(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    max_tr = min(h[i], 1.0 - h[j])
    if max_tr <= 0:
        max_tr = min(h[j], 1.0 - h[i])
        if max_tr <= 0:
            return h.copy()
        donor, recv = j, i
    else:
        donor, recv = i, j
    delta = rng.random() * max_tr
    cand = h.copy()
    cand[donor] -= delta
    cand[recv] += delta
    cand = np.clip(cand, 0.0, 1.0)
    return cand

def random_swap(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    cand = h.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand

def random_perturb(h, target, rng, scale=0.05):
    n = len(h)
    i = rng.integers(n)
    delta = (rng.random() * 2.0 - 1.0) * scale
    cand = h.copy()
    cand[i] += delta
    cand = np.clip(cand, 0.0, 1.0)
    cand = enforce_sum(cand, target)
    return cand

def init_h(n, rng):
    """Create an initial feasible h of length n."""
    if "initial_h_values" in globals():
        old = np.array(initial_h_values, dtype=float)
        if old.size == n:
            h = old.copy()
        else:
            x_old = np.linspace(0.0, 2.0, old.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, old)
    else:
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target)
    return h

    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_val = np.inf
    # try a few problem sizes; 80 is the reference size
    n_list = [80, 100, 120]
    n_idx = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.96:
            break
        n = n_list[n_idx % len(n_list)]
        n_idx += 1
        dx = 2.0 / n
        target = n / 2.0

        cur_val = compute_c5(cur_h)
        if cur_val < best_val:

        # --- local search parameters ---------------------------------
        T0 = 0.05                 # initial temperature
        max_step = 0.3            # max step size in gradient move
        probs = (0.5, 0.3, 0.1, 0.1)   # grad, transfer, swap, noise
        inner_start = time.time()

        while True:
            now = time.time()
            if now - start >= budget_s * 0.96:
                break
            if now - inner_start > budget_s * 0.28:   # cap per size
                break
            # temperature cooling (linear)
            T = T0 * max(0.0, 1.0 - (now - start) / budget_s)

            r = rng.random()
            if r < probs[0]:
                cand = gradient_step(cur_h, target, dx, max_step)
            elif r < probs[0] + probs[1]:
                cand = random_transfer(cur_h, rng)
            elif r < probs[0] + probs[1] + probs[2]:
                cand = random_swap(cur_h, rng)
            else:
                cand = random_perturb(cur_h, target, rng, scale=0.05)

            new_val = compute_c5(cand)
            accept = False
            if new_val < cur_val:
                accept = True
            elif T > 0.0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True
            if accept:
                cur_h, cur_val = cand, new_val
                if new_val < best_val:
                    best_val, best_h = new_val, cand.copy()
                    if best_val < 0.3808:
                        return best_h, best_val, n

        # --- deterministic refinement on the current best of this n ---
        # a few pure gradient steps with reduced step size
        for _ in range(8):

            cand_val = compute_c5(cand)

            else:
                break

        # --- smoothing passes with different kernels ------------------
        kernels = [np.array([0.25, 0.5, 0.25]),

        for ker in kernels:

            sm_val = compute_c5(sm)

        # --- final SLSQP refinement if time permits ------------------
        if time.time() - start < budget_s - 5.0:

            # equality constraint for the sum

                return np.sum(x[:n]) - target
            cons = [{"type": "eq", "fun": sum_eq}]

                if kk == 0:
                    continue
                def make_ineq(k):
                    def fun(x):
                        h = x[:n]
                        if k >= 0:
                            overl = np.dot(h[: n - k], 1.0 - h[k:])
                        else:
                            kk2 = -k
                            overl = np.dot(h[kk2:], 1.0 - h[: n - kk2])
                        return x[-1] - overl * dx
