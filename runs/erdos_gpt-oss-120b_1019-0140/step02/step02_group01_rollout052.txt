def proj_cap_simplex(v, M):
    low = -1e5
    high = 1e5
    for _ in range(60):
        tau = (low + high) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if s > M:
            low = tau
        else:
            high = tau
    h = np.clip(v - (low+high)/2.0, 0.0, 1.0)
    # Due to floating errors adjust:
    s = h.sum()
    if abs(s - M) > 1e-9:
        # minor correction: distribute difference uniformly across non-clipped entries.
        # find indices where 0 < h < 1
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.any():
            delta = (M - s) / free.sum()
            h[free] += delta
            # Clip again
            h = np.clip(h, 0.0, 1.0)
    return h

def cap_simplex_proj(v, M):
    # binary search for tau such that sum(clipped(v - tau,0,1)) = M
    low = -1.0
    high = 1.0
    # adjust bounds based on v
    # Actually tau range can be -max(v) to max(v) maybe.
    # To guarantee we bracket root: when tau large positive, all clipped to 0 => sum=0 < M; when tau large negative, all clipped to 1 => sum=n > M.
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(60):
        tau = (low + high) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if s > M:
            low = tau
        else:
            high = tau
    h = np.clip(v - (low+high)/2.0, 0.0, 1.0)
    # small adjustment to fix sum due to rounding
    diff = M - h.sum()
    if abs(diff) > 1e-9:
        # distribute diff across free entries
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothed_max_and_grad(h, alpha):
    n = h.shape[0]
    O = np.correlate(h, 1.0 - h, mode='full')
    maxO = O.max()
    # compute softmax weights
    exp_w = np.exp(alpha * (O - maxO))
    sum_w = exp_w.sum()
    w = exp_w / sum_w
    grad = np.zeros_like(h)
    # compute contributions
    # use vectorized loop over indices
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            i_end = n - k
            if i_end > 0:
                grad[:i_end] += wk * (1 - h[k:])
                grad[k:] -= wk * h[:i_end]
        else:
            kk = -k
            i_end = n - kk
            if i_end > 0:
                grad[kk:] += wk * (1 - h[:i_end])
                grad[:i_end] -= wk * h[kk:]
    # smoothed objective value (without scaling by dx)
    f = np.dot(w, O)
    return f, grad

import time
import numpy as np

# ---- projection onto [0,1] simplex (sum = M) ----
def _proj_cap_simplex(v, M):
    # binary search on tau so that sum(clipped(v - tau,0,1)) = M
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(60):
        tau = (low + high) * 0.5
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if s > M:
            low = tau
        else:
            high = tau
    h = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    # tiny correction for rounding
    diff = M - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

# ---- C5 evaluation (exact) ----
def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

# ---- smoothed max and its gradient (alpha > 0) ----
def _smooth_max_and_grad(h, alpha):
    n = h.shape[0]
    O = np.correlate(h, 1.0 - h, mode="full")
    maxO = O.max()
    w = np.exp(alpha * (O - maxO))
    w_sum = w.sum()
    w = w / w_sum
    grad = np.zeros_like(h)
    # loop over all lags
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)      # shift amount
        if k >= 0:
            end = n - k
            if end > 0:
                grad[:end] += wk * (1.0 - h[k:])
                grad[k:]   -= wk * h[:end]
        else:
            kk = -k
            end = n - kk
            if end > 0:
                grad[kk:]  += wk * (1.0 - h[:end])
                grad[:end] -= wk * h[kk:]
    smooth_val = float(np.dot(w, O))   # no dx, same ordering as _c5_value
    return smooth_val, grad

# ---- generate a feasible start vector ----
def _initial_h(n, rng):
    M = n / 2.0
    method = rng.integers(4)
    if method == 0:               # binary random
        h = np.zeros(n)
        idx = rng.choice(n, int(M), replace=False)
        h[idx] = 1.0
    elif method == 1:             # uniform random then project
        h = rng.random(n)
        h = h / h.sum() * M
        h = np.clip(h, 0.0, 1.0)
        h = _proj_cap_simplex(h, M)
    elif method == 2:             # linear decreasing shape
        lin = np.linspace(1.0, 0.0, n)
        h = lin * (M / np.sum(lin))
        h = np.clip(h, 0.0, 1.0)
        h = _proj_cap_simplex(h, M)
    else:                         # sinusoidal shape
        x = np.arange(n)
        h = (np.sin(2.0 * np.pi * x / n) + 1.0) / 2.0
        h = _proj_cap_simplex(h, M)
    return h

# ---- optimization for a single resolution ----
def _optimise_n(n, rng, budget):
    M = n / 2.0
    best_h = None
    best_c5 = np.inf
    start = time.time()

    parent_h = None
    if "initial_h_values" in globals():
        if len(initial_h_values) == n:
            parent_h = np.array(initial_h_values, dtype=float)
            parent_h = _proj_cap_simplex(parent_h, M)
            c5p = _c5_value(parent_h)
            best_h, best_c5 = parent_h.copy(), c5p

    while time.time() - start < budget - 0.1:
        # pick start point
        if parent_h is not None:
            h = parent_h.copy()
            parent_h = None
        else:
            h = _initial_h(n, rng)

        step = 0.5
        alpha = 1.0

        for _ in range(2000):
            if time.time() - start > budget - 0.2:
                break
            f, g = _smooth_max_and_grad(h, alpha)
            h_new = _proj_cap_simplex(h - step * g, M)
            f_new, _ = _smooth_max_and_grad(h_new, alpha)

            if f_new < f:
                h = h_new
                step = min(step * 1.02, 1.0)
                alpha = min(alpha * 1.005, 200.0)
            else:
                step *= 0.5

        c5 = _c5_value(h)
        if c5 < best_c5 - 1e-12:
            best_c5, best_h = c5, h.copy()
            if best_c5 < 0.3808:
                return best_h, best_c5

    return best_h, best_c5

# ---- main entry point ----
def run(seed=42, budget_s=1100, **kwargs):
    start_all = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    candidate_ns = [80, 96, 112, 128, 144, 160]

    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 not in candidate_ns:
            candidate_ns.insert(0, n0)
        elif n0 % 2 == 0:
            candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_all > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_all)
        alloc = min(200.0, remaining * 0.5)
        h_opt, c5_opt = _optimise_n(n, rng, alloc)
        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global, best_c5_global, best_n_global = h_opt, c5_opt, n
            if best_c5_global < 0.3808:
                break

    # final local random-transfer hill climbing
    if best_h_global is not None:
        n = best_n_global
        M = n / 2.0
        while time.time() - start_all < budget_s - 0.5:
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(1.0 - best_h_global[i], best_h_global[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer * 0.2
            h_new = best_h_global.copy()
            h_new[i] += delta
            h_new[j] -= delta
            h_new = _proj_cap_simplex(h_new, M)
            c5_new = _c5_value(h_new)
            if c5_new + 1e-12 < best_c5_global:
                best_h_global, best_c5_global = h_new, c5_new
    else:
        # fallback constant function
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = _c5_value(best_h_global)
        best_n_global = n

    # final safety projection
    best_h_global = np.clip(best_h_global, 0.0, 1.0)
    M = best_n_global / 2.0
    if abs(best_h_global.sum() - M) > 1e-9:
        best_h_global = _proj_cap_simplex(best_h_global, M)

    return best_h_global, best_c5_global, best_n_global
