def project_feasible(h, target, max_iter=10):
    # ensure feasibility: bound [0,1] and sum=target
    for _ in range(max_iter):
        # compute sum diff
        cur_sum = h.sum()
        diff = target - cur_sum
        if abs(diff) < 1e-12:
            break
        # adjust uniformly
        h += diff / h.size
        # clip to bounds
        np.clip(h, 0.0, 1.0, out=h)
    # final small correction if still off
    cur_sum = h.sum()
    if abs(cur_sum - target) > 1e-9:
        # adjust smallest slack and enforce bounds
        # Use simple water-filling: find indices not at bound to adjust
        # For simplicity just distribute diff to entries not at 0 or 1 proportionally.
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += (target - cur_sum) / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_optimize(h_start, t_start, target_sum, deadline, rng):
    import time, numpy as np
    n = h_start.shape[0]
    dx = 2.0 / n

    # define overlap function
    def overlap_shift(h, k):
        if k >= 0:
            return np.dot(h[:n - k], 1.0 - h[k:]) * dx
        else:
            kk = -k
            return np.dot(h[kk:], 1.0 - h[:n - kk]) * dx

    constraints = []
    # sum equality
    constraints.append({"type":"eq", "fun": lambda x: np.sum(x[:n]) - target_sum})
    # inequality for each shift
    for k in range(-n+1, n):
        if k == 0:
            continue
        constraints.append({"type":"ineq", "fun": lambda x, k=k: x[-1] - overlap_shift(x[:n], k)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    x0 = np.concatenate([h_start, [t_start]])
    try:
        res = minimize(fun=lambda x: x[-1], x0=x0, method="SLSQP", bounds=bounds, constraints=constraints,
                       options={"maxiter": 2000, "ftol":1e-9, "disp":False})
    except Exception:
        return h_start, t_start
    if not res.success:
        return h_start, t_start
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # adjust sum if needed
    h_opt = enforce_sum_and_bounds(h_opt, target_sum)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum_and_bounds(v, target):
    """Project onto [0,1]^n ∩ {sum = target} by simple iterative correction."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        v += diff / v.size
        for _ in range(10):
            v = np.clip(v, 0.0, 1.0)
            diff = target - v.sum()
            if abs(diff) < 1e-12:
                break
            v += diff / v.size
    return v

def random_start_vector(n, rng):
    """Generate a random feasible vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v / v.sum() * target
    v = np.clip(v, 0.0, 1.0)
    v = enforce_sum_and_bounds(v, target)
    return v

def gradient_worst_shift(h, full_corr):
    """Gradient of the worst (max) overlap w.r.t. h."""
    n = h.shape[0]
    k_idx = np.argmax(full_corr)
    k_shift = k_idx - (n - 1)
    g = np.zeros_like(h)
    if k_shift >= 0:
        k = k_shift
        i1 = np.arange(0, n - k)
        g[i1] += 1.0 - h[i1 + k]
        j2 = np.arange(k, n)
        g[j2] += -h[j2 - k]
    else:
        k = -k_shift
        i1 = np.arange(k, n)
        g[i1] += 1.0 - h[i1 - k]
        j2 = np.arange(0, n - k)
        g[j2] += -h[j2 + k]
    return g, k_shift

def hill_climb(h_start, target_sum, deadline, rng):
    """Simple projected gradient descent on the worst overlap."""
    h = h_start.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    it = 0
    while time.time() < deadline:
        full = np.correlate(h, 1.0 - h, mode="full")
        g, _ = gradient_worst_shift(h, full)
        step = 0.05 / (1.0 + it * 0.001)
        h_new = h - step * g
        h_new = enforce_sum_and_bounds(h_new, target_sum)
        new_val = compute_c5(h_new)
        if new_val < cur_val:
            h, cur_val = h_new, new_val
            if new_val < best_val:
                best_h, best_val = h_new.copy(), new_val
                if best_val < 0.3808:
                    break
        else:
            # occasional random jitter to escape plateaus
            if rng.random() < 0.01:
                pert = rng.normal(scale=0.02, size=n)
                h = enforce_sum_and_bounds(h + pert, target_sum)
                cur_val = compute_c5(h)
        it += 1
        if best_val < 0.3808:
            break
    return best_h, best_val

def anneal_refine(h_start, target_sum, deadline, rng):
    """Mass‑transfer simulated annealing focusing on the worst shift."""
    h = h_start.copy()
    cur_val = compute_c5(h)
    n = h.shape[0]
    while time.time() < deadline:
        full = np.correlate(h, 1.0 - h, mode="full")
        k_idx = np.argmax(full)
        k = k_idx - (n - 1)
        if k >= 0:
            valid = np.arange(0, n - k)
            contrib = h[valid] * (1.0 - h[valid + k])
        else:
            kk = -k
            valid = np.arange(kk, n)
            contrib = h[valid] * (1.0 - h[valid - kk])
        if contrib.size == 0:
            break
        probs = contrib / contrib.sum()
        src = rng.choice(valid, p=probs)
        slack = np.where(h < 1.0 - 1e-12)[0]
        if slack.size == 0:
            break
        dst = rng.choice(slack)
        max_tx = min(h[src], 1.0 - h[dst])
        if max_tx <= 0:
            continue
        delta = max_tx * rng.random()
        new_h = h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        new_h = np.clip(new_h, 0.0, 1.0)
        new_val = compute_c5(new_h)
        if new_val < cur_val:
            h, cur_val = new_h, new_val
        else:

            if np.exp((cur_val - new_val) / T) > rng.random():
                h, cur_val = new_h, new_val
    return h, cur_val

def slsqp_optimize(h_start, t_start, target_sum, deadline, rng):
    """One SLSQP refinement with explicit max‑overlap variable."""
    n = h_start.shape[0]
    dx = 2.0 / n

    def overlap_shift(h, k):
        if k >= 0:
            return np.dot(h[: n - k], 1.0 - h[k:]) * dx
        else:
            kk = -k
            return np.dot(h[kk:], 1.0 - h[: n - kk]) * dx

    constraints = [{"type": "eq", "fun": lambda x: np.sum(x[:n]) - target_sum}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        constraints.append(
            {"type": "ineq", "fun": lambda x, k=k: x[-1] - overlap_shift(x[:n], k)}
        )
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_start, [t_start]])
    try:
        res = minimize(
            fun=lambda x: x[-1],
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 1500, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return h_start, t_start
    if not res.success:
        return h_start, t_start
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = enforce_sum_and_bounds(h_opt, target_sum)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
