def compute_c5(h):
    """Maximum overlap value multiplied by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # simple equal-distribution repair
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # incremental fix
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

    dx = 2.0 / n

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.05
    t = 0

    # soft‑max smoothing
    beta_soft = 2.0
    max_beta = 30.0

    no_improve = 0
    start = time.time()
    while time.time() - start < time_budget:
        # correlation and current C5
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = np.max(corr) * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # soft‑max weights
        vmax = corr.max()
        w = np.exp(beta_soft * (corr - vmax))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the smoothed objective
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # zero‑mean to reduce drift (optional)
        g -= np.mean(g)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_sum_and_bounds(h_new, target)

        # accept if improving or by tiny random chance
        new_c5 = compute_c5(h_new)
        if new_c5 < cur_c5 or rng.random() < 0.001:
            h = h_new

        # increase smoothing slowly
        if beta_soft < max_beta:
            beta_soft = min(max_beta, beta_soft * 1.001)

        # random shake when stuck
        if no_improve > 3000:
            h = random_feasible(n, rng)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        # occasional tiny jitter
        if rng.random() < 0.01:
            h = project_sum_and_bounds(h + rng.normal(scale=0.01, size=n), target)

        # safety clip
        h = np.clip(h, 0.0, 1.0)

        if time.time() - start > time_budget - 0.05:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 80
        h0 = random_feasible(n, rng)

    h0 = project_sum_and_bounds(h0, n / 2.0)

    # overall tracking
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)

    min_run_time = 2.0
    run_idx = 0
    while remaining > min_run_time:
        # choose start point: sometimes the current best, sometimes fresh
        if rng.random() < 0.5:
            init = best_h.copy()
        else:
            init = random_feasible(n, rng)

        # allocate a slice of remaining time (up to 30 s)
        time_slice = min(remaining * 0.5, 30.0)
        time_slice = max(min_run_time, time_slice)

        h_opt, c5_opt = optimize_h_adam(init, time_slice, seed + run_idx + 1000)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        elapsed = time.time() - start
        remaining = max(0.0, budget_s - elapsed)
        run_idx += 1

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
