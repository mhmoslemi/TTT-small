def gradient_descent_momentum(h0, n_iter, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        # compute correlation: corr = np.correlate...
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection onto box and sum constraint
        h_new = project_onto_box_sum(h_new, s)
        # clamp within [0,1] already done in projection
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def compute_c5(h):
    N = h.size
    M = int(2**np.ceil(np.log2(2*N-1)))
    # rfft expects integer size
    H = np.fft.rfft(h, M)
    O = np.fft.rfft(1.0 - h, M)
    corr = np.fft.irfft(H * np.conj(O), M)[:2*N-1]
    return np.max(corr) * (2.0/N)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    # trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand search interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def random_binary_balanced(n, rng):
    """Return a balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    N = h.size
    # smallest power of two >= 2*N-1
    M = 1 << ((2 * N - 2).bit_length())
    H = np.fft.rfft(h, M)
    O = np.fft.rfft(1.0 - h, M)
    corr = np.fft.irfft(H * np.conj(O), M)[: 2 * N - 1]
    return np.max(corr) * (2.0 / N)

def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient for the max‑correlation term (averaged over given shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term1[n - shift :] = 0.0
            term2[shift:] = h[: n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term1[:s] = 0.0
            term2[: n - s] = h[s:]
            term2[n - s :] = 0.0
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    N = h.size
    s = N / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (N - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    """Linear interpolation from arr (length m) to length new_n."""
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    # ---------- small grid (256) ----------
    n_small = 256
    target_sum_small = n_small / 2.0

    # initial construction
    if 'initial_h_values' in globals():
        full = np.asarray(initial_h_values, dtype=float)
        init = upsample(full, n_small)
        idx = np.argsort(-init)[: n_small // 2]
        cur_h = np.zeros(n_small, dtype=float)
        cur_h[idx] = 1.0
    else:
        cur_h = random_binary_balanced(n_small, rng)

    alt_h = thue_morse(n_small)
    alt_score = compute_c5(alt_h)
    cur_score = compute_c5(cur_h)
    if alt_score < cur_score:
        cur_h = alt_h
        cur_score = alt_score

    best_h = cur_h.copy()
    best_score = cur_score

    # helper lists for fast swaps
    ones = list(np.where(cur_h == 1)[0])
    zeros = list(np.where(cur_h == 0)[0])
