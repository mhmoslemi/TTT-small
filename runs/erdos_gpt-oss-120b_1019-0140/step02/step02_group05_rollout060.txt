def compute_c5(h):
    """Maximum of ∫ h(x)(1‑h(x+k))dx approximated on the grid."""

            if max_tr <= 0:
                max_tr = min(cur_h[j], 1.0 - cur_h[i])
                if max_tr <= 0:
                    continue
                donor, recv = j, i
            else:
                donor, recv = i, j
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        else:
            cand = gradient_move(cur_h, dx, target_sum)

        new_val = compute_c5(cand)
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0 and rng.random() < np.exp(-(new_val - cur_val) / T):
            accept = True
        if accept:
            cur_h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break

    # -----------------------------------------------------------------
    # optional smoothing passes
    if time.time() - start < budget_s - 5.0:
        for _ in range(15):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
            else:
                break

    # -----------------------------------------------------------------
    # final refinement with SLSQP if time permits
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        n = best_h.shape[0]
        target_sum = n / 2.0
        constraints = [{"type": "eq", "fun": SumEquality(n, target_sum)}]
        for kk in range(-n + 1, n):
            if kk == 0:
                continue
            constraints.append({"type": "ineq", "fun": OverlapIneq(kk, n, 2.0 / n)})
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])
        try:
            res = minimize(
                fun=objective_last,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                cur = np.sum(h_opt)
                if abs(cur - target_sum) > 1e-9:
                    h_opt += (target_sum - cur) / n
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # -----------------------------------------------------------------
    # final safety correction
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = np.sum(best_h)
    target_sum = best_h.shape[0] / 2.0
    if abs(cur - target_sum) > 1e-9:
        best_h += (target_sum - cur) / best_h.shape[0]
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, best_h.shape[0]
