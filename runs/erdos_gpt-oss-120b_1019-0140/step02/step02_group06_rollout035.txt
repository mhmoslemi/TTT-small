def compute_c5(h):

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # move all entries uniformly
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # distribute remaining diff greedily
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

    rng = np.random.default_rng(seed)
    h = h0.copy()

    target = n / 2.0
    dx = 2.0 / n

    max_beta = 20.0
    alpha0 = 0.2        # base step size
    no_improve = 0

        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx

            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase smoothing slowly
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        # soft‑max weights (stable)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth objective
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):

            k = idx - offset
            if k >= 0:

                head = slice(k, n)

        v = np.correlate(h, 1.0 - h, mode="full")
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        s = w.sum()
        if s == 0:
            w.fill(1.0 / w.size)
        else:
            w /= s

        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        g -= np.mean(g)

        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    return best_h, best_c5

def simulated_annealing_binary(n, target_ones, time_budget, seed, start_bin=None):
    """SA on binary vectors with exact sum constraint."""
    rng = np.random.default_rng(seed)
    if start_bin is not None:
        current = start_bin.copy()
    else:
        current = np.zeros(n, dtype=int)
        current[:target_ones] = 1
        rng.shuffle(current)

    cur_c5 = compute_c5(current.astype(float))
    best = current.copy()
    best_c5 = cur_c5

    start = time.time()
    T0 = max(0.1, cur_c5)

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * max(0.001, 1.0 - frac)

        i = rng.integers(n)
        j = rng.integers(n)
        if current[i] == current[j]:
            continue
        current[i] ^= 1
        current[j] ^= 1

        new_c5 = compute_c5(current.astype(float))
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best = current.copy()
        else:
            current[i] ^= 1
            current[j] ^= 1

    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.time()

    if "initial_h_values" in globals():
        h_initial = np.array(initial_h_values, dtype=float)
        n = len(h_initial)
    else:
        n = 80
        h_initial = None

    target_sum = n / 2.0
    target_ones = int(round(target_sum))

    # time split: 55 % SA, remainder gradient descent
    t_sa = budget_s * 0.55
    t_gd = budget_s - t_sa

    # binary start from initial if available
    start_bin = None
    if h_initial is not None:
        idx = np.argsort(-h_initial)[:target_ones]
        start_bin = np.zeros(n, dtype=int)
        start_bin[idx] = 1

    best_bin, best_c5_bin = simulated_annealing_binary(
        n, target_ones, t_sa, seed + 123, start_bin=start_bin
    )

    # gradient‑descent refinement
    h_start = best_bin.astype(float)
    h_start = project_sum_and_bounds(h_start, target_sum)

    remaining = max(0.0, budget_s - (time.time() - start_total))
    best_h_cont, best_c5_cont = optimize_h(h_start, remaining, seed + 456)

    if best_c5_cont < best_c5_bin:
        final_h = best_h_cont
        final_c5 = best_c5_cont
    else:
        final_h = best_bin.astype(float)
        final_c5 = best_c5_bin

    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = project_sum_and_bounds(final_h, target_sum)
    final_c5 = compute_c5(final_h)

    return final_h, final_c5, n
