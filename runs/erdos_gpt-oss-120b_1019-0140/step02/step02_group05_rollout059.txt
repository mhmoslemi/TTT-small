def enforce_sum_and_bounds(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # free entries where not at bounds
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
        # residual
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def enforce_sum_and_bounds(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc

            diff = target_sum - np.sum(h)

            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add

                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_for_shift(h, k, dx):
    n = len(h)
    grad = np.zeros(n)

        grad[:n - k] = 1.0 - h[k:]
        grad[k:] -= h[:n - k]

        diff = target_sum - cur_sum
        if abs(diff) > 1e-9:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and best_h[i] < 1.0:
                    add = min(diff, 1.0 - best_h[i])
                    best_h[i] += add
                    diff -= add
                elif diff < 0 and best_h[i] > 0.0:
                    sub = min(-diff, best_h[i])
                    best_h[i] -= sub
                    diff += sub
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, n
