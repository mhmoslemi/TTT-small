def simulated_annealing(h, budget_s, start_time):
    n = len(h)
    M = n // 2
    dx = 2.0/n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    T0 = 0.01
    T = T0
    # Compute number of iterations based on time
    while time.time() - start_time < budget_s:
        # Choose two distinct indices
        i, j = rng.choice(n, 2, replace=False)
        # Determine max delta
        # Want to move mass from j to i
        max_inc = min(1.0 - cur_h[i], cur_h[j])
        max_dec = min(cur_h[i], 1.0 - cur_h[j]) # moving opposite direction
        # Choose direction randomly
        if rng.random() < 0.5:
            max_delta = max_inc
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
        else:
            max_delta = max_dec
            if max_delta <= 0:
                continue
            delta = -rng.random() * max_delta
        # Apply move
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # Compute new C5
        new_c5 = compute_c5(new_h)
        dC = new_c5 - cur_c5
        if dC < 0 or rng.random() < np.exp(-dC / T):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        # Cooling schedule
        # we could update T each iteration: T = T0 * (1 - (elapsed / budget_s)) maybe.
        elapsed = time.time() - start_time
        T = T0 * max(0.001, 1.0 - elapsed / budget_s) # linear cooling
    return best_h, best_c5

def init_h(n, rng):
    M = n // 2
    # try to use parent
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        if len(parent) == n:
            h = parent.copy()
        else:
            # upsample using linear interpolation
            old_n = len(parent)
            old_x = np.linspace(0, 2, old_n, endpoint=False)
            new_x = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(new_x, old_x, parent)
    else:
        # sin^2 shape
        x = np.linspace(0, 2, n, endpoint=False)
        h = np.sin(np.pi * x / 2) ** 2  # sin^2 from 0 to 1 to 0
    # ensure within [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust sum exactly
    cur = h.sum()
    diff = M - cur
    if abs(diff) > 1e-12:
        # simple correction: add diff/n equally, then clip again
        h += diff / n
        # clip and re-balance if needed
        h = np.clip(h, 0.0, 1.0)
        # recompute and adjust remaining diff
        cur = h.sum()
        diff2 = M - cur
        if abs(diff2) > 1e-12:
            # distribute remaining diff among non-bound entries iteratively
            for _ in range(5):
                mask = (h > 0) & (h < 1)
                if not np.any(mask):
                    break
                adjust = diff2 / np.count_nonzero(mask)
                h[mask] += adjust
                h = np.clip(h, 0.0, 1.0)
                cur = h.sum()
                diff2 = M - cur
                if abs(diff2) < 1e-12:
                    break
    return h

def anneal(h0, time_budget, start_time, rng):

    dx = 2.0 / n

    T0 = 0.005  # starting temperature
    Tmin = 1e-7
    # cooling factor per iteration; compute based on budget; derive target number of iterations ~ 200k maybe
    # We'll compute cooling per iteration: T = max(Tmin, T0 * (Tmin/T0) ** (iter / max_iter))
    max_iter = int(time_budget * 200)  # assume 200 iterations per second approx; adjust dynamic
    if max_iter < 1:
        max_iter = 1
    for it in range(max_iter):
        # check time
        if time.time() - start_time > start_time + time_budget:  # unrealistic guard
            break
        # Choose random pair
        i, j = rng.choice(n, 2, replace=False)

        # Move mass from j to i (delta positive) or from i to j (delta negative)
        if rng.random() < 0.5:
            max_delta = min(1.0 - cur_h[i], cur_h[j])
            if max_delta <= 0:

            delta = rng.random() * max_delta

                continue

        new_h[i] += delta
        new_h[j] -= delta
        new_c5 = compute_c5(new_h)
        dC = new_c5 - cur_c5
        # Temperature

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    while time.time() < time_limit:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)

        if k == 0:

            # compute second max

        g = np.zeros(n)

                g[i] += 1 - h[i + k]
                g[i + k] -= h[i]

                j = i + kk
                # overl for negative shift: overl(-kk) = sum_i h[i+kk] * (1 - h[i])

        neg = np.argmin(g)  # most negative gradient: want to increase

            h[j] -= delta

        if max_transfer <= 0:

        # compute new c5

    cur_h = h.copy()

    best_h = cur_h.copy()
    best_c5 = cur_c5
    while time.time() < time_limit:

        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        if k == 0:
            # find second worst
            temp_corr = corr.copy()
            temp_corr[max_idx] = -np.inf
            max_idx = int(np.argmax(temp_corr))
            k = max_idx - (n - 1)
        # compute gradient
        g = np.zeros(n)
        if k >= 0:
            kk = k
            i = np.arange(n - kk)
            g[i] += 1.0 - cur_h[i + kk]

        elapsed = time.time() - start_time

        step_factor = 0.5 * (1.0 - progress) + 0.01
        if max_transfer > 0 and g[pos] > 1e-12 and g[neg] < -1e-12:

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def initial_h(n, rng):
    """Construct a feasible initial h of length n."""
    M = n // 2
    parent = globals().get("initial_h_values")
    if parent is not None:
        parent = np.asarray(parent, dtype=float)
        if len(parent) == n:
            h = parent.copy()
        else:
            old_n = len(parent)
            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(new_x, old_x, parent)
    else:
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.sin(np.pi * x / 2.0) ** 2
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = M - cur
    if abs(diff) > 1e-12:
        h += diff / n
        for _ in range(5):
            h = np.clip(h, 0.0, 1.0)
            cur = h.sum()
            diff = M - cur
            if abs(diff) < 1e-12:
                break
            h += diff / n
    return h

def improve_h(start_h, run_start, time_limit, rng):
    """Heuristic improvement using gradient‑guided mass transfers."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    while time.time() < time_limit:
        # worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        # gradient of the worst overlap w.r.t. h
        g = np.zeros(n)
        if k >= 0:
            kk = k
            i = np.arange(n - kk)
            g[i] += 1.0 - cur_h[i + kk]
            g[i + kk] -= cur_h[i]
        else:
            kk = -k
            i = np.arange(n - kk)
            g[i] -= cur_h[i + kk]
            g[i + kk] += 1.0 - cur_h[i]

        pos = int(np.argmax(g))   # positive gradient → should be decreased
        neg = int(np.argmin(g))   # negative gradient → should be increased

        # step size that shrinks as time runs out
        elapsed = time.time() - run_start
        total = time_limit - run_start
        prog = elapsed / total if total > 0 else 0.0
        base = 0.5 * (1.0 - prog) + 0.01

            new_h[pos] -= delta
            new_h[neg] += delta

            max_inc = min(1.0 - cur_h[i], cur_h[j])
            if max_inc <= 0:
