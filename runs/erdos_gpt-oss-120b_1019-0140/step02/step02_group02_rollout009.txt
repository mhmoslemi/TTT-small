                if abs(diff2) < 1e-12:

                if diff2 > 0 and h[i] < 1.0:
                    inc = min(diff2, 1.0 - h[i])
                    h[i] += inc
                    diff2 -= inc
                elif diff2 < 0 and h[i] > 0.0:
                    dec = min(-diff2, h[i])
                    h[i] -= dec
                    diff2 += dec
    return h

def _local_optimize(start_h, budget, rng):
    """Improved annealing that moves mass from low‑neighbor to high‑neighbor indices."""
    n = start_h.shape[0]
    target = n / 2.0
    dx = 2.0 / n
    best_h = start_h.copy()
    best_val = _compute_c5(best_h)
    cur_h = start_h.copy()
    cur_val = best_val
    start_time = time.time()
    deadline = start_time + budget
    # temperature schedule parameters
    T0 = 0.01
    Tend = 1e-7
    eps = 1e-12
    while time.time() < deadline - 0.001:
        # auto‑correlation and worst shift
        auto = np.correlate(cur_h, cur_h, mode='full')
        center = n - 1
        auto[center] = np.inf
        min_idx = np.argmin(auto)
        k = min_idx - center
        # neighbor sums for this shift
        neighbor = np.zeros(n)
        if k > 0:
            neighbor[:n - k] += cur_h[k:]
            neighbor[k:] += cur_h[:n - k]
        elif k < 0:
            kk = -k
            neighbor[kk:] += cur_h[:n - kk]
            neighbor[:n - kk] += cur_h[kk:]
        else:
            continue
        # choose source (low neighbor) and target (high neighbor)
        src_cand = np.where(cur_h > eps)[0]
        tgt_cand = np.where(cur_h < 1 - eps)[0]
        if src_cand.size == 0 or tgt_cand.size == 0:
            break
        src_vals = neighbor[src_cand]
        tgt_vals = neighbor[tgt_cand]
        src_idx = src_cand[np.argmin(src_vals)]
        tgt_idx = tgt_cand[np.argmax(tgt_vals)]
        if src_idx == tgt_idx:
            # fallback random pair
            i, j = rng.choice(n, 2, replace=False)
            src_idx, tgt_idx = (i, j) if cur_h[i] > cur_h[j] else (j, i)
        max_transfer = min(cur_h[src_idx], 1 - cur_h[tgt_idx])
        if max_transfer <= 0:
            # try a random small move
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(cur_h[i], 1 - cur_h[j])
            if max_transfer <= 0:
                continue
            src_idx, tgt_idx = i, j
        delta = max_transfer * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[src_idx] -= delta
        new_h[tgt_idx] += delta
        new_val = _compute_c5(new_h)
        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
                if best_val < 0.3808:
                    return best_h, best_val
        else:
            # annealing acceptance
            remaining = deadline - time.time()
            total = deadline - start_time
            T = T0 * (remaining / total) + Tend
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_h, cur_val = new_h, new_val
        # occasional random perturb to avoid stalling
        if rng.random() < 0.001:
            i, j = rng.choice(n, 2, replace=False)
            max_tr = min(cur_h[i], 1 - cur_h[j])
            if max_tr > 0:
                d = max_tr * rng.random() * 0.3
                cur_h[i] -= d
                cur_h[j] += d
                cur_val = _compute_c5(cur_h)
                if cur_val < best_val:
                    best_h, best_val = cur_h.copy(), cur_val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # choose resolution
    n = 200
    target = n / 2.0

    # build initial h
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = init.shape[0]
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(xs, x0, init)
    else:
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    # main multi‑start loop
    while time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        # decide start point
        if rng.random() < 0.25:
            # fresh random binary start
            start_h = np.zeros(n, dtype=float)
            start_h[rng.choice(n, n // 2, replace=False)] = 1.0
        else:
            # perturb current best
            start_h = best_h.copy()
            i, j = rng.choice(n, 2, replace=False)
            max_tr = min(start_h[i], 1 - start_h[j])
            if max_tr > 0:
                d = max_tr * rng.random() * 0.2
                start_h[i] -= d
                start_h[j] += d
        start_h = np.clip(start_h, 0.0, 1.0)
        start_h = _enforce_sum(start_h, target, rng)

        # allocate a chunk of time to local optimisation
        local_budget = min(8.0, remaining * 0.9)
        loc_h, loc_val = _local_optimize(start_h, local_budget, rng)
        if loc_val < best_val:
            best_h, best_val = loc_h.copy(), loc_val
            if best_val < 0.3808:
                break

    # final SLSQP refinement
    # prepare variables with slack t
    dx = 2.0 / n
    def overlap_k(h, k):
        if k >= 0:
            return np.dot(h[: n - k], 1.0 - h[k:]) * dx
        else:
            kk = -k
            return np.dot(h[kk:], 1.0 - h[: n - kk]) * dx

    # constraints
    constraints = []
    # equality: sum h = target
    constraints.append({"type": "eq", "fun": lambda x, t=target: np.sum(x[:n]) - t})
    # inequality for each shift
    for kk in range(-n + 1, n):
        if kk == 0:
            continue
        def cons_factory(k):
            return lambda x, k=k: x[-1] - overlap_k(x[:n], k)
        constraints.append({"type": "ineq", "fun": cons_factory(kk)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    x0 = np.concatenate([best_h, [best_val]])
    try:
        res = minimize(
            fun=lambda x: x[-1],
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 800, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = _enforce_sum(h_opt, target, rng)
            val_opt = _compute_c5(h_opt)
            if val_opt < best_val:
                best_h, best_val = h_opt, val_opt
    except Exception:
        pass

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _enforce_sum(best_h, target, rng)
    final_val = _compute_c5(best_h)

    return best_h.astype(float), final_val, n
