def compute_c5(arr):
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / len(arr))

def project_onto_cap_simplex(v, target_sum):
    # v may be any real vector
    n = len(v)
    # initial bounds
    lo = -np.max(v) - 1.0  # guarantee v+lo <= -1, clipped to 0
    hi = 1.0 - np.min(v) + 1.0  # guarantee v+hi >= 1, clipped to 1
    # binary search for lambda achieving target sum
    for _ in range(40):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + hi, 0.0, 1.0)
    # adjust residual (tiny)
    residual = target_sum - w.sum()
    if abs(residual) > 1e-12:
        # distribute residual to non-saturated entries
        # find indices where w not at bound
        free = np.where((w > 0) & (w < 1))[0]
        if free.size > 0:
            # distribute equally among free
            per = residual / free.size
            w[free] += per
            # clip again
            w = np.clip(w, 0.0, 1.0)
            # recalc residual; if still not zero, adjust iteratively
            # but usually will be tiny
        # If still not zero, adjust by adjusting any entries
        residual = target_sum - w.sum()
        if abs(residual) > 1e-12:
            for i in range(n):
                if residual > 0 and w[i] < 1.0:
                    step = min(residual, 1.0 - w[i])
                    w[i] += step
                    residual -= step
                elif residual < 0 and w[i] > 0.0:
                    step = min(-residual, w[i])
                    w[i] -= step
                    residual += step
                if abs(residual) < 1e-12:
                    break
    return w

def es_optimize(initial_h, target_sum, time_end, rng):
    n = len(initial_h)
    best = project_onto_cap_simplex(initial_h, target_sum)
    best_c5 = compute_c5(best)
    global_best = best.copy()
    global_best_c5 = best_c5
    sigma = 0.2
    stagnation = 0
    lambda_offspring = max(10, n // 5)  # around 16
    while time.time() < time_end - 0.2:
        improved = False
        # generate offspring
        for _ in range(lambda_offspring):
            cand = best + sigma * rng.standard_normal(n)
            cand = project_onto_cap_simplex(cand, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5 - 1e-12:

                best_c5 = cand_c5
                improved = True

                global_best = cand.copy()

def binary_hill_climb(start_bin, time_end, rng):
    # start_bin: binary 0/1 map that sum = target_sum (i.e., n/2)
    n = len(start_bin)
    target_sum = n // 2
    best_bin = start_bin.copy()
    best_c5 = compute_c5(best_bin.astype(float))
    # hill climbing: try random swaps
    no_improve = 0
    # limit iterations to time or large number
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(best_bin)
        zeros = np.flatnonzero(1 - best_bin)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_bin.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_bin = cand
            best_c5 = cand_c5
            no_improve = 0
        else:
            no_improve += 1
    return best_bin.astype(float), best_c5

        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + hi, 0.0, 1.0)
    # tiny residual correction
    residual = target_sum - w.sum()
    if abs(residual) > 1e-12:
        free = np.where((w > 0.0) & (w < 1.0))[0]
        if free.size:
            per = residual / free.size
            w[free] += per
            w = np.clip(w, 0.0, 1.0)
        residual = target_sum - w.sum()
        if abs(residual) > 1e-12:
            for i in range(n):
                if residual > 0 and w[i] < 1.0:
                    step = min(residual, 1.0 - w[i])
                    w[i] += step
                    residual -= step
                elif residual < 0 and w[i] > 0.0:
                    step = min(-residual, w[i])
                    w[i] -= step
                    residual += step
                if abs(residual) < 1e-12:
                    break
    return w

def es_optimize(start_h, target_sum, time_end, rng):
    """Simple (1,λ)-ES with projection onto capped simplex."""
    n = start_h.shape[0]
    best = project_onto_cap_simplex(start_h, target_sum)
    best_c5 = compute_c5(best)
    global_best = best.copy()
    global_best_c5 = best_c5
    sigma = 0.2
    stagnation = 0
    lam = max(20, n // 4)

    while time.time() < time_end - 0.1:
        improved = False
        for _ in range(lam):
            cand = best + sigma * rng.standard_normal(n)
            cand = project_onto_cap_simplex(cand, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 + 1e-12 < best_c5:
                best = cand
                best_c5 = cand_c5
                improved = True
            if cand_c5 + 1e-12 < global_best_c5:
                global_best = cand.copy()
                global_best_c5 = cand_c5
        # sigma adaptation
        if improved:
            sigma = min(sigma * 1.2, 1.0)
            stagnation = 0
        else:
            sigma *= 0.9
            stagnation += 1
        # restart if stuck
        if stagnation >= 60:
            best = project_onto_cap_simplex(rng.random(n), target_sum)
            best_c5 = compute_c5(best)
            sigma = 0.2
            stagnation = 0
        # early exit
        if global_best_c5 <= 0.3808:
            break
    return global_best, global_best_c5

def continuous_improve(h, target_sum, time_end, rng):
    """Greedy pairwise mass transfer that preserves sum."""
    n = h.shape[0]
    best = h.copy()
    best_c5 = compute_c5(best)
    while time.time() < time_end:
        inc = np.where(best < 1.0 - 1e-12)[0]
        dec = np.where(best > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1.0 - best[i], best[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best = cand
            best_c5 = cand_c5
    return best, best_c5

def binary_hill_climb(start_bin, time_end, rng):
    """Swap-based hill climbing on binary vectors (exact half‑ones)."""
    n = start_bin.shape[0]
    best = start_bin.copy()
    best_c5 = compute_c5(best.astype(float))
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best = cand
            best_c5 = cand_c5
            no_improve = 0
        else:
            no_improve += 1
    return best.astype(float), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = rng.random(n)

    target_sum = n / 2.0
    h0 = project_onto_cap_simplex(h0, target_sum)

    # primary ES optimisation
    time_end = start_time + budget_s - 1.0
    best_h, best_c5 = es_optimize(h0, target_sum, time_end, rng)

    # secondary continuous greedy improvement (if time permits)
    now = time.time()
    if now < start_time + budget_s - 0.7:
        cont_end = start_time + budget_s - 0.6
        cand_h, cand_c5 = continuous_improve(best_h, target_sum, cont_end, rng)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h, cand_c5

    # final binary hill climb from rounded version (if time permits)
    now = time.time()
    if now < start_time + budget_s - 0.3:
        bin_start = (best_h >= 0.5).astype(int)
        cur = bin_start.sum()
        diff = int(target_sum - cur)
        if diff > 0:
            zeros = np.flatnonzero(1 - bin_start)
            sel = rng.choice(zeros, diff, replace=False)
            bin_start[sel] = 1
        elif diff < 0:
            ones = np.flatnonzero(bin_start)
            sel = rng.choice(ones, -diff, replace=False)
            bin_start[sel] = 0
        final_h, final_c5 = binary_hill_climb(bin_start, start_time + budget_s - 0.1, rng)
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5

    # final safety clip & exact sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - target_sum) > 1e-12:
        best_h = project_onto_cap_simplex(best_h, target_sum)

    return best_h, best_c5, n
