def compute_c5(h, dx):
    """Maximum overlap integral for a given h."""
    # np.correlate implements linear cross‑correlation (full mode)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Adjust vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    # greedy fix
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible vector (values in [0,1] and exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def simulated_annealing_continuous(start_h, dx, target_sum, deadline, rng):
    """SA on continuous h, preserving sum and bounds."""
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    T0 = max(cur_val * 0.05, 1e-4)  # initial temperature
    start_local = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_local
        total = deadline - start_local
        frac = elapsed / total if total > 0 else 1.0
        T = T0 * (1.0 - frac)

        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5(cand, dx)
        dval = cand_val - cur_val
        accept = False
        if dval < 0:
            accept = True
        else:
            if T > 0:
                if rng.random() < np.exp(-dval / T):
                    accept = True
        if accept:
            cur_h = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val
                if best_val < 0.3808:
                    break
    return best_h, best_val


def binary_hill_climb(h_bin, cur_val, dx, deadline, rng):
    """Swap‑based hill climbing on a binary vector."""
    best_h = h_bin.copy()
    best_val = cur_val
    while time.time() < deadline:
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best_h = cand
            best_val = cand_val
            if best_val < 0.3808:
                break
    return best_h.astype(float), best_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        target_sum = n / 2.0
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_sum(h0, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = generate_random_feasible(n, target_sum, rng)

    dx = 2.0 / n

    best_h = h0.copy()
    best_val = compute_c5(best_h, dx)

    # Main search: many short SA runs
    while True:
        now = time.time()
        time_left = budget_s - (now - start_time)
        if time_left < 5.0:
            break
        run_time = min(30.0, time_left * 0.7)

        # Choose start point
        if rng.random() < 0.6 and best_val is not None:
            start_h = best_h.copy()
            # small perturbation to escape possible plateau
            for _ in range(3):
                donors = np.where(start_h > 1e-12)[0]
                recipients = np.where(start_h < 1.0 - 1e-12)[0]
                if donors.size == 0 or recipients.size == 0:
                    break
                i = rng.choice(donors)
                j = rng.choice(recipients)
                max_delta = min(start_h[i], 1.0 - start_h[j])
                if max_delta <= 0:
                    continue
                delta = rng.uniform(0.0, max_delta)
                start_h[i] -= delta
                start_h[j] += delta
        else:
            start_h = generate_random_feasible(n, target_sum, rng)

        cand_h, cand_val = simulated_annealing_continuous(
            start_h, dx, target_sum, now + run_time, rng
        )
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val
            if best_val < 0.3808:
                break

    # Optional binary refinement
    time_left = budget_s - (time.time() - start_time)
    if time_left > 6.0:
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argsort(-best_h)[: int(target_sum)]
        binary_h[top_idx] = 1
        bin_val = compute_c5(binary_h.astype(float), dx)
        bin_h, bin_val = binary_hill_climb(
            binary_h, bin_val, dx, start_time + budget_s - 2.0, rng
        )
        if bin_val < best_val:
            best_h, best_val = bin_h, bin_val

    # Final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5(best_h, dx)

    return best_h, best_val, n
