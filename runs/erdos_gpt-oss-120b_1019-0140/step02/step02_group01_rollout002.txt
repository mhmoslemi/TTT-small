                best_c5 = new_c5

import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def thue_morse(n):
    # generate TM of length >= n then slice
    k = int(np.ceil(np.log2(n)))
    a = np.array([0.0])
    for _ in range(k):
        a = np.concatenate([a, 1.0 - a])
    return a[:n]
