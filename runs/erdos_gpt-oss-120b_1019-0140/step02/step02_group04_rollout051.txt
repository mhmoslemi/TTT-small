import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Adjust vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible vector (values in [0,1] and exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def refine_advanced(h_start, dx, target_sum, time_limit, rng):
    """Targeted local search improving the worst autocorrelation."""
    n = len(h_start)
    h = project_to_sum(h_start, target_sum)
    # initial correlation and value
    c = np.correlate(h, h, mode="full")
    c_center = c.copy()
    c_center[n - 1] = np.inf  # ignore zero lag
    min_c = np.min(c_center)
    cur_val = (target_sum - min_c) * dx
    step = 0.1
    eps = 1e-12
    while time.time() < time_limit:
        # locate worst (minimum) autocorrelation lag
        idx = np.argmin(c_center)
        k = idx - (n - 1)  # shift value, k != 0 because we masked it
        # gradient w.r.t. each entry for this lag
        grad = np.zeros(n)
        if k > 0:
            grad[: n - k] += h[k:]
            grad[k:] += h[: n - k]
        elif k < 0:
            kk = -k
            grad[kk:] += h[: n - kk]
            grad[: n - kk] += h[kk:]
        # select donor (low gradient, positive mass) and receiver (high gradient, room)
        donors = np.where(h > eps)[0]
        receivers = np.where(h < 1.0 - eps)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        donor = donors[np.argmin(grad[donors])]
        receiver = receivers[np.argmax(grad[receivers])]
        max_delta = min(step, h[donor], 1.0 - h[receiver])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = h.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        cand_c = np.correlate(cand, cand, mode="full")
        cand_c_center = cand_c.copy()
        cand_c_center[n - 1] = np.inf
        cand_min_c = np.min(cand_c_center)
        cand_val = (target_sum - cand_min_c) * dx
        if cand_val < cur_val - 1e-12:
            h = cand
            c = cand_c
            c_center = cand_c_center
            min_c = cand_min_c
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    # final safety projection
    h = project_to_sum(h, target_sum)
    cur_val = compute_c5_vec(h, dx)
    return h, cur_val


def binary_hill_climb(h_bin, cur_val, dx, budget_s, start_time, rng):
    """Swap based hill climbing on a binary (0/1) vector."""
    best = h_bin.copy()
    best_val = cur_val
    n = len(best)
    while time.time() - start_time < budget_s - 1.0:
        ones = np.flatnonzero(best > 0.5)
        zeros = np.flatnonzero(best < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = compute_c5_vec(cand, dx)
        if val < best_val - 1e-12:
            best = cand
            best_val = val
            if best_val < 0.3808:
                break
    return best, best_val

            # perturb current best
            h0 = best_h + rng.normal(scale=0.01, size=n)
            h0 = np.clip(h0, 0.0, 1.0)
        elif "initial_h_values" in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
        else:
            h0 = generate_random_feasible(n, target_sum, rng)

        # allocate time for this refinement
        time_for_refine = min(5.0, remaining * 0.6)
        t_limit = time.time() + time_for_refine
        h_ref, val_ref = refine_advanced(h0, dx, target_sum, t_limit, rng)

        if val_ref < best_val - 1e-12:
            best_h, best_val, best_n = h_ref, val_ref, n
            if best_val < 0.3808:
                # good enough; keep searching a bit for potential improvement
                pass

    # optional binary rounding and hill climbing on the best size found
    if best_h is not None:
        n = best_n
        dx = 2.0 / n
        binary_h = np.zeros(n, dtype=float)
        top_idx = np.argpartition(-best_h, int(n / 2))[: int(n / 2)]
        binary_h[top_idx] = 1.0
        bin_val = compute_c5_vec(binary_h, dx)
        binary_h, bin_val = binary_hill_climb(binary_h, bin_val, dx, budget_s, start_time, rng)
        if bin_val < best_val - 1e-12:
            best_h, best_val = binary_h, bin_val

    # final enforcement of constraints
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_val = compute_c5_vec(best_h, 2.0 / best_n)

    return best_h, best_val, best_n
