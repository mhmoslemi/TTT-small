```python
def compute_gradient_for_shift(h, shift):
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        # term1 = h[shift:] for positions i where i+shift exists?
        # correlation term: sum_i h[i] * (1 - h[i+shift])
        # derivative w.r.t h[i] = (1 - h[i+shift]) (for i such that i+shift < n)
        # derivative w.r.t h[i+shift] = -h[i] (since derivative of -h[i]*h[i+shift] w.r.t h[i+shift] is -h[i])
        # So gradient[i] includes (1 - h[i+shift]) if shift forward.
        # For indices i >= n - shift, term missing (outside range). So handle accordingly.
        grad[:n - shift] = 1.0 - h[shift:]
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        # shift negative: correlation sum_i h[i] * (1 - h[i - s])
        # gradient[i] includes (1 - h[i - s]) for i >= s, and -h[i + s] for i < n - s
        grad[s:] = 1.0 - h[:n - s]
        grad[:n - s] -= h[s:]
    return grad
```

```python
def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # index of max
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient_for_shift(h, shift)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)  # schedule
        h_new = h - eta * momentum
        # project onto box and sum constraint
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

```python
def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=100):
    n = v.shape[0]
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # adjust bounds if needed
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    # final
    return np.clip(v - lam_mid, lo, hi)
```

```python
def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)
```

import time
import numpy as np

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = target}."""
    n = v.shape[0]
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target) < 1e-12:
            return proj
        if s > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral on the discretised grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradient_for_shift(h, shift):
    """Gradient of correlation at a given shift w.r.t. h."""
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[: n - shift] = 1.0 - h[shift:]
        grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        grad[s:] = 1.0 - h[: n - s]
        grad[: n - s] -= h[s:]
    return grad

def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient_for_shift(h, shift)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse_seq(m):
    """Binary Thue–Morse sequence of length m (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(m)), dtype=float, count=m)

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    """Linear interpolation from arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 256                     # grid size
    target_sum = n / 2.0

    # ----- helpers -------------------------------------------------
    def update_best(hc):
        nonlocal best_h, best_score
        sc = compute_c5(hc)
        if sc < best_score:
            best_h, best_score = hc.copy(), sc

    # ----- initial candidates ---------------------------------------
    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)

    candidates.append(project_onto_box_sum(thue_morse_seq(n), target_sum))
    for _ in range(4):
        candidates.append(random_binary_balanced(n, rng))

    # ----- stochastic gradient descent on several starts ----------
    best_h = None
    best_score = np.inf

    while time_left() > 12.0:

        # allocate iterations proportionally to remaining time
