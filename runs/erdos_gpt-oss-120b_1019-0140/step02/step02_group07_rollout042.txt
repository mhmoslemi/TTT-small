def solve_milp(n, time_limit, rng):
    import cvxpy as cp
    weight = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(nonneg=True)
    constraints = [cp.sum(h) == weight]
    # add shift constraints
    for k in range(1, n):
        length = n - k
        if length == 0:
            continue
        x = cp.Variable(length, boolean=True)
        # constraints linking x with h
        constraints += [x <= h[:length]]
        constraints += [x <= h[k:]]
        constraints += [x >= h[:length] + h[k:] - 1]
        # correlation constraint: t >= sum(h[:length] - x)
        constraints.append(t >= cp.sum(h[:length] - x))
    # objective
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    # solve
    prob.solve(solver=cp.CBC, maximumSeconds=time_limit, verbose=False)

    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        raise RuntimeError(f"Solver status {prob.status}")
    h_val = np.rint(h.value).astype(float)
    # ensure sum matches exactly
    cur_sum = np.sum(h_val)
    if cur_sum != weight:
        # adjust minor rounding issues
        diff = weight - cur_sum
        # if diff>0, set some zeros to 1
        zeros = np.where(h_val == 0)[0]
        if diff > 0:
            idx = rng.choice(zeros, int(diff), replace=False)
            h_val[idx] = 1
        else:
            ones = np.where(h_val == 1)[0]
            idx = rng.choice(ones, int(-diff), replace=False)
            h_val[idx] = 0
    c5 = compute_c5(h_val)
    return h_val, c5

def solve_milp_binary(n, target_sum, time_limit, rng):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(nonneg=True)
    constraints = [cp.sum(h) == target_sum]
    for k in range(1, n):
        length = n - k
        x = cp.Variable(length, boolean=True)
        # linking
        constraints.append(x <= h[:length])
        constraints.append(x <= h[k:])
        constraints.append(x >= h[:length] + h[k:] - 1)
        # correlation constraint
        constraints.append(t >= cp.sum(h[:length] - x))
    prob = cp.Problem(cp.Minimize(t), constraints)
    # Solve with CBC: may support boolean variables.
    prob.solve(solver=cp.CBC, maximumSeconds=time_limit, verbose=False)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        # maybe return best feasible anyway if any
        if h.value is None:
            raise RuntimeError("MILP failed")
    # extract binary solution
    h_val = np.rint(np.array(h.value)).astype(float)
    # adjust rounding if necessary
    cur = np.sum(h_val)
    if cur != target_sum:
        diff = target_sum - cur
        if diff > 0:
            zeros = np.where(h_val == 0)[0]
            if len(zeros) >= diff:
                idx = rng.choice(zeros, int(diff), replace=False)
                h_val[idx] = 1

            ones = np.where(h_val == 1)[0]
            if len(ones) >= -diff:
                idx = rng.choice(ones, int(-diff), replace=False)
                h_val[idx] = 0
    c5 = compute_c5(h_val)
    return h_val, c5

import time, math, numpy as np, cvxpy as cp

def compute_c5(arr):
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def continuous_anneal(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 300000
    it = 0
    while time.time() < time_end and it < max_iter:
        # pick two indices to transfer mass
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # final correction to fix sum exactly
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

def binary_hill_climb(start_h, time_end, rng):
    # start_h may be float; convert to binary
    bin_vec = (np.array(start_h) >= 0.5).astype(int)
    n = len(bin_vec)
    best = bin_vec.copy()
    best_c5 = compute_c5(best.astype(float))
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best = cand
            best_c5 = cand_c5
            no_improve = 0
        else:
            no_improve += 1
    return best.astype(float), best_c5

def solve_milp_binary(n, time_limit, rng):
    weight = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(nonneg=True)
    constraints = [cp.sum(h) == weight]
    for k in range(1, n):
        length = n - k
        if length == 0:
            continue
        x = cp.Variable(length, boolean=True)
        constraints.append(x <= h[:length])
        constraints.append(x <= h[k:])
        constraints.append(x >= h[:length] + h[k:] - 1)
        constraints.append(t >= cp.sum(h[:length] - x))
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.CBC, maximumSeconds=time_limit, verbose=False)
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        raise RuntimeError(f"Solver failed: {prob.status}")
    h_val = np.rint(np.array(h.value)).astype(float)
    cur = np.sum(h_val)
    if cur != weight:
        diff = weight - cur
        if diff > 0:
            zeros = np.where(h_val == 0)[0]
            if len(zeros) >= diff:
                idx = rng.choice(zeros, int(diff), replace=False)
                h_val[idx] = 1
        else:
            ones = np.where(h_val == 1)[0]
            if len(ones) >= -diff:
                idx = rng.choice(ones, int(-diff), replace=False)
                h_val[idx] = 0
    c5 = compute_c5(h_val)
    return h_val, c5

        h0 = np.clip(h0, 0.0, 1.0)
        residual = target_sum - np.sum(h0)
        for i in range(n):

            adj = np.clip(residual, -h0[i], 1 - h0[i])
            h0[i] += adj
            residual = target_sum - np.sum(h0)
    # try MILP binary search if time permits
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30:  # need at least some time
        milp_time = min(remaining * 0.6, 600)
        try:
            h_bin, c5_bin = solve_milp_binary(n, milp_time, rng)
            if c5_bin <= 0.3808:
                return h_bin, c5_bin, n
        except Exception:
            pass
    # fallback to continuous annealing
    time_end = start_time + budget_s - 0.5
    h_cont, c5_cont = continuous_anneal(h0, time_end, rng)
    # optional final binary hill climb if time left
    if time.time() < time_end:
        bin_start = (h_cont >= 0.5).astype(int)
        h_final, c5_final = binary_hill_climb(bin_start.astype(float), time_end, rng)
        if c5_final < c5_cont:

    # final exact-sum enforcement
    diff = target_sum - np.sum(h_cont)
    if abs(diff) > 1e-12:
        h_cont += diff / n
        h_cont = np.clip(h_cont, 0.0, 1.0)

        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -h_cont[i], 1 - h_cont[i])
            h_cont[i] += adj
            residual = target_sum - np.sum(h_cont)
    c5_val = compute_c5(h_cont)
    return h_cont, c5_val, n

import time, math, numpy as np, cvxpy as cp

def compute_c5(arr):
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def continuous_anneal(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 300000
    it = 0
    while time.time() < time_end and it < max_iter:
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

def binary_hill_climb(start_h, time_end, rng):
    bin_vec = (np.array(start_h) >= 0.5).astype(int)
    n = len(bin_vec)
    best = bin_vec.copy()
    best_c5 = compute_c5(best.astype(float))
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best = cand
            best_c5 = cand_c5
            no_improve = 0
        else:
            no_improve += 1
    return best.astype(float), best_c5

def solve_milp_binary(n, time_limit, rng):
    weight = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(nonneg=True)
    constraints = [cp.sum(h) == weight]
    for k in range(1, n):
        length = n - k
        if length == 0:
            continue
        x = cp.Variable(length, boolean=True)
        constraints.append(x <= h[:length])
        constraints.append(x <= h[k:])
        constraints.append(x >= h[:length] + h[k:] - 1)
        constraints.append(t >= cp.sum(h[:length] - x))
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in (cp.CBC, cp.GLPK_MI):
        try:
            prob.solve(solver=solver, maximumSeconds=time_limit, verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                break
        except Exception:
            continue
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        raise RuntimeError("MILP failed")
    h_val = np.rint(np.array(h.value)).astype(float)
    cur = np.sum(h_val)
    if cur != weight:
        diff = weight - cur
        if diff > 0:
            zeros = np.where(h_val == 0)[0]
            if len(zeros) >= diff:
                idx = rng.choice(zeros, int(diff), replace=False)
                h_val[idx] = 1
        else:
            ones = np.where(h_val == 1)[0]
            if len(ones) >= -diff:
                idx = rng.choice(ones, int(-diff), replace=False)
                h_val[idx] = 0
    c5 = compute_c5(h_val)
    return h_val, c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():

    else:
        n = 80
        h0 = np.zeros(n, dtype=float)

    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = np.sum(h0)
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        h0 += diff / n
        h0 = np.clip(h0, 0.0, 1.0)
        residual = target_sum - np.sum(h0)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -h0[i], 1 - h0[i])
            h0[i] += adj
            residual = target_sum - np.sum(h0)

    # try MILP binary solution
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30:

        try:

            if c5_bin <= 0.3808:

        except Exception:

    # hybrid stochastic search
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    while time.time() - start_time < budget_s - 0.5:

        else:

            bin_start = (start_vec >= 0.5).astype(int)
            for _ in range(rng.integers(1, 4)):

                zeros = np.flatnonzero(1 - bin_start)
                if ones.size and zeros.size:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    bin_start[i] = 0
                    bin_start[j] = 1
            start_vec = bin_start.astype(float)

        if rng.random() < 0.6:
            cand_h, cand_c5 = continuous_anneal(start_vec, end_time, rng)
        else:
            cand_h, cand_c5 = binary_hill_climb(start_vec, end_time, rng)

        if cand_c5 < best_c5:
            best_h = cand_h.copy()
            best_c5 = cand_c5
        if best_c5 <= 0.3808:

    if time.time() - start_time < budget_s - 0.3:
        bin_start = (best_h >= 0.5).astype(int)

        for i in range(n):

                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    c5_val = compute_c5(best_h)
    return best_h, c5_val, n
