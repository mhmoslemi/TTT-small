def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1 - h)[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)[:2*n-1]
    centre = n-1
    # Exclude centre
    max_val = conv[:centre].max()
    if conv[centre+1:].size > 0:
        other_max = conv[centre+1:].max()
        if other_max > max_val:
            max_val = other_max
    return max_val * dx

def anneal_binary(h_init, time_limit, rng):
    h = h_init.astype(float)  # for computations
    n = len(h)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    # Precompute positions of ones and zeros for fast selection.
    # But each iteration updates them.
    start = time.time()
    iter = 0
    # Temperature schedule: T = T0 * (1 - frac) where frac = elapsed / time_limit
    T0 = 0.02  # adjust 
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # compute temperature
        frac = elapsed / time_limit
        T = T0 * (1 - frac)  # linear cooldown
        # Choose random 1 and 0 positions to swap
        ones = np.where(h > 0.5)[0]  # but for binary, values are exactly 1.0 or 0.0
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Perform swap: h_i=0, h_j=1
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            # accept with probability exp(-delta / T)
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            current_c5 = new_c5
            if current_c5 < best_c5 - 1e-12:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
        iter += 1

    return best_h, best_c5

def binary_anneal(n, rng, time_budget):
    M = n // 2
    # init random binary vector with M ones.
    h = np.zeros(n, dtype=float)
    ones_idx = rng.choice(n, M, replace=False)
    h[ones_idx] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_budget:
        # propose swap
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # swap if different
        if h[i] == h[j]:
            continue
        # evaluate after swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        # Accept if improvement or with some probability
        # Use simple hill climbing: accept if new_c5 <= best_c5 - small epsilon else revert.
        # But we want to allow uphill moves, use simulated annealing.
        # temperature schedule based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = 0.02 * (1 - frac) + 1e-6
        delta = new_c5 - best_c5
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept as new current state
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # compute convolution of h and (1-h) reversed
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)[:2 * n - 1]
    centre = n - 1
    # exclude centre
    # compute max of conv except centre
    # Use np.max with masked
    # But for speed:
    max_val = conv[0]
    # handle left side
    if centre > 0:
        left_max = conv[:centre].max()
        max_val = left_max if left_max > max_val else max_val
    if centre + 1 < conv.size:
        right_max = conv[centre + 1:].max()
        max_val = right_max if right_max > max_val else max_val
    return max_val * dx

import time
import numpy as np

def _compute_c5(h):
    """Maximum overlap (excluding zero shift) for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    centre = n - 1
    # max of all positions except centre
    max_left = conv[:centre].max() if centre > 0 else -np.inf
    max_right = conv[centre + 1 :].max() if centre + 1 < conv.size else -np.inf
    return max(max_left, max_right) * dx


def _anneal_binary(n, h0, time_limit, rng):
    """Simulated‑annealing on binary vectors (swap 0/1) to minimise C5."""
    h = h0.astype(float).copy()
    cur = _compute_c5(h)
    best_h = h.copy()
    best_val = cur
    start = time.time()
    T0 = 0.05  # initial temperature
    while time.time() - start < time_limit:
        i, j = rng.choice(n, 2, replace=False)
        if h[i] == h[j]:
            continue
        # propose swap
        h[i], h[j] = h[j], h[i]
        new = _compute_c5(h)
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1.0 - frac) + 1e-12
        delta = new - cur
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_val:
                best_val = new
                best_h = h.copy()
        else:
            # reject: revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_val


def _continuous_refine(h, time_limit, rng):
    """Local hill‑climbing with small mass transfers (preserves sum)."""
    n = h.shape[0]
    cur = _compute_c5(h)
    best_h = h.copy()
    best_val = cur
    start = time.time()
    while time.time() - start < time_limit:
        i, j = rng.choice(n, 2, replace=False)
        if rng.random() < 0.5:
            max_transfer = min(1.0 - h[i], h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            h[i] += delta
            h[j] -= delta
        else:
            max_transfer = min(1.0 - h[j], h[i])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            h[j] += delta
            h[i] -= delta
        new = _compute_c5(h)
        if new + 1e-12 < best_val:
            best_val = new
            best_h = h.copy()
            cur = new
        else:
            # revert
            if rng.random() < 0.5:
                h[i] -= delta
                h[j] += delta
            else:
                h[j] -= delta
                h[i] += delta
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # candidate discretisations (must be even)
    candidate_ns = [80, 96, 112, 128, 144]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 not in candidate_ns:
            candidate_ns.insert(0, n0)
        elif n0 % 2 == 0:
            candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        # time guard for outer loop
        if time.time() - start_total > budget_s - 5:
            break

        remaining = budget_s - (time.time() - start_total)
        # allocate at most 250 s per resolution, otherwise use half of remaining
        alloc_n = min(250.0, remaining * 0.5)

        # initialise a feasible binary vector for this n
        M = n // 2
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            parent = np.array(initial_h_values, dtype=float)
            # binarise by threshold 0.5 then fix weight
            h0 = (parent >= 0.5).astype(float)
            cur_sum = h0.sum()
            if cur_sum != M:
                # adjust excess/deficit by flipping random bits
                diff = int(cur_sum - M)
                if diff > 0:

            h0[idx] = 1.0

        h_bin, c5_bin = _anneal_binary(n, h0, alloc_n, rng)

        best_h_global = np.clip(best_h_global, 0.0, 1.0)

    return best_h_global, best_c5_global, n
