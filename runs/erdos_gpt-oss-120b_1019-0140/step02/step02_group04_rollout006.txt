def compute_c5(h, dx):
    """Maximum overlap integral for a given step function."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Project vector onto the simplex {0≤x_i≤1, sum x = target_sum}."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - vec.sum()
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Return a random feasible vector (entries in [0,1], exact sum)."""
    x = rng.random(n)
    return project_to_sum(x, target_sum)

        cand_h[i] -= delta
        cand_h[j] += delta
        cand_val = compute_c5(cand_h, dx)

        if cand_val < cur_val:
            cur_h = cand_h
            cur_val = cand_val
            if cand_val < best_val:
                best_h = cand_h.copy()
                best_val = cand_val
                if best_val < 0.3808:
                    break
        else:
            if temp > 0.0:
                prob = math.exp(-(cand_val - cur_val) / temp)
                if rng.random() < prob:
                    cur_h = cand_h
                    cur_val = cand_val
    return best_h, best_val


# ---------- Local continuous refinement ----------
def refine_continuous(h, dx, target_sum, deadline, start_time, rng):
    cur_h = h.copy()
    cur_val = compute_c5(cur_h, dx)
    step = 0.02
    n = len(h)
    while time.time() < deadline:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5(cur_h, dx)
    return cur_h, cur_val


# ---------- Binary hill climbing ----------
def binary_hill_climb(h_bin, cur_val, dx, deadline, rng):
    best = h_bin.copy()
    best_val = cur_val
    n = len(best)
    while time.time() < deadline:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best = cand
            best_val = cand_val
    return best.astype(float), best_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # Initialise best solution
    if h0 is not None:
        best_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
        best_val = compute_c5(best_h, dx)
    else:
        best_h = generate_random_feasible(n, target_sum, rng)
        best_val = compute_c5(best_h, dx)

    # Simulated annealing with multiple restarts
    while time.time() - start_time < budget_s * 0.93:
        # random start
        start_h = generate_random_feasible(n, target_sum, rng)
        start_val = compute_c5(start_h, dx)

        # allocate a slice of remaining budget for this anneal
        remaining = budget_s - (time.time() - start_time)
        slice_time = min(1.0, remaining * 0.25)  # at most 1 s or 25 % of remaining
        deadline = time.time() + slice_time

        cand_h, cand_val = anneal(
            start_h, start_val, dx, target_sum, rng, deadline, start_time, budget_s
        )
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val
            if best_val < 0.3808:
                break

    # Final continuous refinement if time permits
    remaining = budget_s - (time.time() - start_time)
    if remaining > 2.0:
        deadline = time.time() + remaining * 0.6
        refined_h, refined_val = refine_continuous(
            best_h, dx, target_sum, deadline, start_time, rng
        )
        if refined_val < best_val:
            best_h, best_val = refined_h, refined_val

    # Optional binary hill climb using the top entries of the current best
    remaining = budget_s - (time.time() - start_time)
    if remaining > 1.0:
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argsort(-best_h)[:int(target_sum)]
        binary_h[top_idx] = 1
        bin_val = compute_c5(binary_h.astype(float), dx)
        deadline = time.time() + remaining * 0.4
        bin_h, bin_val = binary_hill_climb(binary_h, bin_val, dx, deadline, rng)
        if bin_val < best_val:
            best_h = bin_h
            best_val = bin_val

    # Final safety enforcement
    best_h = project_to_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_val = compute_c5(best_h, dx)

    return best_h, best_val, n
