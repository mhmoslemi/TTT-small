def binary_annealing_search(n, time_budget, rng):
    M = n // 2
    dx = 2.0/n
    best_h = None
    best_c5 = np.inf
    start = time.time()
    global_iter = 0
    # temperature schedule parameters
    T0 = 0.5  # adjust later.
    # loop until time budget
    while time.time() - start < time_budget - 0.1:
        # random init binary
        ones = list(rng.choice(n, M, replace=False))
        h = np.zeros(n, dtype=np.int8)
        h[ones] = 1
        aut = np.correlate(h.astype(np.int16), h.astype(np.int16), mode='full')
        min_aut = aut.min()
        # compute actual C5
        c5 = compute_c5(h.astype(float))
        # maybe update best via actual c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            if best_c5 < 0.3808:
                return best_h.astype(float), best_c5

        # local annealing for this start
        # set steps based on remaining time
        max_local_steps = 2000
        for _ in range(max_local_steps):
            # time check
            if time.time() - start >= time_budget - 0.05:
                break
            global_iter += 1
            T = T0 * (0.9995 ** global_iter)
            i = rng.choice(ones)
            j = rng.integers(n)
            while h[j] == 1:
                j = rng.integers(n)

            new_aut = aut.copy()
            # removal i
            new_aut[n-1] -= 1
            for p in ones:
                if p == i:

                d = p - i
                new_aut[n-1 + d] -= 1

            new_aut[n-1] += 1
            for p in ones:

import time
import numpy as np

def compute_c5(h):
    """Exact overlap integral C5 for a (float) step function."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def binary_annealing_search(n, time_budget, rng):
    """Search a binary vector (0/1) of length n with sum n/2 that minimizes C5.
    Uses a simple simulated‑annealing scheme with O(n) updates per swap."""
    M = n // 2
    dx = 2.0 / n
    best_h = None
    best_c5 = np.inf
    start = time.time()
    global_iter = 0
    T0 = 0.5  # initial temperature for the metric (M‑min_aut)*dx

    while time.time() - start < time_budget - 0.1:
        # random binary start
        ones = list(rng.choice(n, M, replace=False))
        h = np.zeros(n, dtype=np.int8)
        h[ones] = 1
        # autocorrelation (ordered) as int16 to avoid overflow
        aut = np.correlate(h.astype(np.int16), h.astype(np.int16), mode='full')
        min_aut = aut.min()
        # true C5 for the start (cheap, only once per start)
        c5_true = compute_c5(h.astype(float))
        if c5_true < best_c5:
            best_c5 = c5_true
            best_h = h.copy()
            if best_c5 < 0.3808:
                return best_h.astype(float), best_c5

        # local annealing moves
        max_local_steps = 2000
        for _ in range(max_local_steps):
            if time.time() - start >= time_budget - 0.05:
                break
            global_iter += 1
            T = T0 * (0.9995 ** global_iter)

            i = rng.choice(ones)                     # a 1‑position
            j = rng.integers(n)                      # a 0‑position
            while h[j] == 1:
                j = rng.integers(n)

            # propose swap i↔j
            new_aut = aut.copy()
            # remove i
            new_aut[n - 1] -= 1
            for p in ones:
                if p == i:
                    continue
                d = p - i
                idx1 = n - 1 + d
                idx2 = n - 1 - d
                new_aut[idx1] -= 1
                new_aut[idx2] -= 1
            # add j
            new_aut[n - 1] += 1
            for p in ones:
                if p == i:
                    continue
                d = p - j
                idx1 = n - 1 + d
                idx2 = n - 1 - d
                new_aut[idx1] += 1
                new_aut[idx2] += 1

            new_min_aut = new_aut.min()
            cur_metric = (M - min_aut) * dx
            new_metric = (M - new_min_aut) * dx

            accept = False
            if new_metric < cur_metric - 1e-12:
                accept = True
            else:
                delta = new_metric - cur_metric
                if T > 0 and rng.random() < np.exp(-delta / T):
                    accept = True

            if accept:
                # apply swap
                h[i] = 0
                h[j] = 1
                ones.remove(i)
                ones.append(j)
                aut = new_aut
                min_aut = new_min_aut

                # occasional exact evaluation of C5
                if global_iter % 100 == 0:
                    c5_true = compute_c5(h.astype(float))
                    if c5_true < best_c5 - 1e-12:
                        best_c5 = c5_true
                        best_h = h.copy()
                        if best_c5 < 0.3808:
                            return best_h.astype(float), best_c5

        # end of one random start

    # final exact evaluation of the best binary discovered
    if best_h is not None:
        best_c5 = compute_c5(best_h.astype(float))
        return best_h.astype(float), best_c5
    return None, np.inf
