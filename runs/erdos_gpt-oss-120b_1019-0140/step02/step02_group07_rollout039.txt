def enforce_sum(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    s = np.sum(v)
    diff = target_sum - s
    if abs(diff) > 1e-12:
        # try to distribute diff uniformly
        v += diff / len(v)
        v = np.clip(v, 0.0, 1.0)
        # compute new diff
        s = np.sum(v)
        diff = target_sum - s
        # if still diff, adjust in small increments
        # loop up to 5 times
        for _ in range(5):
            if abs(diff) < 1e-12:
                break
            # find indices not at bound
            mask = (v > 0.0) & (v < 1.0)
            if mask.any():
                adjust = diff / mask.sum()
                v[mask] += adjust
                v = np.clip(v, 0.0, 1.0)
            else:
                # if all at bounds, distribute arbitrarily
                v += diff / len(v)
                v = np.clip(v, 0.0, 1.0)
            s = np.sum(v)
            diff = target_sum - s
    return v

def local_optimize(v0, time_limit, rng):
    n = len(v0)
    target_sum = n/2
    v = v0.copy()
    best_v = v.copy()
    best_c5 = compute_c5(v)
    # Parameters
    max_iter = int(1e6)  # adapt but break earlier
    start = time.time()
    # schedule step size initial, decaying
    eta0 = 0.5  # step size maybe up to 0.5 (max delta)
    for it in range(max_iter):
        t = time.time()
        if t >= time_limit:
            break
        # decide type of move: gradient step with prob 0.7, random swap with prob 0.3
        if rng.random() < 0.7:
            # gradient
            # compute worst shift k
            corr = np.correlate(v, 1 - v, mode='full')
            idx = np.argmax(corr)
            shift = idx - (n - 1)
            # compute gradient
            grad = np.zeros(n)
            # term1: (1 - v[i+shift])
            if shift >= 0:
                grad[:n-shift] = 1 - v[shift:]
                # term2: -v[i-shift]
                grad[shift:] -= v[:-shift]
            else:
                s = -shift
                grad[s:] = 1 - v[:n-s]
                grad[:n-s] -= v[s:]
            # step size
            eta = eta0 * (1 - it / max_iter)
            # new vector
            v_new = v - eta * grad
            # project
            v_new = enforce_sum(v_new, target_sum)
        else:
            # random swap of mass between two positions
            # choose inc idx with v < 1, dec idx with v > 0
            inc_idxs = np.where(v < 1 - 1e-12)[0]
            dec_idxs = np.where(v > 1e-12)[0]
            if inc_idxs.size == 0 or dec_idxs.size == 0:
                continue
            i = rng.choice(inc_idxs)
            j = rng.choice(dec_idxs)
            # define max delta
            max_delta = min(1 - v[i], v[j])
            if max_delta <= 0:
                continue
            # propose delta scaled by current step size maybe
            delta = max_delta * rng.random() * 0.2
            v_new = v.copy()

            # enforce sum (should already sum)

    n = len(v)
    target_sum = n / 2.0
    # ensure feasible
    v = enforce_sum(v, target_sum)

        if now >= time_limit:
            break
        # decide move
        if rng.random() < 0.6:
            # gradient
            corr = np.correlate(v, 1 - v, mode='full')
            idx = np.argmax(corr)
            shift = idx - (n - 1)
            # compute gradient
            grad = np.zeros(n)

                grad[:n-shift] = 1 - v[shift:]
                grad[shift:] -= v[:n-shift]
            else:
                s = -shift
                grad[s:] = 1 - v[:n-s]
                grad[:n-s] -= v[s:]

            eta = 0.5 * (1 - it / max_iters)

            v_new = enforce_sum(v_new, target_sum)

            v_new[j] -= delta
            # clip

Implementation details of enforce_sum: We'll implement function `project_onto_sum_and_bounds(v, target_sum)` that enforces constraints exactly, using water-filling.

Better approach: This is a classic projection of a vector onto the capped simplex: each component in [0,1] with sum = target_sum. This can be solved efficiently via sorting and thresholding. For moderate n, we can implement simple iterative algorithm: using "simplex projection with upper bounds". Since n is small (<=200), we can do sorting approach.

Procedure for projection onto [0,1] simplex:

Given v, we want to find w = argmin ||w - v||^2 s.t. sum w = target_sum and 0 ≤ w_i ≤ 1.

We can adapt algorithm:

- Let w = v.

- While True:
   - Clip w to [0,1]; compute sum_w.
   - If sum_w == target_sum within tolerance: break
   - Compute diff = sum_w - target_sum.
   - If diff > 0: need to reduce sum, find indices where w > 0 and reduce equally among those not at lower bound (0). Actually we need to reduce across all entries that are not at lower bound? Wait we need to reduce sum: we can subtract diff / count_active where active = indices not at bounds after clipping? But after clipping, some are at 0 or 1; adjusting them could re-enter bounds. So we can do iterative water-filling: maintain set of free variables (not at bounds). Then adjust them by subtracting diff / |free|. If any become out of bounds, clip them and remove from free, repeat.

- Similarly for diff < 0: need to increase sum; distribute among free variables.

Given small n, this works.

```
def project_simplex(v, target):
    # project onto [0,1] hypercube with sum target
    w = v.copy()
    # iterative water-filling
    for _ in range(10):
        w = np.clip(w, 0.0, 1.0)
        s = np.sum(w)
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # identify free variables (0 < w < 1)
        free = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if free.size == 0:
            # distribute among all
            w -= diff / len(w)
            continue
        w[free] -= diff / free.size
    # final clip
    w = np.clip(w, 0.0, 1.0)
    # adjust any residual
    residual = target - np.sum(w)
    if abs(residual) > 1e-12:
        # distribute to any variable not at bounds, else adjust first entry
        idxs = np.where((w > 0) & (w < 1))[0]
        if idxs.size:
            w[idxs[0]] += residual
        else:
            w[0] += residual
        w = np.clip(w, 0.0, 1.0)
    return w

import time, math, numpy as np

def compute_c5(v):
    """Maximum overlap integral approximated by discrete correlation."""
    n = len(v)
    return np.max(np.correlate(v, 1 - v, mode="full")) * (2.0 / n)

def project_sum_bounds(v, target):
    """Euclidean projection of v onto the simplex {0≤x≤1, sum x = target}."""
    w = v.copy()
    for _ in range(10):
        w = np.clip(w, 0.0, 1.0)
        s = np.sum(w)
        diff = s - target
        if abs(diff) < 1e-12:
            break
        free = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if free.size == 0:
            w -= diff / len(w)
        else:
            w[free] -= diff / free.size
    w = np.clip(w, 0.0, 1.0)
    residual = target - np.sum(w)
    if abs(residual) > 1e-12:
        idx = np.where((w > 0) & (w < 1))[0]
        if idx.size:
            w[idx[0]] += residual
        else:
            w[0] += residual
        w = np.clip(w, 0.0, 1.0)
    return w

def local_optimize(v0, deadline, rng):
    """Hybrid gradient‑swap simulated annealing on a single start vector."""
    n = len(v0)
    target = n / 2.0
    v = project_sum_bounds(v0, target)
    cur_c5 = compute_c5(v)
    best_v, best_c5 = v.copy(), cur_c5
    max_iters = 200000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # choose move type
        if rng.random() < 0.65:                     # gradient move
            corr = np.correlate(v, 1 - v, mode="full")
            idx = int(np.argmax(corr))
            shift = idx - (n - 1)
            grad = np.zeros(n)
            if shift >= 0:
                grad[: n - shift] = 1 - v[shift:]
                grad[shift:] -= v[: n - shift]
            else:
                s = -shift
                grad[s:] = 1 - v[: n - s]
                grad[: n - s] -= v[s:]
            eta = 0.6 * (1 - it / max_iters)
            v_new = project_sum_bounds(v - eta * grad, target)
        else:                                        # random mass swap
            inc = np.where(v < 1 - 1e-12)[0]
            dec = np.where(v > 1e-12)[0]
            if inc.size == 0 or dec.size == 0:
                continue
            i, j = rng.choice(inc), rng.choice(dec)
            max_delta = min(1 - v[i], v[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random() * 0.25
            v_new = v.copy()
            v_new[i] += delta
            v_new[j] -= delta
            v_new = np.clip(v_new, 0.0, 1.0)
        new_c5 = compute_c5(v_new)
        # simulated‑annealing acceptance
        if new_c5 < cur_c5:
            accept = True
        else:
            T = 0.05 * (1 - it / max_iters) + 0.001
            accept = rng.random() < math.exp(-(new_c5 - cur_c5) / (T + 1e-12))
        if accept:
            v, cur_c5 = v_new, new_c5
            if new_c5 < best_c5:
                best_c5, best_v = new_c5, v_new.copy()
                if best_c5 <= 0.3808:
                    break
    return best_v, best_c5

def binary_hill_climb(start_bin, deadline, rng):
    """Deterministic hill climbing on binary vectors (swap moves)."""
    n = len(start_bin)
    h = start_bin.astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    while time.time() < deadline and no_improve < 5000:
        ones = np.flatnonzero(start_bin)
        zeros = np.flatnonzero(1 - start_bin)
        if ones.size == 0 or zeros.size == 0:
            break
        i, j = rng.choice(ones), rng.choice(zeros)
        cand = start_bin.copy()
        cand[i], cand[j] = 0, 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5, best_h = cand_c5, cand.astype(float)
            start_bin = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=int, count=n)

def rudin_shapiro_seq(n):
    a = np.array([1], dtype=int)
    while len(a) < n:
        a = np.concatenate([a, a * ((-1) ** np.arange(len(a)))])
    a = a[:n]
    return (a + 1) // 2
