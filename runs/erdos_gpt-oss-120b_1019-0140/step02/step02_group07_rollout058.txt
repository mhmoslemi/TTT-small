def binary_steepest_descent(arr_bin):
    # arr_bin: 0/1 integer array
    n = len(arr_bin)
    arr = arr_bin.astype(float)
    best_c5 = compute_c5(arr)
    while True:
        ones_idx = np.flatnonzero(arr_bin)
        zeros_idx = np.flatnonzero(1 - arr_bin)
        improved = False
        best_swap = None
        best_swap_c5 = best_c5
        for i in ones_idx:
            for j in zeros_idx:
                # test swap
                arr_bin[i] = 0
                arr_bin[j] = 1
                c5 = compute_c5(arr_bin.astype(float))
                if c5 < best_swap_c5:
                    best_swap_c5 = c5
                    best_swap = (i, j)
                # revert
                arr_bin[i] = 1
                arr_bin[j] = 0
        if best_swap is not None and best_swap_c5 < best_c5 - 1e-12:
            i, j = best_swap
            arr_bin[i] = 0
            arr_bin[j] = 1
            best_c5 = best_swap_c5
            improved = True
        if not improved:
            break
    return arr_bin.astype(float), best_c5

def enforce_constraints(arr, target_sum):
    arr = np.clip(arr, 0.0, 1.0)
    diff = target_sum - np.sum(arr)
    if abs(diff) > 1e-12:
        # distribute to first element not hitting bounds
        # loop over indices
        for i in range(len(arr)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and arr[i] < 1.0:
                inc = min(diff, 1.0 - arr[i])
                arr[i] += inc
                diff -= inc
            elif diff < 0 and arr[i] > 0.0:
                dec = min(-diff, arr[i])
                arr[i] -= dec
                diff += dec
    return arr

def compute_c5(arr):
    """Approximate C5 via correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse_seq(n):
    """Return Thue‑Morse binary sequence of length n (0/1)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)
    return seq


def rudin_shapiro_seq(n):
    """Return Rudin‑Shapiro binary sequence of length n (0/1)."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)  # a_{2k}
            nxt.append(v * (1 if (k % 2 == 0) else -1))  # a_{2k+1}=(-1)^k a_k
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2  # map +1→1, -1→0


def enforce_constraints(arr, target_sum):
    """Clip to [0,1] and enforce exact sum by simple redistribution."""
    arr = np.clip(arr, 0.0, 1.0)
    diff = target_sum - np.sum(arr)
    if abs(diff) > 1e-12:
        # distribute diff greedily over entries that are not at a bound
        for i in range(len(arr)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and arr[i] < 1.0:
                inc = min(diff, 1.0 - arr[i])
                arr[i] += inc
                diff -= inc
            elif diff < 0 and arr[i] > 0.0:
                dec = min(-diff, arr[i])
                arr[i] -= dec
                diff += dec
    return arr

def binary_steepest_descent(bin_arr):
    """Exhaustive pairwise swap hill‑climb on binary vectors."""
    n = len(bin_arr)
    best_h = bin_arr.astype(float)
    best_c5 = compute_c5(best_h)
    while True:
