def local_search(h, dx, target_sum, time_deadline, rng):
    n = len(h)
    cur_val = compute_c5_vec(h, dx)
    step = target_sum * 0.1 / n  # maybe 0.1 average?
    # temperature schedule
    T = 0.01
    cooling = 0.9995
    eps = 1e-12

    while time.time() < time_deadline:
        # compute full correlation
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        max_val = c.max()
        # break if cannot improve beyond best known? Actually just accept
        # Determine shift with max
        max_idx = np.argmax(c)
        k = max_idx - (n - 1)
        # compute contributions for that shift
        if k >= 0:
            i_range = np.arange(0, n - k)
        else:
            i_range = np.arange(-k, n)
        # contributions:
        contrib = h[i_range] * (1.0 - h[i_range + k])
        # donors: indexes where contrib high and h not too low
        median = np.median(contrib)
        donors = i_range[contrib > median + eps]
        recipients = i_range[contrib < median - eps]
        if donors.size == 0 or recipients.size == 0:
            # random donors and recipients
            donors = np.where(h > eps)[0]
            recipients = np.where(h < 1 - eps)[0]
        i = rng.choice(donors)
        j = rng.choice(recipients)
        # compute max possible delta preserving bounds
        max_delta = min(step, h[i], 1.0 - h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        # apply transfer
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # evaluate
        new_val = compute_c5_vec(h_new, dx)
        if new_val < cur_val - 1e-12:
            h = h_new
            cur_val = new_val
            step = min(step * 1.1, target_sum)  # enlarge step up to big
        else:
            # SA acceptance
            if T > 0 and rng.random() < np.exp(-(new_val - cur_val) / T):
                h = h_new
                cur_val = new_val
            else:
                step *= 0.99
        # cooling
        T *= cooling
        # occasional random perturbation to escape stagnation
        if rng.random() < 0.001:
            # randomize a small subset of entries
            idxs = rng.choice(n, size=min(5, n), replace=False)
            # reassign random values while preserving sum
            # generate random values for these idxs, clip to [0,1]
            rand_vals = rng.random(idxs.shape) 
            # adjust to keep sum
            delta_sum = np.sum(rand_vals) - np.sum(h[idxs])
            # redistribute delta to other entries proportionally
            # simplest: just assign random and then project sum globally
            h[idxs] = rand_vals
            h = project_to_sum(h, target_sum)
            cur_val = compute_c5_vec(h, dx)
    return h, cur_val

def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    # add uniform offset, then clip again
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    # adjust residual difference precisely
    cur = np.sum(vec)
    diff = target_sum - cur
    # if diff >0, distribute to entries <1
    if diff > 1e-12:
        idx = np.where(vec < 1.0 - 1e-12)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < -1e-12:
        idx = np.where(vec > 1e-12)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec

def generate_random_feasible(n, target_sum, rng):
    # sample from exponential distribution to ensure positivity
    x = rng.random(n)
    x = x * (target_sum / np.sum(x))  # scale to target sum
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # adjust exactly sum
    cur = np.sum(x)
    diff = target_sum - cur
    # distribute diff greedily
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given h (discretized)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Clip to [0,1] and adjust to achieve exact sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if diff > 0:
        idx = np.where(vec < 1.0 - 1e-12)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 1e-12)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Random vector in [0,1]^n with exact sum target_sum."""
    x = rng.random(n)
    if np.sum(x) == 0:
        x.fill(target_sum / n)
    else:
        x = x * (target_sum / np.sum(x))
    x = np.clip(x, 0.0, 1.0)
    cur = np.sum(x)
    diff = target_sum - cur
    if diff > 0:
        idx = np.where(x < 1.0 - 1e-12)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 1e-12)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def triangular_initial(n):
    """Triangular bump h(x)=max(0,1-|x-1|) sampled on n points."""
    xs = np.linspace(0.0, 2.0, n, endpoint=False)  # points in [0,2)
    h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    return h


def cosine_initial(n):
    """Cosine bump h(x)=0.5*(1+cos(pi*(x-1))) sampled on n points."""
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 * (1.0 + np.cos(np.pi * (xs - 1.0)))
    return h


def local_search(h, dx, target_sum, deadline, rng):
    """Targeted stochastic local search focused on the worst shift."""
    n = len(h)
    cur_val = compute_c5_vec(h, dx)
    step = max(0.01, target_sum / (10.0 * n))  # initial transfer size
    T = 0.02
    cooling = 0.9995
    eps = 1e-12

    while time.time() < deadline:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_val = corr.max()
        k = np.argmax(corr) - (n - 1)

        # indices participating in this shift
        if k >= 0:
            idx = np.arange(0, n - k, dtype=int)
        else:
            idx = np.arange(-k, n, dtype=int)

        # contributions for this shift
        contrib = h[idx] * (1.0 - h[idx + k])

        # donor = max contribution, receiver = min contribution
        i_max = idx[np.argmax(contrib)]
        i_min = idx[np.argmin(contrib)]

        # verify feasibility of move
        if h[i_max] > eps and h[i_min] < 1.0 - eps:
            i, j = i_max, i_min
        else:
            donors = np.where(h > eps)[0]
            recs = np.where(h < 1.0 - eps)[0]
            if donors.size == 0 or recs.size == 0:
                break
            i = rng.choice(donors)
            j = rng.choice(recs)

        max_delta = min(step, h[i], 1.0 - h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_val = compute_c5_vec(h_new, dx)

        if new_val < cur_val - 1e-12:
            h = h_new
            cur_val = new_val
            step = min(step * 1.1, target_sum)  # allow larger steps after improvement
        else:
            # simulated‑annealing acceptance
            if T > 0 and rng.random() < np.exp(-(new_val - cur_val) / T):
                h = h_new
                cur_val = new_val
            else:
                step *= 0.99

        T *= cooling

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.001:
            subset = rng.choice(n, size=min(5, n), replace=False)
            h[subset] = rng.random(subset.shape)
            h = project_to_sum(h, target_sum)
            cur_val = compute_c5_vec(h, dx)

    return h, cur_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # candidate problem sizes
    n_candidates = [80]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 not in n_candidates:
            n_candidates.insert(0, n0)
    # add some neighbours (odd/even) and a few larger sizes
    for d in (1, -1, 2, -2, 5, -5):
        n_try = n_candidates[0] + d
        if n_try > 0 and n_try not in n_candidates:
            n_candidates.append(n_try)
    n_candidates.extend([96, 100, 120])
    n_candidates = [n for n in n_candidates if n <= 200]

    best_h = None
    best_val = float("inf")
    best_n = None

    # main search loop (reserve a small margin for final cleanup)
    while time.time() - start < budget_s * 0.94:
        for n in n_candidates:
            if time.time() - start >= budget_s * 0.94:
                break
            dx = 2.0 / n
            target_sum = n / 2.0

            # choose an initial vector
            init_options = []

            if "initial_h_values" in globals() and n == len(initial_h_values):
                init_options.append(np.clip(np.array(initial_h_values, dtype=float), 0.0, 1.0))

            init_options.append(triangular_initial(n))
            init_options.append(cosine_initial(n))
            init_options.append(generate_random_feasible(n, target_sum, rng))

            for h0 in init_options:
                h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
                cur_val = compute_c5_vec(h, dx)

                if cur_val < best_val:
                    best_h, best_val, best_n = h.copy(), cur_val, n
                    if best_val <= 0.3808:
                        return best_h, best_val, best_n

                # allocate time for a focused local search
                time_left = start + budget_s * 0.96 - time.time()
                if time_left <= 0:
                    break
                # each local search gets at most 6 s or the remaining budget fraction
                search_deadline = time.time() + min(6.0, time_left * 0.5)
                h_opt, val_opt = local_search(h, dx, target_sum, search_deadline, rng)

                if val_opt < best_val:
                    best_h, best_val, best_n = h_opt.copy(), val_opt, n
                    if best_val <= 0.3808:
                        return best_h, best_val, best_n

    # final safety projection and evaluation
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_val = compute_c5_vec(best_h, 2.0 / best_n)

    return best_h, best_val, best_n
