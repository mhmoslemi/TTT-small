def hill_climb(h, target, time_budget, rng, start_temp=0.01):
    best = h.copy()
    best_f = compute_c5(best)
    cur = h.copy()
    cur_f = best_f
    T0 = start_temp
    start = time.time()
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / time_budget) # linear cooling to zero
        # random transfer
        i,j = rng.choice(len(cur), size=2, replace=False)
        # max transfer
        max_transfer = min(cur[i], 1.0 - cur[j])
        # sample delta
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        # apply
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # already sums maintain
        # clip not needed as delta chosen based on bounds
        new_f = compute_c5(new)
        d = new_f - cur_f
        if d < 0 or rng.random() < np.exp(-d / (T + 1e-12)):
            cur = new
            cur_f = new_f
            if cur_f < best_f:
                best = cur.copy()
                best_f = cur_f
    return best, best_f

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap value multiplied by dx (full discrete correlation)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # try a global adjustment
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # fall back to element‑wise fixing
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def random_feasible(rng, n, target):
    """Generate a random vector in [0,1]^n with sum == target."""
    x = rng.random(n)
    return project_sum_and_bounds(x, target)

def de_optimize(n, target, time_budget, rng, h0=None):
    """Simple differential evolution with projection onto the feasible set."""
    popsize = max(5, min(30, n // 4))
    # initialise population
    pop = np.empty((popsize, n))
    if h0 is not None:
        pop[0] = project_sum_and_bounds(h0.copy(), target)
        start_idx = 1
    else:
        start_idx = 0
    for i in range(start_idx, popsize):
        pop[i] = random_feasible(rng, n, target)

    # evaluate initial fitness
    fitness = np.empty(popsize)
    for i in range(popsize):
        fitness[i] = compute_c5(pop[i])

    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    start = time.time()
    F = 0.5          # differential weight
    CR = 0.9         # crossover probability
    while time.time() - start < time_budget:
        for i in range(popsize):
            # choose three distinct indices not equal to i
            idxs = rng.choice([j for j in range(popsize) if j != i], size=3, replace=False)
            a, b, c = idxs
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)

            # binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_sum_and_bounds(trial, target)

            trial_f = compute_c5(trial)
            if trial_f <= fitness[i]:
                pop[i] = trial
                fitness[i] = trial_f
                if trial_f < best_f:
                    best_f = trial_f
                    best_h = trial.copy()
        # optional early stop if perfect (unlikely)
        if best_f <= 0.0:
            break
    return best_h, best_f

def hill_climb(h, target, time_budget, rng, start_temp=0.02):
    """Simple simulated‑annealing style hill climbing on feasible vectors."""
    best = h.copy()
    best_f = compute_c5(best)
    cur = h.copy()
    cur_f = best_f
    start = time.time()
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = start_temp * max(0.0, 1.0 - elapsed / time_budget)  # linear cooling

        # propose a mass transfer between two coordinates
        i, j = rng.choice(len(cur), size=2, replace=False)
        max_transfer = min(cur[i], 1.0 - cur[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # sum is preserved, bounds respected by construction
        new_f = compute_c5(new)
        diff = new_f - cur_f
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            cur = new
            cur_f = new_f
            if cur_f < best_f:
                best = cur.copy()
                best_f = cur_f
    return best, best_f

def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 200  # fallback size if no parent solution
        h0 = random_feasible(rng, n, n / 2.0)

    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    # Allocate time: 60 % for DE, remainder for hill climbing
    elapsed = time.time() - start_total
    remaining = max(0.0, budget_s - elapsed)
    de_budget = remaining * 0.6
    hc_budget = remaining - de_budget

    # Stage 1: differential evolution
    best_h, best_f = de_optimize(n, target, de_budget, rng, h0)

    # Stage 2: local refinement
    if hc_budget > 0.0:
        best_h, best_f = hill_climb(best_h, target, hc_budget, rng)

    # Final safety projection and evaluation
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
