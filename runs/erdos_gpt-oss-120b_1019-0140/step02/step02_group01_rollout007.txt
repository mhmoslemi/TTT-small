import time
import numpy as np

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _random_feasible_h(n, M, rng):
    # generate a random vector in [0,1] with exact sum M
    h = rng.random(n)
    h = h / np.sum(h) * M
    for _ in range(4):
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = M - cur
        if abs(diff) < 1e-12:
            break
        h += diff / n
    # final safety adjustment on a single entry
    cur = np.sum(h)
    if abs(cur - M) > 1e-9:
        h[0] += M - cur
        h = np.clip(h, 0.0, 1.0)
    return h
