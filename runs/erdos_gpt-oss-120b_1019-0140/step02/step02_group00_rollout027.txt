def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    # handle trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # Initialize lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # Function f(lam) = sum clip(v - lam, lo, hi)
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure bounds bracket s
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time
import math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Exact C5 value for discretised h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n (periodic not assumed)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def smooth_c5_grad(h, t):
    """Log‑sum‑exp smooth approximation of max correlation and its gradient."""
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    max_raw = np.max(corr_raw)
    exps = np.exp(t * (corr_raw - max_raw))
    sum_exps = np.sum(exps)
    weights = exps / sum_exps
    # gradient via convolution with reversed weights
    w_rev = weights[::-1]
    grad = np.convolve(w_rev, (1.0 - h), mode='full')
    n = h.size
    grad = grad[n - 1 : n - 1 + n]
    smooth_raw = max_raw + np.log(sum_exps) / t
    return smooth_raw, grad


def gradient_descent_smooth(h0, n_iter=10000,
                            t_start=0.5, t_end=20.0, step0=0.3):
    """Projected gradient descent on the smoothed objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        t = t_start + (t_end - t_start) * (it - 1) / (n_iter - 1)
        _, grad = smooth_c5_grad(h, t)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
