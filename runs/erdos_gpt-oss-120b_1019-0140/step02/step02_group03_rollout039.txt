def binary_swap_local_search(h_float, start_time, deadline):
    n = len(h_float)
    target_int = int(round(n/2))
    # threshold and adjust
    h_bin = (h_float >= 0.5).astype(np.int8)
    current_sum = h_bin.sum()
    rng_local = np.random.default_rng()
    if current_sum > target_int:
        # turn some ones to zero
        idx = np.where(h_bin == 1)[0]
        to_zero = rng_local.choice(idx, size=current_sum - target_int, replace=False)
        h_bin[to_zero] = 0
    elif current_sum < target_int:
        idx = np.where(h_bin == 0)[0]
        to_one = rng_local.choice(idx, size=target_int - current_sum, replace=False)
        h_bin[to_one] = 1
    # Now ensure sum exactly target_int.
    cur_c5 = compute_c5(h_bin.astype(float))
    best_h = h_bin.copy()
    best_c5 = cur_c5
    # Precompute ones and zeros indexes
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    # iterate with time check
    while time.time() < deadline:
        # random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng_local.choice(ones)
        j = rng_local.choice(zeros)

        h_new[j] = 1

def gradient_max_overlap(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    # tie-breaking choose first max
    k = int(np.argmax(corr) - (n - 1))
    # allocate shifted arrays
    shift_plus = np.zeros_like(h)
    shift_minus = np.zeros_like(h)
    if k >= 0:
        if k < n:
            shift_plus[:n - k] = h[k:]
            shift_minus[k:] = h[:n - k]
    else:
        kk = -k
        if kk < n:
            shift_plus[kk:] = h[:n - kk]
            shift_minus[:n - kk] = h[kk:]
    grad = shift_plus + shift_minus - 1.0
    return grad
```

Note that for k = 0 (self overlap), shift plus = h, shift minus = h, gradient = 2h-1. Might push values away from 0.5; may help reduce max overlap for other k? Might not be necessary.

import time
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_max_overlap(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = int(np.argmax(corr) - (n - 1))
    shift_plus = np.zeros_like(h)
    shift_minus = np.zeros_like(h)
    if k >= 0:
        if k < n:
            shift_plus[:n - k] = h[k:]
            shift_minus[k:] = h[:n - k]
    else:
        kk = -k
        if kk < n:
            shift_plus[kk:] = h[:n - kk]
            shift_minus[:n - kk] = h[kk:]
    grad = shift_plus + shift_minus - 1.0
    return grad

def de_obj(x):
    n = x.shape[0]
    target = n / 2.0
    h = project_to_box_sum(x, target)
    return compute_c5(h)

def local_refine_gradient(h_init, deadline):
    h = h_init.copy()
    n = h.shape[0]
    target = n / 2.0
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.5
    while time.time() < deadline:
        grad = gradient_max_overlap(h)
        h_new = project_to_box_sum(h - step * grad, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
        else:
            step *= 0.9
            if step < 1e-6:
                step = 0.5
    return best_h, best_c5

    target_int = int(round(n / 2.0))
    h_bin = (h_float >= 0.5).astype(np.int8)

    if cur_sum > target_int:
        idx = np.where(h_bin == 1)[0]
        to_zero = rng_local.choice(idx, size=cur_sum - target_int, replace=False)
        h_bin[to_zero] = 0
    elif cur_sum < target_int:
        idx = np.where(h_bin == 0)[0]
        to_one = rng_local.choice(idx, size=target_int - cur_sum, replace=False)
        h_bin[to_one] = 1
    return h_bin

def binary_annealing_search(h_bin_start, start_time, deadline):
    h = h_bin_start.copy()
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    rng_local = np.random.default_rng()
    total = max(deadline - start_time, 1e-12)
    while time.time() < deadline:
        # temperature decays linearly
        remaining = max(deadline - time.time(), 1e-12)
        T = remaining / total
        # pick random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        i = rng_local.choice(ones)
        j = rng_local.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0

            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    return best_h.astype(float), best_c5

    if 'initial_h_values' in globals():

        n = h0.size

        if c5_de < best_c5 - 1e-12:

        if c5_gd < best_c5 - 1e-12:
            best_h, best_c5 = h_gd, c5_gd
    # Phase 3: Binary annealing (remaining time)
    final_deadline = start_time + budget_s - 0.5
    if time.time() < final_deadline and n % 2 == 0:

        h_bin_opt, c5_bin_opt = binary_annealing_search(h_bin_start, time.time(), final_deadline)
        if c5_bin_opt < best_c5 - 1e-12:
            best_h, best_c5 = h_bin_opt, c5_bin_opt
    # Final safety projection and evaluation
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

        if c5_de < best_c5 - 1e-12:
            best_h, best_c5 = h_de, c5_de

    # Phase 2: Gradient descent refinement (≈40% of budget)
    phase2_end = start_time + budget_s * 0.7
    if time.time() < phase2_end:
        h_gd, c5_gd = local_refine_gradient(best_h, phase2_end)
        if c5_gd < best_c5 - 1e-12:
            best_h, best_c5 = h_gd, c5_gd

    # Phase 3: Binary annealing (remaining time)
    final_deadline = start_time + budget_s - 0.5
    if time.time() < final_deadline and n % 2 == 0:
        h_bin_start = make_binary_feasible(best_h)
        h_bin_opt, c5_bin_opt = binary_annealing_search(h_bin_start, time.time(), final_deadline)
        if c5_bin_opt < best_c5 - 1e-12:
            best_h, best_c5 = h_bin_opt, c5_bin_opt

        # Optional final gradient refinement if time permits
        if time.time() < final_deadline:
            h_final, c5_final = local_refine_gradient(best_h, final_deadline)
            if c5_final < best_c5 - 1e-12:
                best_h, best_c5 = h_final, c5_final

    # Final safety projection and evaluation
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
