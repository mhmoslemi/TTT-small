import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target via bisection on λ."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)


# ---------- Constraint classes ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx for a given lag k."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx


def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500):
    """Run a single SLSQP optimisation from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Initialise h
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---------- Phase 1: Simulated Annealing (≈70% budget) ----------
    phase1_deadline = start_time + budget_s * 0.7
    cur_h = best_h.copy()
    cur_c5 = best_c5

    T_initial = 1.0
    T = T_initial
    T_min = 1e-5
    cooling = 0.9995
    max_step = 0.5
    no_improve = 0

    while time.time() < phase1_deadline:
        # temperature schedule
        if T > T_min:
            T = max(T * cooling, T_min)
        sigma = max_step * (T / T_initial)

        # propose candidate
        cand = cur_h + rng.normal(scale=sigma, size=n)
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)

        delta = cand_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c5 = cand, cand_c5
            no_improve = 0
            if cur_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cur_h.copy(), cur_c5
                if best_c5 < 0.3808:
                    break
        else:
            no_improve += 1

        # occasional random restart if stagnating
        if no_improve > 8000:
            cur_h = project_to_box_sum(rng.random(n), target_sum)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0

    # ---------- Phase 2: SLSQP refinement ----------
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    t_start = compute_c5(best_h)
    res = slsqp_optimize(best_h, t_start, sum_con, overlap_cons, maxiter=800)
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-12:
            best_h, best_c5 = h_opt, c5_opt

    # ---------- Phase 3: Light hill‑climbing (remaining time) ----------
    phase3_deadline = start_time + budget_s - 0.3
    cur_h = best_h.copy()
    cur_c5 = best_c5
    base_step = 0.2
    while time.time() < phase3_deadline:
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2)
        cand = cur_h + step * d
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cur_h.copy(), cur_c5
        if rng.random() < 0.001:
            base_step *= 0.9

    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
