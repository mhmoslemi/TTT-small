def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    from scipy.optimize import minimize

    start_time = time.time()
    deadline = start_time + budget_s

    rng = np.random.default_rng(seed)

    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)

    dx = 2.0 / n
    target_sum = n / 2.0

    # Enforce sum and bounds on h0
    def enforce_sum(h):
        # clip first
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) > 1e-12:
            # adjust uniformly as first attempt
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            # final fix small discrepancy
            cur = h.sum()
            diff = target_sum - cur
            if abs(diff) > 1e-12:
                idxs = np.arange(n)
                rng.shuffle(idxs)
                for i in idxs:
                    if abs(diff) < 1e-12:
                        break
                    if diff > 0 and h[i] < 1.0:
                        inc = min(diff, 1.0 - h[i])
                        h[i] += inc
                        diff -= inc
                    elif diff < 0 and h[i] > 0.0:
                        dec = min(-diff, h[i])
                        h[i] -= dec
                        diff += dec
        return h

    h0 = enforce_sum(h0)

    # compute initial value
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ------- SLSQP multi-start -------
    # compute time slice for SLSQP: 0.6 of budget (or up to deadline-5)
    slsqp_end = start_time + budget_s * 0.6
    # pre-prepare constraints function factories
    def sum_con_func(x):
        return np.sum(x[:n]) - target_sum

    # inequality constraints for each lag
    def make_overlap_con(k):
        def con(x, k=k):
            h = x[:n]
            if k >= 0:
                overlap = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overlap = np.dot(h[kk:], 1.0 - h[: n - kk])
            return x[-1] - overlap * dx
        return con

    # Build constraints once
    constraints = [{"type": "eq", "fun": sum_con_func}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        constraints.append({"type": "ineq", "fun": make_overlap_con(k)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # number of SLSQP iterations limited by time
    while time.time() < slsqp_end - 2.0:
        # generate start
        h_start = random_start_vector(n, rng)
        # optionally perturb best_h slightly to diversify
        if rng.random() < 0.2:
            # perturb by small random transfer
            h_start = best_h.copy()
            # apply few random transfers
            for _ in range(3):
                i, j = rng.choice(n, size=2, replace=False)
                max_transfer = min(h_start[i], 1.0 - h_start[j])
                if max_transfer > 0:
                    delta = rng.random() * max_transfer * 0.5
                    h_start[i] -= delta
                    h_start[j] += delta
            h_start = enforce_sum(h_start)

        t_start = compute_c5(h_start)
        x0 = np.concatenate([h_start, [t_start]])
        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 2000, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            break
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # re-enforce exact sum
        cur = h_opt.sum()
        if abs(cur - target_sum) > 1e-9:
            h_opt += (target_sum - cur) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                # early exit
                break

    # --------- Simulated Annealing ----------
    # We'll run annealing until deadline (remaining time)
    def anneal(h_init, cur_val, deadline, rng):
        h = h_init.copy()
        best_local_h = h.copy()
        best_local_val = cur_val
        total_time = deadline - time.time()
        max_temp = 0.01  # max temperature
        it = 0
        while True:
            now = time.time()
            if now >= deadline - 0.1:
                break
            remaining = deadline - now
            # compute O_k array
            corr = np.correlate(h, 1.0 - h, mode="full")
            O_vals = corr * dx
            max_idx = np.argmax(O_vals)
            k = max_idx - (n - 1)  # lag
            # overlapped region
            if k >= 0:
                i_range = np.arange(n - k)
                contrib = h[: n - k] * (1.0 - h[k:])
            else:
                kk = -k
                i_range = np.arange(kk, n)
                contrib = h[kk:] * (1.0 - h[: n - kk])
            i_local = np.argmax(contrib)
            i = i_range[i_local]

            # candidate for destination j
            # restrict j to overlapping region too, if possible
            if k >= 0:
                cand = np.where((h < 1.0 - 1e-12) & (np.arange(n) < n - k))[0]
                partner_vals = h[k:][cand] if cand.size>0 else None
            else:
                cand = np.where((h < 1.0 - 1e-12) & (np.arange(n) >= -k))[0]
                # partner index = j + k (negative k)
                partner_vals = h[: n + k][cand - k] if cand.size>0 else None

            if cand.size == 0:
                # fallback any index with room
                cand = np.where(h < 1.0 - 1e-12)[0]
                if cand.size == 0:
                    break
                j = rng.choice(cand)
            else:
                # choose j with maximal partner h (i.e., minimal 1 - h[partner])
                # partner_vals may be empty if cand empty; but we have cand non-empty.
                # compute partner h values safely
                # note: partner_vals aligned with cand
                # choose index with max partner value
                max_part_idx = np.argmax(partner_vals)
                j = cand[max_part_idx]

            if i == j:
                # avoid self-transfer
                # choose a different random j
                others = np.where((h < 1.0 - 1e-12) & (np.arange(n) != i))[0]
                if others.size == 0:
                    break
                j = rng.choice(others)

            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                continue
            # choose delta
            delta = max_transfer * rng.random() * 0.5  # up to half of possible
            h[i] -= delta
            h[j] += delta

            new_val = compute_c5(h)
            # temperature schedule
            temp = max_temp * (remaining / (deadline - start_time))
            if new_val < cur_val:
                cur_val = new_val
                if new_val < best_local_val:
                    best_local_val = new_val
                    best_local_h = h.copy()
                    if best_local_val < 0.3808:
                        # early exit
                        break
            else:
                # accept uphill with probability
                prob = math.exp((cur_val - new_val) / (temp + 1e-12))
                if rng.random() < prob:
                    cur_val = new_val
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta

            it += 1
            # occasional random shuffle to escape plateau
            if it % 500 == 0:
                # apply random small redistribution among 5 random indices
                idxs = rng.choice(n, size=5, replace=False)
                total = h[idxs].sum()
                # generate random Dirichlet
                rand = rng.random(5)
                rand = rand / rand.sum() * total
                # enforce bounds [0,1] by clipping and renormalizing iteratively
                # Simple clip and rescale a few times
                for _ in range(3):
                    over = rand - 1.0
                    under = -rand
                    if np.any(over > 0):
                        excess = over[over > 0].sum()
                        rand[over > 0] = 1.0
                        # distribute excess to those under 1
                        mask = rand < 1.0
                        if mask.any():
                            rand[mask] += excess * (rand[mask] / rand[mask].sum())
                    if np.any(under > 0):
                        deficit = under[under > 0].sum()
                        rand[under > 0] = 0.0
                        mask = rand > 0
                        if mask.any():
                            rand[mask] += deficit * (rand[mask] / rand[mask].sum())
                # assign back
                h[idxs] = np.clip(rand, 0.0, 1.0)
                h = enforce_sum(h)
                # recompute cur_val after large perturbation
                cur_val = compute_c5(h)
                if cur_val < best_local_val:
                    best_local_val = cur_val
                    best_local_h = h.copy()
                    if best_local_val < 0.3808:
                        break

        return best_local_h, best_local_val

    # Run annealing on current best
    if time.time() < deadline - 0.5:
        best_h, best_val = anneal(best_h, best_val, deadline, rng)

    # final enforce
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h += (target_sum - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n

from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def random_start_vector(n, rng, target):
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        inc = diff / n
        v = np.clip(v + inc, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    inc = min(diff, 1.0 - v[i])
                    v[i] += inc
                    diff -= inc
                elif diff < 0 and v[i] > 0.0:
                    dec = min(-diff, v[i])
                    v[i] -= dec
                    diff += dec
    return v

def enforce_sum(h, target, rng):

    diff = target - cur

def anneal(h_init, cur_val, dx, target, deadline, rng):
    # Simple simulated annealing with targeted moves
    h = h_init.copy()
    best_h = h.copy()
    best_val = cur_val
    total_time = deadline - time.time()
    max_temp = 0.015  # starting temperature
    it = 0
    eps = 1e-12
    n = h.size
    while True:
        now = time.time()
        if now >= deadline - 0.1:
            break
        remaining = deadline - now
        # compute O_k
        corr = np.correlate(h, 1.0 - h, mode="full")
        O_vals = corr * dx
        max_idx = np.argmax(O_vals)
        k = max_idx - (n - 1)
        # overlapped index range
        if k >= 0:
            i_range = np.arange(n - k)
            contrib = h[: n - k] * (1.0 - h[k:])
        else:
            i_range = np.arange(-k, n)
            contrib = h[-k:] * (1.0 - h[: n + k])
        i_local = np.argmax(contrib)
        i = i_range[i_local]
        # candidate destination indices in same range with room
        cand = i_range[h[i_range] < 1.0 - eps]
        if cand.size == 0:
            cand = np.where(h < 1.0 - eps)[0]
            if cand.size == 0:
                break
        # choose j with largest partner value h[j+k]
        partner_vals = h[cand + k]  # works for positive or negative k
        j = cand[np.argmax(partner_vals)]
        if j == i:
            other = cand[cand != i]
            if other.size > 0:
                j = rng.choice(other)
            else:
                others = np.where((h < 1.0 - eps) & (np.arange(n) != i))[0]
                if others.size == 0:
                    break
                j = rng.choice(others)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * 0.5
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h, dx)
        temp = max_temp * remaining / total_time
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    return best_h, best_val
        else:
            if temp > eps:
                if math.exp((cur_val - new_val) / temp) > rng.random():
                    cur_val = new_val
                else:
                    # revert move
                    h[i] += delta
                    h[j] -= delta
            else:
                # revert
                h[i] += delta
                h[j] -= delta
        it += 1
        if it % 400 == 0:
            # occasional perturbation among a few indices
            idxs = rng.choice(n, size=5, replace=False)
            tot = h[idxs].sum()
            # random dirichlet
            rand = rng.random(5)
            rand = rand / rand.sum() * tot
            rand = np.clip(rand, 0.0, 1.0)
            # adjust sum if clipping changed total
            diff = tot - rand.sum()
            if abs(diff) > 1e-12:
                # distribute diff proportionally to avoid bounds violation
                for _ in range(3):
                    mask = (rand > 0) & (rand < 1)
                    if not mask.any():
                        break
                    adjust = diff * (rand[mask] / rand[mask].sum())
                    rand[mask] += adjust
                    diff = tot - rand.sum()
                    if abs(diff) < 1e-12:
                        break
            h[idxs] = rand
            h = enforce_sum(h, target, rng)
            cur_val = compute_c5(h, dx)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val < 0.3808:
                    return best_h, best_val
    return best_h, best_val

    deadline = start + budget_s
    rng = np.random.default_rng(seed)

        n = h0.shape[0]

    dx = 2.0 / n
    target = n / 2.0

    h0 = enforce_sum(h0, target, rng)
    best_h = h0.copy()
    best_val = compute_c5(best_h, dx)

    # Phase 1: refine the initial solution
    if time.time() < deadline - 5.0:
        best_h, best_val = anneal(best_h, best_val, dx, target, deadline, rng)

    # Phase 2: random restarts with limited annealing time
    # allocate remaining time
    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # allocate at most 30 seconds per restart
        alloc = min(30.0, remaining)
        h_start = random_start_vector(n, rng, target)
        cur_val = compute_c5(h_start, dx)
        # run anneal on this start for allocated time
        sub_deadline = time.time() + alloc
        h_opt, val_opt = anneal(h_start, cur_val, dx, target, sub_deadline, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # final enforce sum and clip
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)

    final_val = compute_c5(best_h, dx)
    return best_h.astype(float), final_val, n

        if k >= 0:
            i_range = np.arange(n - k)
            contrib = h[: n - k] * (1.0 - h[k:])
        else:
            i_range = np.arange(-k, n)
            contrib = h[-k:] * (1.0 - h[: n + k])
        i_local = np.argmax(contrib)
        i = i_range[i_local]
        # candidate destination indices in same overlapped region with room
        cand = i_range[h[i_range] < 1.0 - eps]
        if cand.size == 0:
            cand = np.where(h < 1.0 - eps)[0]
            if cand.size == 0:
                break
        # pick j whose partner (j+k) has largest value
        partner_vals = h[cand + k]          # works for both signs of k
        j = cand[np.argmax(partner_vals)]
        if j == i:
            other = cand[cand != i]
            if other.size > 0:
                j = rng.choice(other)
            else:
                others = np.where((h < 1.0 - eps) & (np.arange(n) != i))[0]
                if others.size == 0:
                    break
                j = rng.choice(others)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * 0.5
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h, dx)
        temp = max_temp * remaining / total_time
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    return best_h, best_val
        else:
            if temp > eps and math.exp((cur_val - new_val) / temp) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        it += 1
        if it % 400 == 0:
            # occasional random redistribution among a few indices
            idxs = rng.choice(n, size=5, replace=False)
            tot = h[idxs].sum()
            rand = rng.random(5)
            rand = rand / rand.sum() * tot
            rand = np.clip(rand, 0.0, 1.0)
            diff = tot - rand.sum()
            if abs(diff) > 1e-12:
                for _ in range(3):
                    mask = (rand > 0) & (rand < 1)
                    if not mask.any():
                        break
                    adjust = diff * (rand[mask] / rand[mask].sum())
                    rand[mask] += adjust
                    diff = tot - rand.sum()
                    if abs(diff) < 1e-12:
                        break
            h[idxs] = rand
            h = enforce_sum(h, target, rng)
            cur_val = compute_c5(h, dx)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val < 0.3808:
                    return best_h, best_val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine number of points and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_start_vector(n, rng, n / 2.0)

    dx = 2.0 / n
    target = n / 2.0

    h0 = enforce_sum(h0, target, rng)
    best_h = h0.copy()
    best_val = compute_c5(best_h, dx)

    # Phase 1: fine‑tune the initial solution
    if time.time() < deadline - 5.0:
        best_h, best_val = anneal(best_h, best_val, dx, target, deadline, rng)

    # Phase 2: random restarts with bounded annealing time
    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        alloc = min(30.0, remaining)
        h_start = random_start_vector(n, rng, target)
        cur_val = compute_c5(h_start, dx)
        sub_deadline = time.time() + alloc
        h_opt, val_opt = anneal(h_start, cur_val, dx, target, sub_deadline, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # Final safety clipping and exact sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h, dx)
    return best_h.astype(float), final_val, n
