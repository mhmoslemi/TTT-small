def project_sum_and_bounds(h, target):
    # Clip first to bounds
    h = np.clip(h, 0.0, 1.0)
    # adjust sum using water-filling
    max_iter = 10
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # cannot adjust without violating bounds: distribute diff uniformly and then clip
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    return h
```

def local_search(initial_h, target, time_limit, rng):
    n = len(initial_h)
    dx = 2.0 / n
    h = initial_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    T0 = 0.2
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)
        r = rng.random()
        if r < 0.4:
            # swap
            i, j = rng.choice(n, size=2, replace=False)
            cand = h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.8:
            # transfer move
            i, j = rng.choice(n, size=2, replace=False)
            max_tr = min(h[i], 1.0 - h[j])
            if max_tr <= 0:
                max_tr = min(h[j], 1.0 - h[i])
                if max_tr <= 0:
                    continue
                donor, recv = j, i
            else:
                donor, recv = i, j
            delta = rng.random() * max_tr
            cand = h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        else:
            # gradient move aiming to improve worst overlap
            # compute correlation (auto-correlation)
            corr = np.correlate(h, h, mode='full')
            mid = n - 1
            # mask out centre (shift zero)
            corr_no_center = corr.copy()
            corr_no_center[mid] = corr_no_center.max() + 1.0
            idx_min = np.argmin(corr_no_center)
            shift = idx_min - mid
            # compute neighbor sum
            neigh = np.zeros_like(h)
            if shift >= 0:
                neigh[:n-shift] += h[shift:]
                neigh[shift:] += h[:n-shift]
            else:
                kk = -shift
                neigh[kk:] += h[:n-kk]
                neigh[:n-kk] += h[kk:]
            # gradient direction: increase correlation => add neighbor
            # to keep sum, subtract mean
            direction = neigh - np.mean(neigh)
            max_abs = np.max(np.abs(direction))
            if max_abs > 0.2:
                direction = direction * (0.2 / max_abs)
            cand = h + direction
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_sum_and_bounds(cand, target)
        cand_val = compute_c5(cand)
        # acceptance
        if cand_val < cur_val or (T > 0 and rng.random() < math.exp(-(cand_val - cur_val) / T)):
            h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break
    return best_h, best_val

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
        # early exit if target achieved
        if best_val < 0.3808:
            break

    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:

            sm = smooth_and_fix(best_h, target_sum)

                best_h = sm
            else:
                break

    # final SLSQP refinement if time

    if remaining > 10.0:

        def sum_eq(x, target=target_sum):
            return np.sum(x[:n]) - target

            h = x[:n]
            t = x[-1]
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            return t - corr

        x0 = np.concatenate([best_h, [best_val]])
        # Use options to limit iterations
        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=[{"type":"eq","fun":sum_eq},

                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )

                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_sum_and_bounds(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:

                    best_h = h_opt
        except Exception:
            pass

    # final safety fix
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target_sum)

    # compute final value to return
    final_val = compute_c5(best_h)

    return best_h, final_val, n

import time
import math
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_and_bounds(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust so that sum(h) == target (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        cur = np.sum(h)
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # distribute uniformly then re‑clip
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_and_fix(h: np.ndarray, target: float) -> np.ndarray:
    """Gaussian‑type smoothing followed by sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    sm = project_sum_and_bounds(sm, target)
    return sm

def local_search(initial_h: np.ndarray, target: float, time_limit: float, rng: np.random.Generator):
    """Simple simulated‑annealing / gradient search."""
    n = initial_h.size
    dx = 2.0 / n
    h = initial_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    T0 = 0.5  # initial temperature
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)
        r = rng.random()
        # ----- swap move -----
        if r < 0.4:
            i, j = rng.choice(n, size=2, replace=False)
            cand = h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        # ----- transfer move -----
        elif r < 0.8:
            i, j = rng.choice(n, size=2, replace=False)
            max_tr = min(h[i], 1.0 - h[j])
            if max_tr <= 0:
                max_tr = min(h[j], 1.0 - h[i])
                if max_tr <= 0:
                    continue
                donor, recv = j, i
            else:
                donor, recv = i, j
            delta = rng.random() * max_tr
            cand = h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        # ----- gradient move -----
        else:
            # auto‑correlation to locate the worst shift (minimal overlap)
            corr = np.correlate(h, h, mode="full")
            mid = n - 1
            corr_no_center = corr.copy()
            corr_no_center[mid] = corr_no_center.max() + 1.0
            idx_min = np.argmin(corr_no_center)
            shift = idx_min - mid
            neigh = np.zeros_like(h)
            if shift >= 0:
                neigh[: n - shift] += h[shift:]
                neigh[shift:] += h[: n - shift]
            else:
                kk = -shift
                neigh[kk:] += h[: n - kk]
                neigh[: n - kk] += h[kk:]
            direction = neigh - np.mean(neigh)          # zero‑sum direction
            mabs = np.max(np.abs(direction))
            if mabs > 0.2:
                direction = direction * (0.2 / mabs)
            cand = h + direction
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_sum_and_bounds(cand, target)
        cand_val = compute_c5(cand)
        # acceptance rule
        accept = False
        if cand_val < cur_val:
            accept = True
        elif T > 0.0:
            prob = math.exp(-(cand_val - cur_val) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break
    return best_h, best_val

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points)."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # problem size – even number for convenience
    n = 256
    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- generate initial candidates -----
    candidates = []

    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # up‑sample to the chosen resolution
        x_old = np.linspace(0.0, 1.0, init.size)
        x_new = np.linspace(0.0, 1.0, n)
        h0 = np.interp(x_new, x_old, init)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_sum_and_bounds(h0, target_sum)
        candidates.append(h0)

    # binary start smoothed once
    idx = rng.choice(n, size=int(target_sum), replace=False)
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    candidates.append(smooth_and_fix(h_bin, target_sum))

    # pure random start
    h_rand = rng.random(n)
    candidates.append(project_sum_and_bounds(h_rand, target_sum))

    best_h = None
    best_val = math.inf
    finished_early = False

    # allocate ~80 % of the budget to the stochastic searches
    time_for_sa = budget_s * 0.80
    per_candidate_time = time_for_sa / len(candidates)

    for h_start in candidates:
        h_opt, val_opt = local_search(h_start, target_sum, per_candidate_time, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
        if best_val < 0.3808:
            finished_early = True
            break

    # ----- optional smoothing passes -----
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        for _ in range(5):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
            else:
                break

    # ----- final convex refinement (SLSQP) -----
    remaining = budget_s - (time.time() - start)
    if remaining > 10.0 and best_h is not None:
        def sum_eq(x, target=target_sum):
            return np.sum(x[:n]) - target

        def corr_ineq(x, dx=dx, n=n):
            h = x[:n]
            t = x[-1]
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            return t - corr  # each component must be >=0

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[
                    {"type": "eq", "fun": sum_eq},
                    {"type": "ineq", "fun": corr_ineq},
                ],
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_sum_and_bounds(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # final safety correction
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target_sum)
    final_val = compute_c5(best_h)

    return best_h, final_val, n
