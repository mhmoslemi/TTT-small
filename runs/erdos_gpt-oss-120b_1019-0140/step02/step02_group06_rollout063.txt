def refine_h(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    alpha0 = 0.5
    # schedule: alpha = alpha0 * (1 - elapsed/time_budget), min 0.01
    while time.time() - start < time_budget:
        # compute correlation v and find active shifts
        v = np.correlate(h, 1.0 - h, mode='full')
        maxv = v.max()
        # find shift indices where v equals max
        idxs = np.where(v == maxv)[0]
        ks = idxs - (n - 1)
        g = np.zeros(n)
        for k in ks:
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
                g[tail] += (1.0 - h[head])
                g[head] -= h[tail]
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
                g[tail] += (1.0 - h[head])
                g[head] -= h[tail]
        # average if multiple ks
        if len(ks) > 1:
            g /= len(ks)
        # enforce zero mean direction? Not needed as we project sum later.
        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.01, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # accept improvement
        if c5_new < compute_c5(h) or rng.random() < 0.001:
            h = h_new
        # random perturbation occasionally
        if rng.random() < 0.01:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
    return best_h, best_c5

def project_sum_and_bounds(x, target, tol=1e-10):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        # distribute diff equally
        x += diff / len(x)
        # clip again
        x = np.clip(x, 0.0, 1.0)
        # recompute diff
        cur = x.sum()
        diff = target - cur
        if abs(diff) > tol:
            # adjust iteratively
            for i in range(len(x)):
                if abs(diff) < tol:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

import time
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for a discrete sample."""
    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full')
    return np.max(v) * dx

def project_sum_and_bounds(x, target, tol=1e-12):
    """Clip to [0,1] and adjust so sum(x)==target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > tol:
            for i in range(len(x)):
                if abs(diff) < tol:
                    break
                if diff > 0.0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0.0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def eval_step_c5_cont(l, h):
    """C5 for a continuous step function defined by lengths l and heights h."""
    K = len(l)
    a = np.concatenate(([0.0], np.cumsum(l)[:-1]))
    b = a + l
    ks = set()
    for i in range(K):
        ai, bi = a[i], b[i]
        for j in range(K):
            aj, bj = a[j], b[j]
            ks.update((ai - aj, ai - bj, bi - aj, bi - bj))
    ks.update((-2.0, 2.0))
    min_A = float('inf')
    for k in ks:
        A = 0.0
        for i in range(K):
            hi, ai, bi = h[i], a[i], b[i]
            for j in range(K):
                hj = h[j]
                left = max(ai, a[j] + k)
                right = min(bi, b[j] + k)
                ov = right - left
                if ov > 0.0:
                    A += hi * hj * ov
        if A < min_A:
            min_A = A
    return 1.0 - min_A   # C5 = 1 – min_k A(k)

def discretize_step(l, h, n):
    """Create an n‑point sample of the step function."""
    dx = 2.0 / n
    pos = (np.arange(n) + 0.5) * dx
    a = np.concatenate(([0.0], np.cumsum(l)[:-1]))
    b = a + l
    idx = np.searchsorted(b, pos, side='right')
    return h[idx]

def subgradient_refine(h0, time_budget, seed):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    alpha0 = 0.5
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        maxv = v.max()
        idxs = np.where(v == maxv)[0] - (n - 1)   # active shifts
        g = np.zeros(n)
        for k in idxs:
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
                g[tail] += (1.0 - h[head])
                g[head] -= h[tail]
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
                g[tail] += (1.0 - h[head])
                g[head] -= h[tail]
        if len(idxs) > 1:
            g /= len(idxs)
        elapsed = time.time() - start
        alpha = alpha0 * max(0.01, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < compute_c5(h) or rng.random() < 0.001:
            h = h_new
        if rng.random() < 0.01:  # occasional shake
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_all = time.time()
    rng = np.random.default_rng(seed)

    # target discretisation size – keep it modest for speed but > initial if any
    n = 200
    if n % 2:
        n += 1
    target = n / 2.0
    dx = 2.0 / n

    # start from parent if available
    h_initial = None
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(init)
        dx0 = 2.0 / n0
        pos0 = (np.arange(n0) + 0.5) * dx0
        pos = (np.arange(n) + 0.5) * dx
        h_initial = np.interp(pos, pos0, init, left=init[0], right=init[-1])
        h_initial = project_sum_and_bounds(h_initial, target)

    # --------------------------------------------------------------
    # 1) Random search over low‑dimensional step functions
    # --------------------------------------------------------------
    time_spent = time.time() - start_all
    t_interval = max(0.0, budget_s * 0.45 - time_spent)
    best_grid = None
    best_c5 = np.inf
    interval_start = time.time()
    K = 5  # number of intervals (kept small for speed)

    while time.time() - interval_start < t_interval:
        # random lengths (Dirichlet)
        raw_len = rng.random(K)
        l = 2.0 * raw_len / raw_len.sum()
        if np.min(l) < 1e-4:
            continue

        # random heights for first K‑1 slots, compute the last one to satisfy the integral
        h_first = rng.random(K - 1)
        area_without_last = np.dot(l[:-1], h_first)
        h_last = (1.0 - area_without_last) / l[-1]
        if not (0.0 <= h_last <= 1.0):
            continue
        h_vec = np.concatenate((h_first, [h_last]))

        # quick analytical C5
        c5_cont = eval_step_c5_cont(l, h_vec)
        if c5_cont >= best_c5:
            continue

        # discretise to n points for a proper check
        grid = discretize_step(l, h_vec, n)
        grid = project_sum_and_bounds(grid, target)
        c5_grid = compute_c5(grid)

        if c5_grid < best_c5:
            best_c5 = c5_grid
            best_grid = grid.copy()

    # fallback to a random feasible vector if nothing found
    if best_grid is None:
        best_grid = rng.random(n)
        best_grid = project_sum_and_bounds(best_grid, target)
        best_c5 = compute_c5(best_grid)

    # --------------------------------------------------------------
    # 2) Refine the best grid with sub‑gradient descent
    # --------------------------------------------------------------
    time_spent = time.time() - start_all
    t_refine = max(0.0, budget_s - time_spent - 0.05)  # small safety margin
    best_h, best_c5 = subgradient_refine(best_grid, t_refine, seed + 100)

    # include the optional parent start as an extra refinement attempt
    if h_initial is not None:
        extra_h, extra_c5 = subgradient_refine(h_initial, t_refine * 0.2, seed + 200)
        if extra_c5 < best_c5:
            best_h, best_c5 = extra_h, extra_c5

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
