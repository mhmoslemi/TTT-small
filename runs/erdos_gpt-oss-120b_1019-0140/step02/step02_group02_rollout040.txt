    best_h = np.clip(best_h, 0.0, 1.0)
    total = best_h.sum()

        # final exact fix
        residual = target_sum - best_h.sum()
        if abs(residual) > 1e-12:
            idxs = np.arange(n)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def random_start_vector(n, rng):
    v = rng.random(n)
    # Scale to have mean 0.5 (since target sum n/2 -> avg 0.5)
    v = v - v.mean() + 0.5
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
        diff2 = target - v.sum()
        if abs(diff2) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff2) < 1e-12:
                    break
                if diff2 > 0 and v[i] < 1.0:
                    inc = min(diff2, 1.0 - v[i])
                    v[i] += inc
                    diff2 -= inc
                elif diff2 < 0 and v[i] > 0.0:
                    dec = min(-diff2, v[i])
                    v[i] -= dec
                    diff2 += dec
    return v

def derivative_for_shift(h, k):
    n = h.shape[0]
    d = np.zeros(n)
    if k >= 0:
        nk = n - k
        if nk > 0:
            d[:nk] += 1.0 - h[k:]
            d[k:] += -h[:nk]
    else:
        l = -k
        nl = n - l
        if nl > 0:
            d[l:] += 1.0 - h[:nl]
            d[:nl] += -h[l:]
    return d

def project_to_sum_and_bounds(v, target):
    # Ensure 0<=v<=1 and sum == target. Use water-filling.
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # iterative redistribution
    max_iter = 10 * len(v)
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size == 0:
            # distribute to all not full ones
            free = np.where(v < 1.0 - 1e-12)[0] if diff > 0 else np.where(v > 1e-12)[0]
            if free.size == 0:
                break
        delta = diff / free.size
        # clamp to bounds
        add = np.minimum(1.0 - v[free], np.maximum(-v[free], delta))
        v[free] += add
        diff = target - v.sum()
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v

    deadline = start + budget_s

    best_h = project_to_sum_and_bounds(best_h, target_sum)

import time
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    """Create a random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + 0.5            # shift to mean 0.5
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
        diff2 = target - v.sum()
        if abs(diff2) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff2) < 1e-12:
                    break
                if diff2 > 0 and v[i] < 1.0:
                    inc = min(diff2, 1.0 - v[i])
                    v[i] += inc
                    diff2 -= inc
                elif diff2 < 0 and v[i] > 0.0:
                    dec = min(-diff2, v[i])
                    v[i] -= dec
                    diff2 += dec
    return v

def derivative_for_shift(h, k):
    """Gradient of O_k = ∑ h_i(1−h_{i+k}) with respect to h, assuming zero padding."""
    n = h.shape[0]
    d = np.zeros(n)
    if k >= 0:
        nk = n - k
        if nk > 0:
            d[:nk] += 1.0 - h[k:]
            d[k:] += -h[:nk]
    else:
        l = -k
        nl = n - l
        if nl > 0:
            d[l:] += 1.0 - h[:nl]
            d[:nl] += -h[l:]
    return d

def project_to_sum_and_bounds(v, target):
    """Project v onto the hyperplane sum(v)=target with 0≤v≤1 (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    for _ in range(10 * len(v)):
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size == 0:
            # no interior points, use whichever side can still move
            free = np.where(v < 1.0 - 1e-12)[0] if diff > 0 else np.where(v > 1e-12)[0]
            if free.size == 0:
                break
        delta = diff / free.size
        add = np.minimum(1.0 - v[free], np.maximum(-v[free], delta))
        v[free] += add
        diff = target - v.sum()
    return np.clip(v, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # initialise h (use provided construction if available)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target_sum = n / 2.0
        h0 = project_to_sum_and_bounds(h0, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
        h0 = project_to_sum_and_bounds(h0, target_sum)

    dx = 2.0 / n
    best_h = h0.copy()
    best_val = compute_c5(best_h)          # scaled value

    max_step_factor = 0.5
    min_step_factor = 0.02

    # ------------------------------------------------------------------ #
    # outer loop: multiple restarts
    while time.time() < deadline - 0.5:
        # start either from the current best or a fresh random point
        if rng.random() < 0.4:
            h = random_start_vector(n, rng)
        else:
            h = best_h.copy()
        h = project_to_sum_and_bounds(h, target_sum)

        # inner local search
        step_factor = max_step_factor
        while time.time() < deadline - 0.2:
            # correlation and current maximum (unscaled)
            O_vec = np.correlate(h, 1.0 - h, mode="full")
            O_max = O_vec.max()
            O_max_scaled = O_max * dx

            # global update
            if O_max_scaled < best_val - 1e-9:
                best_val = O_max_scaled
                best_h = h.copy()
                if best_val < 0.3808:
                    break

            # shift attaining the current maximum
            k_idx = O_vec.argmax()
            k_shift = k_idx - (n - 1)

            # gradient for that shift
            d = derivative_for_shift(h, k_shift)

            # candidate indices for mass transfer
            src_mask = (d > 1e-12) & (h > 1e-12)
            dst_mask = (d < -1e-12) & (h < 1.0 - 1e-12)
            src_candidates = np.where(src_mask)[0]
            dst_candidates = np.where(dst_mask)[0]

            if src_candidates.size == 0 or dst_candidates.size == 0:
                # fallback: any feasible pair
                src_possible = np.where(h > 1e-12)[0]
                dst_possible = np.where(h < 1.0 - 1e-12)[0]
                if src_possible.size == 0 or dst_possible.size == 0:
                    break
                i = src_possible[rng.integers(src_possible.size)]
                j = dst_possible[rng.integers(dst_possible.size)]
            else:
                i = src_candidates[np.argmax(d[src_candidates])]
                j = dst_candidates[np.argmin(d[dst_candidates])]

            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                break
            delta = rng.random() * max_transfer * step_factor

            # perform tentative move
            h[i] -= delta
            h[j] += delta
            # minor clipping to keep within bounds
            if h[i] < 0.0:
                h[i] = 0.0
            if h[j] > 1.0:
                h[j] = 1.0

            # evaluate after move
            new_O_max = np.max(np.correlate(h, 1.0 - h, mode="full"))
            new_O_max_scaled = new_O_max * dx

            if new_O_max_scaled <= O_max_scaled + 1e-12:
                # accept
                O_max = new_O_max
                O_max_scaled = new_O_max_scaled
                step_factor = min(max_step_factor, step_factor * 1.02)
            else:
                # reject
                h[i] += delta
                h[j] -= delta
                step_factor = max(min_step_factor, step_factor * 0.95)

            # occasional random jitter to escape plateaus
            if rng.random() < 0.005:
                a, b = rng.choice(n, 2, replace=False)
                avail = min(h[a], 1.0 - h[b])
                if avail > 0:
                    eps = rng.random() * avail * 0.5
                    h[a] -= eps
                    h[b] += eps
                    step_factor = max_step_factor

        if best_val < 0.3808:
            break

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum_and_bounds(best_h, target_sum)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
