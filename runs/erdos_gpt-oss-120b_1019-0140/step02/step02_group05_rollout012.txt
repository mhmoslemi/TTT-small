def fix_sum_and_bounds(h, target_sum):
    # Clip
    h = np.clip(h,0,1)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # indices where 0 < h < 1
        free = np.where((h>0) & (h<1))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h,0,1)
        # residual
        diff = target_sum - np.sum(h)
        # Adjust sequentially
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h,0,1)
    return h

def smoothed_max_and_grad(h, alpha=20):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    max_ov = np.max(overlaps)
    exp_weights = np.exp(alpha * (overlaps - max_ov))
    sum_exp = np.sum(exp_weights)
    weights = exp_weights / sum_exp

    grad = np.zeros(n)
    # compute contributions:
    for shift_idx, w in enumerate(weights):
        if w == 0:
            continue
        k = shift_idx - (n - 1)
        if k >= 0:
            # masks
            # part 1: i=0..n-k-1 => +1
            # part 2: i=0..n-k-1 => -h[i+k]
            # part 3: i=k..n-1 => -h[i-k]
            # vectorize:
            i_range = np.arange(n - k)
            grad[i_range] += w * (1 - h[i_range + k])
            i_range2 = np.arange(k, n)
            grad[i_range2] += w * (-h[i_range2 - k])
        else:
            kk = -k
            # shift negative
            # i = kk..n-1 => +1
            # i = kk..n-1 => -h[i-kk]
            # i = 0..n-kk-1 => -h[i+kk]
            i_range = np.arange(kk, n)
            grad[i_range] += w * (1 - h[i_range - kk])
            i_range2 = np.arange(0, n - kk)
            grad[i_range2] += w * (-h[i_range2 + kk])

def fix_sum_and_bounds(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:

        if free.size:
            inc = diff / free.size
            h[free] += inc

def smoothed_max_and_grad(h, alpha=10.0):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    max_ov = np.max(overlaps)
    exp_w = np.exp(alpha * (overlaps - max_ov))
    sum_exp = np.sum(exp_w)
    if sum_exp == 0:
        weights = exp_w
    else:
        weights = exp_w / sum_exp
    grad = np.zeros(n)
    # vector of indices
    idx = np.arange(n)
    for shift_idx, w in enumerate(weights):
        if w == 0:
            continue
        k = shift_idx - (n - 1)

            # first part i < n - k
            part_len = n - k
            if part_len > 0:
                i = idx[:part_len]
                grad[i] += w * (1 - h[i + k])

            else:
                break

    # ----- random hill climbing for remaining time -----
    if time.time() - start < budget_s - 2.0:
        while time.time() - start < budget_s - 1.0:
            cand = best_h.copy()
            if rng.random() < 0.5:
                i, j = rng.choice(n, 2, replace=False)
                max_tr = min(cand[i], 1.0 - cand[j])
                if max_tr <= 0:
                    max_tr = min(cand[j], 1.0 - cand[i])
                    if max_tr <= 0:
                        continue
                    donor, recv = j, i
                else:
                    donor, recv = i, j
                delta = rng.random() * max_tr
                cand[donor] -= delta
                cand[recv] += delta
            else:
                L = rng.integers(1, n // 10 + 1)
                i = rng.integers(0, n - L + 1)
                j = rng.integers(0, n - L + 1)
                if abs(i - j) < L:
                    continue
                tmp = cand[i:i+L].copy()
                cand[i:i+L] = cand[j:j+L]
                cand[j:j+L] = tmp
            cand = np.clip(cand, 0.0, 1.0)
            cand = fix_sum_and_bounds(cand, target_sum)
            cand_val = compute_c5(cand)
            if cand_val < best_val:
                best_val, best_h = cand_val, cand
                if best_val < 0.3808:
                    break

    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = np.sum(best_h)
    if abs(cur_sum - target_sum) > 1e-9:
        best_h += (target_sum - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, n
