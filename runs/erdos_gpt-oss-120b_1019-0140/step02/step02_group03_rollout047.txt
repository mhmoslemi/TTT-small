def slsqp_refine(h, t0=None, maxiter=500):
    n = len(h)
    dx = 2.0 / n
    if t0 is None:
        t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # constraints list
    cons = []
    # sum equality
    cons.append({'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - n/2.0})
    # overlap constraints
    for k in range(-n+1, n):
        if k == 0: continue
        def constr_factory(k):
            def fun(x, k=k):
                h_vec = x[:n]
                if k >= 0:
                    overlap = np.dot(h_vec[:n - k], 1.0 - h_vec[k:])
                else:
                    kk = -k
                    overlap = np.dot(h_vec[kk:], 1.0 - h_vec[:n - kk])
                return x[-1] - overlap * dx
            return fun
        cons.append({'type': 'ineq', 'fun': constr_factory(k)})
    res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # enforce sum exactly
        cur_sum = h_opt.sum()
        if abs(cur_sum - n/2.0) > 1e-9:
            # project to box sum
            h_opt = project_to_box_sum(h_opt, n/2.0)
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt
    else:
        # fallback to original
        return h, compute_c5(h)
```

def anneal_binary(h, start_score, deadline, rng):
    n = h.shape[0]
    cur_h = h.copy()
    cur_score = start_score
    best_h = cur_h.copy()
    best_score = cur_score
    iteration = 0
    init_T = 0.01  # maybe 0.05
    while time.time() < deadline:
        iteration += 1
        T = init_T * (0.9999 ** iteration)
        # random swap
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                if best_score < 0.3808:
                    break
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_score

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the box [lo,hi]^n while enforcing sum == target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def slsqp_refine(h_start, t_start=None, maxiter=400):
    """Local refinement of a feasible h using SLSQP on the max‑overlap formulation."""
    n = h_start.shape[0]
    dx = 2.0 / n
    if t_start is None:
        t_start = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t_start]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - n / 2.0}]

    # inequality: t >= overlap(k)*dx for all non‑zero shifts
    for k in range(-n + 1, n):
        if k == 0:
            continue

        def make_fun(k):
            def fun(x, k=k):
                h = x[:n]
                if k >= 0:
                    overlap = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overlap * dx
            return fun

        cons.append({'type': 'ineq', 'fun': make_fun(k)})

    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, n / 2.0)
        return h_opt, compute_c5(h_opt)
    else:
        return h_start, compute_c5(h_start)

def anneal_binary(h_start, start_score, deadline, rng):
    """Simple simulated‑annealing on binary vectors (swap a 1 with a 0)."""
    n = h_start.shape[0]
    cur_h = h_start.copy()
    cur_score = start_score
    best_h = cur_h.copy()
    best_score = cur_score
    iteration = 0
    init_T = 0.02

    while time.time() < deadline:
        iteration += 1
        T = init_T * (0.9999 ** iteration)

        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        i = rng.choice(ones)
        j = rng.choice(zeros)

        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]   # swap
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                if best_score < 0.3808:
                    break
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    return best_h, best_score
