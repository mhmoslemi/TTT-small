def continuous_anneal(start_h, time_end, rng):
    n = start_h.size
    target = n / 2.0
    # copy
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # schedule parameters
    init_temp = 1e-2  # larger start to explore more
    final_temp = 1e-7
    max_iter = 200000
    it = 0
    while time.time() < time_end and it < max_iter:
        # pick inc and dec indices
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # Ensure sum constraint automatically satisfied cause transferred delta

        # project maybe to safe bounds
        # but after transfer sum unchanged, still within [0,1]
        # compute
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # final sum correction (should be same)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / arr.size)

def project_to_simplex_bounds(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(40):
        mid = (lo + hi) * 0.5
        y = x - mid
        y = np.clip(y, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = x - (lo + hi) * 0.5
    y = np.clip(y, 0.0, 1.0)
    s = y.sum()
    if abs(s - target) > 1e-10:
        diff = target - s
        free = np.where((y > 1e-12) & (y < 1 - 1e-12))[0]
        if free.size > 0:
            adj = diff / free.size
            y[free] += adj
            y = np.clip(y, 0.0, 1.0)
        else:
            # allocate to first element within bounds
            i0 = 0
            if diff > 0:
                inc = min(diff, 1 - y[i0])
                y[i0] += inc
            else:
                dec = min(-diff, y[i0])
                y[i0] -= dec
    return y

def continuous_anneal(start_h, time_end, rng):
    n = start_h.size
    target = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 200000
    it = 0
    while time.time() < time_end and it < max_iter:
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # enforce exact sum
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    target_sum = n / 2.0
    # ensure feasible start
    h0 = project_to_simplex_bounds(h0, target_sum)

    # DE parameters
    popsize = max(10, min(30, n // 2))
    # initial population
    population = []
    # include deterministic start
    population.append(h0)
    while len(population) < popsize:
        vec = rng.random(n)
        vec = project_to_simplex_bounds(vec, target_sum)
        population.append(vec)
    pop = np.stack(population, axis=0)  # shape (popsize,n)
    # evaluate
    scores = np.apply_along_axis(compute_c5, 1, pop)  # compute_c5 already expects feasible
    best_idx = scores.argmin()
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    # DE loop
    F = 0.8
    CR = 0.9
    # time buffer for later refinement
    buffer = 5.0
    while time.time() - start_time < budget_s - buffer:
        improvement = False
        for i in range(popsize):
            # generate mutant
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # crossover
            jrand = rng.integers(n)
            cross_mask = (rng.random(n) < CR) | (np.arange(n) == jrand)
            trial = pop[i].copy()
            trial[cross_mask] = mutant[cross_mask]
            # repair to feasibility
            trial = project_to_simplex_bounds(trial, target_sum)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                improvement = True
                if trial_score < best_score:
                    best = trial.copy()
                    best_score = trial_score
                    if best_score <= 0.3808:
                        break
        if best_score <= 0.3808:
            break
        # optional early stop if no improvement after full generation
        # But we continue anyway
    # Local refinement with annealing if time permits
    time_left = budget_s - (time.time() - start_time)
    if time_left > 1.0:
        end_time = time.time() + time_left - 0.1
        annealed_h, annealed_c5 = continuous_anneal(best, end_time, rng)
        if annealed_c5 < best_score:
            best = annealed_h
            best_score = annealed_c5

    # final repair
    best = project_to_simplex_bounds(best, target_sum)

    # final clip (just in case)
    best = np.clip(best, 0.0, 1.0)

    return best, best_score, n

import time
import math
import numpy as np

def compute_c5(arr):
    """Compute C5 for a feasible h array."""
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / arr.size)

def project_to_simplex_bounds(x, target):
    """
    Euclidean projection of x onto the set:
        { y : sum(y)=target, 0<=y_i<=1 }.
    """
    x = np.clip(x, 0.0, 1.0)
    lo = np.min(x) - 1.0   # ensures sum=n at lo
    hi = np.max(x)         # ensures sum=0 at hi
    for _ in range(40):

        y = x - mid
