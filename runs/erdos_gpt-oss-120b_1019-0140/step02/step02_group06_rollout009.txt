def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Use np.correlate with mode='full'
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_sum_and_bounds(x, target, tol=1e-12, max_iter=100):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) <= tol:
        return x
    # add diff uniformly
    x = x + diff / len(x)
    x = np.clip(x, 0.0, 1.0)
    # now iterate
    for _ in range(max_iter):
        cur = x.sum()
        diff = target - cur
        if abs(diff) <= tol:
            break
        # try to distribute diff across non-boundary entries
        # find indices which are not at bounds
        free = (x > 0.0 + tol) & (x < 1.0 - tol)
        if free.any():
            # distribute diff equally among free entries
            nfree = free.sum()
            inc = diff / nfree
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
        else:
            # if all at bounds, cannot adjust; break
            break
    # final check; if still off by more than tol, do simple loop adjustment
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        # incremental adjust
        for i in range(len(x)):
            if diff > 0 and x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
            if abs(diff) <= tol:
                break
    return x

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for step function h."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target, tol=1e-12, max_iter=100):
    """Clip x to [0,1] and adjust to make sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) <= tol:
        return x
    # uniform adjustment
    x = x + diff / len(x)
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = x.sum()
        diff = target - cur
        if abs(diff) <= tol:
            break
        free = (x > tol) & (x < 1.0 - tol)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
        else:
            break
    # final sweep if needed
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        for i in range(len(x)):
            if diff > 0.0 and x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0.0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
            if abs(diff) <= tol:
                break
    return x
