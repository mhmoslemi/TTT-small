def compute_c5(arr):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by full correlation."""
    n = len(arr)

        zeros = np.flatnonzero(1 - cur)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 < cur_c5 - 1e-12:
            cur = cand
            cur_c5 = cand_c5
            no_improve = 0
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
        else:
            no_improve += 1
    return best.astype(float), best_c5

def continuous_anneal(start_h, time_end, rng):
    """Simulated annealing on continuous variables preserving the sum."""
    n = len(start_h)
    target_sum = n / 2.0
    h = np.clip(start_h, 0.0, 1.0)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 200000
    it = 0

    while time.time() < time_end and it < max_iter:
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1

    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for k in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[k], 1 - best_h[k])
            best_h[k] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values, dtype=float)
        base_n = len(base_h)
        factor = 2
        n = base_n * factor
        init_bin = (base_h >= 0.5).astype(int)
        init_bin = np.repeat(init_bin, factor)
        target_sum_int = n // 2
        cur_sum = int(np.sum(init_bin))
        diff_int = target_sum_int - cur_sum
        if diff_int != 0:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for idx in idxs:
                if diff_int == 0:
                    break
                if diff_int > 0 and init_bin[idx] == 0:
                    init_bin[idx] = 1
                    diff_int -= 1
                elif diff_int < 0 and init_bin[idx] == 1:
                    init_bin[idx] = 0
                    diff_int += 1
        init_h = init_bin.astype(float)
    else:
        base_n = 80
        factor = 2
        n = base_n * factor
        init_h = np.zeros(n, dtype=float)
        init_h[rng.choice(n, n // 2, replace=False)] = 1.0

    target_sum = n / 2.0
    init_h = np.clip(init_h, 0.0, 1.0)
    cur_sum = np.sum(init_h)
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        init_h += diff / n
        init_h = np.clip(init_h, 0.0, 1.0)
        residual = target_sum - np.sum(init_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -init_h[i], 1 - init_h[i])
            init_h[i] += adj
            residual = target_sum - np.sum(init_h)

    best_h = init_h.copy()
    best_c5 = compute_c5(best_h)

    # Prepare a few seeds
    seeds = []
    seeds.append(best_h.copy())
    rand_bin = np.zeros(n, dtype=float)
    rand_bin[rng.choice(n, n // 2, replace=False)] = 1.0
    seeds.append(rand_bin)

    tm_seq = thue_morse_seq_len(n)
    target_int = n // 2
    diff_int = target_int - int(np.sum(tm_seq))
    if diff_int != 0:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for idx in idxs:
            if diff_int == 0:
                break
            if diff_int > 0 and tm_seq[idx] == 0:
                tm_seq[idx] = 1
                diff_int -= 1
            elif diff_int < 0 and tm_seq[idx] == 1:
                tm_seq[idx] = 0
                diff_int += 1
    seeds.append(tm_seq.astype(float))

    # Main search loop
    while time.time() - start_time < budget_s - 0.5:
        time_left = budget_s - (time.time() - start_time)
        chunk = min(4.0, time_left * 0.4)
        end_time = time.time() + chunk

        if rng.random() < 0.2:
            start_vec = seeds[rng.integers(len(seeds))].copy()
        else:
            start_vec = best_h.copy()
            bin_vec = (start_vec >= 0.5).astype(int)
            for _ in range(rng.integers(1, 4)):
                ones = np.flatnonzero(bin_vec)
                zeros = np.flatnonzero(1 - bin_vec)
                if ones.size and zeros.size:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    bin_vec[i] = 0
                    bin_vec[j] = 1
            start_vec = bin_vec.astype(float)

        cand_h, cand_c5 = hill_climb_binary(start_vec.astype(int), end_time, rng)
        if cand_c5 < best_c5:
            best_h = cand_h.copy()
            best_c5 = cand_c5

        now = time.time()
        if now < start_time + budget_s - 0.6:
            cont_end = now + min(2.0, (budget_s - (now - start_time)) * 0.3)
            cand_h2, cand_c5_2 = continuous_anneal(best_h, cont_end, rng)
            if cand_c5_2 < best_c5:
                best_h = cand_h2.copy()
                best_c5 = cand_c5_2

        if best_c5 <= 0.3808:
            break

    # Final refinement from rounded version
    if time.time() - start_time < budget_s - 0.3:
        final_bin = (best_h >= 0.5).astype(int)
        final_end = start_time + budget_s - 0.1
        final_h, final_c5 = hill_climb_binary(final_bin, final_end, rng)
        if final_c5 < best_c5:
            best_h = final_h.copy()
            best_c5 = final_c5

    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    return best_h, best_c5, n
