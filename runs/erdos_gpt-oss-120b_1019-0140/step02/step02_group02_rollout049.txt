def enforce_sum(v, target, rng):
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # make adjustments greedily
    indices = np.arange(len(v))
    rng.shuffle(indices)
    for i in indices:
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            space = 1.0 - v[i]
            inc = min(diff, space)
            if inc > 0:
                v[i] += inc
                diff -= inc
        else:
            space = v[i]
            dec = min(-diff, space)
            if dec > 0:
                v[i] -= dec
                diff += dec
    # Final small fix if needed
    if abs(diff) > 1e-9:
        # adjust first element within bounds
        v[0] = np.clip(v[0] + diff, 0.0, 1.0)
    return v

def _c5_and_shift(h: np.ndarray):
    """Return (C5, k_max) for array h."""
    n = h.shape[0]
    dx = 2.0 / n
    size = 2 * n
    a = np.concatenate([h, np.zeros(n)])
    b_rev = np.concatenate([ (1.0 - h)[::-1], np.zeros(n) ])

    max_idx = int(np.argmax(conv_slice))
    c5 = float(conv_slice[max_idx] * dx)
    k = max_idx - (n - 1)
    return c5, k

    return _c5_and_shift(h)[0]

def _enforce_sum(v: np.ndarray, target: float, rng: np.random.Generator):
    """Adjust v in‑place so that sum(v) == target while keeping 0≤v≤1."""
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v

    rng.shuffle(idxs)
    for i in idxs:

                diff -= inc

            dec = min(-diff, space)

                diff += dec

        # choose a destination index with room
        cand = np.where(cur < 1.0 - 1e-12)[0]
        if cand.size == 0 or cur[i] <= 1e-12:
            srcs = np.where(cur > 1e-12)[0]
            dsts = np.where(cur < 1.0 - 1e-12)[0]
            if srcs.size == 0 or dsts.size == 0:
                break
            i = int(rng.choice(srcs))
            j = int(rng.choice(dsts))
        else:
            j_candidates = cand[cand != i]
            if j_candidates.size == 0:
                j = (i + 1) % n
            else:
                j = int(rng.choice(j_candidates))

        # capacity for a transfer i -> j
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            # try opposite direction
            max_delta = min(cur[j], 1.0 - cur[i])
            if max_delta <= 0:
                continue
            src, dst = j, i
        else:
            src, dst = i, j

        delta = rng.random() * max_delta * 0.5
        if delta <= 0:
            continue

        # apply move
        cur[src] -= delta
        cur[dst] += delta

        new_val = _c5(cur)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur.copy()
                if best_val < 0.3808:
                    return best_h, best_val
        else:
            # Metropolis acceptance
            total = deadline - start_time
            remain = deadline - time.time()
            frac = max(remain / total, 0.0)
            T0 = 0.001
            T = T0 * frac
            if T > 0 and np.exp(-(new_val - cur_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert
                cur[src] += delta
                cur[dst] -= delta

    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    n = 256                      # resolution (fast FFT, decent granularity)
    target_sum = n / 2.0

    # ---- initialise ----
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        # linear up‑sampling to the chosen resolution
        x_old = np.linspace(0, 1, init.shape[0], endpoint=False)
        x_new = np.linspace(0, 1, n, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
    else:
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h0[idx] = 1.0

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _enforce_sum(h0, target_sum, rng)

    best_h = h0.copy()
    best_val = _c5(best_h)

    # ---- first annealing from the (possibly good) start ----
    if time.time() < deadline - 0.2:
        cand_h, cand_val = _anneal_one(best_h, best_val,
                                       start_time, deadline, rng)
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val

    # ---- stochastic restarts ----
    while time.time() < deadline - 0.5:
        v = _random_vector(n, target_sum, rng)
        v_val = _c5(v)
        cand_h, cand_val = _anneal_one(v, v_val,
                                       start_time, deadline, rng)
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val
            if best_val < 0.3808:
                break

    # final safety adjustment
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _enforce_sum(best_h, target_sum, rng)
    final_val = _c5(best_h)

    return best_h.astype(float), float(final_val), n
