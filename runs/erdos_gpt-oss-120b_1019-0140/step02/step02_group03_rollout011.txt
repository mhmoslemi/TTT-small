def run(seed=42,budget_s=1100,**kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 200
        h0 = rng.random(n)
    target = n/2.0
    h0 = project_to_box_sum(np.clip(h0,0.,1.), target)

    # population
    pop_size = 30
    population = []
    # include h0 as first individual
    population.append(h0.copy())
    for _ in range(pop_size-1):
        v = np.clip(rng.random(n), 0., 1.)
        v = project_to_box_sum(v, target)
        population.append(v)

    # compute scores
    scores = [compute_c5(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # GA loop
    while True:
        cur_time = time.time()
        if cur_time - start > budget_s * 0.85:
            break
        # parent selection
        # tournament size 3
        def select():
            cand = rng.choice(pop_size, size=3, replace=False)
            cand_scores = [scores[i] for i in cand]
            return cand[np.argmin(cand_scores)]
        p1 = select()
        p2 = select()
        # crossover uniform
        mask = rng.random(n) < 0.5
        child = np.where(mask, population[p1], population[p2])
        # mutation Gaussian
        if rng.random() < 0.7:  # mutation probability
            sigma = 0.05
            child += rng.normal(scale=sigma, size=n)
        # clip and repair
        child = np.clip(child, 0., 1.)
        child = project_to_box_sum(child, target)
        # evaluate
        child_score = compute_c5(child)

        # replacement: if child better than worst, replace worst
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()
        # occasional local improvement: apply hill climbing after, for best
        if rng.random() < 0.02:
            # apply hill climbing to best_h
            cand_h, cand_score = hill_climb(best_h, best_score, rng, target, n, start, budget_s)
            if cand_score < best_score:
                best_h = cand_h
                best_score = cand_score
                # replace random individual with improved
                replace_idx = rng.integers(pop_size)
                population[replace_idx] = cand_h
                scores[replace_idx] = cand_score

    # final local refinement using SLSQP on best_h
    refined_h, refined_score = slsqp_refine(best_h, best_score, n, target)
    # ensure final projection
    refined_h = np.clip(refined_h, 0., 1.)
    refined_h = project_to_box_sum(refined_h, target)
    final_score = compute_c5(refined_h)

    return refined_h, final_score, n

def hill_climb(h, cur_score, rng, target, n, start_time, budget_s, max_iters=200):
    best = h.copy()
    best_score = cur_score
    dx = 2.0/n
    base_step = 0.3
    for i in range(max_iters):
        if time.time() - start_time > budget_s * 0.9:
            break
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2)
        cand = best + step * d
        cand = np.clip(cand, 0., 1.)
        cand = project_to_box_sum(cand, target)
        cand_score = compute_c5(cand)
        if cand_score + 1e-12 < best_score:
            best = cand
            best_score = cand_score
            base_step *= 1.05  # maybe inc
        else:
            base_step *= 0.99
    return best, best_score

def slsqp_refine(h, cur_score, n, target):
    # variable vector x = [h..., t]
    x0 = np.concatenate([h, [cur_score]])
    bounds = [(0.0,1.0)]*n + [(0.0, None)]
    dx = 2.0 / n
    def sum_con(x):
        return np.sum(x[:n]) - target
    constraints = [{'type':'eq','fun': sum_con}]
    # overlap constraints
    for k in range(-n+1, n):
        if k == 0:
            continue
        def make_con(k):
            def con(x):
                hvec = x[:n]
                if k >= 0:
                    overlap = np.dot(hvec[:n-k], (1.0 - hvec)[k:])
                else:
                    kk = -k
                    overlap = np.dot(hvec[kk:], (1.0 - hvec)[:n-kk])
                return x[-1] - overlap * dx
            return con
        constraints.append({'type':'ineq','fun': make_con(k)})
    res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter':500, 'ftol':1e-9, 'disp': False})
    if not res.success:
        # fallback: return original
        return h, cur_score
    h_opt = np.clip(res.x[:n], 0., 1.)
    h_opt = project_to_box_sum(h_opt, target)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
```

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project onto [lo,hi]^n while enforcing sum == target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def repair(v, target):
    """Clip to [0,1] and enforce exact sum."""
    v = np.clip(v, 0.0, 1.0)
    return project_to_box_sum(v, target)

def hill_climb(h, cur_score, rng, target, n, start_time, budget_s, max_iters=300):
    """Simple projected hill climbing."""
    best = h.copy()
    best_score = cur_score
    dx = 2.0 / n
    step_scale = 0.4
    for i in range(max_iters):
        if time.time() - start_time > budget_s * 0.95:
            break
        d = rng.normal(size=n)
        d -= d.mean()
        step = step_scale * (rng.random() ** 2)
        cand = best + step * d
        cand = repair(cand, target)
        cand_score = compute_c5(cand)
        if cand_score + 1e-12 < best_score:
            best = cand
            best_score = cand_score
            step_scale *= 1.02
        else:
            step_scale *= 0.99
    return best, best_score

def make_overlap_constraint(k, n, dx):
    """Factory for the constraint t >= overlap(k)*dx."""
    def con(x, kk=k):
        h = x[:n]
        if kk >= 0:
            ov = np.dot(h[:n-kk], (1.0 - h)[kk:])
        else:
            kk2 = -kk
            ov = np.dot(h[kk2:], (1.0 - h)[:n-kk2])
        return x[-1] - ov * dx
    return con

def slsqp_refine(h_start, t_start, n, target, maxiter=500):
    """Refine a candidate using SLSQP with overlap constraints."""
    dx = 2.0 / n
    x0 = np.concatenate([h_start, [t_start]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq',
                    'fun': lambda x: np.sum(x[:n]) - target}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        constraints.append({'type': 'ineq',
                            'fun': make_overlap_constraint(k, n, dx)})
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if not res.success:
        # Return original if optimizer fails
        return h_start, t_start
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = repair(h_opt, target)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) aiming for a low C5."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine dimensionality
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 150                     # larger than 80 for more flexibility
        h0 = rng.random(n)

    target_sum = n / 2.0
    h0 = repair(h0, target_sum)

    # GA parameters
    pop_size = 35
    population = [h0]
    for _ in range(pop_size - 1):
        v = rng.random(n)
        population.append(repair(v, target_sum))

    scores = [compute_c5(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # Main GA loop (steady‑state)
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.80:
            break

        # tournament selection
        def tournament():
            idx = rng.choice(pop_size, size=3, replace=False)
            vals = [scores[i] for i in idx]
            return idx[int(np.argmin(vals))]
        p1 = tournament()
        p2 = tournament()
        while p2 == p1:
            p2 = tournament()

        # uniform crossover
        mask = rng.random(n) < 0.5
        child = np.where(mask, population[p1], population[p2])

        # mutation (Gaussian)
        if rng.random() < 0.7:
            sigma = 0.07
            child += rng.normal(scale=sigma, size=n)

        child = repair(child, target_sum)
        child_score = compute_c5(child)

        # replace worst if better
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()

        # occasional modest hill‑climb on the current best
        if rng.random() < 0.03:
            hc_h, hc_score = hill_climb(best_h, best_score, rng,
                                        target_sum, n, start_time, budget_s)
            if hc_score + 1e-12 < best_score:
                best_h, best_score = hc_h, hc_score
                # inject into population
                replace = rng.integers(pop_size)
                population[replace] = best_h
                scores[replace] = best_score

    # Final refinement with SLSQP
    best_h, best_score = slsqp_refine(best_h, best_score, n, target_sum,
                                      maxiter=800)

    # If still not meeting target, try a few random SLSQP restarts
    while time.time() - start_time < budget_s * 0.97 and best_score > 0.3808:
        rand_h = repair(rng.random(n), target_sum)
        rand_t = compute_c5(rand_h)
        opt_h, opt_t = slsqp_refine(rand_h, rand_t, n, target_sum,
                                    maxiter=400)
        if opt_t < best_score:
            best_h, best_score = opt_h, opt_t

    # Ensure feasibility one last time
    best_h = repair(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
