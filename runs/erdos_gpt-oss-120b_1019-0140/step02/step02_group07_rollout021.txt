def project_to_bounds_sum(x, target):
    # Clip to [0,1] first? Not needed if we handle via clip inside.
    lo = -1.0
    hi = 2.0  # as shift can be enough
    for _ in range(30):
        lam = (lo + hi) / 2.0
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    # final projection
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    return proj

def evolve_es(mean, sigma, target_sum, time_end, rng):
    n = len(mean)
    best_h = mean.copy()
    best_score = compute_c5(best_h)
    # parameters
    lam = 12 # number of offspring
    mu = 4
    # Simple weights: equal.
    no_improve = 0
    # start iteration
    while time.time() < time_end:
        candidates = []
        scores = []
        for _ in range(lam):
            cand = mean + sigma * rng.standard_normal(n)
            cand = project_to_bounds_sum(cand, target_sum)
            score = compute_c5(cand)
            candidates.append(cand)
            scores.append(score)
        # sort by score
        idx_sorted = np.argsort(scores)
        # compute new mean as average of top mu
        top = idx_sorted[:mu]
        # update mean
        mean = np.mean([candidates[i] for i in top], axis=0)
        # adapt sigma based on successes
        successes = sum(scores[i] < best_score for i in range(lam))
        if successes > 0.2 * lam:

def binary_hill_climb(start_bin, time_end, rng):
    n = len(start_bin)
    target_sum = n // 2
    # Ensure start_bin balanced
    sum_bin = start_bin.sum()
    diff = target_sum - sum_bin
    if diff > 0:
        zeros = np.flatnonzero(start_bin == 0)
        rng.shuffle(zeros)
        start_bin[zeros[:diff]] = 1
    elif diff < 0:
        ones = np.flatnonzero(start_bin == 1)
        rng.shuffle(ones)
        start_bin[ones[: -diff]] = 0
    # Now hill climb
    h = start_bin.copy()
    best_score = compute_c5(h.astype(float))
    best_h = h.copy()
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        score = compute_c5(h.astype(float))
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            # revert
            h[i] = 1
            h[j] = 0
            no_improve += 1
    return best_h.astype(float), best_score

def project_to_bounds_sum(x, target):
    # binary search for lambda
    lo = -1.0
    hi = 2.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust residual
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        proj += diff / len(proj)
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def rudin_shapiro_seq(n):

import time
import numpy as np

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_bounds_sum(x, target):
    lo, hi = -1.0, 2.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        proj += diff / len(proj)
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def random_feasible_vector(n, target, rng):
    return project_to_bounds_sum(rng.random(n), target)

def make_balanced_binary(arr, target, rng):
    arr = arr.copy()
    cur = int(arr.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(arr == 0)
        rng.shuffle(zeros)
        arr[zeros[:diff]] = 1
    elif diff < 0:
        ones = np.flatnonzero(arr == 1)
        rng.shuffle(ones)
        arr[ones[: -diff]] = 0
    return arr

def thue_morse_seq(n):
    seq = np.empty(n, dtype=int)
    for i in range(n):
        seq[i] = bin(i).count('1') & 1
    return seq

def rudin_shapiro_seq(n):
    a = np.array([1], dtype=int)
    while len(a) < n:
        k = np.arange(len(a))
        sign = 1 - 2 * (k % 2)
        new = np.empty(2 * len(a), dtype=int)
        new[0::2] = a
        new[1::2] = a * sign
        a = new
    a = a[:n]
    return (a + 1) // 2

def evolve_es(mean, sigma, target, time_end, rng):
    n = len(mean)
    best_h = mean.copy()
    best_score = compute_c5(best_h)
    lam = 20
    mu = 5
    no_improve = 0
    while time.time() < time_end:
        candidates = []
        scores = []
        for _ in range(lam):
            cand = mean + sigma * rng.standard_normal(n)
            cand = project_to_bounds_sum(cand, target)
            sc = compute_c5(cand)
            candidates.append(cand)
            scores.append(sc)
        idx = np.argsort(scores)
        top_idx = idx[:mu]
        mean = np.mean([candidates[i] for i in top_idx], axis=0)
        mean_score = np.mean(scores)
        successes = sum(sc < mean_score for sc in scores)
        if successes > 0.2 * lam:
            sigma = min(sigma * 1.2, 0.5)
        elif successes < 0.2 * lam:
            sigma = max(sigma * 0.8, 1e-6)
        if scores[idx[0]] < best_score - 1e-12:
            best_score = scores[idx[0]]
            best_h = candidates[idx[0]].copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 500:
            mean = random_feasible_vector(n, target, rng)
            sigma = 0.2
            no_improve = 0
    return best_h, best_score

def binary_hill_climb(start_bin, time_end, rng):
    n = len(start_bin)
    target = n // 2
    # ensure exact balance
    cur = int(start_bin.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(start_bin == 0)
        rng.shuffle(zeros)
        start_bin[zeros[:diff]] = 1
    elif diff < 0:
        ones = np.flatnonzero(start_bin == 1)
        rng.shuffle(ones)
        start_bin[ones[: -diff]] = 0
    h = start_bin.copy()
    best_score = compute_c5(h.astype(float))
    best_h = h.copy()
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        sc = compute_c5(h.astype(float))
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = h.copy()
            no_improve = 0
        else:
            # revert
            h[i] = 1
            h[j] = 0
            no_improve += 1
    return best_h.astype(float), best_score

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 128
        h0 = random_feasible_vector(n, n / 2.0, rng)

    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_bounds_sum(h0, target_sum)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # seed pool
    seeds = [h0]
    bin_rand = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    bin_rand[idx] = 1.0
    seeds.append(bin_rand)

    try:
        tm = thue_morse_seq(n).astype(float)
        tm_bal = make_balanced_binary(tm.astype(int), n // 2, rng).astype(float)
        seeds.append(tm_bal)
    except Exception:
        pass

    try:
        rs = rudin_shapiro_seq(n).astype(float)
        rs_bal = make_balanced_binary(rs.astype(int), n // 2, rng).astype(float)
        seeds.append(rs_bal)
    except Exception:
        pass

    target_c5 = 0.3808

    while time.time() - start_time < budget_s - 0.8:
        remaining = budget_s - (time.time() - start_time)
        chunk = min(5.0, remaining * 0.3)
        end_time = time.time() + chunk
        seed_vec = seeds[rng.integers(len(seeds))].copy()
        cand_h, cand_score = evolve_es(seed_vec, 0.2, target_sum, end_time, rng)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
        if best_score <= target_c5:
            break

    # final binary refinement
    if time.time() - start_time < budget_s - 0.3:
        bin_start = (best_h >= 0.5).astype(int)
        bin_start = make_balanced_binary(bin_start, n // 2, rng)
        end_time = start_time + budget_s - 0.1
        fin_h, fin_score = binary_hill_climb(bin_start, end_time, rng)
        if fin_score < best_score:
            best_h, best_score = fin_h, fin_score

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_bounds_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n

    best_h = project_to_bounds_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
