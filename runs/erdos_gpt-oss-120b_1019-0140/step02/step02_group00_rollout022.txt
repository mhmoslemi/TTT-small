def hill_climb_binary(h, time_left_fn, best_score):
    n = h.size
    # Precompute list of indices for 1 and 0
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    current_score = compute_c5(h)
    while time_left_fn() > 0.01:
        # random pick
        i1 = rng.choice(ones)
        i2 = rng.choice(zeros)
        # swap
        h[i1], h[i2] = 0.0, 1.0
        new_score = compute_c5(h)
        if new_score <= current_score:
            # keep swap, update lists accordingly
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i1], h[i2] = 1.0, 0.0
        # early exit if best_score <= target
    return best_h, best_score

def binarize_and_balance(h, target_sum):
    n = h.size
    idx = np.argsort(np.abs(h - 0.5))
    h_bin = (h >= 0.5).astype(float)
    cur_sum = int(h_bin.sum())
    if cur_sum > target_sum:
        # need to turn (cur_sum - target_sum) ones to zero
        # select from np.where(h_bin==1) the ones with smallest (h - 0.5) (i.e., near 0.5)
        ones_idx = np.where(h_bin == 1)[0]
        # sort these by (h - 0.5) ascending (closer to 0.5)
        scores = np.abs(h[ones_idx] - 0.5)
        flip_idx = ones_idx[np.argsort(scores)[:cur_sum - target_sum]]

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximated on a uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Linear interpolation to a smaller size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average sub‑gradient of the max‑correlation term."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def slsqp_optimize_candidate(h0, max_iter=500):
    """Short SLSQP run for a candidate on the small scale."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def binarize_and_balance(h, target_sum):
    """Convert to 0/1 while preserving the exact sum target."""
    h_bin = (h >= 0.5).astype(float)
    cur = int(h_bin.sum())
    if cur > target_sum:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(np.abs(h[ones] - 0.5))
        flip = ones[order[:cur - target_sum]]
        h_bin[flip] = 0.0
    elif cur < target_sum:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(np.abs(h[zeros] - 0.5))
        flip = zeros[order[:target_sum - cur]]
        h_bin[flip] = 1.0
    return h_bin


def hill_climb_binary_start(h_start, max_iters, rng):
    """Greedy 1‑swap hill climbing on a balanced binary vector."""
    h = h_start.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    for _ in range(max_iters):
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = rng.choice(ones)
        i2 = rng.choice(zeros)
        h[i1] = 0.0
        h[i2] = 1.0
        new_score = compute_c5(h)
        if new_score <= cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i2] = 0.0
    return best_h, best_score

    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():

    # ---------------- Small‑scale SLSQP ----------------
    n_small = 64

        if sc_loc < best_score:

            if best_score <= TARGET:

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
