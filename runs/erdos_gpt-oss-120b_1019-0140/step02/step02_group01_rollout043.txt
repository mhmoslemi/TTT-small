Let's derive formally.

Let old vector be h. New vector h' is same as h except h'_i = 0, h'_j = 1, where originally h_i = 1, h_j=0.

We want for each lag l (k) new s_l' = sum_p h'_p * h'_{p+l}.

Let’s compute delta = s'_l - s_l.

We can break sum into contributions where p = i, p = j, p+l = i, p+l = j. Because only indices changed.

Specifically:

- Terms where p = i: old term = h_i * h_{i+l} = 1 * h_{i+l} = h_{i+l}. New term = h'_i * h'_{i+l} = 0 * h'_{i+l} = 0. So delta from p=i left side = - h_{i+l} (but note that h'_{i+l} might be different if i+l = j; but for l = j-i, i+l = j which changes value from 0 to 1 after swap. We need careful.)

def delta_s_for_swap(h, i_one, j_zero):
    # h binary, dtype int
    n = len(h)
    di = -1
    # We'll compute delta array of length 2*n-1
    L = 2*n - 1
    delta = np.zeros(L, dtype=int)
    # compute for each lag l via vectorized
    lags = np.arange(-n+1, n)
    idx = lags + n - 1
    # contributions from i_one
    # left: i+l
    ip = i_one + lags
    valid_ip = (ip >= 0) & (ip < n)
    delta[idx[valid_ip]] -= h[ip[valid_ip]]  # di=-1, so di * h = -h
    # right: i-l
    im = i_one - lags
    valid_im = (im >= 0) & (im < n)
    delta[idx[valid_im]] -= h[im[valid_im]]
    # contributions from j_zero
    # left: j+l, but exclude when j+l == i_one (since new h is 0)
    jp = j_zero + lags
    valid_jp = (jp >= 0) & (jp < n)
    # For those, need to add h[jp] except when jp == i_one
    mask_jp = valid_jp & (jp != i_one)
    delta[idx[mask_jp]] += h[jp[mask_jp]]
    # For jp == i_one, add 0 (since h'_i=0)
    # right: j-l
    jm = j_zero - lags
    valid_jm = (jm >= 0) & (jm < n)
    mask_jm = valid_jm & (jm != i_one)
    delta[idx[mask_jm]] += h[jm[mask_jm]]
    # This excludes double counting of the (i,j) pair.
    return delta

def binary_optimize_n(n, rng, time_limit):
    M = n // 2
    center = n - 1
    best_h = None
    best_min_s = -1
    start = time.time()
    # precompute lags and idx for delta function (constant)
    lags = np.arange(-n+1, n, dtype=np.int32)
    idxs = lags + n - 1
    while time.time() - start < time_limit - 0.01:
        # generate random binary h
        ones = rng.choice(n, M, replace=False)
        h = np.zeros(n, dtype=np.int8)
        h[ones] = 1
        s = np.correlate(h, h, mode='full')
        # compute min_s excluding center
        # Use masked min
        # Avoid creating new concatenated array each time: we can compute min via np.minimum:
        # Temporarily set s[center] = large number then min
        s_center_val = s[center]
        s[center] = np.iinfo(s.dtype).max  # temporarily ensure excluded
        cur_min_s = int(s.min())
        s[center] = s_center_val
        # hill climb
        # define patience param (no improvement attempts)
        patience = max(5000, n * 30)
        no_improve = 0
        while no_improve < patience and time.time() - start < time_limit - 0.01:
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            if h[i] == h[j]:
                continue
            # ensure i_one is where h=1
            if h[i] == 1:
                i_one, j_zero = i, j
            else:
                i_one, j_zero = j, i
            # compute delta using vectorized function
            delta = compute_delta_s(h, i_one, j_zero, lags, idxs, center)
            # compute new min_s
            new_s = s + delta
            # exclude center
            new_s_center = new_s[center]
            new_s[center] = np.iinfo(new_s.dtype).max
            new_min_s = int(new_s.min())
            new_s[center] = new_s_center
            if new_min_s > cur_min_s:
                # accept swap
                # update h
                h[i_one] = 0
                h[j_zero] = 1
                s = new_s
                cur_min_s = new_min_s
                no_improve = 0
                if cur_min_s > best_min_s:
                    best_min_s = cur_min_s
                    best_h = h.copy()
            else:
                no_improve += 1
        # after hill climb, maybe also try greedily swap some zeros with ones? Already done.
    # return best_h (float) and best_min_s
    return best_h, best_min_s

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap for a (float) step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _delta_autocorr(h, i_one, j_zero, lags, idxs):
    """
    Compute change in autocorrelation array (len 2n-1) caused by swapping
    1 at i_one with 0 at j_zero (binary vector h).
    """
    n = len(h)
    delta = np.zeros(lags.shape[0], dtype=np.int32)

    # contributions from i_one (going from 1 to 0)
    ip = i_one + lags
    vm = (ip >= 0) & (ip < n)
    delta[idxs[vm]] -= h[ip[vm]]
    im = i_one - lags
    vm = (im >= 0) & (im < n)
    delta[idxs[vm]] -= h[im[vm]]

    # contributions from j_zero (0 -> 1), but skip the position i_one
    jp = j_zero + lags
    vm = (jp >= 0) & (jp < n) & (jp != i_one)
    delta[idxs[vm]] += h[jp[vm]]
    jm = j_zero - lags
    vm = (jm >= 0) & (jm < n) & (jm != i_one)
    delta[idxs[vm]] += h[jm[vm]]

    return delta

def binary_optimize_n(n, rng, time_limit):
    """
    Search binary vectors of length n (sum = n/2) that maximize the minimal
    autocorrelation (equivalently minimise C5). Returns best vector and its
    minimal autocorrelation (excluding lag zero).
    """
    M = n // 2
    center = n - 1
    lags = np.arange(-n + 1, n, dtype=np.int32)
    idxs = lags + n - 1

    best_h = None
    best_min_s = -1
    start = time.time()
    # outer restarts
    while time.time() - start < time_limit - 0.02:
        # random binary start
        ones = rng.choice(n, M, replace=False)
        h = np.zeros(n, dtype=np.int8)
        h[ones] = 1
        s = np.correlate(h, h, mode='full')
        # exclude centre for min
        tmp = s[center]
        s[center] = np.iinfo(s.dtype).max
        cur_min_s = int(s.min())
        s[center] = tmp

        patience = max(2000, n * 15)
        no_improve = 0
        # hill climbing
        while no_improve < patience and time.time() - start < time_limit - 0.02:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if h[i] == h[j]:
                continue
            if h[i] == 1:
                i_one, j_zero = i, j
            else:
                i_one, j_zero = j, i

            delta = _delta_autocorr(h, i_one, j_zero, lags, idxs)

            new_s = s + delta
            tmp = new_s[center]
            new_s[center] = np.iinfo(new_s.dtype).max
            new_min_s = int(new_s.min())
            new_s[center] = tmp

            if new_min_s > cur_min_s:
                # accept swap
                h[i_one] = 0
                h[j_zero] = 1
                s = new_s
                cur_min_s = new_min_s
                no_improve = 0
                if cur_min_s > best_min_s:
                    best_min_s = cur_min_s
                    best_h = h.copy()
            else:
                no_improve += 1

    return best_h, best_min_s

def continuous_refinement(h, rng, time_limit):
    """Local mass‑transfer hill‑climbing on a real‑valued step function."""
    n = len(h)
    M = n // 2
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_limit - 0.01:
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - best_h[i], best_h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            best_h = h_new
            best_c5 = c5_new
    # final sum correction
    cur = np.sum(best_h)
    if abs(cur - M) > 1e-9:
        best_h += (M - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate discretisations (even)
    cand_ns = [80, 96, 112, 128, 144, 160, 176, 192]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 not in cand_ns:
            cand_ns.insert(0, n0)

    # binary optimisation phase (≈80 % of budget)
    for idx, n in enumerate(cand_ns):
        if time.time() - start_time > budget_s - 60:
            break
        remaining = budget_s - (time.time() - start_time)
        # split remaining time roughly equally among remaining n's, capped
        alloc = min(150.0, remaining / max(1, len(cand_ns) - idx) * 0.9)
        h_bin, min_s = binary_optimize_n(n, rng, alloc)
        if h_bin is None:
            continue
        c5_bin = (n // 2 - min_s) * (2.0 / n)
        if c5_bin < best_c5:
            best_h = h_bin.astype(float)
            best_c5 = c5_bin
            best_n = n
            if best_c5 < 0.3808:
                break

    # consider the provided parent construction
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
        if n_parent % 2 == 0:
            M = n_parent // 2
            cur = np.sum(h_parent)
            if abs(cur - M) > 1e-12:
                h_parent += (M - cur) / n_parent
                h_parent = np.clip(h_parent, 0.0, 1.0)
            c5_parent = compute_c5(h_parent)
            if c5_parent < best_c5:
                best_h = h_parent
                best_c5 = c5_parent
                best_n = n_parent

    # continuous refinement phase (use remaining time)
    time_left = budget_s - (time.time() - start_time)
    if best_h is not None and time_left > 5.0:
        refine_time = time_left * 0.9
        h_ref, c5_ref = continuous_refinement(best_h, rng, refine_time)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
            best_n = len(best_h)

    # safety fallback
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)

    # final constraint enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    n = best_n
    M = n // 2
    cur = np.sum(best_h)
    if abs(cur - M) > 1e-9:
        best_h += (M - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
