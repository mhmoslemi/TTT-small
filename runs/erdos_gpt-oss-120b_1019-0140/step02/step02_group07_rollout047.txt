def local_search(initial_h, time_limit, rng):
    n = len(initial_h)
    target_sum = n/2.0
    h = np.clip(initial_h, 0.0, 1.0)
    # enforce sum exactly by adjusting random entries if needed
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # adjust uniformly
        h += diff / n
        # clip and iterate fix
        for _ in range(10):
            overflow = h - 1.0
            underflow = -h
            if overflow.max() > 0:
                h[overflow>0] = 1.0
            if underflow.max() > 0:
                h[underflow>0] = 0.0
            cur_sum = h.sum()
            diff = target_sum - cur_sum
            if abs(diff) < 1e-12:
                break
            h += diff / n

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    T0 = 1e-2 # initial temp

        T = T0 * (1 - it / max_iter) + T_end * (it / max_iter)

import time
import math
import numpy as np

def compute_c5(arr):
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated by discrete correlation."""
    n = len(arr)
    # correlation of h with (1−h); mode='full' gives all integer shifts
    return np.max(np.correlate(arr, 1 - arr, mode="full")) * (2.0 / n)

def thue_morse_seq(n):
    """Length‑n Thue‑Morse binary sequence (0/1)."""
    seq = np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    """Length‑n Rudin‑Shapiro binary sequence (0/1)."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def repair_sum(vec, target):
    """Adjust vec to satisfy sum == target while staying in [0,1]."""
    v = np.clip(vec, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        v += diff / len(v)
        v = np.clip(v, 0.0, 1.0)
        # residual fix
        residual = target - v.sum()
        for i in range(len(v)):
            if abs(residual) < 1e-12:
                break
            adj = min(residual, 1 - v[i]) if residual > 0 else max(residual, -v[i])
            v[i] += adj
            residual = target - v.sum()
    return v

def anneal(start_h, end_time, rng):
    """Simulated annealing preserving the exact sum."""
    n = len(start_h)
    target = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5

    T0 = 0.10          # start temperature
    T_end = 1e-7       # final temperature
    t0 = time.time()
    duration = max(end_time - t0, 1e-9)

    while time.time() < end_time:
        # temperature decreasing linearly with elapsed fraction
        frac = (time.time() - t0) / duration
        T = T_end + (T0 - T_end) * (1.0 - frac)

        # choose indices that can exchange mass
        inc = np.where(h < 1.0 - 1e-12)[0]
        dec = np.where(h > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            continue
        # propose delta scaled by temperature (larger when T large)
        scale = max(0.1, T / T0)          # at least 0.1 of max_delta
        delta = max_delta * rng.random() * scale
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        if dE <= 0.0 or rng.random() < math.exp(-dE / (T + 1e-30)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    # final repair (tiny numerical drift)
    best_h = repair_sum(best_h, target)
    return best_h, best_c5

def binary_hill_climb(start_bin, end_time, rng):
    """Swap‑based hill climbing on binary vectors (0/1)."""
    n = len(start_bin)
    h = start_bin.astype(float)
    best_c5 = compute_c5(h)
    best_bin = start_bin.copy()
    no_improve = 0
    while time.time() < end_time and no_improve < 2500:
        ones = np.flatnonzero(best_bin)
        zeros = np.flatnonzero(1 - best_bin)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand_bin = best_bin.copy()
        cand_bin[i] = 0
        cand_bin[j] = 1
        cand_c5 = compute_c5(cand_bin.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_bin = cand_bin
            no_improve = 0
        else:
            no_improve += 1
    return best_bin.astype(float), best_c5

def random_balanced_binary(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    arr[rng.choice(n, n // 2, replace=False)] = 1.0
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- Initialise -----
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values, dtype=float)
        n = len(best_h)
    else:
        n = 120                      # fallback size (must be even)
        best_h = random_balanced_binary(n, rng)

    target_sum = n / 2.0
    best_h = repair_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    # ----- Prepare deterministic seeds -----
    seeds = []
    seeds.append(random_balanced_binary(n, rng))
    tm = thue_morse_seq(n)
    tm = repair_sum(tm.astype(float), target_sum)
    seeds.append(tm)
    rs = rudin_shapiro_seq(n)
    rs = repair_sum(rs.astype(float), target_sum)
    seeds.append(rs)

    # ----- Main iterative search -----
    while time.time() - start_time < budget_s - 0.6:
        remaining = budget_s - (time.time() - start_time)
        chunk = min(5.0, remaining * 0.45)          # seconds for this sub‑search
        end_chunk = time.time() + chunk

        # Choose starting point
        if rng.random() < 0.2:
            start_vec = rng.choice(seeds)
        elif rng.random() < 0.5:
            # Perturb current best slightly (Gaussian noise)
            noise = rng.normal(0.0, 0.03, size=n)
            start_vec = best_h + noise
            start_vec = repair_sum(start_vec, target_sum)
        else:
            start_vec = best_h.copy()

        # Run annealing
        cand_h, cand_c5 = anneal(start_vec, end_chunk, rng)

        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h, cand_c5
            if best_c5 <= 0.3808:
                break

        # Occasionally attempt a binary hill climb from rounded version
        if rng.random() < 0.15:
            bin_start = (best_h >= 0.5).astype(int)
            hop_end = time.time() + min(2.0, remaining * 0.25)
            bin_h, bin_c5 = binary_hill_climb(bin_start, hop_end, rng)
            if bin_c5 < best_c5:
                best_h, best_c5 = bin_h, bin_c5
                if best_c5 <= 0.3808:
                    break

    # ----- Final refinement with binary hill climbing -----
    if time.time() - start_time < budget_s - 0.3:
        bin_start = (best_h >= 0.5).astype(int)
        final_end = start_time + budget_s - 0.05
        final_h, final_c5 = binary_hill_climb(bin_start, final_end, rng)
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5

    # Ensure feasibility exactly before returning
    best_h = repair_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
