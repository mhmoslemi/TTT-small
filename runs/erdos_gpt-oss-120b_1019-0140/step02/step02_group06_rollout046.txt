def optimize_h_local(h0, time_budget, rng):

    target = n/2.0
    dx = 2.0/n
    best_h = h.copy()

    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.3
    no_improve = 0
    cooling = lambda elapsed: max(0.1, 1.0 - elapsed / time_budget)
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        # compute correlation and c5
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx

            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        vmax = v.max()
        w = np.exp(beta * (v - vmax))

def project_to_feasible(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    # We will distribute diff iteratively across entries not at bounds
    if abs(diff) < 1e-12:
        return x
    # If need to increase
    if diff > 0:
        # capacity to increase
        while diff > 1e-12:
            mask = x < 1.0 - 1e-12
            if not np.any(mask):
                break
            # amount we can add per such entry if equal distribution
            inc = min(diff / np.count_nonzero(mask), 1.0 - x[mask].max())
            # inc may be too large for some entries, thus we do incremental per element
            # simpler: loop over indices
            for idx in np.flatnonzero(mask):
                if diff <= 1e-12:
                    break
                avail = 1.0 - x[idx]
                add = min(diff, avail)
                x[idx] += add
                diff -= add
    else: # diff < 0
        while diff < -1e-12:
            mask = x > 1e-12
            if not np.any(mask):
                break
            for idx in np.flatnonzero(mask):
                if diff >= -1e-12:
                    break
                avail = x[idx]
                sub = min(-diff, avail)
                x[idx] -= sub
                diff += sub
    # final clip and return
    return np.clip(x, 0.0, 1.0)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_feasible(x, target):
    """Clip to [0,1] and adjust to achieve exact sum == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # distribute diff uniformly a few times, re‑clip each time
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        inc = diff / len(x)
        x += inc
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
    # any leftover is fixed greedily
    if abs(diff) > 1e-12:
        if diff > 0.0:
            for i in range(len(x)):
                if diff <= 0.0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            for i in range(len(x)):
                if diff >= 0.0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return np.clip(x, 0.0, 1.0)

def _subgradient_step(h, target, alpha):
    """One subgradient descent step for the max‑overlap objective."""
    n = h.shape[0]
    v = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(v))
    k = idx_max - (n - 1)
    if k >= 0:
        tail = slice(0, n - k)
        head = slice(k, n)
    else:
        k2 = -k
        tail = slice(k2, n)
        head = slice(0, n - k2)
    g = np.zeros(n)
    g[tail] += (1.0 - h[head])
    g[head] -= h[tail]
    g -= np.mean(g)                     # keep total mass unchanged
    h_new = h - alpha * g
    return project_to_feasible(h_new, target)

def _random_pair_move(h, rng):
    """Propose a feasible move that transfers mass between two coordinates."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # delta such that 0<=h[i]+delta<=1 and 0<=h[j]-delta<=1
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    if delta_max <= delta_min:
        return None
    delta = rng.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def _simulated_annealing(h, target, time_budget, rng):
    """Simple SA using pairwise moves."""
    start = time.time()
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    T0 = 0.004
    half = time_budget * 0.5
    # Phase 1: annealing
    while time.time() - start < half:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / half)
        cand = _random_pair_move(cur, rng)
        if cand is None:
            continue
        cand_c5 = compute_c5(cand)
        if cand_c5 < cur_c5 or rng.random() < np.exp(-(cand_c5 - cur_c5) / (T + 1e-12)):
            cur, cur_c5 = cand, cand_c5
            if cur_c5 < best_c5:
                best, best_c5 = cur.copy(), cur_c5
    # Phase 2: deterministic subgradient refinement
    remaining = time_budget - (time.time() - start)
    if remaining <= 0.0:
        return best, best_c5
    start_ref = time.time()
    alpha0 = 0.5
    while time.time() - start_ref < remaining:
        elapsed = time.time() - start_ref
        alpha = alpha0 * max(0.1, 1.0 - elapsed / remaining)
        cur = _subgradient_step(cur, target, alpha)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best, best_c5 = cur.copy(), cur_c5
    return best, best_c5

    """Hybrid SA + subgradient optimization."""
    n = h0.shape[0]
    target = n / 2.0
    return _simulated_annealing(h0, target, budget, rng)

        # pick grid size
        n = int(rng.choice(n_choices))
        h0 = rng.random(n)
        h0 = project_to_feasible(h0, n / 2.0)
        # allocate a fraction of the remaining time, but keep at least min_run
        time_left = remaining_budget()
        run_budget = min(30.0, max(min_run, time_left * 0.25))
        h_opt, c5_opt = optimize_h_local(h0, run_budget, rng)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n

    # 3) Final safety projection and evaluation
    if best_h is None:
        # fallback: simple half‑interval step
        best_n = 80
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        best_h = project_to_feasible(best_h, best_n / 2.0)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
