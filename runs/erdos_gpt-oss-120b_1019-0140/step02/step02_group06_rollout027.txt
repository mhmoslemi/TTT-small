    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # naive distribution
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        # Recompute diff
        cur = x.sum()
        diff = target - cur
        # if still off, fix iteratively
        if abs(diff) > 1e-12:
            # distribute remaining diff over free entries
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def random_feasible(n, rng, target):
    x = rng.random(n)
    return project_sum_and_bounds(x, target)

def mutate(h, rng, sigma):
    h_new = h + rng.normal(scale=sigma, size=h.shape)
    # No need to clip before projection, but do for stability
    #h_new = np.clip(h_new, 0.0, 1.0)
    return project_sum_and_bounds(h_new, h_new.shape[0]/2.0)

def subgradient_optimize(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    alpha0 = 0.2
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(v))
        L = max_idx - (n - 1)
        # subgradient of v_L
        g = np.zeros(n)
        if L >= 0:
            tail = slice(0, n - L)
            head = slice(L, n)
            g[tail] += 1.0 - h[head]
            g[head] -= h[tail]
        else:
            k = -L
            tail = slice(k, n)
            head = slice(0, n - k)
            g[tail] += 1.0 - h[head]
            g[head] -= h[tail]
        elapsed = time.time() - start
        alpha = alpha0 * max(0.01, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine grid size

        # Interpolate to n_desired points

        x_target = np.linspace(0.0, 2.0, n_desired, endpoint=False)
        h0 = np.interp(x_target, x0, init)
    else:
        h0 = None

    n = n_desired
    target_sum = n / 2.0

    # Initial population
    pop_size = 20
    population = []
    # first candidate: from parent if available else random
    if h0 is not None:
        candidate = project_sum_and_bounds(h0, target_sum)
        population.append((compute_c5(candidate), candidate))
    # fill rest
    while len(population) < pop_size:
        cand = random_feasible(n, rng, target_sum)
        population.append((compute_c5(cand), cand))

    # sort
    population.sort(key=lambda x: x[0])
    best_c5 = population[0][0]
    best_h = population[0][1].copy()

    # Evolutionary loop
    # Reserve some time for final refinement
    reserve_time = 8.0  # seconds
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= reserve_time:
            break
        # adapt sigma
        sigma = 0.2 * max(0.01, 1.0 - elapsed / budget_s) + 0.01
        new_pop = []
        # generate pop_size offspring
        for _ in range(pop_size):
            # select parent from top half
            parent_idx = rng.integers(0, max(1, pop_size // 2))
            parent = population[parent_idx][1]
            child = mutate(parent, rng, sigma)
            # occasional crossover

                other_idx = rng.integers(0, max(1, pop_size // 2))
                other = population[other_idx][1]
                child = (child + other) * 0.5
                child = project_sum_and_bounds(child, target_sum)
            c5 = compute_c5(child)
            new_pop.append((c5, child))
        # combine and select best
        population.extend(new_pop)
        population.sort(key=lambda x: x[0])
        population = population[:pop_size]
        # update best
        if population[0][0] < best_c5:
            best_c5 = population[0][0]
            best_h = population[0][1].copy()
        # optional random shake if stagnation? skip for simplicity

    # Final local refinement using subgradient method
    final_budget = max(0.0, budget_s - (time.time() - start_time))
    if final_budget > 2.0:
        refined_h, refined_c5 = subgradient_optimize(best_h, final_budget, seed + 12345)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n

def softmax_optimize(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.2
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # increase smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
        # softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w /= w_sum
        # gradient of weighted sum
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # zero-sum direction
        g -= np.mean(g)
        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        # accept if improvement or some random acceptance
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        # random shake if stuck
        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
    return best_h, best_c5

In worst case, if diff is beyond total capacity, the loop will adjust all entries to bound and diff remains, but then in the loop we will break after going through all entries but diff may remain non-zero, then loop ends because for loop ends; diff remains but function returns x with sum not exactly target. That may violate constraints.

But in practice, diff is small because we started from values within [0,1] and total sum after clipping is near target. But after any gradient update, may cause diff to be moderate, but not huge; the loop may be enough.

Alternatively, we could use a more robust projection that ensures sum exactly target within tolerance. Could implement a simple algorithm: After clipping, compute cur sum; diff = target - cur. If diff > 0: need to add mass to entries not at 1. Compute indices where x < 1. Compute capacity = sum(1 - x_i). If diff <= capacity: add diff * ( (1 - x_i) / capacity ) to each such i. If diff > capacity: set those to 1 and diff = diff - capacity; then repeat with new set (maybe none left). Similarly for diff < 0: reduce away from entries > 0.

This ensures sum exactly target or as close as possible. Implementing this is fine.

def project_sum_and_bounds(x, target, max_iter=10):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase
            free = (x < 1.0)
            capacity = np.sum(1.0 - x[free])
            if capacity < 1e-12:
                break
            inc = diff * ( (1.0 - x[free]) / capacity )
            x[free] += inc
        else:
            # need to decrease
            free = (x > 0.0)
            capacity = np.sum(x[free])
            if capacity < 1e-12:
                break
            dec = -diff * ( x[free] / capacity )
            x[free] -= dec
        # Clip again to avoid tiny overshoot
        x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap value multiplied by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target, max_iter=10):
    """Clip to [0,1] and enforce sum(x) == target (iterative proportional adjustment)."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase some values (those <1)
            free = x < 1.0
            capacity = np.sum(1.0 - x[free])
            if capacity < 1e-12:
                break
            inc = diff * (1.0 - x[free]) / capacity
            x[free] += inc
        else:
            # need to decrease some values (those >0)
            free = x > 0.0
            capacity = np.sum(x[free])
            if capacity < 1e-12:
                break
            dec = -diff * x[free] / capacity
            x[free] -= dec
        x = np.clip(x, 0.0, 1.0)
    return x

def random_feasible(n, rng, target):
    """Uniform random vector projected onto feasibility."""
    x = rng.random(n)
    return project_sum_and_bounds(x, target)

def mutate(h, rng, sigma):
    """Gaussian perturbation followed by projection."""
    h_new = h + rng.normal(scale=sigma, size=h.shape)
    return project_sum_and_bounds(h_new, h.shape[0] / 2.0)

def softmax_optimize(h0, time_budget, seed):
    """Gradient descent on smooth approximation of the max‐overlap."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.2
    no_improve = 0

    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase smoothing slowly
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w /= w_sum

        # gradient of weighted sum of overlaps
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # enforce zero‑sum direction (preserve total mass)
        g -= np.mean(g)

        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Target resolution
    n = 200
    target_sum = n / 2.0

    # Initialise from parent if available
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = len(init)
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_target = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_target, x0, init)
        h0 = project_sum_and_bounds(h0, target_sum)
    else:
        h0 = None

    best_h = None
    best_c5 = np.inf

    # Parameters for multi‑start optimisation
    attempts = 0
    max_attempts = 10
    reserve_time = 8.0  # seconds kept for final polishing

    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= reserve_time or attempts >= max_attempts:
            break
        # allocate time per attempt (at least 5 s)
        per_attempt = min(200.0, max(5.0, remaining - reserve_time))
        # choose start point
        if attempts == 0 and h0 is not None:
            start_vec = h0
        else:
            start_vec = random_feasible(n, rng, target_sum)
        # run local optimisation
        h_opt, c5_opt = softmax_optimize(start_vec, per_attempt, seed + attempts + 1)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        attempts += 1

    # Final polishing with any leftover budget
    remaining = budget_s - (time.time() - start_time)
    if remaining > 2.0 and best_h is not None:
        h_refined, c5_refined = softmax_optimize(best_h, remaining, seed + 9999)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined

    # Safety projection and final evaluation
    if best_h is None:
        # fallback to a random feasible solution
        best_h = random_feasible(n, rng, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
