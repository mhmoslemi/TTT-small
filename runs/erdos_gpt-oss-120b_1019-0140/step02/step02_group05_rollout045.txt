def compute_c5(h):
    n = len(h); dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr)*dx

def compute_corr_vector(h):
    return np.correlate(h, 1-h, mode='full')

def enforce_sum_and_bounds(h, target, tol=1e-12):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) <= tol:
        return h
    # find indices not at bound
    free = np.where((h > 0) & (h < 1))[0]
    if free.size > 0:
        h[free] += diff / free.size
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = target - cur
    # residual pass
    for i in range(len(h)):
        if abs(diff) < tol:
            break
        space_up = 1.0 - h[i]
        space_down = h[i]
        if diff > 0 and space_up > 0:
            add = min(space_up, diff)
            h[i] += add
            diff -= add
        elif diff < 0 and space_down > 0:
            sub = min(space_down, -diff)
            h[i] -= sub
            diff += sub
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for discretized h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum_and_bounds(h: np.ndarray, target: float, tol: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and adjust linearly to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) <= tol:
        return h
    # distribute diff uniformly over free variables
    free = np.where((h > 0) & (h < 1))[0]
    if free.size:
        h[free] += diff / free.size
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
    # residual pass – greedy fill
    for i in range(h.size):
        if abs(diff) < tol:
            break
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0.0, 1.0)

def objective_t(x: np.ndarray) -> float:
    """Objective for SLSQP – the auxiliary variable holding the max overlap."""
    return x[-1]

class SumConstraint:
    """Equality constraint: sum of h equals target."""
    def __init__(self, target: float):
        self.target = target
    def __call__(self, x: np.ndarray) -> float:
        return np.sum(x[:-1]) - self.target

class OverlapConstraint:
    """Inequality constraint: auxiliary t ≥ overlap for a given shift."""
    def __init__(self, shift: int, n: int, dx: float):
        self.shift = shift
        self.n = n
        self.dx = dx
    def __call__(self, x: np.ndarray) -> float:
        h = x[:-1]
        k = self.shift
        n = self.n
        if k >= 0:
            overlap = np.dot(h[: n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[: n - kk])
        return x[-1] - overlap * self.dx

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 120                     # a modest resolution giving flexibility
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target_sum = n / 2.0
    cur_h = enforce_sum_and_bounds(h0, target_sum)
    best_h = cur_h.copy()
    best_val = compute_c5(best_h)

    # ---- simulated annealing parameters ----------------------------
    T0 = 0.12                       # initial temperature
    Tmin = 1e-6
    p_swap = 0.30
    p_transfer = 0.30
    p_gradient = 0.25
    p_noise = 0.15
    dx = 2.0 / n
    last_smooth = start

    # ---- main stochastic search ------------------------------------
    while True:
        elapsed = time.time() - start
        # keep a small safety margin for final refinement
        if elapsed >= budget_s - 6.0:
            break
        T = max(Tmin, T0 * (1.0 - elapsed / budget_s))

        r = rng.random()
        # ----- swap a 1 with a 0 ------------------------------------
        if r < p_swap:
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # ----- transfer a random amount -------------------------------
        elif r < p_swap + p_transfer:
            i, j = rng.choice(n, size=2, replace=False)
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr <= 0:
                max_tr = min(cur_h[j], 1.0 - cur_h[i])
                if max_tr <= 0:
                    continue
                donor, recv = j, i
            else:
                donor, recv = i, j
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta

        # ----- gradient step for the worst shift ---------------------
        elif r < p_swap + p_transfer + p_gradient:
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = int(np.argmax(corr))
            k_shift = idx_max - (n - 1)
            neigh = np.zeros(n)
            if k_shift >= 0:
                neigh[: n - k_shift] += cur_h[k_shift:]
                neigh[k_shift:] += cur_h[: n - k_shift]
            else:
                kk = -k_shift
                neigh[kk:] += cur_h[: n - kk]
                neigh[: n - kk] += cur_h[kk:]
            grad = dx * (1.0 - neigh)            # d(overlap)/dh_i
            delta_vec = -grad
            delta_vec -= np.mean(delta_vec)      # preserve sum
            max_step = T * 4.0
            max_abs = np.max(np.abs(delta_vec))
            if max_abs > max_step:
                delta_vec = delta_vec * (max_step / max_abs)
            cand = cur_h + delta_vec
            cand = np.clip(cand, 0.0, 1.0)
            cand = enforce_sum_and_bounds(cand, target_sum)

        # ----- small random noise on a fraction of entries -----------
        else:
            cand = cur_h.copy()
            subset_sz = max(1, int(0.18 * n))
            subset = rng.choice(n, size=subset_sz, replace=False)
            noise = (rng.random(subset_sz) - 0.5) * T * 0.6
            cand[subset] += noise
            cand = np.clip(cand, 0.0, 1.0)
            cand = enforce_sum_and_bounds(cand, target_sum)

        new_val = compute_c5(cand)

        # ----- acceptance test ---------------------------------------
        accept = False
        if new_val < best_val:
            accept = True
        elif new_val < compute_c5(cur_h):
            # downhill move only if it improves current state
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - compute_c5(cur_h)) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_h = cand
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break

        # ----- periodic smoothing of the best candidate -------------
        if elapsed - last_smooth > 30.0:
            last_smooth = elapsed
            sm = np.convolve(best_h, np.array([0.25, 0.5, 0.25]), mode="same")
            sm = np.clip(sm, 0.0, 1.0)
            sm = enforce_sum_and_bounds(sm, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm.copy()
                cur_h = sm.copy()

    # ---- final convex‑penalty refinement using SLSQP ---------------
    remaining = budget_s - (time.time() - start)
    if remaining > 8.0:
        # build constraints
        constraints = [{"type": "eq", "fun": SumConstraint(target_sum)}]
        for kk in range(-n + 1, n):
            if kk == 0:
                continue
            constraints.append({"type": "ineq", "fun": OverlapConstraint(kk, n, dx)})
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]  # last variable is t
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=objective_t,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 1800, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:-1], 0.0, 1.0)
                h_opt = enforce_sum_and_bounds(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_and_bounds(best_h, target_sum)

    return best_h, best_val, n
