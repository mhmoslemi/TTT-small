global_target_sum = None

def compute_c5(arr):
    n = global_n
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_capped_simplex(v):
    target = global_target_sum
    y = np.clip(v, 0.0, 1.0)

        else:
            movable = (y > 1e-12)
        if not np.any(movable):
            break
        delta = residual / np.count_nonzero(movable)
        y[movable] += delta
        y = np.clip(y, 0.0, 1.0)
    return y

def de_objective(x):
    y = project_onto_capped_simplex(x)
    return compute_c5(y)

def thue_morse_seq(n):
    # generate binary sequence of length n
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def rudin_shapiro_seq(n):
    # generate Rudin-Shapiro binary sequence (0/1)
    a = np.array([1], dtype=int)
    while a.size < n:
        # a_{2k} = a_k, a_{2k+1} = (-1)^k * a_k
        k = np.arange(a.size)
        b = np.concatenate([a, a * ((-1) ** k)])
        a = b
    a = a[:n]
    # map -1->0, 1->1
    return ((a + 1) // 2).astype(float)

def adjust_to_target(bin_seq, target_sum, rng):
    seq = bin_seq.copy()
    diff = target_sum - seq.sum()
    if diff == 0:
        return seq
    idxs = np.arange(seq.size)
    rng.shuffle(idxs)
    for idx in idxs:
        if diff == 0:
            break
        if diff > 0 and seq[idx] < 0.5:
            seq[idx] = 1.0
            diff -= 1
        elif diff < 0 and seq[idx] > 0.5:
            seq[idx] = 0.0
            diff += 1
    return seq

def local_refine(h, time_end, rng):
    cur = h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    n = cur.shape[0]
    it = 0
    max_iters = 200000
    init_temp = 5e-3
    final_temp = 1e-7
    while time.time() < time_end:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # move delta from i to j
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        T = init_temp * (1 - it / max_iters) + final_temp * (it / max_iters)
        delta = max_delta * rng.random() * 0.5

        cand[j] += delta

        dE = cand_val - cur_val

            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
        it += 1

import time
import math
import numpy as np
from scipy.optimize import differential_evolution

# globals used by objective/repair functions
global_n = None
global_target_sum = None

def compute_c5(arr):
    n = global_n
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_capped_simplex(v):
    target = global_target_sum
    y = np.clip(v, 0.0, 1.0)
    for _ in range(50):
        residual = target - y.sum()
        if abs(residual) <= 1e-12:
            break
        if residual > 0:
            movable = (y < 1.0 - 1e-12)
        else:
            movable = (y > 1e-12)
        if not np.any(movable):
            break
        delta = residual / np.count_nonzero(movable)
        y[movable] += delta
        y = np.clip(y, 0.0, 1.0)
    return y

def de_objective(x):
    y = project_onto_capped_simplex(x)
    return compute_c5(y)

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def rudin_shapiro_seq(n):
    a = np.array([1], dtype=int)
    while a.size < n:
        k = np.arange(a.size)
        b = np.concatenate([a, a * ((-1) ** k)])
        a = b
    a = a[:n]
    return ((a + 1) // 2).astype(float)

def adjust_to_target(seq, target_sum, rng):
    seq = seq.copy()
    diff = int(round(target_sum - seq.sum()))
    if diff == 0:
        return seq
    idxs = np.arange(seq.size)
    rng.shuffle(idxs)
    for idx in idxs:
        if diff == 0:
            break
        if diff > 0 and seq[idx] < 0.5:
            seq[idx] = 1.0
            diff -= 1
        elif diff < 0 and seq[idx] > 0.5:
            seq[idx] = 0.0
            diff += 1
    return seq

def local_refine(h, time_end, rng):
    cur = h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    n = cur.shape[0]
    it = 0
    max_iters = 200000
    init_temp = 5e-3
    final_temp = 1e-7
    while time.time() < time_end:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        T = init_temp * (1 - it / max_iters) + final_temp * (it / max_iters)
        delta = max_delta * rng.random() * 0.5
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5(cand)
        dE = cand_val - cur_val
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
        it += 1
        if it >= max_iters:
            it = 0
    return best, best_val

def binary_hill_climb(start_bin, time_end, rng):
    n = start_bin.shape[0]
    cur_bin = start_bin.copy()
    cur_h = cur_bin.astype(float)
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(cur_bin == 1)
        zeros = np.flatnonzero(cur_bin == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand_bin = cur_bin.copy()
        cand_bin[i] = 0
        cand_bin[j] = 1
        cand_h = cand_bin.astype(float)
        cand_val = compute_c5(cand_h)
        if cand_val + 1e-12 < best_val:
            best_val = cand_val
            best_h = cand_h.copy()
            cur_bin = cand_bin
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    global global_n, global_target_sum
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        init_h = rng.random(n)

    target_sum = n / 2.0
    global_n = n
    global_target_sum = target_sum

    init_h = project_onto_capped_simplex(init_h)

    popsize_factor = 5
    dim = n
    total_pop = popsize_factor * dim
    init_pop = np.empty((total_pop, dim), dtype=float)
    init_pop[0] = init_h
    tm = adjust_to_target(thue_morse_seq(dim), target_sum, rng)
    rs = adjust_to_target(rudin_shapiro_seq(dim), target_sum, rng)
    init_pop[1] = tm
    init_pop[2] = rs
    if total_pop > 3:
        init_pop[3:] = rng.random((total_pop - 3, dim))

    best_h = init_h.copy()
    best_c5 = compute_c5(best_h)

    result = differential_evolution(
        de_objective,
        bounds=[(0.0, 1.0)] * dim,
        init=init_pop,
        popsize=popsize_factor,
        maxiter=500,
        polish=False,
        seed=seed
    )
    y_final = project_onto_capped_simplex(result.x)
    val_final = compute_c5(y_final)
    if val_final < best_c5:
        best_c5 = val_final
        best_h = y_final.copy()

    time_left = budget_s - (time.time() - start_time)
    if time_left > 1.0:
        end_refine = time.time() + time_left * 0.6
        h_refined, c5_refined = local_refine(best_h, end_refine, rng)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()

    time_left = budget_s - (time.time() - start_time)
    if time_left > 0.5:
        end_bin = time.time() + time_left * 0.6
        start_bin = (best_h >= 0.5).astype(int)
        h_bin, c5_bin = binary_hill_climb(start_bin, end_bin, rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
