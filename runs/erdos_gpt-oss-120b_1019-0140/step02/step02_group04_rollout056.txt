def get_mismatch_indices(h, k):
    """Return two arrays: up_idx where h[i]==1 and h[i+k]==0, and down_idx where h[i]==0 and h[i+k]==1.
    k can be positive or negative; handle bounds accordingly.
    """
    n = len(h)
    if k > 0:
        i = np.arange(0, n - k)
        cond_up = (h[i] == 1) & (h[i + k] == 0)
        cond_down = (h[i] == 0) & (h[i + k] == 1)
    elif k < 0:
        k = -k
        i = np.arange(k, n)
        cond_up = (h[i] == 1) & (h[i - k] == 0)
        cond_down = (h[i] == 0) & (h[i - k] == 1)
    else:

        return np.array([], dtype=int), np.array([], dtype=int)
    up_idx = i[cond_up]
    down_idx = i[cond_down]
    return up_idx, down_idx

import time
import math
import numpy as np

# ---------- Helper functions ----------
def compute_c5(h, dx):
    """Maximum overlap integral C5 for binary or continuous h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Clip to [0,1] and enforce exact sum (used for safety)."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # small uniform adjustment
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    # fix any remaining discrepancy exactly
    cur = np.sum(vec)
    diff = target_sum - cur
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec


def generate_random_binary(n, target_sum, rng):
    """Create a binary vector with exactly target_sum ones."""
    idx = rng.permutation(n)
    h = np.zeros(n, dtype=float)
    h[idx[:target_sum]] = 1.0
    return h


def get_mismatch_indices(h, k):
    """
    For shift k (integer, may be negative), return two arrays:
    up_idx: indices i where h[i]==1 and h[i+k]==0 (contribute to overlap)
    down_idx: indices i where h[i]==0 and h[i+k]==1 (can be swapped to reduce)
    """
    n = len(h)
    if k > 0:
        i = np.arange(0, n - k)
        up = (h[i] == 1.0) & (h[i + k] == 0.0)
        down = (h[i] == 0.0) & (h[i + k] == 1.0)
    elif k < 0:
        kk = -k
        i = np.arange(kk, n)
        up = (h[i] == 1.0) & (h[i - kk] == 0.0)
        down = (h[i] == 0.0) & (h[i - kk] == 1.0)
    else:
        return np.array([], dtype=int), np.array([], dtype=int)
    up_idx = i[up]
    down_idx = i[down]
    return up_idx, down_idx


def greedy_reduce(h, dx, max_iter=10000):
    """Deterministic reduction: repeatedly swap mismatched pairs on the worst shift."""
    n = len(h)
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        shift_k = max_idx - (n - 1)
        if shift_k == 0:
            break
        up_idx, down_idx = get_mismatch_indices(h, shift_k)
        if up_idx.size == 0 or down_idx.size == 0:
            break
        i = up_idx[0]
        j = down_idx[0]
        h[i] = 0.0
        h[j] = 1.0
    return h


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # --- problem size and initial binary vector ---
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        target_sum = n // 2
        # binary start from top entries of initial_h_values
        idx_desc = np.argsort(-h0)
        cur_h = np.zeros(n, dtype=float)
        cur_h[idx_desc[:target_sum]] = 1.0
    else:
        n = 80
        target_sum = n // 2
        cur_h = generate_random_binary(n, target_sum, rng)

    dx = 2.0 / n

    # --- evaluate initial solution ---
    cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
    cur_val = cur_corr.max() * dx
    best_h = cur_h.copy()
    best_val = cur_val

    # --- annealing parameters ---
    T = 0.02               # initial temperature
    T_min = 1e-6
    decay = 0.9995
    p_target = 0.8         # probability of using a targeted swap
    no_improve = 0
    restart_limit = 20000
    time_end = start_time + budget_s

    # --- main simulated annealing loop ---
    while time.time() < time_end - 0.01:
        # pick move type
        use_target = rng.random() < p_target
        i = j = None

        if use_target:
            max_idx = np.argmax(cur_corr)
            shift_k = max_idx - (n - 1)
            if shift_k != 0:
                up_idx, down_idx = get_mismatch_indices(cur_h, shift_k)
                if up_idx.size > 0 and down_idx.size > 0:
                    i = rng.choice(up_idx)
                    j = rng.choice(down_idx)

        if i is None:
            ones = np.flatnonzero(cur_h == 1.0)
            zeros = np.flatnonzero(cur_h == 0.0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # perform tentative swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0

        new_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        new_val = new_corr.max() * dx
        delta = new_val - cur_val

        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-delta / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_corr = new_corr
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = cur_h.copy()
                no_improve = 0
                if best_val <= 0.3808:
                    break
            else:
                no_improve += 1
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_improve += 1

        # temperature cooling
        T = max(T * decay, T_min)

        # restart if stuck
        if no_improve >= restart_limit:
            cur_h = generate_random_binary(n, target_sum, rng)
            cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            cur_val = cur_corr.max() * dx
            T = 0.02
            no_improve = 0

    # --- final deterministic reduction ---
    best_h = greedy_reduce(best_h.copy(), dx, max_iter=20000)

    # --- safety projection and final evaluation ---
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5(best_h, dx)

    return best_h, best_val, n
