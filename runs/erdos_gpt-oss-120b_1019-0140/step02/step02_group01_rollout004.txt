def anneal_h(h0, time_budget, rng):

    h = h0.copy()
    n = len(h)
    M = n // 2
    dx = 2.0 / n
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    start = time.time()
    # parameters:
    T0 = 0.001
    while time.time() - start < time_budget:
        # compute progress
        elapsed = time.time() - start
        progress = elapsed / time_budget
        # step factor
        step_factor = max(0.01, 0.5 * (1 - progress)) # early 0.5 reduces to 0.01

        # random pair
        i, j = rng.choice(n, 2, replace=False)
        # decide direction (increase i, decrease j) with 50% chance
        if rng.random() < 0.5:
            target, source = i, j
        else:
            target, source = j, i
        max_transfer = min(1.0 - h[target], h[source])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer * step_factor
        # small threshold to avoid too small moves?
        if delta < 1e-12:
            continue
        h_new = h.copy()
        h_new[target] += delta
        h_new[source] -= delta
        # ensure bounds incase numerical error
        # compute new c5
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-12:
            h = h_new
            c5 = c5_new
            # update best
            if c5_new < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = c5_new
        else:
            # accept with prob
            T = T0 * (1 - progress)
            if T > 0 and rng.random() < np.exp(-(c5_new - c5) / T):
                h = h_new
                c5 = c5_new
        # early break if best_c5 < target threshold maybe 0.3808
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

def generate_candidates(n, rng):
    M = n // 2
    candidates = []
    # binary random
    h_bin = np.zeros(n)
    idx = rng.choice(n, M, replace=False)
    h_bin[idx] = 1.0
    candidates.append(h_bin)
    # uniform random scaled
    h_uni = rng.random(n)
    h_uni *= M / np.sum(h_uni)
    # clip may disturb sum; adjust as before
    h_uni = np.clip(h_uni, 0., 1.)
    cur = np.sum(h_uni)
    if cur < M:
        deficit = M - cur
        for i in np.argsort(h_uni):
            inc = min(deficit, 1.0 - h_uni[i])
            h_uni[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    elif cur > M:
        excess = cur - M
        for i in np.argsort(-h_uni):
            dec = min(excess, h_uni[i])
            h_uni[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    candidates.append(h_uni)
    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n)
    scaling = M / np.sum(lin)
    h_lin = lin * scaling
    h_lin = np.clip(h_lin, 0., 1.)
    cur = np.sum(h_lin)
    if cur < M:
        deficit = M - cur
        for i in np.argsort(h_lin):
            inc = min(deficit, 1.0 - h_lin[i])
            h_lin[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    elif cur > M:
        excess = cur - M
        for i in np.argsort(-h_lin):
            dec = min(excess, h_lin[i])
            h_lin[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    candidates.append(h_lin)
    # if parent exists and matches n, add
    if 'initial_h_values' in globals():
        if len(initial_h_values) == n:
            h_parent = np.array(initial_h_values, dtype=float)
            # ensure exact sum
            cur = np.sum(h_parent)
            if abs(cur - M) > 1e-12:
                h_parent += (M - cur) / n
                h_parent = np.clip(h_parent, 0., 1.)
            candidates.append(h_parent)
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    candidate_ns = [80,96,112,128,144,160] # maybe extended

    # iterate over n
    for n in candidate_ns:

        if elapsed > budget_s - 5:  # keep margin

        # allocate per candidate equally
        if not candidates:

        alloc_per = alloc_n / len(candidates)
        for h0 in candidates:

            # compute sum and correct if needed:
            M = n // 2
            cur = np.sum(h0)

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _fix_sum(h: np.ndarray, M: int) -> np.ndarray:
    """Clamp to [0,1] and enforce exact sum M."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) > 1e-9:
        h = h + (M - cur) / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _generate_candidates(n: int, rng: np.random.Generator) -> list:
    """Return a list of feasible initial vectors for given n."""
    M = n // 2
    cand = []

    # parent if present
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        parent = np.array(initial_h_values, dtype=float)
        cand.append(_fix_sum(parent, M))

    # binary random (exact M ones)
    bin_h = np.zeros(n, dtype=float)
    bin_h[rng.choice(n, M, replace=False)] = 1.0
    cand.append(bin_h)

    # uniform random scaled to sum M, then projected
    uni_h = rng.random(n)
    uni_h *= M / np.sum(uni_h)
    uni_h = _fix_sum(uni_h, M)
    cand.append(uni_h)

    # linear decreasing shape scaled to sum M, then projected
    lin = np.linspace(1.0, 0.0, n)
    lin *= M / np.sum(lin)
    lin = _fix_sum(lin, M)
    cand.append(lin)

    return cand

def _anneal_h(h0: np.ndarray, time_budget: float, rng: np.random.Generator) -> tuple:
    """Simple simulated‑annealing hill‑climber for a single start."""
    n = h0.size
    M = n // 2
    h = _fix_sum(h0, M)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    cur_c5 = best_c5

    start = time.time()
    T0 = 0.001  # initial temperature

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        prog = elapsed / time_budget
        # step size shrinks from ~0.5 to 1e-3
        step_factor = max(1e-3, 0.5 * (1.0 - prog))
        # choose two distinct indices
        i, j = rng.choice(n, 2, replace=False)
        # pick direction randomly
        if rng.random() < 0.5:
            tgt, src = i, j
        else:
            tgt, src = j, i
        max_tr = min(1.0 - h[tgt], h[src])
        if max_tr <= 0.0:
            continue
        delta = rng.random() * max_tr * step_factor
        if delta < 1e-12:
            continue
        h_new = h.copy()
        h_new[tgt] += delta
        h_new[src] -= delta
        # evaluate
        new_c5 = _compute_c5(h_new)
        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = new_c5
        else:
            # occasional uphill acceptance
            T = T0 * (1.0 - prog)
            if T > 0 and rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                h = h_new
                cur_c5 = new_c5
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate discretisations (even numbers)
    n_list = [80, 96, 112, 128, 144, 160]

    # put parent size first if it exists
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 not in n_list:
            n_list.insert(0, n0)
        elif n0 % 2 == 0:
            n_list = [n0] + [v for v in n_list if v != n0]

    for n in n_list:
        # stop if running out of time
        if time.time() - start_time > budget_s - 5.0:
            break
        remaining = budget_s - (time.time() - start_time)
        # allocate at most 250 s or half of what's left
        alloc_n = min(250.0, remaining * 0.5)
        candidates = _generate_candidates(n, rng)
        if not candidates:
            continue
        alloc_per = alloc_n / len(candidates)

        for h0 in candidates:
            # ensure feasibility before annealing
            h0 = _fix_sum(h0, n // 2)
            h_opt, c5_opt = _anneal_h(h0, alloc_per, rng)
            if c5_opt < best_c5 - 1e-12:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n = n
                if best_c5 < 0.3808:
                    break
        if best_c5 < 0.3808:
            break

    # fallback if nothing succeeded
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = _compute_c5(best_h)

    # final projection onto constraints
    best_h = _fix_sum(best_h, best_n // 2)

    return best_h, best_c5, best_n
