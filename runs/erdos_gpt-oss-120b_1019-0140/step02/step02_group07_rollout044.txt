import time
import math
import numpy as np
from scipy.optimize import differential_evolution

def _compute_c5(arr, n):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated by correlation."""
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def _repair(x, target_sum, n):
    """
    Project x onto the feasible set:
        0 ≤ x_i ≤ 1, sum(x) = target_sum.
    Simple iterative water‑filling.
    """
    x = np.clip(x, 0.0, 1.0)
    # early exit if already feasible (within tolerance)
    s = np.sum(x)
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(x < 1.0 - 1e-14)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            # limit by space to upper bound
            max_inc = np.min(1.0 - x[free])
            inc = min(inc, max_inc)
            x[free] += inc
        else:
            free = np.where(x > 1e-14)[0]
            if free.size == 0:
                break
            dec = diff / free.size  # diff negative
            # limit by lower bound
            max_dec = -np.min(x[free])
            dec = max(dec, max_dec)
            x[free] += dec
        s = np.sum(x)
        diff = target_sum - s
    # final clip (should already be feasible)
    return np.clip(x, 0.0, 1.0)

def _objective_raw(raw, target_sum, n):
    """Objective for DE: repaired C5."""
    h = _repair(raw, target_sum, n)
    return _compute_c5(h, n)

def _continuous_anneal(start_h, time_end, rng, n, target_sum):
    """Simple simulated‑annealing refinement preserving the sum."""
    h = start_h.copy()
    cur_c5 = _compute_c5(h, n)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 200000
    it = 0
    while time.time() < time_end and it < max_iter:
        # pick inc/dec indices respecting bounds
        inc_idx = np.where(h < 1.0 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # repair (should already satisfy sum, but clip)
        h_new = np.clip(h_new, 0.0, 1.0)
        new_c5 = _compute_c5(h_new, n)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target_sum = n / 2.0

    # Ensure the starting vector is feasible
    h0 = _repair(h0, target_sum, n)

    # ---------- Differential Evolution ----------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        end_de = time.time() + remaining * 0.6  # keep some time for local refine
        # bounds for each coordinate
        bounds = [(0.0, 1.0)] * n

        # Build an initial population: one individual is the repaired start,
        # the rest are random.
        popsize = 15
        init_pop = rng.random((popsize, n))
        init_pop[0] = h0

        # Callback to respect the time limit
        def de_callback(xk, _):
            if time.time() >= end_de:
                return True
            return False

        result = differential_evolution(
            func=lambda raw: _objective_raw(raw, target_sum, n),
            bounds=bounds,
            strategy='best1bin',
            maxiter=2000,
            popsize=popsize,
            init=init_pop,
            polish=False,
            seed=seed,
            callback=de_callback,
            updating='deferred',
            disp=False,
        )
        best_h = _repair(result.x, target_sum, n)
        best_c5 = _compute_c5(best_h, n)
    else:
        best_h = h0
        best_c5 = _compute_c5(best_h, n)

    # ---------- Local refinement ----------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 2.0:
        time_end = time.time() + remaining * 0.9
        refined_h, refined_c5 = _continuous_anneal(best_h, time_end, rng, n, target_sum)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # Final safety: enforce feasibility exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1.0 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    # Return the solution
    return best_h, best_c5, n
