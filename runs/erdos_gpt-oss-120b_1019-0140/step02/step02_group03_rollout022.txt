import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # Clip and adjust via binary search
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k > 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        elif k < 0:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        else:
            # k==0 -> overlap = np.dot(h, 1.0 - h)
            overlap = np.dot(h, 1.0 - h)
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=600):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type':'eq','fun':sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol':1e-9, 'disp':False})
    return res

def hill_climb_random(start_h, start_c5, rng, time_deadline):
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_c5 = start_c5
    cur_h = start_h.copy()
    cur_c5 = start_c5
    step = 0.1  # initial move size
    # iter limit maybe 2000 or until time
    iter_count = 0
    while time.time() < time_deadline and iter_count < 2000:
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1   # ensure j != i
        # transfer amount
        max_transfer = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:
            # maybe try reverse direction
            max_transfer = min(step, cur_h[j], 1.0 - cur_h[i])
            if max_transfer <= 0:
                iter_count += 1
                continue
            # swap i and j (move from j to i)
            i, j = j, i
        delta = max_transfer * rng.random()
        if delta <= 0:
            iter_count += 1
            continue
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            step = min(step * 1.02, 0.5)
            # if improves global
            if cur_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cur_h.copy(), cur_c5
            # possible early exit if very good
            if best_c5 < 0.3806:
                break
        else:
            # no improvement
            step = max(step * 0.99, 1e-4)
        iter_count += 1
    return best_h, best_c5

        # local hill climb

                break

    # Phase 2: SLSQP refinement on best found
    # Allocate remaining time
    remaining = start_time + budget_s - time.time()
    if remaining > 5.0 and best_c5 > 0.3805:
        # Prepare constraints for current n
        n = best_h.shape[0]
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        # Run SLSQP a few times with random perturbations around best
        attempts = 3
        for _ in range(attempts):
            # small random perturbation to escape possible saddle

            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = project_to_box_sum(h_start, n / 2.0)
            t_start = compute_c5(h_start)
            res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500)
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_to_box_sum(h_opt, n / 2.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt + 1e-12 < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
                    if best_c5 < 0.3805:
                        break
            # update remaining time
            if time.time() - start_time > budget_s - 0.5:
                break

    # After all, final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, best_h.size / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_h.size

import time
import numpy as np
from scipy.optimize import minimize

# ----- Core utilities -----
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the box [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# ----- Constraints for SLSQP -----
class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k > 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        elif k < 0:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        else:
            overlap = np.dot(h, 1.0 - h)
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=600):
    """Run a single SLSQP optimisation starting from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

# ----- Stochastic hill‑climbing -----
def hill_climb_random(start_h, start_c5, rng, time_deadline):
    """Simple hill‑climb with pairwise mass transfers."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_c5 = start_c5
    cur_h = start_h.copy()
    cur_c5 = start_c5
    step = 0.1
    it = 0
    while time.time() < time_deadline and it < 2000:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_transfer = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0.0:
            max_transfer = min(step, cur_h[j], 1.0 - cur_h[i])
            if max_transfer <= 0.0:
                it += 1
                continue
            i, j = j, i
        delta = max_transfer * rng.random()
        if delta <= 0.0:
            it += 1
            continue
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            step = min(step * 1.02, 0.5)
            if cur_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cur_h.copy(), cur_c5
            if best_c5 < 0.3806:
                break
        else:
            step = max(step * 0.99, 1e-4)
        it += 1
    return best_h, best_c5

# ----- Main entry point -----
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Initialise from parent if available ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
    else:
        n0 = 80
        h0 = rng.random(n0)
    target_sum0 = n0 / 2.0
    h0 = project_to_box_sum(h0, target_sum0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---- Phase 1: stochastic hill‑climbing (≈55% of budget) ----
    phase1_deadline = start_time + budget_s * 0.55
    while time.time() < phase1_deadline:
        # Random start: half the time binary, half continuous
        if rng.random() < 0.5:
            h_start = np.zeros(n0)
            idx = rng.choice(n0, size=n0 // 2, replace=False)
            h_start[idx] = 1.0
        else:
            h_start = rng.random(n0)
            h_start = project_to_box_sum(h_start, target_sum0)
        c5_start = compute_c5(h_start)
        sub_deadline = min(time.time() + 0.25, phase1_deadline)
        h_opt, c5_opt = hill_climb_random(h_start, c5_start, rng, sub_deadline)
        if c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt.copy(), c5_opt
            if best_c5 < 0.3806:
                break

    # ---- Phase 2: SLSQP refinement (use remaining time) ----
    remaining = start_time + budget_s - time.time()
    if remaining > 5.0 and best_c5 > 0.3805:
        n = best_h.shape[0]
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        attempts = 3
        for _ in range(attempts):
            # Small random perturbation around the current best
            h_start = best_h + 0.02 * (rng.random(n) - 0.5)
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = project_to_box_sum(h_start, n / 2.0)
            t_start = compute_c5(h_start)
            res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500)
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_to_box_sum(h_opt, n / 2.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt + 1e-12 < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
                    if best_c5 < 0.3805:
                        break
            if time.time() - start_time > budget_s - 0.5:
                break

    # ---- Final projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, best_h.size / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_h.size
