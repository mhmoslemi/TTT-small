def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    get n: if 'initial_h_values' in globals(): n = len(initial_h_values)

    m = n // 2

    dx = 2.0 / n

    # try MILP
    try:
        import cvxpy as cp
        # define variables
        h = cp.Variable(n, boolean=True)
        t = cp.Variable(integer=True)
        constraints = []
        constraints.append(cp.sum(h) == m)
        constraints.append(t >= 0)
        constraints.append(t <= m)
        # for each shift
        for k in range(1, n):
            # define p_k
            p_k = cp.Variable(n - k, boolean=True)
            constraints.append(p_k <= h[: n - k])
            constraints.append(p_k <= h[k:])
            constraints.append(p_k >= h[: n - k] + h[k:] - 1)
            overlap_expr = cp.sum(h[: n - k]) - cp.sum(p_k)
            constraints.append(t >= overlap_expr)
        prob = cp.Problem(cp.Minimize(t), constraints)
        # remaining time
        elapsed = time.time() - start_time
        time_left = max(0.0, budget_s - elapsed - 5.0)  # subtract some buffer
        # Solve with time limit
        # The solver argument may need to be cp.CBC or cp.GLPK_MI etc.
        solver = None
        # Choose solver based on availability
        if hasattr(cp, "CBC"):
            solver = cp.CBC
        elif hasattr(cp, "GLPK_MI"):
            solver = cp.GLPK_MI
        else:
            solver = None
        solution_status = None
        if solver is not None:
            prob.solve(solver=solver, verbose=False, max_seconds=time_left)
            solution_status = prob.status
        else:
            # fallback: try default
            prob.solve(verbose=False)
            solution_status = prob.status

        if solution_status in ["optimal", "optimal_inaccurate", "feasible"]:
            h_sol = np.rint(np.array(h.value)).astype(int)

import time, numpy as np

def compute_c5_vec(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def random_binary_vector(n, m, rng):
    vec = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=m, replace=False)
    vec[idx] = 1
    return vec

def repair_binary(vec, target_sum, rng):
    cur = int(vec.sum())
    diff = target_sum - cur
    if diff > 0:
        zeros = np.where(vec == 0)[0]
        if len(zeros) >= diff:
            chosen = rng.choice(zeros, size=diff, replace=False)
            vec[chosen] = 1
    elif diff < 0:
        ones = np.where(vec == 1)[0]
        d = -diff
        if len(ones) >= d:
            chosen = rng.choice(ones, size=d, replace=False)
            vec[chosen] = 0
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80
    m = n // 2
    dx = 2.0 / n

    # try integer programming formulation
    try:
        import cvxpy as cp
        # variables
        h = cp.Variable(n, boolean=True)
        t = cp.Variable(integer=True)
        constr = [cp.sum(h) == m, t >= 0, t <= m]

        # build constraints for each shift
        for k in range(1, n):
            hi = h[: n - k]
            hj = h[k:]
            p = cp.Variable(n - k, boolean=True)
            constr.append(p <= hi)
            constr.append(p <= hj)
            constr.append(p >= hi + hj - 1)
            overlap = cp.sum(hi) - cp.sum(p)  # S_k - sum(p_k)
            constr.append(overlap <= t)

        prob = cp.Problem(cp.Minimize(t), constr)

        # time allocation for solver
        elapsed = time.time() - start
        time_left = max(0.0, budget_s - elapsed - 5.0)
        # choose solver
        solver = None
        if hasattr(cp, 'CBC'):
            solver = cp.CBC
        elif hasattr(cp, 'GLPK_MI'):
            solver = cp.GLPK_MI
        # solve with time limit if possible
        solve_kwargs = {'verbose': False}
        if solver is not None:
            solve_kwargs['solver'] = solver
        # many cvxpy solvers accept 'max_seconds'
        solve_kwargs['max_seconds'] = time_left
        prob.solve(**solve_kwargs)

        if prob.status in ('optimal', 'optimal_inaccurate', 'feasible'):
            h_val = np.rint(np.array(h.value)).astype(int)
            # ensure exact sum
            cur_sum = int(h_val.sum())
            if cur_sum != m:
                h_val = repair_binary(h_val, m, rng)
            best_h = h_val.astype(float)
            best_val = compute_c5_vec(best_h, dx)
            return best_h, best_val, n
    except Exception:
        # any import or solve error falls back to heuristic
        pass

    # fallback: simple binary hill climbing
    best_h = random_binary_vector(n, m, rng)
    best_val = compute_c5_vec(best_h.astype(float), dx)
    stagnation = 0
    while time.time() - start < budget_s - 0.5:
        i, j = rng.integers(n, size=2)
        if best_h[i] == best_h[j]:
            continue
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best_h, best_val = cand, cand_val
            stagnation = 0
            if best_val < 0.3808:
                break

import time
import numpy as np

def compute_c5_vec(h, dx):
    """Maximum overlap integral for a given step function."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_binary_vector(n, m, rng):
    """Binary vector of length n with exactly m ones."""
    vec = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=m, replace=False)
    vec[idx] = 1
    return vec

def repair_binary(vec, target_sum, rng):
    """Adjust binary vector to achieve the exact target sum."""
    cur = int(vec.sum())
    diff = target_sum - cur
    if diff > 0:
        zeros = np.where(vec == 0)[0]
        if len(zeros) >= diff:
            chosen = rng.choice(zeros, size=diff, replace=False)
            vec[chosen] = 1
    elif diff < 0:
        ones = np.where(vec == 1)[0]
        d = -diff
        if len(ones) >= d:
            chosen = rng.choice(ones, size=d, replace=False)
            vec[chosen] = 0
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # problem dimension
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 80
    m = n // 2
    dx = 2.0 / n

    # ---------- MILP attempt ----------
    try:
        import cvxpy as cp

        # decision variables
        h = cp.Variable(n, boolean=True)
        t = cp.Variable(integer=True)

        constraints = [
            cp.sum(h) == m,
            t >= 0,
            t <= m,
        ]

        # constraints for each positive shift
        for k in range(1, n):
            hi = h[: n - k]
            hj = h[k:]
            p = cp.Variable(n - k, boolean=True)          # AND of hi and hj
            constraints.append(p <= hi)
            constraints.append(p <= hj)
            constraints.append(p >= hi + hj - 1)
            overlap = cp.sum(hi) - cp.sum(p)               # S_k - pair count
            constraints.append(overlap <= t)

        prob = cp.Problem(cp.Minimize(t), constraints)

        # allocate part of the budget to the solver
        elapsed = time.time() - start
        time_left = max(0.0, budget_s - elapsed - 5.0)

        solver = None
        if hasattr(cp, "CBC"):
            solver = cp.CBC
        elif hasattr(cp, "GLPK_MI"):
            solver = cp.GLPK_MI

        solve_opts = {"verbose": False, "max_seconds": time_left}
        if solver is not None:
            solve_opts["solver"] = solver

        prob.solve(**solve_opts)

        if prob.status in ("optimal", "optimal_inaccurate", "feasible"):
            h_val = np.rint(np.array(h.value)).astype(int)
            # ensure exact sum
            if int(h_val.sum()) != m:
                h_val = repair_binary(h_val, m, rng)
            h_vec = h_val.astype(float)
            c5 = compute_c5_vec(h_vec, dx)
            return h_vec, c5, n
    except Exception:
        pass  # fall back to heuristic if MILP fails

    # ---------- Simple binary hill‑climbing fallback ----------
    best_h = random_binary_vector(n, m, rng)
    best_val = compute_c5_vec(best_h.astype(float), dx)
    stagnation = 0

    while time.time() - start < budget_s - 0.5:
        i, j = rng.integers(n, size=2)
        if best_h[i] == best_h[j]:
            continue
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]   # swap a 0 and a 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best_h, best_val = cand, cand_val
            stagnation = 0
            if best_val < 0.3808:
                break
        else:
            stagnation += 1
        if stagnation > 10000:
            # random restart to avoid stagnation
            best_h = random_binary_vector(n, m, rng)
            best_val = compute_c5_vec(best_h.astype(float), dx)
            stagnation = 0

    return best_h.astype(float), best_val, n
