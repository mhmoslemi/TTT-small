def subgradient_descent(h, dx, target_sum, time_limit, start_time, rng):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5_vec(best_h, dx)
    cur_h = h.copy()
    cur_val = best_val
    step = 0.2
    while time.time() < time_limit:
        # compute prefix sums
        pref = np.concatenate(([0], np.cumsum(cur_h)))
        # compute overlap via correlation
        corr = np.correlate(cur_h, cur_h, mode='full')
        overlap = np.empty(n, dtype=float)
        overlap[0] = target_sum
        overlap[1:] = corr[n:]
        # compute c5 values for each k (1..n-1)
        ks = np.arange(1, n)
        a = pref[n - ks]
        b = target_sum - pref[ks]
        c5_vals = np.maximum(a, b) - overlap[ks]
        k_max = int(ks[np.argmax(c5_vals)])  # shift with max
        # gradient
        g = np.zeros(n, dtype=float)
        mask = np.arange(n) < n - k_max
        g[mask] = 1.0
        if k_max < n:
            g[:n - k_max] -= cur_h[k_max:]
        if k_max > 0:
            g[k_max:] -= cur_h[:n - k_max]
        # step
        new_h = cur_h - step * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_sum(new_h, target_sum)
        new_val = compute_c5_vec(new_h, dx)
        # update best
        if new_val < best_val:
            best_val = new_val
            best_h = new_h.copy()
        # maybe adjust step based on improvement
        if new_val < cur_val - 1e-9:
            cur_h = new_h
            cur_val = new_val
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.95
        # optional early break
        if best_val < 0.3808:
            break
    return best_h, best_val

def subgradient_descent(h, dx, target_sum, time_limit, rng):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5_vec(h, dx)
    cur_h = h.copy()
    cur_val = best_val
    step = 0.2
    while time.time() < time_limit:
        # compute prefix
        pref = np.concatenate(([0.0], np.cumsum(cur_h)))
        # compute overlap
        corr = np.correlate(cur_h, cur_h, mode='full')
        overlap = np.empty(n, dtype=float)
        overlap[0] = target_sum
        overlap[1:] = corr[n:] # shape n-1
        # compute c5 values
        ks = np.arange(1, n)
        a = pref[n - ks]
        b = target_sum - pref[ks]
        c5_vals = np.maximum(a, b) - overlap[ks]
        k_max = int(ks[np.argmax(c5_vals)])
        # gradient
        g = np.zeros(n, dtype=float)
        mask = np.arange(n) < n - k_max
        g[mask] = 1.0
        if k_max < n:
            g[:n - k_max] -= cur_h[k_max:]
        if k_max > 0:
            g[k_max:] -= cur_h[:n - k_max]
        # step update
        new_h = cur_h - step * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_sum(new_h, target_sum)
        new_val = compute_c5_vec(new_h, dx)
        # update best
        if new_val < best_val:
            best_val = new_val
            best_h = new_h.copy()
        # accept or not
        if new_val < cur_val - 1e-12:
            cur_h = new_h
            cur_val = new_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.9
        # early stop if reaches target
        if best_val < 0.3808:
            break
    return best_h, best_val

import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Project vec onto the feasible set: 0≤vec≤1 and sum(vec)=target_sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible vector with values in [0,1] and exact sum."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

def subgradient_descent(h_start, dx, target_sum, time_limit, rng):
    """Projected subgradient descent on max_k (w_k - overlap_k)."""
    n = len(h_start)
    best_h = h_start.copy()
    best_val = compute_c5_vec(best_h, dx)

    cur_h = h_start.copy()
    cur_val = best_val
    step = 0.2
    while time.time() < time_limit:
        # prefix sums
        pref = np.concatenate(([0.0], np.cumsum(cur_h)))
        # overlap via full correlation of h with itself
        corr = np.correlate(cur_h, cur_h, mode="full")
        overlap = np.empty(n, dtype=float)
        overlap[0] = target_sum
        overlap[1:] = corr[n:]          # shifts 1 .. n-1

        ks = np.arange(1, n)
        a = pref[n - ks]               # sum of first n-k entries
        b = target_sum - pref[ks]      # sum of last n-k entries
        c5_vals = np.maximum(a, b) - overlap[ks]
        k_max = int(ks[np.argmax(c5_vals)])

        # gradient of the active shift
        g = np.zeros(n, dtype=float)
        mask = np.arange(n) < n - k_max
        g[mask] = 1.0
        if k_max < n:
            g[: n - k_max] -= cur_h[k_max:]
        if k_max > 0:
            g[k_max:] -= cur_h[: n - k_max]

        # step and projection
        new_h = cur_h - step * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_sum(new_h, target_sum)

        new_val = compute_c5_vec(new_h, dx)

        if new_val < best_val:
            best_val = new_val
            best_h = new_h.copy()
        if new_val < cur_val - 1e-12:
            cur_h = new_h
            cur_val = new_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.9

        if best_val < 0.3808:
            break
    return best_h, best_val


# ---------- Local continuous refinement ----------
def refine_continuous(h, dx, target_sum, time_limit, rng):
    """Simple mass‑transfer hill‑climbing."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.02
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-8)[0]
        recips = np.where(cur_h < 1.0 - 1e-8)[0]
        if donors.size == 0 or recips.size == 0:
            break
        i = int(rng.choice(donors))
        j = int(rng.choice(recips))
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_sum(cand, target_sum)
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.95
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # initial feasible vectors
    start_vectors = []
    if h0 is not None:
        start_vectors.append(project_to_sum(np.clip(h0, 0.0, 1.0), target_sum))
    start_vectors.append(generate_random_feasible(n, target_sum, rng))

    best_h = None
    best_val = float("inf")
    for vec in start_vectors:
        val = compute_c5_vec(vec, dx)
        if val < best_val:
            best_val = val
            best_h = vec.copy()
        if best_val < 0.3808:
            return best_h, best_val, n

    # main optimisation loop (random restarts + subgradient)
    deadline = start_time + budget_s - 0.5
    while time.time() < deadline - 7.0:
        h_start = generate_random_feasible(n, target_sum, rng)
        opt_h, opt_val = subgradient_descent(
            h_start, dx, target_sum, time.time() + 7.0, rng
        )
        if opt_val < best_val:
            best_val = opt_val
            best_h = opt_h.copy()
        if best_val < 0.3808:
            break

    # final fine‑tuning on the best candidate
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_h, best_val = refine_continuous(
            best_h, dx, target_sum, time.time() + remaining, rng
        )

    # final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
