            j = i + k
            if 0 <= j < n and best_h[j] < 1.0:
                max_transfer = min(best_h[i], 1.0 - best_h[j])
            else:
                cand = np.where(best_h < 1.0 - 1e-12)[0]
                if cand.size == 0:
                    continue
                j = rng.choice(cand)
                max_transfer = min(best_h[i], 1.0 - best_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # random small mass transfer
            src = np.where(best_h > 1e-12)[0]
            dst = np.where(best_h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_transfer = min(best_h[i], 1.0 - best_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta

        new_val = compute_c5(h_new)
        if new_val < best_val:
            best_h = h_new
            best_val = new_val
            if best_val < 0.3808:
                break
        else:
            # SA acceptance
            if rng.random() < math.exp(-(new_val - best_val) / max(T, 1e-12)):
                best_h = h_new
                best_val = new_val

    # ----- final safety clip and repair -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_sum(best_h, target, rng)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
