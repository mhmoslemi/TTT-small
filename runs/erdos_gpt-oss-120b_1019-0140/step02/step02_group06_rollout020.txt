def optimize_continuous(h0, target, budget, start_time):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.1
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iter_count = 0
    no_improve = 0
    # parameters for binary rounding/check
    binary_check_interval = max(1000, n)
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget:
            break
        # compute correlation vector v
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # update beta smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        # softmax weights
        vmax = corr.max()
        w = np.exp(beta * (corr - vmax))
        w_sum = w.sum()
        w = w / w_sum

        # gradient of weighted sum O = sum w_i * O_i
        grad = np.zeros_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
                grad[tail] += weight * (1.0 - h[head])
                grad[head] -= weight * h[tail]
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
                grad[tail] += weight * (1.0 - h[head])
                grad[head] -= weight * h[tail]

        # ensure zero-sum direction
        grad -= grad.mean()

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow = beta1 ** (iter_count + 1)
        beta2_pow = beta2 ** (iter_count + 1)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        alpha = alpha0 * max(0.2, 1.0 - elapsed / budget)  # keep some learning
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        h_new = h - step
        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        # accept if improves or sometimes even if not to escape
        if c5_new < c5 or np.random.rand() < 0.001:
            h = h_new

        # binary rounding occasional
        if iter_count % binary_check_interval == 0:
            # binary candidate using top half values
            s = int(n // 2)
            idx = np.argpartition(h, -s)[-s:]
            binary = np.zeros_like(h)
            binary[idx] = 1.0
            binary = project_sum_and_bounds(binary, target)
            c5_bin = compute_c5(binary)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = binary.copy()

        # random shake after stagnation
        if no_improve > 4000:
            h = project_sum_and_bounds(h + np.random.normal(scale=0.03, size=n), target)
            no_improve = 0

        iter_count += 1
    return best_h, best_c5

def compute_c5(h):

    dx = 2.0 / n
    # correlation of h and 1-h, full mode
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_sum_and_bounds(x, target):
    # Clip values to [0,1]
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # try to distribute diff uniformly a few times
    for _ in range(5):
        add = diff / x.size
        x += add
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            return x
    # leftover adjustment by scanning
    if diff > 0:
        # need to increase some entries
        for i in range(x.size):
            if diff <= 0:
                break
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
    elif diff < 0:
        # need to decrease some entries
        for i in range(x.size):
            if diff >= 0:
                break
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
    return x

def grad_softmax_weighted(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    vmax = corr.max()
    w = np.exp(beta * (corr - vmax))
    w_sum = w.sum()
    w = w / w_sum
    grad = np.zeros_like(h)
    offset = n - 1
    # loop over each shift
    for idx, weight in enumerate(w):
        if weight < 1e-16:
            continue
        k = idx - offset
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
            grad[tail] += weight * (1.0 - h[head])
            grad[head] -= weight * h[tail]
        else:

            tail = slice(k2, n)
            head = slice(0, n - k2)
            grad[tail] += weight * (1.0 - h[head])
            grad[head] -= weight * h[tail]
    # make gradient tangent to sum constraint
    grad -= grad.mean()
    return grad, corr

def compute_c5(h):
    """Maximum overlap integral multiplied by dx."""

    dx = 2.0 / n

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce exact sum == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # try uniform redistribution a few times
    for _ in range(5):
        add = diff / x.size
        x += add
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            return x
    # finish with a greedy scan
    if diff > 0:
        for i in range(x.size):
            if diff <= 0:

            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
    elif diff < 0:
        for i in range(x.size):
            if diff >= 0:

            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec

            best_c5 = candidate_best_c5

    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Choose discretization size
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.size
        n = max(200, n0)
        if n != n0:
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x0, base)
        else:
            h0 = base.copy()
    else:
        n = 200
        h0 = rng.random(n)

    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    total_budget = budget_s
    stage1_end = start + total_budget * 0.8

    # Stage 1: continuous optimization
    best_h, best_c5 = optimize_continuous(h0, target, stage1_end, rng)

    # Stage 2: binary local search with remaining time
    final_end = start + total_budget
    half = int(target)
    idx = np.argpartition(best_h, -half)[-half:]
    binary_h = np.zeros_like(best_h)
    binary_h[idx] = 1.0
    binary_h = project_sum_and_bounds(binary_h, target)

    best_bin_h, best_bin_c5 = binary_local_search(binary_h, target, final_end, rng)

    if best_bin_c5 < best_c5:
        best_h, best_c5 = best_bin_h, best_bin_c5

    # Final safety projection and exact evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
