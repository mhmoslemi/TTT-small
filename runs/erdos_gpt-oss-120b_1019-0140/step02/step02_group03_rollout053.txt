import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, shift, n):
        self.shift = shift
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.shift
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_refine(h):
    n = h.shape[0]
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    res = minimize(lambda x: x[-1],
                   x0=x0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': 500, 'ftol':1e-9, 'disp': False})
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, n / 2.0)
        return h_opt, compute_c5(h_opt)
    else:
        return h, t0

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the box [lo,hi]^n while enforcing sum(v)=target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[: self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, shift, n):
        self.shift = shift
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[: self.n]
        k = self.shift
        if k >= 0:
            overlap = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overlap * self.dx

    rng = np.random.default_rng(seed)
    start = time.time()

    # Initialise
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)

    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ---- Phase 0: quick SLSQP restarts for a better start ----
    init_deadline = start + budget_s * 0.15
    while time.time() < init_deadline:
        h_rand = rng.random(n)
        h_rand = project_to_box_sum(h_rand, target_sum)
        h_opt, c5_opt = slsqp_refine(h_rand)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            cur_h, cur_c5 = best_h.copy(), best_c5
        if best_c5 < 0.3808:
            break

    # ---- Phase 1: simulated‑annealing / pairwise hill‑climb ----
    max_step_start = 0.5   # initial step magnitude
    min_step_end = 0.001   # final step magnitude
    temp_start = 0.01      # initial temperature
    temp_end = 0.0001      # final temperature
    global_prob = 0.01     # probability to perform a global perturbation
    no_improve = 0

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s - 2.0:
            break
        frac = elapsed / budget_s
        step = max_step_start * (1 - frac) + min_step_end * frac
        temp = temp_start * (1 - frac) + temp_end * frac

        # --- generate candidate ---
        if rng.random() < global_prob:
            cand = cur_h + rng.normal(scale=step, size=n)
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_to_box_sum(cand, target_sum)
        else:
            i, j = rng.choice(n, size=2, replace=False)
            hi, hj = cur_h[i], cur_h[j]
            lower = max(-hi, hj - 1.0)   # keep within [0,1] after move
            upper = min(1.0 - hi, hj)
            lower = max(lower, -step)
            upper = min(upper, step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            cand = cur_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta

        cand_c5 = compute_c5(cand)

        # --- acceptance test ---
        if cand_c5 < cur_c5 or rng.random() < np.exp(-(cand_c5 - cur_c5) / max(temp, 1e-12)):
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cand.copy(), cand_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # --- restart if stuck ---
        if no_improve >= 20000:
            if rng.random() < 0.5:
                cur_h, cur_c5 = best_h.copy(), best_c5
            else:
                cur_h = rng.random(n)
                cur_h = project_to_box_sum(cur_h, target_sum)
                cur_c5 = compute_c5(cur_h)
            no_improve = 0

        if best_c5 < 0.3808:
            break

    # ---- Phase 2: final SLSQP refinement (time permitting) ----
    if time.time() - start < budget_s - 5.0:
        refined_h, refined_c5 = slsqp_refine(best_h)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
