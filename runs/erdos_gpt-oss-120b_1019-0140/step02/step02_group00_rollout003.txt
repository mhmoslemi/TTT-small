def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.shape[0]
    # If s <= lo*n or s >= hi*n, return constants
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    # Find initial bounds for lambda such that sum clip(v - lam) = s
    # lam lower bound: min(v) - hi ensures all v - lam >= hi?
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Adjust bounds if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

        init_full = upsample(init_full, n_small) if init_full.shape[0] != n_small else init_full.copy()
    else:
        init_full = random_binary_balanced(n_small, rng)
    # Convert to binary top half
    # we will set first half of sum to ones based on largest values
    half = n_small // 2
    idx = np.argpartition(init_full, -half)[-half:]
    h_bin = np.zeros(n_small, dtype=float)
    h_bin[idx] = 1.0
    # ensure sum target exact
    h_bin = project_onto_box_sum(h_bin, n_small/2.0)
    # initial best
    best_h = h_bin.copy()

    # Simulated annealing on binary at n_small
    current_h = best_h.copy()
    current_score = best_score
    ones_set = set(np.flatnonzero(current_h).tolist())
    zeros_set = set(np.where(current_h == 0)[0].tolist())
    # parameters for SA
    T_start = 5e-4  # maybe 0.0005
    T_end = 1e-7
    # Loop
    while time_left() > 0.5:
        elapsed = time.time() - start
        fraction = elapsed / budget_s
        T = T_start * (T_end / T_start) ** fraction
        # sample swap
        i1 = rng.choice(list(ones_set))
        i0 = rng.choice(list(zeros_set))
        # perform swap
        current_h[i1] = 0.0
        current_h[i0] = 1.0
        new_score = compute_c5(current_h)
        delta = new_score - current_score
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            # update sets
            ones_set.remove(i1); ones_set.add(i0)
            zeros_set.remove(i0); zeros_set.add(i1)
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = current_h.copy()
        else:
            # revert swap
            current_h[i1] = 1.0
            current_h[i0] = 0.0
        # optionally break early if best_score <= 0.3808 (target)
        if best_score <= 0.3808:
            break

    # Phase: Upsample to n_large if possible
    if n_large > n_small and time_left() > 1.0:
        h_up = upsample(best_h, n_large)
        # project to box sum
        h_up = project_onto_box_sum(h_up, n_large/2.0)
        sc_up = compute_c5(h_up)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up.copy()
        # Now continuous gradient descent refinement on h_up
        # Determine iteration budget based on remaining time
        remaining = time_left()
        # Rough estimate: each iteration ~ 0.001s for n_large? Let's be conservative.
        est_iters = int(max(1000, min(50000, remaining * 400)))  # approx 0.0025s per iter maybe
        if est_iters > 0:
            refined_h, refined_score = gradient_descent_momentum(best_h, n_iter=est_iters,
                                                                step0=0.5, beta=0.95)
            # project final
            refined_h = project_onto_box_sum(refined_h, n_large/2.0)
            refined_score = compute_c5(refined_h)
            if refined_score < best_score:
                best_score = refined_score
                best_h = refined_h.copy()
    # Final projection for safety
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score
