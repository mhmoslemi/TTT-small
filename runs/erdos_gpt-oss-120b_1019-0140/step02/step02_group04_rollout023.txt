    # using lambda search.

    # adjust any small difference via distributing to non-bound elements

import time
import numpy as np

def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given discretized h."""
    # correlation of h with 1-h, full mode
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_to_capped_sum(v, target):
    """
    Euclidean projection of v onto the set {w : 0 <= w_i <= 1, sum(w) = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - hi, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        free = (w > 1e-12) & (w < 1.0 - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w


def generate_random_feasible(n, target, rng):
    """Random vector in [0,1]^n with exact sum = target via projection."""
    v = rng.random(n)
    return project_to_capped_sum(v, target)
