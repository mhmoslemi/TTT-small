def project_to_sum_continuous(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    diff = target - cur
    if abs(diff) < 1e-12:
        return v
    # adjust by adding diff uniformly then clip again then fix
    v += diff / len(v)
    v = np.clip(v, 0.0, 1.0)
    # final correction via iterative pass
    diff = target - np.sum(v)
    # Now adjust greedily.
    for i in range(len(v)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and v[i] < 1.0:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        elif diff < 0 and v[i] > 0.0:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    return v

def binary_sa_one_run(h_bin, dx, time_limit, rng):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_val = compute_c5(cur.astype(float), dx)

    best_val = cur_val

    # exponential cooling schedule parameters
    T0 = 0.02
    decay = 0.9999

    iter_count = 0
    while time.time() < time_limit:
        iter_count += 1
        # temperature
        T = T0 * (decay ** iter_count)
        # choose random swap
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cand = cur.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand.astype(float), dx)
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val:
                best = cur.copy()
                best_val = cur_val
    return best, best_val

def generate_random_feasible(n, target_sum, rng):
    x = rng.random(n)
    x = np.clip(x, 0.0, 1.0)
    # scale to sum
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    # clip again
    x = np.clip(x, 0.0, 1.0)
    # fix residual sum
    diff = target_sum - np.sum(x)
    # adjust up or down
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

import time
import numpy as np

TARGET = 0.3808  # target bound for early termination


def compute_c5(h, dx):
    """Maximum overlap integral C5 for a step function h (vector)."""
    n = h.shape[0]
    N = 2 * n - 1
    w_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=N)
    W = np.fft.rfft(w_rev, n=N)
    conv = np.fft.irfft(H * W, n=N)
    return np.max(conv) * dx


def project_to_sum_continuous(vec, target_sum):
    """Clip vec to [0,1] and adjust so that sum(vec) == target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    cur = np.sum(v)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return v
    v += diff / len(v)
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - np.sum(v)
    n = len(v)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and v[i] < 1.0:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        elif diff < 0 and v[i] > 0.0:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    return v


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible vector (values in [0,1] with exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def upscale_h(old_h, new_n):
    """Linearly interpolate old_h onto a grid of size new_n."""
    old_n = old_h.shape[0]
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, old_h)


def simulated_annealing_continuous(h0, dx, target_sum, time_limit, rng):
    """SA on continuous vectors preserving the sum constraint."""
    cur = project_to_sum_continuous(np.clip(h0, 0.0, 1.0), target_sum)
    cur_val = compute_c5(cur, dx)
    best = cur.copy()
    best_val = cur_val

    step_max = 0.1
    T0 = 0.01
    decay = 0.9999
    it = 0
    while time.time() < time_limit:
        it += 1
        T = T0 * (decay ** it)
        donors = np.where(cur > 1e-12)[0]
        recipients = np.where(cur < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step_max, cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5(cand, dx)
        dval = cand_val - cur_val
        if dval < 0 or rng.random() < np.exp(-dval / (T + 1e-12)):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val:
                best = cur.copy()
                best_val = cur_val
                if best_val <= TARGET:
                    break
    best = project_to_sum_continuous(best, target_sum)
    best_val = compute_c5(best, dx)
    return best, best_val


def greedy_refine(h0, dx, target_sum, time_limit, rng):
    """Simple greedy refinement (mass transfer) with adaptive step size."""
    cur = h0.copy()
    cur_val = compute_c5(cur, dx)
    best = cur.copy()
    best_val = cur_val
    step = 0.05
    while time.time() < time_limit:
        donors = np.where(cur > 1e-12)[0]
        recipients = np.where(cur < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break

        j = rng.choice(recipients)
        max_delta = min(step, cur[i], 1.0 - cur[j])
        if max_delta <= 0:

            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5(cand, dx)
        if cand_val < cur_val:

    best_h = project_to_sum_continuous(np.clip(best_h, 0.0, 1.0), target_sum)
