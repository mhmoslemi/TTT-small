def project_to_feasible(v, target):
    v = np.clip(v, 0.0, 1.0)
    sum_v = v.sum()
    # adjust using waterfilling
    # while not close:
    for _ in range(10):
        diff = target - sum_v
        if abs(diff) < 1e-12:
            break
        # compute indices that can be adjusted
        if diff > 0:
            idx = np.where(v < 1.0)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            v[idx] += inc
            # clip again
            v = np.clip(v, 0.0, 1.0)
        else:
            idx = np.where(v > 0.0)[0]
            if idx.size == 0:
                break
            dec = diff / idx.size  # diff negative
            v[idx] += dec  # dec negative
            v = np.clip(v, 0.0, 1.0)
        sum_v = v.sum()
    return v
```

def random_start_vector(n, rng, target):
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    # attempt simple fill
    if abs(diff) > 1e-12:
        inc = diff / n
        v = np.clip(v + inc, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
    if abs(diff) > 1e-12:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and v[i] < 1.0:
                inc = min(diff, 1.0 - v[i])
                v[i] += inc
                diff -= inc
            elif diff < 0 and v[i] > 0.0:
                dec = min(-diff, v[i])
                v[i] -= dec
                diff += dec

def build_slsqp_constraints(n, dx, target):
    def sum_con(x):
        return np.sum(x[:n]) - target
    constraints = [{'type':'eq','fun':sum_con}]
    for k in range(-n+1, n):
        if k == 0:
            continue
        def con_factory(k):
            def con(x, k=k):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[: n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[: n - kk])
                return x[-1] - overl * dx
            return con
        constraints.append({'type':'ineq','fun':con_factory(k)})
    return constraints

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2] discretized."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_feasible(v, target):
    """Clip to [0,1] and adjust to hit exact sum target (water-filling)."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v < 1.0)[0]
        else:
            idx = np.where(v > 0.0)[0]
        if idx.size == 0:
            break
        inc = diff / idx.size
        v[idx] += inc
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    return v

def random_start_vector(n, rng, target):
    """Random feasible vector with entries in [0,1] and sum == target."""
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        inc = diff / n
        v = np.clip(v + inc, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
    if abs(diff) > 1e-12:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and v[i] < 1.0:
                d = min(diff, 1.0 - v[i])
                v[i] += d
                diff -= d
            elif diff < 0 and v[i] > 0.0:
                d = min(-diff, v[i])
                v[i] -= d
                diff += d
    return v

def gradient_descent(h0, target, dx, deadline):
    """Subgradient descent on max‑overlap with projection onto feasible set."""
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h, best_val = h.copy(), cur_val
    η = 0.5
    while time.time() < deadline - 0.1:
        n = h.shape[0]
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = np.argmax(corr) - (n - 1)
        grad = np.zeros_like(h)
        if k >= 0:
            grad[: n - k] = (1.0 - h[k:]) * dx
            grad[k:] -= h[: n - k] * dx
        else:
            kk = -k
            grad[kk:] = (1.0 - h[: n - kk]) * dx
            grad[: n - kk] -= h[kk:] * dx
        h_new = h - η * grad
        h_new = project_to_feasible(h_new, target)
        new_val = compute_c5(h_new)
        if new_val < best_val:
            best_val, best_h = new_val, h_new.copy()
        if new_val < cur_val:
            h, cur_val = h_new, new_val
            η = min(η * 1.1, 1.0)
        else:
            η = η * 0.7
        if best_val < 0.3808:
            break
    return best_h, best_val

def build_slsqp_constraints(n, dx, target):
    """Linear equality for sum and inequality t >= overlap_k for all shifts."""
    def sum_con(x):
        return np.sum(x[:n]) - target
    cons = [{"type": "eq", "fun": sum_con}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        def con_factory(k):
            def con(x, k=k):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[: n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[: n - kk])
                return x[-1] - overl * dx
            return con
        cons.append({"type": "ineq", "fun": con_factory(k)})
    return cons

def local_optimize(x0, n, dx, target, bounds, constraints):
    """One SLSQP run to minimize the auxiliary variable t."""
    try:
        res = minimize(lambda x: x[-1], x0, method="SLSQP",
                       bounds=bounds, constraints=constraints,
                       options={"maxiter": 2000, "ftol": 1e-9, "disp": False})
    except Exception:
        return None
    if not res.success:
        return None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = h_opt.sum()
    if abs(cur - target) > 1e-8:
        h_opt += (target - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    t_opt = compute_c5(h_opt)
    return np.concatenate([h_opt, [t_opt]])

    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ----- problem size -----
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n = h_init.shape[0]
    else:

        h_init = None
    target = n / 2.0
    dx = 2.0 / n

    if h_init is not None:
        h0 = project_to_feasible(h_init, target)
    else:

    # ----- constraints and bounds for SLSQP -----
    constraints = build_slsqp_constraints(n, dx, target)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_val = np.inf

    max_trials = 40  # number of independent starts
    for trial in range(max_trials):
        if time.time() >= deadline - 2.0:
            break
        # generate start
        if trial == 0 and h_init is not None:
            h_start = h0.copy()
        else:
            h_start = random_start_vector(n, rng, target)

        # subgradient refinement
        h_gd, val_gd = gradient_descent(h_start, target, dx, deadline)

        # one SLSQP refinement
        x0 = np.concatenate([h_gd, [val_gd]])
        x_opt = local_optimize(x0, n, dx, target, bounds, constraints)
        if x_opt is not None:
            h_cur = x_opt[:n]
            val_cur = x_opt[-1]
        else:
            h_cur = h_gd
            val_cur = val_gd

        if val_cur < best_val:
            best_val = val_cur
            best_h = h_cur.copy()
            if best_val < 0.3808:
                break

    # ----- final perturbation phase (simple hill‑climbing) -----
    if best_h is not None and time.time() < deadline - 1.0:
        h_cur = best_h.copy()
        cur_val = best_val
        while time.time() < deadline - 0.5 and cur_val > 0.3807:
            sigma = 0.03
            h_new = h_cur + sigma * rng.standard_normal(n)
            h_new = project_to_feasible(h_new, target)
            val_new = compute_c5(h_new)
            if val_new < cur_val:
                h_cur, cur_val = h_new, val_new
                if cur_val < best_val:
                    best_val, best_h = cur_val, h_cur.copy()
                    if best_val < 0.3808:
                        break

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target) > 1e-8:
        best_h += (target - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
