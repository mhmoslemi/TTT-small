def anneal_optimize(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    n = len(h0)
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    # parameters
    T0 = 0.01
    step_max = 0.5
    step_min = 0.001
    iter_without_improve = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1 - frac)
        step_factor = step_max * (1 - frac) + step_min * frac
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            # increase i, decrease j
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0:
                continue
            delta = (rng.random() * 2 - 1) * step_factor * max_delta
            h_new_i = h[i] + delta
            h_new_j = h[j] - delta
        else:
            # increase j, decrease i
            max_delta = min(1.0 - h[j], h[i])
            if max_delta <= 0:
                continue
            delta = (rng.random() * 2 - 1) * step_factor * max_delta
            h_new_i = h[i] - delta
            h_new_j = h[j] + delta
        # apply
        h_candidate = h.copy()
        h_candidate[i] = h_new_i
        h_candidate[j] = h_new_j
        # compute
        new_c5 = compute_c5(h_candidate)
        delta_c5 = new_c5 - cur_c5
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / (T + 1e-12)):
            h = h_candidate
            cur_c5 = new_c5
            iter_without_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            iter_without_improve += 1
        # optional random shake if stuck
        if iter_without_improve > 20000:
            # large shake: random small Gaussian
            h = best_h.copy()
            h += rng.normal(scale=0.02, size=n)
            # clip and project
            h = np.clip(h, 0.0, 1.0)
            h = project_sum_and_bounds(h, n/2.0)
            cur_c5 = compute_c5(h)
            iter_without_improve = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 100
        h0 = rng.random(n)
        h0 = project_sum_and_bounds(h0, n/2.0)
    # ensure feasible
    h0 = project_sum_and_bounds(h0, n/2.0)

    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)

    # allocate part of time to annealing (e.g., 60%)
    t_anneal = remaining * 0.6
    best_h, best_c5 = anneal_optimize(h0, t_anneal, seed+1)

    # remaining time for gradient descent
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    if remaining > 5.0:
        # run gradient descent refinement
        refined_h, refined_c5 = optimize_h(best_h, remaining, seed+2)
        best_h, best_c5 = refined_h, refined_c5

    # final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, n/2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (scaled by dx)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and adjust to satisfy sum(x)==target (exact)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # First uniform adjustment
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # Greedy fill remaining difference
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def anneal_optimize(h0, time_budget, seed):
    """Simple simulated annealing with pairwise delta moves and swaps."""
    rng = np.random.default_rng(seed)
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    swap_prob = 0.1
    iter_no_improve = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = 0.01 * (1.0 - frac) + 1e-12
        step_factor = 0.5 * (1.0 - frac) + 0.001 * frac
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < swap_prob:
            # propose a swap
            if h[i] == h[j]:
                continue
            h_candidate = h.copy()
            h_candidate[i], h_candidate[j] = h_candidate[j], h_candidate[i]
        else:
            # propose a delta transfer between i and j
            increase_i = rng.random() < 0.5
            if increase_i:
                max_delta = min(1.0 - h[i], h[j])
                if max_delta <= 0:
                    continue
                delta = (rng.random() * 2.0 - 1.0) * step_factor * max_delta
                h_candidate = h.copy()
                h_candidate[i] = h[i] + delta
                h_candidate[j] = h[j] - delta
            else:
                max_delta = min(1.0 - h[j], h[i])
                if max_delta <= 0:
                    continue
                delta = (rng.random() * 2.0 - 1.0) * step_factor * max_delta
                h_candidate = h.copy()
                h_candidate[i] = h[i] - delta
                h_candidate[j] = h[j] + delta
        # evaluate
        new_c5 = compute_c5(h_candidate)
        delta_c5 = new_c5 - cur_c5
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
            h = h_candidate
            cur_c5 = new_c5
            iter_no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            iter_no_improve += 1
        if iter_no_improve > 20000:
            # large random shake around current best
            h = best_h.copy()
            h += rng.normal(scale=0.02, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_sum_and_bounds(h, target)
            cur_c5 = compute_c5(h)
            iter_no_improve = 0
    return best_h, best_c5

def optimize_h(h0, time_budget, seed):
    """Gradient descent on a smooth approximation of the max-overlap."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 5.0          # initial smoothing
    max_beta = 50.0
    alpha0 = 0.2        # base step size
    no_improve = 0

    while time.time() - start < time_budget:
        # correlation (raw, without dx)
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase smoothing slowly
        if beta < max_beta:
            beta = min(max_beta, beta * 1.002)

        # soft‑max weights (stable)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth objective
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # enforce zero‑sum direction (preserve total mass)
        g -= np.mean(g)

        # step‑size schedule (linear decay)
        elapsed = time.time() - start
        alpha = alpha0 * max(0.05, 1.0 - elapsed / time_budget)

        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        # accept if we improve or with tiny probability
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        # random perturbation when stuck
        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    return best_h, best_c5

def generate_triangle(n):
    """Symmetric triangular step function on [0,2] with peak 1 at centre."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.where(xs <= 1.0, xs, 2.0 - xs)
    h = project_sum_and_bounds(h, n / 2.0)
    return h

def generate_cosine(n):
    """Half‑cosine bump on [0,2] with integral 1."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = 0.5 * (1.0 + np.cos(np.pi * (xs - 1.0)))
    h = project_sum_and_bounds(h, n / 2.0)
    return h
