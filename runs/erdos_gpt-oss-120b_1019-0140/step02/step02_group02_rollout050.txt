def anneal(h, best_val, start_time, deadline, rng):
    cur_h = h.copy()
    cur_val = best_val
    best_h = cur_h.copy()
    best = best_val
    n = len(h)
    while time.time() < deadline - 0.2:
        # compute correlation and index of max
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # Get contributions for that shift:
        if k >= 0:
            idx_i = np.arange(n - k)
            idx_j = idx_i + k
        else:
            kk = -k
            idx_i = np.arange(kk, n)
            idx_j = idx_i - kk
        prod = cur_h[idx_i] * (1 - cur_h[idx_j])
        # choose i that contributes most
        i_local = np.argmax(prod)
        i = idx_i[i_local]
        j_candidates = np.where(cur_h < 1 - 1e-12)[0]
        if len(j_candidates) == 0 or cur_h[i] <= 1e-12:
            # fallback random src/dst
            src_idxs = np.where(cur_h > 1e-12)[0]
            dst_idxs = np.where(cur_h < 1 - e-12)[0]
            if len(src_idxs) == 0 or len(dst_idxs) == 0:
                break
            i = rng.choice(src_idxs)
            j = rng.choice(dst_idxs)
        else:
            j = rng.choice(j_candidates)
        # move mass
        max_transfer = min(cur_h[i], 1 - cur_h[j])

import time
import numpy as np

def compute_c5(h):
    """Return max overlap integral approximated via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def anneal(h_start, best_val, start_time, deadline, rng):
    """Simple simulated annealing on continuous h keeping total mass fixed."""
    h = h_start.copy()
    cur_val = best_val
    best_h = h.copy()
    best = best_val
    n = h.shape[0]
    total_time = deadline - start_time
    while time.time() < deadline - 0.2:
        # find shift that currently attains the max overlap
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        if k >= 0:
            i_idx = np.arange(n - k)
            j_idx = i_idx + k
        else:
            kk = -k
            i_idx = np.arange(kk, n)
            j_idx = i_idx - kk

        prod = h[i_idx] * (1.0 - h[j_idx])
        i_local = np.argmax(prod)
        i = i_idx[i_local]

        # choose a destination that can receive mass
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta

        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            # simulated-annealing acceptance
            elapsed = time.time() - start_time
            remaining = max(0.0, deadline - start_time - elapsed)
            T = max(remaining / total_time * 0.01, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta

        if best < 0.3808:
            break
    return best_h, best

                            h_sol[i] = 0.0
                            diff += 1.0
                val = compute_c5(h_sol)
                if val < best_val:
                    best_val = val
                    best_h = h_sol.copy()
        except Exception:
            pass  # any issue, just continue with heuristic

    # ----- continuous annealing refinement ---------------------------------
    if time.time() < deadline - 0.5:
        best_h, best_val = anneal(best_h, best_val, start, deadline, rng)

    # ----- final safety clip & sum enforcement ------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-9:
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            allowed = (1.0 - best_h[i]) if diff > 0 else best_h[i]
            delta = np.clip(diff, -allowed, allowed)
            best_h[i] += delta
            diff -= delta

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
