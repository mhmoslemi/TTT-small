import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_sum_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        s = np.sum(x)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
    return x

def repair(x, target):
    return project_sum_bounds(x, target)

def random_binary_vector(n, rng):
    arr = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    arr[idx] = 1.0
    return arr

def random_continuous_vector(n, rng, target):
    x = rng.random(n)
    return repair(x, target)

def continuous_anneal(start_h, time_end, rng):
    n = len(start_h)
    target = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 1e-2
    final_temp = 1e-8
    max_iter = 200000
    for it in range(max_iter):
        if time.time() >= time_end:
            break
        # select inc/dec indices
        inc_candidates = np.where(h < 1 - 1e-12)[0]
        dec_candidates = np.where(h > 1e-12)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            break
        i = rng.choice(inc_candidates)
        j = rng.choice(dec_candidates)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()

import time
import math
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated by full correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_sum_bounds(x, target):
    """Project onto the box [0,1] and enforce sum == target by uniform adjustments."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        s = np.sum(x)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
    return x

def repair(x, target):
    return project_sum_bounds(x, target)

def random_binary_vector(n, rng):
    v = np.zeros(n, dtype=float)
    v[rng.choice(n, n // 2, replace=False)] = 1.0
    return v

def random_continuous_vector(n, rng, target):
    v = rng.random(n)
    return repair(v, target)

def rudin_shapiro_seq(n):
    """Binary Rudin‑Shapiro sequence (+1/−1 → 1/0) of length n."""
    a = np.array([1], dtype=int)
    while len(a) < n:
        b = np.empty(2 * len(a), dtype=int)
        b[0::2] = a
        # (-1)^k factor
        signs = ((np.arange(len(a)) % 2) * 2 - 1)  # = 1 for even k, -1 for odd k
        b[1::2] = a * signs
        a = b
    a = a[:n]
    return (a + 1) // 2

        j = rng.choice(dec_idx)

        if max_delta <= 0:
