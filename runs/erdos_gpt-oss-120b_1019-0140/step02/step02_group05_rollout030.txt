def refine_with_slsqp(h_initial, c5_initial):
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0
    # compute O_k
    corr = np.correlate(h_initial, 1.0 - h_initial, mode='full')
    O = corr * dx
    # threshold for relevant constraints
    threshold = c5_initial - 1e-4
    ks = [i - (n-1) for i, val in enumerate(O) if val >= threshold and i != (n-1)]
    # If empty, just skip.
    if not ks:
        return h_initial, c5_initial

    # Define objective: minimize t (the last variable)
    # Variables: h (size n) + t (scalar)
    def objective(x):
        return x[-1]

    # Equality: sum h = target_sum
    def eq_sum(x):
        return np.sum(x[:n]) - target_sum

    # Inequality: for each k, t - O_k(h) >= 0 => t >= O_k
    def make_ineq(k):
        def fun(x):
            h = x[:n]
            # compute overlap for shift k
            if k >= 0:
                overlap = np.dot(h[:n - k], 1.0 - h[k:]) * dx
            else:
                kk = -k
                overlap = np.dot(h[kk:], 1.0 - h[:n - kk]) * dx
            return x[-1] - overlap
        return fun

    constraints = [{'type': 'eq', 'fun': eq_sum}]
    for k in ks:
        constraints.append({'type': 'ineq', 'fun': make_ineq(k)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    x0 = np.concatenate([h_initial, [c5_initial]])

    res = minimize(fun=objective, x0=x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': 500, 'ftol':1e-9, 'disp':False})

        if abs(cur - target_sum) > 1e-9:

            h_opt = project_box_sum(h_opt, target_sum)
        val_opt = compute_c5(h_opt)
        return h_opt, val_opt

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_box_sum(v, target):
    """Euclidean projection of v onto {0<=x_i<=1, sum x = target}."""
    # Clip first, then adjust with a uniform shift (water-filling via bisection)
    w = np.clip(v, 0.0, 1.0)
    s = w.sum()
    if abs(s - target) <= 1e-12:
        return w
    # find bounds for lambda
    lo, hi = -1.0, 1.0
    while np.clip(v + hi, 0.0, 1.0).sum() < target:
        hi *= 2.0
    while np.clip(v + lo, 0.0, 1.0).sum() > target:
        lo *= 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        wtmp = np.clip(v + lam, 0.0, 1.0)
        s = wtmp.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    sm = np.clip(sm, 0.0, 1.0)

        return x[-1]

    def eq_sum(x):
        return np.sum(x[:n]) - target

    def make_ineq(k):
        def fun(x):
            h = x[:n]
            if k >= 0:
                ov = np.dot(h[: n - k], 1.0 - h[k:]) * dx
            else:
                kk = -k
                ov = np.dot(h[kk:], 1.0 - h[: n - kk]) * dx
            return x[-1] - ov
        return fun

    cons = [{"type": "eq", "fun": eq_sum}]
    for k in ks:
        cons.append({"type": "ineq", "fun": make_ineq(k)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h0, [c5_0]])

    try:
        res = minimize(
            fun=obj,
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 500, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_box_sum(h_opt, target)
            c_opt = compute_c5(h_opt)
            return h_opt, c_opt
    except Exception:
        pass
    return h0, c5_0


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # fixed resolution (can be altered if time permits)
    n = 80
    dx = 2.0 / n
    target_sum = n / 2.0

    # ----- generate a small pool of feasible starts ----------------
    init_pool = []

    if "initial_h_values" in globals():
        init_pool.append(np.array(initial_h_values, dtype=float))

    init_pool.append(np.full(n, 0.5))

    # random binary masks
    for _ in range(3):
        a = np.zeros(n, dtype=float)
        a[rng.choice(n, n // 2, replace=False)] = 1.0
        init_pool.append(a)

    # random continuous feasible vectors
    for _ in range(2):
        a = rng.random(n)
        a = project_box_sum(a, target_sum)
        init_pool.append(a)

    # ----- bookkeeping ------------------------------------------------
    best_h = None
    best_val = float("inf")
    cur_h = None
    cur_val = None
    it = 0
    stagnate = 0
    max_stagnate = 800  # restart after many non‑improving steps

    # ----- main iterative search --------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.98:
            break

        # restart if needed
        if cur_h is None or stagnate >= max_stagnate:
            cur_h = init_pool[rng.integers(len(init_pool))].copy()
            cur_h = project_box_sum(cur_h, target_sum)
            cur_val = compute_c5(cur_h)
            stagnate = 0
            continue

        # compute overlaps and objective
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        O = corr * dx
        cur_val = O.max()

        # track best
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()
            stagnate = 0
            if best_val < 0.3808:
                break

        # ----- soft‑max weighting for gradient -----------------------
        beta = min(30.0, 0.02 * (it + 1) + 1.0)  # start ~1, grow to ~30
        w = np.exp(beta * O)
        w /= w.sum()  # normalized weights

        # ----- gradient of smoothed max -------------------------------
        G = np.full(n, dx)  # base term from sum(w)=1
        for idx, wk in enumerate(w):
            if wk == 0.0:
                continue
            shift = idx - (n - 1)
            if shift >= 0:
                G[: n - shift] -= wk * dx * cur_h[shift:]
                G[shift:] -= wk * dx * cur_h[: n - shift]
            else:
                l = -shift
                G[l:] -= wk * dx * cur_h[: n - l]
                G[: n - l] -= wk * dx * cur_h[l:]

        # ----- step with backtracking ---------------------------------
        step = 0.1 / math.sqrt(it + 1)
        accepted = False
        for _ in range(6):
            cand = cur_h - step * G
            cand = project_box_sum(cand, target_sum)
            new_val = compute_c5(cand)
            if new_val <= cur_val:
                cur_h, cur_val = cand, new_val
                accepted = True
                stagnate = 0
                break
            step *= 0.5
        if not accepted:
            stagnate += 1

        it += 1

    # ----- optional smoothing passes ----------------------------------
    while time.time() - start < budget_s - 7.0:
        sm = smooth_and_fix(best_h, target_sum)
        sm_val = compute_c5(sm)
        if sm_val + 1e-12 < best_val:
            best_val, best_h = sm_val, sm
        else:
            break

    # ----- final SLSQP refinement ------------------------------------
    if time.time() - start < budget_s - 3.0:
        best_h, best_val = refine_with_slsqp(best_h, best_val)

    # final safety projection
    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_val = compute_c5(best_h)

    return best_h, best_val, n
