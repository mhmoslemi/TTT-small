        # a_{2k+1} = (-1)^k * a_k

    a = a[:n]

import time
import math
import numpy as np

def compute_c5(arr):
    """Return the approximate C5 value for a step function."""
    n = arr.shape[0]
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Binary Thue‑Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=int, count=n)

def rudin_shapiro_seq(n):
    """Rudin‑Shapiro binary sequence (+1/‑1 → 1/0) of length n."""
    a = np.array([1], dtype=int)
    while a.shape[0] < n:
        k = np.arange(a.shape[0])
        even = a
        odd = a * ((-1) ** k)
        a = np.concatenate([even, odd])
    a = a[:n]
    return ((a + 1) // 2).astype(int)

def binary_simulated_anneal(start_bin, time_end, rng):
    """Binary simulated annealing with swap moves, preserving the exact sum."""
    n = start_bin.shape[0]
    cur_bin = start_bin.copy()
    cur_h = cur_bin.astype(float)
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    init_temp = 1e-2
    final_temp = 1e-8
    start_time = time.time()

    while time.time() < time_end:
        elapsed = time.time() - start_time
        frac = elapsed / max(time_end - start_time, 1e-12)
        T = init_temp * (1 - frac) + final_temp * frac

        ones = np.flatnonzero(cur_bin)
        zeros = np.flatnonzero(1 - cur_bin)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur_bin[i] = 0
        cur_bin[j] = 1
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)
        dE = new_c5 - cur_c5

        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert
            cur_bin[i] = 1
            cur_bin[j] = 0
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    return best_h, best_c5

def continuous_anneal(start_h, time_end, rng):
    """Continuous annealing on values in [0,1] preserving sum."""
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 200000
    it = 0
    while time.time() < time_end and it < max_iter:
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1

        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()

        h_new[j] -= delta

            candidates.insert(0, n0)

    for n in candidates:
        if n % 2:
            continue
        target_sum = n / 2.0

        # build seed pool
        seeds = []

        # random balanced binary
        rand_bin = np.zeros(n, dtype=int)
        rand_bin[rng.choice(n, n // 2, replace=False)] = 1
        seeds.append(rand_bin.copy())

        # Thue‑Morse (adjust sum)
        tm = thue_morse_seq(n).copy()
        diff = target_sum - np.sum(tm)
        if diff != 0:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for idx in idxs:
                if diff == 0:
                    break
                if diff > 0 and tm[idx] == 0:
                    tm[idx] = 1
                    diff -= 1
                elif diff < 0 and tm[idx] == 1:
                    tm[idx] = 0
                    diff += 1
        seeds.append(tm.copy())

        # Rudin‑Shapiro (adjust sum)
        rs = rudin_shapiro_seq(n).copy()
        diff = target_sum - np.sum(rs)
        if diff != 0:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for idx in idxs:
                if diff == 0:
                    break
                if diff > 0 and rs[idx] == 0:
                    rs[idx] = 1
                    diff -= 1
                elif diff < 0 and rs[idx] == 1:
                    rs[idx] = 0
                    diff += 1
        seeds.append(rs.copy())

        # use provided initial construct if compatible
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            ini = np.array(initial_h_values, dtype=float)
            bin_ini = (ini >= 0.5).astype(int)
            cur_sum = np.sum(bin_ini)
            if cur_sum != target_sum:
                diff = target_sum - cur_sum
                idxs = np.arange(n)
                rng.shuffle(idxs)
                for idx in idxs:
                    if diff == 0:
                        break
                    if diff > 0 and bin_ini[idx] == 0:
                        bin_ini[idx] = 1
                        diff -= 1
                    elif diff < 0 and bin_ini[idx] == 1:
                        bin_ini[idx] = 0
                        diff += 1
            seeds.append(bin_ini.copy())

        # search loop for this n
        while time.time() - start_time < budget_s - 0.5:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 1.0:
                break
            chunk = min(5.0, remaining * 0.3)
            end_chunk = time.time() + chunk

            start_bin = seeds[rng.integers(len(seeds))].copy()
            cand_h, cand_c5 = binary_simulated_anneal(start_bin, end_chunk, rng)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand_h.copy()
                best_n = n
                if best_c5 <= TARGET:
                    break

            # occasional continuous refinement (short)
            if time.time() - start_time < budget_s - 1.0:
                short_end = time.time() + min(2.0, budget_s - (time.time() - start_time) - 0.5)
                start_cont = (best_h >= 0.5).astype(float) if best_h is not None else start_bin.astype(float)
                cont_h, cont_c5 = continuous_anneal(start_cont, short_end, rng)
                if cont_c5 < best_c5:
                    best_c5 = cont_c5
                    best_h = cont_h.copy()
                    best_n = n
                    if best_c5 <= TARGET:
                        break
            if best_c5 <= TARGET:
                break
        if best_c5 <= TARGET:
            break

    # Final refinement with remaining time
    time_left = budget_s - (time.time() - start_time)
    if best_h is not None and time_left > 1.0:
        final_end = time.time() + time_left - 0.2
        final_h, final_c5 = continuous_anneal(best_h.copy(), final_end, rng)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            best_n = len(final_h)

    # enforce exact feasibility
    if best_h is not None:
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        target_sum = n / 2.0
        diff = target_sum - np.sum(best_h)
        if abs(diff) > 1e-12:
            best_h += diff / n
            best_h = np.clip(best_h, 0.0, 1.0)
            residual = target_sum - np.sum(best_h)
            for i in range(n):
                if abs(residual) < 1e-12:
                    break
                adj = np.clip(residual, -best_h[i], 1 - best_h[i])
                best_h[i] += adj
                residual = target_sum - np.sum(best_h)
        best_c5 = compute_c5(best_h)
        return best_h, best_c5, n

    # fallback (should not happen)
    n = 80
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    return h, compute_c5(h), n
