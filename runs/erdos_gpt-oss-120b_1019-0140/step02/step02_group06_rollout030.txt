import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # Uniform adjustment
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def sa_optimize(h0, time_budget, seed):
    """Simulated annealing with pairwise mass swaps."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5

    # schedules
    step_initial, step_min = 0.08, 0.001
    T_initial, T_final = 0.02, 1e-6

    start = time.time()
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        step = step_initial * (1 - frac) + step_min * frac
        T = T_initial * (1 - frac) + T_final * frac

        # choose a pair i (decrease) and j (increase)
        for _ in range(10):
            i = rng.integers(n)
            if h[i] <= 0.0:
                continue
            j = rng.integers(n)
            if h[j] >= 1.0 or i == j:
                continue
            break
        else:
            continue  # failed to find a pair this round

        max_delta = min(step, h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        h[i] -= delta
        h[j] += delta

        new_c5 = compute_c5(h)
        delta_c5 = new_c5 - c5_cur
        if delta_c5 <= 0.0 or rng.random() < np.exp(-delta_c5 / max(T, 1e-12)):
            c5_cur = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert move
            h[i] += delta
            h[j] -= delta

    return best_h, best_c5

def gradient_descent_optimize(h0, time_budget, seed):
    """Short soft‑max gradient descent, starting from h0."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.1

    c5_cur = best_c5
    while time.time() - start < time_budget:
        # correlation vector (raw, without dx)
        v = np.correlate(h, 1.0 - h, mode='full')
        # smooth weights
        vmax = v.max()
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
        w = np.exp(beta * (v - vmax))
        w /= w.sum()

        # gradient of the smoothed objective
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # zero‑sum direction to keep total mass
        g -= np.mean(g)

        # step‑size schedule (linear decay)
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        new_c5 = compute_c5(h_new)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()

        if new_c5 < c5_cur or rng.random() < 0.001:
            h = h_new
            c5_cur = new_c5

        # occasional random shake
        if rng.random() < 0.01:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            c5_cur = compute_c5(h)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # initial construction
    if 'initial_h_values' in globals():

    target = n / 2.0

    # allocate time
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    if remaining <= 0.0:
        final_h = h0
        return final_h, compute_c5(final_h), n

    # 80 % SA, 20 % gradient polishing
    sa_time = remaining * 0.8
    best_h, best_c5 = sa_optimize(h0, sa_time, seed + 1)

    remaining = max(0.0, budget_s - elapsed)
    if remaining > 0.2:
        best_h, best_c5 = gradient_descent_optimize(best_h, remaining, seed + 2)

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
