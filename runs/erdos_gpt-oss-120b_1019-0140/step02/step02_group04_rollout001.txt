import time, math
import numpy as np

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def binary_from_continuous(h_cont, target_sum):
    idx = np.argsort(-h_cont)[:target_sum]
    h_bin = np.zeros_like(h_cont, dtype=float)
    h_bin[idx] = 1.0
    return h_bin

def random_binary(n, target_sum, rng):
    idx = rng.choice(n, size=target_sum, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n // 2
    dx = 2.0 / n

    # Starting binary solution
    if h0 is not None:
        cur = binary_from_continuous(h0, target_sum)
    else:
        cur = random_binary(n, target_sum, rng)

    cur_val = compute_c5(cur, dx)
    best_h = cur.copy()
    best_val = cur_val

    # Precompute ones and zeros lists
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))

    # Parameters
    max_no_improve = 20000
    no_improve = 0
    T_initial = 0.02
    T_min = 1e-5

    while True:
        if time.time() - start > budget_s - 0.2:
            break

        # Choose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # Perform swap
        cur[i] = 0.0
        cur[j] = 1.0

        # Compute new value
        new_val = compute_c5(cur, dx)

        # Temperature schedule
        elapsed = time.time() - start
        T = max(T_min, T_initial * (1.0 - elapsed / budget_s))

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            # update ones/zeros lists
            # i was 1->0, j was 0->1
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)

            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_h = cur.copy()
                # Early exit if great enough
                if best_val < 0.3808:
                    break
        else:
            # revert swap
            cur[i] = 1.0
            cur[j] = 0.0
            no_improve += 1

        if no_improve >= max_no_improve:
            # random restart
            cur = random_binary(n, target_sum, rng)
            cur_val = compute_c5(cur, dx)
            ones = list(np.flatnonzero(cur == 1))
            zeros = list(np.flatnonzero(cur == 0))
            no_improve = 0
            # maybe update best if new random is better (unlikely)
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur.copy()

    # Ensure final best respects constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    # Force exact sum in case of rounding errors
    current_sum = np.sum(best_h)
    diff = target_sum - current_sum
    # diff should be integer (within tolerance). Adjust using smallest changes.
    if abs(diff) > 1e-9:

            best_h[cand[: -diff_int]] = 0.0

    final_val = compute_c5(best_h, dx)

    return best_h, final_val, n

import time
import math
import numpy as np

def _compute_c5(h, dx):
    """Maximum overlap integral for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _binary_from_continuous(h_cont, target_sum):
    """Turn a continuous vector into a balanced binary one."""
    idx = np.argsort(-h_cont)[:target_sum]
    hb = np.zeros_like(h_cont, dtype=float)
    hb[idx] = 1.0
    return hb

def _random_binary(n, target_sum, rng):
    """Generate a random balanced binary vector."""
    idx = rng.choice(n, size=target_sum, replace=False)
    hb = np.zeros(n, dtype=float)
    hb[idx] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low C5 bound.

    Returns
    -------
    h_values : np.ndarray
        The best found step values (float array in [0,1]).
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of discretisation points used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- problem size & initial vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n // 2          # exact number of ones
    dx = 2.0 / n

    # Initialise a binary candidate
    if h0 is not None:
        cur = _binary_from_continuous(h0, target_sum)
    else:
        cur = _random_binary(n, target_sum, rng)

    cur_val = _compute_c5(cur, dx)
    best_h = cur.copy()
    best_val = cur_val

    # Lists of present ones / zeros (for fast sampling)
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))

    # ----- simulated‑annealing / biased hill climbing -----
    max_no_improve = 25000          # restart after this many rejections
    no_improve = 0
    T0 = 0.02                       # initial temperature
    Tmin = 1e-5

    while True:
        # Time guard
        if time.time() - start_time > budget_s - 0.2:
            break

        # Current correlation to detect the worst shift
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        k_max = int(np.argmax(corr) - (n - 1))

        # Identify ones that directly contribute to the worst shift
        if k_max >= 0:
            i_range = np.arange(0, n - k_max)
            bad_ones = i_range[(cur[i_range] == 1) & (cur[i_range + k_max] == 0)]
        else:
            kk = -k_max
            i_range = np.arange(kk, n)
            bad_ones = i_range[(cur[i_range] == 1) & (cur[i_range - kk] == 0)]

        # Choose donor index (prefer a “bad” one)
        if bad_ones.size > 0 and rng.random() < 0.7:
            i = int(rng.choice(bad_ones))
        else:
            i = int(rng.choice(ones))

        # Choose recipient zero uniformly
        j = int(rng.choice(zeros))

        # Perform tentative swap
        cur[i] = 0.0
        cur[j] = 1.0
        new_val = _compute_c5(cur, dx)

        # Temperature schedule (linear cooling)
        elapsed = time.time() - start_time
        T = max(Tmin, T0 * (1.0 - elapsed / budget_s))

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if rng.random() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            # Update bookkeeping
            cur_val = new_val
            no_improve = 0
            ones = list(np.flatnonzero(cur == 1))
            zeros = list(np.flatnonzero(cur == 0))
            if new_val < best_val:
                best_val = new_val
                best_h = cur.copy()
                if best_val < 0.3808:
                    break
        else:
            # Revert swap
            cur[i] = 1.0
            cur[j] = 0.0
            no_improve += 1

        # Random restart when stuck
        if no_improve >= max_no_improve:
            cur = _random_binary(n, target_sum, rng)
            cur_val = _compute_c5(cur, dx)
            ones = list(np.flatnonzero(cur == 1))
            zeros = list(np.flatnonzero(cur == 0))
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur.copy()

    # ----- final safety enforcement -----
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure the exact integral constraint (should already hold for binary)
    total = np.sum(best_h)
    diff = target_sum - total
    if abs(diff) > 1e-9:
        # diff is integer; fix by toggling the smallest bits
        diff_int = int(round(diff))
        if diff_int > 0:
            cand = np.where(best_h < 1)[0]
            best_h[cand[:diff_int]] = 1.0
        elif diff_int < 0:
            cand = np.where(best_h > 0)[0]
            best_h[cand[: -diff_int]] = 0.0

    final_val = _compute_c5(best_h, dx)
    return best_h, final_val, n
