def optimise_binary(n, max_time, rng, init_h=None):
    M = n // 2

    best_h = None
    best_c5 = np.inf

    start_time = time.time()
    # first try given init_h, else random.
    attempts = 0
    while time.time() - start_time < max_time:
        # Determine remaining time for this restart
        remaining = max_time - (time.time() - start_time)
        # allocate a fraction for this restart: e.g., min(30, remaining*0.5)
        restart_time = min(30.0, remaining * 0.5)
        if restart_time < 1.0:
            break

        if attempts == 0 and init_h is not None:
            h = init_h.copy()
        else:
            # random binary: choose M ones
            h = np.zeros(n, dtype=float)
            idx = rng.choice(n, M, replace=False)
            h[idx] = 1.0

        cur_c5 = compute_c5(h)
        # local best for this restart
        local_best_h = h.copy()

        restart_start = time.time()

            elapsed = time.time() - restart_start
            frac = elapsed / restart_time

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C₅ for a discrete step function h on [0,2].

    Uses FFT to evaluate the full linear correlation of h with (1‑h)
    and multiplies by the step size dx = 2/n.
    """
    n = h.shape[0]
    dx = 2.0 / n
    # length for linear (non‑circular) correlation
    N = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)[: 2 * n - 1]
    return np.max(corr.real) * dx


def _initial_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Construct a feasible start vector (sum = n/2, values in [0,1]).

    Uses the global `initial_h_values` when it has the right size,
    otherwise returns a random binary vector.
    """
    M = n // 2
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            # simple scaling; parent vectors are usually already close
            factor = M / cur
            h *= factor
            h = np.clip(h, 0.0, 1.0)
            cur = h.sum()
            if abs(cur - M) > 1e-12:
                h += (M - cur) / n
                h = np.clip(h, 0.0, 1.0)
        return h
    # random binary start
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, M, replace=False)
    h[idx] = 1.0
    return h


def optimise_binary(
    n: int,
    time_budget: float,
    rng: np.random.Generator,
    init_h: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style optimisation for a binary‑compatible h.

    The algorithm repeatedly performs mass‑transfer moves (including full
    swaps) while accepting occasional worsening moves according to a
    linear temperature schedule.
    """
    M = n // 2
    best_h = None
    best_c5 = np.inf

    outer_start = time.time()
    restart = 0
    while time.time() - outer_start < time_budget:
        # allocate a short window for this restart
        remaining = time_budget - (time.time() - outer_start)
        if remaining < 2.0:
            break
        restart_time = min(30.0, remaining * 0.5)

        # initialise h for this restart
        if restart == 0 and init_h is not None:
            h = init_h.copy()
        else:
            h = np.zeros(n, dtype=float)
            idx = rng.choice(n, M, replace=False)
            h[idx] = 1.0

        cur_c5 = compute_c5(h)
        # keep a local best for this restart
        local_best_h = h.copy()
        local_best_c5 = cur_c5

        # temperature schedule parameters
        T0 = max(cur_c5 * 0.2, 0.01)

        inner_start = time.time()
        while time.time() - inner_start < restart_time:
            # linear temperature decay
            elapsed = time.time() - inner_start
            frac = elapsed / restart_time
            T = T0 * (1.0 - frac) + 1e-12

            # propose a mass‑transfer move
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0.0:
                continue

            # sometimes perform a full swap (binary move)
            if rng.random() < 0.3:
                delta = max_transfer
            else:
                delta = rng.random() * max_transfer

            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta

            c5_new = compute_c5(h_new)
            d = c5_new - cur_c5
            if d <= 0.0 or rng.random() < np.exp(-d / T):
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < local_best_c5:
                    local_best_c5 = cur_c5
                    local_best_h = h.copy()
                # early exit if already better than target
                if cur_c5 < 0.3808:
                    return h, cur_c5

        # end of this restart
        if local_best_c5 < best_c5:
            best_c5 = local_best_c5
            best_h = local_best_h.copy()
        restart += 1

    if best_h is None:
        # fallback: random binary
        best_h = np.zeros(n, dtype=float)
        idx = rng.choice(n, M, replace=False)
        best_h[idx] = 1.0
        best_c5 = compute_c5(best_h)

    return best_h, best_c5


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Main entry point returning (h_values, c5_bound, n_points)."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Decide primary resolution
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 != 0:
            n0 = 80
    else:
        n0 = 80

    # Allocate most time to the primary size
    time_left = budget_s - (time.time() - start_time)
    primary_budget = max(0.0, time_left * 0.85)
    init_h = _initial_h(n0, rng)
    best_h, best_c5 = optimise_binary(n0, primary_budget, rng, init_h=init_h)
    best_n = n0

    # Try a few larger even sizes if time permits
    extra_sizes = [96, 112, 128, 144, 160]
    for n in extra_sizes:
        if time.time() - start_time > budget_s - 5.0:
            break
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5.0:
            break
        alloc = min(200.0, remaining * 0.5)
        h_n, c5_n = optimise_binary(n, alloc, rng, init_h=None)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n
            best_n = n
        if best_c5 < 0.3808:
            break

    # Final greedy refinement (deterministic improving swaps)
    n = best_n
    M = n // 2
    while time.time() - start_time < budget_s - 0.5:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0.0:
            continue
        # Try a full swap
        h_new = best_h.copy()
        h_new[i] -= max_transfer
        h_new[j] += max_transfer
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Safety checks and final projection onto the feasible set
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    target_sum = best_n // 2
    if abs(cur_sum - target_sum) > 1e-9:
        best_h += (target_sum - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
