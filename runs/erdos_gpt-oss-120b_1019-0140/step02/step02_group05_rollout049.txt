import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # zero pad to length 2n
    L = 2 * n
    # compute convolution via FFT
    # reverse (1-h) and pad
    a = np.concatenate([h, np.zeros(n)])
    b = np.concatenate([1.0 - h[::-1], np.zeros(n)])
    # real FFT
    fa = np.fft.rfft(a)
    fb = np.fft.rfft(b)
    conv = np.fft.irfft(fa * fb, n=L)
    corr = conv[:2*n-1]  # same size as np.correlate
    return np.max(corr) * dx
```

import time
import math
import numpy as np

def _full_corr(a, b):
    """Full correlation of a and b (both length n) using FFT."""
    n = a.shape[0]
    L = 2 * n
    # pad with zeros
    aa = np.concatenate([a, np.zeros(n)])
    bb = np.concatenate([b[::-1], np.zeros(n)])  # reverse b
    fa = np.fft.rfft(aa)
    fb = np.fft.rfft(bb)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: 2 * n - 1]

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = _full_corr(h, 1.0 - h)
    return np.max(corr) * dx

def _adjust_sum(vec, target):
    """Shift vec so that sum(vec)=target while staying in [0,1]."""
    cur = np.sum(vec)
    diff = target - cur
    if abs(diff) < 1e-14:
        return vec
    # first spread equally over interior points
    free = np.where((vec > 0) & (vec < 1))[0]
    if free.size:
        inc = diff / free.size
        vec[free] += inc
        np.clip(vec, 0.0, 1.0, out=vec)
    # fix any residual by a simple scan
    diff = target - np.sum(vec)
    if abs(diff) > 1e-14:
        for i in range(vec.size):
            if abs(diff) < 1e-14:
                break
            if diff > 0 and vec[i] < 1.0:
                add = min(diff, 1.0 - vec[i])
                vec[i] += add
                diff -= add
            elif diff < 0 and vec[i] > 0.0:
                sub = min(-diff, vec[i])
                vec[i] -= sub
                diff += sub
    np.clip(vec, 0.0, 1.0, out=vec)
    return vec

def _smooth_and_fix(h, target):
    """Gaussian-like smoothing (kernel 0.25,0.5,0.25) with sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    np.clip(sm, 0.0, 1.0, out=sm)
    sm = _adjust_sum(sm, target)
    return sm

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Grid size – larger than the previous 80 gives finer control.
    n = 200
    target_sum = n / 2.0
    dx = 2.0 / n

    best_h = None
    best_val = math.inf
    early_target = 0.3808  # stop if we beat this

    # Number of independent restarts (each gets a slice of the budget)
    restarts = 5
    time_per_restart = budget_s * 0.85 / restarts

    for _ in range(restarts):
        # ---- initialise -------------------------------------------------
        if "initial_h_values" in globals():
            init = np.array(initial_h_values, dtype=float)
            if init.shape[0] != n:
                # Interpolate to the new grid
                x_old = np.linspace(0.0, 2.0, init.shape[0], endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, init)
            else:
                h = init.copy()
            np.clip(h, 0.0, 1.0, out=h)
            h = _adjust_sum(h, target_sum)
        else:
            h = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1.0

        cur_h = h.copy()
        cur_val = compute_c5(cur_h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()

        restart_start = time.time()
        # Simulated annealing parameters
        T0 = 0.12
        while True:
            elapsed = time.time() - restart_start
            if elapsed >= time_per_restart:
                break
            # Linear temperature schedule
            T = T0 * max(0.0, 1.0 - elapsed / time_per_restart)

            if r < 0.4:  # swap two entries

    while time.time() - start < budget_s - 5.0:

        sm_val = compute_c5(sm)
        if sm_val + 1e-12 < best_val:
            best_val = sm_val
            best_h = sm
        else:

    # ---- final greedy refinement -----------------------------------------
    while time.time() - start < budget_s - 2.0:
        i, j = rng.choice(n, 2, replace=False)
        donor, recv = i, j
