    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    # While diff not ~0 and there are free variables
    for _ in range(10):

        nfree = np.sum(free)
        if nfree == 0:

        inc = diff / nfree
        x[free] += inc
        # clip again
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - np.sum(x)
        if abs(diff) < 1e-12:

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    if "initial_h_values" in globals():

        n = len(init_h)
    else:
        n = 80
        init_h = None

    target_sum = n / 2.0
    dx = 2.0 / n

    def project(h):
        # clip to [0,1] and enforce sum
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        # iterative
        for _ in range(10):
            free = (h > 0) & (h < 1)
            nfree = np.sum(free)
            if nfree == 0:
                break
            inc = diff / nfree
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
            if abs(diff) < 1e-12:
                break
        if abs(diff) > 1e-12:
            # fix residual
            if diff > 0:
                idx = np.where(h < 1)[0]
                for i in idx:
                    add = min(diff, 1 - h[i])
                    h[i] += add
                    diff -= add
                    if diff < 1e-12:
                        break
            else:
                idx = np.where(h > 0)[0]
                for i in idx:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
                    if -diff < 1e-12:
                        break
        return np.clip(h, 0.0, 1.0)

    def compute_OVals(h):
        # correlation without dx
        return np.correlate(h, 1.0 - h, mode='full')

    best_h = None

import time
import numpy as np
from scipy.optimize import minimize

def _project_onto_constraints(x, target_sum):
    """Project x onto [0,1]^n with sum == target_sum."""
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    for _ in range(10):
        free = (x > 0) & (x < 1)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            break
        inc = diff / nfree
        x[free] += inc
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - np.sum(x)
        if abs(diff) < 1e-12:
            break
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1)[0]
            for i in idx:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            idx = np.where(x > 0)[0]
            for i in idx:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
                if -diff <= 1e-12:
                    break
    return np.clip(x, 0.0, 1.0)


def _correlation_vals(h):
    """Return the full correlation vector of h with (1-h)."""
    return np.correlate(h, 1.0 - h, mode="full")


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Initialise problem size and target sum
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    else:
        n = 80
        h_init = None
    target_sum = n / 2.0
    dx = 2.0 / n

    # Helper to evaluate C5 for a given h
    def eval_c5(h):
        return np.max(_correlation_vals(h)) * dx

    # Keep the best solution seen
    best_h = None
    best_val = np.inf

    # Use provided initial vector if present
    if h_init is not None:
        h0 = _project_onto_constraints(h_init.copy(), target_sum)
        val0 = eval_c5(h0)
        best_h = h0.copy()
        best_val = val0

    # Main optimisation loop: many random restarts with smoothed gradient descent
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.94:
            break

        # Random start respecting the constraints
        h = rng.random(n)
        h = _project_onto_constraints(h, target_sum)

        # Gradient‑descent parameters
        lr = 0.20                # initial step size
        decay = 0.9995           # geometric decay per iteration
        alpha = 20.0             # smooth‑max parameter
        max_iter = 2000

        for it in range(max_iter):
            # occasional time check
            if (it & 31) == 0:
                if time.time() - start >= budget_s * 0.94:
                    break

            O = _correlation_vals(h)                 # raw overlap values (no dx)
            cur_val = np.max(O) * dx
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break

            # Smoothed max (log‑sum‑exp) weights
            O_max = np.max(O)
            exp_vals = np.exp(alpha * (O - O_max))
            w = exp_vals / np.sum(exp_vals)

            # Gradient of the smoothed objective
            grad = np.zeros_like(h)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    length = n - k
                    if length <= 0:
                        continue
                    grad[:length] += wk * (1.0 - h[k:])
                    grad[k:]      -= wk * h[:length]
                else:
                    l = -k
                    length = n - l
                    if length <= 0:

                    grad[l:]    += wk * (1.0 - h[:length])

            # Gradient step and projection back onto the feasible set
            h = h - lr * grad

            best_h = h.copy()
            if best_val < 0.3808:
                break

    # Optional local refinement with SLSQP if time permits
    if time.time() - start < budget_s - 5.0:

                return x[-1]                    # minimise the upper bound t

                        kk = -k

    final_c5 = eval_c5(best_h)
