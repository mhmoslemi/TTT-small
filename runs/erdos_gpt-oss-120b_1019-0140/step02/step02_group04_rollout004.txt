import time
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for step function h."""
    # Use np.correlate with zero padding (full mode)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum(vec, target_sum):
    """Clip to [0,1] then adjust to reach exact sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # distribute diff uniformly
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    # fix residual
    diff = target_sum - np.sum(vec)
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff < 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff > -1e-12:
                break
    return vec

def generate_random_binary(n, rng):
    """Return binary vector of length n with exactly n/2 ones."""
    k = n // 2
    arr = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=k, replace=False)
    arr[idx] = 1
    return arr

def refine_continuous(h, dx, target_sum, time_limit, rng):
    """Small continuous adjustments to improve C5."""
    cur_h = h.copy()
    cur_val = compute_c5(cur_h, dx)
    step = 0.05
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recips = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recips.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recips)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5(cand, dx)

            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.99
    # final projection to correct possible drift
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5(cur_h, dx)

    best_val = compute_c5(best_h, dx)
