import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bracket
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

    rng = np.random.default_rng(seed)
    start = time.time()
    n = 256
    # initialize h
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.shape[0] != n:
            h = upsample(init_full, n)
        else:
            h = init_full.copy()
    else:
        h = random_binary_balanced(n, rng)
    h = project_onto_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    dx = 2.0 / n
    # optimizer parameters
    lr0 = 0.6
    beta = 0.9
    momentum = np.zeros_like(h)
    alpha = 5.0
    alpha_max = 200.0
    it = 0
    last_improve = 0
    # main loop
    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 0.6:
            break
        it += 1
        # correlation and softmax weights
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_corr = np.max(corr)
        exp_arg = alpha * (corr - max_corr)   # max exponent = 0
        w = np.exp(exp_arg)
        w_sum = np.sum(w)
        p = w / w_sum
        # gradient via convolution (linear, zero padding)
        b = 1.0 - h
        conv_b = np.convolve(b, p, mode='full')
        conv_h = np.convolve(h, p, mode='full')

        grad = dx * (conv_b[offset:offset + n] - conv_h[offset:offset + n])
        # momentum and step
        momentum = beta * momentum + (1.0 - beta) * grad
        lr = lr0 / np.sqrt(it)
        h_new = h - lr * momentum
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5(h_new)
        if score + 1e-12 < best_score:
            best_score = score
            best_h = h_new.copy()
            last_improve = it
        # stagnation handling

            momentum = np.zeros_like(h)

    best_score = compute_c5(best_h)
    return best_h, best_score, n
