def binary_swap_hill_climb(h_bin, time_limit, rng):

    S_corr = np.correlate(h_bin, h_bin, mode='full')
    min_S = min(np.min(S_corr[:n-1]), np.min(S_corr[n:]))
    cur_c5 = (M - min_S) * dx
    best_h = h_bin.copy()

    start = time.time()
    idx_ones = np.flatnonzero(h_bin == 1)
    idx_zeros = np.flatnonzero(h_bin == 0)
    offset = n - 1
    hpad = np.concatenate([np.zeros(offset), h_bin, np.zeros(offset)])

def swap_hill_climb_continuous(h, time_limit, rng):

    S_corr = np.correlate(h, h, mode='full')
    # compute min_S excluding center

    cur_c5 = (M - min_S) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    offset = n - 1
    while time.time() - start < time_limit:
        # sample batch of candidate swaps:
        batch = 200
        # update idx lists
        idx_ones = np.flatnonzero(h > 1e-9)  # >0
        idx_zeros = np.flatnonzero(h < 1 - 1e-9)  # <1
        if len(idx_ones)==0 or len(idx_zeros)==0:
            break
        i_s = rng.choice(idx_ones, size=batch, replace=True)
        j_s = rng.choice(idx_zeros, size=batch, replace=True)
        max_transfer = np.minimum(1.0 - h[i_s], h[j_s])
        # ignore cases where max_transfer == 0
        valid = max_transfer > 1e-12
        if not np.any(valid):
            continue
        i_s = i_s[valid]
        j_s = j_s[valid]
        max_transfer = max_transfer[valid]
        # compute contributions for each i,j
        idx_range = np.arange(2*n - 1)
        hpad = np.concatenate([np.zeros(offset), h, np.zeros(offset)])

        idx_range = np.arange(2 * n - 1)
        hpad = np.concatenate([np.zeros(offset), h, np.zeros(offset)])

        contrib_i = hpad[i_s[:, None] + idx_range] + hpad[i_s[:, None] + 2 * offset - idx_range]
        contrib_j = hpad[j_s[:, None] + idx_range] + hpad[j_s[:, None] + 2 * offset - idx_range]

        delta_S = (contrib_i - contrib_j) * max_transfer[:, None]
        S_new = S_corr + delta_S

        # new minima (ignore centre)
        min_S_new = np.minimum(np.min(S_new[:, :n-1], axis=1), np.min(S_new[:, n:], axis=1))
        c5_new = (M - min_S_new) * dx
        best_idx = np.argmin(c5_new)

        if c5_new[best_idx] + 1e-12 < cur_c5:
            i = i_s[best_idx]
            j = j_s[best_idx]
            delta = max_transfer[best_idx]
            h[i] += delta
            h[j] -= delta
            h[i] = min(1.0, h[i])
            h[j] = max(0.0, h[j])
            S_corr = np.correlate(h, h, mode='full')
            min_S = min(np.min(S_corr[:n-1]), np.min(S_corr[n:]))
            cur_c5 = (M - min_S) * dx
            if cur_c5 + 1e-12 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            continue

        # no improvement in batch – try a modest random move
        i = rng.choice(idx_pos)
        j = rng.choice(idx_neg)
        max_t = min(1.0 - h[i], h[j])
        if max_t <= 0:
            continue
        delta = rng.random() * max_t
        h[i] += delta
        h[j] -= delta
        S_corr = np.correlate(h, h, mode='full')
        min_S = min(np.min(S_corr[:n-1]), np.min(S_corr[n:]))
        cur_c5 = (M - min_S) * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None
    target = 0.3808

    # candidate discretisations (even sizes)
    candidate_ns = [80, 96, 112, 128, 144, 160, 176, 192]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        alloc = min(250.0, remaining * 0.5)   # time for this resolution
        M = n // 2

        # try parent construction first
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h_parent = np.array(initial_h_values, dtype=float)
            cur = np.sum(h_parent)
            if abs(cur - M) > 1e-12:
                h_parent += (M - cur) / n
                h_parent = np.clip(h_parent, 0.0, 1.0)
            parent_time = min(alloc * 0.3, 20.0)
            h_ref, c5_ref = swap_hill_climb_continuous(h_parent.copy(), parent_time, rng)
            if c5_ref + 1e-12 < best_c5_global:
                best_c5_global = c5_ref
                best_h_global = h_ref
                best_n_global = n
                if best_c5_global < target:
                    break
            alloc -= parent_time

        start_n = time.time()
        # many random restarts within remaining time for this n
        while time.time() - start_n < alloc - 0.5:
            h0 = generate_initial_h(n, M, rng)
            time_left = alloc - (time.time() - start_n)
            per_inst = min(8.0, time_left)
            if per_inst <= 0.5:
                break
            h_opt, c5_opt = swap_hill_climb_continuous(h0, per_inst, rng)
            if c5_opt + 1e-12 < best_c5_global:
                best_c5_global = c5_opt
                best_h_global = h_opt
                best_n_global = n
                if best_c5_global < target:
                    break
        if best_c5_global < target:
            break

    # final safety fallback
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        cur = np.sum(best_h_global)
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h_global += (M - cur) / n
            best_h_global = np.clip(best_h_global, 0.0, 1.0)
        best_c5_global = compute_c5(best_h_global)

    return best_h_global, best_c5_global, n
