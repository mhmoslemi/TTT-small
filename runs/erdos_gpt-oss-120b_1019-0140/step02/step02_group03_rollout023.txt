import time

# Utility functions

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_k(h):
    """Return (max overlap, lag giving the max)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return corr[idx] * dx, idx - (n - 1)

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# Constraint classes for SLSQP
# -------------------------------------------------
class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Enforce t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = int(k)
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k > 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=300):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-7, 'disp': False}
    )
    return res

def gradient_worst_lag(h, k):
    """Gradient of -sum_i h_i h_{i+k} (to increase autocorrelation)."""
    n = h.shape[0]

        g[:-k] += h[k:]
    elif k < 0:
        kk = -k
        g[:-kk] += h[kk:]
        g[kk:] += h[:-kk]
    # k==0 gives zero gradient (should not happen)
    return g

    while time.time() < sa_deadline:
        elapsed = time.time() - start_time - budget_s * 0.3
        frac = min(1.0, max(0.0, elapsed / total_sa_time))
        T = t0 * ((t_end / t0) ** frac)  # exponential cooling

        # Choose direction
        if rng.random() < 0.65:
            # gradient based on worst lag
            _, worst_k = compute_c5_and_k(cur_h)
            d = gradient_worst_lag(cur_h, worst_k)
            if np.allclose(d, 0):
                d = rng.normal(size=n)
                d -= d.mean()
        else:
            d = rng.normal(size=n)
            d -= d.mean()

        # Adaptive step size
        step = base_step0 * (rng.random() ** 2) * (1.0 - frac) ** 2
        cand = cur_h + step * d
        cand = project_to_box_sum(cand, target_sum)

        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5

        if delta < -1e-13 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_h = cand
            cur_c5 = cand_c5
            no_improve = 0
            if cur_c5 < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                if best_c5 < 0.3805:
                    break
        else:
            no_improve += 1

        # Occasionally re‑seed to escape stagnation
        if no_improve > 8000:
            cur_h = best_h + 0.03 * rng.normal(size=n)
            cur_h = project_to_box_sum(cur_h, target_sum)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0

    # -------------------------------------------------
    # Final projection and evaluation
    # -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
