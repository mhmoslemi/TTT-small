                best_val = cand_val
                best_h = h_candidate.copy()
                if best_val < 0.3808:
                    break
        else:
            lr *= 0.7
            if lr < 1e-5:
                lr = 1e-5

        if t % 250 == 0:
            noisy = h + rng.normal(scale=0.02, size=n)
            noisy = project_onto_simplex_interval(noisy, target_sum)
            noisy_val = compute_c5(noisy)
            if noisy_val < best_val:
                h = noisy
                best_val = noisy_val
                best_h = noisy.copy()
                if best_val < 0.3808:
                    break

    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - target_sum) > 1e-9:
        best_h = project_onto_simplex_interval(best_h, target_sum)

    final_val = compute_c5(best_h)
    return best_h, final_val, n
