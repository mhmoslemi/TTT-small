def subgradient_max_corr(h):
    n = h.shape[0]
    v = np.correlate(h, 1.0 - h, mode='full')
    vmax = v.max()
    # tolerance for ties:
    tol = 1e-12
    mask = v >= vmax - tol
    # Compute gradient.
    g = np.zeros_like(h)
    offset = n - 1
    for idx, active in enumerate(mask):
        if not active:
            continue
        k = idx - offset
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
        g[tail] += (1.0 - h[head])
        g[head] -= h[tail]
    # Average if multiple maxima:
    count = mask.sum()
    if count > 1:
        g /= count
    return g
```

def simulated_annealing_optimize(h0, time_limit, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    no_improve = 0
    # Temperature schedule: start T0 = 0.01 maybe? Since difference in C5 small.
    # Since objective scale approx ~0.3-0.4, we set initial T = 0.01.
    T0 = 0.01
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac)  # linear decay to 0
        # propose perturbation: pick i,j distinct random indices
        i, j = rng.choice(n, size=2, replace=False)
        # delta uniformly in [-d, d] where d decays with time maybe.
        max_delta = 0.1 * (1 - frac) + 0.01
        delta = rng.uniform(-max_delta, max_delta)
        # adjust h[i] and h[j] to keep sum constant: h[i] += delta, h[j] -= delta
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = np.clip(h_new, 0.0, 1.0)
        # after clipping sum might change, fix via projection:
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        c5_curr = compute_c5(h)
        delta_c5 = c5_new - c5_curr
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(T, 1e-12)):
            h = h_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 5000:
            # random restart
            h = project_sum_and_bounds(rng.random(n), target)
            no_improve = 0
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Return C5 = max correlation * dx for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and adjust so that sum(x) == target (exactly)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # uniform adjustment
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # greedy correction
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def adam_optimize(h0, time_limit, seed):
    """Adam optimizer using a soft‑max approximation of the max‑overlap."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    dx = 2.0 / n

    # initial best
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = np.max(corr) * dx
    best_h = h.copy()

    # Adam state
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # smoothing for soft‑max
    beta = 2.0
    max_beta = 30.0

    start = time.time()
    no_improve = 0
    while time.time() - start < time_limit:
        t += 1
        # correlation
        corr = np.correlate(h, 1.0 - h, mode="full")
        vmax = corr.max()
        # soft‑max weights
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
        w = np.exp(beta * (corr - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # weighted gradient
        g = np.zeros_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # scale by dx (objective = vmax*dx)
        g *= dx

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate decay (linear)
        elapsed = time.time() - start
        lr0 = 0.2
        lr = lr0 * max(0.01, 1.0 - elapsed / time_limit)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_sum_and_bounds(h_new, target)

        # evaluate new C5
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")

            no_improve = 0

    return best_h, best_c5

def simulated_annealing_optimize(h0, time_limit, seed):
    """Simple SA that preserves total mass by pairwise moves."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)

    start = time.time()
    no_improve = 0
    T0 = 0.02  # initial temperature
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * max(0.0, 1.0 - frac)  # linear cooling

        # propose a pairwise delta
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = 0.12 * (1.0 - frac) + 0.02
        delta = rng.uniform(-max_delta, max_delta)

        h_new[i] += delta
        h_new[j] -= delta
        h_new = np.clip(h_new, 0.0, 1.0)

                no_improve = 0
