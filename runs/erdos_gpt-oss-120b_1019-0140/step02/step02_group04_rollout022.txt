    cur = np.sum(vec)

import time
import numpy as np

def compute_c5_vec(h, dx):
    """Maximum overlap integral for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target):
    """Project vec onto the simplex {0≤v≤1, sum(v)=target}."""
    v = np.clip(vec, 0.0, 1.0)
    cur = np.sum(v)
    diff = target - cur
    if abs(diff) < 1e-12:
        return v
    v += diff / len(v)
    v = np.clip(v, 0.0, 1.0)
    diff = target - np.sum(v)
    # greedy fix
    for _ in range(2 * len(v)):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v < 1.0)[0]
            if idx.size == 0:
                break
            i = idx[0]
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            idx = np.where(v > 0.0)[0]
            if idx.size == 0:
                break
            i = idx[0]
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    return v

def generate_random_feasible(n, target, rng):
    """Generate a random vector in [0,1]^n with exact sum = target."""
    x = rng.random(n)
    x = x / np.sum(x) * target
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
    return x

        if max_diff is not None:
            max_delta = min(max_delta, max_diff / 2.0)
        delta = float(rng.uniform(0.0, max_delta))
        if delta <= 0:
            continue

        cand = cur_h.copy()
        cand[donor_idx] -= delta
        cand[rec_idx] += delta
        cand_val = compute_c5_vec(cand, dx)

        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            no_improve = 0
            step = min(step * 1.1, 0.5)
            if cand_val < best_val - 1e-12:
                best_h = cand.copy()
                best_val = cand_val
        else:
            # simulated annealing acceptance
            elapsed = time.time() - start_time
            frac = elapsed / budget_s
            T = 0.02 * (1.0 - frac)
            if T > 0 and rng.random() < np.exp(-(cand_val - cur_val) / T):
                cur_h = cand
                cur_val = cand_val
            no_improve += 1
            if no_improve > 200:
                step = max(step * 0.5, 1e-3)
                cur_h = best_h.copy()
                cur_h += rng.normal(scale=0.01, size=n)
                cur_h = np.clip(cur_h, 0.0, 1.0)
                cur_h = project_to_sum(cur_h, target_sum)
                cur_val = compute_c5_vec(cur_h, dx)
                no_improve = 0

        # occasional random perturbation
        if rng.random() < 0.01:
            donors = np.where(cur_h > 1e-12)[0]
            recs = np.where(cur_h < 1.0 - 1e-12)[0]
            if donors.size and recs.size:
                d = int(rng.choice(donors))
                r = int(rng.choice(recs))
                md = min(step, cur_h[d], 1.0 - cur_h[r])
                if md > 0:
                    delta2 = float(rng.uniform(0.0, md))
                    cand2 = cur_h.copy()
                    cand2[d] -= delta2
                    cand2[r] += delta2
                    val2 = compute_c5_vec(cand2, dx)
                    if val2 < cur_val - 1e-12:
                        cur_h = cand2
                        cur_val = val2
                        if val2 < best_val - 1e-12:
                            best_h = cand2.copy()
                            best_val = val2

    # final safeguard
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)
    return best_h, best_val, n
