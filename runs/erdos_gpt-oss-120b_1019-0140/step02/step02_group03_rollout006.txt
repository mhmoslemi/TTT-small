import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def refine_by_hill_climb(h, score, target_sum, rng, max_iter=15):
    best_h = h.copy()
    best_score = score
    base_step = 0.3
    n = h.shape[0]
    for _ in range(max_iter):
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2)
        cand = best_h + step * d
        cand = project_to_box_sum(cand, target_sum)
        cand_score = compute_c5(cand)
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score
            base_step = 0.3
        else:
            base_step *= 0.9
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        target_sum = n / 2.0
        h0 = project_to_box_sum(init, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target_sum)

    # GA parameters
    pop_size = 40
    mutation_rate = 0.2
    mutation_std = 0.05

    # Initialize population
    population = [h0]
    for _ in range(pop_size - 1):
        v = rng.random(n)
        v = project_to_box_sum(v, target_sum)
        population.append(v)

    scores = [compute_c5(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # Main GA loop
    while time.time() - start_time < budget_s * 0.85:
        # tournament selection
        def tournament():
            idxs = rng.choice(pop_size, size=3, replace=False)
            return population[min(idxs, key=lambda i: scores[i])]

        parent_a = tournament()
        parent_b = tournament()

        # uniform crossover
        mask = rng.random(n) < 0.5
        child = np.where(mask, parent_a, parent_b)

        # mutation
        mutate_mask = rng.random(n) < mutation_rate
        child[mutate_mask] += rng.normal(scale=mutation_std, size=np.sum(mutate_mask))
        # clip and project to sum
        child = np.clip(child, 0.0, 1.0)
        child = project_to_box_sum(child, target_sum)

        child_score = compute_c5(child)

        # occasional hill‑climb refinement
        if rng.random() < 0.3:
            child, child_score = refine_by_hill_climb(child, child_score, target_sum, rng)

        # replacement: replace worst if better
        worst_idx = max(range(pop_size), key=lambda i: scores[i])
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_h = child.copy()
                best_score = child_score
                # early exit if already good enough
                if best_score < 0.3808:
                    break

    # Final local refinement on the best solution within remaining time
    remaining = budget_s - (time.time() - start_time)
    if remaining > 1.0:
        # perform hill climbing with more iterations within time limit
        # allocate half of remaining time for refinement
        refine_time = remaining * 0.5
        start_refine = time.time()
        while time.time() - start_refine < refine_time:
            best_h, best_score = refine_by_hill_climb(best_h, best_score, target_sum, rng, max_iter=20)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while enforcing sum(v) == target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def refine_by_hill_climb(h, score, target_sum, rng, max_iter=15):
    """Small hill‑climbing local improvement."""
    best_h = h.copy()
    best_score = score
    base_step = 0.3
    n = h.shape[0]
    for _ in range(max_iter):
        d = rng.normal(size=n)
        d -= d.mean()                     # zero‑mean direction preserves sum
        step = base_step * (rng.random() ** 2)
        cand = best_h + step * d
        cand = project_to_box_sum(cand, target_sum)
        cand_score = compute_c5(cand)
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score
            base_step = 0.3
        else:
            base_step *= 0.9
    return best_h, best_score
