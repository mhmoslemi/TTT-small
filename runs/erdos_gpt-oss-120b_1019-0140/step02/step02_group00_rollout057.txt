def compute_c5_binary(h):
    n = h.size
    # compute circular autocorrelation
    # using real FFT for speed
    h_fft = np.fft.rfft(h)
    # autocorr = np.fft.irfft(h_fft * np.conj(h_fft), n)
    # but need integer result; result is real.
    autocorr = np.fft.irfft(h_fft * np.conj(h_fft), n)
    # Values may have tiny floating errors; round to nearest integer.
    min_overlap = np.min(autocorr)
    c5 = 1.0 - (2.0 / n) * min_overlap
    return float(c5)

```
def compute_c5_autocorr(h):
    n = h.size
    # compute autocorrelation via FFT, real
    H = np.fft.rfft(h)
    autocorr = np.fft.irfft(H * np.conj(H), n)
    # due to float errors, clip to [0, n/2]
    min_overlap = np.min(autocorr)
    # c5 = 1 - 2*min_overlap/n
    return 1.0 - (2.0 / n) * min_overlap
```

```
        cur_sum = h.sum()

            h[i] = 0.0
```

```
def gradient_descent_momentum(h0, n_iter, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5_general(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project onto box [0,1] and sum constraint via Euclidean projection
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_general(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

```
def sa_binary_optimize(n, rng, time_left_func, init_h=None):
    if init_h is None:
        h = random_binary_balanced(n, rng)
    else:
        h = init_h.copy()
    best_h = h.copy()
    best_score = compute_c5_binary(best_h)
    cur_h = h.copy()

    start = time.time()
    total_time = time_left_func()
    if total_time <= 0:
        return best_h, best_score
    # schedule
    start_T = 0.02
    end_T = 0.0005
    while True:
        elapsed = time.time() - start
        if elapsed >= total_time - 0.01:  # leave tiny buffer
            break
        progress = elapsed / total_time
        T = start_T * (1 - progress) + end_T * progress
        # propose swap
        i = rng.choice(np.flatnonzero(cur_h))
        j = rng.choice(np.flatnonzero(1 - cur_h))

        cur_h[j] = 1.0

            cur_h[j] = 0.0

    return best_h, best_score
```

```
def upsample(arr, new_n):
    m = arr.shape[0]

    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5_general(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_c5_binary(h):
    n = h.shape[0]
    corr = np.correlate(h, h, mode='full')
    min_ov = int(round(np.min(corr)))
    return 1.0 - (2.0 / n) * min_ov

def compute_gradients_for_shifts(h, shifts):
    grad = np.zeros_like(h)
    for d in shifts:
        term1 = np.roll(h, -d)
        term2 = np.roll(h, d)
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_general(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_general(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def sa_binary_optimize_fixed_iters(n, rng, max_iters, init_h=None):
    if init_h is None:
        h = random_binary_balanced(n, rng)
    else:
        h = init_h.copy()
    best_h = h.copy()
    best_score = compute_c5_binary(best_h)
    cur_h = h.copy()
    cur_score = best_score
    start_T = 0.02
    end_T = 0.0005
    for it in range(max_iters):
        progress = it / max_iters
        T = start_T * (1 - progress) + end_T * progress
        i = rng.choice(np.flatnonzero(cur_h))
        j = rng.choice(np.flatnonzero(1 - cur_h))
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5_binary(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_score

        max_iters = int(run_sec * 2000)  # approx 2000 swaps/second
        if max_iters <= 0:
            break
        init = None
        if best_h is not None:
            init = (best_h >= 0.5).astype(float)
            cur_sum = init.sum()
            need = n // 2 - cur_sum
            while need > 0:
                i = rng.choice(np.flatnonzero(1 - init))
                init[i] = 1.0
                need -= 1
            while need < 0:
                i = rng.choice(np.flatnonzero(init))
                init[i] = 0.0
                need += 1
        cand_h, _ = sa_binary_optimize_fixed_iters(n, rng, max_iters, init)
        sc = compute_c5_general(cand_h)
        if sc < best_score:
            best_score, best_h = sc, cand_h.copy()
        if best_score <= target:
            break
    # ---- gradient‑momentum refinement ----
    if best_h is not None and time_left() > 2.0:
        est_iters = int(min(50000, max(2000, time_left() * 3000)))
        h_ref, sc_ref = gradient_descent_momentum(best_h, est_iters,
                                                  step0=0.5, beta=0.9)
        sc_ref = compute_c5_general(h_ref)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
    best_h = project_onto_box_sum(best_h, n/2.0)
    best_score = compute_c5_general(best_h)
    return best_h, best_score, n
