import time
import math
import numpy as np

def compute_c5(arr):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Generate Thue‑Morse binary sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    """Generate Rudin‑Shapiro binary sequence (+1/-1 → 1/0) of length n."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)                                   # a_{2k}
            nxt.append(v * (1 if (k % 2 == 0) else -1))    # a_{2k+1}=(-1)^k * a_k
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def adjust_to_target(vec, target_sum, rng):
    """Flip binary entries to reach exact target sum."""
    diff = target_sum - np.sum(vec)
    if diff > 0:
        zero_idxs = np.where(vec == 0)[0]
        rng.shuffle(zero_idxs)
        for idx in zero_idxs:
            if diff <= 0:
                break
            vec[idx] = 1.0
            diff -= 1.0
    elif diff < 0:
        one_idxs = np.where(vec == 1)[0]
        rng.shuffle(one_idxs)
        for idx in one_idxs:
            if diff >= 0:
                break
            vec[idx] = 0.0
            diff += 1.0
    # final numeric fix
    residual = target_sum - np.sum(vec)
    if abs(residual) > 1e-12:
        vec += residual / len(vec)
        vec = np.clip(vec, 0.0, 1.0)
        residual = target_sum - np.sum(vec)
        for i in range(len(vec)):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -vec[i], 1 - vec[i])
            vec[i] += adj
            residual = target_sum - np.sum(vec)
    return vec

def continuous_anneal(start_h, time_end, rng):
    """Simulated annealing on continuous variables preserving the sum."""
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 0.01
    final_temp = 1e-8
    start = time.time()
    eps = 1e-12
    while time.time() < time_end:
        inc_idx = np.where(h < 1 - eps)[0]
        dec_idx = np.where(h > eps)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        elapsed = time.time() - start
        total = max(time_end - start, 1e-12)
        prog = elapsed / total
        T = init_temp * (1 - prog) + final_temp * prog
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
    # final numeric correction (should be unnecessary but safe)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

def binary_hill_climb(start_bin, time_end, rng):
    """Deterministic hill‑climbing on binary vectors (swap moves)."""
    bin_vec = start_bin.copy()
    n = len(bin_vec)
    h = bin_vec.astype(float)
    best_c5 = compute_c5(h)

        j = rng.choice(zeros)
        bin_vec[i] = 0
        bin_vec[j] = 1
        cand_c5 = compute_c5(bin_vec.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_bin = bin_vec.copy()
            no_improve = 0
        else:
            bin_vec[i] = 1
            bin_vec[j] = 0
            no_improve += 1
    return best_bin.astype(float), best_c5

def directed_improvement(h, time_end, rng):
    """Greedy pairwise transfer aimed at reducing the maximal correlation."""
    best = h.copy()
    best_c5 = compute_c5(best)
    n = len(best)
    eps = 1e-12
    while time.time() < time_end:
        corr = np.correlate(best, 1 - best, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)                     # shift corresponding to max correlation
        # derivative d_i = 1 - h[i+k] - h[i-k] (out‑of‑range terms ignored)
        d = np.ones(n)
        if k >= 0:
            if k < n:
                d[:n - k] -= best[k:]
                d[k:] -= best[:n - k]
        else:
            kp = -k
            if kp < n:
                d[kp:] -= best[:n - kp]
                d[:n - kp] -= best[kp:]
        inc_idx = np.where(best < 1 - eps)[0]
        dec_idx = np.where(best > eps)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i_sel = dec_idx[np.argmax(d[dec_idx])]
        j_sel = inc_idx[np.argmin(d[inc_idx])]
        if d[i_sel] <= d[j_sel] + 1e-12:
            break
        max_delta = min(best[i_sel], 1 - best[j_sel])
        if max_delta < eps:
            break
        improved = False
        for scale in (1.0, 0.5, 0.25, 0.125, 0.0625):
            delta = max_delta * scale
            new_h = best.copy()
            new_h[i_sel] -= delta
            new_h[j_sel] += delta
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5 - 1e-12:
                best, best_c5 = new_h, new_c5
                improved = True
                break
        if not improved:
            # small random fallback
            i_rand = rng.choice(dec_idx)
            j_rand = rng.choice(inc_idx)
            max_delta2 = min(best[i_rand], 1 - best[j_rand])
            if max_delta2 < eps:
                break
            delta2 = max_delta2 * rng.random() * 0.1
            new_h = best.copy()
            new_h[i_rand] -= delta2
            new_h[j_rand] += delta2
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5 - 1e-12:
                best, best_c5 = new_h, new_c5
                continue
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    diff = target_sum - np.sum(h0)
    if abs(diff) > 1e-12:
        h0 += diff / n
        h0 = np.clip(h0, 0.0, 1.0)
        residual = target_sum - np.sum(h0)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -h0[i], 1 - h0[i])
            h0[i] += adj
            residual = target_sum - np.sum(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # seed pool
    seeds = [best_h.copy()]
    # random binary
    rand_bin = np.zeros(n, dtype=float)
    rand_bin[rng.choice(n, n // 2, replace=False)] = 1.0
    seeds.append(rand_bin)
    # Thue‑Morse
    tm = thue_morse_seq(n).astype(float)
    tm = adjust_to_target(tm, target_sum, rng)
    seeds.append(tm)
    # Rudin‑Shapiro
    rs = rudin_shapiro_seq(n).astype(float)
    rs = adjust_to_target(rs, target_sum, rng)
    seeds.append(rs)
    # a few random continuous seeds
    for _ in range(2):
        vec = rng.random(n)
        vec = np.clip(vec, 0.0, 1.0)
        s = np.sum(vec)
        if s > 0:
            vec *= target_sum / s
        vec = np.clip(vec, 0.0, 1.0)
        diff = target_sum - np.sum(vec)
        if abs(diff) > 1e-12:
            vec += diff / n
            vec = np.clip(vec, 0.0, 1.0)
            residual = target_sum - np.sum(vec)
            for i in range(n):
                if abs(residual) < 1e-12:
                    break
                adj = np.clip(residual, -vec[i], 1 - vec[i])
                vec[i] += adj
                residual = target_sum - np.sum(vec)
        seeds.append(vec)

    # main optimization loop
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 2.0:
            break
        chunk = min(5.0, remaining * 0.4)
        end_time = time.time() + chunk

        if rng.random() < 0.2:
            start_vec = seeds[rng.integers(len(seeds))]
        else:
            start_vec = best_h.copy()
            for _ in range(rng.integers(1, 4)):
                inc_idx = np.where(start_vec < 1 - 1e-12)[0]
                dec_idx = np.where(start_vec > 1e-12)[0]
                if inc_idx.size == 0 or dec_idx.size == 0:
                    break
                i = rng.choice(inc_idx)
                j = rng.choice(dec_idx)
                max_delta = min(1 - start_vec[i], start_vec[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random() * 0.5
                start_vec[i] += delta
                start_vec[j] -= delta

        if rng.random() < 0.7:
            cand_h, cand_c5 = continuous_anneal(start_vec, end_time, rng)
        else:
            start_bin = (start_vec >= 0.5).astype(int)
            cand_h, cand_c5 = binary_hill_climb(start_bin, end_time, rng)

        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h.copy(), cand_c5

        if best_c5 <= 0.3808:
            break

    # directed greedy improvement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 1.0:
        finish_time = time.time() + remaining - 0.5
        best_h, best_c5 = directed_improvement(best_h, finish_time, rng)

    # final binary hill climb on rounded version
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0.5:
        bin_start = (best_h >= 0.5).astype(int)
        final_bin, final_c5 = binary_hill_climb(bin_start, time.time() + remaining - 0.2, rng)
        if final_c5 < best_c5:
            best_h, best_c5 = final_bin.astype(float), final_c5

    # final numeric cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    return best_h, best_c5, n
