def pairwise_local_search(h, best_c5, time_deadline):
    n = len(h)
    target = n/2
    cur_h = h.copy()
    cur_c5 = best_c5
    step = 0.2  # max delta to move
    while time.time() < time_deadline:
        # randomly pick i and j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # ensure we can move mass from i to j
        if cur_h[i] <= 1e-12 or cur_h[j] >= 1-1e-12:
            continue
        max_delta = min(step, cur_h[i], 1-cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            # optional: shrink step
            step = min(step, 0.5) # keep moderate
        else:
            # occasionally reduce step
            if rng.random() < 0.01:
                step *= 0.9
    return cur_h, cur_c5

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    for oc in overlap_cons:
        constraints.append({'type': 'ineq', 'fun': oc})
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res
```

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while forcing sum(v)=target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    """Equality constraint sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    """Run one SLSQP optimisation on (h,t)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{'type': 'eq', 'fun': sum_con}]
    cons += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- choose discretization size -----
    n = 120
    target_sum = n / 2.0

    # ----- initialise h -----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != n:
            # linear upscale/downscale to required size
            old_n = h0.size
            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(new_x, old_x, h0)
    else:
        h0 = rng.random(n)
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----- prepare constraints for SLSQP -----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # ----- Phase 1: many SLSQP restarts -----
    phase1_deadline = start_time + budget_s * 0.65
    while time.time() < phase1_deadline:
        # random feasible start
        h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, target_sum)
        t_start = compute_c5(h_start)

        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800)
        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # enforce exact sum after optimisation
        if abs(h_opt.sum() - target_sum) > 1e-9:
            h_opt = project_to_box_sum(h_opt, target_sum)

        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            if best_c5 < 0.3806:
                break

    # ----- Phase 2: pairwise mass‑transfer hill climbing -----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    step = 0.2  # maximum mass moved per trial
    phase2_deadline = start_time + budget_s - 0.5
    while time.time() < phase2_deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                if best_c5 < 0.3806:
                    break
        else:
            if rng.random() < 0.01:
                step *= 0.9

    # ----- Optional final polishing with SLSQP (if time permits) -----
    remaining = start_time + budget_s - time.time()
    if remaining > 5.0:
        t_start = compute_c5(best_h)
        res = slsqp_optimize(best_h, t_start, sum_con, overlap_cons, maxiter=500)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # ----- final projection & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
