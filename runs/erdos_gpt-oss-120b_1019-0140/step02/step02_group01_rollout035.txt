def optimise_for_n(n, time_budget, rng):
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    constraints = [{'type':'eq','fun': sum_con}]
    constraints += [{'type':'ineq','fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf
    start = time.time()

    # use parent if available
    if 'initial_h_values' in globals():
        if len(initial_h_values) == n:
            parent_h = np.array(initial_h_values, dtype=float)
            # adjust sum
            cur = np.sum(parent_h)
            if abs(cur - M) > 1e-12:
                parent_h += (M - cur) / n
                parent_h = np.clip(parent_h, 0.0, 1.0)
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h
            best_c5 = parent_c5

    while time.time() - start < time_budget - 0.05:
        # generate initial h
        if best_h is not None and np.random.rand() < 0.1:
            h0 = best_h.copy()

def adjust_sum(h, M):
    cur = np.sum(h)
    if abs(cur - M) <= 1e-12:
        return h
    diff = M - cur
    # if diff >0, need increase values
    if diff > 0:
        # increase smallest entries not at 1
        for idx in np.argsort(h):
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        for idx in np.argsort(-h):
            dec = min(diff, h[idx])
            h[idx] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    return h
```

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- Constraint helpers ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.M = n // 2

    def __call__(self, x):
        return np.sum(x[:self.n]) - self.M


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx


# ---------- Objective ----------
def compute_c5(h):
    """Maximum overlap integral for a given (discrete) h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def objective_t(x):
    """Auxiliary variable t to be minimised."""
    return x[-1]


# ---------- Helper to fix sum ----------
def adjust_sum(h, M):
    cur = np.sum(h)
    if abs(cur - M) <= 1e-12:
        return h
    diff = M - cur
    if diff > 0:
        for idx in np.argsort(h):
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        for idx in np.argsort(-h):
            dec = min(diff, h[idx])
            h[idx] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    return h

        h = adjust_sum(h, M)
    elif method == 2:

        x = np.linspace(0, np.pi, n)
        h = np.sin(x)
        h = h / np.sum(h) * M
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum(h, M)
    elif method == 3:
        # linear decreasing
        lin = np.linspace(1.0, 0.0, n)
        h = lin / np.sum(lin) * M
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum(h, M)
    else:
        # parabolic bump (peak in the centre)
        x = np.arange(n, dtype=float)
        centre = (n - 1) / 2.0
        h = 1.0 - ((x - centre) / centre) ** 2
        h = np.clip(h, 0.0, None)
        h = h / np.sum(h) * M
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum(h, M)
    return h


# ---------- Optimisation for a fixed n ----------
def optimise_for_n(n, time_budget, rng):
    """Run SLSQP restarts for a given n within time_budget seconds."""
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf
    start = time.time()

    # include parent construction if it matches this size
    use_parent = False
    parent_h = None
    if 'initial_h_values' in globals():
        if len(initial_h_values) == n:
            use_parent = True
            parent_h = np.array(initial_h_values, dtype=float)
            cur = np.sum(parent_h)
            if abs(cur - M) > 1e-12:
                parent_h += (M - cur) / n
                parent_h = np.clip(parent_h, 0.0, 1.0)
            parent_c5 = compute_c5(parent_h)
            best_h, best_c5 = parent_h, parent_c5

    while time.time() - start < time_budget - 0.1:
        # generate starting point
        if use_parent:
            h0 = parent_h.copy()
            use_parent = False
        else:
            h0 = generate_initial_h(n, M, rng)

        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])

        try:
            res = minimize(fun=objective_t,
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 1500, 'ftol': 1e-9, 'disp': False})
        except Exception:
            continue

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur = np.sum(h_opt)
        if abs(cur - M) > 1e-8:
            h_opt += (M - cur) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        c5_opt = compute_c5(h_opt)

        if c5_opt + 1e-12 < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            if best_c5 < 0.3808:
                break

    return best_h, best_c5


# ---------- Simulated annealing refinement ----------
def refine_sa(h, c5, n, rng, start_time, budget_s):
    """Hill‑climbing with occasional uphill moves."""
    best_h = h.copy()
    best_c5 = c5
    cur_h = h.copy()
    cur_c5 = c5
    M = n // 2

    remaining_total = budget_s - (time.time() - start_time) - 0.5
    if remaining_total <= 0.5:
        return best_h, best_c5

    end_time = time.time() + remaining_total
    T0 = 0.01  # initial temperature

    while time.time() < end_time:
        # linear temperature schedule
        T = T0 * (end_time - time.time()) / remaining_total
        # propose a donor‑receiver move
        i, j = rng.choice(n, 2, replace=False)
        if rng.random() < 0.5:
            donor, rec = i, j
        else:
            donor, rec = j, i
        max_transfer = min(cur_h[donor], 1.0 - cur_h[rec])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        if delta <= 1e-14:
            continue
        new_h = cur_h.copy()
        new_h[donor] -= delta
        new_h[rec] += delta
        new_c5 = compute_c5(new_h)

        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
        else:
            # uphill acceptance
            if T > 0 and rng.random() < math.exp(-(new_c5 - cur_c5) / T):
                cur_h, cur_c5 = new_h, new_c5

        # early exit if we already beat the target
        if best_c5 < 0.3808:
            break

    return best_h, best_c5


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # candidate discretisations (even numbers, modest size)
    candidate_ns = [80, 96, 112, 128, 144, 160, 176, 192]

    # ensure parent size is tried first
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 not in candidate_ns:
            candidate_ns.insert(0, n0)
        elif n0 % 2 == 0:
            candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    # allocate roughly equal time per resolution, but keep a safety margin
    per_n_budget = max(5.0, min(250.0, (budget_s - 10) / max(1, len(candidate_ns))))

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 8:
            break
        remaining = budget_s - (time.time() - start_time)
        alloc = min(per_n_budget, remaining * 0.5)
        h_opt, c5_opt = optimise_for_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global, best_c5_global, best_n_global = h_opt, c5_opt, n
            if best_c5_global < 0.3808:
                break

    # ----- local refinement via simulated annealing -----
    if best_h_global is not None:
        best_h_global, best_c5_global = refine_sa(
            best_h_global,
            best_c5_global,
            best_n_global,
            rng,
            start_time,
            budget_s
        )

    # ----- final safety checks / fallback -----
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        cur = np.sum(best_h_global)
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h_global += (M - cur) / n
            best_h_global = np.clip(best_h_global, 0.0, 1.0)

    return best_h_global, float(best_c5_global), n
