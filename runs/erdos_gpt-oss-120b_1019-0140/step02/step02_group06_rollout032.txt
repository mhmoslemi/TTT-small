def subgradient_descent(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    no_improve = 0
    while time.time() - start < time_budget:
        # compute correlation and max indices
        v = np.correlate(h, 1.0 - h, mode='full')
        max_v = v.max()
        # find indices where v near max (within tolerance)
        # it's integer so exact max.
        max_idxs = np.where(v == max_v)[0]
        # choose random index among them
        idx = rng.choice(max_idxs)
        # shift k index: k = idx - (n-1)
        k = idx - (n-1)
        # compute gradient for this shift
        g = np.zeros(n)
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
            g[tail] += (1.0 - h[head])
            g[head] -= h[tail]
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
            g[tail] += (1.0 - h[head])
            g[head] -= h[tail]
        # zero-sum direction
        g -= np.mean(g)
        # step-size schedule
        elapsed = time.time() - start
        alpha = 0.2 * max(0.05, 1.0 - elapsed / time_budget)
        # perturb? acceptance?
        h_new = h - alpha * g
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        # Accept if improves else with small probability
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        # Accept to move state (for exploring) if improves or with small prob
        if c5_new < compute_c5(h) or rng.random() < 0.001:
            h = h_new
        else:
            # maybe occasional random restart
            if no_improve > 2000:
                # random restart near best
                h = best_h + rng.normal(scale=0.02, size=n)
                h = project_h(h, target)
                no_improve = 0
        # occasional random perturbation to escape local minima
        if rng.random() < 0.01:
            h = project_h(h + rng.normal(scale=0.05, size=n), target)
    return best_h, best_c5

    start = time.time()

    ones = np.where(cur_h > 0.5)[0].tolist()
    zeros = np.where(cur_h < 0.5)[0].tolist()
    # Temperature schedule for SA
    init_temp = 0.02
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        temp = init_temp * max(0.0, 1.0 - elapsed / time_budget)
        # if not enough ones or zeros then break
        if not ones or not zeros:
            break
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        # swap
        cur_h[i_one] = 0.0
        cur_h[i_zero] = 1.0
        # update ones and zeros lists
        # To avoid deep copying each time, we can just recalc after swap
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        accept = False
        if new_c5 < cur_c5 - 1e-12:
            accept = True
        else:
            # probabilistic acceptance
            if temp > 0 and rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            cur_c5 = new_c5
            # update indices lists
            ones.remove(i_one)
            zeros.remove(i_zero)
            ones.append(i_zero)
            zeros.append(i_one)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i_one] = 1.0
            cur_h[i_zero] = 0.0
        # occasional random shake to escape deeper local minima
        if rng.random() < 0.002:
            # random new binary pattern with same number of ones
            cur_h[:] = 0.0
            cur_h[rng.choice(n, target, replace=False)] = 1.0
            cur_c5 = compute_c5(cur_h)
            ones = np.where(cur_h > 0.5)[0].tolist()
            zeros = np.where(cur_h < 0.5)[0].tolist()

def subgradient_descent(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    no_improve = 0
    while time.time() - start < time_budget:
        # compute correlation vector
        v = np.correlate(h, 1.0 - h, mode='full')
        max_v = v.max()
        max_idxs = np.where(v == max_v)[0]
        idx = rng.choice(max_idxs)
        k = idx - (n - 1)
        # gradient for that shift
        g = np.zeros(n)
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
            g[tail] += (1.0 - h[head])
            g[head] -= h[tail]
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
            g[tail] += (1.0 - h[head])
            g[head] -= h[tail]
        # zero sum
        g -= np.mean(g)
        # step size schedule
        elapsed = time.time() - start
        alpha = 0.2 * max(0.001, 1.0 - elapsed / time_budget)
        # update
        h_new = h - alpha * g
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        # accept if improve or small prob
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if rng.random() < 0.001:
                h = h_new
                cur_c5 = c5_new
        # update best
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        # random shake if stuck
        if no_improve > 2000:
            # random perturbation
            h = project_h(h + rng.normal(scale=0.02, size=n), target)
            cur_c5 = compute_c5(h)
            no_improve = 0
    return best_h, best_c5

def project_h(x, target):
    x = np.clip(x, 0.0, 1.0)
    # adjust sum to target
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # iterative adjustment
    # We'll use water-filling approach: adjust uniformly within bounds
    # Continue until diff is sufficiently small
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase some entries
            # capacity = 1 - x
            capacity = 1.0 - x
            total_cap = capacity.sum()
            if total_cap < 1e-12:
                break
            inc = diff * capacity / total_cap
            # ensure we don't exceed 1
            inc = np.minimum(inc, capacity)
            x += inc
        else:
            # need to decrease
            capacity = x
            total_cap = capacity.sum()
            if total_cap < 1e-12:
                break
            dec = (-diff) * capacity / total_cap
            dec = np.minimum(dec, capacity)
            x -= dec
        cur = x.sum()
        diff = target - cur
    # final clip
    x = np.clip(x, 0.0, 1.0)
    return x
```

This water-filling ensures total sum matches target accurately in finite steps.

import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_h(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - x
            total_cap = cap.sum()
            if total_cap <= 0:
                break
            inc = diff * cap / total_cap
            inc = np.minimum(inc, cap)
            x += inc
        else:
            cap = x
            total_cap = cap.sum()
            if total_cap <= 0:
                break
            dec = (-diff) * cap / total_cap
            dec = np.minimum(dec, cap)
            x -= dec
        cur = x.sum()
        diff = target - cur
    return np.clip(x, 0.0, 1.0)

def subgradient_descent(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        max_v = v.max()
        max_idxs = np.where(v == max_v)[0]
        idx = rng.choice(max_idxs)
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
            g[tail] += (1.0 - h[head])
            g[head] -= h[tail]
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
            g[tail] += (1.0 - h[head])
            g[head] -= h[tail]
        g -= np.mean(g)
        elapsed = time.time() - start
        alpha = 0.2 * max(0.001, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if rng.random() < 0.001:
                h = h_new
                cur_c5 = c5_new
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        if no_improve > 2000:
            h = project_h(h + rng.normal(scale=0.02, size=n), target)
            cur_c5 = compute_c5(h)
            no_improve = 0
    return best_h, best_c5

def binary_hill_climb(h_start, time_budget, seed):
    rng = np.random.default_rng(seed)
    n = len(h_start)
    target = int(round(h_start.sum()))
    cur_h = h_start.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur_h > 0.5)[0])
    zeros = list(np.where(cur_h < 0.5)[0])
    start = time.time()
    init_temp = 0.02
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        temp = init_temp * max(0.0, 1.0 - elapsed / time_budget)
        if not ones or not zeros:
            break
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        cur_h[i_one] = 0.0
        cur_h[i_zero] = 1.0
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        accept = False
        if new_c5 < cur_c5 - 1e-12:
            accept = True
        elif temp > 0 and rng.random() < np.exp(-delta / temp):
            accept = True
        if accept:
            cur_c5 = new_c5
            ones.remove(i_one)
            zeros.remove(i_zero)
            ones.append(i_zero)
            zeros.append(i_one)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            cur_h[i_one] = 1.0
            cur_h[i_zero] = 0.0
        if rng.random() < 0.002:
            cur_h[:] = 0.0
            cur_h[rng.choice(n, target, replace=False)] = 1.0
            cur_c5 = compute_c5(cur_h)
            ones = list(np.where(cur_h > 0.5)[0])
            zeros = list(np.where(cur_h < 0.5)[0])
    return best_h, best_c5
