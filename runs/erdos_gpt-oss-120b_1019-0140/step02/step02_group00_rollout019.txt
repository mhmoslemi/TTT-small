def gradient_descent_momentum_custom(h0, max_iters, time_left_func):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    step0 = 0.5
    # choose top_k for gradient: e.g., top 3 shifts
    top_k = 3
    for it in range(1, max_iters+1):
        if time_left_func() < 0.01:
            break
        # compute correlation full
        corr = np.correlate(h, 1 - h, mode='full')
        # find top_k shifts
        # use argpartition: for top k largest values
        if top_k >= corr.size:
            idxes = np.arange(corr.size)
        else:
            idxes = np.argpartition(corr, -top_k)[-top_k:]
        shifts = idxes - (n - 1)
        # compute gradient sum over these shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[sft:] = h[:n - sft]
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection to box and sum constraint
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse_seq(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradient_for_shifts(h, shifts):
    """Gradient of max‑correlation objective averaged over given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift == 0:
            # skip shift zero (its contribution is zero for binary, and symmetric)
            continue
        if shift > 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:  # shift < 0
            sft = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[sft:] = h[:n - sft]
            term2[:n - sft] = h[sft:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= max(len(shifts), 1)
    return grad


def gradient_descent_momentum_custom(h0, max_iters, start_time, budget_s,
                                     top_k=5, beta=0.9, step0=0.5):
    """Projected gradient descent with momentum on top‑k shift gradients."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, max_iters + 1):
        # time guard
        if budget_s - (time.time() - start_time) < 0.01:
            break

        # correlation and top‑k shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        if top_k >= corr.size:
            idxes = np.arange(corr.size)
        else:
            idxes = np.argpartition(corr, -top_k)[-top_k:]
        shifts = idxes - (n - 1)

        # gradient
        grad = compute_gradient_for_shifts(h, shifts)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        eta = step0 / np.sqrt(it)

        # projected update
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)

        # scoring
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def binary_hill_climb(h_cont, start_time, budget_s, rng,
                      max_trials=20000):
    """Convert continuous to binary and perform greedy swap improvement."""
    n = h_cont.size
    half = n // 2
    # threshold to binary (top half ones)
    idx_desc = np.argsort(-h_cont)
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx_desc[:half]] = 1.0

    best_h = h_bin.copy()
    best_score = compute_c5(best_h)

    for _ in range(max_trials):
        if budget_s - (time.time() - start_time) < 0.001:
            break
        one_idxs = np.where(best_h == 1)[0]
        zero_idxs = np.where(best_h == 0)[0]
        if one_idxs.size == 0 or zero_idxs.size == 0:
            break
        i = rng.choice(one_idxs)
        j = rng.choice(zero_idxs)
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        sc = compute_c5(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand
            if best_score <= 0.3808:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    n = 256  # target resolution (even)

    # ----- Build initial candidates -----
    candidates = []

    # use parent construction if available
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, init_full.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            cand = np.interp(x_new, x_old, init_full)
        else:
            cand = init_full.copy()
        cand = project_onto_box_sum(cand, n / 2.0)
        candidates.append(cand)

    # Thue‑Morse pattern
    candidates.append(project_onto_box_sum(thue_morse_seq(n), n / 2.0))

    # a few random continuous starts
    for _ in range(3):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))

    # linear ramp (scaled to satisfy integral)
    ramp = np.linspace(0.0, 1.0, n)
    ramp = ramp * ((n / 2.0) / np.sum(ramp))
    candidates.append(project_onto_box_sum(ramp, n / 2.0))

    best_h = None
    best_score = np.inf

    # ----- Main optimisation passes -----
    for cand in candidates:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5.0:
            break

        # allocate generous iteration budget
        max_iters = int(min(200000, remaining * 500))
        h_opt, sc_opt = gradient_descent_momentum_custom(
            cand, max_iters, start_time, budget_s,
            top_k=5, beta=0.9, step0=0.5)

        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

        # binary hill climbing on the best found so far
        remaining = budget_s - (time.time() - start_time)
        if remaining > 5.0 and best_h is not None:
            h_bin, sc_bin = binary_hill_climb(best_h, start_time, budget_s, rng)
            if sc_bin < best_score:
                best_score = sc_bin
                best_h = h_bin.copy()

        if best_score <= 0.3808:
            break

    # ----- Random binary search (fallback) -----
    while (budget_s - (time.time() - start_time)) > 5.0:
        h_rand = random_balanced_binary(n, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score = sc_rand
            best_h = h_rand.copy()
        if best_score <= 0.3808:
            break

    # ----- Final refinement if time permits -----
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0 and best_h is not None:
        extra_iters = int(min(50000, remaining * 500))
        h_ref, sc_ref = gradient_descent_momentum_custom(
            best_h, extra_iters, start_time, budget_s,
            top_k=5, beta=0.9, step0=0.5)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # ----- Guard against constraint drift -----
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, n
