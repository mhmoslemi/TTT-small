def binary_sa(init_bin, dx, time_limit, rng):
    n = len(init_bin)
    cur = init_bin.copy()
    cur_val = compute_c5_vec(cur.astype(float), dx)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    last_improve_time = start
    iteration = 0
    while time.time() < time_limit:
        iteration += 1
        elapsed = time.time() - start
        frac = elapsed / (time_limit - start)
        # temperature schedule:
        temp = 0.02 * (1 - frac) + 1e-5 * frac  # start 0.02 end 1e-5
        # choose swap
        ones_idx = np.where(cur == 1)[0]
        zeros_idx = np.where(cur == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        cand = cur.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val - 1e-12:
                best = cur.copy()
                best_val = cur_val
                last_improve_time = time.time()
        # optional restart if no improvement for many iterations
        if iteration % 5000 == 0:
            # check time left: if >some fraction, possibly restart
            if time.time() - last_improve_time > (time_limit - start) * 0.2:
                cur = generate_random_binary(n, rng)
                cur_val = compute_c5_vec(cur.astype(float), dx)
    return best, best_val

def refine_continuous(h, dx, target_sum, time_limit, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.1
    while time.time() < time_limit:
        # pick donor and recipient
        donor = rng.choice(np.where(cur_h > 1e-12)[0])
        recipient = rng.choice(np.where(cur_h < 1.0 - 1e-12)[0])
        max_delta = min(step, cur_h[donor], 1.0 - cur_h[recipient])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.random() * max_delta
        cand = cur_h.copy()
        cand[donor] -= delta
        cand[recipient] += delta
        # project to sum maybe not needed as we maintain sum exactly via transfer
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.99
    # final projection enforce sum exactly
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

import time
import math
import numpy as np

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given discretized h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """
    Adjust `vec` so that sum(vec) == target_sum while keeping each entry in [0,1].
    Simple iterative projection: clip, then distribute residual uniformly,
    finally fix any remaining deficit/excess greedily.
    """
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # distribute uniformly
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    # greedy fix
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_binary(n, rng):
    """Return a binary vector of length n with exactly n//2 ones."""
    v = np.zeros(n, dtype=int)
    ones = rng.choice(n, n // 2, replace=False)
    v[ones] = 1
    return v


# ---------- Continuous Simulated Annealing ----------
def continuous_sa(init_h, dx, deadline, rng):
    """Continuous SA that preserves the sum by transferring mass between entries."""
    cur = np.clip(init_h.copy(), 0.0, 1.0)
    target_sum = np.sum(cur)
    cur = project_to_sum(cur, target_sum)
    cur_val = compute_c5_vec(cur, dx)

    best = cur.copy()
    best_val = cur_val

    n = len(cur)
    start = time.time()
    # SA schedule parameters
    T0 = 0.04
    Tmin = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        temp = T0 * (1.0 - frac) + Tmin * frac

        # pick donor (with some mass) and recipient (room for mass)
        donors = np.where(cur > 1e-12)[0]
        recipients = np.where(cur < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)

        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = compute_c5_vec(cand, dx)
        delta_val = cand_val - cur_val

        if delta_val < 0 or rng.random() < math.exp(-delta_val / max(temp, 1e-12)):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val - 1e-12:
                best = cur.copy()
                best_val = cur_val
    return best, best_val


# ---------- Local refinement (coordinate descent) ----------
def refine_continuous(h, dx, target_sum, deadline, rng):
    """Simple coordinate descent moving a random amount of mass between entries."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.1

    while time.time() < deadline:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.random() * max_delta
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.99

    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where
    h_values satisfies 0<=h<=1, sum(h)*dx == 1 and tries to minimize C5.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # Starting vector
    if h0 is not None:
        # Clip and enforce feasibility
        cur_start = np.clip(h0, 0.0, 1.0)
        cur_start = project_to_sum(cur_start, target_sum)
    else:
        cur_start = np.full(n, target_sum / n)  # uniform 0.5

    # Allocate time slices
    remaining = budget_s - (time.time() - start_time)
    if remaining <= 0:
        best_h = cur_start
        best_val = compute_c5_vec(best_h, dx)
        return best_h, best_val, n

    # 65% of remaining time for continuous SA, rest for refinement
    time_sa = remaining * 0.65
    deadline_sa = time.time() + time_sa

    best_h, best_val = continuous_sa(cur_start, dx, deadline_sa, rng)

    # Final local refinement with remaining time (minus tiny buffer)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        deadline_ref = time.time() + remaining - 0.5
        h_ref, val_ref = refine_continuous(best_h, dx, target_sum, deadline_ref, rng)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref

    # Safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
