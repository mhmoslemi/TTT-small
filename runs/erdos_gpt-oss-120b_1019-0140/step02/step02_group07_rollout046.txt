def repair(x):
    y = np.clip(x, 0.0, 1.0)
    # aim to approximate sum constraint
    target = n/2
    diff = target - np.sum(y)
    y += diff / n
    y = np.clip(y, 0.0, 1.0)
    # second pass
    diff = target - np.sum(y)
    if abs(diff) > 1e-12:
        # distribute residual one by one
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            # compute available room: can increase if diff>0, decrease if diff<0
            if diff > 0 and y[i] < 1.0:
                inc = min(diff, 1.0 - y[i])
                y[i] += inc
                diff -= inc
            elif diff < 0 and y[i] > 0.0:
                dec = min(-diff, y[i])
                y[i] -= dec
                diff += dec
    return y
```

    h = start_bin.copy()
    best_c5 = compute_c5(h.astype(float))

import time
import numpy as np
from scipy.optimize import differential_evolution

# Global placeholders for DE objective (avoid closures)
_DE_N = None
_DE_TARGET_SUM = None

def compute_c5(arr):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_sum(x, n, target):
    """Clip to [0,1] and adjust to satisfy the sum constraint exactly."""
    y = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(y)
    y += diff / n
    y = np.clip(y, 0.0, 1.0)

    residual = target - np.sum(y)
    if abs(residual) > 1e-12:
        # distribute the remaining residual one by one
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            if residual > 0.0:
                inc = min(residual, 1.0 - y[i])
                y[i] += inc
                residual -= inc
            else:
                dec = min(-residual, y[i])
                y[i] -= dec
                residual += dec
    return y

def de_objective(x):
    """Objective for differential evolution: C5 after repairing sum."""
    y = repair_sum(x, _DE_N, _DE_TARGET_SUM)
    return compute_c5(y)

def binary_hill_climb(start_bin, time_limit, rng):
    """Swap‑based hill climbing on binary vectors (exactly n/2 ones)."""
    n = len(start_bin)
    h = start_bin.copy()
    best_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    no_improve = 0
    while time.time() < time_limit and no_improve < 5000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial feasible vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target_sum = n / 2.0

    # Ensure exact feasibility of the starting vector
    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = np.sum(h0)
    if abs(cur_sum - target_sum) > 1e-12:
        h0 += (target_sum - cur_sum) / n
        h0 = np.clip(h0, 0.0, 1.0)
        residual = target_sum - np.sum(h0)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -h0[i], 1 - h0[i])
            h0[i] += adj
            residual = target_sum - np.sum(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Prepare DE (global parameters for the objective)
    global _DE_N, _DE_TARGET_SUM
    _DE_N = n
    _DE_TARGET_SUM = target_sum

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 2.0:
        # Allocate most of the remaining time to DE
        max_iter = 500
        popsize = 12
        bounds = [(0.0, 1.0)] * n
        de_seed = int(rng.integers(0, 2**32 - 1))

        result = differential_evolution(
            de_objective,
            bounds,
            strategy='best1bin',
            maxiter=max_iter,
            popsize=popsize,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=de_seed,
            polish=False,
            init='latinhypercube'
        )

        h_de = repair_sum(result.x, n, target_sum)
        c5_de = compute_c5(h_de)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de

    # Final local refinement via binary hill climbing
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 2.0:
        bin_start = (best_h >= 0.5).astype(int)
        time_limit = time.time() + min(remaining, 20.0)
        final_bin, final_c5 = binary_hill_climb(bin_start, time_limit, rng)
        if final_c5 < best_c5:
            best_h = final_bin.astype(float)
            best_c5 = final_c5

    # Final safety: clip and enforce exact sum
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    return best_h, best_c5, n
