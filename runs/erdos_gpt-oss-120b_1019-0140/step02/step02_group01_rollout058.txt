def propose_move(h, rng):
    n = len(h)
    # find candidates
    # pick random i; if h[i] == 1, find j where h[j] >0? Actually we need move from j to i (i gets more, j loses)
    # We'll pick i and j random, then compute possible max_transfer from j to i: if i is receiving (increase) and j is giving (decrease).
    # But we can choose direction based on which has capacity.
    i, j = rng.choice(n, 2, replace=False)
    # compute max amount that can be transferred from j to i while staying in [0,1]:
    max_transfer = min(1 - h[i], h[j])
    if max_transfer <= 0:
        # try opposite direction
        max_transfer = min(1 - h[j], h[i])
        if max_transfer <= 0:
            return None, None, 0
        # we will transfer from i to j
        direction = -1
        # but we will swap i and j such that we always transfer from j to i
        i, j = j, i
    else:
        direction = 1 # from j to i

    # choose delta
    delta = rng.random() * max_transfer
    return i, j, delta

def compute_c5(h, M, dx):
    n = len(h)
    # compute convolution of h and h (reverse? for correlation)
    # autocorrelation via FFT: corr = np.fft.ifft(np.fft.fft(h, n*2) * np.conj(np.fft.fft(h, n*2))).real
    # length = 2n-1 (taking first 2n-1)
    # actually compute using np.fft.fft with size = 2*n
    f = np.fft.fft(h, 2*n)
    corr = np.fft.ifft(f * np.conj(f)).real
    # corr length 2*n, we need relevant part: indices 0..2n-2 correspond to lags -(n-1)..(n-1)
    # shift zero at index n-1
    # extract inner = corr[:2*n-1]
    inner = corr[:2*n-1]
    # Use dot product sum = M, so overlap = M - inner
    # C5 = max(overlap) * dx = (M - min_inner) * dx
    # Exclude lag 0 (index n-1)
    # Use mask:
    mask = np.ones_like(inner, dtype=bool)
    mask[n-1] = False
    min_inner = np.min(inner[mask])
    return (M - min_inner) * dx

def anneal_optimize(n, time_limit, rng):

    dx = 2.0/n
    # initial h: if initial_h_values exists and length matches n: use it else generate random binary

        cur = np.sum(h)

def propose_move(h, rng):
    n = len(h)
    i, j = rng.choice(n, 2, replace=False)
    # potential transfer from j to i
    max_transfer = min(1.0 - h[i], h[j])
    if max_transfer > 0:
        delta = rng.random() * max_transfer
        return i, j, delta
    # else try opposite direction
    max_transfer = min(1.0 - h[j], h[i])
    if max_transfer > 0:
        delta = rng.random() * max_transfer
        return j, i, delta
    # no transfer possible
    return i, j, 0.0

def anneal_optimize_with_initial(h0, time_limit, rng):
    n = len(h0)
    M = n // 2
    dx = 2.0 / n
    cur_h = np.clip(h0, 0.0, 1.0)
    # Project cur_h to sum exactly M via simple water-filling
    diff = M - np.sum(cur_h)
    for _ in range(5):
        cur_h += diff / n
        np.clip(cur_h, 0, 1, out=cur_h)
        diff = M - np.sum(cur_h)
        if abs(diff) < 1e-12:
            break

    cur_c5 = compute_c5(cur_h, M, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    T0 = 0.001  # initial temperature
    # loops
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # propose move
        i, j, delta = propose_move(cur_h, rng)
        if delta == 0.0:
            continue
        h_new = cur_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new, M, dx)
        if c5_new <= cur_c5:
            accept = True
        else:
            T = T0 * max(0.0, 1.0 - elapsed / time_limit)
            # Avoid division by zero
            prob = np.exp(-(c5_new - cur_c5) / (T + 1e-12))
            accept = rng.random() < prob
        if accept:
            cur_h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
    return best_h, best_c5

import time
import numpy as np

# ---------- Core evaluation ----------
def compute_c5(h, M, dx):
    """
    Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx for the discretized vector h.
    Using the identity C5 = (M - min_corr) * dx, where
    min_corr = min_{k≠0} Σ_i h_i * h_{i+k} (linear correlation, zero‑padded).
    """
    n = len(h)
    # linear correlation of h with itself
    corr = np.correlate(h, h, mode='full')
    # mask out the zero shift (index n-1)
    mask = np.ones_like(corr, dtype=bool)
    mask[n - 1] = False
    min_corr = np.min(corr[mask])
    return (M - min_corr) * dx

# ---------- Move proposal ----------
def propose_move(h, rng):
    """Return a feasible (i, j, delta) moving mass from j to i."""
    n = len(h)
    i, j = rng.choice(n, 2, replace=False)
    # try moving from j to i
    max_transfer = min(1.0 - h[i], h[j])
    if max_transfer > 0.0:
        delta = rng.random() * max_transfer
        return i, j, delta
    # otherwise try opposite direction
    max_transfer = min(1.0 - h[j], h[i])
    if max_transfer > 0.0:
        delta = rng.random() * max_transfer
        # swap roles so we always add to the first index
        return j, i, delta
    return i, j, 0.0

# ---------- Deterministic smooth start ----------
def initial_h_sin2(n):
    """Smooth start: h_i = sin^2(pi*i/(n-1)), projected onto sum=n/2."""
    x = np.arange(n, dtype=float)
    h = np.sin(np.pi * x / (n - 1)) ** 2
    M = n // 2
    diff = M - np.sum(h)
    for _ in range(5):
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        diff = M - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return h

# ---------- Simulated annealing ----------
def anneal_optimize_with_initial(h0, time_limit, rng):
    """
    Simulated‑annealing optimisation starting from h0.
    Returns best discovered (h, C5) within time_limit seconds.
    """
    n = len(h0)
    M = n // 2
    dx = 2.0 / n
    # ensure feasibility
    h = np.clip(h0, 0.0, 1.0)
    diff = M - np.sum(h)
    for _ in range(5):
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        diff = M - np.sum(h)
        if abs(diff) < 1e-12:
            break

    cur_h = h
    cur_c5 = compute_c5(cur_h, M, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    T0 = 0.005  # initial temperature

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break

        i, j, delta = propose_move(cur_h, rng)
        if delta == 0.0:
            continue

        h_new = cur_h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        c5_new = compute_c5(h_new, M, dx)

        if c5_new <= cur_c5:
            accept = True
        else:
            T = T0 * max(0.0, 1.0 - elapsed / time_limit)
            prob = np.exp(-(c5_new - cur_c5) / (T + 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()

    return best_h, best_c5

# ---------- Main entry point ----------
