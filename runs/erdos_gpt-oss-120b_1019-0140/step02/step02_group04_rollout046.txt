def compute_gradient(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # part from derivative of h_i term
        g[:n - k] += 1 - h[k:]
        # part from derivative of h_{i+k} term (negative)
        g[k:] -= h[:n - k]
    else:
        m = -k
        # part from h_i term, i >= m
        g[m:] += 1 - h[:n - m]
        # part from h_{i+m} term (negative)
        g[:n - m] -= h[m:]
    return g

import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible vector (values in [0,1] with exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def project_to_box_sum(v, target_sum, tol=1e-12, max_iter=60):
    """Projection onto [0,1]^n ∩ {sum = target_sum}."""
    n = v.shape[0]
    lower = -np.max(v)          # all entries become ≤0 → clipped to 0
    upper = 1.0 - np.min(v)     # all entries become ≥1 → clipped to 1
    for _ in range(max_iter):
        mid = (lower + upper) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(w)
        if abs(s - target_sum) <= tol:
            return w
        if s < target_sum:
            lower = mid
        else:
            upper = mid
    return np.clip(v + (lower + upper) * 0.5, 0.0, 1.0)


def compute_gradient(h, k):
    """Gradient of overlap O_k = Σ h_i (1 - h_{i+k}) w.r.t. h, with zero padding."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        # term from h_i * (1 - h_{i+k})
        g[: n - k] += 1.0 - h[k:]
        # term from -h_{i-k} * h_i
        g[k:] -= h[: n - k]
    else:
        m = -k
        g[m:] += 1.0 - h[: n - m]
        g[: n - m] -= h[m:]
    return g


def subgradient_descent(h0, dx, target_sum, time_limit, rng):
    """Subgradient descent on the max-overlap objective."""
    h = h0.copy()
    val = compute_c5_vec(h, dx)
    step = 0.05
    max_step = 0.5
    min_step = 1e-8
    n = h.shape[0]
    stagnation = 0
    while time.time() < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        g = compute_gradient(h, k)
        cand = h - step * g
        cand = project_to_box_sum(cand, target_sum)
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < val - 1e-12:
            h = cand
            val = cand_val
            step = min(step * 1.1, max_step)
            stagnation = 0
            if val < 0.3808:
                break
        else:
            step = max(step * 0.9, min_step)
            stagnation += 1
            if stagnation > 400:
                # random perturbation to escape flat regions
                pert = rng.normal(scale=0.02, size=n)
                cand = h + pert
                cand = project_to_box_sum(cand, target_sum)
                cand_val = compute_c5_vec(cand, dx)
                if cand_val < val:
                    h = cand
                    val = cand_val
                stagnation = 0
    return h, val

        top_idx = np.argsort(-best_h)[: int(target_sum)]
        binary_h[top_idx] = 1
        cur_val = compute_c5_vec(binary_h.astype(float), dx)
        cur_bin = binary_h.copy()
        max_iter = int(min(1e5, remaining * 2000))
        for _ in range(max_iter):
            ones = np.flatnonzero(cur_bin)
            zeros = np.flatnonzero(1 - cur_bin)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cand = cur_bin.copy()
            cand[i] = 0
            cand[j] = 1
            cand_val = compute_c5_vec(cand.astype(float), dx)
            if cand_val < cur_val - 1e-12:
                cur_bin = cand
                cur_val = cand_val
                if cur_val < best_val:
                    best_h = cur_bin.astype(float)
                    best_val = cur_val
                if cur_val < 0.3808:
                    break

    # Final safety projection and evaluation
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
