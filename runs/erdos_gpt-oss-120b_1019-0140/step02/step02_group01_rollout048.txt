def project_onto_box_simplex(v, M):
    # v may violate bounds.
    # Step 1: clip to [0,1]
    v_clipped = np.clip(v, 0, 1)
    # Step 2: compute excess = sum(v_clipped) - M
    excess = np.sum(v_clipped) - M
    if abs(excess) < 1e-10:
        return v_clipped
    # If excess > 0, need to reduce values; if excess <0 need to increase values.
    # Reduce (or increase) uniformly among indices not at bound (0 or 1).
    # However uniform adjustment may push some variables beyond bound; iterate until convergence.
    # Use while loop.
    while True:
        free = (v_clipped > 0) & (v_clipped < 1)
        n_free = np.sum(free)
        if n_free == 0:
            # cannot adjust further, force adjust with tolerance/random?
            break
        delta = excess / n_free
        v_clipped[free] -= delta
        # clip again
        v_clipped = np.clip(v_clipped, 0, 1)
        new_excess = np.sum(v_clipped) - M
        if abs(new_excess - excess) < 1e-12:
            # Should converge
            pass
        if abs(new_excess) < 1e-9:
            return v_clipped
        excess = new_excess
    # final fallback: scale linearly?
    # but should not happen
    return v_clipped

def project_onto_box_simplex(v, M):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Compute excess
    excess = np.sum(v) - M
    if abs(excess) < 1e-12:
        return v
    # If excess > 0: need to reduce sum; else increase sum
    # Use iterative redistribution among free variables (0 < v < 1)
    max_iter = 100
    for _ in range(max_iter):
        free = (v > 0) & (v < 1)
        n_free = np.sum(free)
        if n_free == 0:
            break
        delta = excess / n_free
        v[free] -= delta
        # clip again
        v = np.clip(v, 0.0, 1.0)
        new_excess = np.sum(v) - M
        if abs(new_excess - excess) < 1e-12:
            # No change; break
            break
        excess = new_excess
        if abs(excess) < 1e-12:
            break
    return v

import time
import numpy as np

# ---------- Core evaluation ----------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# ---------- Gradient for a given shift ----------
def gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the overlap for shift k (ignoring dx factor)."""
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        i = np.arange(0, n - k)
        g[i] += (1.0 - h[i + k])
        i = np.arange(k, n)
        g[i] += -h[i - k]
    else:
        m = -k
        i = np.arange(m, n)
        g[i] += (1.0 - h[i + k])          # i+k == i-m
        i = np.arange(0, n - m)
        g[i] += -h[i - k]                # i-k == i+m
    return g

# ---------- Projection onto [0,1] simplex ----------
def project_onto_box_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {0≤x≤1, sum x = target_sum}."""
    x = np.clip(v, 0.0, 1.0)
    excess = np.sum(x) - target_sum
    if abs(excess) < 1e-12:
        return x
    for _ in range(100):
        free = (x > 0.0) & (x < 1.0)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = excess / n_free
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
        new_excess = np.sum(x) - target_sum
        if abs(new_excess - excess) < 1e-12:
            break
        excess = new_excess
        if abs(excess) < 1e-12:
            break
    return x

                    i, j = j, i
                delta = rng.random() * max_tf
                h[i] -= delta
                h[j] += delta
                new_c5 = compute_c5(h)
                if new_c5 <= c5 + 1e-12:
                    c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5, best_h = new_c5, h.copy()
                        if best_c5 < 0.3808:
                            return best_h, best_c5
                else:
                    h[i] += delta
                    h[j] -= delta

    return best_h, best_c5

# ---------- Main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate discretisations (even numbers, modest size)
    candidate_ns = [80, 96, 112, 128, 144, 160]

    # prioritize parent size if available
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        # allocate at most 250 s per resolution, otherwise half of what remains
        alloc = min(250.0, remaining * 0.5)
        h_opt, c5_opt = optimise_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # fallback to a simple feasible candidate if nothing succeeded
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)

    # final safety projection (enforce bounds and exact sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_n // 2
    if abs(best_h.sum() - target_sum) > 1e-9:
        best_h = project_onto_box_simplex(best_h, target_sum)

    return best_h, best_c5, best_n
