import time, numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_and_bounds(h, target_sum):
    # ensure bounds [0,1] and sum exactly target_sum via simple correction
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # distribute diff uniformly to entries not at bound
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        # residual fix
        cur = np.sum(h)
        diff = target_sum - cur
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_and_fix(h, target_sum):
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode='same')
    sm = np.clip(sm, 0.0, 1.0)
    sm = project_sum_and_bounds(sm, target_sum)
    return sm

    target_sum = n / 2.0

    # compute initial C5
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    # precompute indices sets for speed

    # we will store arrays of indices of ones and zeros each iteration
    # but we can update them after each move.
    # Determine temperature schedule

    # limit for acceptance probability when T=0

        if ones.size == 0 or zeros.size == 0:

        i = rng.choice(ones)
        j = rng.choice(zeros)

import time
import numpy as np

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx on the grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_and_bounds(h, target):
    """Clip to [0,1] and enforce sum == target (up to tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        # residual fix
        cur = np.sum(h)
        diff = target - cur
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_and_fix(h, target):
    """Gaussian‑type smoothing (3‑point) with sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return project_sum_and_bounds(sm, target)

def gradient_step(h, target):
    """Single gradient‑like move aimed at the worst shift."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)          # shift giving maximal overlap
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = dx * (1.0 - neigh)       # heuristic gradient
    delta = -(grad - np.mean(grad))
    max_step = 0.2
    mabs = np.max(np.abs(delta))
    if mabs > max_step:
        delta *= max_step / mabs
    cand = h + delta
    cand = np.clip(cand, 0.0, 1.0)
    return project_sum_and_bounds(cand, target)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.size
        # binarise (values are already near 0/1)
        base_bin = (base >= 0.5).astype(int)
        up_factor = 3                     # enlarge the grid (still < 1000 points)
        n = n0 * up_factor
        h = np.repeat(base_bin, up_factor).astype(float)
    else:
        n = 200
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    target_sum = n / 2.0
    h = project_sum_and_bounds(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    # ----- simulated annealing (binary swaps) ------------------------
    T0 = 0.02
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.85:
            break
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        # choose a random 1 and a random 0 to swap
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:

        i = rng.choice(ones)
        j = rng.choice(zeros)

        cand = best_h.copy()

        new_val = compute_c5(cand)

        accept = False

        elif T > 0:

                accept = True
        if accept:

            best_val = new_val
            if best_val < 0.3808:
                break
