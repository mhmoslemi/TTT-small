import cvxpy as cp
import numpy as np
import time

def solve_milp_binary(n):
    # n must be even else can't have sum=n/2 integer
    target = n // 2
    m = n * (n - 1) // 2  # number of unordered pairs
    # mapping pair index to (i, j)
    pairs = []
    for i in range(n):
        for j in range(i+1, n):
            pairs.append((i, j))
    # create variables
    h = cp.Variable(n, boolean=True)
    w = cp.Variable(m, boolean=True)
    t = cp.Variable()
    constraints = []
    # sum constraint
    constraints.append(cp.sum(h) == target)
    # pair linking constraints
    for idx, (i, j) in enumerate(pairs):

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur_sum = np.sum(h)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return h
    # distribute diff uniformly, then clip
    h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    # residual fix by per-element adjustments
    cur_sum = np.sum(h)
    residual = target_sum - cur_sum
    # iterate to correct residual
    for i in range(len(h)):
        if abs(residual) < 1e-12:
            break
        adj = np.clip(residual, -h[i], 1 - h[i])
        h[i] += adj
        residual = target_sum - np.sum(h)
    return h

def solve_milp_h(n, time_limit, rng):
    target = n // 2
    m = n*(n-1)//2
    pairs = []
    for i in range(n):
        for j in range(i+1, n):
            pairs.append((i,j))
    # cvxpy definitions
    h = cp.Variable(n, boolean=True)
    w = cp.Variable(m, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == target]
    # linking constraints
    for idx, (i,j) in enumerate(pairs):
        constraints.append(w[idx] <= h[i])
        constraints.append(w[idx] <= h[j])
        constraints.append(w[idx] >= h[i] + h[j] - 1)
    # precompute masks for distances
    dist_to_w_idxs = {}
    for idx, (i,j) in enumerate(pairs):
        d = j - i
        dist_to_w_idxs.setdefault(d, []).append(idx)
    # shift constraints (k from 1..n-1 and -k)
    for k in range(1, n):
        abs_k = k
        # overlapping indices for positive shift
        mask_pos = np.zeros(n)
        mask_pos[:n-abs_k] = 1
        # overlapping indices for negative shift
        mask_neg = np.zeros(n)
        mask_neg[abs_k:] = 1
        # w mask for distance abs_k
        w_mask = np.zeros(m)
        w_mask[dist_to_w_idxs.get(abs_k, [])] = 1
        # constraints: t >= sum_i mask_pos[i]*h_i - sum w_mask*w
        constraints.append(t >= cp.sum(cp.multiply(mask_pos, h)) - cp.sum(cp.multiply(w_mask, w)))
        # negative shift
        constraints.append(t >= cp.sum(cp.multiply(mask_neg, h)) - cp.sum(cp.multiply(w_mask, w)))
    # objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    # solve with time limit
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, max_seconds=time_limit, verbose=False)
        except Exception:
            return None, None
    if h.value is None:
        return None, None
    # round to binary
    h_val = np.rint(h.value).astype(int)
    # Ensure correct sum; if not, repair
    cur_sum = np.sum(h_val)
    if cur_sum != target:
        # adjust by flipping bits
        diff = target - cur_sum
        # if diff>0, need to turn some zeros to ones
        zeros = np.where(h_val == 0)[0]
        ones = np.where(h_val == 1)[0]
        if diff > 0:
            if len(zeros) >= diff:
                idxs = rng.choice(zeros, diff, replace=False)
                h_val[idxs] = 1
        elif diff < 0:
            diff = -diff
            if len(ones) >= diff:
                idxs = rng.choice(ones, diff, replace=False)
                h_val[idxs] = 0
    # compute c5
    c5 = compute_c5(h_val.astype(float))
    return h_val, c5

import time
import numpy as np

# optional cvxpy for a MILP stage
try:
    import cvxpy as cp
    _cvxpy_available = True
except Exception:
    _cvxpy_available = False

def compute_c5(arr):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def _repair(h, target):
    """Clip to [0,1] and enforce exact sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # distribute uniformly, then fix residual
    h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    for i in range(len(h)):
        if abs(residual) < 1e-12:
            break
        adj = np.clip(residual, -h[i], 1 - h[i])
        h[i] += adj
        residual = target - np.sum(h)
    return h

def _solve_milp_binary(n, time_limit, rng):
    """MILP: binary h with sum=n/2 minimizing max overlap (t)."""
    if n % 2 != 0:
        return None, None
    target = n // 2
    m = n * (n - 1) // 2
    # pair list
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    # distance -> list of indices in w
    dist_to_idx = {}
    for idx, (i, j) in enumerate(pairs):
        d = j - i
        dist_to_idx.setdefault(d, []).append(idx)

    # variables
    h = cp.Variable(n, boolean=True)
    w = cp.Variable(m, boolean=True)
    t = cp.Variable()

    constraints = [cp.sum(h) == target]

    # linking constraints
    for idx, (i, j) in enumerate(pairs):
        constraints.append(w[idx] <= h[i])
        constraints.append(w[idx] <= h[j])
        constraints.append(w[idx] >= h[i] + h[j] - 1)

    # shift constraints (exclude k=0)
    for k in range(1, n):
        d = k
        # masks for h
        mask_pos = np.zeros(n)
        mask_pos[:n - d] = 1
        mask_neg = np.zeros(n)
        mask_neg[d:] = 1
        # mask for w
        wmask = np.zeros(m)
        wmask[dist_to_idx.get(d, [])] = 1
        constraints.append(t >= cp.sum(mask_pos * h) - cp.sum(wmask * w))
        constraints.append(t >= cp.sum(mask_neg * h) - cp.sum(wmask * w))

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try CBC first (supports max_seconds), fallback to GLPK_MI
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
        except Exception:
            return None, None

    if h.value is None:
        return None, None

    h_bin = np.rint(h.value).astype(int)
    # adjust exact sum if needed
    cur_sum = h_bin.sum()
    if cur_sum != target:
        diff = target - cur_sum
        zeros = np.where(h_bin == 0)[0]
        ones = np.where(h_bin == 1)[0]
        if diff > 0 and len(zeros) >= diff:
            idxs = rng.choice(zeros, diff, replace=False)
            h_bin[idxs] = 1
        elif diff < 0 and len(ones) >= -diff:
            idxs = rng.choice(ones, -diff, replace=False)
            h_bin[idxs] = 0

    c5 = compute_c5(h_bin.astype(float))
    return h_bin, c5

def _hill_climb_binary(start_bin, time_end, rng):
    """Simple hill‑climbing on binary vectors (swap moves)."""
    h = start_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h.astype(float))
    n = len(h)
    no_improve = 0
    while time.time() < time_end and no_improve < 3000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_h = cand.copy()
            best_c5 = cand_c5
            h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5

def _continuous_anneal(start_h, time_end, rng):
    """Simulated annealing on continuous vars (keeps total sum)."""
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 200000
    it = 0
    while time.time() < time_end and it < max_iter:
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < np.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # final exact‑sum correction
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target_sum = n / 2.0
    best_h = _repair(h0, target_sum)
    best_c5 = compute_c5(best_h)

    # MILP stage (if time permits)
    time_elapsed = time.time() - start_time
    remaining = budget_s - time_elapsed
    if _cvxpy_available and remaining > 30:
        milp_time = min(remaining * 0.5, 200.0)
        h_bin, c5_bin = _solve_milp_binary(n, milp_time, rng)
        if h_bin is not None and c5_bin < best_c5:
            best_h = h_bin.astype(float)
            best_c5 = c5_bin

    # Local improvement stage
    while True:
        time_elapsed = time.time() - start_time
        if time_elapsed >= budget_s - 0.3:
            break
        chunk = min(5.0, (budget_s - time_elapsed) * 0.4)
        end_time = time.time() + chunk

        # pick a start for hill climbing
        if rng.random() < 0.4:
            start_bin = np.zeros(n, dtype=int)
            start_bin[rng.choice(n, n // 2, replace=False)] = 1
        else:
            start_bin = (best_h >= 0.5).astype(int)

        cand_h, cand_c5 = _hill_climb_binary(start_bin, end_time, rng)
        if cand_c5 < best_c5:
            best_h = cand_h.copy()
            best_c5 = cand_c5

        # occasional annealing from the best found so far
        if time.time() < start_time + budget_s - 0.5 and rng.random() < 0.2:
            cand_f, cand_c5_f = _continuous_anneal(best_h, start_time + budget_s - 0.5, rng)
            if cand_c5_f < best_c5:
                best_h = cand_f.copy()
                best_c5 = cand_c5_f

        if best_c5 <= 0.3808:
            break

    # final feasibility enforcement
    best_h = _repair(best_h, target_sum)
    best_c5 = compute_c5(best_h)
