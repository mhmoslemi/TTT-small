def repair_sum(x):
    n = len(x)
    target_sum = n/2.0
    # First clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    current_sum = np.sum(x)
    # Compute residual to add across all entries
    residual = target_sum - current_sum
    # Add evenly
    x += residual / n
    # Clip again, then handle any leftover residual
    x = np.clip(x, 0.0, 1.0)
    # Resolve any remaining residual via water-filling
    residual = target_sum - np.sum(x)
    # while residual not negligible:
    while abs(residual) > 1e-12:
        # find indices that can absorb residual:
        if residual > 0:
            # can increase up to 1
            inc = 1.0 - x
            total_inc = np.sum(inc)
            if total_inc < 1e-12:
                break
            delta = min(residual, total_inc)
            # proportionally distribute
            x += inc * (delta / total_inc)
        else:
            # residual <0, need to decrease values
            dec = x
            total_dec = np.sum(dec)
            if total_dec < 1e-12:
                break
            delta = min(-residual, total_dec)
            x -= dec * (delta / total_dec)
        residual = target_sum - np.sum(x)
    return x
```
