        j = rng.choice(recipients)

    """Project vec into [0,1] and enforce exact sum == target_sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec


def random_binary(n, target_sum, rng):
    """Generate a binary vector with exactly target_sum ones."""
    vec = np.zeros(n, dtype=int)
    ones = rng.choice(n, size=int(target_sum), replace=False)
    vec[ones] = 1
    return vec


def init_binary_from_initial(n, target_sum, rng):
    """If initial_h_values exists, return a binary start by taking top entries."""
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if len(h0) == n:
            bin_vec = np.zeros(n, dtype=int)
            top = np.argsort(-h0)[: int(target_sum)]
            bin_vec[top] = 1
            return bin_vec
    return None


# ---------- Simulated annealing on binary vectors ----------
def simulated_annealing_binary(rng, n, target_sum, dx, time_limit, start_time):
    cur = init_binary_from_initial(n, target_sum, rng)
    if cur is None:
        cur = random_binary(n, target_sum, rng)
    cur_val = compute_c5_vec(cur.astype(float), dx)

    best = cur.copy()
    best_val = cur_val

    T0 = 0.02
    T_end = 1e-4
    it = 0
    last_improve = 0
    restart_interval = 20000

    while time.time() < time_limit:
        it += 1
        # temperature schedule (log‑linear)
        progress = (time.time() - start_time) / (time_limit - start_time)
        T = T0 * (T_end / T0) ** progress

        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)
        if ones.size == 0 or zeros.size == 0:
            cur = random_binary(n, target_sum, rng)
            cur_val = compute_c5_vec(cur.astype(float), dx)
            continue

        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)

        if cand_val < cur_val or rng.random() < np.exp(-(cand_val - cur_val) / T):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best = cand.copy()
                best_val = cand_val
                last_improve = it
                if best_val < 0.3808:
                    break

        if it - last_improve > restart_interval:
            cur = random_binary(n, target_sum, rng)
            cur_val = compute_c5_vec(cur.astype(float), dx)
            last_improve = it

    return best.astype(float), best_val


# ---------- Local continuous refinement ----------
def refine_continuous(h, dx, target_sum, time_limit, start_time, rng):
    cur = h.copy()
    cur_val = compute_c5_vec(cur, dx)
    step = 0.05
    n = len(h)

    while time.time() < time_limit:
        donors = np.where(cur > 1e-12)[0]
        recips = np.where(cur < 1.0 - 1e-12)[0]
        if donors.size == 0 or recips.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recips)
        max_delta = min(step, cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.99

    cur = project_to_sum(cur, target_sum)
    cur_val = compute_c5_vec(cur, dx)
    return cur, cur_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 80
    target_sum = n / 2.0
    dx = 2.0 / n

    # placeholder for best solution
    best_h = np.zeros(n, dtype=float)
    best_val = np.inf

    # ---- binary simulated annealing (≈80% of budget) ----
    sa_deadline = start + budget_s * 0.8
    bin_h, bin_val = simulated_annealing_binary(rng, n, target_sum, dx, sa_deadline, start)
    if bin_val < best_val:
        best_h = bin_h
        best_val = bin_val

    # ---- continuous refinement with remaining time ----
    remaining = start + budget_s - time.time()
    if remaining > 2.0:
        cont_deadline = start + budget_s
        cont_h, cont_val = refine_continuous(best_h, dx, target_sum,
                                             cont_deadline, start, rng)
        if cont_val < best_val:
            best_h = cont_h
            best_val = cont_val

    # final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
