def project_to_sum(x, target):
    lo = -1.0
    hi = 1.0
    # run for about 50 iterations
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    # Use hi as final shift to ensure sum >= target
    y = np.clip(x + hi, 0.0, 1.0)
    # Adjust any tiny mismatch by distributing residual evenly over non-bound entries
    residual = target - y.sum()
    if abs(residual) > 1e-12:
        # distribute only to entries not at bounds
        mask = (y > 1e-12) & (y < 1 - 1e-12)
        free = mask.sum()
        if free > 0:
            y[mask] += residual / free
            # clip again
            y = np.clip(y, 0.0, 1.0)
        else:
            # cannot adjust; just shift again using small lambda
            y += residual / len(y)
            y = np.clip(y, 0.0, 1.0)
    return y

import time
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(arr):
    """Approximate C5 via full correlation."""
    corr = np.correlate(arr, 1 - arr, mode="full")
    return np.max(corr) * (2.0 / len(arr))

def thue_morse_seq(n):
    """Binary Thue‑Morse of length n."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=int, count=n)

def rudin_shapiro_seq(n):
    """Binary Rudin‑Shapiro (+1/−1 mapped to 1/0)."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def project_to_sum(x, target):
    """Project x∈[0,1]^n onto the hyperplane sum=x_target by uniform shift + clipping."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + hi, 0.0, 1.0)
    residual = target - y.sum()
    if abs(residual) > 1e-12:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        free = mask.sum()
        if free:
            y[mask] += residual / free
            y = np.clip(y, 0.0, 1.0)
        else:
            y += residual / len(y)
            y = np.clip(y, 0.0, 1.0)
    return y

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # dimension
    n = len(initial_h_values) if 'initial_h_values' in globals() else 80
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    # ------------------------------------------------------------------
    # Build a small seed set for the initial DE population
    seeds = []
    def add_seed(v):
        v = np.array(v, dtype=float)
        seeds.append(project_to_sum(v, target_sum))

    # binary random half‑ones
    bin_rand = np.zeros(n, dtype=float)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    add_seed(bin_rand)

    # Thue‑Morse
    add_seed(thue_morse_seq(n).astype(float))

    # Rudin‑Shapiro
    add_seed(rudin_shapiro_seq(n).astype(float))

    # initial_h_values if present
    if 'initial_h_values' in globals():
        add_seed(initial_h_values)

    # a few continuous random seeds
    for _ in range(4):
        add_seed(rng.random(n))

    popsize = max(15, len(seeds))
    while len(seeds) < popsize:
        add_seed(rng.random(n))
    init_pop = np.stack(seeds[:popsize])

    # ------------------------------------------------------------------
    best_h = None
    best_c5 = np.inf

    def objective(x):
        h = project_to_sum(x, target_sum)
        return compute_c5(h)

    def callback(xk, convergence):
        nonlocal best_h, best_c5
        h = project_to_sum(xk, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        if best_c5 <= 0.3808:
            return True
        if time.time() - start_time > budget_s * 0.94:
            return True
        return False

    maxiter = 5000
    differential_evolution(
        objective,
        bounds,
        strategy="best1bin",
        maxiter=maxiter,
        popsize=popsize,
        init=init_pop,
        callback=callback,
        polish=False,
        seed=rng,
    )

    if best_h is None:
        best_h = project_to_sum(rng.random(n), target_sum)
        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Simple local refinement: transfer small mass between two positions
    h_ref = best_h.copy()
    cur_c5 = best_c5
    while time.time() - start_time < budget_s * 0.98:
        inc = np.where(h_ref < 1.0 - 1e-12)[0]
        dec = np.where(h_ref > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1.0 - h_ref[i], h_ref[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.3
        h_new = h_ref.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h_ref, cur_c5 = h_new, new_c5
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h_ref.copy()

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)

    return best_h, best_c5, n
