- At j: becomes 1 * (1 - h_{j+k}) = 1 - h_{j+k}.

Thus net change = (new_i + new_j) - (old_i + old_j) = (0 + (1 - h_{j+k})) - ((1 - h_{i+k}) + 0) = (1 - h_{j+k}) - (1 - h_{i+k}) = h_{i+k} - h_{j+k}.

Thus the correlation reduces if h_{i+k} < h_{j+k} (i.e., if the partner at i+k is lower than at j+k). In binary case, h_{i+k} and h_{j+k} are 0/1. So we want case where h_{i+k}=0 and h_{j+k}=1, then delta = 0 -1 = -1 => correlation decreases by 1.

Thus we can look for a pair (i,j) with h_i=1, h_j=0, such that h_{i+k}=0, h_{j+k}=1. Then swapping reduces correlation at that shift by 1. This is a beneficial move.

def milp_optimize(n, target, dx, time_limit):
    # building
    h = cp.Variable(n, boolean=True)
    # generate pairs
    pairs = []
    diff_to_pair_idxs = {d: [] for d in range(1, n)}
    for i in range(n):
        for j in range(i+1, n):
            idx = len(pairs)
            pairs.append((i, j))
            diff_to_pair_idxs[j - i].append(idx)
    w = cp.Variable(len(pairs), boolean=True)
    constraints = [cp.sum(h) == target]
    # linearization constraints
    for idx, (i, j) in enumerate(pairs):
        constraints.append(w[idx] <= h[i])
        constraints.append(w[idx] <= h[j])
        constraints.append(w[idx] >= h[i] + h[j] - 1)
    c = cp.Variable()
    constraints.append(c >= 0)
    # shift constraints
    for k in range(-n+1, n):
        if k == 0:
            continue
        if k > 0:
            sum_h = cp.sum(h[:n - k])
        else:
            sum_h = cp.sum(h[-k:])
        sum_w = cp.sum(w[diff_to_pair_idxs[abs(k)]]) if diff_to_pair_idxs[abs(k)] else 0
        constraints.append(sum_h - sum_w <= c / dx)
    prob = cp.Problem(cp.Minimize(c), constraints)
    # attempt solve with time limit
    try:
        prob.solve(solver=cp.GLPK_MI, msg=False, time_limit=time_limit)
    except TypeError:
        try:
            prob.solve(solver=cp.CBC, msg=False, timeLimit=time_limit)
        except Exception:
            prob.solve(solver=cp.GLPK_MI, msg=False)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
        return None, None
    h_val = np.rint(h.value).astype(float)
    # enforce exact sum
    cur_sum = h_val.sum()
    if cur_sum != target:
        # adjust by flipping bits
        diff = int(target - cur_sum)
        if diff > 0:
            # set some zeros to 1
            zero_idxs = np.where(h_val < 0.5)[0]
            if len(zero_idxs) >= diff:
                h_val[zero_idxs[:diff]] = 1.0
        elif diff < 0:
            one_idxs = np.where(h_val > 0.5)[0]
            if len(one_idxs) >= -diff:
                h_val[one_idxs[: -diff]] = 0.0
    c_val = c.value if c.value is not None else compute_c5(h_val)  # fallback
    return h_val, c_val
```

import time
import numpy as np

def compute_c5(h):
    """Return approximate C5 for a discretized function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

# ---------- MILP optimizer (binary formulation) ----------
def _milp_optimize(n, target, dx, time_limit):
    """Return (h_opt, c_opt) from a MILP, or (None,None) if not solved."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    # binary vector h
    h = cp.Variable(n, boolean=True)

    # enumerate unordered pairs (i<j) and map differences to pair indices
    pairs = []
    diff_to_pair = {d: [] for d in range(1, n)}
    for i in range(n):
        for j in range(i + 1, n):
            idx = len(pairs)
            pairs.append((i, j))
            diff_to_pair[j - i].append(idx)
    w = cp.Variable(len(pairs), boolean=True)  # w_ij = h_i * h_j

    # core constraints
    constraints = [cp.sum(h) == target]

    # linearization of products
    for idx, (i, j) in enumerate(pairs):
        constraints.append(w[idx] <= h[i])
        constraints.append(w[idx] <= h[j])
        constraints.append(w[idx] >= h[i] + h[j] - 1)

    # auxiliary variable for the max correlation bound
    c = cp.Variable()
    constraints.append(c >= 0)

    # correlation constraints for each shift k != 0
    for k in range(-n + 1, n):
        if k == 0:
            continue
        if k > 0:
            sum_h = cp.sum(h[: n - k])
        else:
            sum_h = cp.sum(h[-k:])  # -k is positive
        sum_w = cp.sum(w[diff_to_pair[abs(k)]])
        constraints.append(sum_h - sum_w <= c / dx)

    prob = cp.Problem(cp.Minimize(c), constraints)

    # try solving; most open-source solvers ignore explicit time limits
    try:
        prob.solve(solver=cp.GLPK_MI, msg=False, time_limit=time_limit)
    except TypeError:
        try:
            prob.solve(solver=cp.CBC, msg=False, timeLimit=time_limit)
        except Exception:
            prob.solve(solver=cp.GLPK_MI, msg=False)

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    h_val = np.rint(h.value).astype(float)
    # enforce exact weight if rounding disturbed it
    cur = int(round(h_val.sum()))
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.where(h_val < 0.5)[0]
            if len(zeros) >= diff:
                h_val[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.where(h_val > 0.5)[0]
            if len(ones) >= -diff:
                h_val[ones[: -diff]] = 0.0
    c_val = float(c.value) if c.value is not None else compute_c5(h_val)
    return h_val, c_val

# ---------- Greedy local search ----------
def _greedy_optimize(h, dx, deadline, rng):
    """Improve a binary h by local swaps targeting the worst shift."""
    n = h.shape[0]
    target = n // 2
    best_h = h.copy()
    best_val = compute_c5(best_h)
    no_improve = 0

    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        cur_val = corr[max_idx] * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_val <= 0.3808:
            break
        if no_improve > 300:
            # random perturbation to escape a plateau
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = 0.0, 1.0
            no_improve = 0
            continue

        k = max_idx - (n - 1)  # shift corresponding to max correlation
        if k >= 0:
            i_range = np.arange(n - k)
            partner = i_range + k
        else:
            kk = -k
            i_range = np.arange(kk, n)
            partner = i_range - kk

        # candidates that would reduce the max correlation by swapping
        cand_i = i_range[(h[i_range] > 0.5) & (h[partner] < 0.5)]
        cand_j = i_range[(h[i_range] < 0.5) & (h[partner] > 0.5)]

        if cand_i.size and cand_j.size:
            i = cand_i[0]
            j = cand_j[0]
            h[i], h[j] = 0.0, 1.0
            continue

        # fallback: try a random swap and keep it if it improves the bound
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_val = compute_c5(h)
        if new_val < cur_val:
            continue
        # revert if no improvement
        h[i], h[j] = 1.0, 0.0

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]

    target = n // 2

    # Binarise if necessary and enforce exact weight
    if not np.all(np.logical_or(np.isclose(h0, 0), np.isclose(h0, 1))):
        idx_sorted = np.argsort(-h0)  # descending
        bin_h = np.zeros(n, dtype=float)
        bin_h[idx_sorted[:target]] = 1.0
        h0 = bin_h
    else:
        cur_sum = int(round(h0.sum()))
        if cur_sum > target:
            ones = np.where(h0 > 0.5)[0]
            rng.shuffle(ones)
            for i in ones[target:]:
                h0[i] = 0.0
        elif cur_sum < target:
            zeros = np.where(h0 < 0.5)[0]
            rng.shuffle(zeros)
            for i in zeros[: target - cur_sum]:
                h0[i] = 1.0

    h0 = np.clip(h0, 0.0, 1.0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- try exact MILP (binary) if time allows -----
    remaining = deadline - time.time()
    if remaining > 35.0:
        milp_h, milp_c = _milp_optimize(n, target, dx, max(5.0, remaining - 10.0))
        if milp_h is not None:
            milp_val = compute_c5(milp_h)
            if milp_val < best_val:
                best_h, best_val = milp_h, milp_val

    # ----- greedy local improvement -----
    if time.time() < deadline - 0.5:
        gre_h, gre_val = _greedy_optimize(best_h.copy(), dx, deadline, rng)
        if gre_val < best_val:
            best_h, best_val = gre_h, gre_val

    # ----- final safety adjustments -----
    cur = int(round(best_h.sum()))
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.where(best_h < 0.5)[0]
            best_h[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.where(best_h > 0.5)[0]
            best_h[ones[: -diff]] = 0.0
    best_h = np.clip(best_h, 0.0, 1.0)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
