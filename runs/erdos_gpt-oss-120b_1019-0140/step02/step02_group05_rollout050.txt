def upsample_h(old_h, new_n):
    old_n = len(old_h)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    new_h = np.clip(new_h, 0.0, 1.0)
    # adjust sum exactly
    target_sum = new_n / 2.0
    cur_sum = np.sum(new_h)
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # distribute diff equally among non-boundary positions
        free = np.where((new_h > 0) & (new_h < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            new_h[free] += inc
            new_h = np.clip(new_h, 0.0, 1.0)
        # residual fix loop
        diff = target_sum - np.sum(new_h)

                add = min(diff, 1.0 - new_h[i])

                new_h[i] -= sub
                diff += sub
    new_h = np.clip(new_h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    """Exact discrete C5 for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_and_fix(h, target_sum):

    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")

        for i in range(len(sm)):

            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
        sm = np.clip(sm, 0.0, 1.0)
    return sm

def upsample_h(old_h, new_n):
    """Linear interpolation from old_h (any length) to length new_n."""
    old_n = len(old_h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    new_h = np.clip(new_h, 0.0, 1.0)
    target_sum = new_n / 2.0
    cur_sum = np.sum(new_h)
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        free = np.where((new_h > 0) & (new_h < 1))[0]
        if free.size:
            inc = diff / free.size
            new_h[free] += inc
            new_h = np.clip(new_h, 0.0, 1.0)
        diff = target_sum - np.sum(new_h)
        for i in range(new_n):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and new_h[i] < 1.0:
                add = min(diff, 1.0 - new_h[i])
                new_h[i] += add
                diff -= add
            elif diff < 0 and new_h[i] > 0.0:
                sub = min(-diff, new_h[i])
                new_h[i] -= sub
                diff += sub
        new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----------------------- initialise -----------------------
    n = 200                     # finer grid for better bound
    target_sum = n / 2.0
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        cur_h = upsample_h(h0, n)
    else:
        cur_h = rng.random(n)
        cur_h = np.clip(cur_h, 0.0, 1.0)
        cur_sum = np.sum(cur_h)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            free = np.where((cur_h > 0) & (cur_h < 1))[0]
            if free.size:
                inc = diff / free.size
                cur_h[free] += inc
                cur_h = np.clip(cur_h, 0.0, 1.0)
            diff = target_sum - np.sum(cur_h)
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and cur_h[i] < 1.0:
                    add = min(diff, 1.0 - cur_h[i])
                    cur_h[i] += add
                    diff -= add
                elif diff < 0 and cur_h[i] > 0.0:
                    sub = min(-diff, cur_h[i])
                    cur_h[i] -= sub
                    diff += sub
            cur_h = np.clip(cur_h, 0.0, 1.0)

    best_h = cur_h.copy()
    best_val = compute_c5(best_h)
    cur_val = best_val

    # ------------------- SA parameters ----------------------
    p_swap = 0.25
    p_transfer = 0.25
    p_gradient = 0.25
    p_smooth = 0.15
    p_binary = 0.02
    T0 = 0.02
    no_improve = 0
    max_no_improve = 2000

    # ------------------- simulated annealing ----------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.85:
            break
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)
        r = rng.random()
        if r < p_swap:
            i, j = rng.choice(n, 2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < p_swap + p_transfer:
            i, j = rng.choice(n, 2, replace=False)
            if cur_h[i] > 0 and cur_h[j] < 1:
                donor, recv = i, j
            elif cur_h[j] > 0 and cur_h[i] < 1:
                donor, recv = j, i
            else:
                continue
            max_tr = min(cur_h[donor], 1.0 - cur_h[recv])
            if max_tr <= 0:
                continue
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        elif r < p_swap + p_transfer + p_gradient:
            # gradient move targeting the shift with minimal overlap
            corr_hh = np.correlate(cur_h, cur_h, mode="full")
            center = n - 1
            corr_hh[center] = np.inf
            k_idx = np.argmin(corr_hh)
            k_shift = k_idx - center
            d = np.zeros(n)
            if k_shift >= 0:
                d[: n - k_shift] += cur_h[k_shift:]
                d[k_shift:] += cur_h[: n - k_shift]
            else:
                kk = -k_shift
                d[kk:] += cur_h[: n - kk]
                d[: n - kk] += cur_h[kk:]
            # Choose donor (low d) and receiver (high d) respecting bounds
            sorted_idx = np.argsort(d)
            donor = None
            recv = None
            for idx in sorted_idx:
                if cur_h[idx] > 1e-12:
                    donor = idx
                    break
            for idx in sorted_idx[::-1]:
                if cur_h[idx] < 1.0 - 1e-12:
                    recv = idx
                    break
            if donor is None or recv is None:
                continue
            max_tr = min(cur_h[donor], 1.0 - cur_h[recv])
            if max_tr <= 0:
                continue
            delta = 0.5 * max_tr * rng.random()
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        elif r < p_swap + p_transfer + p_gradient + p_smooth:
            kernel = np.array([0.25, 0.5, 0.25])
            cand = np.convolve(cur_h, kernel, mode="same")
            cand = np.clip(cand, 0.0, 1.0)
            cur_sum = np.sum(cand)
            diff = target_sum - cur_sum
            if abs(diff) > 1e-12:
                free = np.where((cand > 0) & (cand < 1))[0]
                if free.size:
                    inc = diff / free.size
                    cand[free] += inc
                    cand = np.clip(cand, 0.0, 1.0)
                diff = target_sum - np.sum(cand)
                for i2 in range(n):
                    if abs(diff) < 1e-12:
                        break
                    if diff > 0 and cand[i2] < 1.0:
                        add = min(diff, 1.0 - cand[i2])
                        cand[i2] += add
                        diff -= add
                    elif diff < 0 and cand[i2] > 0.0:
                        sub = min(-diff, cand[i2])
                        cand[i2] -= sub
                        diff += sub
                cand = np.clip(cand, 0.0, 1.0)
        else:
            # occasional binary projection move
            half = n // 2
            idx = np.argpartition(cur_h, -half)[-half:]
            cand = np.zeros_like(cur_h)
            cand[idx] = 1.0
            # binary move always respects sum
            # no need for extra correction
        new_val = compute_c5(cand)
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0:
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            cur_h = cand
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break
        else:
            no_improve += 1
        if no_improve >= max_no_improve:
            # gentle restart around the best solution
            cur_h = best_h.copy()
            for _ in range(5):
                i, j = rng.choice(n, 2, replace=False)
                if cur_h[i] > 0 and cur_h[j] < 1:
                    max_tr = min(cur_h[i], 1.0 - cur_h[j])
                    if max_tr > 0:
                        delta = rng.random() * max_tr
                        cur_h[i] -= delta
                        cur_h[j] += delta
            cur_val = compute_c5(cur_h)
            no_improve = 0

    # ------------------- optional smoothing passes ----------
    if time.time() - start < budget_s - 5.0:
        for _ in range(5):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
            else:
                break

    # ------------------- final SLSQP refinement ------------
    remaining = budget_s - (time.time() - start)
    if remaining > 10.0:
        beta = 40.0
        def smooth_obj(x):
            h = x
            corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / len(h))
            return logsumexp(beta * corr) / beta
        def sum_eq(x):
            return np.sum(x) - target_sum
        constraints = [{"type": "eq", "fun": sum_eq}]
        bounds = [(0.0, 1.0)] * n
        x0 = best_h.copy()
        try:
            res = minimize(
                smooth_obj,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x, 0.0, 1.0)
                cur = np.sum(h_opt)
                if abs(cur - target_sum) > 1e-9:
                    h_opt += (target_sum - cur) / n
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # ------------------- final safety correction ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = np.sum(best_h)
    if abs(cur - target_sum) > 1e-9:
        best_h += (target_sum - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, n
