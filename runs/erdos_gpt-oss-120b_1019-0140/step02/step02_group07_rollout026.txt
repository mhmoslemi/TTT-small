Now implement function sample_candidate(mean, sigma, target_sum, rng):

   cand = mean + sigma * rng.standard_normal(mean.shape)

   # project onto target sum hyperplane (average)
   avg = cand.mean()
   cand = cand - (avg - (target_sum / len(mean)))   # target avg = 0.5

   # clip
   cand = np.clip(cand, 0.0, 1.0)

   # adjust sum to target
   total = cand.sum()
   diff = target_sum - total
   if abs(diff) > 1e-12:
       # distribute diff uniformly and clip again
       cand += diff / len(mean)
       cand = np.clip(cand, 0.0, 1.0)
       # residual fix if needed
       total = cand.sum()
       residual = target_sum - total
       if abs(residual) > 1e-12:
           # distribute to indices not at bounds
           # we gather indices where we can move in needed direction
           idxs = np.arange(len(cand))
           # random order
           for i in rng.permutation(idxs):
               if abs(residual) < 1e-12:
                   break
               if residual > 0 and cand[i] < 1.0:
                   capacity = 1.0 - cand[i]
                   delta = min(capacity, residual)
               elif residual < 0 and cand[i] > 0.0:
                   capacity = -cand[i]  # negative
                   delta = max(capacity, residual)
               else:
                    continue
               cand[i] += delta
               residual -= delta
   return cand

import time
import numpy as np

def compute_c5(arr):
    """Approximate C5 = max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def enforce_constraints(vec, target_sum, rng):
    """Clip to [0,1] and adjust to meet exact sum target."""
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - vec.sum()
    if abs(diff) > 1e-12:
        vec += diff / len(vec)
        vec = np.clip(vec, 0.0, 1.0)
        residual = target_sum - vec.sum()
        if abs(residual) > 1e-12:
            for i in rng.permutation(len(vec)):
                if abs(residual) < 1e-12:
                    break
                if residual > 0 and vec[i] < 1.0:
                    cap = 1.0 - vec[i]
                    delta = min(cap, residual)
                elif residual < 0 and vec[i] > 0.0:
                    cap = -vec[i]   # negative capacity
                    delta = max(cap, residual)
                else:
                    continue
                vec[i] += delta
                residual -= delta
    return vec

def random_binary_vector(n, rng):
    """Return a binary vector with exactly n/2 ones."""
    vec = np.zeros(n, dtype=float)
    vec[rng.choice(n, n // 2, replace=False)] = 1.0
    return vec

def sample_candidate(mean_vec, sigma, target_sum, rng):
    """Sample a candidate around mean_vec, then enforce constraints."""
    cand = mean_vec + sigma * rng.standard_normal(mean_vec.shape)
    # project onto hyperplane with correct average (target_sum / n)
    avg = cand.mean()
    cand -= (avg - (target_sum / len(cand)))
    cand = enforce_constraints(cand, target_sum, rng)
    return cand

def sort_by_fitness(pair):
    return pair[1]

def hill_climb_binary(start_bin, time_end, rng):
    """Swap-based hill climbing on binary vectors (keeps sum fixed)."""
    n = len(start_bin)
    best_h = start_bin.copy()
    best_c5 = compute_c5(best_h.astype(float))
    while time.time() < time_end:
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 < best_c5:
            best_h = cand
            best_c5 = cand_c5
    return best_h.astype(float), best_c5
