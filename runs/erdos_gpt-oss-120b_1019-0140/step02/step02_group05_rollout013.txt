def compute_grad_smooth(h, tau):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1 - h, mode='full')
    f = corr * dx  # vector length 2n-1
    # compute weights
    # use stable logsumexp: max_f = np.max(f), then w = np.exp((f - max_f)/tau)
    max_f = np.max(f)
    w = np.exp((f - max_f)/tau)
    w_sum = np.sum(w)
    w = w / w_sum  # these are softmax weights
    # grads
    grad = np.zeros(n)
    # iterate over k and accumulate
    offset = n - 1
    for idx, wk in enumerate(w):
        k = idx - offset
        if wk == 0: continue  # skip
        if k >= 0:
            # part1
            grad[:n-k] += wk * dx * (1 - h[k:])
            # part2
            grad[k:] -= wk * dx * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wk * dx * (1 - h[:n-kk])
            grad[:n-kk] -= wk * dx * h[kk:]
    return grad

def project_onto_sum_and_bounds(v, target, lower=0, upper=1):
    # Solve for lambda such that sum_i min(upper, max(lower, v_i + lambda)) = target
    # monotone increasing in lambda.
    lo = -np.max(v - lower) - 1.0  # ensure lower bound
    hi = np.max(upper - v) + 1.0   # ensure high enough
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, lower, upper)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, lower, upper)

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_sum_and_bounds(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    # bisection on lambda
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, lower, upper)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    proj = np.clip(v + mid, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = np.where((proj > lower + 1e-12) & (proj < upper - 1e-12))[0]
        if free.size:
            inc = diff / free.size
            proj[free] += inc
            proj = np.clip(proj, lower, upper)
    return proj

def compute_grad_smooth(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    # stable softmax
    max_f = np.max(f)
    w = np.exp((f - max_f) / tau)
    w_sum = np.sum(w)
    w = w / w_sum
    grad = np.zeros(n, dtype=float)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            # i in 0..n-1-k
            grad[:n - k] += wk * dx * (1.0 - h[k:])
            # i in k..n-1
            grad[k:] -= wk * dx * h[:n - k]
        else:
            kk = -k
            grad[kk:] += wk * dx * (1.0 - h[:n - kk])
            grad[:n - kk] -= wk * dx * h[kk:]
    return grad

def overlap_shift(h, k, dx):
    n = len(h)
    if k >= 0:
        return dx * np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        return dx * np.dot(h[kk:], 1.0 - h[: n - kk])
