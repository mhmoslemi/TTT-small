def project_sum_box(h, M):
    # h is array
    # First box clip
    h = np.clip(h, 0.0, 1.0)
    # Compute sum
    S = np.sum(h)
    # Simple shift: offset = (M - S) / n

    h += (M - S) / len(h)
    # Clip again
    h = np.clip(h, 0.0, 1.0)

import time
import numpy as np
import math

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    # autocorrelation of h with itself
    corr = np.correlate(h, h, mode='full')
    center = n - 1
    # exclude zero shift
    min_corr = np.min(np.concatenate([corr[:center], corr[center + 1:]]))
    return 1.0 - min_corr * dx

def project_sum_box(h, target_sum):
    """Project onto {0≤h≤1, sum(h)=target_sum} by simple water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        s = np.sum(h)
        if abs(s - target_sum) <= 1e-12:
            break
        delta = (target_sum - s) / h.size
        h += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def linear_initial_h(n, M):
    """Deterministic decreasing linear shape, scaled to meet sum=M."""
    h = np.linspace(1.0, 0.0, n)
    scaling = M / np.sum(h)
    h = h * scaling
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) > 1e-12:
        h = project_sum_box(h, M)
    return h

def gradient_ascent(h0, M, max_iter, time_limit, rng):
    """Increase the minimal autocorrelation via projected gradient ascent."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    eta0 = 0.05

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # autocorrelation
        corr = np.correlate(h, h, mode='full')
        center = n - 1
        corr[center] = np.inf  # ignore zero shift
        idx_min = int(np.argmin(corr))
        shift = idx_min - center

        # gradient of the chosen shift
        g = np.zeros_like(h)
        if shift >= 0:
            sh = shift
            g[:n - sh] += h[sh:]
            g[sh:] += h[:n - sh]
        else:
            ks = -shift
            g[ks:] += h[:n - ks]
            g[:n - ks] += h[ks:]

        eta = eta0 / math.sqrt(1.0 + it)
        h_new = h + eta * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum_box(h_new, M)

        c5_new = compute_c5(h_new)
        c5_curr = compute_c5(h)

        # accept if not worse (tiny tolerance) or with small probability
        if c5_new <= c5_curr + 1e-12:
            h = h_new
        else:
            temp = 0.0005 + it * 1e-5
            if rng.random() < math.exp(-(c5_new - c5_curr) / temp):
                h = h_new

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5

def simulated_annealing(h0, M, time_budget, rng):
    """Refine a solution by random small mass transfers (SA style)."""
    n = h0.shape[0]
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    h = best_h.copy()
    start = time.time()

    while time.time() - start < time_budget:
        # pick source (j) and destination (i)
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        c5_new = compute_c5(h_new)
        elapsed = time.time() - start
        # linear cooling from ~0.01 to 0
        T = max(1e-5, (time_budget - elapsed) / time_budget) * 0.01

        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_new
            h = h_new
        else:
            c5_curr = compute_c5(h)
            if rng.random() < math.exp(-(c5_new - c5_curr) / T):
                h = h_new

    return best_h, best_c5

def optimise_n(n, rng, time_budget):
    """Find a good h for a given discretisation size."""
    M = n // 2
    best_h = None
    best_c5 = np.inf
    start = time.time()

    # first attempt: linear shape (or parent if available)
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h_start = np.array(initial_h_values, dtype=float)
        cur = np.sum(h_start)
        if abs(cur - M) > 1e-12:
            h_start = project_sum_box(h_start, M)
    else:
        h_start = linear_initial_h(n, M)

    # run gradient ascent on the first start
    remaining = time_budget - (time.time() - start)
    if remaining > 1.0:
        slice_time = min(remaining * 0.6, 120.0)
        h_opt, c5_opt = gradient_ascent(h_start, M, max_iter=20000,
                                        time_limit=slice_time, rng=rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # remaining budget for random restarts
    while True:
        elapsed = time.time() - start
        if elapsed > time_budget - 5.0:
            break
        remaining = time_budget - elapsed
        # random binary start
        h_rand = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h_rand[idx] = 1.0
        sub_budget = min(30.0, remaining * 0.4)
        h_opt, c5_opt = gradient_ascent(h_rand, M, max_iter=8000,
                                        time_limit=sub_budget, rng=rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if best_c5 < 0.3808:
            break

    # final SA refinement within whatever time remains
    remaining = time_budget - (time.time() - start)
    if remaining > 1.0 and best_h is not None:
        h_sa, c5_sa = simulated_annealing(best_h, M, remaining, rng)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa, c5_sa

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # candidate even sizes, prefer parent size if present
    candidate_ns = [80, 96, 112, 128, 144, 160]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5.0:
            break
        remaining = budget_s - (time.time() - start_time)
        alloc = min(250.0, remaining * 0.5)
        h_opt, c5_opt = optimise_n(n, rng, alloc)
        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global, best_c5_global, best_n_global = h_opt, c5_opt, n
            if best_c5_global < 0.3808:
                break

    # final safety net
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        M = n // 2
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        cur = np.sum(best_h_global)
        if abs(cur - M) > 1e-9:
            best_h_global = project_sum_box(best_h_global, M)

    return best_h_global, best_c5_global, n
