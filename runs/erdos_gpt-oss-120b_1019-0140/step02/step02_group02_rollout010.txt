def repair(h, target):
    # ensure sum exactly target, values in [0,1]
    # first clip
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # distribute diff to random indices not at bounds
    idxs = np.arange(len(h))
    np.random.shuffle(idxs)
    for i in idxs:
        if diff == 0:
            break
        if diff > 0:
            cap = 1.0 - h[i]
            if cap > 0:
                inc = min(cap, diff)
                h[i] += inc
                diff -= inc
        else: # diff < 0
            cap = h[i]
            if cap > 0:
                dec = min(cap, -diff)
                h[i] -= dec
                diff += dec
    return h

def adjust_sum(h, target, rng):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase
        possible = np.where(h < 1.0 - 1e-12)[0]
        if len(possible) == 0:
            return h
        while diff > 1e-12:
            i = rng.choice(possible)
            cap = 1.0 - h[i]
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
            if cap - inc < 1e-12:
                # now full; remove from possible
                possible = possible[possible != i]
                if len(possible) == 0:
                    break
    else:
        # need to decrease
        possible = np.where(h > 1e-12)[0]
        if len(possible) == 0:

        while diff < -1e-12:
            i = rng.choice(possible)

    diff = target - cur
    if abs(diff) < 1e-12:

    if diff > 0:

        while diff > 1e-12:
            # indices that can increase
            inc_idx = np.where(h < 1.0 - 1e-12)[0]
            if inc_idx.size == 0:
                break
            i = rng.choice(inc_idx)
            capacity = 1.0 - h[i]
            inc = min(capacity, diff)
            h[i] += inc
            diff -= inc
    else:

        while diff < -1e-12:
            dec_idx = np.where(h > 1e-12)[0]
            if dec_idx.size == 0:
                break
            i = rng.choice(dec_idx)
            capacity = h[i]
            dec = min(capacity, -diff)
            h[i] -= dec
            diff += dec
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the max overlap integral approximation for a given vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def adjust_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Modify h in place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    eps = 1e-12
    if abs(diff) < eps:
        return h
    if diff > 0.0:
        while diff > eps:
            inc_idx = np.where(h < 1.0 - eps)[0]
            if inc_idx.size == 0:
                break
            i = rng.choice(inc_idx)
            cap = 1.0 - h[i]
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
    else:
        while diff < -eps:
            dec_idx = np.where(h > eps)[0]
            if dec_idx.size == 0:
                break
            i = rng.choice(dec_idx)
            cap = h[i]
            dec = min(cap, -diff)
            h[i] -= dec
            diff += dec
    return h

def random_start(rng: np.random.Generator, n: int, target: float) -> np.ndarray:
    """Generate a random feasible vector with entries in [0,1] and exact sum."""
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    v = adjust_sum(v, target, rng)
    return v

def gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the overlap for the shift k that currently attains the maximum."""

        i = np.arange(m)
        g[i] += 1.0 - h[i + k]
        g[i + k] -= h[i]

        m = n - kk
        i = np.arange(m)
        g[i + kk] += 1.0 - h[i]

    return g

def find_move(h: np.ndarray, k: int, rng: np.random.Generator):
    """Choose a source index with large contribution and a destination with small contribution."""
    n = h.shape[0]
    if k >= 0:
        idx = np.arange(n - k)
        prod = h[idx] * (1.0 - h[idx + k])
    else:
        kk = -k
        idx = np.arange(n - kk)
        prod = h[idx + kk] * (1.0 - h[idx])
    i = idx[np.argmax(prod)]
    # build a full‑length array where missing positions have zero contribution
    prod_full = np.zeros(n)
    prod_full[idx] = prod
    cand_order = np.argsort(prod_full)  # ascending
    for j in cand_order:
        if j != i and h[j] < 1.0 - 1e-12:
            return i, j
    # fallback: any index that can receive mass
    possible = np.where(h < 1.0 - 1e-12)[0]
    possible = possible[possible != i]
    if possible.size == 0:
        return None, None
    return i, int(rng.choice(possible))
