    score = start_score
    best_h = h.copy()
    best_score = score
    n = len(h)

        i, j = rng.integers(n, size=2)
        if i == j or h[i] == h[j]:

        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # Bisection
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure bounds bracket s
    # maybe adjust if necessary
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def gradient_descent_momentum(h0, time_budget, step0=0.5, beta=0.9):
    start = time.time()
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        it += 1
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        # compute gradient as average over shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[sft:] = h[:n - sft]
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

# ---------- Helper functions ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def resample(arr, new_n):
    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def binarize_balanced(v):
    """Return a binary vector with exactly half ones by thresholding the largest entries."""
    n = v.size
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    out = np.zeros_like(v, dtype=float)
    out[idx] = 1.0
    return out


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def slsqp_optimize_candidate(h0, max_iter=800):
    """Run a short SLSQP on a small‑scale problem to improve a candidate."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def gradient_descent_momentum(h0, time_budget, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum for a fixed time budget."""
    start = time.time()
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    it = 0
    while True:
        if time.time() - start >= time_budget:
            break
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def hill_climb(start_h, start_score, rng, time_end):
    """Simple binary hill‑climbing on balanced 0/1 vectors."""
    h = start_h.copy()
    score = start_score
    best_h = h.copy()
    best_score = score
    n = h.size
    target = 0.3808

    while time.time() < time_end:
        i, j = rng.integers(n, size=2)
        if i == j or h[i] == h[j]:
            continue
        # swap a 1 with a 0
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
        if new_score < score:
            score = new_score
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        if best_score <= target:
            break
    return best_h, best_score

    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    target = 0.3808

    # ---- Phase 1: small‑scale SLSQP ----
    n_small = 128
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n_small), n_small / 2.0)
        candidates.append(cand)

    candidates.append(thue_morse(n_small))
    for _ in range(3):
        v = rng.random(n_small)

    best_h_small = None
    best_score_small = np.inf

    for cand in candidates:

        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()

    if best_h_small is None:
        best_h_small = project_onto_box_sum(rng.random(n_small), n_small / 2.0)
        best_score_small = compute_c5(best_h_small)

    # ---- Phase 2: upsample to larger grid ----
    n_large = 512
    h_large = resample(best_h_small, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    best_h = h_large
    best_score = compute_c5(best_h)

    # ---- Phase 3: continuous refinement (gradient descent) ----
    remain = time_left()
    if remain > 8.0:
        gd_time = remain * 0.7
        h_gd, sc_gd = gradient_descent_momentum(best_h, time_budget=gd_time,
                                                step0=0.5, beta=0.9)
        if sc_gd < best_score:

    # ---- Phase 4: binary hill climbing ----
    remain = time_left()
    if remain > 3.0:

        h_bin = binarize_balanced(best_h)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_h, best_score = h_bin, sc_bin
        end_hc = time.time() + remain
        h_hc, sc_hc = hill_climb(best_h, best_score, rng, time_end=end_hc)
        if sc_hc < best_score:
            best_h, best_score = h_hc, sc_hc

    # final sanity projection and score recompute
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
