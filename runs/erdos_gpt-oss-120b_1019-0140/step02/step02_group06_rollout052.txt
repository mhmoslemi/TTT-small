def propose_move(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    hi = h[i]
    lo = h[j]
    # delta can be positive (increase i and decrease j) or negative (decrease i increase j)
    delta_min = max(-hi, lo-1)
    delta_max = min(1-hi, lo)
    if delta_max <= delta_min:
        return None  # cannot move
    delta = rng.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

Define function gradient_of_softmax(h, beta):
   v = np.correlate(h, 1-h, mode='full')
   vmax = v.max()
   w = np.exp(beta * (v - vmax))
   w /= w.sum()
   # Compute gradient g
   n = len(h)
   offset = n-1
   g = np.zeros(n)
   for idx, weight in enumerate(w):
       if weight == 0: continue
       k = idx - offset
       if k >= 0:
           tail = slice(0, n - k)
           head = slice(k, n)
       else:
           k2 = -k
           tail = slice(k2, n)
           head = slice(0, n - k2)
       g[tail] += weight * (1.0 - h[head])
       g[head] -= weight * h[tail]
   # enforce zero-sum direction: gradient of constraint -> subtract mean
   g -= np.mean(g)
   return g

    h = h0.copy()
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    beta_growth = 1.001 # per iteration
    lr = 0.3
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v2 = np.zeros_like(h)
    t_iter = 0
    while time.time() - start < time_budget:
        t_iter += 1
        # compute correlation for objective for tracking
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral multiplied by dx."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_box_sum(x, target, max_iter=60):
    """
    Project vector x onto the box [0,1]^n while enforcing sum(x) == target.
    Uses a simple bisection on a Lagrange multiplier.
    """
    lo = -np.max(x) - 1.0   # makes all entries <=0 after shift
    hi = 1.0 - np.min(x)    # makes all entries >=1 after shift
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def gradient_of_softmax(h, beta):
    """Gradient of the soft‑max smoothed max‑overlap objective."""
    n = h.shape[0]
    v = np.correlate(h, 1.0 - h, mode='full')
    vmax = v.max()
    w = np.exp(beta * (v - vmax))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / v.size)
    else:
        w = w / w_sum
    offset = n - 1
    g = np.zeros_like(h)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
        g[tail] += weight * (1.0 - h[head])
        g[head] -= weight * h[tail]
    # zero‑sum direction (preserve total mass)
    g -= np.mean(g)
    return g

def adam_optimize(h0, time_budget, seed):
    """
    Adam‑style gradient descent on a smooth approximation of the max‑overlap.
    Returns the best vector seen and its C5 value.
    """
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    # hyper‑parameters
    beta = 2.0
    max_beta = 30.0
    beta_growth = 1.001
    lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v2 = np.zeros_like(h)
    t_iter = 0
    no_improve = 0
    while time.time() - start < time_budget:
        t_iter += 1
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # increase smoothing parameter
        if beta < max_beta:
            beta = min(max_beta, beta * beta_growth)
        # gradient of smoothed objective
        g = gradient_of_softmax(h, beta)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v2 = beta2 * v2 + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v2 / (1.0 - beta2 ** t_iter)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_sum(h_new, target)
        h = h_new
        # occasional random perturbation to escape stagnation
        if t_iter % 2000 == 0:
            h = project_box_sum(h + rng.normal(scale=0.02, size=n), target)
        # hard restart if no improvement for long
        if no_improve >= 5000:
            h = project_box_sum(rng.random(n), target)
            m.fill(0.0)
            v2.fill(0.0)
            no_improve = 0
    return best_h, best_c5

def hill_climb(h_start, time_budget, rng):
    """
    Simple pairwise swap hill climbing with sum preservation.
    Returns the best vector found.
    """
    n = h_start.shape[0]
    target = n / 2.0
    h_curr = h_start.copy()
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    no_improve = 0
    while time.time() - start < time_budget:
        i, j = rng.choice(n, size=2, replace=False)
        hi = h_curr[i]
        hj = h_curr[j]
        # delta added to i and subtracted from j
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max)
        h_new = h_curr.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        # accept improvement for the current walk as well
        if c5_new < compute_c5(h_curr) - 1e-12:
            h_curr = h_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 20000:
            # small random shake around best
            h_curr = best_h + rng.normal(scale=0.01, size=n)
            h_curr = project_box_sum(h_curr, target)
            no_improve = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of sample points."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modest but sufficiently rich discretisation
    n = 200  # target number of points (still well below 1 k)

    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        n_parent = parent.shape[0]
        # Interpolate parent onto the finer grid
        x_parent = np.linspace(0.0, 2.0, n_parent, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_parent, parent)
    else:
        h0 = rng.random(n)

    # Feasibility projection (box + integral constraint)
    target_sum = n / 2.0
    h0 = project_box_sum(h0, target_sum)

    # Allocate time between two phases
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    grad_frac = 0.6
    grad_time = remaining * grad_frac
    hill_time = max(0.0, remaining - grad_time)

    # Phase 1: Adam‑based smooth descent
    best_h, _ = adam_optimize(h0, grad_time, seed + 1)

    # Phase 2: Pairwise hill climbing for fine tuning
    best_h, best_c5 = hill_climb(best_h, hill_time, rng)

    # Final safety projection
    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
