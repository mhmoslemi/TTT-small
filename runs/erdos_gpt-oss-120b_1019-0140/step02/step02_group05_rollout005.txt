def refine_max_descent(h, target_sum, dx, max_iters=200, step0=0.2, time_limit=None, start_time=None):
    # loop, each iteration update and maybe break if time.
    for it in range(max_iters):
        # time check
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        overlaps = compute_overlaps(h, dx)
        k_idx = np.argmax(overlaps) - (len(overlaps)//2)  # shift value
        # compute gradient of max overlap
        grad = gradient_overlap(k_idx, h, dx)
        # Project gradient to sum zero: subtract mean
        grad -= np.mean(grad)
        # step size schedule
        step = step0 * (0.98 ** it)
        h_new = h - step * grad
        # clip and adjust sum
        h_new = np.clip(h_new, 0.0, 1.0)
        cur = np.sum(h_new)
        diff = target_sum - cur
        if abs(diff) > 1e-12:
            # distribute diff equally among free entries
            free = (h_new > 0) & (h_new < 1)
            if free.any():
                inc = diff / np.sum(free)
                h_new[free] += inc
                h_new = np.clip(h_new, 0.0, 1.0)
                diff = target_sum - np.sum(h_new)
            # residual correction
            if abs(diff) > 1e-12:
                for i in range(len(h_new)):
                    if abs(diff) < 1e-12:
                        break
                    if diff > 0 and h_new[i] < 1.0:
                        add = min(diff, 1.0 - h_new[i])
                        h_new[i] += add
                        diff -= add
                    elif diff < 0 and h_new[i] > 0.0:
                        sub = min(-diff, h_new[i])
                        h_new[i] -= sub
                        diff += sub
        h = np.clip(h_new, 0.0, 1.0)
    return h

def gradient_overlap(k, h, dx):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            # term 1: for i in range(0, n - k)
            idx1 = slice(0, n - k)
            grad[idx1] += dx * (1 - h[idx1 + k])
            # term 2: for i in range(k, n)
            idx2 = slice(k, n)
            grad[idx2] -= dx * h[idx2 - k]
    else:
        kk = -k
        # For negative shift, equivalent to overlap with -k in other direction: sum_{i=kk}^{n-1} h[i] * (1 - h[i-kk]).
        # Could derive directly: same formula with k negative.
        # Using symmetry: overlap_{-k} = sum_{i=0}^{n-1-kk} h[i+kk] * (1 - h[i]) * dx.
        # So we can compute gradient similarly swapping roles.
        # Derive: For k = -kk:
        # overlap = sum_{i=0}^{n-1-kk} h[i+kk] * (1 - h[i]) * dx.
        # Derivative w.r.t h[j]:
        # -dx * h[j+kk] when j in [0, n-1-kk] (from derivative of (1 - h[i]) term)
        # +dx * (1 - h[j-kk]) when j in [kk, n-1] (from derivative of h[i+kk]).
        # So:
        idx1 = slice(kk, n)   # j in kk..n-1 -> from h[i+kk] term
        grad[idx1] += dx * (1 - h[idx1 - kk])
        idx2 = slice(0, n - kk)   # j in 0..n-kk-1 -> from -(h[i+kk]) term
        grad[idx2] -= dx * h[idx2 + kk]
    return grad

def soft_obj_and_grad(h, dx, beta):
    overlaps = compute_overlaps(h, dx)
    max_ov = np.max(overlaps)
    shifted = overlaps - max_ov
    exps = np.exp(beta * shifted)
    sum_exps = np.sum(exps)
    f = max_ov + (1.0 / beta) * np.log(sum_exps)
    w = exps / sum_exps
    grad = np.zeros_like(h)
    offset = len(overlaps)//2  # n-1

        k = idx_shift - offset
        grad += w_k * gradient_overlap(k, h, dx)
    return f, grad

def smooth_and_fix(h, target_sum):
    kernel = np.array([0.25,0.5,0.25])
    # convolution with 'same'
    sm = np.convolve(h, kernel, mode='same')
    sm = np.clip(sm, 0.0, 1.0)
    cur = np.sum(sm)
    diff = target_sum - cur
    # distribute diff to free entries
    free = np.where((sm > 0) & (sm < 1))[0]
    if free.size:
        inc = diff / free.size
        sm[free] += inc
        sm = np.clip(sm, 0.0, 1.0)
        diff = target_sum - np.sum(sm)
    # residual fix
    for i in range(len(sm)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and sm[i] < 1.0:
            add = min(diff, 1.0 - sm[i])
            sm[i] += add
            diff -= add
        elif diff < 0 and sm[i] > 0.0:
            sub = min(-diff, sm[i])
            sm[i] -= sub
            diff += sub
    sm = np.clip(sm, 0.0, 1.0)
    return sm

import time
import numpy as np
from scipy.optimize import minimize

# global placeholders for objective parameters
_OBJ_DX = None
_OBJ_BETA = None
_TARGET_SUM = None

def compute_c5(h):
    """Exact C5 value for discrete h on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_overlaps(h, dx):
    """Vector of all overlap integrals (including shift 0)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def gradient_overlap(k, h, dx):
    """Gradient of the overlap for a given shift k w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            # contribution from h[i] * (1-h[i+k])
            idx1 = slice(0, n - k)
            grad[idx1] += dx * (1.0 - h[idx1 + k])
            # contribution from -h[i+k] * h[i]
            idx2 = slice(k, n)
            grad[idx2] -= dx * h[idx2 - k]
    else:
        kk = -k
        if kk < n:
            # overlap_{-kk} = sum_{i=0}^{n-kk-1} h[i+kk]*(1-h[i])
            idx1 = slice(kk, n)               # i+kk term
            grad[idx1] += dx * (1.0 - h[idx1 - kk])
            idx2 = slice(0, n - kk)           # -(1-h[i]) term
            grad[idx2] -= dx * h[idx2 + kk]
    return grad

def project_sum_and_bounds(h, target):
    """Clip to [0,1] and enforce exact sum by redistributing the deficit."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        if np.any(free):
            inc = diff / np.sum(free)
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
        # residual fix
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)

def random_h(n, target, rng):
    """Generate a random feasible h."""
    h = rng.random(n)
    return project_sum_and_bounds(h, target)

def smooth_and_fix(h, target):
    """One Gaussian smoothing step with sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    cur = np.sum(sm)
    diff = target - cur
    if abs(diff) > 1e-12:
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            inc = diff / free.size
            sm[free] += inc
            sm = np.clip(sm, 0.0, 1.0)
            diff = target - np.sum(sm)
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
    return np.clip(sm, 0.0, 1.0)

def soft_obj_and_grad(h, dx, beta):
    """Smooth approximation of max overlap and its gradient."""
    overlaps = compute_overlaps(h, dx)
    max_ov = np.max(overlaps)
    shifted = overlaps - max_ov
    exps = np.exp(beta * shifted)
    sum_exps = np.sum(exps)
    f = max_ov + (1.0 / beta) * np.log(sum_exps)
    w = exps / sum_exps
    grad = np.zeros_like(h)
    offset = len(overlaps) // 2  # index of shift 0
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - offset
        grad += wk * gradient_overlap(k, h, dx)
    return f, grad

def obj_fun(x):
    global _OBJ_DX, _OBJ_BETA
    f, _ = soft_obj_and_grad(x, _OBJ_DX, _OBJ_BETA)
    return f

def obj_jac(x):
    global _OBJ_DX, _OBJ_BETA
    _, g = soft_obj_and_grad(x, _OBJ_DX, _OBJ_BETA)
    return g

def equality_sum(x):
    global _TARGET_SUM
    return np.sum(x) - _TARGET_SUM

def refine_max_descent(h, target, dx, max_iters=300, step0=0.2):
    """Projected gradient descent on the true max-overlap objective."""
    for it in range(max_iters):
        overlaps = compute_overlaps(h, dx)
        k_idx = int(np.argmax(overlaps)) - (len(overlaps) // 2)
        grad = gradient_overlap(k_idx, h, dx)
        grad -= np.mean(grad)  # keep sum unchanged in first order
        step = step0 * (0.98 ** it)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum_and_bounds(h_new, target)
        h = h_new
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _OBJ_DX, _OBJ_BETA, _TARGET_SUM
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_val = np.inf
    best_n = None

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    if "initial_h_values" in globals():

        val = compute_c5(h_ref)
        if val < best_val:

        # one SLSQP pass if time permits
        elapsed = time.time() - start

            _OBJ_BETA = 12.0
            _TARGET_SUM = target0

                    method="SLSQP",

    # -----------------------------------------------------------------
