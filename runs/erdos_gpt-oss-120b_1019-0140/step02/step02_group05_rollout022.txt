def random_feasible_h(rng, n, target_sum):
    # generate random uniform [0,1] values, then project onto sum target via smoothing fix
    h = rng.random(n)
    # scale to have sum target via simple linear adjustment followed by clipping and rebalancing
    cur = np.sum(h)
    diff = target_sum - cur
    # adjust uniformly
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    # re-fix sum by iterating small adjustments (similar to smooth_and_fix but simpler)
    cur = np.sum(h)
    diff = target_sum - cur
    # Iterate over free positions
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = target_sum - cur
    # final correction
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-9:
        # adjust a single index
        idx = np.argmin(np.abs(h - 0.5))  # arbitrary
        h[idx] += target_sum - cur
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    # ensure sum now
    return h

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    from scipy.optimize import minimize

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # Smoothing and sum-fixing helper
    def smooth_and_fix(h):
        # Gaussian smoothing with kernel length 3
        kernel = np.array([0.25, 0.5, 0.25])
        sm = np.convolve(h, kernel, mode='same')
        sm = np.clip(sm, 0.0, 1.0)
        cur_sum = np.sum(sm)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            free = np.where((sm > 0) & (sm < 1))[0]
            if free.size:
                inc = diff / free.size
                sm[free] += inc
                sm = np.clip(sm, 0.0, 1.0)
            # residual fix
            diff = target_sum - np.sum(sm)
            for i in range(n):
                if abs(diff) < 1e-12:

import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h_initial = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        h_initial = None

    target_sum = n / 2.0
    dx = 2.0 / n

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def smooth_and_fix(h):
        kernel = np.array([0.25, 0.5, 0.25])
        sm = np.convolve(h, kernel, mode='same')
        sm = np.clip(sm, 0.0, 1.0)
        cur_sum = np.sum(sm)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            free = np.where((sm > 0) & (sm < 1))[0]
            if free.size:
                inc = diff / free.size
                sm[free] += inc
                sm = np.clip(sm, 0.0, 1.0)
            diff = target_sum - np.sum(sm)
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and sm[i] < 1.0:
                    add = min(diff, 1.0 - sm[i])
                    sm[i] += add
                    diff -= add
                elif diff < 0 and sm[i] > 0.0:
                    sub = min(-diff, sm[i])
                    sm[i] -= sub
                    diff += sub
            sm = np.clip(sm, 0.0, 1.0)
        return sm

    # Build start list
    starts = []
    if h_initial is not None:
        h0 = smooth_and_fix(np.clip(h_initial, 0.0, 1.0))
        starts.append(h0)

    # deterministic start: block pattern
    block = np.concatenate([np.ones(n//2), np.zeros(n - n//2)])
    starts.append(smooth_and_fix(block))

    # reversed block
    block_rev = np.concatenate([np.zeros(n//2), np.ones(n - n//2)])
    starts.append(smooth_and_fix(block_rev))

    # random binary pattern
    binary = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    binary[idx] = 1.0
    starts.append(smooth_and_fix(binary))

    # random uniform feasible patterns
    for _ in range(10):
        h = rng.random(n)
        h = np.clip(h, 0.0, 1.0)
        cur_sum = np.sum(h)
        diff = target_sum - cur_sum
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        cur_sum = np.sum(h)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            free = np.where((h > 0) & (h < 1))[0]
            if free.size:
                inc = diff / free.size
                h[free] += inc
                h = np.clip(h, 0.0, 1.0)
            # final fix
            cur_sum = np.sum(h)
            diff = target_sum - cur_sum
            if abs(diff) > 1e-12:
                idx_adj = rng.integers(n)
                h[idx_adj] = np.clip(h[idx_adj] + diff, 0.0, 1.0)
        starts.append(smooth_and_fix(h))

    best_h = None
    best_val = np.inf

    # constraints
    def sum_eq(x):
        return np.sum(x[:n]) - target_sum

    ks = [k for k in range(-n+1, n) if k != 0]
    inequality_constraints = []
    for k in ks:
        def make_fun(k):
            def fun(x):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[: n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[: n - kk])
                return x[-1] - overl * dx
            return fun
        inequality_constraints.append({'type': 'ineq', 'fun': make_fun(k)})

    constraints = [{'type': 'eq', 'fun': sum_eq}] + inequality_constraints
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    for idx_start, h_start in enumerate(starts):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.94:
            break

        # optional extra smoothing
        h = h_start.copy()
        for _ in range(2):
            h = smooth_and_fix(h)

        # compute initial C5
        c5_init = compute_c5(h)
        x0 = np.concatenate([h, [c5_init]])

        # adapt maxiter based on remaining time and start index
        remaining_time = budget_s - (time.time() - start_time)
        # estimate iterations; larger for early starts
        maxiter = 300 if idx_start < 5 else 120

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
            )
        except Exception:
            continue

        if not res.success:
            # continue anyway
            pass

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur_sum = np.sum(h_opt)
        if abs(cur_sum - target_sum) > 1e-8:
            h_opt += (target_sum - cur_sum) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)

        val = compute_c5(h_opt)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

        # post‑SLSQP local gradient steps
        if time.time() - start_time < budget_s * 0.85:
            h_cur = h_opt.copy()
            for _ in range(3):
                corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                k_idx = int(np.argmax(corr)) - (n - 1)
                grad = np.zeros_like(h_cur)
                k = k_idx
                if k >= 0:
                    grad[: n - k] = (1.0 - h_cur[k:])
                    grad[k:] -= h_cur[: n - k]
                else:
                    kk = -k
                    grad[kk:] = (1.0 - h_cur[: n - kk])
                    grad[: n - kk] -= h_cur[kk:]
                max_abs = np.max(np.abs(grad))
                if max_abs == 0:
                    break
                step = 0.15
                step_vec = -step * grad / max_abs
                h_new = h_cur + step_vec
                h_new = np.clip(h_new, 0.0, 1.0)
                cur_sum = np.sum(h_new)
                diff = target_sum - cur_sum
                if abs(diff) > 1e-12:
                    free = np.where((h_new > 0) & (h_new < 1))[0]
                    if free.size:
                        inc = diff / free.size
                        h_new[free] += inc
                        h_new = np.clip(h_new, 0.0, 1.0)
                h_cur = h_new
            val2 = compute_c5(h_cur)
            if val2 < best_val:
                best_val = val2
                best_h = h_cur.copy()
                if best_val < 0.3808:
                    break

    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = smooth_and_fix(best_h)
        best_val = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = np.sum(best_h)
    if abs(cur_sum - target_sum) > 1e-9:
        best_h += (target_sum - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, n
