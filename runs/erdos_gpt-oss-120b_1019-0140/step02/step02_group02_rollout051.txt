def hill_climb(h, cur_val, start_time, deadline, rng, max_no_improve=200):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = cur_val
    no_improve_counter = 0
    while time.time() < deadline - 0.05:
        # compute corr to find active k
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * dx
        # If current best better than cur_val? Typically cur_val==best_val when best.
        k = max_idx - (n - 1)
        # define overlapping indices
        if k >= 0:
            i_start = 0
            i_end = n - k
            i_range = np.arange(i_start, i_end, dtype=int)
            # product for i_range:
            prod = h[i_range] * (1.0 - h[i_range + k])
        else:
            kk = -k
            i_start = kk
            i_end = n
            i_range = np.arange(i_start, i_end, dtype=int)
            prod = h[i_range] * (1.0 - h[i_range - kk])
        if i_range.size == 0:
            # nothing to adjust
            break
        # heavy index
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # candidate j outside overlap
        # compute mask for index not in i_range
        # Efficient: use a boolean mask to mark overlapping indices
        mask = np.ones(n, dtype=bool)
        mask[i_range] = False
        # also ensure not i (but i not in mask now)
        # among those, need h < 1 to increase
        cand_idxs = np.where(mask & (h < 1.0 - 1e-12))[0]
        if cand_idxs.size == 0:
            # fallback: any index not i where h<1
            cand_idxs = np.where((np.arange(n) != i) & (h < 1.0 - 1e-12))[0]
            if cand_idxs.size == 0:
                # cannot move
                break
        j = rng.choice(cand_idxs)
        max_transfer = min(h[i], 1.0 - h[j])
        # Determine delta: try full transfer, else random fraction
        # We could try random fraction to explore
        delta = max_transfer * (0.2 + 0.8 * rng.random())  # between 20% and 100%
        # apply move
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
                no_improve_counter = 0
                # early exit if below target threshold
                if best_val < 0.3808:
                    break
            else:
                no_improve_counter = 0
        else:
            # revert
            h[i] += delta
            h[j] -= delta
            # maybe try smaller delta? we could try a few attempts; but skip for simplicity
            no_improve_counter += 1
        # if many no improvements, break or random perturb
        if no_improve_counter >= max_no_improve:
            # random perturbation to escape local optimum
            # move a small amount between two random indices
            i2, j2 = rng.choice(n, size=2, replace=False)
            # ensure i2 has mass and j2 capacity
            if h[i2] > 1e-6 and h[j2] < 1.0 - 1e-6:
                delta2 = min(h[i2], 1.0 - h[j2]) * rng.random() * 0.1
                h[i2] -= delta2
                h[j2] += delta2
            no_improve_counter = 0
        # continue loop
    return best_h, best_val

            # tiny random shuffle to escape a plateau
            i2, j2 = rng.choice(n, size=2, replace=False)
            if h[i2] > 1e-6 and h[j2] < 1.0 - 1e-6:
                delta2 = min(h[i2], 1.0 - h[j2]) * rng.random() * 0.1
                h[i2] -= delta2
                h[j2] += delta2
            no_improve = 0
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # obtain an initial feasible h and its length n
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        cur = h0.sum()
        if abs(cur - target) > 1e-12:
            diff = target - cur
            for idx in (0, 1):
                if abs(diff) < 1e-12:
                    break
                max_adj = (1.0 - h0[idx]) if diff > 0 else h0[idx]
                adj = np.clip(diff / 2.0, -max_adj, max_adj)
                h0[idx] += adj
                diff -= adj
            if abs(diff) > 1e-12:
                h0 = np.clip(h0 + diff / n, 0.0, 1.0)
                diff2 = target - h0.sum()
                if abs(diff2) > 1e-12:
                    ids = np.arange(n)
                    rng.shuffle(ids)
                    for i in ids:
                        if abs(diff2) < 1e-12:
                            break
                        max_adj = (1.0 - h0[i]) if diff2 > 0 else h0[i]
                        adj = np.clip(diff2, -max_adj, max_adj)
                        h0[i] += adj
                        diff2 -= adj
    else:
        n = 120                              # higher resolution than the 80‑point record
        target = n / 2.0
        v = rng.random(n)
        v = v - v.mean() + target / n
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            v = np.clip(v + diff / n, 0.0, 1.0)
            cur = v.sum()
            diff = target - cur
            if abs(diff) > 1e-12:
                ids = np.arange(n)
                rng.shuffle(ids)
                for i in ids:
                    if abs(diff) < 1e-12:
                        break
                    max_adj = (1.0 - v[i]) if diff > 0 else v[i]
                    adj = np.clip(diff, -max_adj, max_adj)
                    v[i] += adj
                    diff -= adj
        h0 = v

    h0 = np.clip(h0, 0.0, 1.0)
    target = n / 2.0
    cur = h0.sum()
    if abs(cur - target) > 1e-9:
        h0[0] = np.clip(h0[0] + (target - cur), 0.0, 1.0)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # SLSQP phase – a few random starts with the exact linear‑plus‑bilinear model
    # ------------------------------------------------------------------
    dx = 2.0 / n
    target_sum = n / 2.0

    def sum_eq(x):
        return np.sum(x[:n]) - target_sum

    constraints = [{"type": "eq", "fun": sum_eq}]
    for kk in range(-n + 1, n):
        if kk == 0:
            continue

        def make_con(k):
            def con(x, k=k):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[: n - k], 1.0 - h[k:])
                else:
                    kk_abs = -k
                    overl = np.dot(h[kk_abs:], 1.0 - h[: n - kk_abs])
                return x[-1] - overl * dx
            return con

        constraints.append({"type": "ineq", "fun": make_con(kk)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    attempts = 0
    while time.time() < deadline - 30.0 and attempts < 3:
        attempts += 1
        start_vec = rng.random(n)
        start_vec = start_vec - start_vec.mean() + target_sum / n
        start_vec = np.clip(start_vec, 0.0, 1.0)
        cur_sum = start_vec.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            start_vec = np.clip(start_vec + diff / n, 0.0, 1.0)
        cur_sum = start_vec.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            start_vec[0] = np.clip(start_vec[0] + (target_sum - cur_sum), 0.0, 1.0)

        init_c = compute_c5(start_vec)
        x0 = np.concatenate([start_vec, [init_c]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 1500, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            break
        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur_sum = h_opt.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            h_opt += (target_sum - cur_sum) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # ------------------------------------------------------------------
    # Local hill climbing starting from the best SLSQP solution
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_h, best_val = hill_climb(best_h.copy(), best_val, start, deadline, rng)

    # ------------------------------------------------------------------
    # A couple of extra hill‑climbing restarts (binary seeds) if time permits
    # ------------------------------------------------------------------
    extra = 0
    while time.time() < deadline - 5.0 and extra < 2:
        extra += 1
        # binary random seed: exactly target ones
        idx = rng.choice(n, size=n // 2, replace=False)
        h_rand = np.zeros(n, dtype=float)
        h_rand[idx] = 1.0
        val_rand = compute_c5(h_rand)
        h_rand, val_rand = hill_climb(h_rand.copy(), val_rand, start, deadline, rng)
        if val_rand < best_val:
            best_val = val_rand
            best_h = h_rand.copy()
            if best_val < 0.3808:
                break

    # ------------------------------------------------------------------
    # Final safety projection onto the feasible set
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target_sum) > 1e-9:
        best_h += (target_sum - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
