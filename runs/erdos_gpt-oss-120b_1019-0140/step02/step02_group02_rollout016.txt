def project_to_sum_and_bounds(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Iterate adjusting sum.
    for _ in range(10):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        # Find indices not at bounds
        free = (x > 0.0) & (x < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust further; break
            # just equalize by scaling maybe
            # scale all values proportionally: x * (target / cur)
            # but ensure bounds.
            if cur > 0:
                x = x * (target / cur)
                x = np.clip(x, 0.0, 1.0)
            break
        delta = diff / n_free
        x[free] += delta
        # Clip again
        x = np.clip(x, 0.0, 1.0)
    # final adjustment: small diff due to rounding; maybe adjust via adding to any not full/having capacity
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:

        for i in np.where(x < 1.0)[0]:

                break

            if diff >= 0:
                break
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
    # ensure final sum exact
    return x

import time
import numpy as np

def compute_c5(h):
    """Exact max overlap integral approximation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    """Uniform random vector in [0,1]^n with exact sum = n/2."""
    target = n / 2.0
    v = rng.random(n)
    # bring average to target/n
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # try a uniform shift first
        inc = diff / n
        v = np.clip(v + inc, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    delta = min(diff, 1.0 - v[i])
                    v[i] += delta
                    diff -= delta
                elif diff < 0 and v[i] > 0.0:
                    delta = min(-diff, v[i])
                    v[i] -= delta
                    diff += delta
    return v

def project_to_sum_and_bounds(x, target):
    """Project onto [0,1] box and exact sum constraint."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # scale uniformly if possible
            if cur > 0:
                x = x * (target / cur)
            x = np.clip(x, 0.0, 1.0)
            break
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        for i in np.where(x < 1.0)[0]:
            if diff <= 0:
                break
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
        for i in np.where(x > 0.0)[0]:
            if diff >= 0:
                break
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
    return x

def overlap_and_weights(h, alpha):
    """Return overlaps (dx*(target - correlation)) and softmax weights."""
    n = h.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, h, mode="full")
    overl = target * dx - dx * corr
    # softmax
    m = np.max(overl)
    exp_shift = np.exp(alpha * (overl - m))
    w = exp_shift / np.sum(exp_shift)
    return overl, w

def compute_grad(h, weights):
    """Gradient of softmax-approximated max overlap w.r.t. h."""
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    # weights indexed by shift s = idx - (n-1)
    for s in range(-(n - 1), n):
        w = weights[s + n - 1]
        if w == 0.0:
            continue
        if s >= 0:
            # contribution from h[j+s]
            grad[: n - s] -= dx * w * h[s:]
            # contribution from h[j-s]
            grad[s:] -= dx * w * h[: n - s]
        else:
            kk = -s
            grad[kk:] -= dx * w * h[: n - kk]
            grad[: n - kk] -= dx * w * h[kk:]
    return grad

def gradient_descent_start(h_start, deadline, rng):
    """Run softmax‑gradient descent from a start vector."""
    n = h_start.shape[0]
    target = n / 2.0
    best_h = h_start.copy()
    best_val = compute_c5(best_h)

    max_iters = 2500
    alpha_start = 5.0
    alpha_end = 80.0
    lr_start = 0.7
    lr_end = 0.001

    for it in range(max_iters):
        if time.time() > deadline - 0.2:
            break
        t_frac = it / max_iters
        alpha = alpha_start * ((alpha_end / alpha_start) ** t_frac)
        lr = lr_start * ((lr_end / lr_start) ** t_frac)

        _, weights = overlap_and_weights(h_start, alpha)
        grad = compute_grad(h_start, weights)

        h_new = h_start - lr * grad
        h_new = project_to_sum_and_bounds(h_new, target)

        # occasional tiny random jitter to escape stagnation
        if rng.random() < 0.01:
            h_new += rng.uniform(-0.02, 0.02, size=n)
            h_new = project_to_sum_and_bounds(h_new, target)

        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            if best_val < 0.3808:
                return best_h, best_val

        h_start = h_new
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_sum_and_bounds(h0, target)
    else:
        n = 80
        target = n / 2.0
        h0 = None

    best_h = None
    best_val = np.inf

    # Try the provided construction first
    if h0 is not None:
        bh, bv = gradient_descent_start(h0.copy(), deadline, rng)
        if bv < best_val:
            best_val, best_h = bv, bh
        if best_val < 0.3808:
            final_h = project_to_sum_and_bounds(best_h, target)
            return final_h.astype(float), float(compute_c5(final_h)), int(n)

    # Multi‑restart loop
    while time.time() < deadline - 0.5:
        h_start = random_start_vector(n, rng)
        # occasional mix with best found so far
        if best_h is not None and rng.random() < 0.2:
            mix = rng.random()
            h_start = mix * best_h + (1 - mix) * h_start
            h_start = project_to_sum_and_bounds(h_start, target)

        bh, bv = gradient_descent_start(h_start, deadline, rng)
        if bv < best_val:
            best_val, best_h = bv, bh
            if best_val < 0.3808:
                break

    # Fallback in the unlikely case nothing was found
    if best_h is None:
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0

    best_h = project_to_sum_and_bounds(best_h, target)
    final_val = compute_c5(best_h)
    return best_h.astype(float), float(final_val), int(n)
