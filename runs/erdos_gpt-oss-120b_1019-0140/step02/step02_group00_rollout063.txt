    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    # shift indices: indices 0 to 2n-2 correspond to shifts -(n-1) to (n-1)
    # we need to align with shift k ∈ [-(n-1), n-1]
    # Use array of length 2*n-1.
    corr = corr * (2.0 / n)  # incorporate dx
    # compute weights
    max_corr = np.max(corr)  # not needed
    exp_alpha = np.exp(alpha * corr - alpha * np.max(corr))  # for numerical stability
    weights = exp_alpha / np.sum(exp_alpha)
    # compute gradient
    grad = np.zeros_like(h)
    # For each shift k, compute contributions
    # shift index mapping: shift = idx - (n-1)
    for idx, w in enumerate(weights):
        if w == 0:
            continue
        shift = idx - (n - 1)
        # compute g_i = 1 - h_{i+shift} - h_{i-shift}
        # vectorized: shift modulo n
        g = 1.0 - np.take(h, (np.arange(n) + shift) % n) - np.take(h, (np.arange(n) - shift) % n)
        grad += w * g
    return grad

def smooth_gradient_descent(h0, time_left_func, max_iters=5000, eta0=0.8, beta=0.9):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    alpha = 1.0
    for it in range(1, max_iters+1):
        if time_left_func() < 0.1:
            break
        # compute correlation array
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        # compute softmax weights
        # use stable computation
        max_corr = np.max(corr)
        exp_vals = np.exp(alpha * (corr - max_corr))
        weights = exp_vals / np.sum(exp_vals)
        # compute gradient
        idx = np.arange(n)
        grad = np.zeros_like(h)
        # loop over all shifts
        for shift_idx, w in enumerate(weights):
            if w == 0:
                continue
            k = shift_idx - (n - 1)
            # compute 1 - h[i+k] - h[i-k]
            grad += w * (1.0 - np.take(h, (idx + k) % n) - np.take(h, (idx - k) % n))
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        eta = eta0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project onto box and sum constraint
        h_new = project_onto_box_sum(h_new, s)
        # compute score with hard max
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # schedule alpha update
        if it % 1000 == 0:
            alpha = min(alpha * 2, 100.0)
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from old to new length (periodic endpoint excluded)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def softmax_weights(corr, alpha):
    """Stable softmax on correlation vector."""
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    return exp_vals / np.sum(exp_vals)


def smooth_gradient_descent(h0, time_left, max_iters=5000, eta0=0.8,
                            beta=0.9, alpha_start=1.0, alpha_factor=2.0,
                            alpha_cap=80.0):
    """Projected (soft‑max) gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    alpha = alpha_start
    idx = np.arange(n)

    for it in range(1, max_iters + 1):
        if time_left() < 0.05:
            break

        # correlation and soft‑max weights
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        w = softmax_weights(corr, alpha)

        # accumulate weighted shifted copies of h
        total = np.zeros_like(h)
        for shift_idx, wi in enumerate(w):
            if wi == 0.0:
                continue
            k = shift_idx - (n - 1)
            if k >= 0:
                # i + k part
                total[:n - k] += wi * h[k:]
                # i - k part
                total[k:] += wi * h[:n - k]
            else:
                p = -k
                total[p:] += wi * h[:n - p]
                total[:n - p] += wi * h[p:]

        grad = np.ones(n) - total  # ∇ (soft‑max) of max‑overlap

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)

        # evaluate true max objective
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()

        h = h_new

        # slowly increase smoothing
        if it % 800 == 0 and alpha < alpha_cap:
            alpha = min(alpha * alpha_factor, alpha_cap)

    return best_h, best_score


def random_local_search(h0, rng, time_left, max_iters=20000, delta_scale=0.12):
    """Small random pairwise adjustments keeping the sum constraint."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)

    while max_iters > 0 and time_left() > 0.05:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.uniform(-delta_scale, delta_scale)
        hi, hj = h[i] + delta, h[j] - delta
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            max_iters -= 1
            continue
        h[i], h[j] = hi, hj
        score = compute_c5(h)
        if score < best_score:
            best_score, best_h = score, h.copy()
        else:
            # revert
            h[i], h[j] = hi - delta, hj + delta
        max_iters -= 1
    return best_h, best_score

    # ----- initial candidates -----
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = upsample(init_full, n)
        cand = project_onto_box_sum(cand, target_sum)
        candidates.append(cand)

    # random continuous starts
    for _ in range(2):
        if time_left() < 1.0:
            break
        v = rng.random(n)
        cand = project_onto_box_sum(v, target_sum)
        candidates.append(cand)

    # random binary starts
    for _ in range(2):
        if time_left() < 1.0:
            break
        cand = random_binary_balanced(n, rng)
        candidates.append(cand.astype(float))

    best_h = None
    best_score = float('inf')
    for cand in candidates:
        sc = compute_c5(cand)
        if sc < best_score:
            best_score, best_h = sc, cand.copy()

    # ----- iterative refinement -----
    while time_left() > 6.0:
        # choose a start: either perturb current best or fresh random
        if best_h is not None and rng.random() < 0.6:
            start = best_h + rng.normal(scale=0.06, size=n)
        else:
            start = rng.random(n)
        start = project_onto_box_sum(start, target_sum)

        # gradient descent with soft‑max smoothing
        max_iters = int(min(5000, time_left() * 30))
        h_opt, sc_opt = smooth_gradient_descent(start, time_left,
                                                max_iters=max_iters,
                                                eta0=0.9, beta=0.85,
                                                alpha_start=0.8,
                                                alpha_factor=2.0,
                                                alpha_cap=80.0)

        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= 0.3808:
                break

        # short local random search around the obtained point
        if time_left() > 2.0:
            h_loc, sc_loc = random_local_search(h_opt, rng, time_left,
                                                max_iters=8000,
                                                delta_scale=0.08)
            if sc_loc < best_score:
                best_score, best_h = sc_loc, h_loc.copy()
                if best_score <= 0.3808:
                    break

    # final safeguard projection and evaluation
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
