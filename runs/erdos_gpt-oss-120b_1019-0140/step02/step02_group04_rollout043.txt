def compute_c5_fft(h, dx):
    n = h.shape[0]
    n_fft = 2 * n
    h_fft = np.fft.fft(h, n=n_fft)
    rev_fft = np.fft.fft((1 - h)[::-1], n=n_fft)
    conv = np.fft.ifft(h_fft * rev_fft).real
    max_val = np.max(conv[:2*n-1])  # there may be small negative rounding errors.
    return max_val * dx

def compute_c5_fft(h, dx, fft_len=None):
    n = len(h)
    if fft_len is None:
        L = 1 << ((2*n - 1).bit_length())
    else:
        L = fft_len
    H = np.fft.rfft(h, n=L)
    rev = np.fft.rfft((1 - h)[::-1], n=L)
    conv = np.fft.irfft(H * rev, n=L)
    # conv length L; take first 2*n-1 terms
    max_val = conv[: 2*n - 1].max()
    return max_val * dx

def compute_c5_fft(h, dx, L):

    rev = np.fft.rfft((1 - h)[::-1], n=L)
    conv = np.fft.irfft(H * rev, n=L)
    # conv length L
    max_val = conv[:2*len(h)-1].max()
    return max_val * dx

def generate_continuous_feasible(n, target_sum, rng):
    # generate random numbers and scale
    arr = rng.random(n)
    arr = arr * (target_sum / arr.sum())
    # clip to [0,1]
    arr = np.clip(arr, 0.0, 1.0)
    # adjust sum
    cur_sum = arr.sum()
    diff = target_sum - cur_sum
    # distribute diff
    if diff > 1e-12:
        # add to entries <1
        inc_candidates = np.where(arr < 1.0)[0]
        rng.shuffle(inc_candidates)
        for idx in inc_candidates:
            inc = min(diff, 1.0 - arr[idx])
            arr[idx] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < -1e-12:
        dec_candidates = np.where(arr > 0.0)[0]
        rng.shuffle(dec_candidates)
        for idx in dec_candidates:
            dec = min(-diff, arr[idx])
            arr[idx] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return arr

    else:
        x = x * (target_sum / total)
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target_sum - cur
    if diff > 0:
        inc_idx = np.where(x < 1.0)[0]
        rng.shuffle(inc_idx)
        for i in inc_idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        dec_idx = np.where(x > 0.0)[0]
        rng.shuffle(dec_idx)
        for i in dec_idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

def _local_anneal(h0, dx, target_sum, L, rng, end_time):
    n = len(h0)
    cur_h = h0.copy()
    cur_val = _compute_c5_fft(cur_h, dx, L)
    best_h = cur_h.copy()
    best_val = cur_val
    step = 0.1
    T0 = max(1e-3, cur_val * 0.5)
    start = time.time()
    while time.time() < end_time:
        elapsed = time.time() - start
        frac = elapsed / (end_time - start + 1e-12)
        T = T0 * (1.0 - frac)
        donors = np.where(cur_h > 1e-12)[0]
        if donors.size == 0:
            break
        i = rng.choice(donors)
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if recipients.size == 0:
            break
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.random() * max_delta
        cur_h[i] -= delta
        cur_h[j] += delta
        new_val = _compute_c5_fft(cur_h, dx, L)
        dval = new_val - cur_val
        accept = False
        if dval < -1e-12:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-dval / T)
                accept = rng.random() < prob
        if accept:
            cur_val = new_val
            step = min(step * 1.05, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                if best_val <= 0.3808:
                    break
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
            step *= 0.99
    # tiny sum correction
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-9:
        if diff > 0:
            cand = np.where(best_h < 1.0)[0]
        else:
            cand = np.where(best_h > 0.0)[0]
        if cand.size:
            idx = rng.choice(cand)
            best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)
    best_val = _compute_c5_fft(best_h, dx, L)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_val = np.inf
    best_n = None

    # Try several discretizations, finer ones first.
    sizes = [200, 150, 120, 80]
    for n in sizes:
        target_sum = n / 2.0
        dx = 2.0 / n
        L = 1 << ((2 * n - 1).bit_length())  # power of two ≥ 2n‑1

        # initial vector
        if 'initial_h_values' in globals():
            init = np.array(initial_h_values, dtype=float)
            if init.size == n:
                h0 = _generate_continuous_feasible(n, target_sum, rng)
            else:
                h0 = _generate_continuous_feasible(n, target_sum, rng)
        else:
            h0 = _generate_continuous_feasible(n, target_sum, rng)

        val0 = _compute_c5_fft(h0, dx, L)
        if val0 < best_val:
            best_val = val0
            best_h = h0.copy()
            best_n = n

        time_elapsed = time.time() - start_time
        time_left = budget_s - time_elapsed
        if time_left <= 2.0:
            break
        alloc = min(0.3 * time_left, 200.0)   # max 200 s per size
        end_time = time.time() + alloc

        while time.time() < end_time:
            # occasional fresh start
            if rng.random() < 0.3:
                h_start = _generate_continuous_feasible(n, target_sum, rng)
            else:
                if best_n == n:
                    h_start = best_h.copy()
                else:
                    h_start = _generate_continuous_feasible(n, target_sum, rng)
            h_opt, val_opt = _local_anneal(h_start, dx, target_sum, L, rng, end_time)
            if val_opt < best_val:
                best_val = val_opt
                best_h = h_opt.copy()
                best_n = n
                if best_val <= 0.3808:
                    return best_h.astype(float), best_val, best_n

    # Final safety adjustments
    if best_h is None:
        best_n = 80
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_val = _compute_c5_fft(best_h, 2.0 / best_n,
                                   1 << ((2 * best_n - 1).bit_length()))
    else:
        n = best_n
        target_sum = n / 2.0
        best_h = np.clip(best_h, 0.0, 1.0)
        cur_sum = best_h.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-9:
            if diff > 0:
                cand = np.where(best_h < 1.0)[0]
            else:
                cand = np.where(best_h > 0.0)[0]
            if cand.size:
                idx = rng.choice(cand)
                best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)
        best_val = _compute_c5_fft(best_h, 2.0 / n,
                                   1 << ((2 * n - 1).bit_length()))
    return best_h.astype(float), best_val, best_n
