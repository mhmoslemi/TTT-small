def local_search(h, time_limit):
    n = len(h)
    M = np.sum(h)  # should be n/2
    dx = 2.0 / n
    # initial autocorrelation
    P = np.correlate(h, h, mode='full')
    C = M - P
    mask = np.arange(-n+1, n) != 0   # but we need array bool relative to indices
    # easier: zero_idx = n-1; mask = np.arange(2*n-1) != zero_idx
    zero_idx = n - 1
    idxs = np.arange(2*n -1)
    mask = idxs != zero_idx
    c5 = np.max(C[mask]) * dx

    best_h = h.copy()
    best_c5 = c5

    # random number generator from outer rng
    start = time.time()
    while time.time() - start < time_limit:
        # maybe schedule temperature: decreasing linearly
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = (1.0 - frac) * 0.02  # start temp about 0.02? calibrate
        # choose i,j such that we can move from j to i
        i, j = rng.choice(n, 2, replace=False)

        # determine move direction: we can decide to move from j to i (increase i) or from i to j
        # compute max transfer for direction i_gain = (1 - h[i]), j_loss = h[j]
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            # try opposite direction
            max_transfer = min(1.0 - h[j], h[i])
            if max_transfer <= 0:
                continue
            # swap i and j roles
            i, j = j, i

        delta = rng.random() * max_transfer
        if delta <= 0:
            continue

        # compute deltaP vector
        # pad h to compute needed values
        pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
        # precompute index offset base
        # note: pad indices for term: i - L + (n-1) etc.
        # but we can compute directly using L vector.
        # Precompute L once outside loop for speed: let's compute L_global outside function.
        # We'll precompute L as np.arange(-n+1, n) global within function but we need variable n.

        # We'll put L = np.arange(-n+1, n) defined before loop.

        term = delta * (pad[i - L + n - 1] + pad[i + L + n - 1] - pad[j - L + n - 1] - pad[j + L + n - 1])

        # deltaP = term + delta^2 contributions
        deltaP = term
        # zero idx
        deltaP[zero_idx] += 2.0 * delta * delta
        offset = j - i
        if offset != 0:
            idx_off = zero_idx + offset
            idx_neg = zero_idx - offset
            deltaP[idx_off] -= delta * delta
            deltaP[idx_neg] -= delta * delta

        # Update P and C
        P_new = P + deltaP
        C_new = M - P_new
        c5_new = np.max(C_new[mask]) * dx

        # Acceptance
        if c5_new < c5 - 1e-12:
            accept = True
        else:
            # compute acceptance probability
            # temperature scaling: use (c5_new - c5) / temp?
            # if temp <=0 skip; else accept prob.
            if temp > 0:
                prob = np.exp(-(c5_new - c5) / temp)
                accept = rng.random() < prob
            else:
                accept = False

        if accept:
            # apply changes
            h[i] += delta
            h[j] -= delta
            P = P_new
            C = C_new
            c5 = c5_new
            # record best
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        # maybe occasionally restart or perturb?
        # No additional.

    return best_h, best_c5

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_binary_h(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def local_search(h0, time_limit, rng):
    n = h0.shape[0]
    M = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    # compute initial autocorrelation
    P = np.correlate(h, h, mode='full')
    # mask for non-zero shift
    zero_idx = n - 1
    mask = np.arange(2*n - 1) != zero_idx
    C = M - P
    c5 = np.max(C[mask]) * dx
    best_h = h.copy()
    best_c5 = c5
    # precompute L and pad
    L = np.arange(-n+1, n, dtype=np.int64)
    pad = np.zeros(3*n - 2, dtype=np.float64)
    start = time.time()
    steps_no_improve = 0
    # parameters
    temp_initial = 0.01  # starting temperature
    restart_prob = 0.001
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # random restart
        if rng.random() < restart_prob:
            h_candidate = random_binary_h(n, rng)
            P_candidate = np.correlate(h_candidate, h_candidate, mode='full')
            C_candidate = M - P_candidate
            c5_candidate = np.max(C_candidate[mask]) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                h = h_candidate.copy()
                P = P_candidate.copy()
                c5 = c5_candidate
                steps_no_improve = 0
            continue

        # choose i,j
        i, j = rng.choice(n, 2, replace=False)
        # decide direction
        max_transfer_ij = min(1.0 - h[i], h[j])
        max_transfer_ji = min(1.0 - h[j], h[i])
        if max_transfer_ij >= max_transfer_ji:
            max_transfer = max_transfer_ij
            # i will gain, j will lose
        else:
            max_transfer = max_transfer_ji
            i, j = j, i  # swap so i gains
        if max_transfer <= 0:
            continue

        delta = rng.random() * max_transfer
        if delta <= 0:
            continue

        # update pad
        pad[n-1:n-1+n] = h
        # compute deltaP via vectorized indexing
        idx_i_minus = i - L + n - 1
        idx_i_plus = i + L + n - 1
        idx_j_minus = j - L + n - 1
        idx_j_plus = j + L + n - 1
        term = delta * (pad[idx_i_minus] + pad[idx_i_plus] - pad[idx_j_minus] - pad[idx_j_plus])
        deltaP = term
        # add delta^2 contributions
        deltaP[zero_idx] += 2.0 * delta * delta
        offset = j - i
        if offset != 0:
            deltaP[zero_idx + offset] -= delta * delta
            deltaP[zero_idx - offset] -= delta * delta

        P_new = P + deltaP
        C_new = M - P_new
        c5_new = np.max(C_new[mask]) * dx

        # acceptance
        diff = c5_new - c5
        if diff <= 0:
            accept = True
        else:
            elapsed_frac = elapsed / time_limit
            temp = temp_initial * (1.0 - elapsed_frac)  # linear cooling
            if temp <= 0:
                accept = False
            else:
                prob = np.exp(-diff / temp)
                accept = rng.random() < prob

        if accept:
            # apply move
            h[i] += delta
            h[j] -= delta
            P = P_new
            C = C_new
            c5 = c5_new
            steps_no_improve = 0 if c5_new < best_c5 else steps_no_improve + 1
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            steps_no_improve += 1

        # occasional restart from best if stuck
        if steps_no_improve >= 5000:
            h = best_h.copy()
            P = np.correlate(h, h, mode='full')
            C = M - P
            c5 = np.max(C[mask]) * dx
            steps_no_improve = 0

    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def random_binary_h(n, rng):
    """Generate a binary feasible vector (0/1) with sum n/2."""
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def local_search(h0, time_limit, rng):
    """Simulated‑annealing style hill climbing with random restarts."""
    n = int(h0.shape[0])
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    start = time.time()
    no_improve = 0
    temp_initial = 0.02               # initial temperature for SA
    restart_prob = 0.001              # chance of a random restart each iteration

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break

        # occasional random restart
        if rng.random() < restart_prob:
            cand = random_binary_h(n, rng)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = cand_c5
                h = cand.copy()
                c5 = cand_c5
                no_improve = 0
            continue

        # pick two distinct indices
        i, j = rng.choice(n, 2, replace=False)

        # decide which index receives mass
        max_ij = min(1.0 - h[i], h[j])   # move from j → i
        max_ji = min(1.0 - h[j], h[i])   # move from i → j
        if max_ij >= max_ji:
            max_transfer = max_ij
        else:
            max_transfer = max_ji
            i, j = j, i                 # swap so i is the receiver

        if max_transfer <= 0.0:
            continue

        delta = rng.random() * max_transfer
        if delta <= 0.0:
            continue

        # propose new vector
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)

        diff = c5_new - c5
        if diff <= 0.0:
            accept = True
        else:
            # linear cooling
            temp = temp_initial * (1.0 - elapsed / time_limit)
            accept = (temp > 0.0) and (rng.random() < np.exp(-diff / (temp + 1e-15)))

        if accept:
            h = h_new
            c5 = c5_new
            no_improve = 0 if c5_new < best_c5 else no_improve + 1
            if c5_new < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = c5_new
        else:
            no_improve += 1

        if no_improve >= 5000:
            # reset to best found so far
            h = best_h.copy()
            c5 = best_c5
            no_improve = 0

    return best_h, best_c5
