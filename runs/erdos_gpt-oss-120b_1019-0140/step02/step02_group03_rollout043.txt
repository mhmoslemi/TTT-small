def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

import time
import numpy as np
from scipy.optimize import minimize

# ----- utilities --------------------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing its sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumEquality:
    """Equality constraint: sum(x) == target."""
    def __init__(self, target):
        self.target = target
    def __call__(self, x):
        return np.sum(x) - self.target

def slsqp_refine(h_start):
    """Local refinement with SLSQP."""
    n = h_start.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': SumEquality(target)}
    res = minimize(
        fun=compute_c5,
        x0=h_start,
        method='SLSQP',
        bounds=bounds,
        constraints=[cons],
        options={'maxiter': 400, 'ftol': 1e-12, 'disp': False}
    )
    if res.success:
        x = np.clip(res.x, 0.0, 1.0)
        x = project_to_box_sum(x, target)
        return x
    return h_start

# ----- main entry point -------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ---- dimension and initial vector ---------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    # make feasible
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_box_sum(h0, target_sum)

    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ---- simulated annealing (pairwise delta moves) -------------------------
    T0 = 1e-2                     # initial temperature
    Tf = 1e-6                     # final temperature
    anneal_deadline = start_time + budget_s * 0.78
    while time.time() < anneal_deadline:
        # temperature schedule (linear)
        tnow = time.time()
        frac = (tnow - start_time) / (anneal_deadline - start_time)
        T = T0 * (1.0 - frac) + Tf * frac

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = cur_h[i], cur_h[j]

        # allowed delta that preserves bounds and sum
        max_up   = min(1.0 - hi, hj)      # increase hi, decrease hj
        max_down = min(hi, 1.0 - hj)      # decrease hi, increase hj
        if max_up + max_down == 0.0:
            continue

        # magnitude scaling (smaller as temperature drops)
        scale = T / T0
        max_delta = (max_up + max_down) * 0.5 * scale
        if max_delta <= 0.0:

        # occasional full swap
