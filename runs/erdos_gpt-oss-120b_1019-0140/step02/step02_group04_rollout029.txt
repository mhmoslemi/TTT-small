import time
import numpy as np

def compute_c5_vec(h, dx):
    """Maximum overlap integral for a binary/continuous step vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _max_shift_and_conv(h):
    """Return (max value, shift index, full convolution) for vector h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    shift = max_idx - (len(h) - 1)  # shift = k
    return max_val, shift, conv

def _binary_from_continuous(vals, target_sum):
    """Produce a binary vector with exactly target_sum ones, choosing largest entries."""
    idx = np.argsort(-vals)[:int(target_sum)]
    b = np.zeros_like(vals, dtype=float)
    b[idx] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        init_vals = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        init_vals = None

    target_sum = n // 2  # binary requirement
    dx = 2.0 / n

    # Initial solution (binary)
    if init_vals is not None:
        best_h = _binary_from_continuous(init_vals, target_sum)
    else:
        best_h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, size=target_sum, replace=False)
        best_h[ones_idx] = 1.0

    best_val = compute_c5_vec(best_h, dx)
    cur_h = best_h.copy()
    cur_val = best_val

    # Simulated annealing / targeted hill‑climb
    time_end = start_time + budget_s * 0.95  # reserve a bit for final polishing
    T_start = 0.02
    T_end = 1e-4
    no_improve_iters = 0
    max_no_improve = 8000  # restart after stagnation

    while time.time() < time_end:
        # evaluate current max shift
        max_overlap, shift_k, _ = _max_shift_and_conv(cur_h)

        # donors: ones that currently contribute to the maximum overlap
        inds = np.arange(n, dtype=int)
        shifted = inds + shift_k
        in_range = (shifted >= 0) & (shifted < n)

        donor_mask = np.zeros(n, dtype=bool)
        # out‑of‑range donors (they always contribute if they are ones)
        donor_mask[~in_range] = (cur_h[~in_range] == 1)
        # in‑range donors
        if np.any(in_range):
            idx_in = inds[in_range]
            sh_in = shifted[in_range]
            donor_mask[in_range] = (cur_h[idx_in] == 1) & (cur_h[sh_in] == 0)

        # recipients: zeros that would align with a one after swapping
        recipient_mask = np.zeros(n, dtype=bool)
        if np.any(in_range):
            idx_in = inds[in_range]
            sh_in = shifted[in_range]
            recipient_mask[in_range] = (cur_h[idx_in] == 0) & (cur_h[sh_in] == 1)

        donors = np.flatnonzero(donor_mask)
        recipients = np.flatnonzero(recipient_mask)

        if donors.size and recipients.size:
            i = rng.choice(donors)
            j = rng.choice(recipients)
        else:

            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
            if ones.size == 0 or zeros.size == 0:

            i = rng.choice(ones)

        cand[i] = 0.0

        improved = False

            for _ in range(30):
                i = rng.choice(donors)
                j = rng.choice(recipients)
                cand = best_h.copy()
                cand[i] = 0.0
                cand[j] = 1.0
                cand_val = compute_c5_vec(cand, dx)

            ones = np.flatnonzero(best_h == 1)

            if cand_val < best_val - 1e-12:

    # Ensure feasibility and proper type
    best_h = np.clip(best_h, 0.0, 1.0)

    current_sum = int(best_h.sum())

            chosen = rng.choice(ones, size=-diff, replace=False)
            best_h[chosen] = 0.0

    best_val = compute_c5_vec(best_h, dx)
    return best_h, best_val, n
