    n = len(h)
    if k >= 0:
        i_candidates = np.where(h[:n - k] > 1e-12)[0]
        if i_candidates.size == 0:
            return None
        i = rng.choice(i_candidates)
        donor = i
        recv = i + k
        max_tr = min(h[donor], 1.0 - h[recv])
    else:
        kk = -k

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_sum(h, target_sum):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # free indices where h not at bounds
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size > 0:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)

    while abs(diff) > 1e-12 and i < len(h):
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub

import time

from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_sum(h, target_sum):
    """Clip to [0,1] and adjust to achieve the exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    # residual distribution (deterministic)
    i = 0
    while abs(diff) > 1e-12 and i < h.size:
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        i += 1
    return h
