def gradient_descent_h(h0, time_budget, rng, target):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    # hyperparams
    beta = 2.0
    max_beta = 100.0
    alpha0 = 0.4
    momentum = np.zeros_like(h)
    # track no improvement
    no_improve = 0
    while time.time() - start < time_budget:
        # compute raw overlaps
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # update smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        # softmax weights
        vmax = v.max()
        # to avoid overflow, subtract max
        w = np.exp(beta * (v - vmax))
        w = w / w.sum()

        # compute gradient
        g = np.zeros(n)
        offset = n - 1
        # vectorized? loops as before, but we could use convolution to compute contributions more efficiently.
        # For each shift k, weight w_k contributes gradient:
        # For shift k >=0:
        # For i in [0,n-1-k]: grad[i] += weight * (1 - h[i+k])
        # For i in [k,n-1]: grad[i] -= weight * h[i-k]
        # Equivalent to:
        # tail indices: i -> i, head indices: i+k -> i.
        # Use slice operations per shift.
        # (We'll keep loop)
        for idx, weight in enumerate(w):
            if weight < 1e-15:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            # grad[tail] += weight * (1 - h[head])
            g[tail] += weight * (1.0 - h[head])
            # grad[head] -= weight * h[tail]
            g[head] -= weight * h[tail]
        # enforce zero-sum direction (mass preserving)
        # Since we have sum h must be target, we need gradient component orthogonal to constant vector.
        # Compute mean gradient and subtract to remove net change in sum.
        g = g - g.mean()

        # momentum update
        momentum = 0.9 * momentum + g
        # step size schedule: linear decay based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_budget
        alpha = alpha0 * (1.0 - frac) + 0.01 * frac  # maybe small final step
        # update
        h_new = h - alpha * momentum
        # projection
        h_new = project_sum_and_bounds(h_new, target)
        # accept if better or with small probability
        c5_new = compute_c5(h_new)
        # always accept if improvement:
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        # random shake if stuck
        if no_improve > 2000:
            # add noise
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    return best_h, best_c5

def grad_descent_softmax(h, time_budget, rng, target, max_beta=200.0):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    alpha0 = 0.4
    momentum = np.zeros_like(h)
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # update beta
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
        # softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum
        # gradient via weighted shifts
        g = np.zeros(n)
        offset = n - 1
        # vectorized loop over shifts
        for idx, weight in enumerate(w):
            if weight < 1e-15:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # project gradient to zero-sum
        g -= g.mean()

        # momentum update
        momentum = 0.9 * momentum + g

        # step size schedule based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_budget
        alpha = alpha0 * (1.0 - frac) + 0.01 * frac

        h_new = h - alpha * momentum
        h_new = project_sum_and_bounds(h_new, target)

        # accept if better
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    return best_h, best_c5

def refine_max_gradient(h, time_budget, rng, target):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    momentum = np.zeros_like(h)
    # step size small
    alpha0 = 0.05
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0 / n)
        idx_max = np.argmax(v)
        k = idx_max - (n - 1)
        # gradient of max shift
        g = np.zeros(n)
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
            g[tail] += 1.0 - h[head]
            g[head] -= h[tail]
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
            g[tail] += 1.0 - h[head]
            g[head] -= h[tail]
        g -= g.mean()
        # momentum
        momentum = 0.9 * momentum + g
        # step size schedule decaying
        elapsed = time.time() - start
        frac = elapsed / time_budget
        alpha = alpha0 * (0.2 + 0.8 * (1 - frac))  # start ~ alpha0 then decay
        h_new = h - alpha * momentum
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        # occasional random perturbation after stagnation
        if np.random.rand() < 0.0005:
            h = project_sum_and_bounds(h + rng.normal(scale=0.01, size=n), target)
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap value multiplied by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def _softmax_gradient(h, beta):
    """Weighted gradient of the smoothed max-overlap."""
    n = len(h)
    v = np.correlate(h, 1.0 - h, mode='full')
    vmax = v.max()
    w = np.exp(beta * (v - vmax))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    g = np.zeros(n)
    offset = n - 1
    for idx, weight in enumerate(w):
        if weight < 1e-15:
            continue
        k = idx - offset
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
        g[tail] += weight * (1.0 - h[head])
        g[head] -= weight * h[tail]
    g -= g.mean()
    return g

def _max_gradient(h):
    """Gradient of the raw max-overlap (single worst shift)."""
    n = len(h)
    v = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(v))
    k = idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        tail = slice(0, n - k)
        head = slice(k, n)
        g[tail] += 1.0 - h[head]
        g[head] -= h[tail]
    else:
        k2 = -k
        tail = slice(k2, n)
        head = slice(0, n - k2)
        g[tail] += 1.0 - h[head]
        g[head] -= h[tail]
    g -= g.mean()
    return g

def _softmax_descent(h0, time_budget, rng, target):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 200.0
    alpha0 = 0.4
    momentum = np.zeros_like(h)
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        g = _softmax_gradient(h, beta)
        momentum = 0.9 * momentum + g
        elapsed = time.time() - start
        frac = elapsed / time_budget
        alpha = alpha0 * (0.2 + 0.8 * (1.0 - frac))
        h_new = h - alpha * momentum
        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
    return best_h, best_c5

def _max_descent(h0, time_budget, rng, target):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    momentum = np.zeros_like(h)
    alpha0 = 0.05
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        g = _max_gradient(h)
        momentum = 0.9 * momentum + g
        elapsed = time.time() - start
        frac = elapsed / time_budget
        alpha = alpha0 * (0.2 + 0.8 * (1.0 - frac))
        h_new = h - alpha * momentum
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        if rng.random() < 0.0005:
            h = project_sum_and_bounds(h + rng.normal(scale=0.01, size=n), target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 80
        h0 = rng.random(n)

    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)

    soft_budget = remaining * 0.7

    final_c5 = compute_c5(best_h)
