    """Maximum overlap integral C5 for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    """
    Simulated annealing on a binary vector with exact half sum.
    Returns best vector found and its C5 value.

    n = overall_best_n
    best_h_float = np.clip(overall_best_h.astype(float), 0.0, 1.0)
    # enforce exact sum (tiny floating adjustments if needed)
    current_sum = best_h_float.sum()
    target_sum_float = n / 2.0
    if abs(current_sum - target_sum_float) > 1e-9:
        diff = target_sum_float - current_sum
        best_h_float[0] = np.clip(best_h_float[0] + diff, 0.0, 1.0)

    overall_best_val = compute_c5_vec(best_h_float, 2.0 / n)
    return best_h_float, overall_best_val, n
