def transfer_step(h, max_delta, rng):
    donors = np.where(h > 0)[0]
    recipients = np.where(h < 1)[0]
    if donors.size == 0 or recipients.size == 0:
        return h
    i = rng.choice(donors)
    j = rng.choice(recipients)
    delta = rng.uniform(0, max_delta)
    delta = min(delta, h[i])  # cannot exceed donor
    delta = min(delta, 1 - h[j])  # cannot exceed recipient capacity
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

import time
import numpy as np

def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target_sum):
    """Clip to [0,1] and adjust to meet exact sum."""
    vec = np.clip(vec, 0.0, 1.0)
    for _ in range(5):
        diff = target_sum - np.sum(vec)
        if abs(diff) < 1e-12:
            break
        free = (vec > 1e-12) & (vec < 1.0 - 1e-12)
        free_cnt = np.count_nonzero(free)
        if free_cnt == 0:
            break
        inc = diff / free_cnt
        vec[free] += inc
        np.clip(vec, 0.0, 1.0, out=vec)
    diff = target_sum - np.sum(vec)
    if abs(diff) > 1e-9:
        if diff > 0:
            idx = np.argsort(vec)
            for i in idx:
                add = min(diff, 1.0 - vec[i])
                vec[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            idx = np.argsort(-vec)
            for i in idx:
                sub = min(-diff, vec[i])
                vec[i] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    return vec

def generate_random_feasible(n, target_sum, rng):
    """Random vector in [0,1]^n with exact sum = target_sum."""
    x = rng.random(n)
    x = x * (target_sum / np.sum(x))
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

def random_transfer(h, rng, max_delta=0.2):
    """Move mass from a donor to a recipient while respecting bounds."""
    n = len(h)
    donors = np.where(h > 1e-12)[0]
    recs = np.where(h < 1.0 - 1e-12)[0]
    if donors.size == 0 or recs.size == 0:
        return h
    i = rng.choice(donors)
    j = rng.choice(recs)
    delta = rng.uniform(0.0, max_delta)
    delta = min(delta, h[i], 1.0 - h[j])
    out = h.copy()
    out[i] -= delta
    out[j] += delta
    return out

def subgradient_descent(start_h, dx, target_sum, step_init, time_limit, rng):
    """Subgradient descent on max overlap objective."""
    h = start_h.copy()
    val = compute_c5_vec(h, dx)
    step = step_init
    n = len(h)
    while time.time() < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)
        grad = np.zeros_like(h)
        if k >= 0:
            grad[: n - k] += (1.0 - h[k:])
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[kk:] += (1.0 - h[: n - kk])
            grad[: n - kk] -= h[kk:]
        norm = np.linalg.norm(grad)
        if norm > 0:
            delta = step * grad / norm
        else:
            delta = np.zeros_like(h)
        cand = h - delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_to_sum(cand, target_sum)
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < val - 1e-12:
            h, val = cand, cand_val
            step = min(step * 1.05, 0.5)
            if val < 0.3808:
                break
        else:
            step = max(step * 0.9, 1e-5)
        # occasional jitter
        if rng.random() < 0.01:
            h = random_transfer(h, rng, max_delta=0.1)
            h = project_to_sum(h, target_sum)
            val = compute_c5_vec(h, dx)
    return h, val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        init_h = np.clip(h0, 0.0, 1.0)
    else:
        n = 80
        init_h = generate_random_feasible(n, n / 2.0, rng)
    target_sum = n / 2.0
    dx = 2.0 / n

    best_h = project_to_sum(init_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    end_time = start + budget_s
    # Main multi‑restart subgradient phase (~70% of budget)
    while time.time() < end_time * 0.7:
        remaining = end_time - time.time()
        if remaining < 5.0:
            break
        # random restart with 30 % probability
        if rng.random() < 0.3:
            start_h = generate_random_feasible(n, target_sum, rng)
        else:
            start_h = best_h.copy()
        # allocate a short slice of time for this run
        slice_time = min(30.0, remaining * 0.4)
        cand_h, cand_val = subgradient_descent(
            start_h, dx, target_sum, step_init=0.2,
            time_limit=time.time() + slice_time, rng=rng
        )
        if cand_val < best_val - 1e-12:
            best_h, best_val = cand_h, cand_val
            if best_val < 0.3808:
                break

    # Final random local search with simple transfers
    while time.time() < end_time - 1.0:
        cand_h = random_transfer(best_h, rng, max_delta=0.15)
        cand_h = project_to_sum(cand_h, target_sum)
        cand_val = compute_c5_vec(cand_h, dx)
        if cand_val < best_val - 1e-12:
            best_h, best_val = cand_h, cand_val
            if best_val < 0.3808:
                break
        # occasional full restart
        if rng.random() < 0.02:
            cand_h = generate_random_feasible(n, target_sum, rng)
            cand_val = compute_c5_vec(cand_h, dx)
            if cand_val < best_val:
                best_h, best_val = cand_h, cand_val

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
