def smoothmax_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    # compute softmax weights (normalized)
    exp_vals = np.exp(beta * (corr - max_corr))
    w = exp_vals / exp_vals.sum()
    # gradient: vector of shape n
    grad = np.zeros(n)
    # compute contributions using slices
    # We need mapping from shift index to shift value L = idx - (n-1)
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        L = idx - (n - 1)
        if L >= 0:
            i_end = n - L
            if i_end <= 0:
                continue
            i_slice = slice(0, i_end)
            j_slice = slice(L, n)
            grad[i_slice] += weight * (1.0 - h[j_slice])
            grad[j_slice] -= weight * h[i_slice]
        else:
            L2 = -L
            i_start = L2
            i_end = n
            if i_start >= n:
                continue
            i_slice = slice(i_start, i_end)
            j_slice = slice(0, n - L2)
            grad[i_slice] += weight * (1.0 - h[j_slice])
            grad[j_slice] -= weight * h[i_slice]
    # scale by dx
    dx = 2.0 / n
    grad *= dx
    # smooth max value
    smooth_val = max_corr * dx + (np.log(exp_vals.sum()) / beta) * dx  # because max_corr is before dx
    # Equivalent: smooth_val = (max_corr + np.log(exp_vals.sum())/beta) * dx
    # Indeed max_corr is before scaling; so:
    smooth_val = (max_corr + np.log(exp_vals.sum())/beta) * (2.0/n)
    return smooth_val, grad, corr

def smoothmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    y = corr - maxc  # <=0
    exp_y = np.exp(beta * y)
    sum_exp = exp_y.sum()
    w = exp_y / sum_exp
    grad = np.zeros(n)
    # vectorized loop over shifts
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        L = idx - (n - 1)
        if L >= 0:
            i_end = n - L
            if i_end <= 0:
                continue
            i_slice = slice(0, i_end)
            j_slice = slice(L, n)
            grad[i_slice] += weight * (1.0 - h[j_slice])
            grad[j_slice] -= weight * h[i_slice]
        else:
            L2 = -L
            i_start = L2
            i_end = n
            if i_start >= n:
                continue
            i_slice = slice(i_start, i_end)
            j_slice = slice(0, n - L2)
            grad[i_slice] += weight * (1.0 - h[j_slice])
            grad[j_slice] -= weight * h[i_slice]
    grad *= dx
    smooth_val = (maxc + np.log(sum_exp) / beta) * dx
    return smooth_val, grad, corr
```

```
def project_sum(h, target):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # distribute over free entries
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
        else:
            # no free entries; adjust uniformly
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
    return h
```

def optimize_h_n(init_h, target, start_time, budget):
    import time, numpy as np
    n = len(init_h)
    best_h = init_h.copy()
    best_val = compute_c5(best_h)
    # parameters
    max_restarts = 5
    for restart in range(max_restarts):
        # random perturbation
        h = best_h.copy()
        # add jitter
        jitter = np.random.default_rng().normal(scale=0.05, size=n)
        h += jitter
        h = np.clip(h, 0.0, 1.0)
        h = project_sum(h, target)
        # Perform gradient descent using momentum
        v = np.zeros_like(h)
        for iter_idx in range(2000):
            elapsed = time.time() - start_time
            if elapsed >= budget * 0.95:
                break
            # schedule beta
            beta = min(30.0, 0.5 + iter_idx * 0.005)  # increase slowly
            # compute smoothmax and grad
            smooth_val, grad, _ = smoothmax_grad(h, beta)
            step = 0.2 / np.sqrt(iter_idx + 1)
            v = 0.9 * v - step * grad
            cand = h + v
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_sum(cand, target)
            # evaluate real C5
            cur_val = compute_c5(cand)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = cand.copy()
                # reset momentum
                v = np.zeros_like(h)
                # early exit
                if best_val < 0.3808:
                    return best_h, best_val
            h = cand
        # After gradient descent, do L-BFGS-B refinement with increasing beta
        h_ref, val_ref = refine_with_lbfgsb(best_h, target, start_time, budget)
        if val_ref < best_val:
            best_val = val_ref
            best_h = h_ref.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

def refine_with_lbfgsb(h0, target, start_time, budget):
    import time, numpy as np
    n = len(h0)
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    # penalty weight
    penalty = 1e5
    betas = [5.0, 10.0, 20.0, 40.0]
    for beta in betas:
        elapsed = time.time() - start_time
        if elapsed >= budget * 0.95:
            break
        def fun(x):
            # smoothmax + penalty
            corr = np.correlate(x, 1.0 - x, mode='full')
            maxc = corr.max()
            y = corr - maxc
            exp_y = np.exp(beta * y)
            sum_exp = exp_y.sum()
            smooth = (maxc + np.log(sum_exp) / beta) * (2.0 / n)
            pen = penalty * (np.sum(x) - target) ** 2
            return smooth + pen
        def jac(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            maxc = corr.max()
            y = corr - maxc
            exp_y = np.exp(beta * y)
            sum_exp = exp_y.sum()
            w = exp_y / sum_exp
            grad = np.zeros_like(x)
            # loop as in smoothmax_grad but without recomputing dx factor
            for idx, weight in enumerate(w):
                if weight == 0.0:
                    continue
                L = idx - (n - 1)
                if L >= 0:
                    i_end = n - L
                    if i_end <= 0:
                        continue
                    i_slice = slice(0, i_end)
                    j_slice = slice(L, n)
                    grad[i_slice] += weight * (1.0 - x[j_slice])
                    grad[j_slice] -= weight * x[i_slice]
                else:
                    L2 = -L
                    i_start = L2
                    i_end = n
                    i_slice = slice(i_start, i_end)
                    j_slice = slice(0, n - L2)
                    grad[i_slice] += weight * (1.0 - x[j_slice])
                    grad[j_slice] -= weight * x[i_slice]
            grad = grad * (2.0 / n) + 2.0 * penalty * (np.sum(x) - target)
            return grad
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(fun, x0=best_h, jac=jac, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False})
            if not res.success:
                # still accept result
                pass
            h_candidate = np.clip(res.x, 0.0, 1.0)
            # force exact sum
            h_candidate = project_sum(h_candidate, target)
            val_candidate = compute_c5(h_candidate)
            if val_candidate < best_val:
                best_val = val_candidate
                best_h = h_candidate.copy()
                if best_val < 0.3808:
                    break
        except Exception:
            # ignore and continue
            pass
    return best_h, best_val

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum(h, target):
    """Clip to [0,1] and adjust to satisfy sum(h)=target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
        else:
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
    return h

def smoothmax_grad(h, beta):
    """Smooth approximation of max correlation and its gradient."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = corr.max()
    y = corr - maxc            # ≤0
    exp_y = np.exp(beta * y)
    sum_exp = exp_y.sum()
    w = exp_y / sum_exp        # softmax weights, sum to 1
    grad = np.zeros(n)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        L = idx - (n - 1)
        if L >= 0:
            i_end = n - L
            if i_end <= 0:
                continue
            i_slice = slice(0, i_end)
            j_slice = slice(L, n)
            grad[i_slice] += weight * (1.0 - h[j_slice])
            grad[j_slice] -= weight * h[i_slice]
        else:
            L2 = -L
            i_start = L2
            i_end = n
            i_slice = slice(i_start, i_end)
            j_slice = slice(0, n - L2)
            grad[i_slice] += weight * (1.0 - h[j_slice])
            grad[j_slice] -= weight * h[i_slice]
    grad *= dx
    smooth_val = (maxc + np.log(sum_exp) / beta) * dx
    return smooth_val, grad, corr

def _lbfgsb_obj(x, beta, target, n, penalty):
    corr = np.correlate(x, 1.0 - x, mode="full")
    maxc = corr.max()
    y = corr - maxc
    exp_y = np.exp(beta * y)
    sum_exp = exp_y.sum()
    smooth = (maxc + np.log(sum_exp) / beta) * (2.0 / n)
    pen = penalty * (np.sum(x) - target) ** 2
    return smooth + pen

def _lbfgsb_jac(x, beta, target, n, penalty):
    corr = np.correlate(x, 1.0 - x, mode="full")
    maxc = corr.max()
    y = corr - maxc
    exp_y = np.exp(beta * y)
    sum_exp = exp_y.sum()
    w = exp_y / sum_exp
    grad = np.zeros_like(x)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        L = idx - (n - 1)
        if L >= 0:
            i_end = n - L
            if i_end <= 0:

            i_slice = slice(0, i_end)
            j_slice = slice(L, n)
            grad[i_slice] += weight * (1.0 - x[j_slice])

        else:

            grad[i_slice] += weight * (1.0 - x[j_slice])
            grad[j_slice] -= weight * x[i_slice]
    grad = grad * (2.0 / n) + 2.0 * penalty * (np.sum(x) - target)
    return grad

    """L‑BFGS‑B refinement with increasing beta and quadratic sum penalty."""
    n = h0.shape[0]
