def local_search_binary(h_start, dx, target_val, time_limit, rng):
    h = h_start.copy()
    val = compute_c5_vec(h, dx)
    n = h.shape[0]
    best_local_h = h.copy()
    best_local_val = val
    while time.time() < time_limit:
        # compute correlation
        cross = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cross)
        cur_val = cross[max_idx] * dx
        if cur_val < best_local_val - 1e-12:
            best_local_h = h.copy()
            best_local_val = cur_val
            if cur_val < target_val:
                break
        k = max_idx - (n - 1)
        i_ones, i_zeros = get_mismatch_indices(h, k)
        if i_ones.size > 0 and i_zeros.size > 0:
            i = rng.choice(i_ones)
            j = rng.choice(i_zeros)
            # swap
            h[i] = 0
            h[j] = 1
        else:
            # random swap
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0
            h[j] = 1
        # evaluate new value
        new_val = compute_c5_vec(h, dx)
        if new_val <= cur_val + 1e-12:
            # keep swap
            val = new_val
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # early break if time nearly out.
    return best_local_h, best_local_val

import time
import numpy as np

def compute_c5_vec(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec

def random_binary(n, rng):
    arr = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=n // 2, replace=False)
    arr[idx] = 1
    return arr

def get_mismatch_indices(h, k):
    n = h.shape[0]
    if k >= 0:
        left = h[: n - k]
        right = h[k:]
        i_ones = np.where((left == 1) & (right == 0))[0]
        i_zeros = np.where((left == 0) & (right == 1))[0]
    else:
        kk = -k
        left = h[kk:]
        right = h[: n - kk]
        i_ones = np.where((left == 1) & (right == 0))[0] + kk
        i_zeros = np.where((left == 0) & (right == 1))[0] + kk
    return i_ones, i_zeros

def local_search_binary(h_start, dx, target_val, time_limit, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5_vec(h, dx)
    n = h.shape[0]
    no_improve = 0
    while time.time() < time_limit:
        cross = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(cross))
        cur_val = cross[max_idx] * dx
        k = max_idx - (n - 1)
        i_ones, i_zeros = get_mismatch_indices(h, k)
        if i_ones.size and i_zeros.size:
            i = int(rng.choice(i_ones))
            j = int(rng.choice(i_zeros))
        else:
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        h[i], h[j] = 0, 1
        new_val = compute_c5_vec(h, dx)
        if new_val < cur_val - 1e-12:
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
                if best_val <= target_val:
                    break
        else:
            h[i], h[j] = 1, 0
            no_improve += 1
            if no_improve >= 200:
                break
    return best_h, best_val

def refine_continuous(h, dx, target_sum, time_limit, start_time, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.05
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = int(rng.choice(donors))
        j = int(rng.choice(recipients))
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = float(rng.uniform(0.0, max_delta))
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = None
    dx = 2.0 / n
    target_sum = n / 2.0
    if h0 is not None:
        binary_h = np.zeros(n, dtype=int)
        top = np.argsort(-h0)[: int(target_sum)]
        binary_h[top] = 1
    else:
        binary_h = random_binary(n, rng)
    best_h = binary_h.astype(float)
    best_val = compute_c5_vec(best_h, dx)
    # binary local search phase
    while time.time() - start_time < budget_s * 0.80:
        if best_val <= 0.3808:
            break
        if rng.random() < 0.2:
            start_h = random_binary(n, rng)
        else:
            start_h = best_h.astype(int)
        local_deadline = min(start_time + budget_s * 0.80, time.time() + 5.0)
        cur_h, cur_val = local_search_binary(start_h, dx, 0.3808, local_deadline, rng)
        if cur_val < best_val - 1e-12:
            best_h = cur_h.astype(float)
            best_val = cur_val
    # continuous refinement phase
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        final_h, final_val = refine_continuous(best_h, dx, target_sum, start_time + remaining, start_time, rng)
        if final_val < best_val - 1e-12:
            best_h, best_val = final_h, final_val
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(np.sum(best_h) - target_sum) > 1e-8:
        best_h = project_to_sum(best_h, target_sum)
        best_val = compute_c5_vec(best_h, dx)
    else:
        best_val = compute_c5_vec(best_h, dx)
    return best_h, best_val, n
