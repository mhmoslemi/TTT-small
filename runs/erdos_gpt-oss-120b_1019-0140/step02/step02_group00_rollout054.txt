def adam_optimize(h0, n_iter=20000, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    sum_target = h.shape[0] / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    for t in range(1, n_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # compute subgradient (as in earlier gradient)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)

        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climb_binary(h, time_left_func):
    n = h.shape[0]
    weight = n // 2
    # ensure binary
    assert np.all(np.isin(h, [0,1]))
    A = np.correlate(h, h, mode='full')
    current_min = A.min()
    best_c5 = weight - current_min
    best_h = h.copy()
    # Indices of ones and zeros
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    while time_left_func() > 0.05:
        # Pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta A efficiently
        # compute using loops across s. We'll create delta array of size 2n-1.
        delta = np.zeros(2*n-1, dtype=int)
        # j and i may be in each other's neighborhoods; ensure correct contributions.
        # We'll add contributions as defined: -h[i+s] - h[i-s] + h[j+s] + h[j-s]
        # For speed, we can compute slices:
        # Compute arrays arr_i_plus, arr_i_minus:
        # arr_i_plus = np.concatenate([np.zeros(i), h[i+1:]])? Not exactly; we need h[i+s] for s from -(n-1) to (n-1)
        # We can loop s range
        for s in range(-n+1, n):
            idx = s + n - 1
            if 0 <= i + s < n:
                delta[idx] -= h[i + s]
            if 0 <= i - s < n:
                delta[idx] -= h[i - s]
            if 0 <= j + s < n:
                delta[idx] += h[j + s]
            if 0 <= j - s < n:
                delta[idx] += h[j - s]
        # new min A
        A_new_min = (A + delta).min()
        new_c5 = weight - A_new_min
        if new_c5 < best_c5:
            # Accept swap
            h[i] = 0
            h[j] = 1
            A += delta
            current_min = A_new_min
            best_c5 = new_c5
            best_h = h.copy()
            # update ones/zeros arrays
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        # Possibly also accept occasional non-improving with probability (annealing)
    return best_h, best_c5

import time
import numpy as np

# ---------- Helper utilities ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # enlarge interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of the max‑overlap term (average over all maximal shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s_target)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def adam_optimize(h0, n_iter=5000, lr=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam‑style projected optimizer for the same objective."""
    h = h0.copy()
    n = h.shape[0]
    s_target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for t in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + eps)
        h_new = h - step * m_hat
        h_new = project_onto_box_sum(h_new, s_target)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def hill_climb_binary(h_int, time_left_func, rng):
    """Swap‑based hill climbing on a binary (0/1) vector."""
    n = h_int.size
    half = n // 2
    A = np.correlate(h_int, h_int, mode='full')
    best_c5 = half - A.min()
    best_h = h_int.copy()

    # Pre‑compute index arrays for speed
    while time_left_func() > 0.02:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        delta = np.zeros(2 * n - 1, dtype=int)
        # contributions of i (changing 1→0)
        for s in range(-n + 1, n):
            idx = s + n - 1
            if 0 <= i + s < n:
                delta[idx] -= best_h[i + s]
            if 0 <= i - s < n:
                delta[idx] -= best_h[i - s]
            if 0 <= j + s < n:
                delta[idx] += best_h[j + s]
            if 0 <= j - s < n:
                delta[idx] += best_h[j - s]

        A_new_min = (A + delta).min()
        new_c5 = half - A_new_min
        if new_c5 < best_c5:
            # accept swap
            best_h[i] = 0
            best_h[j] = 1
            A += delta
            best_c5 = new_c5
    return best_h, best_c5


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # target resolution
    n = 512
    half_n = n // 2

    # ----- build initial candidates -----
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = upsample(init_full, n)
        cand = project_onto_box_sum(cand, half_n)
        candidates.append(cand)

    candidates.append(thue_morse(n))
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, half_n))

    best_h = None
    best_score = float('inf')

    # ----- quick momentum runs on each candidate -----
    short_iters = 3000
    for cand in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = gradient_descent_momentum(cand, n_iter=short_iters,
                                                  step0=0.5, beta=0.9)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # ----- long refinement on the best candidate -----
    if best_h is not None and time_left() > 20.0:
        # allocate the bulk of remaining budget
        remaining = time_left()
        long_iters = int(min(200000, remaining * 800))  # empirical scaling
        h_opt, sc_opt = gradient_descent_momentum(best_h, n_iter=long_iters,
                                                  step0=0.5, beta=0.9)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # ----- Adam refinement (optional) -----
    if best_h is not None and time_left() > 10.0:
        remaining = time_left()
        adam_iters = int(min(50000, remaining * 800))
        h_opt, sc_opt = adam_optimize(best_h, n_iter=adam_iters,
                                      lr=0.2, beta1=0.9, beta2=0.999)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # final projection to guarantee feasibility
    best_h = project_onto_box_sum(best_h, half_n)
    best_score = compute_c5(best_h)

    # ----- binary rounding and hill climbing -----
    if time_left() > 5.0:
        # round to binary by taking the top half entries
        idx = np.argpartition(best_h, -half_n)[-half_n:]
        binary_h = np.zeros_like(best_h, dtype=float)
        binary_h[idx] = 1.0
        bin_score = compute_c5(binary_h)
        if bin_score < best_score:
            best_score = bin_score
            best_h = binary_h.copy()

        # hill climbing on the binary version
        if time_left() > 5.0:
            binary_int = binary_h.astype(int)
            refined_bin, refined_score = hill_climb_binary(binary_int,
                                                           time_left,
                                                           rng)
            refined_bin_f = refined_bin.astype(float)
            if refined_score < best_score:
                best_score = refined_score
                best_h = refined_bin_f.copy()

    # final safeguard
    best_h = project_onto_box_sum(best_h, half_n)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
