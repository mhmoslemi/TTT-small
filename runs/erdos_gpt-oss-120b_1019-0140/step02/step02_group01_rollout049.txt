def project_onto_capped_simplex(v, M, tol=1e-9, max_iter=100):
    n = len(v)
    # initial bounds
    lo = min(v) - 1.0
    hi = max(v)
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(x)
        if abs(s - M) <= tol:
            return x
        if s > M:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

import time

import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discrete step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_capped_simplex(v, M, tol=1e-9, max_iter=100):
    """
    Project vector v onto {x : 0 <= x_i <= 1, sum x_i = M}.
    Uses bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(x)
        if abs(s - M) <= tol:
            return x
        if s > M:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # small correction if necessary
    diff = M - np.sum(x)
    if abs(diff) > tol:

            i = interior[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

                        break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C₅ bound, and the number of samples."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidate_ns = [80, 96, 112, 128, 144]

    # prefer parent size if available
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns = [n0] + candidate_ns
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        remaining = budget_s - elapsed
        alloc = min(250.0, remaining * 0.5)
        h_opt, c5_opt = optimise_with_sa(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    if best_h is None:
        # fallback: constant 0.5 function (feasible for any even n)
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        M = n // 2
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = np.sum(best_h)
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
