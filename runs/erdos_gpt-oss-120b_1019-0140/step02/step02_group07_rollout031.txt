def anneal(h0, time_end, rng, init_temp=0.02, final_temp=1e-6):
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n = len(h)
    target_sum = n/2.0
    it = 0
    # determine total iterations based on time limit approximate
    while time.time() < time_end:
        # propose move
        inc_idx = np.where(h < 1-1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        delta = max_delta * rng.random()
        # optionally occasionally select larger delta
        if rng.random() < 0.1:
            delta = max_delta * (0.5 + 0.5 * rng.random())
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # fix any small negatives/over 1 due to floating error
        # compute new C5
        new_c5 = compute_c5(h_new)
        # temperature schedule based on iteration count
        # We need total_iters estimate; use iteration counter into dynamic schedule
        it += 1
        # compute T as linear decay
        T = init_temp * (1 - it / 5000000) + final_temp * (it / 5000000)
        # accept
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        # maybe periodically enforce sum exactly
        if it % 5000 == 0:
            # reproject to exact sum by adjusting smallest deviation
            diff = target_sum - np.sum(h)
            if abs(diff) > 1e-12:
                # distribute diff uniformly small amount to all entries within bounds
                # simple: add diff/n to each then clip and correct residual
                adj = diff / n
                h += adj

import time, math, numpy as np

def compute_c5(arr):
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def anneal(start_h, time_end, rng,
           init_temp=0.1, final_temp=1e-6, max_iter=2000000):

    target_sum = n / 2.0
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    it = 0
    while time.time() < time_end and it < max_iter:
        inc = np.where(h < 1 - 1e-12)[0]
        dec = np.where(h > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random()
        if rng.random() < 0.1:
            delta = max_delta * (0.5 + 0.5 * rng.random())
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if new_c5 <= cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / (T + 1e-18)):
            h, cur_c5 = h_new, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, h_new.copy()
        it += 1

                residual = target_sum - np.sum(vals)
                for k in range(n):
                    if abs(residual) < 1e-12:
                        break
                    adj = np.clip(residual, -vals[k], 1 - vals[k])
                    vals[k] += adj
                    residual = target_sum - np.sum(vals)
            h0 = vals

        remaining = budget_s - (time.time() - start_time)
        chunk = min(8.0, remaining * 0.2)
        if chunk < 0.5:
            break
        h_opt, c5_opt = anneal(h0, time.time() + chunk, rng)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
            if best_c5 <= 0.3808:
                break

    if best_h is not None and time.time() - start_time < budget_s - 1.5:
        end_time = start_time + budget_s - 0.5
        bin_start = (best_h >= 0.5).astype(int)
        h_fin, c5_fin = binary_hill_climb(bin_start, end_time, rng)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin.astype(float), c5_fin

    if best_h is not None:
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        diff = (n / 2.0) - np.sum(best_h)
        if abs(diff) > 1e-12:
            best_h += diff / n
            best_h = np.clip(best_h, 0.0, 1.0)
            residual = (n / 2.0) - np.sum(best_h)
            for i in range(n):
                if abs(residual) < 1e-12:
                    break
                adj = np.clip(residual, -best_h[i], 1 - best_h[i])
                best_h[i] += adj
                residual = (n / 2.0) - np.sum(best_h)

    return best_h, best_c5, best_n
