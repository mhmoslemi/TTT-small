import numpy as np, cvxpy as cp, time

def solve_erdos_milp(n, target_time):
    # define variables
    h = cp.Variable(n, boolean=True)
    W = cp.Variable((n,n), boolean=True)
    T = cp.Variable()
    constraints = []
    # sum constraint
    constraints.append(cp.sum(h) == n/2)
    # linking constraints
    for i in range(n):
        for j in range(n):
            constraints.append(W[i,j] <= h[i])
            constraints.append(W[i,j] <= h[j])
            constraints.append(W[i,j] >= h[i] + h[j] - 1)
    # correlation constraints
    for s in range(-(n-1), n):
        # overlapping i range
        if s >= 0:
            i_start = 0
            i_end = n - s
            # i from 0 to n-s-1 inclusive
        else:
            i_start = -s
            i_end = n
        # sum Over i in [i_start, i_end)
        idx = np.arange(i_start, i_end, dtype=int)
        # compute sum_h = cp.sum(h[idx])
        # sum_W = cp.sum(W[idx, idx + s])  where idx + s may be vector
        # but need integer indices; we can use list comprehension.
        sum_h = cp.sum(h[idx])
        # gather W terms
        W_terms = [W[i, i+s] for i in idx]  # i+s always in range
        sum_W = cp.sum(cp.hstack(W_terms)) if W_terms else 0
        constraints.append(sum_h - sum_W <= T)
    # objective
    obj = cp.Minimize(T)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CBC, time_limit=target_time)
    # retrieve solution
    h_val = np.array(h.value, dtype=float).flatten()
    T_val = T.value
    return h_val, T_val
```

import time
import numpy as np
import cvxpy as cp

def compute_c5(arr: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    n = arr.size
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * (2.0 / n))


def binary_hill_climb(start_bin: np.ndarray, time_end: float, rng: np.random.Generator):
    """Simple swap‑based hill climbing on binary vectors with exact sum."""
    n = start_bin.size
    target_sum = n // 2
    h = start_bin.copy()
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    while time.time() < time_end and no_improve < 3000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5


def solve_milp_binary(n: int, max_seconds: float):
    """Solve the binary Erdős C5 problem via a compact MILP (CBC)."""
    # Decision variables
    h = cp.Variable(n, boolean=True)
    W = cp.Variable((n, n), boolean=True)   # W[i,j] = h_i * h_j
    T = cp.Variable()
    constraints = []

    # Exact sum constraint
    constraints.append(cp.sum(h) == n // 2)

    # Linking constraints for the product variables
    for i in range(n):
        for j in range(n):
            constraints.append(W[i, j] <= h[i])
            constraints.append(W[i, j] <= h[j])
            constraints.append(W[i, j] >= h[i] + h[j] - 1)

    # Correlation constraints for each shift s = -(n-1)..(n-1)
    for s in range(-(n - 1), n):
        if s >= 0:
            i_start, i_end = 0, n - s
            j_start, j_end = s, n
        else:
            i_start, i_end = -s, n
            j_start, j_end = 0, n + s
        # sum of h over the overlapping segment
        sum_h = cp.sum(h[i_start:i_end])
        # sum of products over the same segment
        sum_W = cp.sum(W[i_start:i_end, j_start:j_end])
        constraints.append(sum_h - sum_W <= T)

    constraints.append(T >= 0)

    prob = cp.Problem(cp.Minimize(T), constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=max_seconds, verbose=False)
    except Exception:
        # CBC may be unavailable; try GLPK_MI as a fallback
        prob.solve(solver=cp.GLPK_MI, max_seconds=max_seconds, verbose=False)

    if h.value is None or T.value is None:
        return None, None
    h_sol = np.array(h.value, dtype=float).flatten()
    T_sol = float(T.value)
    return h_sol, T_sol


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80  # default fallback size

    target_sum = n // 2
    time_left = lambda: max(0.0, budget_s - (time.time() - start_time))

    # ------------------------------------------------------------------
    # Stage 1: try exact MILP (binary) with a generous time budget
    # ------------------------------------------------------------------
    milp_budget = min(time_left(), 600.0)  # cap to 10 minutes
    best_h, best_c5 = None, np.inf

    if milp_budget > 5.0:
        h_bin, T_val = solve_milp_binary(n, milp_budget)
        if h_bin is not None:
            # h_bin already respects the sum exactly and binary bounds
            c5_val = compute_c5(h_bin)
            best_h, best_c5 = h_bin, c5_val
            # early exit if we have already beaten the target
            if best_c5 <= 0.3808:
                return best_h, best_c5, n

    # ------------------------------------------------------------------
    # Stage 2: fallback heuristic (binary hill climbing) if needed
    # ------------------------------------------------------------------
    # Use initial_h_values as a starter if it exists and is binary‑compatible
    if best_h is None:
        if 'initial_h_values' in globals():
            starter = np.array(initial_h_values, dtype=float)
            # Force exact feasibility (binary and correct sum)
            starter = np.clip(starter, 0.0, 1.0)
            cur_sum = np.sum(starter)
            diff = target_sum - cur_sum
            if abs(diff) > 1e-12:
                # simple redistribution to meet exact sum
                adjust = diff / n
                starter += adjust
                starter = np.clip(starter, 0.0, 1.0)
                # final integer fix
                cur = int(round(np.sum(starter)))
                while cur != target_sum:
                    if cur < target_sum:
                        idx = rng.choice(np.where(starter < 1 - 1e-12)[0])
                        starter[idx] = 1.0
                        cur += 1
                    else:
                        idx = rng.choice(np.where(starter > 1e-12)[0])
                        starter[idx] = 0.0
                        cur -= 1
            # binarize
            start_bin = (starter >= 0.5).astype(int)
        else:
            # random binary start
            start_bin = np.zeros(n, dtype=int)
            start_bin[rng.choice(n, target_sum, replace=False)] = 1

        best_h, best_c5 = binary_hill_climb(start_bin, end_hc, rng)

    # ------------------------------------------------------------------
    # Final safety: ensure feasibility and clipping
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)

    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)

        residual = target_sum - np.sum(best_h)
        for i in range(n):

            adj = np.clip(residual, -best_h[i], 1.0 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
