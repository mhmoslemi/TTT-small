def compute_correlation_fft(h):
    """Linear correlation of h with (1-h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    a = np.concatenate([h[::-1], np.zeros(L - n, dtype=h.dtype)])

    corr = np.fft.irfft(np.fft.rfft(a) * np.fft.rfft(b), n=L)
    return corr

    return np.max(corr) * (2.0 / n)

            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # ---- parameters ----
    n = 512                     # grid size
    target_sum = n / 2.0
    # ---- initial candidates ----
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init_full, n)
        candidates.append(binary_from_continuous(init_up))
    # deterministic Thue‑Morse pattern upsampled (optional)
    def thue_morse(m):
        return np.fromiter(((bin(i).count('1') & 1) for i in range(m)), dtype=float, count=m)
    candidates.append(binary_from_continuous(upsample(thue_morse(64), n)))
    # a few random starts
    for _ in range(4):
        rand_vec = rng.random(n)
        candidates.append(binary_from_continuous(rand_vec))

    best_h = None
    best_score = np.inf
    for h in candidates:
        sc = compute_c5_fft(h)
        if sc < best_score:
            best_score, best_h = sc, h.copy()

    # ---- Simulated annealing on binary space ----
    T0 = 0.02
    Tmin = 1e-6
    decay = 0.9995
    cur_h = best_h.copy()
    cur_score = best_score
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h <= 0.5)
    T = T0
    it = 0
    while time_left() > 5.0:
        it += 1
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = compute_c5_fft(cand)
        delta = cand_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_h = cand
            cur_score = cand_score
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h <= 0.5)[0]
        if cur_score < best_score:
            best_score, best_h = cur_score, cur_h.copy()
            if best_score <= 0.3808:
                break
        T = max(T * decay, Tmin)

    # ---- Continuous refinement via projected gradient descent with momentum ----
    remaining = time_left()
    if remaining > 1.0:
        gd_start = time.time()
        h_cont = best_h.astype(float).copy()
        momentum = np.zeros_like(h_cont)
        beta = 0.9
        step0 = 0.5
        best_score_gd = best_score
        best_h_gd = h_cont.copy()
        it_gd = 0
        while time.time() - gd_start < remaining - 0.2:
            it_gd += 1
            corr = compute_correlation_fft(h_cont)
            max_corr = np.max(corr)
            eps = 1e-12
            idxs = np.where(np.abs(corr - max_corr) <= eps)[0]
            shifts = idxs - (n - 1)
            grad = compute_gradients_for_shifts(h_cont, shifts)
            momentum = beta * momentum + (1.0 - beta) * grad
            eta = step0 / np.sqrt(it_gd)
            h_new = h_cont - eta * momentum
            h_new = project_onto_box_sum(h_new, target_sum)
            score_new = compute_c5_fft(h_new)
            if score_new < best_score_gd:
                best_score_gd = score_new
                best_h_gd = h_new.copy()
                if best_score_gd <= 0.3808:
                    break
            h_cont = h_new
        if best_score_gd < best_score:
            best_score, best_h = best_score_gd, best_h_gd.copy()

    # ---- final safeguard projection ----
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5_fft(best_h)
    return best_h, float(best_score), int(best_h.size)
