    n = len(start_h)
    h = repair(start_h, target_sum)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    it = 0
    max_iter = 200000
    init_temp = 1e-2
    final_temp = 1e-8
    while time.time() < time_limit and it < max_iter:
        # choose two indices
        i, j = rng.choice(n, size=2, replace=False)
        # figure max delta
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        max_delta = min(max_inc, max_dec)
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        # temperature schedule
        frac = it / max_iter
        T = init_temp * (1 - frac) + final_temp * frac
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # final repair to exact sum
    best_h = repair(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

import time
import math
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(arr):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * (2.0 / n)

def repair(x, target_sum):
    """Project x onto [0,1] box and enforce sum(x) = target_sum."""
    x = np.clip(x, 0.0, 1.0)
    residual = target_sum - np.sum(x)
    # distribute residual iteratively among free variables
    for _ in range(10):
        if abs(residual) < 1e-12:
            break
        if residual > 0:
            free = np.where(x < 1.0 - 1e-12)[0]
        else:
            free = np.where(x > 1e-12)[0]
        if free.size == 0:
            break
        delta = residual / free.size
        if residual > 0:
            inc = np.minimum(delta, 1.0 - x[free])
            x[free] += inc
        else:
            dec = np.minimum(-delta, x[free])
            x[free] -= dec
        residual = target_sum - np.sum(x)
    return x

def continuous_anneal(start_h, end_time, rng, target_sum):
    """Simple simulated annealing that swaps mass between two positions."""
    n = len(start_h)
    h = repair(start_h, target_sum)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    it = 0
    max_iter = 200000
    init_temp = 1e-2
    final_temp = 1e-8
    while time.time() < end_time and it < max_iter:
        i, j = rng.choice(n, size=2, replace=False)
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        max_delta = min(max_inc, max_dec)
        if max_delta <= 0.0:
            it += 1
            continue
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        frac = it / max_iter
        T = init_temp * (1.0 - frac) + final_temp * frac
        if dE <= 0.0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    best_h = repair(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine dimension and initial feasible vector
    if "initial_h_values" in globals():

    target_sum = n / 2.0
    h0 = repair(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    target_c5 = 0.3808

    # ---- Differential Evolution ----
    def de_obj(x):
        x_rep = repair(x, target_sum)
        return compute_c5(x_rep)

    best_holder = [best_c5, best_h]

    def de_cb(xk, convergence):
        cand = repair(xk, target_sum)
        c5 = compute_c5(cand)
        if c5 < best_holder[0]:
            best_holder[0] = c5
            best_holder[1] = cand.copy()
        return c5 <= target_c5

    bounds = [(0.0, 1.0)] * n
