def compute_softmax_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # compute softmax weights:
    # w = exp(beta*(corr - max_corr)), to avoid overflow
    exps = np.exp(beta * (corr - max_corr))
    w_sum = np.sum(exps)
    w = exps / w_sum
    # shift range: -(n-1) to (n-1)
    shifts = np.arange(-(n-1), n)
    # compute grad = sum_k w_k * (1 - h_{i+k} - h_{i-k})
    # We'll compute convolution of w with terms?
    # For each shift k, term vector = 1 - shift_forward - shift_backward
    # Could accumulate using loops over shifts, but n <=512 so loops ok.
    grad = np.zeros_like(h)
    for idx, k in enumerate(shifts):
        wk = w[idx]
        if wk == 0:
            continue
        if k >= 0:
            term1 = np.concatenate((np.zeros(k), h[:n-k]))
            term2 = np.concatenate((h[k:], np.zeros(k)))
        else:
            s = -k
            term1 = np.concatenate((h[s:], np.zeros(s)))
            term2 = np.concatenate((np.zeros(s), h[:n-s]))
        grad += wk * (1.0 - term1 - term2)
    return grad
```

```
def softmax_gradient(h, beta):
    # compute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    w = exps / np.sum(exps)
    n = h.shape[0]
    offset = n - 1
    # conv_bwd: w * h
    conv_bwd = np.convolve(w, h, mode='full')
    # conv_fwd: w_rev * h
    conv_fwd = np.convolve(w[::-1], h, mode='full')
    grad = 1.0 - conv_fwd[offset:offset+n] - conv_bwd[offset:offset+n]
    return grad
```

Check shapes: w length = 2n-1, h length n, conv length = (2n-1 + n - 1) = 3n-2; offset = n - 1; slice conv_fwd[offset:offset+n] yields length n; yep.

Edge: For loops with large beta, exponential may overflow. Use shift max as done. Exps may become huge for large beta, but we shift max_corr ensures exponent arg <=0; exp( <=0 ) <=1 more stable. For large beta, numbers become near zero except one value.

def adam_optimize(h0, max_iter, lr, beta_start, beta_end, s, time_left_func):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter+1):
        if time_left_func() < 0.01:
            break
        # schedule beta
        beta = beta_start + (beta_end - beta_start) * (t / max_iter)
        grad = softmax_gradient(h, beta)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # optional early break if score below target
        if best_score <= 0.3808:
            break
    return best_h, best_score
```

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the interval brackets the target sum
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return the C5 value for a discretized h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def upsample(arr, new_n):
    return downsample(arr, new_n)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def slsqp_optimize_candidate(h0, max_iter=500):
    """Short SLSQP refinement for a small‑scale candidate."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [
        {'type': 'eq', 'fun': eq_sum},
        {'type': 'ineq', 'fun': ineq_fun}
    ]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)


def gradient_from_weights(h, w):
    """Compute gradient = 1 - conv_fwd - conv_bwd from weight vector w."""
    n = h.shape[0]
    offset = n - 1
    conv_bwd = np.convolve(w, h, mode='full')
    conv_fwd = np.convolve(w[::-1], h, mode='full')
    grad = 1.0 - conv_fwd[offset:offset + n] - conv_bwd[offset:offset + n]
    return grad


def adam_optimize(h0, max_iter, lr, beta_start, beta_end, s, start_time, budget_s):
    """Adam optimizer with a soft‑max smoothing of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = beta1
    beta2_pow = beta2

    best_h = h.copy()
    best_score = compute_c5(h)

    for t in range(1, max_iter + 1):
        # time check
        if budget_s - (time.time() - start_time) < 0.01:
            break

        # correlation and soft‑max weights
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # temperature schedule
        beta = beta_start + (beta_end - beta_start) * (t / max_iter)
        exps = np.exp(beta * (corr - max_corr))
        w = exps / np.sum(exps)

        # gradient
        grad = gradient_from_weights(h, w)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h = project_onto_box_sum(h - step, s)

        # evaluate
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            if best_score <= 0.3808:
                break

        # update bias‑correction powers
        beta1_pow *= beta1
        beta2_pow *= beta2

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---------- small‑scale initialization ----------
    n_small = 64
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_full = project_onto_box_sum(init_full, init_full.size / 2.0)
        cand = downsample(init_full, n_small)
        candidates.append(project_onto_box_sum(cand, n_small / 2.0))

    candidates.append(thue_morse(n_small))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    best_h_small = None
    best_score_small = float('inf')
    # short SLSQP refinements
    for cand in candidates:
        if budget_s - (time.time() - start_time) < 2.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=300)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()
        if best_score_small <= 0.3808:
            break

    # extra random restarts while time permits
    while budget_s - (time.time() - start_time) > 10.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=200)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()
        if best_score_small <= 0.3808:
            break

    if best_h_small is None:
        best_h_small = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score_small = compute_c5(best_h_small)

    # ---------- medium resolution (256) ----------
    n_mid = 256
    h_mid = upsample(best_h_small, n_mid)
    h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
    best_h_mid = h_mid.copy()
    best_score_mid = compute_c5(h_mid)

    # Adam refinement on medium size
    if budget_s - (time.time() - start_time) > 5.0:
        remaining = budget_s - (time.time() - start_time)
        max_iter_mid = int(min(50000, max(5000, remaining * 800)))
        best_h_mid, best_score_mid = adam_optimize(
            h_mid, max_iter_mid, lr=0.10,
            beta_start=0.05, beta_end=5.0,
            s=n_mid / 2.0,
            start_time=start_time, budget_s=budget_s
        )

    # ---------- large resolution (512) ----------
    n_large = 512
    h_large = upsample(best_h_mid, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    best_h_large = h_large.copy()
    best_score_large = compute_c5(h_large)

    if budget_s - (time.time() - start_time) > 5.0:
        remaining = budget_s - (time.time() - start_time)
        max_iter_large = int(min(20000, max(2000, remaining * 300)))
        best_h_large, best_score_large = adam_optimize(
            h_large, max_iter_large, lr=0.08,
            beta_start=0.10, beta_end=5.0,
            s=n_large / 2.0,
            start_time=start_time, budget_s=budget_s
        )

    # ---------- optional binary local improvement ----------
    if budget_s - (time.time() - start_time) > 5.0:
        n = best_h_large.size
        half = n // 2
        idx_desc = np.argsort(-best_h_large)
        binary_h = np.zeros(n, dtype=float)
        binary_h[idx_desc[:half]] = 1.0
        cur_h = binary_h.copy()
        cur_score = compute_c5(cur_h)
        improved = True
        while improved and budget_s - (time.time() - start_time) > 0.5:
            improved = False
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
            rng.shuffle(ones)
            rng.shuffle(zeros)
            for i1 in ones[:10]:
                for i0 in zeros[:10]:
                    cur_h[i1] = 0.0
                    cur_h[i0] = 1.0
                    new_score = compute_c5(cur_h)
                    if new_score + 1e-9 < cur_score:
                        cur_score = new_score
                        improved = True
                        break
                    else:
                        cur_h[i1] = 1.0
                        cur_h[i0] = 0.0
                if improved:
                    break
        if cur_score < best_score_large:
            best_h_large = cur_h.copy()
            best_score_large = cur_score

    # ---------- pick the best overall ----------
    final_h, final_score = best_h_mid, best_score_mid
    if best_score_large < final_score:
        final_h, final_score = best_h_large, best_score_large

    # final guarding projection
    final_h = project_onto_box_sum(final_h, final_h.size / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, final_h.size
