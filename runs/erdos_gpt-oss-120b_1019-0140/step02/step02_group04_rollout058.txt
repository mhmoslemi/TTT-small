def project_to_target(v, target_sum, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    # initial bounds for lambda: low = min(v - upper), high = max(v - lower)
    low = np.min(v - upper)
    high = np.max(v - lower)
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = np.sum(x)
        if abs(s - target_sum) < tol:
            return x
        if s > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low+high)/2.0, lower, upper)

def gradient_for_shift(h, k):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if k > 0:
        # first term: i from 0 to n-1-k
        g[:n - k] = (1.0 - h[k:])
        # second term: i from k to n-1
        g[k:] -= h[:n - k]
    elif k < 0:
        kk = -k
        # first term: i from kk to n-1 (since i>=kk)
        g[kk:] = (1.0 - h[:n - kk])
        # second term: i from 0 to n-1-kk (since i+kk < n)
        g[:n - kk] -= h[kk:]
    else:
        # k==0 not used
        pass
    return g

def project_onto_capped_simplex(v, target_sum, lower=0.0, upper=1.0, tol=1e-10, max_iter=100):
    lo = np.min(v) - upper
    hi = np.max(v) - lower
    # ensure brackets
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = np.sum(x)
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    # after loop
    return np.clip(v - (lo + hi) / 2.0, lower, upper)

def simulated_annealing(h, target_sum, dx, time_limit, start_time, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    best_h, best_val = cur_h.copy(), cur_val
    # Pre-compute temperature schedule: start_temp = maybe 0.005? We'll compute initial as (max_val - min_possible) maybe.
    # Let's set T0 = 0.001 (since we want small step acceptance)
    T0 = 0.001
    # Determine total budget for SA: time_limit - start_time
    total = time_limit - start_time
    while time.time() < time_limit:
        elapsed = time.time() - start_time
        t_frac = elapsed / total if total > 0 else 1.0
        T = T0 * (1 - t_frac)  # linear cooling to 0
        # propose move
        i, j = rng.integers(len(cur_h), size=2, replace=False)
        # Determine direction move from i to j (transfer)
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:
            # try reverse
            max_transfer = min(cur_h[j], 1.0 - cur_h[i])
            if max_transfer <= 0:
                continue
            i, j = j, i
        delta = rng.uniform(0, max_transfer)
        # maybe scale delta by factor so typical changes small:
        # delta = rng.uniform(0, max_transfer) * 0.5 (optional)
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = compute_c5_vec(new_h, dx)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
                # optional early stop if below threshold
                if best_val < 0.3808:
                    break
    return best_h, best_val

import time
import numpy as np

def compute_c5_vec(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_onto_capped_simplex(v, target_sum, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    lo = np.min(v) - upper
    hi = np.max(v) - lower
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = np.sum(x)
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, lower, upper)

def generate_random_feasible(n, target_sum, rng):
    v = rng.random(n)
    return project_onto_capped_simplex(v, target_sum)

def gradient_for_shift(h, k):
    n = len(h)
    g = np.zeros(n, dtype=np.float64)
    if k > 0:
        g[:n - k] = 1.0 - h[k:]
        g[k:] -= h[:n - k]
    elif k < 0:
        kk = -k
        g[kk:] = 1.0 - h[:n - kk]
        g[:n - kk] -= h[kk:]
    # k == 0: gradient is (1 - 2*h) but we ignore shift 0
    return g

def gradient_descent(h_start, target_sum, dx, time_limit, start_time, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5_vec(h, dx)
    # thresholds
    target_val = 0.3808
    # count for no improvement
    no_improve = 0
    max_no_improve = 2000
    alphas = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    while time.time() < time_limit and no_improve < max_no_improve:
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (len(h) - 1)
        # compute gradient
        g = gradient_for_shift(h, k)
        improved = False
        cur_val = corr[max_idx] * dx
        # try various alphas

            # projection onto constraints

            val = compute_c5_vec(h_candidate, dx)

                best_h = h_candidate.copy()
                best_val = val
                improved = True
                no_improve = 0
                break
        if not improved:
            # random perturbation to escape
            i, j = rng.integers(len(h), size=2, replace=False)
            # direction transfer
            max_transfer = min(h[i], 1.0 - h[j])

            max_transfer = min(cur_h[j], 1.0 - cur_h[i])
            if max_transfer <= 0:

import time

# ---------- Utility functions ----------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given discretized h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_onto_capped_simplex(v, target_sum, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    """
    Projection of v onto the set {x | sum(x)=target_sum, lower<=x_i<=upper}.
    Uses bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - upper
    hi = np.max(v) - lower
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = np.sum(x)
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, lower, upper)


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible vector (values in [0,1] and exact sum)."""
    v = rng.random(n)
    return project_onto_capped_simplex(v, target_sum)

def simulated_annealing(h_start, target_sum, dx, time_limit, start_time, rng):
    """Simple SA moving mass between two points."""
    cur_h = h_start.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    best_h, best_val = cur_h.copy(), cur_val
    target_val = 0.3808

                    no_improve = 0
        if not improved:
            no_improve += 1

        if best_val < target_val:
            break
    return best_h, best_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Problem size and optional parent construction
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # Initial feasible solution
    if h0 is not None:
        h0 = np.clip(h0, 0.0, 1.0)
        best_h = project_onto_capped_simplex(h0, target_sum)
    else:
        best_h = generate_random_feasible(n, target_sum, rng)

    best_val = compute_c5_vec(best_h, dx)

    # Time allocation
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining <= 0.0:
        return best_h, best_val, n

    # Simulated annealing phase (~60% of remaining time)
    sa_time = remaining * 0.6
    sa_end = start_time + (time.time() - start_time) + sa_time
    best_h, best_val = simulated_annealing(
        best_h, target_sum, dx, sa_end, start_time, rng
    )

    # Gradient descent polishing phase with the rest of the budget
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5.0:
        gd_end = start_time + budget_s - 1.0  # small safety margin
        best_h, best_val = gradient_descent(
            best_h, target_sum, dx, gd_end, start_time, rng
        )

    # Final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
