def anneal_binary(h, time_limit, start_time, rng):
    n = len(h)
    M = n // 2
    # initialize ones and zeros lists
    ones = [int(i) for i in np.where(h == 1)[0]]
    zeros = [int(i) for i in np.where(h == 0)[0]]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # set parameters
    T = 0.01 # initial temperature
    decay = 0.9995
    iter_count = 0
    while time.time() - start_time < time_limit - 0.01:
        # propose swap
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        # compute c5
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            # accept with probability exp(-delta/T)
            if T > 1e-12 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # swap indices in the lists
            ones[i_idx] = j
            zeros[j_idx] = i
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # update temperature

    best_h = h.copy()

        max_decrease = h[j]

def anneal_binary(h, duration_s, start_time, rng):
    deadline = start_time + duration_s
    n = len(h)
    # ones and zeros lists:
    ones = [int(i) for i in np.where(h == 1)[0]]
    zeros = [int(i) for i in np.where(h == 0)[0]]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.05
    decay = 0.99997
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            # probability
            if T > 1e-12:
                if rng.random() < np.exp(-delta / T):
                    accept = True
        if accept:
            # update lists
            ones[i_idx] = j
            zeros[j_idx] = i
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T *= decay
    return best_h, best_c5

import time
import numpy as np

# ---------------------------------------------------------
# Utility: compute C5 for a given vector h (values in [0,1])
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# ---------------------------------------------------------
# Simulated annealing on binary vectors (swap a 1 and a 0)
def anneal_binary(h, duration_s, start_time, rng):
    deadline = start_time + duration_s
    n = len(h)
    # initialise index lists for 1's and 0's
    ones = [int(i) for i in np.where(h == 1.0)[0]]
    zeros = [int(i) for i in np.where(h == 0.0)[0]]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.05                     # initial temperature
    decay = 0.99997              # exponential cooling
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 1e-12 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # update index lists
            ones[i_idx] = j
            zeros[j_idx] = i
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T *= decay
    return best_h, best_c5

# ---------------------------------------------------------
# Small continuous refinement (mass transfer between two entries)
def continuous_refine(h, duration_s, start_time, rng):
    deadline = start_time + duration_s
    n = len(h)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    while time.time() < deadline:
        i, j = rng.choice(n, 2, replace=False)
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        max_transfer = min(max_inc, max_dec)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        if delta <= 0:
            continue
        h[i] += delta
        h[j] -= delta
        new_c5 = compute_c5(h)
        if new_c5 + 1e-12 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            cur_c5 = new_c5
        else:
            # revert
            h[i] -= delta
            h[j] += delta
    return best_h, best_c5

# ---------------------------------------------------------
# Optimisation for a single discretisation size n
def optimise_n(n, allocated_s, rng, start_time):
    M = n // 2
    best_h = None
    best_c5 = np.inf

    # --- parent solution (if supplied) ---------------------------------
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        parent = np.array(initial_h_values, dtype=float)
        cur = np.sum(parent)
        if abs(cur - M) > 1e-12:
            parent += (M - cur) / n
            parent = np.clip(parent, 0.0, 1.0)
        c5_parent = compute_c5(parent)
        best_h, best_c5 = parent.copy(), c5_parent
        # continuous refinement of parent (may be non‑binary)
        time_left = allocated_s - (time.time() - start_time)
        if time_left > 2.0:
            refine_t = min(5.0, time_left * 0.5)
            ph, pc5 = continuous_refine(parent.copy(), refine_t, start_time, rng)
            if pc5 < best_c5:
                best_h, best_c5 = ph.copy(), pc5

    # --- repeated binary annealing restarts ------------------------------
    while time.time() - start_time < allocated_s - 0.5:
        remaining = allocated_s - (time.time() - start_time)
        inner = min(20.0, remaining * 0.8)
        if inner < 0.5:
            break
        # random binary start
        idx = rng.choice(n, M, replace=False)
        h0 = np.zeros(n, dtype=float)
        h0[idx] = 1.0
        h_opt, c5_opt = anneal_binary(h0.copy(), inner, start_time, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            if best_c5 < 0.3808:
                break

    # --- final continuous polishing -------------------------------------
    time_left = allocated_s - (time.time() - start_time)
    if best_h is not None and time_left > 1.0:
        best_h, best_c5 = continuous_refine(best_h.copy(), time_left, start_time, rng)

    return best_h, best_c5
