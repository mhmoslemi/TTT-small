def softmax_gradient_descent(h0, n_iter, lr, alpha):
    h = h0.copy()
    N = h.size
    dx = 2.0 / N
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        tmax = np.max(corr)
        # compute softmax weights
        w_raw = np.exp(alpha * (corr - tmax))
        w = w_raw / np.sum(w_raw)
        # compute sum_forward and sum_backward via correlation
        w_rev = w[::-1]
        sum_fwd = np.correlate(w, h, mode='full')[N-1:N-1+N]
        sum_bwd = np.correlate(w_rev, h, mode='full')[N-1:N-1+N]
        grad = dx * (1.0 - (sum_fwd + sum_bwd))
        # update
        h = h - lr * grad
        # project onto box+sum
        h = project_onto_box_sum(h, N/2.0)
    return h

def softmax_gradient_descent(h0, n_iter, lr, alpha):
    h = h0.copy()
    N = h.shape[0]
    dx = 2.0 / N
    for it in range(1, n_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        tmax = np.max(corr)
        # compute softmax weights
        w_exp = np.exp(alpha * (corr - tmax))
        w = w_exp / np.sum(w_exp)
        w_rev = w[::-1]
        sum_fwd = np.correlate(w, h, mode='full')[N-1:N-1+N]
        sum_bwd = np.correlate(w_rev, h, mode='full')[N-1:N-1+N]
        grad = dx * (1.0 - (sum_fwd + sum_bwd))
        eta = lr / np.sqrt(it)
        h = h - eta * grad
        h = project_onto_box_sum(h, N/2.0)
    return h

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure the interval brackets target sum
    # adjust if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

    score = compute_c5(h)
    best_h = h.copy()
    best_score = score
    N = h.size

    for it in range(1, max_iters+1):
        if time_check():
            break
        # random swap
        i = rng.integers(N)
        while h[i] == 0:
            i = rng.integers(N)
        # choose opposite
        j = rng.integers(N)
        while h[j] == 1:
            j = rng.integers(N)
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)

        # temperature schedule
        T = T0 * (1 - it / max_iters)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):

            if score < best_score:

                best_h = h.copy()

            h[i], h[j] = h[j], h[i]

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure bracketing
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximated on the discrete grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse(n):
    """Binary Thue–Morse sequence of length n as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def upscale_vec(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def to_binary(h):
    """Return binary vector with exactly N/2 ones by taking the largest entries."""
    N = h.size
    idx = np.argpartition(h, -N // 2)[-N // 2 :]
    bin_vec = np.zeros(N, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec


def binary_anneal(h0, rng, max_iters, time_check):
    """Simulated annealing on balanced binary vectors."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    N = h.size
    T0 = 0.005  # initial temperature

    for it in range(1, max_iters + 1):
        if time_check():
            break
        # choose a random 1 and a random 0 to swap
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = h[j], h[i]  # swap
        new_score = compute_c5(h)
        delta = new_score - cur_score
        T = T0 * (1.0 - it / max_iters)
        accept = delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    N = 256                     # grid size (must be even)
    target = 0.3808

    # ----- initial candidate pool -----
    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init_up = upscale_vec(init, N)
        init_up = project_onto_box_sum(init_up, N / 2.0)
        candidates.append(init_up)

    candidates.append(thue_morse(N))
    for _ in range(3):
        candidates.append(random_binary_balanced(N, rng))
    for _ in range(2):
        v = rng.random(N)
        candidates.append(project_onto_box_sum(v, N / 2.0))

    best_h = None
    best_score = float('inf')

    # ----- main search loop -----
    while time_left() > 8.0:
        # pick a start vector (random or from pool)
        start_vec = rng.choice(candidates)
        # if start_vec is not binary, convert it to binary for annealing
        if np.any((start_vec > 0) & (start_vec < 1)):
            start_bin = to_binary(start_vec)
        else:
            start_bin = start_vec.copy()

        # allocate iterations based on remaining time (roughly 2000 swaps per second)

        best_score = compute_c5(best_h)

    # final sanity projection (should be a binary balanced vector already)
    best_h = project_onto_box_sum(best_h, N / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
