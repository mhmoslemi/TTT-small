def project_sum_and_bounds(h, target_sum):
    # Clip to [0,1], then adjust sum to target:
    h = np.clip(h, 0.0, 1.0)
    cur_sum = np.sum(h)
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # Distribute diff across free variables (0< h <1)
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            # if still diff, adjust sequentially
        cur_sum = np.sum(h)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # adjust sequentially
            for i in range(len(h)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
    return h

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_and_bounds(h, target):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        # residual adjust
        cur = np.sum(h)
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(h)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
            h = np.clip(h, 0.0, 1.0)
    return h

    i, j = rng.choice(n, size=2, replace=False)
    donor, recv = i, j
    max_tr = min(h[donor], 1.0 - h[recv])
    if max_tr <= 0:
        max_tr = min(h[recv], 1.0 - h[donor])
        if max_tr <= 0:
            return h.copy()
        donor, recv = recv, donor
    delta = rng.random() * max_tr
    cand = h.copy()
    cand[donor] -= delta
    cand[recv] += delta
    return np.clip(cand, 0.0, 1.0)

def swap_move(h, rng):
    """Swap two entries."""
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    cand = h.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand

def smooth_and_fix(h, target):
    """Gaussian‑type smoothing followed by projection."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return project_sum_and_bounds(sm, target)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # grid size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        target = n / 2.0
        base = np.array(initial_h_values, dtype=float)
        base = project_sum_and_bounds(np.clip(base, 0.0, 1.0), target)
    else:
        n = 120
        target = n / 2.0
        base = None

    # assemble starting points
    starts = []
    if base is not None:
        starts.append(base)
    for _ in range(4):
        starts.append(random_initial_h(n, target, rng))

    best_h = None
    best_val = np.inf

    # multi‑start local search
    for start_h in starts:
        cur_h = start_h.copy()
        cur_val = compute_c5(cur_h)
        if cur_val < best_val:
            best_val, best_h = cur_val, cur_h.copy()
        while True:
            elapsed = time.time() - start
            if elapsed >= budget_s * 0.94:
                break
            T0 = 0.01
            T = T0 * max(0.0, 1.0 - elapsed / budget_s)

            r = rng.random()
            if r < 0.5:
                cand = gradient_step(cur_h, target, rng)
            elif r < 0.8:
                cand = transfer_move(cur_h, rng)
            else:
                cand = swap_move(cur_h, rng)

            new_val = compute_c5(cand)
            accept = False
            if new_val < cur_val:
                accept = True
            elif T > 0 and rng.random() < np.exp(-(new_val - cur_val) / T):
                accept = True
            if accept:
                cur_h, cur_val = cand, new_val
                if cur_val < best_val:
                    best_val, best_h = cur_val, cur_h.copy()
                    if best_val < 0.3808:
                        break
        if best_val < 0.3808:
            break

    # optional smoothing passes
    if time.time() - start < budget_s - 5.0:
        for _ in range(5):
            sm = smooth_and_fix(best_h, target)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_h, best_val = sm, sm_val
            else:
                break

    # final SLSQP refinement
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        n = len(best_h)
        target = n / 2.0
        dx = 2.0 / n

        def sum_eq(x):
            return np.sum(x[:n]) - target

        constraints = [{"type": "eq", "fun": sum_eq}]
        for k in range(-n + 1, n):
            if k == 0:
                continue

            def make_fun(k):
                def fun(x, k=k):
                    h = x[:n]
                    if k >= 0:
                        overl = np.dot(h[: n - k], 1.0 - h[k:]) * dx
                    else:
                        kk = -k
                        overl = np.dot(h[kk:], 1.0 - h[: n - kk]) * dx
                    return x[-1] - overl
                return fun

            constraints.append({"type": "ineq", "fun": make_fun(k)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 600, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_sum_and_bounds(h_opt, target)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_h, best_val = h_opt, val_opt
        except Exception:
            pass

    # final safety projection
    best_h = project_sum_and_bounds(np.clip(best_h, 0.0, 1.0), target)
    best_val = compute_c5(best_h)

    return best_h, best_val, n
