def compute_c5_fft(h):
    n = h.shape[0]
    L = 2*n - 1
    # zero pad to L
    fh = np.fft.rfft(h, n=L)
    fc = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fh * np.conj(fc), n=L)
    # Note: correlation mode 'full' returns length 2*n-1, where index shift = (i - (n-1))
    # corr[i] = sum_{j} h[j] * (1-h[j - (i-(n-1))]) valid.
    # same as np.correlate.
    c5 = np.max(corr) * (2.0 / n)
    return c5

def compute_c5(h, dx=None):
    n = h.size
    if dx is None:
        dx = 2.0 / n
    L = 2*n - 1
    # choose nfft = next_power_of_two(L)
    nfft = 1 << (L - 1).bit_length()
    # compute rfft length nfft
    fh = np.fft.rfft(h, n=nfft)
    fc = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(fh * np.conj(fc), n=nfft)[:L]
    # may have tiny negative due to numerical errors; clip 0.
    return np.max(corr) * dx

def sa_optimize_continuous_with_params(n, time_budget, rng, init_h, nfft, L, FT_one, dx):
    M = n // 2
    # generate initial h if none
    if init_h is None:
        # binary start then small noise
        h = np.zeros(n)
        ones = rng.choice(n, M, replace=False)
        h[ones] = 1.0
        # add small uniform noise and project
        noise = rng.random(n) * 0.1 - 0.05
        h = np.clip(h + noise, 0.0, 1.0)
        # adjust sum to M
        cur = h.sum()
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
        # adjust precisely if needed
        cur = h.sum()
        if abs(cur - M) > 1e-9:
            diff = M - cur
            while abs(diff) > 1e-12:
                if diff > 0:
                    idx = rng.choice(np.where(h < 1)[0])
                    inc = min(diff, 1 - h[idx])
                    h[idx] += inc
                    diff -= inc
                else:
                    idx = rng.choice(np.where(h > 0)[0])
                    dec = min(-diff, h[idx])
                    h[idx] -= dec
                    diff += dec
    else:
        h = np.clip(init_h.copy(), 0.0, 1.0)
        # ensure exact sum M via similar method
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)
            cur = h.sum()
            if abs(cur - M) > 1e-9:
                diff = M - cur
                # adjust via random indices
                while abs(diff) > 1e-12:
                    if diff > 0:
                        idx = rng.choice(np.where(h < 1)[0])
                        inc = min(diff, 1 - h[idx])
                        h[idx] += inc
                        diff -= inc
                    else:
                        idx = rng.choice(np.where(h > 0)[0])
                        dec = min(-diff, h[idx])
                        h[idx] -= dec
                        diff += dec

    # compute current C5
    ft_h = np.fft.rfft(h, n=nfft)
    ft_1mh = FT_one - ft_h
    corr = np.fft.irfft(ft_h * np.conj(ft_1mh), n=nfft)[:L]
    cur_c5 = np.max(corr) * dx

    best_h = h.copy()
    best_c5 = cur_c5

    start = time.time()
    # temperature schedule
    T0 = 0.02
    T_end = 1e-6
    # iteration counter
    iter_no_improve = 0
    while time.time() - start < time_budget:
        # Temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + T_end * frac
        # propose swap from j to i
        # choose i (to increase), j (to decrease)
        # randomly decide direction
        i, j = rng.choice(n, 2, replace=False)
        # compute max delta allowable
        max_up = 1.0 - h[i]
        max_down = h[j]
        if max_up <= 0 or max_down <= 0:
            continue
        max_delta = min(max_up, max_down)
        # propose delta: uniform up to max_delta
        delta = rng.random() * max_delta
        # apply move
        h[i] += delta
        h[j] -= delta
        # recompute C5
        ft_h = np.fft.rfft(h, n=nfft)
        ft_1mh = FT_one - ft_h
        corr = np.fft.irfft(ft_h * np.conj(ft_1mh), n=nfft)[:L]
        new_c5 = np.max(corr) * dx
        # acceptance
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # compute probability
            if rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                accept = True
        if accept:
            cur_c5 = new_c5
            iter_no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                if best_c5 < 0.3808:  # early exit
                    break
        else:
            # revert
            h[i] -= delta
            h[j] += delta
            iter_no_improve += 1
        # occasional restart
        if iter_no_improve >= 5000:
            # reinitialize h randomly (binary + noise)
            h = np.zeros(n)
            ones = rng.choice(n, M, replace=False)
            h[ones] = 1.0
            noise = rng.random(n) * 0.1 - 0.05
            h = np.clip(h + noise, 0.0, 1.0)
            cur = h.sum()
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)
            # compute cur_c5 again
            ft_h = np.fft.rfft(h, n=nfft)
            ft_1mh = FT_one - ft_h
            corr = np.fft.irfft(ft_h * np.conj(ft_1mh), n=nfft)[:L]
            cur_c5 = np.max(corr) * dx
            iter_no_improve = 0
    return best_h, best_c5

import time
import numpy as np

def _prepare_fft(n):
    L = 2 * n - 1
    nfft = 1 << (L - 1).bit_length()  # next power of two
    ones_pad = np.concatenate([np.ones(n), np.zeros(nfft - n)])
    FT_one = np.fft.rfft(ones_pad)
    dx = 2.0 / n
    return nfft, L, FT_one, dx

def _c5_fft(h, nfft, L, FT_one, dx):
    ft = np.fft.rfft(h, n=nfft)
    ft1 = FT_one - ft
    corr = np.fft.irfft(ft * np.conj(ft1), n=nfft)[:L]
    return np.max(corr) * dx

def _feasible_h(n, M, rng, init_h=None):
    if init_h is not None:
        h = np.clip(init_h.astype(float), 0.0, 1.0)
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)
            cur = h.sum()
        # small greedy fix if still off
        diff = M - cur
        while abs(diff) > 1e-12:
            if diff > 0:
                idx = rng.choice(np.where(h < 1)[0])
                inc = min(diff, 1.0 - h[idx])
                h[idx] += inc
                diff -= inc
            else:
                idx = rng.choice(np.where(h > 0)[0])
                dec = min(-diff, h[idx])
                h[idx] -= dec
                diff += dec
        return h
    # random binary start + tiny noise
    h = np.zeros(n)
    h[rng.choice(n, M, replace=False)] = 1.0
    h = np.clip(h + rng.random(n) * 0.1 - 0.05, 0.0, 1.0)
    cur = h.sum()

    return h

def _simulated_annealing(n, M, rng, time_budget, init_h, nfft, L, FT_one, dx):
    h = _feasible_h(n, M, rng, init_h)
    cur_c5 = _c5_fft(h, nfft, L, FT_one, dx)
    best_h, best_c5 = h.copy(), cur_c5
    start = time.time()
    T0, T_end = 0.02, 1e-6
    no_improve = 0
    while time.time() - start < time_budget:
        # temperature schedule
        frac = (time.time() - start) / time_budget
        T = T0 * (1 - frac) + T_end * frac

        i, j = rng.choice(n, 2, replace=False)
        max_up = 1.0 - h[i]
        max_down = h[j]
        if max_up <= 0 or max_down <= 0:
            continue
        delta = rng.random() * min(max_up, max_down)
        h[i] += delta
        h[j] -= delta

        new_c5 = _c5_fft(h, nfft, L, FT_one, dx)
        accept = False
        if new_c5 < cur_c5:

def _upsample(h, target_n):
    src_n = len(h)
    if src_n == target_n:

    x_src = np.arange(src_n)
    x_tgt = np.linspace(0, src_n - 1, target_n)
    return np.interp(x_tgt, x_src, h)
