def continuous_refine(h0, cur_val, start, deadline, rng):
    n = h0.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    # constraints
    def sum_con(x):
        return np.sum(x[:n]) - target
    constraints = [{'type': 'eq', 'fun': sum_con}]
    def make_con(k):
        def con(x, k=k):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[:n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[:n - kk])
            return x[-1] - overl * dx
        return con
    for kk in range(-n + 1, n):
        if kk == 0:
            continue
        constraints.append({'type': 'ineq', 'fun': make_con(kk)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h0, [cur_val]])
    # set maxiter relative to remaining time
    maxiter = 2000
    time_left = deadline - time.time()
    if time_left < 1.0:
        return h0, cur_val
    options = {'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    try:
        res = minimize(fun=lambda x: x[-1],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options=options)
    except Exception:
        return h0, cur_val
    if not res.success:
        # fallback use h0
        return h0, cur_val
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce sum exactly
    cur = h_opt.sum()
    if abs(cur - target) > 1e-9:
        h_opt += (target - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    val = compute_c5(h_opt)
    return h_opt, val
```

def binary_anneal_periodic(m, total_time, rng):
    start = time.time()
    deadline = start + total_time
    best_s = None
    best_val = np.inf
    # initial candidate: maybe use initial_h_values if length matches 2*m
    # else random.
    while time.time() < deadline - 0.2:
        s = np.zeros(m, dtype=float)
        ones = rng.choice(m, m//2, replace=False)
        s[ones] = 1.0
        s_ones = list(ones)
        s_zeros = [i for i in range(m) if i not in s_ones]
        cur_s = s
        cur_val = compute_c5(np.concatenate([cur_s, cur_s]))
        # local annealing
        while time.time() < deadline - 0.2:
            # propose swap within s
            i_one = rng.integers(len(s_ones))
            i_zero = rng.integers(len(s_zeros))
            idx_one = s_ones[i_one]
            idx_zero = s_zeros[i_zero]
            cur_s[idx_one] = 0.0
            cur_s[idx_zero] = 1.0
            s_ones[i_one], s_zeros[i_zero] = idx_zero, idx_one
            new_val = compute_c5(np.concatenate([cur_s, cur_s]))
            # acceptance based on temperature schedule
            elapsed = time.time() - start
            fraction = elapsed / total_time
            T = max(1e-6, 0.02 * (1 - fraction))
            # Accept if lower or probabilistic
            if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / T):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_s = cur_s.copy()
                    if best_val <= 0.3808:
                        return np.concatenate([best_s, best_s]), best_val
            else:
                # revert
                cur_s[idx_one] = 1.0
                cur_s[idx_zero] = 0.0
                s_ones[i_one], s_zeros[i_zero] = idx_one, idx_zero
        # continue restarts
    if best_s is None:
        # fallback random
        best_s = np.zeros(m, dtype=float)
        best_s[:m//2] = 1.0
    h_best = np.concatenate([best_s, best_s])
    best_val = compute_c5(h_best)
    return h_best, best_val

    # inequality constraints for each shift

            if k >= 0:
                overl = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])

        if kk == 0:
            continue
        constraints.append({'type': 'ineq', 'fun': make_con(kk)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        h_best = np.clip(h_best, 0.0, 1.0)

    final_val = compute_c5(h_best)
    return h_best.astype(float), final_val, n
