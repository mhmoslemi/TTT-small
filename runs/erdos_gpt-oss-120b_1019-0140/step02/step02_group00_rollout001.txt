import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project vector v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from size len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    """Linear interpolation from larger to smaller size."""
    return upsample(arr, new_n)

def thue_morse(n):
    """Binary Thue‑Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Subgradient for active shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        active = np.where(corr >= max_corr - eps)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, active)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    """Short SLSQP refinement of a candidate."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_small = 64
    n_large = 256
    half_large = n_large // 2

    # ---- Phase 0: gather small‑scale starts ----
    base_small = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        base_small.append(project_onto_box_sum(downsample(init_full, n_small), n_small / 2.0))
    base_small.append(thue_morse(n_small))
    for _ in range(4):
        base_small.append(random_binary_balanced(n_small, rng))
    for _ in range(4):
        v = rng.random(n_small)
        base_small.append(project_onto_box_sum(v, n_small / 2.0))

    # ---- Phase 1: SLSQP on small grid ----
    best_h_small = None
    best_score_small = float('inf')
    for cand in base_small:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=800)
        if sc_opt < best_score_small:
            best_score_small, best_h_small = sc_opt, h_opt.copy()
    # extra random starts while time permits
    while time_left() > 30.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=500)
        if sc_opt < best_score_small:
            best_score_small, best_h_small = sc_opt, h_opt.copy()

    if best_h_small is None:
        best_h_small = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score_small = compute_c5(best_h_small)

    # ---- Phase 2: Upsample to target resolution ----
    h_up = upsample(best_h_small, n_large)
    h_up = project_onto_box_sum(h_up, half_large)
    best_h = h_up
    best_score = compute_c5(best_h)

    # ---- Phase 3: Continuous momentum refinement ----
    rem = time_left()
    if rem > 12.0:
        iters = int(min(40000, max(5000, rem * 850)))  # roughly 0.001 s/iter
        h_ref, sc_ref = gradient_descent_momentum(best_h, n_iter=iters,
                                                  step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref

    # ---- Phase 4: Binary rounding ----
    idx_top = np.argpartition(best_h, -half_large)[-half_large:]
    h_bin = np.zeros(n_large, dtype=float)
    h_bin[idx_top] = 1.0
    score_bin = compute_c5(h_bin)
    if score_bin < best_score:
        best_h, best_score = h_bin, score_bin

    # ---- Phase 5: Binary simulated‑annealing hill‑climb ----
    # Ensure exact half‑ones invariant
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h <= 0.5)[0]
    # small constant for occasional uphill moves
    uphill_prob = 0.001
    while time_left() > 2.0:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_score = compute_c5(h_new)
        if new_score < best_score or rng.random() < uphill_prob:
            best_h, best_score = h_new, new_score
            # update index sets
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h <= 0.5)[0]
            if best_score <= 0.3808:
                break

    # final safety projection and score recompute
    best_h = project_onto_box_sum(best_h, half_large)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
