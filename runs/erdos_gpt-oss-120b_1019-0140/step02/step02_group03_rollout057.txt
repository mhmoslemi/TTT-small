def local_optimize(h):
    n = len(h)
    dx = 2.0/n
    # initial t = compute_c5(h)
    x0 = np.concatenate([h, [dx*0]])? no, t0 = compute_c5(h).
    x0 = np.concatenate([h, [t0]])

    bounds = [(0,1)]*n + [(0, None)]

    def overlap_k(k, h_vec):
        if k>=0:
            return np.dot(h_vec[:n-k], 1 - h_vec[k:]) * dx
        else:
            kk = -k
            return np.dot(h_vec[kk:], 1 - h_vec[:n-kk]) * dx

    constraints = [{'type':'eq', 'fun': lambda x: x[:n].sum() - n/2}]
    for k in range(-n+1, n):
        if k == 0: continue
        constraints.append({'type': 'ineq',
                            'fun': lambda x, k=k: x[-1] - overlap_k(k, x[:n])})

    res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                    constraints=constraints,
                    options={'maxiter':500, 'ftol':1e-8, 'disp': False})
    if res.success:

        # re-project sum if needed

        t_opt = compute_c5(h_opt)
        return h_opt, t_opt
    else:
        # fallback: return original h
        return h, compute_c5(h)

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
   # using binary search on lambda
   low = -np.max(v) - 1.0
   high = (hi - np.min(v)) + 1.0
   for _ in range(50):
       mid = (low + high) * 0.5
       x = np.clip(v + mid, lo, hi)
       s = x.sum()
       if s < target:
           low = mid
       else:
           high = mid
   return np.clip(v + (low+high)*0.5, lo, hi)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project onto [lo,hi]^n while enforcing sum == target."""
    low = -np.max(v) - 1.0
    high = (hi - np.min(v)) + 1.0
    for _ in range(50):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def local_optimize_slsqp(h, maxiter=200):
    """Refine h using SLSQP to reduce its C5."""
    n = h.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t}]
    res = minimize(lambda x: compute_c5(x), h, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if res.success:
        h_opt = project_to_box_sum(np.clip(res.x, 0.0, 1.0), target)
        return h_opt, compute_c5(h_opt)
    return h, compute_c5(h)
