def worst_shift_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # O = corr * dx
    idx = int(np.argmax(corr))  # maybe multiple maxima but choose first
    shift = idx - (n - 1)
    # compute shifted arrays with zero padding
    if shift >= 0:
        h_plus = np.concatenate([np.zeros(shift), h[:n - shift]])
        h_minus = np.concatenate([h[shift:], np.zeros(shift)])
    else:
        s = -shift
        h_plus = np.concatenate([h[s:], np.zeros(s)])  # h[t + shift] where shift negative
        h_minus = np.concatenate([np.zeros(s), h[:n - s]])  # h[t - shift] where shift negative => t + s
    g = dx * (1.0 - h_plus - h_minus)
    return shift, g, idx, corr

def init_h(n, M, rng):
    method = rng.integers(3)
    if method == 0:
        # binary random
        h = np.zeros(n)
        ids = rng.choice(n, M, replace=False)
        h[ids] = 1.0
    elif method == 1:
        # linear decreasing shape
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        # clip and adjust to meet sum if needed
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        # adjust
        if cur < M:
            deficit = M - cur
            # add to smallest values (still <1)
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12: break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12: break

        x = np.arange(n) / (n - 1)

        h = np.clip(h, 0.0, 1.0)

            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12: break

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given (continuous) h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def worst_shift_gradient(h):
    """Return the shift with maximal overlap and the gradient of that overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # shift = -(n-1) .. (n-1)

    if shift >= 0:
        # h[t+shift]  (zero‑padded on the right)
        h_plus = np.concatenate([h[shift:], np.zeros(shift)])
        # h[t-shift]  (zero‑padded on the left)
        h_minus = np.concatenate([np.zeros(shift), h[:n - shift]])
    else:
        s = -shift
        # h[t+shift] = h[t-s]  (zero‑padded on the left)
        h_plus = np.concatenate([np.zeros(s), h[:n - s]])
        # h[t-shift] = h[t+s]  (zero‑padded on the right)
        h_minus = np.concatenate([h[s:], np.zeros(s)])

    g = dx * (1.0 - h_plus - h_minus)
    return shift, g, idx, corr

def init_h(n, M, rng):
    """Generate a feasible initial h (sum = M, 0≤h≤1)."""
    method = rng.integers(3)
    if method == 0:
        # binary random
        h = np.zeros(n)
        ids = rng.choice(n, M, replace=False)
        h[ids] = 1.0
    elif method == 1:
        # linear decreasing shape
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    else:
        # sinusoidal shape, peaks in the centre
        x = np.arange(n) / (n - 1)
        h = np.sin(np.pi * x)
        scaling = M / np.sum(h)
        h = h * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    return h

def search_h_for_n(n, time_budget, rng):
    """Search for a good h of length n within the allotted time."""
    M = n // 2
    best_h = None
    best_c5 = np.inf
    start = time.time()
    use_parent = ('initial_h_values' in globals()) and (len(initial_h_values) == n)

    while time.time() - start < time_budget - 0.1:
        if use_parent:

            cur = np.sum(h)
            if abs(cur - M) > 1e-12:
                h += (M - cur) / n
                h = np.clip(h, 0.0, 1.0)
            use_parent = False
        else:
            h = init_h(n, M, rng)

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
            if best_c5 < 0.3808:
                break

        # gradient‑descent style refinement on the worst shift
        max_iters = 2000
        for _ in range(max_iters):
            if time.time() - start > time_budget - 0.05:

            shift, g, _, _ = worst_shift_gradient(h)
            i_min = int(np.argmin(g))   # most negative gradient
            i_max = int(np.argmax(g))   # most positive gradient
            if g[i_min] >= g[i_max] - 1e-12:

            max_inc = min(1.0 - h[i_min], h[i_max])
            if max_inc <= 0:
                break
            delta = 0.5 * max_inc        # moderate step size
            h[i_min] += delta
            h[i_max] -= delta

            new_c5 = compute_c5(h)

                    break

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function giving a low C5 bound."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # candidate discretisations (even numbers)
    candidate_ns = [80, 96, 112, 128, 144, 160, 176, 192]

    # prioritize a provided parent construction if its size appears
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns = [n0] + candidate_ns
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        alloc = min(200.0, remaining * 0.5)   # generous but bounded slice per n
        h_opt, c5_opt = search_h_for_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global, best_c5_global, best_n_global = h_opt, c5_opt, n
            if best_c5_global < 0.3808:
                break

    # final safety fallback
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        # enforce constraints one last time
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        cur = np.sum(best_h_global)
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h_global += (M - cur) / n
            best_h_global = np.clip(best_h_global, 0.0, 1.0)

    return best_h_global, best_c5_global, n
