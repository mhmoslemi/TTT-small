def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        h = x[:self.n]
        return np.sum(h) - self.n/2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        # return t - overlap*dx >= 0 => constraint >=0
        t = x[self.n]
        return t - overlap * self.dx

def slsqp_refine(h_start, maxiter=500):
    n = h_start.shape[0]
    dx = 2.0 / n
    # initial t is compute_c5
    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    # Could also include k=0 but not needed
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # Project to fix sum (just in case)
        h_opt = project_to_box_sum(h_opt, n/2.0)
        t_opt = compute_c5(h_opt)
        return h_opt, t_opt
    else:
        # return original
        return h_start, compute_c5(h_start)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project vector v onto [lo, hi]^n while fixing sum(v) = target."""
    low = lo - np.max(v) - 1.0
    high = hi - np.min(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx for a given shift k."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        t = x[self.n]
        return t - overlap * self.dx

def slsqp_refine(h_start, maxiter=500):
    """Local refinement via SLSQP on (h, t) to reduce max overlap."""
    n = h_start.shape[0]
    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    try:
        res = minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    except Exception:
        return h_start, compute_c5(h_start)
    if not res.success:
        return h_start, compute_c5(h_start)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h0 = project_to_box_sum(h0, n / 2.0)
    else:
        n = 200  # a modestly larger resolution when no parent solution
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Simulated annealing parameters
    start_temp = 0.01
    end_temp = 1e-5
    base_step = 0.5
    deadline = start_time + budget_s - 0.5  # safety margin
    it = 0

    while time.time() < deadline:
        # Temperature based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 0.5)
        temp = start_temp * (end_temp / start_temp) ** frac

        # Propose a move
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2) * max(0.1, temp / start_temp)
        cand = cur_h + step * d
        cand = project_to_box_sum(cand, n / 2.0)

        cand_c5 = compute_c5(cand)
        diff = cand_c5 - cur_c5
        if diff <= 0 or rng.random() < np.exp(-diff / max(temp, 1e-12)):
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                if best_c5 < 0.3808:
