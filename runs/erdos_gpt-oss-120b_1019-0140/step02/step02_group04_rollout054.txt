def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # add uniform correction to all entries then clip
    vec = vec + diff / len(vec)
    # clip again
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    # distribute remaining diff greedily
    if diff > 0:
        # need to increase
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec

def directed_local_search(h, dx, target_sum, time_limit, start_time, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.1
    n = len(h)
    # small epsilon for gradient threshold
    eps = 1e-8
    # early break if already below target bound?
    target_bound = 0.3808
    while time.time() < time_limit:
        # compute overlap vector and find worst shift
        O_vec = np.correlate(cur_h, 1.0 - cur_h, mode="full") * dx
        max_idx = int(np.argmax(O_vec))
        k_max = max_idx - (n - 1)
        # if current value already below target bound, we can break early
        if cur_val < target_bound - 1e-12:
            break
        # gradient for that shift
        g = gradient_overlap(cur_h, k_max)
        donors = np.where((g > eps) & (cur_h > eps))[0]
        recipients = np.where((g < -eps) & (cur_h < 1 - eps))[0]
        if donors.size == 0 or recipients.size == 0:
            # cannot improve in this direction; reduce step and continue
            step *= 0.5
            if step < 1e-5:
                break
            continue
        # select donor and recipient — use highest gradient magnitude
        if donors.size > 1:
            donor = donors[np.argmax(g[donors])]
        else:
            donor = donors[0]
        if recipients.size > 1:
            recipient = recipients[np.argmin(g[recipients])]
        else:
            recipient = recipients[0]
        max_delta = min(step, cur_h[donor], 1.0 - cur_h[recipient])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[donor] -= delta
        cand[recipient] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.2, 0.5)
        else:
            step *= 0.9
        # occasionally random shuffle to escape stagnation?

    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)

import time
import numpy as np

TARGET_BOUND = 0.3808  # early stopping target


def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Clip to [0,1] and adjust to exact sum=target_sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Continuous random feasible vector (values in [0,1] and exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x.fill(target_sum / n)
    else:
        x *= target_sum / s
    np.clip(x, 0.0, 1.0, out=x)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        idx = rng.permutation(n)
        for i in idx:
            if diff > 0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            else:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    np.clip(x, 0.0, 1.0, out=x)
    return x


def generate_random_binary_feasible(n, target_sum, rng):
    """Binary feasible vector with exactly target_sum ones (target_sum must be int)."""
    m = int(round(target_sum))
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=m, replace=False)
    h[ones] = 1.0
    return h


def gradient_overlap(h, k):
    """Gradient of overlap_k = sum_i h[i]*(1-h[i+k]) w.r.t h."""
    n = len(h)
    idx = np.arange(n)
    g = np.zeros(n, dtype=float)

    idx_plus = idx + k
    mask_plus = (idx_plus >= 0) & (idx_plus < n)
    if np.any(mask_plus):
        g[mask_plus] = 1.0 - h[idx_plus[mask_plus]]

    idx_minus = idx - k
    mask_minus = (idx_minus >= 0) & (idx_minus < n)
    if np.any(mask_minus):
        g[mask_minus] -= h[idx_minus[mask_minus]]
    return g


def directed_local_search(h, dx, target_sum, time_limit, start_time, rng):
    """Iteratively transfer mass from donors (g>0) to recipients (g<0) to reduce max overlap."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.1
    n = len(cur_h)
    eps = 1e-8

    while time.time() < time_limit:
        if cur_val < TARGET_BOUND - 1e-12:
            break

        # worst shift
        O_vec = np.correlate(cur_h, 1.0 - cur_h, mode="full") * dx
        max_idx = int(np.argmax(O_vec))
        k_max = max_idx - (n - 1)

        # gradient for this shift
        g = gradient_overlap(cur_h, k_max)

        donors = np.where((g > eps) & (cur_h > eps))[0]
        recipients = np.where((g < -eps) & (cur_h < 1.0 - eps))[0]

        if donors.size == 0 or recipients.size == 0:
            step *= 0.5
            if step < 1e-5:
                break
            continue

        # select strongest donor / recipient
        donor = donors[np.argmax(g[donors])]
        recipient = recipients[np.argmin(g[recipients])]

        max_delta = min(step, cur_h[donor], 1.0 - cur_h[recipient])
        if max_delta <= 0:
            step *= 0.5
            continue

        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[donor] -= delta
        cand[recipient] += delta

        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-9:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.2, 0.5)
        else:
            step *= 0.9

        # occasional tiny random perturbation to escape plateaus
        if rng.random() < 0.001:
            i, j = rng.integers(n, size=2)
            if i != j:
                d = min(0.01, cur_h[i], 1.0 - cur_h[j])
                if d > 0:
                    cand2 = cur_h.copy()
                    cand2[i] -= d
                    cand2[j] += d
                    val2 = compute_c5_vec(cand2, dx)
                    if val2 < cur_val:
                        cur_h, cur_val = cand2, val2
    # final projection (should be redundant)
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val


def improve_from(start_h, dx, target_sum, start_time, budget_s, rng):
    """Run directed search with remaining time budget."""
    remaining = budget_s - (time.time() - start_time) - 0.5
    if remaining <= 0.1:
        return start_h, compute_c5_vec(start_h, dx)
    time_limit = time.time() + remaining
    return directed_local_search(start_h, dx, target_sum, time_limit, start_time, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
