def project_to_simplex_bounds(x, target):
    # Clip
    x = np.clip(x, 0.0, 1.0)
    # Use iterative method
    # Equivalent to solving: Find lambda such that y_i = min(max(x_i + lambda, 0), 1), and sum y_i = target.
    # This is a piecewise linear problem; can solve using bisection for lambda.
    # lambda range: lower = -1 (makes all x_i + lambda <= 0) => sum = 0; upper = 1 (makes all x_i + lambda >= 1) => sum = n.
    # Actually x_i in [0,1], lambda in [-1, 1]. We need sum = target between 0 and n.
    # Use bisection to find lambda such that sum_i min(max(x_i + lambda, 0), 1) = target.
    # Do 30 iterations.
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo+hi)/2.0, 0.0, 1.0)
    return y

def binary_hill_climb_once(start_bin, time_end, rng):
    n = len(start_bin)
    cur = start_bin.copy()
    cur_c5 = compute_c5(cur.astype(float))
    best_c5 = cur_c5
    best = cur.copy()
    no_improve = 0
    # while loop
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.integers(ones.size)
        j = rng.integers(zeros.size)
        # swap
        cand = cur.copy()
        cand[ones[i]] = 0
        cand[zeros[j]] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best = cand.copy()
            cur = cand.copy()
            cur_c5 = cand_c5
            no_improve = 0
        else:
            no_improve += 1
    return best.astype(float), best_c5

def project_to_sum_bounds(x, target):

    n = x.size
    # Clip to [0,1]
    y = np.clip(x, 0.0, 1.0)

import time

def compute_c5(arr):
    """Approximate C5 via correlation."""
    n = arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_sum_bounds(x, target):
    """
    Euclidean projection of x onto the hyperplane sum(y)=target with 0<=y<=1.
    Uses bisection on a Lagrange multiplier lambda.
    """
    y = np.clip(x, 0.0, 1.0)
    # Quick check
    s = y.sum()
    if abs(s - target) < 1e-12:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(35):
        mid = (lo + hi) * 0.5
        y_mid = np.clip(y + mid, 0.0, 1.0)
        s_mid = y_mid.sum()
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def binary_hill_climb_once(start_bin, time_end, rng):
    """
    Simple swap hill climbing on binary vectors.
    start_bin: 0/1 int array with correct sum.
    """
    n = start_bin.shape[0]

    best = cur.copy()

    # stop after many failed attempts or when time runs out

        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)
        if ones.size == 0 or zeros.size == 0:

        i = rng.integers(ones.size)

            cur_c5 = cand_c5
