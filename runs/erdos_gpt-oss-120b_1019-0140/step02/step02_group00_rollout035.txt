def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    # compute weights
    max_corr = np.max(corr)  # for numerical stability subtract max before exp
    exp_vals = np.exp(alpha * (corr - max_corr))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # compute contributions
    # For each shift index s (0 to 2n-2), shift k = s - (n-1)
    for s, k in enumerate(range(-(n-1), n)):

        if w == 0:
            continue

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    # stable exp
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # iterate shifts:
    for s, k in enumerate(range(-(n-1), n)):
        w = weights[s]
        if w == 0:
            continue
        if k >= 0:
            # first term: grad[:n-k] += w * (1 - h[k:])
            grad[:n-k] += w * (1.0 - h[k:])
            # second term: grad[k:] -= w * h[:n-k]
            grad[k:] -= w * h[:n-k]
        else:
            p = -k
            # first term: grad[p:] += w * (1 - h[:n-p])
            grad[p:] += w * (1.0 - h[:n-p])
            # second term: grad[:n-p] -= w * h[p:]
            grad[:n-p] -= w * h[p:]
    return grad
```

def smooth_grad_descent(h0, n_iter=2000, step0=0.5, alpha_start=1.0, alpha_end=10.0):

    total = n / 2.0

    h = h0.copy()

    # maintain indices of zeros and ones
    zero_idx = np.where(h == 0.0)[0]
    one_idx = np.where(h == 1.0)[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    # temperature schedule
    T0 = 0.005  # initial temperature
    # compute total allowed iters based on time
    # but we also can stop early if time low.
    it = 0
    # pre-compute max_iters if time left large else reduce
    # let's set max_iters = int(min(200000, time_left_func()*2000))
    max_iters = int(min(200000, max(0, time_left_func() * 2000)))
    while it < max_iters and time_left_func() > 0.01:
        it += 1
        if zero_idx.size == 0 or one_idx.size == 0:
            break
        i = rng.choice(zero_idx)
        j = rng.choice(one_idx)
        # propose swap
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        score = compute_c5(h_new)
        delta = score - best_score
        # temperature decreasing
        T = T0 * (1 - it / max_iters)
        accept = False
        if score < best_score:
            accept = True
        else:
            # probability to accept worse
            if T > 0:

            best_score = score
            best_h = h_new.copy()

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term (averaged over given integer shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    grad /= len(shifts)
    return grad

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing style local search on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(200000, max(1, time_left() * 2000)))
    T0 = 0.005
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # pick a random zero and a random one
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        # propose swap
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # ----- parameters -----
    n_small = 64
    n_large = 256
    target_bound = 0.3808

    # ----- candidate generation -----
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_small = downsample(init_full, n_small) if init_full.size != n_small else init_full.copy()
        candidates.append(project_onto_box_sum(init_small, n_small / 2.0))
    candidates.append(thue_morse(n_small).astype(float))
    for _ in range(2):
        candidates.append(random_binary_balanced(n_small, rng).astype(float))
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    best_h = None
    best_score = float('inf')

    # ----- Phase 1: small‑scale smooth gradient descent -----
    for h0 in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = smooth_grad_descent(

            alpha_start=1.0, alpha_end=10.0,
            beta=0.9, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # ----- Phase 2: upsample and momentum refinement -----

    up_score = compute_c5(h_up)
    if up_score < best_score:

    if time_left() > 4.0:
        iters = int(min(20000, max(5000, time_left() * 800)))
        h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=iters,
                                              step0=0.5, beta=0.9,

    # ----- Phase 3: binary hill‑climb (simulated annealing) -----
    if time_left() > 3.0:

        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)

        # small continuous refinement after binary search

            h_ref2, sc_ref2 = momentum_grad_descent(best_h, n_iter=5000,
                                                    step0=0.3, beta=0.9,
                                                    time_left=time_left)
            if sc_ref2 < best_score:

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
