def repair_sum(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # While diff not zero
    n = h.shape[0]
    attempts=0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            # need to add mass, find idx where h<1
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:  # diff < 0
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        attempts += 1
    return h
```

def compute_c5(h):
    """Return the discrete approximation of C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    # shift to roughly meet the target sum
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        # spread the remaining difference uniformly, then fix up
        if diff > 0:
            inc = diff / n
            v = np.clip(v + inc, 0.0, 1.0)
        else:
            dec = -diff / n
            v = np.clip(v - dec, 0.0, 1.0)
        diff = target - v.sum()
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    add = min(diff, 1.0 - v[i])
                    v[i] += add
                    diff -= add
                elif diff < 0 and v[i] > 0.0:
                    sub = min(-diff, v[i])
                    v[i] -= sub
                    diff += sub
    return v
