def anneal_continuous(h, dx, target_sum, time_limit, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    n = len(h)
    start_time = time.time()
    total_time = time_limit - start_time
    while time.time() < time_limit:
        elapsed = time.time() - start_time
        t_frac = elapsed / total_time
        T = max(0.02 * (1 - t_frac), 1e-6) # linear cooling
        # donors and receivers based on bounds
        donors = np.where(cur_h > 1e-12)[0]
        receivers = np.where(cur_h < 1 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(receivers)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        # scale delta: perhaps propose around max_delta * rng.random() * (0.5 + 0.5*T/0.02) ??? keep simple:
        delta = rng.random() * max_delta
        cand_h = cur_h.copy()
        cand_h[i] -= delta
        cand_h[j] += delta
        cand_val = compute_c5_vec(cand_h, dx)
        # acceptance
        if cand_val < cur_val:
            cur_h, cur_val = cand_h, cand_val
            if cand_val < best_val:
                best_h, best_val = cand_h.copy(), cand_val
        else:
            # accept with probability exp(-(cand_val - cur_val)/T)
            prob = np.exp(-(cand_val - cur_val) / T)
            if rng.random() < prob:
                cur_h, cur_val = cand_h, cand_val
        # optional: if best_val already below threshold 0.3808, break early
        if best_val <= 0.3808:
            break
    # final project to sum (to repair rounding issues)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)
    return best_h, best_val

def generate_random_feasible(n, target_sum, rng):
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:

    else:

    x = np.clip(x, 0.0, 1.0)

    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        # distribute by adding to entries < 1 if diff>0, or subtract from >0
        if diff > 0:
            idx = np.where(x < 1.0)[0]

def grad_for_shift(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # term where i is index of h as first factor
        idx1 = np.arange(0, n - k)
        grad[idx1] += (1.0 - h[idx1 + k])
        # term where i is index of h as second factor (negates h_{i - k})
        idx2 = np.arange(k, n)
        grad[idx2] -= h[idx2 - k]
    else:
        kk = -k
        # O_{-kk}
        # term first factor: i from kk..n-1, (1 - h[i - kk])
        idx1 = np.arange(kk, n)
        grad[idx1] += (1.0 - h[idx1 - kk])
        # term second factor: i from 0..n-kk-1, -h[i + kk]
        idx2 = np.arange(0, n - kk)
        grad[idx2] -= h[idx2 + kk]
    return grad

import time
import numpy as np

# ----- Core utilities -----
def compute_c5_vec(h, dx):
    """Maximum overlap integral for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Project vec onto [0,1]^n while enforcing sum(vec) == target_sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # distribute the residual uniformly, then fix any remaining overflow/underflow
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Return a random vector in [0,1]^n with exact sum target_sum."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            rng.shuffle(idx)
            for i in idx:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
                if diff <= 1e-12:
                    break
        else:
            idx = np.where(x > 0.0)[0]
            rng.shuffle(idx)
            for i in idx:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
                if diff >= -1e-12:
                    break
    return np.clip(x, 0.0, 1.0)


def grad_for_shift(h, k):
    """Gradient of overlap O_k with respect to h (for a fixed shift k)."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # term from h_i * (1 - h_{i+k})
        i1 = np.arange(0, n - k)
        grad[i1] += 1.0 - h[i1 + k]
        # term from -h_{i-k} * h_i (appears when i is the second index)
        i2 = np.arange(k, n)
        grad[i2] -= h[i2 - k]
    else:
        kk = -k
        i1 = np.arange(kk, n)
        grad[i1] += 1.0 - h[i1 - kk]
        i2 = np.arange(0, n - kk)
        grad[i2] -= h[i2 + kk]
    return grad


def anneal_continuous(start_h, dx, target_sum, deadline, rng):
    """
    Simulated annealing on the simplex with box bounds.
    Moves preserve the sum by transferring mass between a donor and a receiver.
    """
    cur_h = start_h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    n = len(cur_h)

    start = time.time()
    init_T = 0.02  # initial temperature (scale of C5)
    while True:
        now = time.time()
        if now >= deadline:
            break
        # linear temperature schedule
        frac = max(0.0, (deadline - now) / (deadline - start))
        T = init_T * frac + 1e-8

        # compute current O_k and locate the most critical shift
        O = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        k_idx = int(np.argmax(O))
        k_shift = k_idx - (n - 1)

        # gradient for that shift
        grad = grad_for_shift(cur_h, k_shift)

        # donors: positive gradient (decreasing h reduces O_k)
        donors = np.where((grad > 1e-12) & (cur_h > 1e-12))[0]
        # receivers: negative gradient (increasing h reduces O_k)
        recvs = np.where((grad < -1e-12) & (cur_h < 1.0 - 1e-12))[0]

        if len(donors) == 0 or len(recvs) == 0:
            # fallback to random donor/receiver
            donors = np.where(cur_h > 1e-12)[0]
            recvs = np.where(cur_h < 1.0 - 1e-12)[0]
            if donors.size == 0 or recvs.size == 0:
                break
            i = rng.choice(donors)
            j = rng.choice(recvs)
        else:
            # pick strongest donor/receiver
            i = donors[np.argmax(grad[donors])]
            j = recvs[np.argmin(grad[recvs])]

        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        # step size scales with temperature
        delta = max_delta * rng.random() * (0.5 + 0.5 * frac)
        cand_h = cur_h.copy()
        cand_h[i] -= delta
        cand_h[j] += delta

        cand_val = compute_c5_vec(cand_h, dx)

        if cand_val < cur_val:
            cur_h, cur_val = cand_h, cand_val
            if cand_val < best_val:
                best_h, best_val = cand_h.copy(), cand_val
                if best_val <= 0.3808:
                    break
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(cand_val - cur_val) / T):
                cur_h, cur_val = cand_h, cand_val

    # final projection to satisfy the sum constraint exactly
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)
    return best_h, best_val


def hill_climb_binary(bin_h, cur_val, dx, deadline, rng):
    """Swap hill climbing on a binary vector."""
    best_h = bin_h.copy()
    best_val = cur_val
    n = len(bin_h)
    while time.time() < deadline:
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val:
            best_h, best_val = cand, cand_val
            if best_val <= 0.3808:
                break
    return best_h.astype(float), best_val


# ----- Main entry point -----
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size: try 100 points (preferred) unless a parent provides another size.
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        use_parent = True
    else:
        n = 100
        use_parent = False

    target_sum = n / 2.0
    dx = 2.0 / n

    best_h = None
    best_val = np.inf

    # Helper to initialise a feasible vector
    def init_feasible():
        if use_parent:
            # start from the parent construction (projected to exact sum)
            h0 = np.array(initial_h_values, dtype=float)
            h0 = np.clip(h0, 0.0, 1.0)
            return project_to_sum(h0, target_sum)
        else:
            return generate_random_feasible(n, target_sum, rng)

    # Initial candidate
    cur0 = init_feasible()
    val0 = compute_c5_vec(cur0, dx)
    if val0 < best_val:
        best_h, best_val = cur0.copy(), val0

    # Main annealing loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.85:
            break
        remaining = budget_s - elapsed
        # allocate at most 12 seconds per annealing run
        run_time = min(remaining * 0.4, 12.0)
        deadline = time.time() + run_time

        h_start = generate_random_feasible(n, target_sum, rng)
        h_opt, val_opt = anneal_continuous(h_start, dx, target_sum, deadline, rng)

        if val_opt < best_val:
            best_h, best_val = h_opt.copy(), val_opt
            if best_val <= 0.3808:
                break

    # Optional binary refinement (rounding + swap hill‑climb)
    time_left = budget_s - (time.time() - start_time)
    if time_left > 4.0 and best_h is not None:
        # round to binary by taking the largest entries
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argpartition(-best_h, int(target_sum))[: int(target_sum)]
        binary_h[top_idx] = 1
        bin_val = compute_c5_vec(binary_h.astype(float), dx)
        deadline = time.time() + time_left
        bin_h_opt, bin_val_opt = hill_climb_binary(binary_h, bin_val, dx, deadline, rng)
        if bin_val_opt < best_val:
            best_h, best_val = bin_h_opt, bin_val_opt

    # Final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, float(best_val), n
