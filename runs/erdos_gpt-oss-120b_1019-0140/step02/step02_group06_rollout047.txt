import time, numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_feasible(x, target):
    # Clip first
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) <= 1e-12:
        return x
    # allocate diff to free capacity
    # Use iterative algorithm
    if diff > 0:
        # need to increase sum
        # indices sorted by current value ascending
        idx = np.argsort(x)
        for i in idx:
            if diff <= 0:
                break
            space = 1.0 - x[i]
            inc = min(space, diff)
            x[i] += inc
            diff -= inc
    else:
        diff = -diff
        # need to decrease sum, decrease largest values first
        idx = np.argsort(-x)
        for i in idx:
            if diff <= 0:
                break
            avail = x[i]
            dec = min(avail, diff)
            x[i] -= dec
            diff -= dec
    # After loop, clip again to be safe
    return np.clip(x, 0.0, 1.0)
```

    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    beta_max = 30.0
    alpha0 = 0.2

        else:
            w /= w_sum

        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_feasible(x, target):
    """Clip to [0,1] and adjust to achieve sum == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    if diff > 0:
        # need to increase some entries
        idx = np.argsort(x)               # smallest first
        for i in idx:
            if diff <= 0:
                break
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
    else:
        diff = -diff
        idx = np.argsort(-x)              # largest first
        for i in idx:
            if diff <= 0:
                break
            dec = min(diff, x[i])
            x[i] -= dec
            diff -= dec
    return np.clip(x, 0.0, 1.0)

def optimize_h(h0, time_budget, rng):
    """Projected gradient descent with soft‑max smoothing, momentum and occasional shake."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    beta_max = 30.0
    alpha0 = 0.2
    gamma = 0.9               # momentum factor
    m = np.zeros_like(h)      # momentum buffer
    no_improve = 0
    while time.time() - start < time_budget:
        # correlation and current C5
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0 / n)

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase smoothing slowly
        if beta < beta_max:
            beta = min(beta_max, beta * 1.002)

        # soft‑max weights (stable)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the smoothed objective
        g = np.zeros_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                kk = -k
                tail = slice(kk, n)
                head = slice(0, n - kk)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # remove any mean component (preserve total mass)
        g -= np.mean(g)

        # momentum update
        m = gamma * m + (1.0 - gamma) * g

        # decaying step size (exponential) with a small floor
        elapsed = time.time() - start
        alpha = alpha0 * np.exp(-elapsed / time_budget) + 0.01 * alpha0

        # tentative step and projection
        h_new = h - alpha * m
        h_new = project_to_feasible(h_new, target)

        # acceptance rule
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        # random shake if stuck too long
        if no_improve > 5000:
            h = project_to_feasible(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    return best_h, best_c5
