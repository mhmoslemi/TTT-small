def gradient_descent(h0, target, time_limit, rng):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 80.0
    alpha0 = 0.15
    no_improve = 0
    while time.time() - start < time_limit:
        # compute gradient
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # update beta
        if beta < max_beta:
            beta = min(max_beta, beta * 1.002)

        # softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w /= w.sum()

        # gradient
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]

        # zero-mean direction
        g -= g.mean()

        # step size decay
        elapsed = time.time() - start
        alpha = alpha0 * max(0.05, 1.0 - elapsed / time_limit)

        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)

        # evaluate improvement
        c5_new = compute_c5(h_new)

        if c5_new < c5 or rng.random() < 0.002:
            h = h_new

        # random shake
        if no_improve > 2500:
            h = project_sum_and_bounds(h + rng.normal(scale=0.015, size=n), target)
            no_improve = 0

    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Return max overlap integral C5 for a step function h."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce exact sum(x)==target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute uniformly, then re‑clip iteratively
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def gradient_descent(h0, target, time_limit, rng):
    """Soft‑max based gradient descent on max‑overlap surrogate."""
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 80.0
    alpha0 = 0.15
    no_improve = 0
    offset = n - 1
    while time.time() - start < time_limit:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w /= w.sum()

        g = np.zeros(n)
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]

        g -= g.mean()
        elapsed = time.time() - start
        alpha = alpha0 * max(0.05, 1.0 - elapsed / time_limit)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)

        if c5_new < c5 or rng.random() < 0.002:
            h = h_new

        if no_improve > 2500:
            h = project_sum_and_bounds(h + rng.normal(scale=0.015, size=n), target)
            no_improve = 0
    return best_h, best_c5

def max_subgradient_descent(h, target, iters, rng):
    """Iterative sub‑gradient descent using only the worst shift."""
    n = len(h)
    for it in range(iters):
        v = np.correlate(h, 1.0 - h, mode='full')
        k = np.argmax(v) - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
            g[tail] = 1.0 - h[head]
            g[head] -= h[tail]
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
            g[tail] = 1.0 - h[head]
            g[head] -= h[tail]
        g -= g.mean()
        alpha = 0.02 / (1.0 + it * 0.01)
        h = h - alpha * g
        h = project_sum_and_bounds(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = rng.random(n)
        h0 = project_sum_and_bounds(h0, n / 2.0)

    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    # Initial descent using a reasonable share of the budget
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    init_time = min(remaining * 0.4, 150.0)
    best_h, best_c5 = gradient_descent(h0, target, init_time, rng)

    # Repeated short restarts
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    restart_time = 4.0
    while remaining > restart_time + 1.0:
        if rng.random() < 0.6:
            h_start = best_h + rng.normal(scale=0.07, size=n)
        else:
            h_start = rng.random(n)
        h_start = project_sum_and_bounds(h_start, target)
        cand_h, cand_c5 = gradient_descent(h_start, target, restart_time, rng)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h
        elapsed = time.time() - start
        remaining = budget_s - elapsed

    # Final refinement with pure max‑subgradient
    remaining = max(0.0, budget_s - (time.time() - start))
    if remaining > 0.2:
        iters = int(min(20000, remaining * 2500))
        best_h = max_subgradient_descent(best_h, target, iters, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
