import time, numpy as np, cvxpy as cp

def solve_milp(n, budget_s, start_time):
    # Build model
    X = cp.Variable(n, boolean=True)
    # y matrix (n x n) boolean - but we only need upper triangle (i < j)
    Y = cp.Variable((n, n), boolean=True)
    constraints = []
    target_sum = n // 2 # n/2 is integer if n even
    constraints.append(cp.sum(X) == target_sum)
    # linking constraints
    # Use broadcasting?
    # We'll loop over i,j.
    for i in range(n):
        for j in range(i+1, n):
            # Y[i,j] <= X[i]
            constraints.append(Y[i,j] <= X[i])
            constraints.append(Y[i,j] <= X[j])
            constraints.append(Y[i,j] >= X[i] + X[j] - 1)
    # For diagonal maybe set Y[i,i] = X[i] for completeness
    for i in range(n):
        constraints.append(Y[i,i] == X[i])
    # R variable
    R = cp.Variable(integer=True)
    constraints.append(R >= 0)
    # Each shift k
    for k in range(1, n):
        # sum Y[i,i+k] for i = 0 .. n-1-k
        idx = np.arange(n-k)
        constraints.append(cp.sum(Y[idx, idx + k]) >= R)
    # maximize R
    obj = cp.Maximize(R)
    prob = cp.Problem(obj, constraints)
    # Solve, with time limit.
    # Determine remaining budget; we want to reserve some time for final check.
    remaining = budget_s - (time.time() - start_time) - 0.5
    if remaining <= 0:
        return None, None
    # Try solver list
    sol = None
    for solver in ["ECOS_BB", "GLPK_MI", "CBC"]:
        try:
            prob.solve(solver=solver, warm_start=True, verbose=False, max_iters=200000)
            # cvxpy has max_iter param for ECOS_BB?
        except Exception as e:
            continue
        if prob.status in ["optimal", "optimal_inaccurate"]:
            sol = X.value
            break
    if sol is None:
        return None, None
    # Round
    x_bin = np.rint(sol).astype(float)
    # enforce sum exactly
    cur_sum = np.sum(x_bin)
    diff = target_sum - cur_sum
    if diff != 0:
        # adjust by flipping bits
        # If diff >0 need to turn some zeros to 1
        idxs = np.where(x_bin == 0)[0]
        if diff > 0:
            # flip diff zeros to 1
            x_bin[idxs[:int(diff)]] = 1
        else:
            idxs1 = np.where(x_bin == 1)[0]
            x_bin[idxs1[:int(-diff)]] = 0
    return x_bin, compute_c5(x_bin)

def solve_milp_binary(n, target_sum, start_h, time_limit):
    # start_h: optional array of shape (n,) for warm start (float)
    # time_limit: seconds allowed for solving (value might not be able to enforce but we can attempt)
    X = cp.Variable(n, boolean=True)
    Y = cp.Variable((n, n), boolean=True)
    C = cp.Variable()
    constraints = []
    constraints.append(cp.sum(X) == target_sum)
    # linking constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(Y[i, j] <= X[i])
            constraints.append(Y[i, j] <= X[j])
            constraints.append(Y[i, j] >= X[i] + X[j] - 1)
    # for completeness, set Y[i,i] = X[i]
    for i in range(n):
        constraints.append(Y[i, i] == X[i])
    # shift constraints
    for d in range(-n+1, n):
        if d == 0:
            continue
        if d > 0:
            idx_i = np.arange(n - d)
            idx_j = idx_i + d
        else:
            idx_i = np.arange(-d, n)
            idx_j = idx_i + d
        constraints.append(cp.sum(X[idx_i]) - cp.sum(Y[idx_i, idx_j]) <= C)
    constraints.append(C >= 0)
    # objective
    obj = cp.Minimize(C)
    prob = cp.Problem(obj, constraints)
    # warm start
    if start_h is not None:
        # round start_h to binary for X
        x0 = np.rint(start_h).astype(int)
        # ensure exactly target sum
        cur = np.sum(x0)
        diff = target_sum - cur
        # adjust bits as needed
        if diff != 0:
            if diff > 0:
                zeros = np.where(x0 == 0)[0]
                x0[zeros[:diff]] = 1
            else:
                ones = np.where(x0 == 1)[0]
                x0[ones[: -diff]] = 0
        X.value = x0
        # set Y initial as outer product
        Y.value = np.outer(x0, x0)
    # solve with attempt of solver list
    sol = None
    for solver in ['GLPK_MI', 'CBC', 'ECOS_BB']:
        try:
            prob.solve(solver=solver, verbose=False, warm_start=True)
        except Exception:
            continue
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            sol = X.value
            C_val = C.value
            break
    if sol is None:
        return None, None
    # round solution
    x_bin = np.rint(sol).astype(float)
    # enforce exact sum
    cur_sum = np.sum(x_bin)
    diff = target_sum - cur_sum
    if diff != 0:
        if diff > 0:
            zeros = np.where(x_bin == 0)[0]
            x_bin[zeros[:int(diff)]] = 1
        else:
            ones = np.where(x_bin == 1)[0]
            x_bin[ones[:int(-diff)]] = 0
    # final safety clip
    x_bin = np.clip(x_bin, 0.0, 1.0)
    return x_bin, compute_c5(x_bin)

            if cand_c5_2 < best_c5:
                best_h = cand_h2
                best_c5 = cand_c5_2
    # final rounding and sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = np.sum(best_h)
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # adjust by uniform distribution
        best_h += diff / n

        new = compute_c5(h_new)
        dE = new - cur
        T = init_T * (1 - it / max_iter) + final_T * (it / max_iter)
        if dE <= 0 or rng.random() < np.exp(-dE / (T + 1e-18)):
            h = h_new
            cur = new
            if new < best:
                best = new
                best_h = h_new.copy()
        it += 1
    # enforce exact sum
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best

def binary_hill_climb(start_bin, time_end, rng):
    n = len(start_bin)
    best = start_bin.copy()
    best_c5 = compute_c5(best.astype(float))
    no_imp = 0
    while time.time() < time_end and no_imp < 4000:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        c5 = compute_c5(cand.astype(float))
        if c5 + 1e-12 < best_c5:
            best = cand
            best_c5 = c5
            no_imp = 0
        else:
            no_imp += 1
    return best.astype(float), best_c5

def solve_milp_binary(n, target_sum, start_h, time_limit):
    X = cp.Variable(n, boolean=True)
    Y = cp.Variable((n, n), boolean=True)
    C = cp.Variable()
    cons = [cp.sum(X) == target_sum, C >= 0]
    # linking constraints
    for i in range(n):
        for j in range(i + 1, n):
            cons.append(Y[i, j] <= X[i])
            cons.append(Y[i, j] <= X[j])
            cons.append(Y[i, j] >= X[i] + X[j] - 1)
    # shift constraints
    for d in range(-n + 1, n):
        if d == 0:
            continue
        if d > 0:
            idx_i = np.arange(n - d)
            idx_j = idx_i + d
        else:
            idx_i = np.arange(-d, n)
            idx_j = idx_i + d
        cons.append(cp.sum(X[idx_i]) - cp.sum(Y[idx_i, idx_j]) <= C)
    prob = cp.Problem(cp.Minimize(C), cons)
    # warm start if possible
    if start_h is not None:
        x0 = np.rint(start_h).astype(int)
        cur = x0.sum()
        diff = target_sum - cur
        if diff != 0:
            if diff > 0:
                zeros = np.where(x0 == 0)[0]
                x0[zeros[:diff]] = 1
            else:
                ones = np.where(x0 == 1)[0]
                x0[ones[: -diff]] = 0
        X.value = x0
        Y.value = np.outer(x0, x0)
    # try solvers
    solved = False
    for solver in ("GLPK_MI", "CBC", "ECOS_BB"):
        try:
            prob.solve(solver=solver, warm_start=True, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                solved = True
                break
        except Exception:
            continue
    if not solved or X.value is None:
        return None, None
    x_bin = np.rint(X.value).astype(float)
    # enforce exact sum
    cur = x_bin.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        if diff > 0:
            zeros = np.where(x_bin == 0)[0]
            x_bin[zeros[:int(diff)]] = 1
        else:
            ones = np.where(x_bin == 1)[0]
            x_bin[ones[:int(-diff)]] = 0
    x_bin = np.clip(x_bin, 0.0, 1.0)
    return x_bin, compute_c5(x_bin)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        n = len(initial_h_values)

        if h_bin is not None and c5_val is not None:
            best_h, best_c5 = h_bin, c5_val
    # fallback heuristics if needed
    if best_h is None or best_c5 > 0.3808:
        # start point for hill climbing
        if init_h is not None:
            start_vec = (init_h >= 0.5).astype(int)
        else:
            start_vec = np.zeros(n, dtype=int)
            start_vec[rng.choice(n, target, replace=False)] = 1
        end = start + budget_s - 0.5
        h_hc, c5_hc = binary_hill_climb(start_vec, end, rng)
        if c5_hc < best_c5:
            best_h, best_c5 = h_hc, c5_hc
        # optional annealing
        if time.time() < end - 1.0:
            h_ann, c5_ann = continuous_anneal(best_h.copy(), end, rng)
            if c5_ann < best_c5:
                best_h, best_c5 = h_ann, c5_ann
    # final feasibility fix
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    diff = target - cur_sum
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target - best_h.sum()
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target - best_h.sum()
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
