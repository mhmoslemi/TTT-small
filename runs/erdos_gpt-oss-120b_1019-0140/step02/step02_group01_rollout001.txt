def project_to_simplex_box(v, M, rng, max_iter=10):
    # ensures 0 <= v <= 1 and sum(v) = M
    for _ in range(max_iter):
        np.clip(v, 0, 1, out=v)
        cur = v.sum()
        diff = M - cur
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        free = (v > 0) & (v < 1)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds; distribute uniformly
            v += diff / len(v)
            continue
        v[free] += diff / cnt
    np.clip(v, 0, 1, out=v)
    cur = v.sum()
    if abs(cur - M) > 1e-9:
        v += (M - cur) / len(v)
        np.clip(v, 0, 1, out=v)
    return v

def anneal_optimize(initial_h, time_budget, rng):
    h_best = initial_h.copy()
    best_c5 = compute_c5(h_best)
    h_curr = h_best.copy()
    curr_c5 = best_c5

def optimize_n(n, total_time, rng, init_h=None):
    M = n // 2
    best_h = None
    best_c5 = np.inf
    # include initial provided if size matches 
    if init_h is not None and len(init_h) == n:
        h0 = init_h.copy()
        c5 = compute_c5(h0)
        best_h, best_c5 = h0, c5
    # calculate minimal time per restart
    restart_time = min(15.0, total_time * 0.2)  # each restart 15 sec or 20% of remaining
    start_global = time.time()
    while time.time() - start_global < total_time:
        # allocate time for this restart
        remaining = total_time - (time.time() - start_global)
        t_alloc = min(restart_time, remaining)
        # generate start h
        if best_h is not None and rng.random() < 0.2:
            # start from best with small perturbation
            h_start = best_h.copy()
            # perturb: random mass transfer
            for _ in range(3):
                h_tmp = propose_move(h_start, rng)
                if h_tmp is not None:
                    h_start = h_tmp
        else:
            h_start = random_feasible(n, rng)
        # run annealing
        h_opt, c5_opt = anneal_optimize(h_start, t_alloc, rng)
        if c5_opt < best_c5 - 1e-12:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break
        # early exit if remaining time small
        if time.time() - start_global > total_time - 0.5:
            break
    # final greedy hill climbing: do pairwise mass transfer moves greedily
    if best_h is not None:
        h_curr = best_h.copy()
        curr_c5 = best_c5
        # define max improvement loops
        start_greedy = time.time()
        while time.time() - start_greedy < min(30.0, total_time - (time.time() - start_global)):
            # try many proposals and accept improvement only
            h_new = propose_move(h_curr, rng)
            if h_new is None:
                continue
            c5_new = compute_c5(h_new)

                curr_c5 = c5_new

                    best_h, best_c5 = h_curr.copy(), curr_c5
                # break to restart
            # maybe break early after some trials without improvement

import time
import numpy as np

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


# ----------------------------------------------------------------------
def project_to_simplex_box(v: np.ndarray, M: float, max_iter: int = 12) -> np.ndarray:
    """Project v onto the box [0,1] with sum(v)=M (approximately, exact after call)."""
    for _ in range(max_iter):
        np.clip(v, 0.0, 1.0, out=v)
        cur = v.sum()
        diff = M - cur
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        cnt = int(free.sum())
        if cnt == 0:
            v += diff / v.size
            continue
        v[free] += diff / cnt
    np.clip(v, 0.0, 1.0, out=v)
    cur = v.sum()
    if abs(cur - M) > 1e-9:
        v += (M - cur) / v.size
        np.clip(v, 0.0, 1.0, out=v)
    return v


# ----------------------------------------------------------------------
def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random h with entries in [0,1] and sum = n/2."""
    M = n // 2
    h = rng.exponential(1.0, size=n)
    h = h / h.sum() * M
    return project_to_simplex_box(h, M)


# ----------------------------------------------------------------------
def propose_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """Create a new feasible h by moving mass from one index to another."""
    n = h.shape[0]
    i, j = rng.choice(n, 2, replace=False)
    # attempt move i <- j
    max_up = 1.0 - h[i]
    max_down = h[j]
    delta_max = min(max_up, max_down)
    if delta_max <= 0.0:
        # try the opposite direction
        max_up = 1.0 - h[j]
        max_down = h[i]
        delta_max = min(max_up, max_down)
        if delta_max <= 0.0:
            return None
        i, j = j, i
    delta = rng.random() * delta_max
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new


# ----------------------------------------------------------------------
def anneal_optimize(initial_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Simulated‑annealing search starting from initial_h."""
    best_h = initial_h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    start = time.time()
    T = 0.005          # initial temperature
    decay = 0.9999     # geometric cooling factor
    it = 0
    while time.time() - start < time_budget:
        h_new = propose_move(cur_h, rng)
        if h_new is None:
            continue
        new_val = compute_c5(h_new)
        dE = new_val - cur_val
        if dE < 0.0 or rng.random() < np.exp(-dE / T):
            cur_h = h_new
            cur_val = new_val
            if cur_val + 1e-12 < best_val:
                best_h = cur_h.copy()
                best_val = cur_val
                if best_val < 0.3808:
                    break
        T *= decay
        it += 1
        if it % 5000 == 0 and T < 1e-8:
            T = 0.005
    return best_h, best_val


# ----------------------------------------------------------------------
def greedy_refine(h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Deterministic hill‑climbing: accept only improving moves."""
    best_h = h.copy()
    best_val = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_budget:
        h_new = propose_move(best_h, rng)
        if h_new is None:
            continue
        new_val = compute_c5(h_new)
        if new_val + 1e-12 < best_val:
            best_h = h_new
            best_val = new_val
            if best_val < 0.3808:
                break
    return best_h, best_val


# ----------------------------------------------------------------------
def optimise_for_n(n: int, total_budget: float, rng: np.random.Generator, init_h: np.ndarray | None):
    """Multi‑restart annealing for a fixed discretisation size n."""
    M = n // 2
    best_h = None
    best_val = np.inf
    start_all = time.time()

    # use given initial h if available
    if init_h is not None and init_h.shape[0] == n:
        cand = init_h.copy()
        val = compute_c5(cand)
        best_h, best_val = cand, val

    # time allocation per restart (15 s or 15 % of remaining, whichever smaller)
    while time.time() - start_all < total_budget:
        remaining = total_budget - (time.time() - start_all)
        if remaining < 5.0:
            break
        restart_time = min(15.0, remaining * 0.15)

        # start point: either perturb the current best or fresh random feasible
        if best_h is not None and rng.random() < 0.3:
            h0 = best_h.copy()
            for _ in range(3):
                cand = propose_move(h0, rng)
                if cand is not None:
                    h0 = cand
        else:
            h0 = random_feasible(n, rng)

        # annealing
        h_opt, val_opt = anneal_optimize(h0, restart_time, rng)
        if val_opt + 1e-12 < best_val:
            best_h, best_val = h_opt, val_opt
            if best_val < 0.3808:
                break

    # final deterministic refinement (at most 30 s or whatever remains)
    time_left = total_budget - (time.time() - start_all)
    if best_h is not None and time_left > 5.0:
        refine_time = min(30.0, time_left)
        h_ref, val_ref = greedy_refine(best_h, refine_time, rng)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref

    return best_h, best_val


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the allotted budget."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # Candidate discretisations (even numbers, modest size)
    candidates = [80, 96, 112, 128, 144, 160]

    # Prefer the parent construction if present
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        n0 = parent.shape[0]
        if n0 % 2 == 0:
            if n0 not in candidates:
                candidates.insert(0, n0)
            else:
                candidates = [n0] + [x for x in candidates if x != n0]

    best_h = None
    best_val = np.inf
    best_n = None

    # Simple allocation: give each n up to 300 s, but never exceed remaining budget
    for n in candidates:
        elapsed = time.time() - start
        if elapsed > budget_s - 5.0:
            break
        remaining = budget_s - elapsed
        alloc = min(300.0, remaining * 0.6)

        init_h = None
        if "initial_h_values" in globals():
            tmp = np.asarray(initial_h_values, dtype=float)
            if tmp.shape[0] == n:
                init_h = tmp

        h_opt, val_opt = optimise_for_n(n, alloc, rng, init_h)

        if h_opt is not None and val_opt < best_val:
            best_h, best_val, best_n = h_opt.copy(), val_opt, n
            if best_val < 0.3808:
                break

    # Fallback in the unlikely event nothing was found
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5, dtype=float)
        best_val = compute_c5(best_h)

    # Final sanity enforcement of constraints
    n = best_n
    M = n // 2
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - M) > 1e-9:
        best_h += (M - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, best_n
