        h_new[i] -= delta
        h_new[j] += delta
        # Ensure bounds (should be)

def project_to_feasible(x, target):
    # Clip
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # if diff>0: need to add mass, choose indices where x<1.
    # if diff<0: need to subtract mass, choose indices where x>0.
    # We'll use while loop with random shuffle to distribute.
    # Use upto maybe 1000 iterations (should converge quickly).
    max_iters = 1000
    for _ in range(max_iters):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(x < 1)[0]
            if idx.size == 0:
                break
            # random select a subset? We'll distribute evenly:
            caps = 1 - x[idx]
            total_cap = caps.sum()
            if total_cap < diff - 1e-12:
                # allocate all possible, then diff remains, cannot satisfy constraints
                # set those to 1
                x[idx] = 1.0
                diff = target - x.sum()
                continue
            # Determine increments proportionally
            inc = diff * caps / total_cap
            x[idx] += inc
        else:
            idx = np.where(x > 0)[0]
            if idx.size == 0:
                break
            caps = x[idx]
            total_cap = caps.sum()
            if total_cap < -diff - 1e-12:
                # all to zero
                x[idx] = 0.0
                diff = target - x.sum()
                continue
            dec = (-diff) * caps / total_cap
            x[idx] -= dec
        # Ensure numeric stability
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
    return x

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given h."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_feasible(x, target):
    """Clip to [0,1] and enforce sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # Distribute the remaining difference proportionally to available capacity
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size == 0:
                break
            caps = 1.0 - x[idx]
            total = caps.sum()
            if total < diff - 1e-12:
                x[idx] = 1.0
                diff = target - x.sum()
                continue
            inc = diff * caps / total
            x[idx] += inc
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size == 0:
                break
            caps = x[idx]
            total = caps.sum()
            if total < -diff - 1e-12:
                x[idx] = 0.0
                diff = target - x.sum()
                continue
            dec = (-diff) * caps / total
            x[idx] -= dec
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
    return x

def subgrad_descent(h0, time_budget, rng):
    """Subgradient descent on the max-overlap objective."""
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    v = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = v.max() * dx
    start = time.time()
    step0 = 0.2
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        maxv = v.max()
        c5 = maxv * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # indices achieving the maximum (tolerance for floating error)
        max_idx = np.where(np.isclose(v, maxv, atol=1e-12))[0]
        grad = np.zeros(n)
        offset = n - 1
        for idx in max_idx:
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            grad[tail] += (1.0 - h[head])
            grad[head] -= h[tail]
        # zero‑mean to keep total mass unchanged
        grad -= grad.mean()
        elapsed = time.time() - start
        step = step0 * max(0.01, 1.0 - elapsed / time_budget)
        h_new = h - step * grad
        h_new = project_to_feasible(h_new, target)
        v_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = v_new.max() * dx
        # accept if better or with tiny random chance
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        # occasional random jitter
        if rng.random() < 0.005:
            h = project_to_feasible(h + rng.normal(scale=0.02, size=n), target)
    return best_h, best_c5

def simulated_annealing(h0, time_budget, rng, target, dx):
    """Simple SA moving mass between two points."""
    h = h0.copy()
    n = len(h)
    cur_c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    T0 = 0.02
    while time.time() - start < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx
        diff = c5_new - cur_c5
        elapsed = time.time() - start
        T = T0 * max(0.01, 1.0 - elapsed / time_budget)
        if diff < 0 or rng.random() < np.exp(-diff / T):
            h = h_new
            cur_c5 = c5_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        init0 = np.array(initial_h_values, dtype=float)
        n = init0.size
    else:
        n = 80
        init0 = None

    target = n / 2.0
    dx = 2.0 / n

    # Prepare a set of starting points (include parent if present)
    start_vectors = []
    if init0 is not None:
        start_vectors.append(project_to_feasible(init0.copy(), target))
    for _ in range(4):
        vec = rng.random(n)
        start_vectors.append(project_to_feasible(vec, target))

    best_h = None
    best_c5 = np.inf

    # Multi‑start subgradient descent
    remaining = budget_s - (time.time() - start)
    for idx, h0 in enumerate(start_vectors):
        remaining = budget_s - (time.time() - start)
        if remaining <= 5.0:
            break
        runs_left = len(start_vectors) - idx
        alloc = min(200.0, max(5.0, remaining * 0.8 / runs_left))
        h_opt, c5_val = subgrad_descent(h0, alloc, rng)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()

    # If nothing succeeded, fall back to a random feasible vector
    if best_h is None:
        best_h = project_to_feasible(rng.random(n), target)
        best_c5 = compute_c5(best_h)

    # Finish with simulated annealing using remaining time
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        best_h, best_c5 = simulated_annealing(best_h, remaining, rng, target, dx)

    # Final safety projection and evaluation
    best_h = project_to_feasible(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
