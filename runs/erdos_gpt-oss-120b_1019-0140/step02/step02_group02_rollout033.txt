def slsqp_optimize(h_start, deadline, rng):
    n = h_start.shape[0]
    target_sum = n/2.0
    dx = 2.0/n
    # Create constraints and bounds as before
    sum_con = lambda x: np.sum(x[:n]) - target_sum
    constraints = [{"type":"eq","fun": sum_con}]
    # overlaps constraints
    def make_con(k):
        def con(x, k=k):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])
            return x[-1] - overl * dx
        return con
    for kk in range(-n+1, n):
        if kk == 0: continue
        constraints.append({"type":"ineq","fun": make_con(kk)})
    bounds = [(0.0,1.0)]*n + [(0.0, None)]
    t_start = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t_start]])
    try:
        res = minimize(fun=lambda x: x[-1], x0=x0, method="SLSQP", bounds=bounds,
                       constraints=constraints, options={"maxiter":2000, "ftol":1e-9, "disp": False})
    except Exception:
        return None, None
    if not res.success:
        return None, None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce sum exactly (project)
    cur = h_opt.sum()
    if abs(cur - target_sum) > 1e-9:
        h_opt += (target_sum - cur)/n
        h_opt = np.clip(h_opt,0.0,1.0)
    val = compute_c5(h_opt)
    return h_opt, val

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        inc = diff / n
        v = np.clip(v + inc, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    d = min(diff, 1.0 - v[i])
                    v[i] += d
                    diff -= d
                elif diff < 0 and v[i] > 0.0:
                    d = min(-diff, v[i])
                    v[i] -= d
                    diff += d
    return v

def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    v[idx] = 1.0
    return v

def slsqp_optimize(h_start, deadline, rng):
    """Run one SLSQP trial from h_start, respecting the exact sum constraint."""
    n = h_start.shape[0]
    target = n / 2.0
    dx = 2.0 / n

    def sum_con(x):
        return np.sum(x[:n]) - target

    constraints = [{"type": "eq", "fun": sum_con}]
    def make_con(k):
        def con(x, k=k):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])
            return x[-1] - overl * dx
        return con
    for kk in range(-n + 1, n):
        if kk == 0:
            continue
        constraints.append({"type": "ineq", "fun": make_con(kk)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])

    try:
        res = minimize(
            fun=lambda x: x[-1],
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 2000, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return None, None

    if not res.success:
        return None, None

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = h_opt.sum()
    if abs(cur - target) > 1e-9:
        h_opt += (target - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    val = compute_c5(h_opt)
    return h_opt, val

def targeted_local_search(h, cur_val, deadline, rng):
    """Hill‑climbing focused on the worst shift, with a tiny Metropolis component."""
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline - 0.05:
        # current worst shift
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr_full))
        cur_val = corr_full[max_idx] * dx
        k = max_idx - (n - 1)

        # region indices for this shift
        if k >= 0:
            i_region = np.arange(n - k)
            i_shift = i_region + k
        else:
            kk = -k
            i_region = np.arange(kk, n)
            i_shift = i_region - kk

        # contributions to the worst overlap
        contrib = h[i_region] * (1.0 - h[i_shift])
        src_local = int(np.argmax(contrib))
        src_i = int(i_region[src_local])
        src_shift_val = float(h[i_shift[src_local]])

        # candidates that can receive mass
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0:
            break

        if k >= 0:
            cand_valid = cand[cand + k < n]
            cand_shift = h[cand_valid + k]
        else:
            kk = -k
            cand_valid = cand[cand - kk >= 0]
            cand_shift = h[cand_valid - kk]

        diff = cand_shift - src_shift_val
        pos = np.where(diff > 1e-12)[0]
        if pos.size > 0:
            best_idx = pos[int(np.argmax(diff[pos]))]
            dst_j = int(cand_valid[best_idx])
        else:
            dst_j = int(rng.choice(cand))

        delta_max = min(float(h[src_i]), 1.0 - float(h[dst_j]))
        if delta_max < 1e-12:
            continue
        delta = rng.random() * delta_max

        # apply move
        h[src_i] -= delta
        h[dst_j] += delta

        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_val = new_corr.max() * dx

        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if cur_val < 0.3808:
                break
        else:
            # Metropolis acceptance
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert
                h[src_i] += delta
                h[dst_j] -= delta
    return h, cur_val
