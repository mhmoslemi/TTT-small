    low = -np.max(v) - 5.0

def compute_c5(h):
    """Maximum overlap integral for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project onto [lo,hi]^n while fixing sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def overlap_gradient(h, k):
    """Gradient of overlap(k) w.r.t. h (subgradient of max)."""
    n = h.shape[0]
    if k > 0:
        shift = k
        G = np.ones(n)
        if shift < n:
            G[:n - shift] -= h[shift:]
            G[shift:] -= h[:n - shift]
    else:
        shift = -k
        G = np.zeros(n)
        if shift < n:
            G[shift:] = 1.0
            G[shift:] -= h[:n - shift]
            G[:n - shift] -= h[shift:]
    return G
