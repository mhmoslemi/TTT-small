import time, numpy as np, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # simple uniform adjustment
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # adjust greedily
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def gradient_descent_optimize(h0, time_budget, seed):
    # similar to previous with possible parameter tuning
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)

import time
import math
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (approximation via discrete correlation)."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x)==target (exact repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def annealing_optimize(h0, c5_0, time_budget, seed):
    """Simple simulated‑annealing with pair transfers and occasional global jitter."""
    rng = np.random.default_rng(seed)
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur_c5 = c5_0
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    T_initial = 0.02
    max_step = 0.20           # max transferred mass per pair move
    max_sigma = 0.30          # scale for global jitter
    it = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T_initial * max(0.0, 1.0 - elapsed / time_budget)
        if it % 50 == 0:
            sigma = max_sigma * (T / T_initial + 0.1)
            cand = h + sigma * rng.normal(size=n)
            cand = project_sum_and_bounds(cand, target)
        else:
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 1e-12:
                it += 1
                continue
            eps = rng.random() * min(max_transfer, max_step)
            if eps < 1e-12:
                it += 1
                continue
            cand = h.copy()
            cand[i] -= eps
            cand[j] += eps
        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        elif T > 1e-12:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            h = cand
            h = project_sum_and_bounds(h, target)
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        it += 1
    return h, cur_c5, best_h, best_c5

def gradient_descent_optimize(h0, time_budget, seed):
    """Projected gradient descent on a smoothed max‑overlap objective."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 40.0
    alpha0 = 0.15
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / len(v)
        else:
            w /= w_sum
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-15:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        g -= np.mean(g)                     # preserve total mass
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        if no_improve > 1500:
            h = project_sum_and_bounds(h + rng.normal(scale=0.03, size=n), target)
            no_improve = 0
    return best_h, best_c5

def greedy_local_search(h0, time_budget, seed):
    """Deterministic hill‑climbing with pair transfers."""
    rng = np.random.default_rng(seed)
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur_c5 = compute_c5(h)
    start = time.time()
    max_step = 0.10
    while time.time() - start < time_budget:
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 1e-12:
            continue
        eps = rng.random() * min(max_transfer, max_step)
        if eps < 1e-12:
            continue
        cand = h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = project_sum_and_bounds(cand, target)
        cand_c5 = compute_c5(cand)
        if cand_c5 < cur_c5:
            h = cand
            cur_c5 = cand_c5
    return h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        factor = 3                      # up‑sample for more flexibility
        h0 = np.repeat(base, factor)
        n = len(h0)
    else:
        n = 120
        h0 = rng.random(n)

    h0 = project_sum_and_bounds(h0, n / 2.0)
    cur_c5 = compute_c5(h0)

    # Allocate time between phases
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    anneal_time = remaining * 0.60
    grad_time   = remaining - anneal_time

    # Phase 1: simulated annealing
    _, _, best_h, best_c5 = annealing_optimize(h0, cur_c5, anneal_time, seed + 1)

    # Phase 2: gradient descent refinement
    if grad_time > 0.2:
        h_opt, c5_opt = gradient_descent_optimize(best_h, grad_time, seed + 2)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: greedy local polishing (if time left)
    remaining = max(0.0, budget_s - (time.time() - start))
    if remaining > 0.5:
        h_greedy, c5_greedy = greedy_local_search(best_h, remaining, seed + 3)
        if c5_greedy < best_c5:
            best_h, best_c5 = h_greedy, c5_greedy

    # Final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
