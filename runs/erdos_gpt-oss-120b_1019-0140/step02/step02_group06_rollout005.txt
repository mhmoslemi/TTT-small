import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target, tol=1e-12):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        # distribute uniformly
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > tol:
            # adjust iteratively
            # Use a simple loop over indices
            for i in range(len(x)):
                if abs(diff) < tol:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    # final clip
    return np.clip(x, 0.0, 1.0)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target, tol=1e-12):
    """Clip to [0,1] and adjust so that sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        # uniform correction
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > tol:
            # incremental fix
            for i in range(len(x)):
                if abs(diff) < tol:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return np.clip(x, 0.0, 1.0)

def smooth_optimize(h, time_budget, rng):
    """Few steps of soft‑max gradient descent to polish a solution."""
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_f = compute_c5(best_h)
    start = time.time()
    beta = 2.0
    max_beta = 20.0
    alpha0 = 0.2
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.ones_like(v) / len(v)

        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        g -= g.mean()

        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)

        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        if f_new < compute_c5(h):
            h = h_new
        else:
            if rng.random() < 0.01:
                h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)

        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
    else:
        n = 80
        h0 = rng.random(n)
    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    pop_size = 30
    population = [h0]
    while len(population) < pop_size:
        cand = rng.random(n)
        cand = project_sum_and_bounds(cand, target)
        population.append(cand)
    population = np.array(population, dtype=float)

    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_f = fitness[best_idx]

    tournament_size = 3
    mutation_rate = 0.3
    mutation_scale = 0.1
    iteration = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        iteration += 1

        # tournament selection for parent 1
        idxs = rng.choice(pop_size, size=tournament_size, replace=False)
        parent1 = population[idxs[np.argmin(fitness[idxs])]]
        # tournament selection for parent 2
        idxs = rng.choice(pop_size, size=tournament_size, replace=False)
        parent2 = population[idxs[np.argmin(fitness[idxs])]]

        # uniform crossover
        mask = rng.random(n) < 0.5
        child = np.where(mask, parent1, parent2).astype(float)

        # mutation
        if rng.random() < mutation_rate:
            scale = mutation_scale * max(0.1, 1.0 - elapsed / budget_s)
            child += rng.normal(scale=scale, size=n)
        child = project_sum_and_bounds(child, target)

        child_f = compute_c5(child)

        # replace worst if better
        worst_idx = int(np.argmax(fitness))
        if child_f < fitness[worst_idx]:
            population[worst_idx] = child
            fitness[worst_idx] = child_f
            if child_f < best_f:
                best_f = child_f
                best_h = child.copy()

        # occasional random injection to keep diversity
        if iteration % 5000 == 0:
            ridx = rng.integers(pop_size)
            newcand = rng.random(n)
            newcand = project_sum_and_bounds(newcand, target)
            newcand_f = compute_c5(newcand)
            if newcand_f < fitness[ridx]:
                population[ridx] = newcand
                fitness[ridx] = newcand_f
                if newcand_f < best_f:
                    best_f = newcand_f
                    best_h = newcand.copy()

    # final polishing with a brief gradient descent phase
    remaining = budget_s - (time.time() - start_time)
    if remaining > 2.0:
        best_h, best_f = smooth_optimize(best_h, remaining, rng)

    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
