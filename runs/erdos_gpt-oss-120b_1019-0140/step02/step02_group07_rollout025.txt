def de_optimize(start_vec, time_end, rng):
    n = len(start_vec)
    target_sum = n/2.0
    bounds = [(0,1)]*n

    def objective(x):
        # x is np.ndarray shape n
        x = np.asarray(x)
        # enforce sum via repair
        s = np.sum(x)
        # apply simple affine shift:
        diff = target_sum - s
        # distribute equally
        x = x + diff / n
        # clip
        x = np.clip(x, 0, 1)
        # after clipping, re-adjust sum again (could adjust residual)
        # loop while residual > tolerance
        for _ in range(5):
            res = target_sum - np.sum(x)
            if abs(res) < 1e-12:
                break
            # distribute residual proportionally to free space
            # free_i = (1 - x_i) if res>0 else -x_i
            # We can try simple addition of equal share again.
            x = x + res / n
            x = np.clip(x, 0, 1)
        # final compute c5
        return compute_c5(x)

    # callback to stop based on time
    def callback(xk, convergence):
        # xk is current best vector
        return time.time() > time_end

    # differential evolution call
    result = differential_evolution(objective,
                                    bounds,
                                    maxiter=200,
                                    popsize=5,
                                    init='random',
                                    recombination=0.9,

def continuous_anneal(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 300000
    it = 0
    while time.time() < time_end and it < max_iter:
        # candidate move: pick i to increase, j to decrease
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # ensure exact sum (tiny drift)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        # residual fix
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            if residual > 0:
                max_inc = 1 - best_h[i]
                adj = min(residual, max_inc)
                best_h[i] += adj
                residual -= adj
            else:
                max_dec = best_h[i]
                adj = min(-residual, max_dec)
                best_h[i] -= adj
                residual += adj
    return best_h, best_c5

import time
import math
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(arr):
    """Approximate C5 via full correlation of h and (1-h)."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Binary Thue‑Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)

def rudin_shapiro_seq(n):
    """Binary Rudin‑Shapiro sequence (+1/‑1 → 1/0) of length n."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)                                   # a_{2k}
            nxt.append(v * (1 if (k % 2 == 0) else -1))    # a_{2k+1}=(-1)^k * a_k
        a = nxt
    a = np.array(a[:n], dtype=int)
    return ((a + 1) // 2).astype(float)

def repair_sum(x):
    """Project x onto the simplex {0≤x_i≤1, sum x_i = n/2}."""
    n = len(x)
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        x = x + diff / n
        x = np.clip(x, 0.0, 1.0)
        residual = target - np.sum(x)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            if residual > 0:
                max_inc = 1.0 - x[i]
                delta = min(residual, max_inc)
                if delta > 0:
                    x[i] += delta
                    residual -= delta
            else:
                max_dec = x[i]
                delta = min(-residual, max_dec)
                if delta > 0:
                    x[i] -= delta
                    residual += delta
    return x

def binary_hill_climb(start_bin, time_end, rng):

    best_h = cur.copy()

        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            cur = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5

        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # exact‑sum correction
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            if residual > 0:
                max_inc = 1.0 - best_h[i]
                delta = min(residual, max_inc)
                best_h[i] += delta
                residual -= delta
            else:
                max_dec = best_h[i]
                delta = min(-residual, max_dec)
                best_h[i] -= delta
                residual += delta
    return best_h, best_c5

def de_optimize(start_vec, time_end, rng):
    """Box‑constrained differential evolution with sum‑repair."""
    n = len(start_vec)
    bounds = [(0.0, 1.0)] * n

    def objective(x):
        x = repair_sum(np.asarray(x))
        return compute_c5(x)

    def callback(xk, convergence):
        return time.time() > time_end

    seed_int = int(rng.integers(0, 2**31 - 1))
    result = differential_evolution(
        objective,
        bounds,
        maxiter=200,
        popsize=5,
        init='random',
        recombination=0.9,
        mutation=(0.5, 1.0),
        polish=False,
        seed=seed_int,
        callback=callback,
        disp=False,
    )
    best = repair_sum(result.x)
    return best, compute_c5(best)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    # ensure feasibility
    best_h = repair_sum(h0)
    best_c5 = compute_c5(best_h)

    # generate a few deterministic seeds
    seeds = [best_h]

    rand_bin = np.zeros(n, dtype=float)
    rand_bin[rng.choice(n, n // 2, replace=False)] = 1.0
    seeds.append(rand_bin)

    tm = thue_morse_seq(n).astype(float)
    diff = n / 2 - np.sum(tm)
    if diff != 0:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for idx in idxs:
            if diff == 0:
                break
            if diff > 0 and tm[idx] == 0:
                tm[idx] = 1
                diff -= 1
            elif diff < 0 and tm[idx] == 1:
                tm[idx] = 0
                diff += 1
    seeds.append(tm)

    rs = rudin_shapiro_seq(n).astype(float)
    diff = n / 2 - np.sum(rs)
    if diff != 0:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for idx in idxs:
            if diff == 0:
                break
            if diff > 0 and rs[idx] == 0:
                rs[idx] = 1
                diff -= 1
            elif diff < 0 and rs[idx] == 1:
                rs[idx] = 0
                diff += 1
    seeds.append(rs)

    # evaluate initial seeds
    for s in seeds:
        vec = repair_sum(s)
        c5 = compute_c5(vec)
        if c5 < best_c5:
            best_h, best_c5 = vec, c5

    # main optimisation loop
    while time.time() - start_time < budget_s - 5.0:
        remaining = budget_s - (time.time() - start_time)
        if rng.random() < 0.4:
            # differential evolution
            time_end = time.time() + min(30.0, remaining * 0.4)
            cand_h, cand_c5 = de_optimize(best_h, time_end, rng)
        else:
            # local searches
            time_end = time.time() + min(15.0, remaining * 0.3)
            if rng.random() < 0.5:
                cand_h, cand_c5 = continuous_anneal(best_h, time_end, rng)
            else:
                bin_start = (best_h >= 0.5).astype(int)
                cand_h, cand_c5 = binary_hill_climb(bin_start, time_end, rng)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h, cand_c5
            if best_c5 <= 0.3808:
                break

    # final refinement with binary hill climbing
    time_left = budget_s - (time.time() - start_time)
    if time_left > 2.0:
        end_time = start_time + budget_s - 0.1
        bin_start = (best_h >= 0.5).astype(int)
        fin_h, fin_c5 = binary_hill_climb(bin_start, end_time, rng)
        if fin_c5 < best_c5:
            best_h, best_c5 = fin_h, fin_c5

    # final feasibility enforcement
    best_h = repair_sum(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
