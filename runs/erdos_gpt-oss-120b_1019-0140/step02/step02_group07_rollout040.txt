def solve_lp(n, w, rng):
    h = cp.Variable(n)
    z = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == w]
    for k in range(1, n):
        size = n - k
        if size == 0: continue
        y = cp.Variable(size)
        constraints += [y <= h[:size]]
        constraints += [y <= h[k:]]
        constraints += [y >= h[:size] + h[k:] - 1]
        constraints += [y >= 0]
        constraints += [cp.sum(y) >= z]
    prob = cp.Problem(cp.Maximize(z), constraints)
    prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
    return h.value, z.value
```

def compute_c5(arr):
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)
```

def binary_hill_climb(start_bin, time_end, rng):
    n = len(start_bin)
    h = start_bin.astype(float)
    best_c5 = compute_c5(h)
    best_bin = start_bin.copy()
    hot_i = np.flatnonzero(start_bin)
    cold = np.flatnonzero(1 - start_bin)
    while time.time() < time_end:
        if hot_i.size == 0 or cold.size == 0:
            break
        i = rng.choice(hot_i)
        j = rng.choice(cold)
        # swap
        cand_bin = best_bin.copy()
        cand_bin[i] = 0
        cand_bin[j] = 1
        cand_c5 = compute_c5(cand_bin.astype(float))
        if cand_c5 < best_c5 - 1e-12:
            best_c5 = cand_c5
            best_bin = cand_bin
            # update lists
            hot_i = np.flatnonzero(best_bin)
            cold = np.flatnonzero(1 - best_bin)
        # else continue
    return best_bin.astype(float), best_c5
```

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, cvxpy as cp
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0

    # Solve LP for continuous h
    try:
        h_cont, z_val = solve_lp(n, target_sum, rng)
    except Exception as e:
        # fallback: use random binary initial
        h_cont = None

    best_h = None
    best_c5 = float('inf')

    if h_cont is not None:
        h_cont = np.clip(h_cont, 0.0, 1.0)
        # enforce sum exactly
        cur_sum = np.sum(h_cont)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-9:
            h_cont += diff / n
            h_cont = np.clip(h_cont, 0.0, 1.0)
        c5_val = compute_c5(h_cont)
        best_h = h_cont
        best_c5 = c5_val

    # consider initial_h_values as candidate
    if h0 is not None:
        h0 = np.clip(h0, 0.0, 1.0)
        # adjust to exact sum if needed
        cur_sum = np.sum(h0)
        if abs(cur_sum - target_sum) > 1e-12:
            diff = target_sum - cur_sum
            h0 += diff / n
            h0 = np.clip(h0, 0.0, 1.0)
        c5_0 = compute_c5(h0)
        if c5_0 < best_c5:
            best_h = h0.copy()
            best_c5 = c5_0

    # binary hill climb from rounded LP solution if possible
    if h_cont is not None:
        bin_start = round_to_binary(h_cont, target_sum)
        # hill climb within remaining time
        remaining = budget_s - (time.time() - start_time)
        if remaining > 5.0:

            if cand_c5 < best_c5:

                best_c5 = cand_c5

    # Additional random restarts hill climbing
    while time.time() - start_time < budget_s - 1.0:
        # generate random binary with exact sum
        bin_rand = np.zeros(n, dtype=int)
        bin_rand[rng.choice(n, int(target_sum), replace=False)] = 1
        remaining = budget_s - (time.time() - start_time)
        time_end = time.time() + min(2.0, remaining*0.5)

        if cand_c5 < best_c5:
            best_h = cand_h.copy()
            best_c5 = cand_c5
            if best_c5 <= 0.3808:

    # final clip and sum correction
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = np.sum(best_h)
    diff = target_sum - cur_sum

        best_h += diff / n

        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:

            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    # compute final c5
    final_c5 = compute_c5(best_h)

import time
import numpy as np

def compute_c5(arr):
    """Maximum overlap integral approximated by discrete correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * (2.0 / n)

def solve_lp(n, target_sum):
    """Linear program that maximises the minimal self‑overlap of h."""
    import cvxpy as cp

    h = cp.Variable(n)                 # real variables in [0,1]
    z = cp.Variable()                  # lower bound on each overlap
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
    ]

    # For every shift k compute a lower‑bound variable y_{i,k} for h_i*h_{i+k}
    for k in range(1, n):
        size = n - k
        if size == 0:
            continue
        y = cp.Variable(size)          # lower bound for the product
        constraints += [
            y <= h[:size],
            y <= h[k:],
            y >= h[:size] + h[k:] - 1,
            y >= 0,
            cp.sum(y) >= z,
        ]

    prob = cp.Problem(cp.Maximize(z), constraints)
    # ECOS is a fast LP/QP solver available in the environment
    prob.solve(solver=cp.ECOS, max_iters=20000, verbose=False)
    if h.value is None:
        raise RuntimeError("LP failed to find a solution")
    return np.clip(np.squeeze(np.array(h.value, dtype=float)), 0.0, 1.0)

def round_to_binary(h, target_ones):
    """Greedy rounding that preserves the exact number of ones."""
    idx = np.argsort(-h)               # descending order
    bin_vec = np.zeros_like(h, dtype=int)
    bin_vec[idx[:target_ones]] = 1
    return bin_vec

def binary_hill_climb(start_bin, time_end, rng):
    """Swap‑based hill climbing on binary vectors."""
    n = len(start_bin)
    best_bin = start_bin.copy()
    best_c5 = compute_c5(best_bin.astype(float))
    # pre‑compute index sets
    ones = np.flatnonzero(best_bin)
    zeros = np.flatnonzero(1 - best_bin)

    while time.time() < time_end:
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_bin.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_bin = cand
            best_c5 = cand_c5
            ones = np.flatnonzero(best_bin)
            zeros = np.flatnonzero(1 - best_bin)
    return best_bin.astype(float), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None
    target_sum = n / 2.0
    target_ones = int(round(target_sum))

    # ----- 1. Linear‑programming construction -----
    try:
        h_lp = solve_lp(n, target_sum)
        c5_lp = compute_c5(h_lp)
        best_h = h_lp
        best_c5 = c5_lp
    except Exception:
        best_h = None
        best_c5 = float("inf")

    # ----- 2. Incorporate any provided initial construction -----
    if h0 is not None:
        h0 = np.clip(h0, 0.0, 1.0)
        # enforce exact integral (tiny numerical fix)
        diff = target_sum - np.sum(h0)
        if abs(diff) > 1e-12:
            h0 += diff / n
            h0 = np.clip(h0, 0.0, 1.0)
        c5_0 = compute_c5(h0)
        if c5_0 < best_c5:
            best_h = h0.copy()
            best_c5 = c5_0

    # ----- 3. Refine the LP solution via binary hill‑climbing -----
    if best_h is not None:
