import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def hill_climb(h, cur_val, rng, deadline):
    while True:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            for j in zeros:
                if time.time() > deadline:
                    return h, cur_val
                cand = h.copy()
                cand[i], cand[j] = cand[j], cand[i]
                cand_val = compute_c5(cand)
                if cand_val + 1e-12 < cur_val:
                    h = cand
                    cur_val = cand_val
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, cur_val

def anneal(start_h, time_limit, rng):
    start = time.time()
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    T0 = 0.5
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        prog = elapsed / time_limit
        T = T0 * max(0.0, 1.0 - prog)
