def continuous_anneal(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n/2.0
    h = start_h.copy()
    # project into [0,1] and enforce sum
    h = np.clip(h,0,1)
    # adjust sum
    diff = target_sum - np.sum(h)
    h += diff / n
    h = np.clip(h,0,1)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # schedule: linear cooling from T0 to Tf across many iterations
    T0 = 1e-2
    Tf = 1e-7
    max_iter = 200000
    for it in range(max_iter):
        if time.time() >= time_end:
            break
        # choose inc and dec indices
        inc_candidates = np.where(h < 1 - 1e-12)[0]
        dec_candidates = np.where(h > 1e-12)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            break
        i = rng.choice(inc_candidates)
        j = rng.choice(dec_candidates)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            continue
        # propose delta proportionally to T?
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = T0 * (1 - it / max_iter) + Tf * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-30)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    # final adjust sum for best_h
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h,0,1)
    return best_h, best_c5
```

def enforce_sum_and_bounds(x, target):
    x = np.clip(x, 0., 1.)
    diff = target - np.sum(x)
    # if diff > 0, add to entries that are < 1
    # if diff < 0, subtract from entries > 0
    if abs(diff) > 1e-12:
        # compute capacities
        if diff > 0:
            capacities = 1.0 - x
        else:
            capacities = -x  # amount can subtract
        # sort indices by capacity descending
        idxs = np.argsort(-capacities)  # we will use large capacities first
        for idx in idxs:
            if abs(diff) < 1e-12:
                break
            cap = capacities[idx]
            if cap <= 0:
                continue
            # amount to transfer
            amount = min(cap, abs(diff))
            if diff > 0:
                x[idx] += amount
                diff -= amount
            else:
                x[idx] -= amount
                diff += amount
    # final clip (just in case)
    x = np.clip(x, 0., 1.)
    return x

def binary_hill_climb(start_bin, time_end, rng):
    n = len(start_bin)
    h = start_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h.astype(float))
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5

import time
import math
import numpy as np

def compute_c5(arr):
    """Approximate C5 = max_k ∫ h(x)(1−h(x+k))dx via discrete correlation."""
    n = arr.size
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def enforce_sum_and_bounds(x, target):
    """Clip to [0,1] and adjust to meet the exact sum `target`."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        # Determine how much each entry can accept/release.
        if diff > 0:
            caps = 1.0 - x
        else:
            caps = -x
        # Sort indices by available capacity (largest first).
        idxs = np.argsort(-caps)
        for idx in idxs:
            if abs(diff) < 1e-12:
                break
            cap = caps[idx]
            if cap <= 0.0:
                continue
            amt = min(cap, abs(diff))
            if diff > 0:
                x[idx] += amt
                diff -= amt
            else:
                x[idx] -= amt
                diff += amt
    # final clip (tiny rounding issues)
    x = np.clip(x, 0.0, 1.0)
    # if a minute residue remains, adjust a single free entry
    resid = target - np.sum(x)
    if abs(resid) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            i = free[0]
            x[i] += resid
            x[i] = np.clip(x[i], 0.0, 1.0)
    return x

def continuous_anneal(start_h, time_end, rng):
    """Simulated annealing on a continuous vector preserving the sum."""
    n = start_h.size
    target = n / 2.0
    h = enforce_sum_and_bounds(start_h, target)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # annealing schedule
    T0 = 5e-3
    Tf = 1e-7
    max_iter = 200000
    for it in range(max_iter):
        if time.time() >= time_end:
            break
        inc_idx = np.where(h < 1.0 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = T0 * (1.0 - it / max_iter) + Tf * (it / max_iter)
        if dE <= 0.0 or rng.random() < math.exp(-dE / (T + 1e-30)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    best_h = enforce_sum_and_bounds(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def binary_hill_climb(start_bin, time_end, rng):
    """Swap‑based hill climbing on binary vectors (exactly n/2 ones)."""
    n = start_bin.size
    h = start_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h.astype(float))
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = enforce_sum_and_bounds(h0, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---------- Differential Evolution (global exploration) ----------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        try:
            from scipy.optimize import differential_evolution
        except Exception:
            differential_evolution = None
        if differential_evolution is not None:
            bounds = [(0.0, 1.0)] * n
            penalty_weight = 1e4

            def de_obj(x):
                x = np.clip(x, 0.0, 1.0)
                sum_diff = np.sum(x) - target_sum
                pen = penalty_weight * (sum_diff ** 2)
                return compute_c5(x) + pen

            # modest iteration count to respect budget
            de_maxiter = 30
            de_res = differential_evolution(
                de_obj,
                bounds,
                seed=seed,
                maxiter=de_maxiter,
                polish=False,
                disp=False,
            )
            cand = enforce_sum_and_bounds(de_res.x, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_h, best_c5 = cand, cand_c5

    # ---------- Continuous annealing with time slicing ----------
    while time.time() - start_time < budget_s - 0.5:
        chunk = min(2.0, budget_s - (time.time() - start_time) - 0.5)
        end_time = time.time() + chunk
        # jitter around current best
        start_vec = best_h + rng.normal(scale=0.05, size=n)
        start_vec = enforce_sum_and_bounds(start_vec, target_sum)
        cand_h, cand_c5 = continuous_anneal(start_vec, end_time, rng)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h, cand_c5
        if best_c5 <= 0.3808:
            break

    # ---------- Final binary hill climb ----------
    if time.time() - start_time < budget_s - 0.2:
        bin_start = (best_h >= 0.5).astype(int)
        final_h, final_c5 = binary_hill_climb(
            bin_start,
            start_time + budget_s - 0.1,
            rng,
        )
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5

    # Enforce feasibility one last time
    best_h = enforce_sum_and_bounds(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
