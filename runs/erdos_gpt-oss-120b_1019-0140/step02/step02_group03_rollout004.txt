def propose_swap(cur_h, step_max, rng):

    n = cur_h.shape[0]

    i, j = rng.choice(n, size=2, replace=False)

    hi, hj = cur_h[i], cur_h[j]

    delta_min = max(-hi, hj - 1.0)

    delta_max = min(1.0 - hi, hj)

    # restrict magnitude to step_max

    delta_max = min(delta_max, step_max)

    delta_min = max(delta_min, -step_max)

    if delta_max <= delta_min:
        return None  # no feasible move

    delta = rng.uniform(delta_min, delta_max)

    cand = cur_h.copy()

    cand[i] = hi + delta

    cand[j] = hj - delta

    # sum preserved, h remains within [0,1]

    return cand

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- Core utilities -------------------

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h (discrete)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_box_sum(v: np.ndarray, target: float, lo: float = 0.0, hi: float = 1.0) -> np.ndarray:
    """Project v onto [lo, hi]^n while forcing its sum to `target`."""
    # binary search for lambda shifting all entries before clipping
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)


def upsample_h(h_small: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation from a smaller vector to a larger one."""
    n_small = h_small.shape[0]
    if n_small == n_target:
        return h_small.copy()
    x_small = np.arange(n_small) / n_small
    x_target = np.arange(n_target) / n_target
    h_target = np.interp(x_target, x_small, h_small)
    return h_target


# ------------------- SLSQP optimizer -------------------

def slsqp_optimize(h_start: np.ndarray, t_start: float, n: int,
                   target_sum: float, maxiter: int = 800):
    """Run a single SLSQP optimisation starting from (h_start, t_start)."""
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def sum_constraint(x):
        return np.sum(x[:n]) - target_sum

    def overlap_slack(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        slack = t - corr * dx
        # drop zero shift (index n-1)
        return np.concatenate([slack[:n - 1], slack[n:]])

    constraints = [
        {'type': 'eq', 'fun': sum_constraint},
        {'type': 'ineq', 'fun': overlap_slack}
    ]

    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=lambda x: x[-1],
                   x0=x0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res


# ------------------- Move proposal for SA -------------------

def propose_swap(cur_h: np.ndarray, step_max: float, rng: np.random.Generator):
    """Generate a feasible candidate by transferring mass between two indices."""
    n = cur_h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    hi, hj = cur_h[i], cur_h[j]

    # feasible interval for delta (increase hi, decrease hj)
    delta_min = max(-hi, hj - 1.0)
    delta_max = min(1.0 - hi, hj)

    # restrict magnitude
    delta_min = max(delta_min, -step_max)
    delta_max = min(delta_max, step_max)

    if delta_max <= delta_min:
        return None
    delta = rng.uniform(delta_min, delta_max)
    cand = cur_h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    return cand


# ------------------- Main entry point -------------------
