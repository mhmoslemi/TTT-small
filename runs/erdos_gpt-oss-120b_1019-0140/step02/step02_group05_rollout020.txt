def enforce_bounds_and_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # find indices where h is interior (0 < h < 1)
        interior = np.where((h > 0) & (h < 1))[0]
        if interior.size > 0:
            inc = diff / interior.size
            h[interior] += inc
            h = np.clip(h, 0.0, 1.0)
        # residual fix
        diff = target_sum - h.sum()
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h
```

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def enforce_bounds_and_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        interior = np.where((h > 0) & (h < 1))[0]
        if interior.size:
            inc = diff / interior.size
            h[interior] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        # residual fix
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap C5 for a discretized h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_bounds_and_sum(h, target):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        interior = np.where((h > 0) & (h < 1))[0]
        if interior.size:
            inc = diff / interior.size
            h[interior] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        for i in range(h.size):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise h and n
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 80
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    target_sum = n / 2.0
    h = enforce_bounds_and_sum(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    dx = 2.0 / n
    T0 = 0.08                     # initial temperature
    p_targeted = 0.45
    p_swap = 0.20
    p_perturb = 0.25
    p_grad = 0.10

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        # temperature schedule (linear)
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        # current overlap and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_val = np.max(corr) * dx
        if cur_val < 0.3808:
            best_h, best_val = h.copy(), cur_val
            break
        k_idx = int(np.argmax(corr))
        k = k_idx - (n - 1)               # shift corresponding to worst overlap

        # propose a move
        r = rng.random()
        cand = None

        if r < p_targeted:
            # targeted transfer based on the worst shift
            if k == 0:
                continue
            if k > 0:
                max_i = n - k - 1
                if max_i < 0:
                    continue
                i = rng.integers(0, max_i + 1)
                donor, recv = i, i + k
            else:
                kk = -k
                max_i = n - kk - 1
                if max_i < 0:
                    continue
                i = rng.integers(0, max_i + 1)
                donor, recv = i + kk, i
            max_tr = min(h[donor], 1.0 - h[recv])
            if max_tr > 0:
                delta = rng.random() * max_tr
                cand = h.copy()
                cand[donor] -= delta
                cand[recv] += delta

        elif r < p_targeted + p_swap:
            i, j = rng.choice(n, size=2, replace=False)
            cand = h.copy()
            cand[i], cand[j] = cand[j], cand[i]

        elif r < p_targeted + p_swap + p_perturb:
            noise = rng.normal(scale=0.08, size=n)
            cand = h + noise
            cand = np.clip(cand, 0.0, 1.0)
            cand = enforce_bounds_and_sum(cand, target_sum)

        else:
            # gradient move on the worst shift
            grad = np.zeros(n)
            if k >= 0:
                grad[: n - k] = 1.0 - h[k:]
                grad[k:] += 1.0 - h[: n - k]
            else:
                kk = -k
                grad[kk:] = 1.0 - h[: n - kk]
                grad[: n - kk] += 1.0 - h[kk:]
            # zero‑mean to keep sum unchanged by the raw step
            grad_centered = grad - np.mean(grad)
            delta = -grad_centered
            mabs = np.max(np.abs(delta))
            max_step = 0.2
            if mabs > max_step:
                delta = delta * (max_step / mabs)
            cand = h + delta
            cand = np.clip(cand, 0.0, 1.0)
            cand = enforce_bounds_and_sum(cand, target_sum)

        if cand is None:
            continue

        cand_val = compute_c5(cand)

        # Metropolis acceptance
        accept = False
        if cand_val < cur_val:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(cand_val - cur_val) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break

    # --------------------------------------------------------
    # Post‑processing: binary projection and a few gradient refinements
    # --------------------------------------------------------
    binary_h = np.zeros_like(best_h)
    top_idx = np.argsort(-best_h)[: n // 2]
    binary_h[top_idx] = 1.0
    binary_val = compute_c5(binary_h)
    if binary_val < best_val:
        best_h, best_val = binary_h, binary_val

    # short deterministic gradient polishing
    for _ in range(200):
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        cur = np.max(corr) * dx
        if cur + 1e-12 < best_val:
            best_val = cur
            best_h = best_h.copy()
        k_idx = int(np.argmax(corr))
        k = k_idx - (n - 1)
        if k == 0:
            break
        grad = np.zeros(n)
        if k >= 0:
            grad[: n - k] = 1.0 - best_h[k:]
            grad[k:] += 1.0 - best_h[: n - k]
        else:
            kk = -k
            grad[kk:] = 1.0 - best_h[: n - kk]
            grad[: n - kk] += 1.0 - best_h[kk:]
        delta = -0.05 * (grad - np.mean(grad))
        cand = best_h + delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = enforce_bounds_and_sum(cand, target_sum)
        cand_val = compute_c5(cand)
        if cand_val + 1e-12 < best_val:
            best_h, best_val = cand, cand_val
        else:
            break

    # final safety correction
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-9:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        diff = target_sum - best_h.sum()
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and best_h[i] < 1.0:
                add = min(diff, 1.0 - best_h[i])
                best_h[i] += add
                diff -= add
            elif diff < 0 and best_h[i] > 0.0:
                sub = min(-diff, best_h[i])
                best_h[i] -= sub
                diff += sub
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, n
