import time
import numpy as np
from scipy.optimize import minimize


def compute_c5_vec(h, dx):
    """Maximum overlap integral for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Adjust vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    # greedy fix
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Random vector in [0,1]^n with exact sum target_sum."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def hill_climb(h, dx, target_sum, time_limit, rng):
    """Simple mass‑transfer hill climbing with adaptive step size."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.2
    min_step = 1e-5
    n = len(cur_h)
    while time.time() < time_limit:
        improved = False
        for _ in range(8):  # try a handful of random moves
            i = rng.integers(n)
            if cur_h[i] <= 1e-12:
                continue
            j = rng.integers(n)
            if cur_h[j] >= 1.0 - 1e-12:
                continue
            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            # try full transfer and a random fraction
            for delta in (max_delta, max_delta * rng.random()):
                cand = cur_h.copy()
                cand[i] -= delta
                cand[j] += delta
                cand_val = compute_c5_vec(cand, dx)
                if cand_val < cur_val - 1e-12:
                    cur_h = cand
                    cur_val = cand_val
                    step = min(step * 1.2, 0.5)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            step = max(step * 0.9, min_step)
        if cur_val < 0.3808:
            break
    return cur_h, cur_val


def binary_hill_climb(h_bin, cur_val, dx, time_limit, rng):
    """Swap‑based hill climbing on a binary (0/1) mask."""
    best = h_bin.copy()
    best_val = cur_val
    n = len(best)
    while time.time() < time_limit:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best = cand
            best_val = cand_val
            if best_val < 0.3808:
                break
    return best.astype(float), best_val


class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        return np.sum(x[: self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overl * self.dx


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem sizes to try
    sizes = []
    if "initial_h_values" in globals():
        sizes.append(len(initial_h_values))
    else:
        sizes = [64, 80, 96]

    best_h = None
    best_val = float("inf")
    best_n = None

    for n in sizes:
        target_sum = n / 2.0
        dx = 2.0 / n

        # Initial candidate from parent if available
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            init = np.array(initial_h_values, dtype=float)
            init = np.clip(init, 0.0, 1.0)
            init = project_to_sum(init, target_sum)
            val = compute_c5_vec(init, dx)
            if val < best_val:
                best_h, best_val, best_n = init, val, n

        # Main random‑restart loop
        while time.time() - start < budget_s - 5.0:
            h0 = generate_random_feasible(n, target_sum, rng)
            deadline = start + budget_s - 1.0
            h_opt, val_opt = hill_climb(h0, dx, target_sum, deadline, rng)
            if val_opt < best_val - 1e-12:
                best_h, best_val, best_n = h_opt.copy(), val_opt, n
                if best_val < 0.3808:
                    break

            # Occasionally polish a binary rounding of the current best
            if rng.random() < 0.03:
                bin_h = np.zeros(n, dtype=int)
                top = np.argsort(-h_opt)[: int(target_sum)]
                bin_h[top] = 1
                bin_val = compute_c5_vec(bin_h.astype(float), dx)

                    if best_val < 0.3808:
                        break

        if best_val < 0.3808:
            break

    # Final SLSQP polishing if time permits
    remaining = budget_s - (time.time() - start)
    if best_h is not None and remaining > 30.0:
        n = best_n
        target_sum = n / 2.0
        dx = 2.0 / n
        sum_con = SumConstraint(n)
        overlap_cons = [
            OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0

        constraints = [{"type": "eq", "fun": sum_con}]
        constraints += [{"type": "ineq", "fun": oc} for oc in overlap_cons]
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

                x0=x0,

                h_new = project_to_sum(h_new, target_sum)
                val_new = compute_c5_vec(h_new, dx)
                if val_new < best_val:
                    best_h, best_val = h_new, val_new
        except Exception:
            pass

    # Safety finalisation
    if best_h is None:
        # fallback simple binary half‑ones
        n = sizes[0]
        target_sum = n / 2.0
        best_h = np.zeros(n)
        best_h[: int(target_sum)] = 1.0
        best_val = compute_c5_vec(best_h, 2.0 / n)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_val = compute_c5_vec(best_h, 2.0 / best_n)

    return best_h, best_val, best_n
