    while time.time() < ref_end:
        donors = np.where(cur_h > 1e-12)[0]
        recips = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recips.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recips)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = compute_c5_vec(cand, dx)
        if cand_score < cur_score - 1e-12:
            cur_h, cur_score = cand, cand_score
            if cur_score < best_score - 1e-12:
                best_h, best_score = cur_h.copy(), cur_score
            step = min(step * 1.2, 0.5)
        else:
            step *= 0.9
        if best_score < 0.3808:
            break

    # ---------- Binary hill climbing ----------
    if time.time() < end_time:
        # round to binary using top entries
        binary_h = np.zeros(n, dtype=int)
        top = int(round(target))
        idx = np.argsort(-best_h)[:top]
        binary_h[idx] = 1
        bin_score = compute_c5_vec(binary_h.astype(float), dx)

        best_bin_h, best_bin_score = binary_h, bin_score
        while time.time() < end_time:
            ones = np.flatnonzero(best_bin_h)
            zeros = np.flatnonzero(1 - best_bin_h)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cand = best_bin_h.copy()
            cand[i] = 0
            cand[j] = 1
            cand_score = compute_c5_vec(cand.astype(float), dx)
            if cand_score < best_bin_score - 1e-12:
                best_bin_h, best_bin_score = cand, cand_score
                if best_bin_score < best_score - 1e-12:
                    best_h = best_bin_h.astype(float).copy()
                    best_score = best_bin_score
                if best_score < 0.3808:
                    break

    # final safety projection
    best_h = project_to_sum(np.clip(best_h, 0.0, 1.0), target)
    best_score = compute_c5_vec(best_h, dx)

    return best_h, float(best_score), n
