def subgradient_optimize(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    n = len(h0)
    target = n/2
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    step0 = 0.4
    it = 0
    while time.time() - start < time_budget:
        # compute correlation
        v = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(v)
        shift = k_max - (n - 1)
        # subgradient
        g = np.ones(n)  # derivative of sum term (1) for each element
        # subtract shifted h
        if shift >= 0:
            # subtract h[i+shift] from g[i] where i+shift < n
            g[:n-shift] -= h[shift:]
            # subtract h[i-shift] for i >= shift
            g[shift:] -= h[:n-shift]
        else:
            shift_neg = -shift
            # shift negative => similar
            g[shift_neg:] -= h[:n-shift_neg]
            g[:n-shift_neg] -= h[shift_neg:]
        # step size
        step = step0 / np.sqrt(1 + it)
        # update
        h_new = h - step * g
        # projection
        h_new = project_sum_and_bounds(h_new, target)
        # evaluate
        c5_new = compute_c5(h_new)
        # accept if improves or with probability
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        it += 1
        # occasional random shake
        if it % 5000 == 0:
            # if no improvement, add noise
            if best_c5 - compute_c5(h) > -1e-9:  # not improved
                h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
    return best_h, best_c5

def local_refine(h, time_budget, rng):
    n = len(h)
    target = n/2
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    # Determine max step value for δ, maybe 0.1.
    max_delta = 0.1
    it = 0
    while time.time() - start < time_budget:
        # pick i where h[i] > 1e-12, j where h[j] < 1 - 1e-12
        # to speed, precompute indices.
        idx_pos = np.where(best_h > 1e-8)[0]
        idx_neg = np.where(best_h < 1 - 1e-8)[0]
        if len(idx_pos)==0 or len(idx_neg)==0:
            break
        i = rng.choice(idx_pos)
        j = rng.choice(idx_neg)
        delta = rng.random()*max_delta
        delta = min(delta, best_h[i], 1 - best_h[j])
        if delta <= 0:
            continue
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap times dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce exact sum == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute uniformly, then fix any remaining error by greedy passes
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def subgradient_optimize(h0, time_budget, seed):
    """Subgradient descent on the exact max-overlap objective."""
    if time_budget <= 0:
        return h0.copy(), compute_c5(h0)
    rng = np.random.default_rng(seed)
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    step0 = 0.5
    it = 0
    no_improve = 0
    while time.time() - start < time_budget:
        # correlation and shift of worst overlap
        v = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(v))
        shift = k_max - (n - 1)          # integer shift (can be negative)
        # subgradient of the active linear term
        g = np.ones(n)
        if shift >= 0:
            k = shift
            if n - k > 0:
                g[:n - k] -= h[k:]
                g[k:]   -= h[:n - k]
        else:
            k = -shift
            if n - k > 0:
                g[k:]   -= h[:n - k]
                g[:n - k] -= h[k:]
        # step size schedule
        step = step0 / np.sqrt(1.0 + it)
        h_new = h - step * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)

        # keep global best
        if c5_new + 1e-12 < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h = h_new
        it += 1

        # occasional random perturbation if stuck
        if no_improve >= 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
    return best_h, best_c5

def local_refine(h0, time_budget, rng):
    """Simple pairwise mass swap refinement."""
    if time_budget <= 0:
        return h0.copy(), compute_c5(h0)
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    max_delta = 0.1
    while time.time() - start < time_budget:
        # choose a donor (h>0) and a receiver (h<1)
        donors = np.where(best_h > 1e-12)[0]
        receivers = np.where(best_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(receivers)
        delta = rng.random() * max_delta
        delta = min(delta, best_h[i], 1.0 - best_h[j])
        if delta <= 0:
            continue
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            best_c5 = c5_new
            best_h = h_new
    # final projection to guarantee feasibility
    best_h = project_sum_and_bounds(best_h, target)
    return best_h, best_c5
