def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
    return res
```

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)


# ----------------------------------------------------------------------
# Constraint classes for SLSQP
# ----------------------------------------------------------------------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx


def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=600):
    """Run a single SLSQP optimisation starting from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
    )
    return res


# ----------------------------------------------------------------------
# Genetic operators
# ----------------------------------------------------------------------
def crossover(a, b, rng):
    mask = rng.random(a.shape) < 0.5
    return np.where(mask, a, b)


def mutate(v, rng, rate=0.1, sigma=0.02):
    mask = rng.random(v.shape) < rate
    perturb = rng.normal(scale=sigma, size=v.shape)
    out = v.copy()
    out[mask] += perturb[mask]
    return out


def tournament_select(pop_c5, rng, size=3):
    idx = rng.choice(len(pop_c5), size=size, replace=False)
    return idx[np.argmin(pop_c5[idx])]


def refine_with_slsqp(h, n, sum_con, overlap_cons, time_limit):
    """Refine a candidate using SLSQP if time permits."""
    if time.time() > time_limit:
        return h, compute_c5(h)
    t0 = compute_c5(h)
    res = slsqp_optimize(h, t0, sum_con, overlap_cons, maxiter=1200)
    if not res.success:
        return h, compute_c5(h)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -------------------- problem size --------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)

    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)

    # -------------------- best record --------------------
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # -------------------- constraints for SLSQP --------------------
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # -------------------- Phase 1: multiple SLSQP restarts --------------------
    phase1_deadline = start_time + budget_s * 0.45
    while time.time() < phase1_deadline:
        # random feasible start
        h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, target_sum)
        t_start = compute_c5(h_start)

        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=600)
        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)

        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            if best_c5 < 0.3808:
                break

    # -------------------- Phase 2: Genetic algorithm --------------------
    pop_size = 30
    population = np.empty((pop_size, n))
    # seed with best and random individuals
    population[0] = best_h
    for i in range(1, pop_size):
        v = rng.random(n)
        v = project_to_box_sum(v, target_sum)
        population[i] = v
    pop_c5 = np.apply_along_axis(compute_c5, 1, population)

    def update_best(idx):
        nonlocal best_h, best_c5
        if pop_c5[idx] + 1e-12 < best_c5:
            best_h = population[idx].copy()
            best_c5 = pop_c5[idx]

    ga_deadline = start_time + budget_s * 0.9
    while time.time() < ga_deadline:
        # select parents via tournament
        p1 = tournament_select(pop_c5, rng)
        p2 = tournament_select(pop_c5, rng)

        # generate two children
        child1 = crossover(population[p1], population[p2], rng)
        child2 = crossover(population[p2], population[p1], rng)

        # mutation
        child1 = mutate(child1, rng)
        child2 = mutate(child2, rng)

        # repair to feasible region
        child1 = project_to_box_sum(child1, target_sum)
        child2 = project_to_box_sum(child2, target_sum)

        # evaluate
        c5_1 = compute_c5(child1)
        c5_2 = compute_c5(child2)

        # replace worst individuals if better
        worst_idx = np.argmax(pop_c5)
        if c5_1 < pop_c5[worst_idx]:
            population[worst_idx] = child1
            pop_c5[worst_idx] = c5_1
            update_best(worst_idx)
            worst_idx = np.argmax(pop_c5)

        if c5_2 < pop_c5[worst_idx]:
            population[worst_idx] = child2
            pop_c5[worst_idx] = c5_2
            update_best(worst_idx)

        # occasional random injection for diversity
        if rng.random() < 0.01:
            rand_idx = rng.integers(pop_size)
            rand_vec = rng.random(n)
            rand_vec = project_to_box_sum(rand_vec, target_sum)
            population[rand_idx] = rand_vec
            pop_c5[rand_idx] = compute_c5(rand_vec)
            update_best(rand_idx)

        if best_c5 < 0.3808:
            break

    # -------------------- Phase 3: refine top individuals --------------------
    final_deadline = start_time + budget_s - 1.0  # safety margin
    # consider top 5 individuals
    top_indices = np.argsort(pop_c5)[:5]
    for idx in top_indices:
        if time.time() > final_deadline:
            break
        refined_h, refined_c5 = refine_with_slsqp(population[idx], n,
                                                 sum_con, overlap_cons,
                                                 final_deadline)
        if refined_c5 + 1e-12 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
            if best_c5 < 0.3808:
                break

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
