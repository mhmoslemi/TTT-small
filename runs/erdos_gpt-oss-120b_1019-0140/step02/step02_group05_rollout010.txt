import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def enforce_constraints(h, target_sum):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    diff = target_sum - cur_sum
    # We'll adjust free indices iteratively
    # Use while diff not small and iteration limit
    for _ in range(5):
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            break
        inc = diff / free.size
        h[free] += inc
        # Clip and recompute diff
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
    # final fix by greedy
    if abs(diff) > 1e-12:
        # iterate over all indices
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
        h = np.clip(h, 0.0, 1.0)
    return h

def random_initial(n, rng):
    # binary random: choose n/2 indices to be 1; rest 0.
    h = np.zeros(n, dtype=float)
    indices = rng.choice(n, n//2, replace=False)
    h[indices] = 1.0
    return h

def upsample_initial(initial, n_target):
    n0 = len(initial)
    x0 = np.linspace(0, 2, n0, endpoint=False)
    x = np.linspace(0, 2, n_target, endpoint=False)
    # linear interpolation; values outside range get 0?
    # But since domain same, we can use np.interp
    h = np.interp(x, x0, initial, left=0.0, right=0.0)
    return h

def gradient_step(h, target_sum):
    n = len(h)
    dx = 2.0 / n
    # compute correlation to find worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    # compute neighbor sum (h[i+k] + h[i-k])
    neigh = np.zeros_like(h)
    if k >= 0:
        # h[i+k] part
        neigh[: n - k] += h[k:]
        # h[i-k] part
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]   # h[i - kk] i.e. h[ i - kk ] = h[ i - kk ]
        neigh[: n - kk] += h[kk:]   # h[i + kk]
    grad = dx * (1.0 - neigh)  # derivative of overl_k
    # ensure zero-sum direction: subtract average
    grad_centered = grad - np.mean(grad)
    return grad_centered

def project_sum_and_clip(h, target_sum):
    # Clip to [0,1] and adjust sum using algorithm similar to enforce_constraints
    return enforce_constraints(h, target_sum)

def sa_optimize(initial_h, target_sum, time_budget, rng):
    n = len(initial_h)
    cur_h = np.copy(initial_h)
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    # parameters
    T0 = 0.02   # initial temperature
    p_swap = 0.3
    p_transfer = 0.3
    p_gradient = 0.25
    p_random = 0.15
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # temperature
        T = T0 * max(0.0, 1.0 - elapsed / time_budget)
        r = rng.random()
        if r < p_swap:
            # swap two indices
            i, j = rng.choice(n, size=2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < p_swap + p_transfer:
            i, j = rng.choice(n, size=2, replace=False)
            # transfer amount from i to j
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr <= 0:
                # try reverse
                max_tr = min(cur_h[j], 1.0 - cur_h[i])
                if max_tr <= 0:
                    continue
                donor, recv = j, i
            else:
                donor, recv = i, j
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        elif r < p_swap + p_transfer + p_gradient:
            grad = gradient_step(cur_h, target_sum)
            # bound step size
            max_step = 0.3
            step_norm = np.max(np.abs(grad))
            if step_norm > max_step:
                grad = grad * (max_step / step_norm)
            cand = cur_h - grad  # move opposite gradient to reduce overl
            # ensure bounds and sum
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_sum_and_clip(cand, target_sum)
        else:
            # random perturbation: choose 5% of indices and assign random values
            cand = cur_h.copy()
            m = max(1, n // 20)
            idx = rng.choice(n, size=m, replace=False)
            # assign uniform random numbers
            cand[idx] = rng.random(m)
            cand = project_sum_and_clip(cand, target_sum)
        cand_val = compute_c5(cand)
        accept = False
        if cand_val < cur_val:
            accept = True
        elif T > 0:
            prob = np.exp(-(cand_val - cur_val) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            cur_h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    # early exit
                    break
    return best_h, best_val

def gradient_refine(h, target_sum, max_iter=200):
    # iterative gradient descent using line search
    n = len(h)
    dx = 2.0 / n
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    for _ in range(max_iter):
        # find worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(np.argmax(corr))
        k = idx_max - (n - 1)
        # compute neighbor sum
        neigh = np.zeros(n)
        if k >= 0:
            neigh[: n - k] += cur_h[k:]
            neigh[k:] += cur_h[: n - k]
        else:
            kk = -k
            neigh[kk:] += cur_h[: n - kk]
            neigh[: n - kk] += cur_h[kk:]
        grad = dx * (1.0 - neigh)
        grad_centered = grad - np.mean(grad)
        # try step sizes
        improved = False
        best_candidate = cur_h
        best_candidate_val = cur_val
        for alpha in (0.3, 0.15, 0.07, 0.03):
            cand = cur_h - alpha * grad_centered
            cand = np.clip(cand, 0.0, 1.0)
            cand = enforce_constraints(cand, target_sum)
            val = compute_c5(cand)
            if val + 1e-12 < best_candidate_val:
                best_candidate_val = val
                best_candidate = cand.copy()
                improved = True
        if not improved:
            break
        cur_h = best_candidate
        cur_val = best_candidate_val
    return cur_h, cur_val

def smooth_and_fix(h, target_sum):
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    # adjust sum
    sm = enforce_constraints(sm, target_sum)
    return sm

def run(seed=42, budget_s=1100, **kwargs):
    # Determine remaining time budget
    start_all = time.time()
    rng = np.random.default_rng(seed)
    # choose n
    n = 200
    target_sum = n / 2.0
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # upsample to n
        h = upsample_initial(h0, n)
    else:
        h = random_initial(n, rng)
    # enforce
    h = enforce_constraints(h, target_sum)
    # Determine number of starts
    num_starts = 3
    # allocate time for SA per start
    total_budget = budget_s
    # reserve time for final refinement (smoothing + gradient)
    reserve = total_budget * 0.2
    sa_budget = (total_budget - reserve) / num_starts
    best_h_overall = h.copy()
    best_val_overall = compute_c5(best_h_overall)
    # initial
    # try each start
    for s in range(num_starts):
        elapsed = time.time() - start_all
        if elapsed + sa_budget + reserve > total_budget:
            # not enough time left
            break
        # generate start:
        if s == 0:
            start_h = h.copy()
        else:
            # random fresh start
            start_h = random_initial(n, rng)
            start_h = enforce_constraints(start_h, target_sum)
        # SA
        cur_h, cur_val = sa_optimize(start_h, target_sum, sa_budget, rng)
        if cur_val < best_val_overall:
            best_val_overall = cur_val
            best_h_overall = cur_h.copy()
        # early exit if good enough
        if best_val_overall < 0.3808:
            break
    # optional smoothing iterations
    elapsed = time.time() - start_all
    while elapsed < total_budget - 5.0:
        # apply smoothing a few times
        sm_h = smooth_and_fix(best_h_overall, target_sum)
        sm_val = compute_c5(sm_h)
        if sm_val + 1e-12 < best_val_overall:
            best_val_overall = sm_val
            best_h_overall = sm_h
        else:
            break
        elapsed = time.time() - start_all
    # gradient refinement
    if time.time() - start_all < total_budget - 2.0:
        refined_h, refined_val = gradient_refine(best_h_overall, target_sum, max_iter=200)
        if refined_val + 1e-12 < best_val_overall:
            best_h_overall = refined_h
            best_val_overall = refined_val
    # final safety projection
    best_h_overall = np.clip(best_h_overall, 0.0, 1.0)
    best_h_overall = enforce_constraints(best_h_overall, target_sum)
    return best_h_overall, best_val_overall, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_constraints(h, target_sum):
    """Clip to [0,1] and enforce exact sum by redistributing the residual."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # distribute on free (non‑boundary) entries
    for _ in range(5):
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            break
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
    # any leftover correction greedily
    if abs(diff) > 1e-12:
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
        h = np.clip(h, 0.0, 1.0)
    return h

def random_initial(n, rng):
    """Binary start: exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample_initial(initial, n_target):
    """Linear interpolation of a shorter seed to a longer grid."""
    n0 = len(initial)
    x0 = np.linspace(0, 2, n0, endpoint=False)
    x = np.linspace(0, 2, n_target, endpoint=False)
    return np.interp(x, x0, initial, left=0.0, right=0.0)

def gradient_step(h):
    """Gradient of the worst overlap shift (zero‑sum direction)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    neigh = np.zeros_like(h)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = dx * (1.0 - neigh)
    grad_centered = grad - np.mean(grad)
    return grad_centered

def smooth_and_fix(h, target_sum):
    """One Gaussian smoothing pass with sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return enforce_constraints(sm, target_sum)

    n = len(start_h)
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
