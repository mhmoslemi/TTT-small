def simulated_annealing(h0, M, time_budget, rng):
    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    curr_h = h0.copy()
    curr_c5 = best_c5

    start = time.time()
    # parameters
    # We set initial temperature proportional to best_c5 maybe 0.01
    T0 = 0.01
    # Determine max iterations maybe based on time, but we will loop until time up
    while time.time() - start < time_budget:
        # adapt temperature linearly decreasing
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1 - frac)  # linear cooling
        # propose move
        i, j = rng.choice(n, 2, replace=False)
        # decide direction of transfer randomly
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_transfer = min(curr_h[donor], 1.0 - curr_h[receiver])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        # propose new h
        h_new = curr_h.copy()
        h_new[donor] -= delta
        h_new[receiver] += delta
        # compute C5
        c5_new = compute_c5(h_new)
        # accept rule
        if c5_new < curr_c5 or rng.random() < np.exp(-(c5_new - curr_c5) / max(T, 1e-12)):
            curr_h = h_new
            curr_c5 = c5_new
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        # early break if best < target
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

def slsqp_refine(h0, M, time_budget, rng):
    n = len(h0)
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # initial t = compute_c5
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    # run minimize with SLSQP once
    try:
        res = minimize(fun=objective_t, x0=x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False})
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            # adjust sum
            cur = np.sum(h_opt)
            if abs(cur - M) > 1e-9:
                h_opt += (M - cur) / n
                h_opt = np.clip(h_opt, 0.0, 1.0)
            c5_opt = compute_c5(h_opt)
            return h_opt, c5_opt
    except Exception:
        pass
    return h0, compute_c5(h0)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helpers ----------
def compute_c5(h):
    """Maximum overlap integral for a given discretized h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def objective_t(x):
    return x[-1]

def generate_initial_h(n, M, rng):
    """Random feasible h with sum M (M = n/2)."""
    method = rng.integers(3)
    if method == 0:
        # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:
        # random uniform then scale, then fix bounds
        h = rng.random(n)
        h = h / np.sum(h) * M
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = M - cur
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            cur = np.sum(h)
            diff = M - cur
            if diff > 0:
                for i in np.argsort(h):
                    inc = min(diff, 1.0 - h[i])
                    h[i] += inc
                    diff -= inc
                    if diff <= 1e-12:
                        break
            elif diff < 0:
                for i in np.argsort(-h):
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
                    if diff >= -1e-12:
                        break
    else:
        # linear decreasing shape
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = M - cur
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)

                for i in np.argsort(-h):
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
                    if diff >= -1e-12:
                        break
    return h

def project_to_feasible(h, M):
    """Clip to [0,1] and adjust to exact sum M."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) < 1e-12:
        return h
    h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = M - cur
    if diff > 0:
        for i in np.argsort(h):
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        for i in np.argsort(-h):
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return h

# ---------- Simulated Annealing ----------
def simulated_annealing(h_start, M, time_budget, rng):
    n = len(h_start)
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    curr_h = best_h.copy()
    curr_c5 = best_c5
    start = time.time()
    T0 = 0.05  # initial temperature
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1.0 - frac)  # linear cooling
        i, j = rng.choice(n, 2, replace=False)
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_transfer = min(curr_h[donor], 1.0 - curr_h[receiver])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        h_new = curr_h.copy()
        h_new[donor] -= delta
        h_new[receiver] += delta
        c5_new = compute_c5(h_new)
        if c5_new < curr_c5 or rng.random() < np.exp(-(c5_new - curr_c5) / max(T, 1e-12)):
            curr_h, curr_c5 = h_new, c5_new
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

# ---------- SLSQP Refinement ----------
def slsqp_refine(h0, M, time_budget, rng):
    n = len(h0)
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    try:
        res = minimize(
            fun=objective_t,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1200, 'ftol': 1e-10, 'disp': False}
        )
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            cur = np.sum(h_opt)
            if abs(cur - M) > 1e-9:
                h_opt += (M - cur) / n
                h_opt = np.clip(h_opt, 0.0, 1.0)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

# ---------- Main entry ----------
