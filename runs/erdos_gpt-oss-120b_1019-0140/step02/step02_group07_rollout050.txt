def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target_sum = n/2.0
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    # adjust uniformly

    # after clipping, sum may be off slightly; fix via greedy
    residual = target_sum - np.sum(h)
    # distribute residual across elements with room
    if abs(residual) > 1e-12:
