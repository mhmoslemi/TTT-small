def optimise_for_n(n, time_budget, rng):

   M = n // 2
   sum_con = SumConstraint(n)
   overlap_constraints = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
   constraints = [{'type':'eq', 'fun': sum_con}]
   for oc in overlap_constraints:
        constraints.append({'type':'ineq','fun':oc})
   bounds = [(0.,1.)]*n + [(0., None)]

   best_h = None; best_c5 = np.inf

   # include parent if matches size
   if 'initial_h_values' in globals() and len(initial_h_values)==n:
        parent_h = np.array(initial_h_values, dtype=float)
        # adjust sum
        cur = np.sum(parent_h)
        if abs(cur - M) > 1e-12:
             parent_h += (M - cur)/n
             parent_h = np.clip(parent_h,0.,1.)
        c5 = compute_c5(parent_h)
        best_h, best_c5 = parent_h, c5

   # also include legendre if n==80
   if n == 80:
        # generate Legendre seq
        p = 79
        leg_seq = legendre_sequence(p)
        # add extra 1
        h_leg = np.concatenate([leg_seq, [1.0]])
        # random shift
        shift = rng.integers(n)
        h_leg = np.roll(h_leg, shift)
        # ensure sum equals M (should)
        c5 = compute_c5(h_leg)
        if c5 < best_c5:
            best_h, best_c5 = h_leg, c5

   start_local = time.time()
   while time.time() - start_local < time_budget - 0.5:
        # generate random binary h
        h0 = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h0[idx] = 1.0
        # optionally also allow continuous random as additional seeds occasionally: method = rng.integers(5) etc.

        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])
        try:
            res = minimize(fun=objective_t,
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
        except Exception:
            continue
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0., 1.)
        cur = np.sum(h_opt)
        if abs(cur - M) > 1e-9:
            h_opt += (M - cur)/n
            h_opt = np.clip(h_opt,0.,1.)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            # early break if good enough
            if best_c5 < 0.3808:
                break

   return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n/2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            # sum over i=0..n-1-k
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
