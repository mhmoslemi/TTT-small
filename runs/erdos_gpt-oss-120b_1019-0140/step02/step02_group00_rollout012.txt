import time
import numpy as np

# ---------- helper functions ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def random_binary_balanced(n, rng):
    """Return a binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    # problem size (chosen for speed/quality trade‑off)
    n = 256
    target = 0.3808

    best_h = None
    best_score = np.inf

    # ---- optionally use provided initial construction ----
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # bring to correct size by simple repeat if needed
        if init.size != n:
            factor = n // init.size
            init = np.repeat(init, factor)[:n]
        cand = project_onto_box_sum(init, n / 2.0)
        sc = compute_c5(cand)
        if sc < best_score:
            best_score, best_h = sc, cand.copy()
        # also try a pure binary version derived from the same data
        idx_desc = np.argsort(-cand)
        bin_cand = np.zeros(n, dtype=float)
        bin_cand[idx_desc[: n // 2]] = 1.0
        scb = compute_c5(bin_cand)
        if scb < best_score:
            best_score, best_h = scb, bin_cand.copy()

    # ---- simulated‑annealing on binary vectors ----
    outer_break = False
    while time_left() > 6.0 and not outer_break:
        h = random_binary_balanced(n, rng)
        score = compute_c5(h)

        if score < best_score:
            best_score, best_h = score, h.copy()
            if best_score <= target:
                break

        # annealing parameters
        T = 0.02
        decay = 0.9999
        stagnancy = 0
        max_stagnancy = 5000

        # estimate step budget for this run
        steps_budget = int(min(20000, max(1000, time_left() * 2000)))

        for _ in range(steps_budget):
            if time_left() < 2.0:
                break

            # propose a swap (1↔0)
            i = rng.integers(n)
            if h[i] < 0.5:
                continue
            j = rng.integers(n)
            if h[j] > 0.5:
                continue

            # execute swap
            h[i] = 0.0
            h[j] = 1.0
            new_score = compute_c5(h)
            delta = new_score - score

            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                score = new_score
                stagnancy = 0
                if score < best_score:
                    best_score, best_h = score, h.copy()
                    if best_score <= target:
                        outer_break = True
                        break
            else:
                # revert
                h[i] = 1.0
                h[j] = 0.0
                stagnancy += 1
                if stagnancy >= max_stagnancy:
                    break

            T *= decay
        # continue with a fresh random start if not terminated
    # -------------------------------------------------

    # ---- final refinement with momentum gradient descent ----
    if best_h is not None and time_left() > 4.0:
        remaining = time_left()
        max_iters = int(min(20000, max(5000, remaining * 800)))
        h_ref, sc_ref = gradient_descent_momentum(best_h, n_iter=max_iters,
                                                  step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
    # ---------------------------------------------------------

    # safeguard final projection
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5, dtype=float), n / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, n
