import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def binary_initial(n, rng):
    """Return a random binary vector of length n with exactly n/2 ones."""
    half = n // 2
    indices = rng.choice(n, size=half, replace=False)
    h = np.zeros(n, dtype=float)
    h[indices] = 1.0
    return h

def to_binary_from_continuous(v):
    """Convert a continuous vector to binary by taking the top half entries."""
    n = v.shape[0]
    half = n // 2
    idx = np.argsort(-v)[:half]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        cur_h = to_binary_from_continuous(init)
    else:
        n = 80  # even length is required for the sum constraint
        cur_h = binary_initial(n, rng)

    half = n // 2
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))

    # ----- simulated annealing parameters ------------------------------------
    T0 = max(1e-3, cur_c5 * 0.05)   # initial temperature
    T = T0
    decay = 0.9995                  # exponential cooling
    stagnation_seconds = 5.0        # restart if no improvement for this long
    last_improve = start_time

    # ----- main optimisation loop --------------------------------------------
    while True:
        # time guard
        if time.time() - start_time >= budget_s - 0.5:
            break

        # propose a swap between a 1 and a 0
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # perform the swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        ones[i_idx] = j
        zeros[j_idx] = i

        cand_c5 = compute_c5(cur_h)

        # ----- acceptance test ------------------------------------------------
        if cand_c5 < cur_c5 - 1e-12:          # improvement
            cur_c5 = cand_c5
            last_improve = time.time()
            if cand_c5 < best_c5 - 1e-12:
                best_c5 = cand_c5
                best_h = cur_h.copy()
        else:
            diff = cand_c5 - cur_c5
            if T > 0.0 and rng.random() < np.exp(-diff / T):
                cur_c5 = cand_c5
                # (no best update here because cand_c5 >= cur_c5)
            else:
                # revert swap
                cur_h[i] = 1.0
                cur_h[j] = 0.0
                ones[i_idx] = i
                zeros[j_idx] = j

        # ----- temperature decay ---------------------------------------------
        T *= decay

        # ----- stagnation handling --------------------------------------------
        if time.time() - last_improve > stagnation_seconds:
            # restart from a fresh random binary configuration
            cur_h = binary_initial(n, rng)
            cur_c5 = compute_c5(cur_h)
            ones = list(np.flatnonzero(cur_h == 1))
            zeros = list(np.flatnonzero(cur_h == 0))
            T = T0
            last_improve = time.time()

    # final safety: ensure binary and correct sum (should already hold)
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h.astype(float), best_c5, n
