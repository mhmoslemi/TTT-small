def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h_initial = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        h_initial = None

    target_sum = n/2

    # Helper to generate binary vector with exactly target_sum ones.
    def random_binary():
        vec = np.zeros(n, dtype=int)
        vec[rng.choice(n, target_sum, replace=False)] = 1
        return vec

    # Repair binary to have exact sum:
    def repair_binary(vec):
        cur = vec.sum()
        if cur == target_sum:
            return vec
        diff = target_sum - cur
        indices = np.arange(n)
        rng.shuffle(indices)
        if diff > 0:
            # need to add ones: pick zeros
            zeros = indices[vec[indices]==0]
            vec[zeros[:diff]] = 1
        else:
            # need to remove ones: pick ones
            ones = indices[vec[indices]==1]
            vec[ones[: -diff]] = 0
        return vec

    # create initial population
    pop_size = 80   # approx
    population = []
    fitness = [] 
    # add initial if available
    if h_initial is not None:
        # convert to binary using threshold (0.5)
        bin_initial = (h_initial >= 0.5).astype(int)
        bin_initial = repair_binary(bin_initial.copy())
        population.append(bin_initial)
        fitness.append(compute_c5(bin_initial.astype(float)))

    # add deterministic seeds: Thue-Morse and Rudin-Shapiro
    # generate sequences using functions
    # maybe also random.
    # ...

    while len(population) < pop_size:
        vec = random_binary()
        population.append(vec)
        fitness.append(compute_c5(vec.astype(float)))
    # main GA loop
    best_idx = np.argmin(fitness)
    best_vec = population[best_idx].copy()
    best_val = fitness[best_idx]

    # parameters
    tournament_k = 3
    mutation_rate = 0.1
    generation = 0
    last_improve_gen = 0

    while time.time() - start < budget_s - 5.0:
        generation += 1
        # selection and create new offspring
        new_pop = []
        new_fit = []
        # elite preservation: keep best individual
        new_pop.append(best_vec.copy())
        new_fit.append(best_val)
        # fill rest
        while len(new_pop) < pop_size:
            # tournament select two parents
            idx1 = tournament_select()
            idx2 = tournament_select()
            parent1 = population[idx1]
            parent2 = population[idx2]
            # crossover
            child = crossover(parent1, parent2)
            # mutation
            if rng.random() < mutation_rate:
                child = mutate_swap(child)
            # repair if needed
            child = repair_binary(child)
            # evaluate
            cval = compute_c5(child.astype(float))
            new_pop.append(child)
            new_fit.append(cval)
        # replace population
        population = new_pop
        fitness = new_fit
        # update best
        cur_best_idx = np.argmin(fitness)
        cur_best_val = fitness[cur_best_idx]
        if cur_best_val < best_val - 1e-12:
            best_val = cur_best_val
            best_vec = population[cur_best_idx].copy()
            last_improve_gen = generation
        # adapt mutation: if no improvement for 50 gens, increase mut_rate
        if generation - last_improve_gen > 50:
            mutation_rate = min(0.5, mutation_rate * 1.5)
            last_improve_gen = generation  # reset to maybe increase at later stages?
        # maybe reduce later, but fine.

    # after GA, possibly run continuous annealing
    # convert best binary to float
    best_h_float = best_vec.astype(float)
    # run continuous anneal for remaining time
    remaining = budget_s - (time.time() - start)
    if remaining > 10.0:
        end_time = time.time() + remaining
        best_h_float, best_val = continuous_anneal(best_h_float, end_time, rng)
    # ensure exact sum and bounds
    best_h_float = np.clip(best_h_float, 0.0, 1.0)
    diff = target_sum - np.sum(best_h_float)
    if abs(diff) > 1e-12:
        best_h_float += diff / n
        # clip again
        best_h_float = np.clip(best_h_float, 0.0, 1.0)
        # final residual correction loop akin prior.
        residual = target_sum - np.sum(best_h_float)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h_float[i], 1 - best_h_float[i])
            best_h_float[i] += adj
            residual = target_sum - np.sum(best_h_float)
    # compute final c5
    final_c5 = compute_c5(best_h_float)
    return best_h_float, final_c5, n

def subgradient_optimize(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n/2.0
    h = start_h.copy()
    # enforce feasibility: clip, adjust sum
    h = np.clip(h, 0.0, 1.0)
    cur_sum = np.sum(h)
    if abs(cur_sum - target_sum) > 1e-12:
        h += (target_sum - cur_sum) / n
        h = np.clip(h, 0.0, 1.0)
        residual = target_sum - np.sum(h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -h[i], 1 - h[i])
            h[i] += adj
            residual = target_sum - np.sum(h)

    # prepare FFT length
    L = 1
    while L < 2*n:
        L <<= 1
    # schedule
    max_iter = 2000000
    init_alpha = 0.5
    final_alpha = 1e-4
    best_h = h.copy()
    best_c5 = compute_c5(h)
    it = 0
    # pre-allocate arrays for speed?
    while time.time() < time_end and it < max_iter:
        # FFT
        fa = np.fft.rfft(h, n=L)
        fb = np.fft.rfft(1.0 - h, n=L)
        conv = np.fft.irfft(fa * fb, n=L)[:2*n-1]
        idx = np.argmax(conv)
        k = idx - (n - 1)
        # gradient
        G = np.zeros_like(h)
        if k >= 0:
            i = np.arange(0, n - k)
            G[i] += (1.0 - h[i + k])
            G[i + k] -= h[i]
        else:
            s = -k
            i = np.arange(s, n)
            G[i] += (1.0 - h[i - s])
            G[i - s] -= h[i]
        # step size
        alpha = init_alpha * (1 - it / max_iter) + final_alpha * (it / max_iter)
        # update
        h = h - alpha * G
        # clip
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        cur_sum = np.sum(h)
        if abs(cur_sum - target_sum) > 1e-12:
            h += (target_sum - cur_sum) / n
            h = np.clip(h, 0.0, 1.0)
            residual = target_sum - np.sum(h)

                    break

                h[i] += adj
                residual = target_sum - np.sum(h)
        # evaluate
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # early exit if perfect

import time
import math
import numpy as np

def compute_c5(arr):
    """Approximate C5 by correlation of h and (1-h)."""
    n = len(arr)
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Thue‑Morse binary sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                      dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    """Rudin‑Shapiro binary sequence (+1/-1 → 1/0) of length n."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def binary_hill_climb(start_bin, time_end, rng):
    """Hill climbing on a binary vector (swap moves)."""
    n = len(start_bin)
    h = start_bin.astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    while time.time() < time_end and no_improve < 3000:
        ones = np.flatnonzero(start_bin)
        zeros = np.flatnonzero(1 - start_bin)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = start_bin.copy()
        cand[i] = 0
        cand[j] = 1
        cand_f = cand.astype(float)
        cand_c5 = compute_c5(cand_f)
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand_f.copy()
            start_bin = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def subgradient_optimize(start_h, time_end, rng):
    """Sub‑gradient descent on the max‑overlap objective."""
    n = len(start_h)
    target = n / 2.0
    h = np.clip(start_h, 0.0, 1.0)
    # enforce exact sum
    cur_sum = np.sum(h)
    if abs(cur_sum - target) > 1e-12:
        h += (target - cur_sum) / n
        h = np.clip(h, 0.0, 1.0)
        residual = target - np.sum(h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -h[i], 1.0 - h[i])
            h[i] += adj
            residual = target - np.sum(h)

    # FFT length (next power of two >= 2n)
    L = 1
    while L < 2 * n:
        L <<= 1

    max_iter = 2_000_000
    init_alpha = 0.5
    final_alpha = 1e-4
    best_h = h.copy()
    best_c5 = compute_c5(h)
    it = 0

    while time.time() < time_end and it < max_iter:
        # correlation via FFT (linear convolution)
        fa = np.fft.rfft(h, n=L)
        fb = np.fft.rfft(1.0 - h, n=L)
        conv = np.fft.irfft(fa * fb, n=L)[:2 * n - 1]
        idx = int(np.argmax(conv))
        k = idx - (n - 1)

        # gradient for the active shift k
        G = np.zeros_like(h)
        if k >= 0:
            i = np.arange(0, n - k, dtype=int)
            G[i] += (1.0 - h[i + k])
            G[i + k] -= h[i]
        else:
            s = -k
            i = np.arange(s, n, dtype=int)
            G[i] += (1.0 - h[i - s])
            G[i - s] -= h[i]

        # step size (linear schedule)
        alpha = init_alpha * (1 - it / max_iter) + final_alpha * (it / max_iter)
        h = h - alpha * G
        h = np.clip(h, 0.0, 1.0)

        # enforce sum after projection
        cur_sum = np.sum(h)
        if abs(cur_sum - target) > 1e-12:
            h += (target - cur_sum) / n
            h = np.clip(h, 0.0, 1.0)
            residual = target - np.sum(h)
            for i in range(n):
                if abs(residual) < 1e-12:
                    break
                adj = np.clip(residual, -h[i], 1.0 - h[i])
                h[i] += adj
                residual = target - np.sum(h)

        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            if best_c5 <= 0.3808:
                break
        it += 1

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modest size that balances resolution and speed
    n = 200
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # Helper to force binary vectors to have exactly target_sum ones
    # ------------------------------------------------------------------
    def fix_binary_sum(vec):
        cur = int(vec.sum())
        if cur == target_sum:
            return vec
        diff = int(target_sum - cur)  # positive -> need more ones
        idxs = np.arange(n)
        rng.shuffle(idxs)
        if diff > 0:
            zeros = idxs[vec[idxs] == 0]
            vec[zeros[:diff]] = 1
        else:
            ones = idxs[vec[idxs] == 1]
            vec[ones[: -diff]] = 0
        return vec

    # ------------------------------------------------------------------
    # Build a list of diverse starting vectors
    # ------------------------------------------------------------------
    starts = []

    # 1. Resample the provided parent (if any) to length n
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        m = len(parent)
        if m != n:
            old_x = np.linspace(0.0, 1.0, m, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            parent = np.interp(new_x, old_x, parent)
        parent = np.clip(parent, 0.0, 1.0)
        parent += (target_sum - np.sum(parent)) / n
        parent = np.clip(parent, 0.0, 1.0)
        residual = target_sum - np.sum(parent)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -parent[i], 1.0 - parent[i])
            parent[i] += adj
            residual = target_sum - np.sum(parent)
        starts.append(parent)

    # 2. Thue‑Morse binary pattern (adjusted to exact sum)
    tm = thue_morse_seq(n).astype(float)
    tm = fix_binary_sum(tm.astype(int)).astype(float)
    starts.append(tm)

    # 3. Rudin‑Shapiro binary pattern (adjusted)
    rs = rudin_shapiro_seq(n).astype(float)
    rs = fix_binary_sum(rs.astype(int)).astype(float)
    starts.append(rs)

    # 4. Uniform 0.5 vector (feasible)
    uniform = np.full(n, 0.5, dtype=float)
    starts.append(uniform)

    # 5–9. A handful of random binary starts with tiny noise
    for _ in range(5):
        bin_vec = np.zeros(n, dtype=int)
        bin_vec[rng.choice(n, int(target_sum), replace=False)] = 1
        # add tiny random perturbation to escape binary basin
        noise = rng.random(n) * 0.02
        cand = bin_vec.astype(float) + noise
        cand = np.clip(cand, 0.0, 1.0)
        cand += (target_sum - np.sum(cand)) / n
        cand = np.clip(cand, 0.0, 1.0)
        residual = target_sum - np.sum(cand)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -cand[i], 1.0 - cand[i])
            cand[i] += adj
            residual = target_sum - np.sum(cand)
        starts.append(cand)

    # ------------------------------------------------------------------
    # Main optimisation: run sub‑gradient descent on each start,
    # allocating time proportionally.
    # ------------------------------------------------------------------
    best_h = None
    best_c5 = np.inf
    for start_h in starts:
        now = time.time()
        time_left = budget_s - (now - start_time)
        if time_left <= 5.0:
            break
        # spend at most 30 s or half the remaining time on a single start
        chunk = min(30.0, time_left * 0.5)
        end_time = now + chunk
        cand_h, cand_c5 = subgradient_optimize(start_h, end_time, rng)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h.copy()
        if best_c5 <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Final refinement: binary hill‑climb from the rounded best solution
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start_time < budget_s - 2.0:
        bin_start = (best_h >= 0.5).astype(int)
        end_time = start_time + budget_s - 0.5
        final_h, final_c5 = binary_hill_climb(bin_start, end_time, rng)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()

    # ------------------------------------------------------------------
    # Final safety: enforce bounds and exact sum
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = np.sum(best_h)
    if abs(cur_sum - target_sum) > 1e-12:
        best_h += (target_sum - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1.0 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
