project_sum_and_bounds(x, target):
    # Clip between 0 and 1
    x = np.clip(x, 0.0, 1.0)
    # Adjust sum to target.
    # Often diff may be small; adjust uniformly.
    sum_x = x.sum()
    diff = target - sum_x
    if abs(diff) > 1e-12:
        # While diff not zero adjust
        # We'll try to add diff proportionally to non-saturated entries.
        for _ in range(10):
            mask = (x > 0) & (x < 1)
            if mask.sum() == 0:
                break
            # Distribute diff equally among mask entries
            delta = diff / mask.sum()
            # Clip each entry
            deltas = np.clip(delta, -x[mask], 1 - x[mask])
            x[mask] += deltas
            sum_x = x.sum()
            diff = target - sum_x
            if abs(diff) < 1e-12:
                break
        # Final fallback: greedy fill
        if abs(diff) > 1e-12:
            # Increase smallest entries if diff>0 else decrease largest
            indices = np.argsort(x) if diff > 0 else np.argsort(-x)
            for idx in indices:
                if diff <= 0:
                    break
                cap = 1 - x[idx] if diff > 0 else x[idx]
                inc = min(diff, cap)
                if inc > 0:
                    x[idx] += inc
                    diff -= inc
    # final clip
    x = np.clip(x, 0.0, 1.0)
    # ensure sum approx (maybe due to floating errors)
    return x

def project_sum_and_bounds(x, target):
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(x + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(x + (lo+hi)/2.0, 0.0, 1.0)

def gradient_descent(h0, time_budget, rng):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best = h.copy()
    best_c5 = compute_c5(best)
    start = time.time()
    beta = 2.0
    max_beta = 50.0
    alpha0 = 0.2
    no_improve = 0

    while time.time() - start < time_budget:
        # compute correlation vector
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0 / n)

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase beta gradually
        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)

        # softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / len(v)
        else:
            w /= w_sum

        # gradient: each term contributes weight * derivative of correlation
        # derivative of correlation w.r.t h[i] is (1 - h[i-k]) for lag k plus -h[i+k] for opposite orientation? Use earlier code.
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # enforce zero-sum direction
        g = g - np.mean(g)

        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        # add small random noise
        h_new = h - alpha * g + 0.01 * rng.normal(size=n)
        h_new = project_sum_and_bounds(h_new, target)

        # accept if improvement
        new_c5 = compute_c5(h_new)
        if new_c5 < c5 or rng.random() < 0.001:
            h = h_new

        # random shake to escape
        if no_improve > 2000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.05, size=n), target)
            no_improve = 0

    return best, best_c5

def cem_search(h0, time_budget, rng):
    n = len(h0)
    target = n / 2.0
    mu = h0.copy()
    sigma = np.full(n, 0.2) # standard deviation
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    iteration = 0
    while time.time() - start < time_budget:
        # Determine remaining time
        remaining = time_budget - (time.time() - start)
        # allocate iteration time of maybe 0.5 sec; dynamic?
        # Decide sample size
        N = 200
        # generate candidates
        samples = rng.normal(loc=mu, scale=sigma, size=(N, n))
        # evaluate
        cands = []
        for s in samples:
            s = np.clip(s, 0.0, 1.0)
            s = project_sum_and_bounds(s, target)
            c5 = compute_c5(s)
            cands.append((c5, s))
        # sort
        cands.sort(key=lambda x: x[0])
        # update best
        if cands[0][0] < best_c5:
            best_c5, best_h = cands[0][0], cands[0][1].copy()
        # select elite
        elite = np.stack([c[1] for c in cands[:max(5, N // 10)]], axis=0)
        # update mu and sigma (std dev)
        mu = elite.mean(axis=0)
        sigma = elite.std(axis=0) + 1e-4  # avoid zero
        iteration += 1
        # early break if good
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

    return best_h, final_c5, n

def compute_c5(h):

def project_sum_and_bounds(x, target):
    # binary search for lambda shift that makes sum after clipping equal target
    low, high = -1.0, 1.0
    for _ in range(60):
        lam = (low + high) * 0.5
        proj = np.clip(x + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            high = lam
        else:
            low = lam
    proj = np.clip(x + (low + high) * 0.5, 0.0, 1.0)
    # tiny correction if needed
    diff = target - proj.sum()
    if abs(diff) > 1e-9:

            mask = (proj > 0) & (proj < 1)
            if mask.sum() == 0:
                break
            delta = diff / mask.sum()
            delta = np.clip(delta, -proj[mask], 1 - proj[mask])
            proj[mask] += delta
            diff = target - proj.sum()
            if abs(diff) < 1e-9:
                break
    return proj

            low = lam
    proj = np.clip(x + (low + high) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        for _ in range(10):
            mask = (proj > 0) & (proj < 1)
            if mask.sum() == 0:
                break
            delta = diff / mask.sum()
            delta = np.clip(delta, -proj[mask], 1 - proj[mask])
            proj[mask] += delta
            diff = target - proj.sum()
            if abs(diff) < 1e-9:
                break
    return proj

def gradient_descent(h0, time_budget, rng):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    beta = 2.0
    max_beta = 50.0
    alpha0 = 0.2
    no_improve = 0

    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0 / n)

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)

        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        ws = w.sum()
        w = w / ws if ws > 0 else np.full_like(w, 1.0 / w.size)

        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        g -= np.mean(g)

        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        h_new = h - alpha * g + 0.01 * rng.normal(size=n)
        h_new = project_sum_and_bounds(h_new, target)

        new_c5 = compute_c5(h_new)
        if new_c5 < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 2000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.05, size=n), target)
            no_improve = 0

    return best_h, best_c5

def cem_search(h0, time_budget, rng):
    n = len(h0)
    target = n / 2.0
    mu = h0.copy()
    sigma = np.full(n, 0.2)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()

    while time.time() - start < time_budget:
        N = 200
        samples = rng.normal(loc=mu, scale=sigma, size=(N, n))
        c5_vals = np.empty(N)
        proj_samples = np.empty_like(samples)

        for i in range(N):
            s = np.clip(samples[i], 0.0, 1.0)
            s = project_sum_and_bounds(s, target)
            proj_samples[i] = s
            c5_vals[i] = compute_c5(s)

        idx_best = np.argmin(c5_vals)
        if c5_vals[idx_best] < best_c5:
            best_c5 = c5_vals[idx_best]
            best_h = proj_samples[idx_best].copy()

        elite_num = max(5, N // 10)
        elite_idx = np.argpartition(c5_vals, elite_num)[:elite_num]
        elite = proj_samples[elite_idx]
        mu = elite.mean(axis=0)
        sigma = elite.std(axis=0) + 1e-4

        if best_c5 < 0.3808:
            break

    return best_h, best_c5

def binary_local_search(h_bin, time_budget, rng):
    n = len(h_bin)
    target = int(round(n / 2))
    current = h_bin.copy()
    cur_sum = int(current.sum())
    if cur_sum != target:
        diff = target - cur_sum
        if diff > 0:
            zeros = np.where(current == 0)[0]
            if len(zeros) >= diff:
                pick = rng.choice(zeros, size=diff, replace=False)
                current[pick] = 1
        elif diff < 0:
            ones = np.where(current == 1)[0]
            if len(ones) >= -diff:
                pick = rng.choice(ones, size=-diff, replace=False)
                current[pick] = 0

    best = current.copy()
    best_c5 = compute_c5(best)
    start = time.time()
    no_improve = 0

    while time.time() - start < time_budget:
        ones_idx = np.where(current == 1)[0]
        zeros_idx = np.where(current == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        improved = False
        for _ in range(500):
            i1 = rng.choice(ones_idx)
            i0 = rng.choice(zeros_idx)
            cand = current.copy()
            cand[i1] = 0
            cand[i0] = 1
            c5cand = compute_c5(cand)
            if c5cand + 1e-12 < best_c5:
                best_c5 = c5cand
                best = cand.copy()
                current = cand
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                for _ in range(3):
                    i1 = rng.choice(ones_idx)
                    i0 = rng.choice(zeros_idx)
                    current[i1] = 0
                    current[i0] = 1
                no_improve = 0
        else:
            no_improve = 0

        if best_c5 < 0.3808:
            break

    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = rng.random(n)

    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    # Phase 1: gradient descent (~30% of budget)
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    t1 = remaining * 0.3
    best_h, best_c5 = gradient_descent(h0, t1, rng)

    # Phase 2: cross‑entropy method (~40% of budget)
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    t2 = remaining * 0.4
    if t2 > 0:
        h_cem, c5_cem = cem_search(best_h, t2, rng)
        if c5_cem < best_c5:
            best_h, best_c5 = h_cem, c5_cem

    # Phase 3: binary local search (remaining time)
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    if remaining > 0:
        binary = (best_h >= 0.5).astype(float)
        cur_sum = int(binary.sum())
        diff = int(round(target - cur_sum))
        if diff > 0:
            zeros = np.where(binary == 0)[0]
            order = np.argsort(-best_h[zeros])
            binary[zeros[order[:diff]]] = 1
        elif diff < 0:
            ones = np.where(binary == 1)[0]
            order = np.argsort(best_h[ones])
            binary[ones[order[: -diff]]] = 0
        h_bin, c5_bin = binary_local_search(binary, remaining, rng)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin

    # Final refinement if any time left
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    if remaining > 5.0:
        h_ref, c5_ref = gradient_descent(best_h, remaining, rng)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
