import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions (top level) ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Adjust v (in [0,1]) so that its sum equals target (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # scale first, then fix residual
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # distribute diff iteratively
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    """Random vector in [0,1]^n with exact sum==target."""
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

def random_binary_vector(n, target, rng):
    """Binary vector with exactly target ones (target assumed integer)."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def sum_constraint_eq(x, n, target):
    """Equality: sum of first n entries equals target."""
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    """Inequality: t - overlap(k) >= 0."""
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def objective_fun(x):
    """Objective for SLSQP: minimize the auxiliary variable t."""
    return x[-1]

def build_constraints(n, target, dx):
    """Create SLSQP constraints (equality + one inequality per non-zero shift)."""
    cons = [
        {"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}
    ]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append(
            {"type": "ineq", "fun": overlap_constraint_ineq, "args": (k, n, dx)}
        )
    return cons

def continuous_anneal(h, cur_val, best_h, best_val, start_time, deadline, rng):
    """Simple simulated annealing that transfers mass between points."""
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        # contributions for the worst shift
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)

        i_local = np.argmax(prod)
        i = i_range[i_local]

        # choose a destination index with headroom
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta

        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine dimension and initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n / 2.0, rng)

    target = n / 2.0
    dx = 2.0 / n
    h0 = enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Build constraints once
    constraints = build_constraints(n, target, dx)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # ---------- SLSQP phase ----------
    while time.time() < deadline - 5.0:
        # Random start: binary 70% of the time, else perturb best
        if rng.random() < 0.7:
            start_h = random_binary_vector(n, target, rng)
        else:
            start_h = best_h.copy()
            # small random perturbation while preserving sum
            for _ in range(3):
                src = np.where(start_h > 1e-8)[0]
                dst = np.where(start_h < 1.0 - 1e-8)[0]
                if len(src) == 0 or len(dst) == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(start_h[i], 1.0 - start_h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                start_h[i] -= delta
                start_h[j] += delta
            start_h = enforce_sum(start_h, target, rng)

        t0 = compute_c5(start_h)
        x0 = np.concatenate([start_h, [t0]])

        try:
            res = minimize(
                fun=objective_fun,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 2000, "ftol": 1e-10, "disp": False},
            )
        except Exception:
            break

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = enforce_sum(h_opt, target, rng)
        val = compute_c5(h_opt)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # ---------- Annealing phase ----------
    if time.time() < deadline - 0.5:
        cur_h = best_h.copy()
        cur_val = best_val
        best_h, best_val = continuous_anneal(
            cur_h, cur_val, best_h, best_val, start, deadline, rng
        )

    # Final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
