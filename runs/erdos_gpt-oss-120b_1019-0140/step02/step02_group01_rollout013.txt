        j = i + k
    a, b = h[i], h[j]
    avg = (a + b) * 0.5
    if np.isclose(avg, a) and np.isclose(avg, b):
        return None
    h2 = h.copy()
    h2[i] = avg
    h2[j] = avg
    return h2


# ------------------------------------------------------------
def transfer_move(h, rng):
    """Move a small amount from one index to another (preserves sum)."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    # decide direction
    if rng.integers(2) == 0:
        # i receives from j
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            return None
        delta = rng.random() * max_transfer * 0.5
        h2 = h.copy()
        h2[i] += delta
        h2[j] -= delta
    else:
        # j receives from i
        max_transfer = min(1.0 - h[j], h[i])
        if max_transfer <= 0:
            return None
        delta = rng.random() * max_transfer * 0.5
        h2 = h.copy()
        h2[j] += delta
        h2[i] -= delta
    return h2


# ------------------------------------------------------------
def local_search(h0, budget, rng):
    """Hill‑climbing with smoothing and transfer moves."""
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    no_improve = 0

    while time.time() - start < budget:
        # worst lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        worst_k = max_idx - (n - 1)

        # try smoothing on worst lag
        cand = smooth_pair(h, worst_k, rng)
        if cand is not None:
            c5c = np.max(np.correlate(cand, 1.0 - cand, mode='full')) * dx
            if c5c + 1e-12 < best_c5:
                h, best_h, best_c5 = cand, cand, c5c
                no_improve = 0
                continue

        # try a transfer move
        cand = transfer_move(h, rng)
        if cand is not None:
            c5c = np.max(np.correlate(cand, 1.0 - cand, mode='full')) * dx
            if c5c + 1e-12 < best_c5:
                h, best_h, best_c5 = cand, cand, c5c
                no_improve = 0
                continue

        no_improve += 1
        # occasional random restart if stuck
        if no_improve >= 2000:
            h = generate_random_h(n, rng)
            no_improve = 0

    return best_h, best_c5


# ------------------------------------------------------------
def search_n(n, total_budget, rng):
    """Multiple short restarts for a given resolution."""
    best_h = None
    best_c5 = np.inf
    start = time.time()
    first = True

    while time.time() - start < total_budget:
        remaining = total_budget - (time.time() - start)
        # each restart gets at most 5 s (or the remaining time)
        chunk = min(5.0, remaining)
        if first and 'initial_h_values' in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
            target = n // 2
            cur = np.sum(h0)
            if abs(cur - target) > 1e-12:
                h0 += (target - cur) / n
                h0 = np.clip(h0, 0.0, 1.0)
            first = False
        else:
            if 'initial_h_values' in globals() and n > len(initial_h_values):
                h0 = upsample_initial_h(initial_h_values, n)
            else:
                h0 = generate_random_h(n, rng)

        h_opt, c5_opt = local_search(h0, chunk, rng)

        if c5_opt < best_c5 - 1e-12:
            best_c5, best_h = c5_opt, h_opt
            if best_c5 < 0.3808:
                break

        if time.time() - start > total_budget - 0.5:
            break

    return best_h, best_c5


# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h with a low Erdős C₅ bound."""
    start_global = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate discretisations (even numbers, modest size)
    candidates = [80, 96, 112, 128, 144, 160, 192, 224, 256]

    # prioritize a parent construction if present
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidates:
                candidates = [n0] + candidates
            else:
                candidates = [n0] + [n for n in candidates if n != n0]

    for n in candidates:
        if time.time() - start_global > budget_s - 5:
            break
        # allocate up to 200 s per resolution, otherwise half of the remaining budget
        remain = budget_s - (time.time() - start_global)
        alloc = min(200.0, remain * 0.5)

        h_opt, c5_opt = search_n(n, alloc, rng)

        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # fallback to a simple constant if nothing better is found
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)

    # final projection onto the feasible set
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n // 2
    cur = np.sum(best_h)
    if abs(cur - target) > 1e-9:
        best_h += (target - cur) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
