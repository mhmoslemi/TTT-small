def hill_climb_binary(h0, dx, deadline):
    h = h0.copy()
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    while time.time() < deadline - 0.1:
        # compute shift matrix sums
        shifts = np.arange(-(n-1), n)
        # Precompute Hsum
        Hsum = np.zeros((n, 2*n-1), dtype=float)
        for i in range(n):
            idx_plus = i + shifts
            valid_plus = (idx_plus >= 0) & (idx_plus < n)
            Hsum[i, valid_plus] = h[idx_plus[valid_plus]]  # H_plus
            idx_minus = i - shifts
            valid_minus = (idx_minus >= 0) & (idx_minus < n)
            Hsum[i, valid_minus] += h[idx_minus[valid_minus]]  # add H_minus
        # Now Hsum contains sum of plus and minus for each i
        # Compute indices sets
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_improvement = 0.0
        best_swap = None
        best_delta = None
        best_new_max = cur_max
        for i in ones:
            hi = Hsum[i]
            for j in zeros:
                delta = hi - Hsum[j]
                new_corr = corr + delta
                new_max = new_corr.max()
                improvement = cur_max - new_max
                if improvement > best_improvement + 1e-14:
                    best_improvement = improvement
                    best_swap = (i, j)
                    best_delta = delta
                    best_new_max = new_max
        if best_improvement > 1e-12:
            i, j = best_swap
            # perform swap
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + best_delta
            cur_max = best_new_max
            # update best if improved
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            break
    return best_h, best_max * dx

import time
import numpy as np

def compute_c5(h):
    """Return C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def hill_climb_binary(h_start, dx, deadline):
    """
    Improve a binary (0/1) vector h by swapping 1s and 0s to reduce max correlation.
    Returns best vector found and its C5 value.
    """
    h = h_start.copy()
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max

    shifts = np.arange(-(n - 1), n)                     # length 2*n-1
    center_idx = n - 1                                 # index of shift 0

    while time.time() < deadline - 0.1:
        # build shift matrices
        H_plus = np.zeros((n, 2 * n - 1), dtype=float)
        H_minus = np.zeros((n, 2 * n - 1), dtype=float)
        for i in range(n):
            idx_plus = i + shifts
            v = (idx_plus >= 0) & (idx_plus < n)
            H_plus[i, v] = h[idx_plus[v]]
            idx_minus = i - shifts
            v = (idx_minus >= 0) & (idx_minus < n)
            H_minus[i, v] = h[idx_minus[v]]
        Hsum = H_plus + H_minus

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improve = 0.0
        best_swap = None
        best_delta = None
        best_new_max = cur_max

        for i in ones:
            hi = Hsum[i]
            for j in zeros:
                delta = hi - Hsum[j]

                # correction for shift 0 (term3 = 2)
                delta[center_idx] -= 2.0
                # correction for shifts ±(j-i) (term3 = -1)
                d = j - i
                if d != 0:
                    idx1 = d + center_idx
                    if 0 <= idx1 < delta.size:
                        delta[idx1] += 1.0
                    idx2 = -d + center_idx
                    if 0 <= idx2 < delta.size:
                        delta[idx2] += 1.0

                new_corr = corr + delta
                new_max = new_corr.max()
                improve = cur_max - new_max
                if improve > best_improve + 1e-14:
                    best_improve = improve
                    best_swap = (i, j)
                    best_delta = delta
                    best_new_max = new_max

        if best_improve > 1e-12:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + best_delta
            cur_max = best_new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            break

    return best_h, best_max * dx

        i = i_range[i_local]

            remaining = deadline - (time.time() - start_time)
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta

        if best < 0.3808:
            break
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine dimensionality
    if "initial_h_values" in globals():
        init_raw = np.array(initial_h_values, dtype=float)
        n = init_raw.shape[0]
    else:
        n = 80
    dx = 2.0 / n
    target = n / 2.0

    # Prepare several binary start vectors
    init_vectors = []

    if "initial_h_values" in globals():
        # Turn the provided values into a binary vector (top half ones)
        idx_desc = np.argsort(-init_raw)
        binary = np.zeros(n, dtype=float)
        half = int(np.floor(target))
        binary[idx_desc[:half]] = 1.0
        if not np.isclose(target, half):
            # add a fractional entry if needed (rare, n odd)
            binary[idx_desc[half]] = target - half
        init_vectors.append(binary)

    # add a few random binary vectors
    for _ in range(8):
        if time.time() >= deadline - 5.0:
            break
        h_rand = np.zeros(n, dtype=float)
        ones = rng.choice(n, int(target), replace=False)
        h_rand[ones] = 1.0
        init_vectors.append(h_rand)

    best_h = None
    best_val = np.inf

    # Hill‑climbing on each start
    for h0 in init_vectors:
        if time.time() >= deadline - 5.0:
            break
        # Hill‑climb works only for integer target (even n)
        if n % 2 == 0:
            h_opt, val = hill_climb_binary(h0, dx, deadline)
        else:
            # fallback to continuous optimisation for odd n
            h_opt = h0.copy()
            val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    if best_h is None:
        # guaranteed fallback: simple half‑ones pattern
        best_h = np.zeros(n, dtype=float)
        best_h[: int(target)] = 1.0
        best_val = compute_c5(best_h)

    # Continuous annealing phase for fine tuning
    if time.time() < deadline - 0.5:
        best_h, best_val = continuous_anneal(best_h, best_val, start, deadline, rng)

    # Final safety clip and sum correction
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h += (target - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
