def repair_h(x):
    # x is array shape (n,)
    # clip to [0,1]
    y = np.clip(x, 0.0, 1.0)
    target = len(x) / 2.0
    # iterative distribute residual over interior points
    for _ in range(10):
        sum_y = y.sum()
        if abs(sum_y - target) < 1e-12:
            break
        residual = target - sum_y
        # find interior indices
        interior = np.where((y > 0) & (y < 1))[0]
        if interior.size == 0:
            # cannot adjust further, break
            break
        delta = residual / interior.size
        y[interior] += delta
        # clip again
        y = np.clip(y, 0.0, 1.0)
    # final small correction: if sum still off, adjust few indices
    sum_y = y.sum()
    if abs(sum_y - target) > 1e-12:
        # adjust by adding small amount uniformly and clipping after each step
        diff = target - sum_y
        # we can add diff/n to all points and clip (maybe simple)
        y += diff / len(y)
        y = np.clip(y, 0.0, 1.0)
        # after this maybe still off; try iterative again
        for _ in range(10):
            sum_y = y.sum()
            if abs(sum_y - target) < 1e-12:
                break
            residual = target - sum_y
            interior = np.where((y > 0) & (y < 1))[0]
            if interior.size == 0:
                break
            delta = residual / interior.size
            y[interior] += delta
            y = np.clip(y, 0.0, 1.0)
    # should satisfy constraints
    return y

def periodic_search(n, time_deadline):
    best_c5 = np.inf
    best_h = None
    target_ones = n // 2
    # get divisors of n up to certain max
    max_d = min(16, n)  # don't exceed n

    for d in divisors:

        # generate combinations of half positions out of d
        for ones_pos in itertools.combinations(range(d), half):
            pattern = np.zeros(d, dtype=float)
            pattern[list(ones_pos)] = 1.0

                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

def repair_h(x):

    no_imp = 0
    while time.time() < time_end and no_imp < 3000:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best = cand
            best_c5 = cand_c5
            no_imp = 0
        else:
            no_imp += 1
    return best.astype(float), best_c5

def continuous_anneal(start_h, time_end, rng):
    """Simple simulated annealing moving a small amount of mass between two coordinates."""
    n = len(start_h)
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    it = 0
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 200000
    while time.time() < time_end:
        inc = np.where(h < 1 - 1e-12)[0]
        dec = np.where(h > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            continue
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        it += 1
    best_h = repair_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def periodic_search(n, deadline):
    """Exhaustive search over binary patterns of small period (≤20) and all cyclic shifts."""
    best_c5 = np.inf
    best_h = None
    target = 0.3808
    max_period = min(20, n)
    divisors = [d for d in range(2, max_period + 1, 2) if n % d == 0]
    for d in divisors:
        half = d // 2
        repeats = n // d
        for ones_pos in itertools.combinations(range(d), half):
            pattern = np.zeros(d, dtype=float)
            pattern[list(ones_pos)] = 1.0
            base = np.tile(pattern, repeats)
            for shift in range(d):
                if time.time() > deadline:
                    return best_h, best_c5
                h = np.roll(base, shift)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    if best_c5 <= target:
                        return best_h, best_c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # initialise
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = repair_h(np.array(initial_h_values, dtype=float))
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
        h0 = repair_h(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: exhaustive periodic search (≈40% of time budget)
    phase1_end = start_time + budget_s * 0.4
    periodic_h, periodic_c5 = periodic_search(n, phase1_end)
    if periodic_h is not None and periodic_c5 < best_c5:
        best_h, best_c5 = periodic_h, periodic_c5

    # Phase 2: binary hill climbing from best rounded solution
    now = time.time()
    remaining = budget_s - (now - start_time)
    if remaining > 5.0:
        bin_start = (best_h >= 0.5).astype(int)
        cur_ones = bin_start.sum()
        target_ones = n // 2
        if cur_ones != target_ones:
            if cur_ones > target_ones:
                ones_idx = np.flatnonzero(bin_start)
                to_flip = ones_idx[np.argsort(best_h[ones_idx])[:cur_ones - target_ones]]
                bin_start[to_flip] = 0
            else:
                zeros_idx = np.flatnonzero(1 - bin_start)
                to_flip = zeros_idx[np.argsort(-best_h[zeros_idx])[:target_ones - cur_ones]]
                bin_start[to_flip] = 1
        end_time = start_time + budget_s - 0.1
        bin_h, bin_c5 = binary_hill_climb(bin_start, end_time, rng)
        if bin_c5 < best_c5:
            best_h, best_c5 = bin_h, bin_c5

    # Phase 3: continuous annealing if time remains
    now = time.time()
    remaining = budget_s - (now - start_time)
    if remaining > 5.0:
        end_time = start_time + budget_s - 0.05
        ann_h, ann_c5 = continuous_anneal(best_h, end_time, rng)
        if ann_c5 < best_c5:
            best_h, best_c5 = ann_h, ann_c5

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_h(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
