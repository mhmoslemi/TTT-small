def sa_optimize(h0, time_budget, rng):

    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    start = time.time()
    iteration = 0
    # initial temperature based on expected scale of C5
    T0 = 0.01 # or 0.001? need acceptance for small uphill moves.
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget - 0.01:
            break
        # Determine move
        # Compute worst shift and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_vals = corr * dx
        k_idx = np.argmax(c_vals) - (n - 1)
        worst_k = k_idx
        grad = np.ones(n)
        if worst_k > 0:
            k = worst_k
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        elif worst_k < 0:
            k = -worst_k
            grad[k:] -= h[:n - k]
            grad[:n - k] -= h[k:]
        else:
            grad = 1 - 2 * h
        # Determine pairs
        pos_idx = np.where(grad > 1e-6)[0]
        neg_idx = np.where(grad < -1e-6)[0]
        # With probability 0.7 use directed move if both sets non empty
        if pos_idx.size > 0 and neg_idx.size > 0 and rng.random() < 0.7:
            i = int(rng.choice(neg_idx))
            j = int(rng.choice(pos_idx))
            max_inc = min(1.0 - h[i], h[j])
            if max_inc <= 0:
                continue
            # delta up to perhaps 0.2 * max_inc
            delta = max_inc * rng.random() * 0.3
        else:
            i, j = rng.choice(n, 2, replace=False)
            # choose direction
            if rng.random() < 0.5:
                max_inc = min(1.0 - h[i], h[j])
                if max_inc <= 0:
                    continue
                delta = max_inc * rng.random() * 0.2
            else:
                max_inc = min(1.0 - h[j], h[i])
                if max_inc <= 0:
                    continue
                delta = max_inc * rng.random() * 0.2
                i, j = j, i  # swap so i receives, j loses

        # apply delta
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # ensure bounds (clamp if needed)
        if h_new[i] > 1.0 or h_new[j] < 0.0:
            # numeric issues - skip
            continue
        new_c5 = compute_c5(h_new)
        # acceptance schedule
        # temperature linearly decays
        T = T0 * max(0.0, 1.0 - elapsed / time_budget)
        if new_c5 < current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / max(T, 1e-12)):
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        iteration += 1

    return best_h, best_c5

def project_onto_feasible(h, M):
    # Ensure within bounds and sum exactly M via iterative correction.
    # Clip to [0,1]
    h = np.clip(h,0.0,1.0)
    cur = np.sum(h)
    # Since after clipping sum may differ, we adjust by adding uniformly.
    # We'll clamp adjustments to keep within bounds again.
    # Use simple iteration up to few times.
    for _ in range(5):
        diff = M - cur
        if abs(diff) < 1e-12:
            break
        # distribute diff equally
        delta = diff / len(h)
        h += delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
    # In extreme case, if still not match due to bound saturation, we could adjust by adding from indices below 1, etc.
    # Implement iterative fill: while diff> tolerance, add to smallest components not at 1.

def sa_optimize(h0, n, time_budget, rng):
    M = n //2
    h = proj_box_simplex(h0, M)  # ensure feasible
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    start = time.time()
    # scaling for delta
    max_delta_factor = 0.3
    # temperature parameters
    T0 = 0.005  # low but we can start higher, maybe 0.02
    while True:
        elapsed = time.time() - start
        if elapsed > time_budget - 0.01:
            break
        # compute corr and overlaps
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * (2.0 / n)
        k_idx = np.argmax(overlaps) - (n - 1)
        worst_k = k_idx
        # gradient for directed move
        grad = np.ones(n)
        if worst_k > 0:
            k = worst_k
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        elif worst_k < 0:
            k = -worst_k
            grad[k:] -= h[:n - k]
            grad[:n - k] -= h[k:]
        else:
            grad = 1 - 2 * h
        # find pos and neg indices
        pos_idx = np.where(grad > 1e-6)[0]
        neg_idx = np.where(grad < -1e-6)[0]
        # choose move
        if pos_idx.size > 0 and neg_idx.size > 0 and rng.random() < 0.7:
            i = int(rng.choice(neg_idx))
            j = int(rng.choice(pos_idx))
            max_inc = min(1.0 - h[i], h[j])
            if max_inc <= 0:
                continue
            delta = max_inc * rng.random() * max_delta_factor
        else:
            i, j = rng.choice(n, 2, replace=False)
            # decide direction random
            if rng.random() < 0.5:
                max_inc = min(1.0 - h[i], h[j])
                if max_inc <= 0:
                    continue
                delta = max_inc * rng.random() * max_delta_factor
            else:
                max_inc = min(1.0 - h[j], h[i])
                if max_inc <= 0:
                    continue
                delta = max_inc * rng.random() * max_delta_factor
                i, j = j, i
        # apply
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # compute new C5
        new_c5 = compute_c5(h_new)
        # acceptance
        # temperature schedule
        T = T0 * max(0.0, 1.0 - elapsed / time_budget)
        if new_c5 < current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / max(T, 1e-12)):
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                # early exit if good enough
                if best_c5 < 0.3808:
                    # break entire loop
                    break
    return best_h, best_c5

def polish_slsqp(h, n):
    # using constraints same as previous
    M = n//2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k,n) for k in range(-n+1,n) if k!=0]
    constraints = [{'type':'eq','fun':sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]
    bounds = [(0.0,1.0)] * n + [(0.0,None)]
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    try:
        res = minimize(fun=lambda x: x[-1],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter':300, 'ftol':1e-9, 'disp':False})
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            # enforce sum
            cur = np.sum(h_opt)
            if abs(cur - M) > 1e-9:
                h_opt += (M - cur) / n
                h_opt = np.clip(h_opt,0.0,1.0)
            c5_opt = compute_c5(h_opt)
            return h_opt, c5_opt
    except Exception:
        pass
    return h, compute_c5(h)

def proj_box_simplex(v, M, max_iter=50, tol=1e-9):
    # v: array; want h = min(1, max(0, v - theta)) s.t. sum = M.
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        theta = (lo + hi) / 2.0
        h = np.clip(v - theta, 0.0, 1.0)
        s = h.sum()
        if abs(s - M) <= tol:
            return h
        if s > M:
            lo = theta
        else:
            hi = theta
    # final
    theta = (lo + hi) / 2.0
    return np.clip(v - theta, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def proj_box_simplex(v, M, max_iter=60, tol=1e-9):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        th = (lo + hi) * 0.5
        h = np.clip(v - th, 0.0, 1.0)
        s = h.sum()
        if abs(s - M) <= tol:
            return h
        if s > M:
            lo = th
        else:
            hi = th
    th = (lo + hi) * 0.5
    return np.clip(v - th, 0.0, 1.0)

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n * 0.5

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k > 0:
            ov = np.dot(h[:self.n - k], 1.0 - h[k:])
        elif k < 0:
            kk = -k
            ov = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        else:
            return x[-1] - np.dot(h, 1.0 - h) * self.dx
        return x[-1] - ov * self.dx

def sa_optimize(h0, n, budget, rng):
    M = n // 2
    h = proj_box_simplex(h0, M)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    max_delta_factor = 0.35
    T0 = 0.015
    while True:
        if time.time() - start > budget - 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        ov = corr * (2.0 / n)
        k_idx = np.argmax(ov) - (n - 1)
        worst_k = k_idx
        grad = np.ones(n)
        if worst_k > 0:
            k = worst_k
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        elif worst_k < 0:
            k = -worst_k
            grad[k:] -= h[:n - k]
            grad[:n - k] -= h[k:]
        else:
            grad = 1.0 - 2.0 * h
        pos = np.where(grad > 1e-6)[0]
        neg = np.where(grad < -1e-6)[0]
        if pos.size and neg.size and rng.random() < 0.7:
            i = int(rng.choice(neg))
            j = int(rng.choice(pos))
            mx = min(1.0 - h[i], h[j])
            if mx <= 0:
                continue
            delta = mx * rng.random() * max_delta_factor
        else:
            i, j = rng.choice(n, 2, replace=False)
            if rng.random() < 0.5:
                mx = min(1.0 - h[i], h[j])
                if mx <= 0:
                    continue
                delta = mx * rng.random() * max_delta_factor
            else:
                mx = min(1.0 - h[j], h[i])
                if mx <= 0:
                    continue
                delta = mx * rng.random() * max_delta_factor
                i, j = j, i
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / budget)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                if best_c5 < 0.3808:
                    break
    return best_h, best_c5

def polish_slsqp(h, n, rng):
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    cons = [{'type': 'eq', 'fun': sum_con}]
    cons += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    try:
        res = minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            cur = h_opt.sum()
            if abs(cur - M) > 1e-9:
                h_opt += (M - cur) / n
                h_opt = np.clip(h_opt, 0.0, 1.0)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate sizes – larger first
    cand = [200, 176, 160, 144, 128, 112, 96, 80]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in cand:
                cand.insert(0, n0)
            else:
                cand.remove(n0)
                cand = [n0] + cand

    for n in cand:
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        alloc = min(250.0, remaining * 0.5)

        M = n // 2
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
            h0 = proj_box_simplex(h0, M)
        else:
            h0 = np.full(n, 0.5)

        h_opt, c5_opt = sa_optimize(h0, n, alloc, rng)
        if c5_opt < best_c5:
            best_c5, best_h, best_n = c5_opt, h_opt, n
        if best_c5 < 0.3808:
            break

    if best_h is not None and time.time() - start < budget_s - 2:
        # final local refinement
        best_h, best_c5 = polish_slsqp(best_h, best_n, rng)

    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    else:
        best_h = proj_box_simplex(best_h, best_n // 2)

    return best_h, best_c5, best_n
