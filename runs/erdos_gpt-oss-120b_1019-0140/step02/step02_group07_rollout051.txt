import time

import numpy as np

def compute_c5(arr):
    """Approximate C5 via correlation on a discretised step function."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_box_and_sum(v, target_sum, tol=1e-12, max_iter=60):
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane sum=v_target.
    Uses bisection on the Lagrange multiplier for the box‑simplex projection.
    """
    v = np.clip(v, 0.0, 1.0)
    # if already satisfies sum, return
    s = v.sum()
    if abs(s - target_sum) <= tol:
        return v
    lo, hi = -1.0, 1.0  # bounds for lambda (sufficient for [0,1] box)
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s_proj = proj.sum()
        if s_proj > target_sum:
            lo = lam
        else:
            hi = lam
        if abs(s_proj - target_sum) <= tol:
            break
    return proj
