def binary_anneal(start_bin, deadline, rng):
    n = len(start_bin)
    target_sum = n // 2
    h = start_bin.copy()
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    # schedule: temperature start T0, final Tf
    T0 = 0.01
    Tf = 1e-6
    max_iters = int(1e6)  # large
    it = 0
    while time.time() < deadline and it < max_iters:
        # propose swap (i zero -> 1, j one -> 0)
        zero_idxs = np.flatnonzero(h == 0)
        one_idxs = np.flatnonzero(h == 1)
        if zero_idxs.size == 0 or one_idxs.size == 0:
            break
        i = rng.choice(zero_idxs)
        j = rng.choice(one_idxs)
        # swap
        cand = h.copy()
        cand[i] = 1
        cand[j] = 0
        cand_c5 = compute_c5(cand.astype(float))
        dE = cand_c5 - cur_c5
        # compute temperature based on iteration fraction
        T = T0 * (1 - it / max_iters) + Tf * (it / max_iters)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
        it += 1
    return best_h, best_c5

import time, math, numpy as np, cvxpy as cp

def compute_c5(arr):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    n = arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Binary Thue‑Morse sequence of length n (0/1)."""
    seq = np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=int, count=n)
    return seq

def binary_hill_climb(start_bin, deadline, rng):
    """Deterministic hill climbing on binary vectors (swap moves)."""
    n = len(start_bin)
    h = start_bin.copy()
    best_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    no_improve = 0
    while time.time() < deadline and no_improve < 3000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5

def binary_anneal(start_bin, deadline, rng):
    """Simulated annealing on binary vectors preserving the exact half‑sum."""
    n = len(start_bin)
    target_sum = n // 2
    h = start_bin.copy()
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    T0, Tf = 1e-2, 1e-7
    it = 0
    max_iter = 2000000
    while time.time() < deadline and it < max_iter:
        zeros = np.flatnonzero(h == 0)
        ones = np.flatnonzero(h == 1)
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        cand = h.copy()
        cand[i] = 1
        cand[j] = 0
        cand_c5 = compute_c5(cand.astype(float))
        dE = cand_c5 - cur_c5
        T = T0 * (1 - it / max_iter) + Tf * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
                if best_c5 <= 0.3808:
                    break
        it += 1
    return best_h, best_c5

def continuous_anneal(start_h, deadline, rng):
    """Simulated annealing on continuous variables preserving sum."""
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    T0, Tf = 5e-3, 1e-7
    max_iter = 400000
    it = 0
    while time.time() < deadline and it < max_iter:
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = T0 * (1 - it / max_iter) + Tf * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # final exact sum correction
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1.0 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_c5

def milp_optimize_fixed_n(n, time_limit, rng):
    """Exact MILP for given n (binary h) using CBC. Returns (h_vec, c5) or (None,None)."""
    dx = 2.0 / n
    target = n // 2
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    t = cp.Variable()  # upper bound on S_k = sum h_i - sum w_ij over each shift

    cons = [cp.sum(h) == target, t >= 0]

    # product linking
    h_col = cp.reshape(h, (n, 1))
    h_row = cp.reshape(h, (1, n))
    cons.append(w <= h_col)
    cons.append(w <= h_row)
    cons.append(w >= h_col + h_row - 1)

    # shift constraints
    for k in range(-n + 1, n):
        idx = [(i, i + k) for i in range(n) if 0 <= i + k < n]
        if not idx:
            continue
        sum_h = cp.sum([h[i] for (i, j) in idx])
        sum_w = cp.sum([w[i, j] for (i, j) in idx])
        cons.append(t >= sum_h - sum_w)

    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.CBC, cbc={'seconds': max(time_limit, 1.0)}, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    h_opt = np.rint(np.array(h.value)).astype(int)
    # ensure feasibility (rare rounding issues)
    if h_opt.sum() != target:
        # simple repair by flipping bits
        diff = target - h_opt.sum()
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for idx in idxs:
            if diff == 0:
                break
            if diff > 0 and h_opt[idx] == 0:
                h_opt[idx] = 1
                diff -= 1
            elif diff < 0 and h_opt[idx] == 1:
                h_opt[idx] = 0
                diff += 1
    c5 = compute_c5(h_opt.astype(float))
    return h_opt.astype(float), c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
