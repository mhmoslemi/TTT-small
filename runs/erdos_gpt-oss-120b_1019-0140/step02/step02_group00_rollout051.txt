def simulated_annealing_binary(h0, budget_s, start_time):
    n = len(h0)
    cur_h = h0.copy()
    cur_score = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        # temperature schedule
        fraction = elapsed / budget_s
        T = max(1e-6, 0.02 * (1 - fraction))  # start 0.02, end near 0
        # propose swap
        i = rng.integers(n)
        # select j such that opposite value
        # find opposite
        if cur_h[i] == 1.0:
            # need j where 0
            # quick random j until find
            for _ in range(10):
                j = rng.integers(n)
                if cur_h[j] == 0.0:
                    break
            else:
                # if fail, pick from indices of zeros
                zeros = np.where(cur_h == 0)[0]
                if zeros.size == 0:
                    continue
                j = rng.choice(zeros)
        else:
            for _ in range(10):
                j = rng.integers(n)
                if cur_h[j] == 1.0:
                    break
            else:
                ones = np.where(cur_h == 1)[0]

def run_sa(h_start, time_budget, start_time):
    # run until time_budget seconds exhausted
    cur_h = h_start.copy()
    cur_score = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = len(cur_h)
    while time.time() - start_time < time_budget:
        # propose swap
        i = rng.integers(n); j = rng.integers(n-1)
        if j >= i: j+=1
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5_fft(cur_h)
        delta = new_score - cur_score
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start_time
        fraction = elapsed / time_budget  # but this uses full budget
        T = max(1e-6, 0.02 * (1 - fraction))
        accept = delta < 0 or rng.random() < np.exp(-delta / T)
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_score

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        corr = compute_corr_fft(h)  # raw correlation
        max_corr = corr.max()
        # find shifts achieving max within tolerance
        eps = 1e-12
        lags = np.where(corr >= max_corr - eps)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, lags)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from old grid (length m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_fft(h):
    """Linear correlation of h with (1-h) using FFT (no scaling)."""
    n = h.shape[0]
    n2 = 2 * n
    a = np.fft.fft(h, n2)
    b = np.fft.fft(1.0 - h, n2)
    corr = np.fft.ifft(a * np.conj(b)).real
    return corr[:2 * n - 1]


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1 - h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return corr.max() * (2.0 / n)


def compute_gradients_for_shifts(h, lags):
    """Gradient of max‑correlation term for given integer lags."""
    n = h.size
    grad = np.zeros_like(h)
    for lag in lags:
        if lag >= 0:
            s = lag
            # contribution from first factor (1 - h[i+s])
            grad[:n - s] += 1.0 - h[s:]
            # contribution from second factor (-h[i])
            grad[s:] += -h[:n - s]
        else:
            s = -lag
            # lag negative: analogous contributions swapped
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] += -h[s:]
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        eps = 1e-12
        lags = np.where(corr >= max_corr - eps)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, lags)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def run_sa(h0, start_time, end_time, rng):
    """Simulated annealing on binary balanced vectors."""
    cur_h = h0.copy()
    cur_score = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.size
    T0 = 0.05
    Tmin = 1e-8

    while True:
        now = time.time()
        if now >= end_time:
            break
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5_fft(cur_h)
        delta = new_score - cur_score
        # temperature based on allocated interval
        frac = (now - start_time) / (end_time - start_time)
        T = max(Tmin, T0 * (1.0 - frac))
        accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    n = 512  # target resolution
    # ----- Build initial candidates -----
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init_full, n)
        # turn into balanced binary by thresholding top n/2 entries
        idx = np.argpartition(init_up, -n // 2)[-n // 2:]
        bin_vec = np.zeros(n, dtype=float)
        bin_vec[idx] = 1.0
        candidates.append(bin_vec)

    candidates.append(thue_morse(n))
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))

    # ----- Evaluate candidates -----
    best_h = None
    best_score = np.inf
    for cand in candidates:
        sc = compute_c5_fft(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()

    # ----- Main search loop -----
    while time_left() > 8.0:
        # allocate a chunk for simulated annealing
        chunk = min(30.0, time_left() - 5.0)
        end_chunk = time.time() + chunk
        h_sa, sc_sa = run_sa(best_h, start, end_chunk, rng)
        if sc_sa < best_score:
            best_score, best_h = sc_sa, h_sa.copy()

        # gradient refinement on whatever time remains
        remaining = time_left()
        if remaining > 5.0:
            # estimate iterations; roughly 800 iters per second on this size
            n_iters = int(min(20000, max(5000, remaining * 800)))
            h_gd, sc_gd = gradient_descent_momentum(best_h, n_iter=n_iters)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()

    # final safety projection and evaluation
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)

    return best_h, best_score, n
