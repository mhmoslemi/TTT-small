def project_onto_bounds_sum(v, target, max_iter=10):
    # start with clip
    h = np.clip(v, 0.0, 1.0)
    # define mask for free variables (those not at bounds)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine free indices where h not at bound (0 or 1)
        free = (h > 1e-12) & (h < 1 - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds, can't adjust further; break
            break
        # distribute diff equally among free variables
        delta = diff / n_free
        # adjust
        h[free] += delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
    # final sanity: ensure sum close to target, if not adjust using more thorough algorithm (e.g., sorting)
    # fallback: use algorithm for projection onto capped simplex (like sorting thresholds)
    # For simplicity, after few passes, adjust using sorted approach:
    if abs(h.sum() - target) > 1e-9:
        # Use projection onto simplex with upper bound 1
        # Solve by sorting: we find lambda s.t. sum_i clip(v_i - lambda, 0,1) = target
        # do binary search for lambda
        lo, hi = -1.0, 2.0  # lambda range plausible
        for _ in range(50):
            lam = (lo + hi) / 2
            proj = np.clip(v - lam, 0.0, 1.0)
            s = proj.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        h = np.clip(v - (lo + hi) / 2, 0.0, 1.0)
    return h

def project_onto_bounds_sum(v, target):
    # Clip to [0,1] first to have feasible domain for lambda search.
    # v may be outside; assumption initial v can be any.
    # Use lambda but we need to ensure after clip and shift sum matches target.
    # Solve using binary search on lambda.
    lo, hi = -1.0, 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def gradient_active_shift(h, k):
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    idx = np.arange(n)
    if k >= 0:
        mask1 = idx <= n - 1 - k
        # term1
        grad[mask1] += dx
        # term2
        grad[mask1] -= dx * h[idx[mask1] + k]
        # term3
        mask2 = idx >= k
        grad[mask2] -= dx * h[idx[mask2] - k]
    else:
        kp = -k
        mask1 = idx >= kp
        grad[mask1] += dx
        grad[mask1] -= dx * h[idx[mask1] - kp]
        mask2 = idx <= n - 1 - kp
        grad[mask2] -= dx * h[idx[mask2] + kp]
    return grad

def random_three_segment(n, target, rng):
    # Sample p in [0,n]
    p = rng.integers(0, n+1)
    # ensure p <= target (can't exceed sum)
    if p > target:
        p = int(target)
    # sample q such that q>0 and target - p >=0.
    remaining_sum = target - p
    if remaining_sum <= 0:
        # pure binary: just p ones, rest zeros
        q = 0
        f = 0.0
    else:
        # Choose q >= 1 up to n - p
        max_q = n - p
        # randomly choose q that yields f<=1
        # possible q range: remaining_sum <= q (since f <=1) -> q >= remaining_sum
        q_min = int(np.ceil(remaining_sum))  # smallest integer q satisfying f<=1
        q_max = n - p
        if q_min > q_max:
            # fallback to pure binary
            q = 0
            f = 0.0
        else:
            q = rng.integers(q_min, q_max+1)  # inclusive high bound
            f = remaining_sum / q
    # Now choose left_pad start position for the block+fraction
    total_block = p + q
    left_max = n - total_block
    left = rng.integers(0, left_max+1) if left_max > 0 else 0
    # Build h
    h = np.zeros(n, dtype=float)
    if p > 0:
        h[left:left+p] = 1.0
    if q > 0:
        h[left+p:left+p+q] = f
    return h

def project_onto_bounds_sum(v, target):
    # binary search for lambda such that sum(clip(v - lam,0,1)) = target
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_active_shift(h, k):
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    idx = np.arange(n)
    if k >= 0:
        mask1 = idx <= n - 1 - k
        if mask1.any():
            grad[mask1] += dx
            grad[mask1] -= dx * h[idx[mask1] + k]
        mask2 = idx >= k
        if mask2.any():
            grad[mask2] -= dx * h[idx[mask2] - k]
    else:
        kp = -k
        mask1 = idx >= kp
        if mask1.any():
            grad[mask1] += dx
            grad[mask1] -= dx * h[idx[mask1] - kp]
        mask2 = idx <= n - 1 - kp
        if mask2.any():
            grad[mask2] -= dx * h[idx[mask2] + kp]
    return grad

def gradient_descent(h_start, target, max_iters, deadline, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    n = h.shape[0]
    for it in range(max_iters):
        if time.time() > deadline - 0.1:
            break
        # compute active shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # gradient
        grad = gradient_active_shift(h, k)
        # step size: linearly decreasing from 1.0 to 0.0
        step = 1.0 * (1.0 - it / max_iters)
        if step < 1e-6:
            step = 0.0
        # propose new
        h_new = h - step * grad
        # project onto feasible set
        h_new = project_onto_bounds_sum(h_new, target)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_onto_bounds_sum(v, target):
    """Euclidean projection onto {0<=x<=1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def gradient_active_shift(h, k):
    """Gradient of the active shift term of C5."""
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    idx = np.arange(n)
    if k >= 0:
        mask1 = idx <= n - 1 - k
        if mask1.any():
            grad[mask1] += dx
            grad[mask1] -= dx * h[idx[mask1] + k]
        mask2 = idx >= k
        if mask2.any():
            grad[mask2] -= dx * h[idx[mask2] - k]
    else:
        kp = -k
        mask1 = idx >= kp
        if mask1.any():
            grad[mask1] += dx
            grad[mask1] -= dx * h[idx[mask1] - kp]
        mask2 = idx <= n - 1 - kp
        if mask2.any():
            grad[mask2] -= dx * h[idx[mask2] + kp]
    return grad


def random_binary_vector(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return project_onto_bounds_sum(v, target)


def random_three_segment(n, target, rng):
    """Generate a piecewise constant pattern: [1]*p + [f]*q + [0]*rest."""
    p = rng.integers(0, int(target) + 1)
    remaining = target - p
    if remaining <= 0:
        q, f = 0, 0.0
    else:
        max_q = n - p
        q_min = int(np.ceil(remaining))
        q_max = max_q
        if q_min > q_max:
            p = int(target)
            q, f = 0, 0.0
        else:
            q = rng.integers(q_min, q_max + 1)
            f = remaining / q
    total = p + q
    left = rng.integers(0, n - total + 1) if n - total > 0 else 0
    h = np.zeros(n, dtype=float)
    if p > 0:
        h[left:left + p] = 1.0
    if q > 0:
        h[left + p:left + p + q] = f
    return h


def gradient_descent(h_start, target, max_iters, deadline, rng):
    """Subgradient descent on the active shift with projection."""
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    n = h.shape[0]
    max_step = 5.0
    for it in range(max_iters):
        if time.time() > deadline - 0.1:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = np.argmax(corr) - (n - 1)
        grad = gradient_active_shift(h, k)
        step = max_step * (1.0 - it / max_iters)
        if step < 1e-8:
            step = 0.0
        h_new = h - step * grad
        h_new = project_onto_bounds_sum(h_new, target)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            if best_val < 0.3808:
                return best_h, best_val
        h = h_new
    return best_h, best_val


def continuous_anneal(h_start, best_val, start_time, deadline, rng):
    """Simple simulated annealing on continuous h (mass transfers)."""
    h = h_start.copy()
    cur_val = best_val
    best_h = h.copy()
    best = best_val
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        if best < 0.3808:
            break
    return best_h, best

    best_h = np.clip(best_h, 0.0, 1.0)
