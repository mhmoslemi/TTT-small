    n = len(h)

    a = h

def gradient_descent_smooth(h0, time_budget, rng, target):
    h = h0.copy()
    n = len(h)
    dx = 2.0/n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 2.0
    max_beta = 50.0
    alpha0 = 0.15
    no_improve = 0
    start = time.time()
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # increase beta more slowly
        if beta < max_beta:
            beta = min(max_beta, beta * 1.0015)
        # softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / v.size
        else:
            w /= w_sum
        # compute gradient
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
        # zero-mean to preserve sum
        g -= np.mean(g)
        # step size schedule decays linearly
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        # accept if improve or random small chance
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        # random shake when stuck
        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
    return best_h, best_c5

def project_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) <= 1e-12:
        return x

    while abs(diff) > 1e-12 and iteration < 10:

            # all at bounds, distribute diff to nearest bound
            # apply diff/len(x) naive
            x += diff / len(x)
        else:

            delta = diff / n_free
            # Apply but clip to bounds
            x[free] += delta
            x = np.clip(x, 0.0, 1.0)

        diff = target - cur

                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
    return x

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full')
    return np.max(v) * dx

def project_sum_and_bounds(x, target, tol=1e-12, max_iter=20):
    """Clip x to [0,1] and adjust sum(x) to target (exactly if possible)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) <= tol:
        return x
    for _ in range(max_iter):
        # indices that can move without hitting bounds
        free = (x > 0.0) & (x < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds: distribute uniformly (may violate bounds)
            delta = diff / len(x)
            x += delta
            x = np.clip(x, 0.0, 1.0)
        else:
            delta = diff / n_free
            x[free] += delta
            x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) <= tol:
            break
    # final tiny correction if needed
    if abs(diff) > tol:
        for i in range(len(x)):
            if abs(diff) <= tol:
                break
            if diff > 0 and x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
    return x

def gradient_descent_smooth(h0, time_budget, rng, target):
    """Smooth max‑overlap gradient descent."""
    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 5.0               # start smoothing
    max_beta = 80.0
    alpha0 = 0.3             # initial step size
    no_improve = 0
    start = time.time()
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)

        # soft‑max weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(v, 1.0 / v.size)
        else:
            w /= w_sum

        # gradient of weighted sum of overlaps
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]

        # remove zero‑sum component
        g -= np.mean(g)

        # decaying step size
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)

        # accept if better or tiny random acceptance
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        # random shake when stuck
        if no_improve > 2500:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
    return best_h, best_c5

def simulated_annealing(h0, time_budget, rng, target):
    """Simple SA on the max‑overlap objective."""
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n = h.shape[0]
    start = time.time()
    T0, T_end = 0.02, 1e-5
    sigma0, sigma_end = 0.1, 0.001
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + T_end * frac
        sigma = sigma0 * (1 - frac) + sigma_end * frac
        cand = h + rng.normal(scale=sigma, size=n)
        cand = project_sum_and_bounds(cand, target)
        cand_c5 = compute_c5(cand)
        if cand_c5 < cur_c5 or rng.random() < np.exp(-(cand_c5 - cur_c5) / (T + 1e-12)):
            h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best_h = h.copy()
                best_c5 = cur_c5
        # occasional restart
        if rng.random() < 0.005:
            h = project_sum_and_bounds(rng.normal(scale=0.2, size=n), target)
            cur_c5 = compute_c5(h)
    return best_h, best_c5

def improve_h(h_start, time_budget, rng, target):
    """Hybrid: SA → GD."""
    start = time.time()
    t_sa = time_budget * 0.6
    h_sa, c_sa = simulated_annealing(h_start, t_sa, rng, target)
    remaining = time_budget - (time.time() - start)
    if remaining > 0.5:
        h_gd, c_gd = gradient_descent_smooth(h_sa, remaining, rng, target)
        if c_gd < c_sa:
            return h_gd, c_gd
    return h_sa, c_sa

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():

    best_c5 = compute_c5(best_h)

    # Main iterative improvement loop
    while True:
        remaining = budget_s - (time.time() - start)
        if remaining < 5.0:
            break

        chunk = min(30.0, remaining * 0.8)

        if rng.random() < 0.2:

        else:

        start_h = project_sum_and_bounds(start_h, target)

        if cur_c5 < best_c5:

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
