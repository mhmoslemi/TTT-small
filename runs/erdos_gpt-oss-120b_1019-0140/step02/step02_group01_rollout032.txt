def repair_h(h, M):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) < 1e-12:
        return h
    diff = M - cur
    # Distribute diff to entries that are not at bounds
    # If diff > 0 (need to increase sum), add diff proportionally to entries <1
    if diff > 0:
        # compute capacities
        caps = 1.0 - h
        total_cap = np.sum(caps)
        if total_cap > 0:
            # distribute proportionally
            inc = diff * caps / total_cap
            h += inc
        else:
            # all at 1; can't increase, just leave
            pass
    else: # diff < 0 need to decrease sum
        caps = h
        total_cap = np.sum(caps)
        if total_cap > 0:
            dec = -diff * caps / total_cap
            h -= dec
        else:
            pass
    # final clip and adjust tiny residual
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    # If still off due to rounding, adjust a random element within bounds
    if abs(cur - M) > 1e-9:
        # find index with room
        for _ in range(10):
            idx = np.random.randint(len(h))
            room = M - cur
            if room > 0:
                inc = min(room, 1.0 - h[idx])
                h[idx] += inc
                cur += inc
                if cur >= M - 1e-12:
                    break
            else:
                dec = min(-room, h[idx])
                h[idx] -= dec
                cur -= dec
                if cur <= M + 1e-12:

def compute_S(h):
    n = h.shape[0]
    S = np.empty(n, dtype=float)
    for k in range(n):
        S[k] = np.dot(h[:n - k], h[k:])
    return S

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, M, rng):

    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) < 1e-12:
        return h
    # Simple iterative redistribution
    for _ in range(1000):

        if diff > 0:

import time
import numpy as np

def compute_S(h):
    """Autocorrelation S[k] = sum_i h[i] * h[i+k] for k>=0."""
    n = h.shape[0]
    S = np.empty(n, dtype=float)
    for k in range(n):
        S[k] = np.dot(h[: n - k], h[k:])
    return S

def compute_c5(h):
    """Exact C5 from definition: max overlap * dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, M, rng):
    """Clip to [0,1] and adjust sum to exactly M."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) < 1e-12:
        return h
    # iterative redistribution
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idxs = np.where(h < 1.0 - 1e-12)[0]
            if idxs.size == 0:
                break
            idx = rng.choice(idxs)
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
        else:
            idxs = np.where(h > 1e-12)[0]
            if idxs.size == 0:
                break
            idx = rng.choice(idxs)
            dec = min(-diff, h[idx])
            h[idx] -= dec
            diff += dec
    return h

def random_binary_h(n, rng):
    """Binary feasible vector with exactly n/2 ones."""
    M = n // 2
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, M, replace=False)] = 1.0
    return h

def upscale_h(h_small, n_new, rng):
    """Linear interpolation to a larger resolution, then repair."""
    n_old = h_small.shape[0]
    xs_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(xs_new, xs_old, h_small)
    return repair_h(h_interp, n_new / 2.0, rng)

def local_search(h_init, M, time_limit, rng):
    """Simulated‑annealing style pairwise mass transfer."""
    n = h_init.shape[0]
    h = h_init.copy()
    S = compute_S(h)
    cur_min_S = S.min()
    best_h = h.copy()
    best_min_S = cur_min_S
    ks = np.arange(n, dtype=int)
    dS = np.empty(n, dtype=float)
    start = time.time()
    init_T = 0.5
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        progress = elapsed / time_limit
        T = init_T * (1.0 - progress)
        i, j = rng.choice(n, 2, replace=False)
        if rng.random() < 0.5:
            target, source = i, j
        else:
            target, source = j, i
        max_transfer = min(1.0 - h[target], h[source])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        dS.fill(0.0)
        t_plus = target + ks
        mask = t_plus < n
        dS[mask] += delta * h[t_plus[mask]]
        t_minus = target - ks
        mask = t_minus >= 0
        dS[mask] += delta * h[t_minus[mask]]
        s_plus = source + ks
        mask = s_plus < n
        dS[mask] -= delta * h[s_plus[mask]]
        s_minus = source - ks
        mask = s_minus >= 0
        dS[mask] -= delta * h[s_minus[mask]]
        dS[0] += 2.0 * delta * delta
        d = abs(source - target)
        if d > 0:
            dS[d] -= delta * delta
        S_new = S + dS
        new_min_S = S_new.min()
        accept = False
        if new_min_S > cur_min_S + 1e-12:
            accept = True
        elif T > 1e-12:
            prob = np.exp((new_min_S - cur_min_S) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            h[target] += delta
            h[source] -= delta
            S = S_new
            cur_min_S = new_min_S
            if cur_min_S > best_min_S + 1e-12:
                best_min_S = cur_min_S
                best_h = h.copy()
    c5 = (M - best_min_S) * (2.0 / n)
    return best_h, c5

    # candidate even discretisations (larger sizes give finer bounds)

    best_n_global = None

    for idx, n in enumerate(candidate_ns):

        remaining = budget_s - (time.time() - start)
        remaining_ns = len(candidate_ns) - idx
        time_alloc = min(remaining * 0.5 / remaining_ns, 300.0)  # cap per resolution
        M = n / 2.0

        if "initial_h_values" in globals() and n == len(initial_h_values):

            cur = np.sum(h0)
            if abs(cur - M) > 1e-12:
                h0 = repair_h(h0, M, rng)
        elif best_h_global is not None and n > best_n_global:
            h0 = upscale_h(best_h_global, n, rng)
        else:

        # stochastic local improvement
        h_opt, c5_opt = local_search(h0, M, time_alloc, rng)
