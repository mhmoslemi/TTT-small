def repair_sum(v, target):
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # Add diff equally, clip
    v = v + diff / v.size
    v = np.clip(v, 0, 1)
    # iterate until diff small (max couple loops)
    for _ in range(10):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        # distribute diff proportionally to free capacity
        if diff > 0:
            # capacity to increase
            cap = 1 - v
            total_cap = cap.sum()
            if total_cap <= 0:
                break
            inc = diff * cap / total_cap
            v = v + inc
        else:
            # capacity to decrease
            cap = v
            total_cap = cap.sum()
            if total_cap <= 0:
                break
            dec = diff * cap / total_cap
            v = v + dec   # diff negative, dec negative
        v = np.clip(v, 0, 1)
    # Final tiny adjustment by uniform distribution just in case
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        v = v + diff / v.size
        v = np.clip(v, 0, 1)
    return v

def genetic_algorithm(n, target_sum, rng, budget_end):
    pop_size = 50
    elite_frac = 0.1
    n_elite = max(1, int(pop_size * elite_frac))
    mutation_rate = 0.2
    sigma = 0.05
    tournament_k = 3
    # initialize population
    population = []
    # include initial_h if exists
    if 'initial_h_values' in globals():
        v0 = np.array(initial_h_values, dtype=float)
        if len(v0) != n:
            # if mismatch, ignore
            v0 = random_h(n, target_sum, rng)
    else:
        v0 = random_h(n, target_sum, rng)
    population.append(v0)
    # add some seeds: Thue-Morse and Rudin-Shapiro if n small, but we can generate them using previous code.
    # But we don't have these functions defined; we can in this file define small versions using numpy. There's limit on data length; but n=80, fine.
    # Let's implement simple tm_seq and rs_seq functions.
    # We'll add them later.

    # Fill the rest with random individuals
    while len(population) < pop_size:
        population.append(random_h(n, target_sum, rng))

    # compute fitness array
    fitness = np.empty(pop_size)
    for i, ind in enumerate(population):
        fitness[i] = compute_c5(ind)

    best_idx = np.argmin(fitness)
    best_ind = population[best_idx].copy()
    best_fitness = fitness[best_idx]

    # main loop
    while time.time() < budget_end:
        # elitism
        elite_indices = np.argsort(fitness)[:n_elite]
        new_population = [population[i].copy() for i in elite_indices]

        # generate offspring
        while len(new_population) < pop_size:
            # select parents
            p1_idx = tournament_select(fitness, tournament_k, rng)
            p2_idx = tournament_select(fitness, tournament_k, rng)
            p1 = population[p1_idx]
            p2 = population[p2_idx]
            # crossover
            alpha = rng.random()
            child = alpha * p1 + (1 - alpha) * p2
            # repair sum after crossover
            child = repair_sum(child, target_sum)
            # mutation
            if rng.random() < 0.5:  # apply mutation with probability 0.5 (adjust)
                # per gene mutation
                mutation_mask = rng.random(n) < mutation_rate
                # add gaussian noise
                child[mutation_mask] += sigma * rng.standard_normal(np.count_nonzero(mutation_mask))
                # clip
                child = np.clip(child, 0, 1)
                # repair sum again
                child = repair_sum(child, target_sum)
            new_population.append(child)

        population = new_population
        # evaluate fitness
        fitness = np.empty(pop_size)
        for i, ind in enumerate(population):
            fitness[i] = compute_c5(ind)

        # update best
        cur_best_idx = np.argmin(fitness)
        if fitness[cur_best_idx] < best_fitness:
            best_fitness = fitness[cur_best_idx]
            best_ind = population[cur_best_idx].copy()
            # early exit if target reached
            if best_fitness <= 0.3808:
                break
    return best_ind, best_fitness

def continuous_anneal(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n/2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start_t = time.time()
    # iteration limit maybe high
    while True:
        now = time.time()
        if now >= time_end:
            break
        # compute temperature as linear decay based on elapsed fraction
        frac = (now - start_t) / (time_end - start_t)
        T = init_temp * (1 - frac) + final_temp * frac

        # choose two indices i where h[i] not at bounds to transfer mass
        # find indices with h[i] > 0 and h[j] < 1
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            continue
        # propose delta ~ max_delta * random * (some factor depending on T)
        # Use factor: delta = max_delta * rng.random() * 0.5
        delta = max_delta * rng.random() * 0.5
        # optionally scale delta with temperature: multiply by factor = T/ (init_temp + 1e-12)?
        # Might be okay.

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        # accept if better or random
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        # else reject
    # final repair to enforce exact sum
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0, 1)
        # additional repair if needed
        best_h = repair_sum(best_h, target_sum)
    return best_h, best_c5

def binary_hill_climb(start_bin, time_end, rng):
    n = len(start_bin)
    h_bin = start_bin.copy()
    best_c5 = compute_c5(h_bin.astype(float))
    best_h = h_bin.copy()
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(h_bin)
        zeros = np.flatnonzero(1 - h_bin)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h_bin.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 < best_c5 - 1e-12:
            best_c5 = cand_c5
            best_h = cand.copy()
            h_bin = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5

def rudin_shapiro_seq(n):
    # construct recursively
    a = np.array([1], dtype=int)
    while len(a) < n:
        # generate next
        b = np.empty(len(a) * 2, dtype=int)
        b[0::2] = a
        signs = np.where(np.arange(len(a)) % 2 == 0, 1, -1)
        b[1::2] = a * signs
        a = b
    a = a[:n]
    # map to 0/1
    return (a + 1) // 2

import time
import math
import numpy as np

def compute_c5(arr):
    """Approximate C5 via discrete correlation."""
    n = arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_sum(v, target, max_iter=10):
    """Project v onto the simplex sum(v)=target with box constraints [0,1]."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - v
            total_cap = cap.sum()
            if total_cap == 0:
                break
            inc = diff * cap / total_cap
            v += inc
        else:
            cap = v
            total_cap = cap.sum()
            if total_cap == 0:
                break
            dec = diff * cap / total_cap
            v += dec    # diff is negative, dec is negative
        v = np.clip(v, 0.0, 1.0)
    # final uniform tweak if needed
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    return v

def random_h(n, target, rng):
    """Generate a random feasible vector."""
    v = rng.random(n)
    v = v / v.sum() * target
    v = repair_sum(v, target)
    return v

def thue_morse_seq(n):
    """Binary Thue‑Morse sequence of length n."""
    seq = np.empty(n, dtype=int)
    for i in range(n):
        seq[i] = bin(i).count('1') & 1
    return seq.astype(float)

def rudin_shapiro_seq(n):
    """Binary Rudin‑Shapiro sequence (+1/-1 → 1/0)."""
    a = np.array([1], dtype=int)
    while a.size < n:
        b = np.empty(a.size * 2, dtype=int)
        b[0::2] = a
        signs = np.where(np.arange(a.size) % 2 == 0, 1, -1)
        b[1::2] = a * signs
        a = b
    a = a[:n]
    return ((a + 1) // 2).astype(float)

def tournament_select(fitness, k, rng):
    """Return index of best individual among k random choices."""
    idxs = rng.integers(0, fitness.size, size=k)
    best = idxs[np.argmin(fitness[idxs])]
    return best

def genetic_algorithm(n, target, rng, deadline, init_vec=None):
    pop_size = 50
    elite_frac = 0.12
    n_elite = max(1, int(pop_size * elite_frac))
    mut_rate = 0.22
    sigma = 0.06
    tour_k = 3

    pop = []
    # seed with provided vector if any
    if init_vec is not None:
        pop.append(init_vec.copy())
    # deterministic seeds
    tm = thue_morse_seq(n)
    diff = target - tm.sum()
    if diff != 0:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for idx in idxs:
            if diff == 0:
                break
            if diff > 0 and tm[idx] == 0:
                tm[idx] = 1.0
                diff -= 1.0
            elif diff < 0 and tm[idx] == 1.0:
                tm[idx] = 0.0
                diff += 1.0
    pop.append(repair_sum(tm, target))

    rs = rudin_shapiro_seq(n)
    diff = target - rs.sum()
    if diff != 0:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for idx in idxs:
            if diff == 0:
                break
            if diff > 0 and rs[idx] == 0:
                rs[idx] = 1.0
                diff -= 1.0
            elif diff < 0 and rs[idx] == 1:
                rs[idx] = 0.0
                diff += 1.0
    pop.append(repair_sum(rs, target))

    # fill remainder randomly
    while len(pop) < pop_size:
        pop.append(random_h(n, target, rng))

    fitness = np.empty(pop_size)
    for i, ind in enumerate(pop):
        fitness[i] = compute_c5(ind)

    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    while time.time() < deadline:
        # elitism
        elite_ids = np.argsort(fitness)[:n_elite]
        new_pop = [pop[i].copy() for i in elite_ids]

        # generate offspring
        while len(new_pop) < pop_size:
            p1 = pop[tournament_select(fitness, tour_k, rng)]
            p2 = pop[tournament_select(fitness, tour_k, rng)]
            alpha = rng.random()
            child = alpha * p1 + (1.0 - alpha) * p2
            child = repair_sum(child, target)

            # mutation
            if rng.random() < 0.5:
                mask = rng.random(n) < mut_rate
                n_mut = mask.sum()
                if n_mut:
                    child[mask] += sigma * rng.standard_normal(n_mut)
                    child = np.clip(child, 0.0, 1.0)
                    child = repair_sum(child, target)

            new_pop.append(child)

        pop = new_pop
        for i, ind in enumerate(pop):
            fitness[i] = compute_c5(ind)

        cur_best = np.argmin(fitness)
        if fitness[cur_best] < best_f:
            best_f = fitness[cur_best]
            best_h = pop[cur_best].copy()
            if best_f <= 0.3808:
                break

    return best_h, best_f

def continuous_anneal(start_h, deadline, rng):
    n = start_h.shape[0]
    target = n / 2.0
    h = start_h.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur

    t0 = time.time()
    init_temp = 1e-2
    final_temp = 1e-7

    while time.time() < deadline:
        now = time.time()
        frac = (now - t0) / (deadline - t0)
        T = init_temp * (1.0 - frac) + final_temp * frac

        inc_idx = np.where(h < 1.0 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new = compute_c5(h_new)
        dE = new - cur
        if dE <= 0.0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur = new
            if new < best:
                best = new
                best_h = h_new.copy()

    best_h = repair_sum(best_h, target)
    return best_h, best

def binary_hill_climb(start_bin, deadline, rng):
    n = start_bin.shape[0]
    cur = compute_c5(start_bin.astype(float))
    best = cur
    best_h = start_bin.copy()
    no_improve = 0
    while time.time() < deadline and no_improve < 5000:
        ones = np.flatnonzero(start_bin)
        zeros = np.flatnonzero(1 - start_bin)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = start_bin.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best:
            best = cand_c5
            best_h = cand.copy()
            start_bin = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        init_vec = np.array(initial_h_values, dtype=float)
        init_vec = np.clip(init_vec, 0.0, 1.0)
        init_vec = repair_sum(init_vec, n / 2.0)
    else:
        n = 80
        init_vec = None

    target = n / 2.0

    # Time allocation
    total = budget_s
    ga_deadline = start_time + total * 0.55
    anneal_deadline = ga_deadline + total * 0.35
    hill_deadline = start_time + total - 0.2

    # Genetic algorithm phase
    best_h, best_c5 = genetic_algorithm(n, target, rng, ga_deadline, init_vec)

    # Simulated annealing refinement
    if time.time() < anneal_deadline:
        ha, ca = continuous_anneal(best_h, anneal_deadline, rng)
        if ca < best_c5:
            best_h, best_c5 = ha, ca

    # Binary hill climbing from rounded solution
    if time.time() < hill_deadline:
        bin_start = (best_h >= 0.5).astype(int)
        hb, cb = binary_hill_climb(bin_start, hill_deadline, rng)
        if cb < best_c5:
            best_h, best_c5 = hb, cb

    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_sum(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
