def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    def get_initial():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values, dtype=float)
            n = h.shape[0]
            # enforce sum if needed
            target = n/2.0
            cur = h.sum()
            if abs(cur-target) > 1e-12:
                h = np.clip(h + (target-cur)/n, 0.0, 1.0)
                # final fix if needed
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    idxs = np.arange(n)
                    rng.shuffle(idxs)
                    for i in idxs:
                        if abs(diff) < 1e-12:
                            break
                        if diff > 0 and h[i] < 1.0:
                            inc = min(diff, 1.0 - h[i])
                            h[i] += inc
                            diff -= inc
                        elif diff < 0 and h[i] > 0.0:
                            dec = min(-diff, h[i])
                            h[i] -= dec
                            diff += dec
            return h
        else:
            n = 80
            # start with random binary pattern with exact sum
            h = np.zeros(n, dtype=float)
            idx = rng.choice(n, n//2, replace=False)
            h[idx] = 1.0
            return h
    best_h = get_initial()
    n = best_h.shape[0]
    target = n/2.0
    dx = 2.0/n

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    best_val = compute_c5(best_h)

    # define projection function
    def project_feasible(h):
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            return h
        # get indices that can be increased/decreased
        if diff > 0:
            # need to increase some entries
            inc_idxs = np.where(h < 1.0)[0]
            if inc_idxs.size == 0:
                return h
            # distribute diff equally (with caps)
            for i in inc_idxs:
                if diff <= 0:
                    break
                cap = 1.0 - h[i]
                add = min(cap, diff/ (inc_idxs.size))
                h[i] += add
                diff -= add
            # if still diff > 0, distribute again
            i = 0
            while diff > 1e-12:
                idx = inc_idxs[i % inc_idxs.size]
                cap = 1.0 - h[idx]
                if cap > 1e-12:
                    add = min(cap, diff)
                    h[idx] += add
                    diff -= add
                i += 1
        else:
            diff = -diff
            dec_idxs = np.where(h > 0.0)[0]
            if dec_idxs.size == 0:
                return h
            for i in dec_idxs:
                if diff <= 0:
                    break
                cap = h[i]
                sub = min(cap, diff/ (dec_idxs.size))
                h[i] -= sub
                diff -= sub
            i = 0
            while diff > 1e-12:
                idx = dec_idxs[i % dec_idxs.size]
                cap = h[idx]
                if cap > 1e-12:
                    sub = min(cap, diff)
                    h[idx] -= sub
                    diff -= sub
                i += 1
        return np.clip(h, 0.0, 1.0)

    # Annealing function
    def anneal(h_start, cur_val, deadline, rng):
        h = h_start.copy()
        best_local_h = h.copy()
        best_local_val = cur_val
        n_local = h.shape[0]
        while time.time() < deadline - 0.1:
            # compute max overlap and shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = np.argmax(corr)
            max_val = corr[max_idx] * dx
            k = max_idx - (n_local - 1)
            # contributions:
            if k >= 0:
                i_range = np.arange(n_local - k)
                prod = h[i_range] * (1.0 - h[i_range + k])
            else:
                kk = -k
                i_range = np.arange(kk, n_local)
                prod = h[i_range] * (1.0 - h[i_range + k])  # k negative
            # pick i with max contribution
            i_local = np.argmax(prod)
            i = i_range[i_local]
            j = i + k  # counterpart
            # compute possible delta
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 1e-12:
                # fallback: random feasible src/dst
                src = np.where(h > 1e-12)[0]
                dst = np.where(h < 1.0 - 1e-12)[0]
                if src.size == 0 or dst.size == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_transfer = min(h[i], 1.0 - h[j])
                if max_transfer <= 0:
                    continue
            delta = rng.random() * max_transfer
            if delta <= 0:
                continue
            h[i] -= delta
            h[j] += delta
            # ensure feasibility (sum unchanged)
            # compute new val
            new_val = compute_c5(h)
            if new_val < cur_val:
                cur_val = new_val
                if new_val < best_local_val:
                    best_local_val = new_val
                    best_local_h = h.copy()
                if best_local_val < 0.3808:
                    break
            else:
                # accept w/ prob
                remaining = deadline - time.time()
                T = max(remaining * 0.001, 1e-12)
                if np.exp((cur_val - new_val) / T) > rng.random():
                    cur_val = new_val
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta
            # occasional random perturbation to escape
            if rng.random() < 0.001:
                # random transfer
                src = np.where(h > 1e-12)[0]
                dst = np.where(h < 1.0 - 1e-12)[0]
                if src.size and dst.size:
                    i = rng.choice(src)
                    j = rng.choice(dst)
                    max_transfer = min(h[i], 1.0 - h[j])
                    if max_transfer > 0:
                        delta = rng.random() * max_transfer
                        h[i] -= delta
                        h[j] += delta
        return best_local_h, best_local_val

    # Main loop with restarts
    while time.time() < deadline - 5.0:
        # start from current best (or random)
        if rng.random() < 0.5:
            h_start = best_h.copy()
            cur_val = best_val
        else:
            # new random start
            n = best_h.shape[0]
            h_start = np.zeros(n, dtype=float)
            idx = rng.choice(n, n//2, replace=False)
            h_start[idx] = 1.0
            cur_val = compute_c5(h_start)
        # anneal
        cur_deadline = min(deadline, time.time() + 30.0)  # limit each run to 30 seconds
        h_opt, val_opt = anneal(h_start, cur_val, cur_deadline, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break
        # early exit if time low
        if time.time() > deadline - 2.0:
            break

    # final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h)
    final_val = compute_c5(best_h)
    # ensure best_val is final
    if final_val < best_val:
        best_val = final_val
        best_h = best_h

    return best_h.astype(float), best_val, best_h.shape[0]

    n = h.shape[0]
    dx = 2.0/n
    # conv = np.correlate(h, h, mode='full') but we need overlap = target - conv_k
    conv = np.correlate(h, h, mode='full')

    k_idx = np.arange(-n+1, n)
    target = n/2.0
    O = (target - conv) * dx  # actual overlap values

    max_O = np.max(O)
    exp_beta = np.exp(beta * (O - max_O))
    w = exp_beta / np.sum(exp_beta)
    # compute gradient vector
    grad = np.zeros_like(h)
    # contributions of each shift to each index:
    # For shift s (positive), derivative of overlap w.r.t h[j] = -(h[j+s] + h[j-s]) (if within bounds)
    # So grad[j] = - sum_s w[s] * (h[j+s] + h[j-s])
    for s in range(-n+1, n):

        if w_s == 0:
            continue
        # shift s >=0
        if s >= 0:

            # add -w_s * h[j-s] for j where j-s >=0 -> j >= s

def gradient_descent(h, steps, beta, rng):
    for s in range(steps):
        # compute O and gradient
        n = h.shape[0]
        target = n/2.0
        dx = 2.0/n
        conv = np.correlate(h, h, mode='full')
        O = (target - conv) * dx
        max_O = np.max(O)
        exp_beta = np.exp(beta * (O - max_O))
        w = exp_beta / np.sum(exp_beta)
        grad = np.zeros_like(h)
        # loop over shift indices
        for kk in range(-n+1, n):
            wk = w[kk + n - 1]
            if wk == 0:
                continue
            if kk >= 0:
                grad[:n - kk] -= wk * h[kk:]
                grad[kk:] -= wk * h[:n - kk]
            else:
                k2 = -kk
                grad[:n - k2] -= wk * h[k2:]
                grad[k2:] -= wk * h[:n - k2]
        grad *= dx
        # step
        step_size = 0.1  # could decay
        h_new = h - step_size * grad
        # project
        h = project_feasible(h_new)
    return h

import time
import numpy as np

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) gives raw overlap sums
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_feasible(h, target):
    """Clip to [0,1] and adjust to achieve exact sum=target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # indices that can be increased / decreased
    if diff > 0:
        inc = np.where(h < 1.0)[0]
        if inc.size == 0:
            return h
        # distribute diff uniformly, respecting caps
        for i in inc:
            if diff <= 0:
                break
            cap = 1.0 - h[i]
            add = min(cap, diff / inc.size)
            h[i] += add
            diff -= add
        i = 0
        while diff > 1e-12:
            idx = inc[i % inc.size]
            cap = 1.0 - h[idx]
            if cap > 0:
                add = min(cap, diff)
                h[idx] += add
                diff -= add
            i += 1
    else:
        diff = -diff
        dec = np.where(h > 0.0)[0]
        if dec.size == 0:
            return h
        for i in dec:
            if diff <= 0:
                break
            cap = h[i]
            sub = min(cap, diff / dec.size)
            h[i] -= sub
            diff -= sub
        i = 0
        while diff > 1e-12:
            idx = dec[i % dec.size]
            cap = h[idx]
            if cap > 0:
                sub = min(cap, diff)
                h[idx] -= sub
                diff -= sub
            i += 1
    return np.clip(h, 0.0, 1.0)

def _anneal(h_start, cur_val, deadline, rng):
    """Targeted simulated annealing that moves mass from a heavy contributor to its counterpart."""
    h = h_start.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_val = cur_val
    it = 0
    while time.time() < deadline - 0.1 and it < 5000:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)          # shift corresponding to maximum overlap
        # contributions for this shift
        if k >= 0:
            i_range = np.arange(n - k)
            prod = h[i_range] * (1.0 - h[i_range + k])
        else:
            kk = -k
            i_range = np.arange(kk, n)
            prod = h[i_range] * (1.0 - h[i_range + k])   # k negative
        i_local = np.argmax(prod)
        i = i_range[i_local]
        j = i + k   # partner index (may equal i only when k==0)
        max_transfer = min(h[i], 1.0 - h[j]) if 0 <= j < n else 0.0
        if max_transfer <= 1e-12:
            # fallback: any feasible src/dst pair
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                continue
        delta = rng.random() * max_transfer
        if delta <= 0:
            continue
        h[i] -= delta
        h[j] += delta
        new_val = _compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
            if best_val < 0.3808:
                break
        else:
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        # occasional random jiggle to escape plateaus
        if rng.random() < 0.001:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size and dst.size:
                i = rng.choice(src)
                j = rng.choice(dst)
                max_transfer = min(h[i], 1.0 - h[j])
                if max_transfer > 0:
                    delta = rng.random() * max_transfer
                    h[i] -= delta
                    h[j] += delta
    return best_h, best_val

def _smooth_max_grad(h, beta):
    """Return (smooth_max_overlap, gradient) for the soft‑max approximation."""
    n = h.shape[0]
    target = n / 2.0
    dx = 2.0 / n
    conv = np.correlate(h, h, mode="full")                 # ∑ h_i h_{i+k}
    O = (target - conv) * dx                               # actual overlaps for each shift
    # soft‑max weights
    max_O = np.max(O)
    w = np.exp(beta * (O - max_O))
    w /= np.sum(w)
    grad = np.zeros_like(h)
    # accumulate -dx * Σ w_k (h_{j+k}+h_{j-k})
    for shift in range(-n + 1, n):
        wk = w[shift + n - 1]
        if wk == 0:
            continue
        if shift >= 0:
            grad[: n - shift] -= wk * h[shift:]
            grad[shift:]      -= wk * h[: n - shift]
        else:
            s = -shift
            grad[: n - s] -= wk * h[s:]
            grad[s:]      -= wk * h[: n - s]
    grad *= dx
    smooth_max = np.max(O)
    return smooth_max, grad

def _gradient_descent(h, steps, beta, rng):
    """Simple projected gradient descent on the smooth‑max objective."""
    n = h.shape[0]
    target = n / 2.0
    step_size = 0.2
    for _ in range(steps):
        _, grad = _smooth_max_grad(h, beta)
        h_new = h - step_size * grad
        h = _project_feasible(h_new, target)
        step_size *= 0.99                # mild decay
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, C5_bound, n_points)."""
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        cur = h0.sum()
        if abs(cur - target) > 1e-12:
            h0 = np.clip(h0 + (target - cur) / n, 0.0, 1.0)
            diff = target - h0.sum()
            if abs(diff) > 1e-12:
                idxs = np.arange(n)
                rng.shuffle(idxs)
                for i in idxs:
                    if abs(diff) < 1e-12:
                        break
                    if diff > 0 and h0[i] < 1.0:
                        inc = min(diff, 1.0 - h0[i])
                        h0[i] += inc
                        diff -= inc
                    elif diff < 0 and h0[i] > 0.0:
                        dec = min(-diff, h0[i])
                        h0[i] -= dec
                        diff += dec
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target = n / 2.0
    h0 = _project_feasible(h0, target)
    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    # ----- main optimisation loop -----
    while time.time() < deadline - 5.0:
        # choose start point: either current best or a fresh random binary vector
        if rng.random() < 0.4:
            h_start = best_h.copy()
        else:
            h_start = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h_start[idx] = 1.0
        cur_val = _compute_c5(h_start)

        # targeted annealing (time‑boxed)
        anneal_deadline = min(deadline, time.time() + 12.0)
        h_opt, val_opt = _anneal(h_start, cur_val, anneal_deadline, rng)

        # gradient‑descent refinement on the smooth max
        h_opt = _gradient_descent(h_opt, steps=60, beta=20.0, rng=rng)
        val_opt = _compute_c5(h_opt)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

        # early exit if time is low
        if time.time() > deadline - 4.0:
            break

    # final safety projection and evaluation
    best_h = _project_feasible(best_h, target)
    final_val = _compute_c5(best_h)
    if final_val < best_val:
        best_val = final_val

    return best_h.astype(float), best_val, n
