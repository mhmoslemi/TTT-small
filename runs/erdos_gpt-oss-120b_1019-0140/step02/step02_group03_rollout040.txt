Implementation details:

```
def random_delta_move(h, step, rng):
    n = h.size
    i, j = rng.integers(n, size=2)
    while i == j:
        j = rng.integers(n)
    # maximum delta we can transfer from i to j while staying within bounds
    max_minus = min(step, h[i]) # moving from i to j reduces i
    max_plus = min(step, 1 - h[j])
    delta_max = min(max_minus, max_plus)
    # also we can move opposite direction (from j to i)
    max_minus_rev = min(step, h[j])
    max_plus_rev = min(step, 1 - h[i])
    delta_max_rev = min(max_minus_rev, max_plus_rev)
    # Choose direction randomly
    if rng.random() < 0.5:
        # move amount delta within [0, delta_max]
        if delta_max <= 0:
            return None  # no feasible move
        delta = rng.random() * delta_max
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        if delta_max_rev <= 0:
            return None
        delta = rng.random() * delta_max_rev
        h_new = h.copy()
        h_new[j] -= delta
        h_new[i] += delta
    return h_new
```

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

            kk = -k

        else:
            overlap = 0.0

    return x[-1]

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""

    return np.max(corr) * dx

    """Project v onto [lo,hi]^n while fixing its sum to target via a scalar shift."""
    low = -np.max(v) - 5.0

        x = np.clip(v + mid, lo, hi)

        if s < target:

            h_start = best_h + rng.normal(scale=0.1, size=n)
        else:
            h_start = rng.random(n)
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = project_to_box_sum(h_start, n / 2.0)
        t_start = compute_c5(h_start)
        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=400)
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # enforce exact sum after optimisation
        if abs(h_opt.sum() - n / 2.0) > 1e-9:
            h_opt = project_to_box_sum(h_opt, n / 2.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            if best_c5 <= 0.3808:
                break

    # ---- Phase 2: Projected Gradient Descent refinement (≈35% of budget) ----
    phase2_deadline = start_time + budget_s * 0.85
    best_h, best_c5 = pgd_optimize(best_h, n / 2.0, phase2_deadline)

    # ---- Phase 3: Simple simulated‑annealing style perturbations (remainder) ----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    T = 0.02
    cooling = 0.9995
    while time.time() < start_time + budget_s - 0.2:
        cand = cur_h + rng.normal(scale=0.05) * (rng.random() ** 2)
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_to_box_sum(cand, n / 2.0)
        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                if best_c5 <= 0.3808:
                    break
        T = max(T * cooling, 1e-7)

    # ---- Final safety projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
