    max_val = np.max(corr)

            # term1: for indices where h[t+shift] appears
            # We need gradient of corr_s w.r.t h (for each shift).

import time
import numpy as np
import scipy.optimize as opt

# ---------- Helper functions ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradient_max_corr(h):
    """Gradient of the max correlation term (subgradient)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= max_val - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[: n - s] += 1.0 - h[s:]
            grad[s:] -= h[: n - s]
        else:
            ss = -s
            grad[ss:] += 1.0 - h[: n - ss]
            grad[: n - ss] -= h[ss:]
    grad /= len(shifts)
    return grad

def gradient_descent_momentum_refine(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        grad = compute_gradient_max_corr(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=200):
    """Short SLSQP refinement for a given candidate (exactly enforces constraints)."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

# ---------- Main routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    n = 512                     # target resolution
    dx = 2.0 / n

    # ---- Initialization ----
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        h = upsample(init_full, n)
    else:
        h = random_binary_balanced(n, rng).astype(float)
    h = project_onto_box_sum(h, n / 2.0)
    a = 1.0 - h
    corr = np.correlate(h, a, mode='full')
    current_score = np.max(corr) * dx
    best_h = h.copy()
    best_score = current_score

    # ---- Simulated‑annealing hill climbing ----
    it = 0
    max_iters = 10_000_000
    recompute_every = 5000
    T0 = 5e-4
    Tmin = 5e-7

    while time_left() > 0.5 and it < max_iters:
        it += 1
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            continue
        # larger moves early, smaller later
        frac = rng.random() * 0.5 + 0.5 * (time_left() / budget_s)
        delta = max_delta * frac

        # keep copies for a possible revert
        h_old = h.copy()
        a_old = a.copy()
        corr_old = corr.copy()
        score_old = current_score

        # pre‑compute reverse of old h for linear terms
        h_rev = h_old[::-1]

        # update h and a
        h[i] = h_old[i] - delta
        h[j] = h_old[j] + delta
        a[i] = 1.0 - h[i]
        a[j] = 1.0 - h[j]

        # linear contributions from Δh_i and Δh_j using old a
        k_i = (n - 1) - i
        corr[k_i:k_i + n] -= delta * a_old
        k_j = (n - 1) - j
        corr[k_j:k_j + n] += delta * a_old

        # contributions from Δa_i and Δa_j using old h
        corr[i:i + n] += delta * h_rev
        corr[j:j + n] -= delta * h_rev

        # quadratic correction
        mid = n - 1
        dq = delta * delta
        corr[mid] -= 2.0 * dq
        diff = j - i
        corr[mid + diff] += dq
        corr[mid - diff] += dq

        current_score = np.max(corr) * dx

        # temperature schedule
        T = max(Tmin, T0 * (time_left() / budget_s))

        # accept/reject
        if current_score <= score_old or rng.random() < np.exp((score_old - current_score) / T):
            # accept
            if current_score < best_score:
                best_score = current_score
                best_h = h.copy()
        else:
            # revert
            h = h_old
            a = a_old
            corr = corr_old
            current_score = score_old

        # periodic exact recomputation to avoid drift
        if it % recompute_every == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            current_score = np.max(corr) * dx

    # ---- Local refinement with SLSQP (if time permits) ----
    if time_left() > 10.0:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=300)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # ---- Gradient‑descent momentum refinement (if time permits) ----
    remaining = time_left()
    if remaining > 2.0:
        # estimate number of iterations we can afford (roughly 800 iters per second)
        est_iters = int(min(20000, max(2000, remaining * 800)))
        h_ref, sc_ref = gradient_descent_momentum_refine(best_h,
                                                          n_iter=est_iters,
                                                          step0=0.4,
                                                          beta=0.9)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # final projection and exact evaluation
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
