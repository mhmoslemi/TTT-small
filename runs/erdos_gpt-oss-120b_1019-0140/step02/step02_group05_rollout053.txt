def project_feasible(h, target):
    # ensure bounds
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    sum_h = h.sum()
    diff = target - sum_h
    # distribute diff among free indices (0<h<1)
    free = np.where((h > 0) & (h < 1))[0]
    # iterative distribution
    while abs(diff) > 1e-12 and free.size > 0:
        inc = diff / free.size
        # add inc, but keep within [0,1]
        new_vals = h[free] + inc
        # find which exceed bounds
        overflow_pos = new_vals > 1.0
        overflow_neg = new_vals < 0.0
        # clip
        h[free[overflow_pos]] = 1.0
        h[free[overflow_neg]] = 0.0
        # for the rest, assign new_vals
        idx_ok = ~ (overflow_pos | overflow_neg)
        h[free[idx_ok]] = new_vals[idx_ok]
        # recompute diff
        sum_h = h.sum()
        diff = target - sum_h
        # recompute free
        free = np.where((h > 0) & (h < 1))[0]
    # final clip
    return np.clip(h, 0.0, 1.0)

    sum_h = h.sum()
    diff = target - sum_h
    # tolerance
    tol = 1e-12
    # Try to distribute diff to free entries
    for _ in range(10):  # limit iterations
        if abs(diff) <= tol:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # cannot adjust via interior; adjust boundary
            # find index with capacity in direction of diff
            if diff > 0:
                # need increase sum: find i where h[i] < 1
                idx = np.where(h < 1)[0]
                if idx.size == 0:
                    break

            # update

            # clip
            h = np.clip(h, 0.0, 1.0)

        diff = target - sum_h

import time

from scipy.optimize import minimize

# ---------- helper functions ----------
def full_corr(h):
    """Linear correlation of h with (1-h) using FFT."""
    n = len(h)
    N = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.fft(h, n=N)
    G = np.fft.fft(1.0 - h, n=N)
    corr = np.fft.ifft(H * np.conj(G)).real
    return corr[: 2 * n - 1]


def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx on the grid."""
    dx = 2.0 / len(h)
    return np.max(full_corr(h)) * dx


def gradient_of_overlap(h, k):
    """Derivative of overlap_k = Σ h[i]*(1-h[i+k]) w.r.t. h."""
    n = len(h)
    grad = np.zeros_like(h)
    if k >= 0:
        i_end = n - k
        grad[:i_end] += 1.0 - h[k:]
        grad[k:] -= h[:i_end]
    else:
        kk = -k
        i_start = kk
        grad[i_start:] += 1.0 - h[: n - kk]
        i_end = n - kk
        grad[:i_end] -= h[kk:]
    return grad


def project_feasible(vec, target):
    """Project a vector onto {0≤x≤1, sum(x)=target}."""
    x = np.clip(vec, 0.0, 1.0)
    tol = 1e-12
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size == 0:
            # adjust any variable that can still move
            if diff > 0:
                idx = np.where(x < 1.0)[0]
                if idx.size == 0:
                    break
                inc = min(diff, 1.0 - x[idx[0]])
                x[idx[0]] += inc
            else:
                idx = np.where(x > 0.0)[0]
                if idx.size == 0:
                    break
                dec = min(-diff, x[idx[0]])
                x[idx[0]] -= dec
        else:
            inc = diff / free.size
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
    return np.clip(x, 0.0, 1.0)

    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)

        i, j = rng.choice(n, size=2, replace=False)
        donor, recv = i, j
        max_tr = min(cur_h[donor], 1.0 - cur_h[recv])
        if max_tr <= 0:
            donor, recv = j, i
            max_tr = min(cur_h[donor], 1.0 - cur_h[recv])
        if max_tr <= 0:
            return cur_h.copy()
        delta = rng.random() * max_tr
        cand = cur_h.copy()
        cand[donor] -= delta
        cand[recv] += delta
    elif move == "jitter":
        cand = cur_h.copy()
        m = rng.integers(1, 5)
        idxs = rng.choice(n, size=m, replace=False)
        deltas = rng.normal(scale=0.02, size=m)
        deltas -= deltas.mean()
        cand[idxs] += deltas
    else:
        cand = cur_h.copy()
    return project_feasible(cand, target)


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -----
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        init_n = len(init_h)
        n_target = max(init_n, 160)
        if n_target % 2 != 0:
            n_target += 1
        if n_target != init_n:
            x_old = np.linspace(0, 2, init_n, endpoint=False)
            x_new = np.linspace(0, 2, n_target, endpoint=False)
            h = np.interp(x_new, x_old, init_h)
        else:
            h = init_h.copy()
    else:
        n_target = 160
        h = rng.random(n_target)

    n = n_target
    target_sum = n / 2.0  # because dx = 2/n, sum*dx = 1
    h = project_feasible(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    # ----- simulated annealing parameters -----
    T0 = 0.02
    p_swap = 0.2
    p_transfer = 0.2
    p_gradient = 0.3
    p_jitter = 0.3

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.85:
            break
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        r = rng.random()
        if r < p_swap:
            move = "swap"
        elif r < p_swap + p_transfer:
            move = "transfer"
        elif r < p_swap + p_transfer + p_gradient:
            move = "gradient"
        else:
            move = "jitter"

        if move == "gradient":
            corr = full_corr(cur_h)
            k = int(np.argmax(corr)) - (n - 1)
            dx = 2.0 / n
            grad = gradient_of_overlap(cur_h, k) * dx
            # step size proportional to temperature (scaled for effect)
            step = T * 5.0
            cand = cur_h - step * grad
            cand = project_feasible(cand, target_sum)
        else:
            cand = generate_candidate(cur_h, move, rng, target_sum, n)

        new_val = compute_c5(cand)

        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0:
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            cur_h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break

    # ----- optional smoothing passes -----
    if time.time() - start < budget_s - 5.0:
        for _ in range(10):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
            else:
                break

    # ----- final refinement with SLSQP (if time permits) -----
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0

        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

        constraints = [{"type": "eq", "fun": sum_eq}]
        for k in range(-n + 1, n):
            if k == 0:
                continue

            def make_ineq(k_shift):
                def fun(x):
                    h_vec = x[:n]
                    if k_shift >= 0:
                        overlap = np.dot(h_vec[: n - k_shift], 1.0 - h_vec[k_shift:])
                    else:
                        kk = -k_shift
                        overlap = np.dot(h_vec[kk:], 1.0 - h_vec[: n - kk])
                    return x[-1] - overlap * dx

                return fun

            constraints.append({"type": "ineq", "fun": make_ineq(k)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_feasible(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # final safety correction
    best_h = project_feasible(best_h, n / 2.0)

    return best_h, best_val, n
