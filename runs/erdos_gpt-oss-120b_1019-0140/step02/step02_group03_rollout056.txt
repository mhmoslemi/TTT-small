    import time, numpy as np
    from scipy.optimize import minimize

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine n
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        n = h_init.size
    else:
        n = 80
        h_init = None
    target = n / 2.0
    dx = 2.0 / n

    # define helper functions
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def project_to_box_sum(v, target_sum, lo=0.0, hi=1.0):
        # binary search λ
        low = -np.max(v) - 5.0
        high = (hi - np.min(v)) + 5.0
        for _ in range(60):
            mid = (low + high) * 0.5
            x = np.clip(v + mid, lo, hi)
            s = x.sum()
            if s < target_sum:
                low = mid
            else:
                high = mid
        return np.clip(v + (low + high) * 0.5, lo, hi)

    def gradient_overlap(h, k):
        nloc = h.shape[0]
        grad = np.zeros_like(h)
        if k >= 0:
            if k < nloc:
                grad[:nloc - k] += (1 - h[k:])
                grad[k:] += -h[:nloc - k]
        else:
            d = -k
            if d < nloc:
                grad[d:] += (1 - h[:nloc - d])
                grad[:nloc - d] += -h[d:]
        # scale by dx because C5 includes dx
        return grad * dx

    # initial best
    if h_init is not None:
        # ensure feasible
        if np.any(h_init < 0) or np.any(h_init > 1) or abs(h_init.sum() - target) > 1e-9:
            h_init = project_to_box_sum(np.clip(h_init, 0.0, 1.0), target)

    else:
        # random feasible start
        h_rand = rng.random(n)
        h_rand = project_to_box_sum(h_rand, target)
        best_h = h_rand.copy()
        best_c5 = compute_c5(best_h)

    # if already good enough return early
    if best_c5 <= 0.3808:
        return best_h, best_c5, n

    # ----- Binary hill climbing -----
    binary_deadline = start_time + budget_s * 0.3
    target_ones = int(target)

        h_bin = np.zeros(n, dtype=float)

                    if best_c5 <= 0.3808:

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions (top‑level) ----------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_box_sum(v: np.ndarray, target: float, lo: float = 0.0, hi: float = 1.0) -> np.ndarray:
    """Project v onto [lo,hi]^n while enforcing sum(v) == target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)


def gradient_overlap(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the overlap for the shift that attains the max (scaled by dx)."""
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if k >= 0:
        if k < n:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:]    += -h[:n - k]
    else:
        d = -k
        if d < n:
            grad[d:]    += (1.0 - h[:n - d])
            grad[:n - d] += -h[d:]
    return grad * dx


class SumEquality:
    """Equality constraint: sum(x) == target."""
    def __init__(self, target: float):
        self.target = target
    def __call__(self, x: np.ndarray) -> float:
        return np.sum(x) - self.target


# ---------- Main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- Determine problem size -----
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        n = h_init.size
    else:
        n = 80
        h_init = None
    target_sum = n / 2.0

    # ----- Initialise best solution -----
    if h_init is not None:
        # make sure feasible
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_to_box_sum(h_init, target_sum)
        best_h = h_init.copy()
        best_c5 = compute_c5(best_h)
    else:
        rnd = rng.random(n)
        best_h = project_to_box_sum(rnd, target_sum)
        best_c5 = compute_c5(best_h)

    # Early exit if already good enough
    if best_c5 <= 0.3808:
        return best_h, best_c5, n

    # ----- Phase 1: Binary hill‑climbing (global exploration) -----
    binary_deadline = start_time + budget_s * 0.30
    ones_needed = int(target_sum)

    while time.time() < binary_deadline:
        # random binary start
        h_bin = np.zeros(n, dtype=float)
        idx_one = rng.choice(n, size=ones_needed, replace=False)
        h_bin[idx_one] = 1.0
        cur_c5 = compute_c5(h_bin)

        # inner improvement loop
        inner_limit = 12000
        moves = 0
        while time.time() < binary_deadline and moves < inner_limit:
            # choose a 1‑position and a 0‑position
            i = rng.choice(np.where(h_bin == 1)[0])
            j = rng.choice(np.where(h_bin == 0)[0])
