def local_refine(h, target_sum):
    n = len(h)
    dx = 2.0 / n

    def overlap_vec(x):
        # compute O_k for all shifts, return array length 2n-1
        corr = np.correlate(x, 1.0 - x, mode='full')
        return corr * dx

    def objective(v):
        # v = [h..., t]
        return v[-1]

    def constraint_eq(v):
        # sum constraint
        return np.sum(v[:n]) - target_sum

    def make_ineq(k):
        def fun(v):
            x = v[:n]
            # compute O_k for shift k
            # we can compute inner product quickly:
            # O_k = dx * (target_sum - sum_i x_i * x_{i - k})
            # but we need sum_i x_i*x_{i-k}
            if k >= 0:
                prod = np.dot(x[k:], x[:n - k])
            else:
                kk = -k
                prod = np.dot(x[:n - kk], x[kk:])
            O_k = dx * (target_sum - prod)
            return v[-1] - O_k  # >=0
        return fun

    constraints = [{'type':'eq', 'fun': constraint_eq}]
    for kk in range(-n+1, n):
        if kk == 0:
            continue
        constraints.append({'type':'ineq', 'fun': make_ineq(kk)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    x0 = np.concatenate([h, [compute_c5(h)]])
    try:
        res = minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            # re-impose sum
            cur = np.sum(h_opt)
            if abs(cur - target_sum) > 1e-9:
                h_opt += (target_sum - cur) / n
                h_opt = np.clip(h_opt, 0.0, 1.0)
            val_opt = compute_c5(h_opt)
            return h_opt, val_opt
    except Exception:
        pass
    # fallback: return original
    return h, compute_c5(h)

def simulated_annealing(h0, target_sum, t_run, rng):
    n = len(h0)
    dx = 2.0 / n
    cur_h = h0.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    # schedule: linear cooling from T0 to 0 over t_run
    T0 = 0.02  # maybe larger for better exploration
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= t_run:
            break
        T = T0 * max(0.0, 1.0 - elapsed / t_run)
        # choose move
        move_type = rng.random()
        # probabilities
        if move_type < 0.3:
            # swap
            # pick two distinct positions with different values
            # sample i then j until values differ
            i = rng.integers(n)
            # find j with opposite value
            # there may be many same, so we sample repeatedly
            for _ in range(5):
                j = rng.integers(n)
                if cur_h[i] != cur_h[j]:
                    break
            else:
                # fallback random different
                j = (i + 1) % n
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.6:
            # transfer
            i, j = rng.choice(n, size=2, replace=False)
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr <= 0:
                max_tr = min(cur_h[j], 1.0 - cur_h[i])
                if max_tr <= 0:
                    continue
                donor, recv = j, i
            else:
                donor, recv = i, j
            delta = max_tr * rng.random()
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        else:
            # gradient move
            # compute correlation
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            idx_max = int(np.argmax(corr))
            k = idx_max - (n - 1)
            # compute neigh = contributions
            neigh = np.zeros(n)
            if k >= 0:
                neigh[: n - k] += cur_h[k:]
                neigh[k:] += cur_h[: n - k]
            else:
                kk = -k
                neigh[kk:] += cur_h[: n - kk]
                neigh[: n - kk] += cur_h[kk:]
            grad = dx * (1.0 - neigh)  # gradient of O_k
            # project to zero-sum direction
            grad_centered = grad - np.mean(grad)
            # step
            step_size = 0.1  # base step size maybe
            delta_vec = -step_size * grad_centered
            # enforce bounds
            cand = cur_h + delta_vec
            cand = np.clip(cand, 0.0, 1.0)
            # re-impose sum
            diff = target_sum - np.sum(cand)
            # distribute diff to non-bound elements
            free = np.where((cand > 0) & (cand < 1))[0]
            if free.size > 0:
                inc = diff / free.size
                cand[free] += inc
                cand = np.clip(cand, 0.0, 1.0)
            # if still diff, adjust iteratively
            diff = target_sum - np.sum(cand)
            if abs(diff) > 1e-12:
                for ii in range(n):
                    if abs(diff) < 1e-12:
                        break
                    if diff > 0 and cand[ii] < 1.0:
                        add = min(diff, 1.0 - cand[ii])
                        cand[ii] += add
                        diff -= add
                    elif diff < 0 and cand[ii] > 0.0:
                        sub = min(-diff, cand[ii])
                        cand[ii] -= sub
                        diff += sub
                cand = np.clip(cand, 0.0, 1.0)
        # project again to ensure sum (some moves may violate)
        cand = project_onto_constraints(cand, target_sum)
        new_val = compute_c5(cand)
        delta = new_val - cur_val
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / (T + 1e-12))):
            cur_h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
    return best_h, best_val

def project_onto_constraints(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    # iterate to correct diff
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        else:
            # all at bounds; adjust any that can move
            for i in range(len(h)):
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
            h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = target_sum - cur
    return h

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return the maximum overlap integral C5 for a discretised step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraints(h, target_sum):
    """Clip to [0,1] and adjust to meet the exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        else:
            # all at bounds, distribute greedily
            for i in range(len(h)):
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
            h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = target_sum - cur
    return h

def upsample_h(h, old_n, new_n):
    """Linear interpolation from old_n samples to new_n samples on [0,2)."""
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def smooth_and_fix(h, target_sum):
    """Gaussian‑type smoothing followed by projection onto constraints."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return project_onto_constraints(sm, target_sum)

def simulated_annealing(h0, target_sum, time_budget, rng):
    """Simple SA with swap, transfer and gradient moves."""
    n = len(h0)
    dx = 2.0 / n
    cur_h = h0.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    T0 = 0.03  # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        T = T0 * max(0.0, 1.0 - elapsed / time_budget)

        # choose move type
        r = rng.random()
        if r < 0.35:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            if cur_h[i] == cur_h[j]:
                continue
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.70:  # transfer
            i, j = rng.choice(n, size=2, replace=False)
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr <= 0:
                max_tr = min(cur_h[j], 1.0 - cur_h[i])
                if max_tr <= 0:
                    continue
                donor, recv = j, i
            else:
                donor, recv = i, j
            delta = max_tr * rng.random()
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        else:  # gradient move on the worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = int(np.argmax(corr))
            k = idx_max - (n - 1)
            neigh = np.zeros(n)
            if k >= 0:
                neigh[: n - k] += cur_h[k:]
                neigh[k:] += cur_h[: n - k]
            else:
                kk = -k
                neigh[kk:] += cur_h[: n - kk]
                neigh[: n - kk] += cur_h[kk:]
            grad = dx * (1.0 - neigh)                 # ∂O_k/∂h
            grad_centered = grad - np.mean(grad)      # zero‑sum direction
            step = 0.12
            cand = cur_h - step * grad_centered
            cand = np.clip(cand, 0.0, 1.0)

        cand = project_onto_constraints(cand, target_sum)
        new_val = compute_c5(cand)
        delta = new_val - cur_val
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            prob = np.exp(-delta / (T + 1e-15))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    return best_h, best_val
    return best_h, best_val

def sum_eq(x, target_sum, n):
    """Equality constraint: sum of h equals target_sum."""
    return np.sum(x[:n]) - target_sum

def overlap_ineq(x, n, dx):
    """Inequality constraints: t - O_k >= 0 for all shifts."""
    h = x[:n]
    t = x[-1]
    O_vec = np.correlate(h, 1.0 - h, mode="full") * dx
    return t - O_vec

def local_refine(h, target_sum, time_left):
    """Run a short SLSQP refinement (if time permits)."""
    n = len(h)
    if time_left < 4.0:
        return h, compute_c5(h)
    dx = 2.0 / n
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {"type": "eq", "fun": sum_eq, "args": (target_sum, n)},
        {"type": "ineq", "fun": overlap_ineq, "args": (n, dx)},
    ]
    try:
        res = minimize(
            fun=lambda v: v[-1],
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 300, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            cur = np.sum(h_opt)
            diff = target_sum - cur
            if abs(diff) > 1e-9:
                free = np.where((h_opt > 0) & (h_opt < 1))[0]
                if free.size:
                    inc = diff / free.size
                    h_opt[free] += inc
                else:
                    h_opt += diff / n
                h_opt = np.clip(h_opt, 0.0, 1.0)
            val_opt = compute_c5(h_opt)
            return h_opt, val_opt
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine base resolution and initialise h
    # ------------------------------------------------------------------
    TARGET_N = 200  # higher resolution gives a finer search space
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n0 = len(init_h)
        if n0 != TARGET_N:
            cur_h = upsample_h(init_h, n0, TARGET_N)
        else:
            cur_h = init_h.copy()
    else:
        cur_h = rng.random(TARGET_N)

    target_sum = TARGET_N / 2.0
    cur_h = project_onto_constraints(cur_h, target_sum)

    # ------------------------------------------------------------------
    # Simulated annealing phase (majority of the budget)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 10.0:
        t_sa = remaining * 0.85
        best_h, best_val = simulated_annealing(cur_h, target_sum, t_sa, rng)
    else:
        best_h = cur_h
        best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Optional smoothing passes
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        for _ in range(6):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val < best_val - 1e-12:
                best_val = sm_val
                best_h = sm
            else:
                break

    # ------------------------------------------------------------------
    # Final local refinement with SLSQP (if time permits)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    best_h, best_val = local_refine(best_h, target_sum, remaining)

    # ------------------------------------------------------------------
    # Final safety projection and return
    # ------------------------------------------------------------------
    best_h = project_onto_constraints(best_h, target_sum)
    best_val = compute_c5(best_h)
    return best_h, best_val, TARGET_N
