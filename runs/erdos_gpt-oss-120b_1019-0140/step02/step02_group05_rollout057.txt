import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_and_fix(h, target_sum):
    # Gaussian smoothing kernel [0.25, 0.5, 0.25]
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    cur = np.sum(sm)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            inc = diff / free.size
            sm[free] += inc
            sm = np.clip(sm, 0.0, 1.0)
        # residual
        diff = target_sum - np.sum(sm)
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
        sm = np.clip(sm, 0.0, 1.0)
    return sm

def project_sum_and_bounds(h, target_sum):
    # clip to bounds
    np.clip(h, 0.0, 1.0, out=h)
    cur_sum = np.sum(h)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return h
    # first distribute uniformly among non‑fixed entries
    for _ in range(10):
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            break
        inc = diff / free.size
        h[free] += inc
        np.clip(h, 0.0, 1.0, out=h)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    # if still off, adjust any entries sequentially
    if abs(diff) >= 1e-12:
        for i in range(len(h)):
            if abs(diff) < 1e-12:

                sub = min(-diff, h[i])
                h[i] -= sub

import time

def compute_c5(h):
    """Maximum overlap for a given step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_and_fix(h, target_sum):
    """One Gaussian smoothing pass followed by sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    diff = target_sum - np.sum(sm)
    if abs(diff) > 1e-12:
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            inc = diff / free.size
            sm[free] += inc
            sm = np.clip(sm, 0.0, 1.0)
        diff = target_sum - np.sum(sm)
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
        sm = np.clip(sm, 0.0, 1.0)
    return sm

def project_sum_and_bounds(h, target_sum):
