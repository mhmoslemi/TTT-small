def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project onto box [lo,hi] and sum = target."""
    # Binary search for lambda offset such that sum(clip(v + lam, lo, hi)) = target.
    lam_low = -np.max(v) - 10.0  # ensure all v+lam <= lo
    lam_high = hi - np.min(v) + 10.0  # ensure all >= hi
    for _ in range(50):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(v + lam_mid, lo, hi)
        s = proj.sum()
        if s < target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v + (lam_low + lam_high) / 2.0, lo, hi)

def run(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    from scipy.optimize import minimize

    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n and initial h
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)

    target = n / 2.0
    h0 = project_to_box_sum(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    base_step = 0.05
    iter_count = 0

    # main loop: 85% of budget
    while time.time() - start < budget_s * 0.85:
        # compute autocorrelation
        S = np.correlate(cur_h, cur_h, mode='full')
        # positive lag values
        # shift range 1..n-1
        # shift = argmin(S[n:]) + 1
        shift = np.argmin(S[n:]) + 1

        # gradient = left + right shift
        left = np.concatenate([cur_h[shift:], np.zeros(shift)])
        right = np.concatenate([np.zeros(shift), cur_h[:n - shift]])
        grad = left + right

        # step size schedule
        step = base_step * (0.995 ** iter_count)

        cand = cur_h + step * grad
        cand = project_to_box_sum(cand, target)

        cand_c5 = compute_c5(cand)

        if cand_c5 < cur_c5 - 1e-12:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                if best_c5 < 0.3806:
                    break
        else:
            # simulated annealing acceptance
            elapsed = time.time() - start
            frac = elapsed / (budget_s * 0.85)
            T = max(1e-8, 0.005 * (1.0 - frac))
            if rng.random() < math.exp(-(cand_c5 - cur_c5) / T):
                cur_h = cand
                cur_c5 = cand_c5

        # occasional random perturbation
        if rng.random() < 0.003:
            # add small gaussian noise
            noise = rng.normal(scale=0.05, size=n)
            cand2 = cur_h + noise
            cand2 = project_to_box_sum(cand2, target)
            cand2_c5 = compute_c5(cand2)
            if cand2_c5 < cur_c5:
                cur_h = cand2
                cur_c5 = cand2_c5
                if cur_c5 < best_c5:
                    best_h = cur_h.copy()
                    best_c5 = cur_c5

        # occasional random restart
        if rng.random() < 0.001:
            rand_vec = rng.random(n)
            rand_vec = project_to_box_sum(rand_vec, target)
            rand_c5 = compute_c5(rand_vec)
            if rand_c5 < cur_c5:
                cur_h = rand_vec
                cur_c5 = rand_c5

        iter_count += 1

    # refinement via SLSQP if time allows
    remaining = budget_s - (time.time() - start)
    if remaining > 30.0:
        # define constraints
        class SumConstraint:
            def __init__(self, n):
                self.n = n
            def __call__(self, x):
                return np.sum(x[:self.n]) - self.n / 2.0

        class OverlapConstraint:
            def __init__(self, k, n):
                self.k = k
                self.n = n
                self.dx = 2.0 / n
            def __call__(self, x):
                h = x[:self.n]
                k = self.k
                if k >= 0:
                    overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
                return x[-1] - overlap * self.dx

        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

        t_start = compute_c5(best_h)
        x0 = np.concatenate([best_h, [t_start]])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        res = minimize(
            fun=lambda x: x[-1],
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=([{'type': 'eq', 'fun': sum_con}]
                         + [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]),
            options={'maxiter': 400, 'ftol': 1e-9, 'disp': False}
        )
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target)
            opt_c5 = compute_c5(h_opt)
            if opt_c5 < best_c5:
                best_h = h_opt
                best_c5 = opt_c5

    # final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

import time
import math
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while forcing the sum to equal target."""
    low = -np.max(v) - 10.0
    high = hi - np.min(v) + 10.0
    for _ in range(50):
        mid = (low + high) * 0.5
        proj = np.clip(v + mid, lo, hi)
        s = proj.sum()
        if s < target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(v + lam, lo, hi)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target = n / 2.0
    h0 = project_to_box_sum(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    base_step = 0.05       # initial step magnitude for gradient updates
    T0 = 0.004             # initial temperature for SA acceptance
    it = 0

    # ----- main exploration (≈85 % of budget) -----------------------
    while time.time() - start < budget_s * 0.85:
        # autocorrelation and worst lag
        S = np.correlate(cur_h, cur_h, mode='full')
        shift = np.argmin(S[n:]) + 1                     # 1 ≤ shift ≤ n‑1

        # gradient that raises the autocorrelation at this lag
        left  = np.concatenate([cur_h[shift:], np.zeros(shift)])
        right = np.concatenate([np.zeros(shift), cur_h[:n - shift]])
        grad = left + right

        # linear decay of step size
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.85)
        step = base_step * max(0.01, 1.0 - frac)

        cand = cur_h + step * grad
        cand = project_to_box_sum(cand, target)
        cand_c5 = compute_c5(cand)

        if cand_c5 < cur_c5 - 1e-12:
            cur_h, cur_c5 = cand, cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cur_h.copy(), cur_c5
                if best_c5 < 0.3806:

        else:

                self.n = n
            def __call__(self, x):
                return np.sum(x[:self.n]) - self.n / 2.0

        class OverlapConstraint:

                self.k = k
                self.n = n

            def __call__(self, x):

                k = self.k

    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
