def sa_binary(n, target_sum, dx, rng, time_limit):
   # start from random binary vector with target sum (choose by random shuffle).
   h = np.zeros(n, dtype=int)
   idx = rng.choice(n, size=target_sum, replace=False)
   h[idx] = 1
   cur_score = compute_c5(h.astype(float), dx)
   best_h = h.copy()
   best_score = cur_score
   # schedule: initial temperature = 0.01 * cur_score? Or maybe 0.02.
   start = time.time()
   iter = 0
   while time.time() < time_limit:
        iter += 1
        # pick random swap
        i = rng.choice(np.flatnonzero(h))
        j = rng.choice(np.flatnonzero(1 - h))
        # perform swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5(h.astype(float), dx)
        delta = new_score - cur_score
        # temperature schedule: linear decrease over iterations or time-based
        elapsed = time.time() - start
        fraction = elapsed / (time_limit - start)
        temp = (0.01) * (1 - fraction) + 1e-6
        # Acceptance
        if delta <= 0 or rng.random() < np.exp(-delta / temp):
            cur_score = new_score
            # keep changes
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
   return best_h.astype(float), best_score

def binary_simulated_annealing(n, target_sum, dx, rng, time_deadline):

    idx = rng.choice(n, size=target_sum, replace=False)
    h[idx] = 1
    cur_score = compute_c5_vec(h.astype(float), dx)
    best_h = h.copy()

    start = time.time()

        # Choose random i (1) and j (0)
        i_candidates = np.flatnonzero(h)
        if i_candidates.size == 0:
            break
        i = rng.choice(i_candidates)
        zero_candidates = np.flatnonzero(1 - h)
        if zero_candidates.size == 0:
            break
        j = rng.choice(zero_candidates)
        # swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5_vec(h.astype(float), dx)
        delta = new_score - cur_score
        # compute temperature based on time
        elapsed = time.time() - start
        total = time_deadline - start
        fraction = elapsed / total if total > 0 else 1.0
        temp = max(0.005, (0.02 * (1 - fraction)))  # base temperature 0.02 linearly decreasing to near 0
        if delta <= 0 or rng.random() < np.exp(-delta / temp):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
    return best_h.astype(float), best_score

def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    # if diff near zero, done
    eps = 1e-12
    if abs(diff) < eps:
        return vec
    # distribute diff uniformly
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    # fix any remaining diff by adding/subtracting to/from entries not at caps
    if diff > 0:
        # need to add diff
        # iterate over indices not at 1
        for i in range(len(vec)):
            if diff <= eps:
                break
            if vec[i] < 1.0:
                inc = min(diff, 1.0 - vec[i])
                vec[i] += inc
                diff -= inc
    elif diff < 0:
        # need to subtract
        for i in range(len(vec)):
            if diff >= -eps:
                break
            if vec[i] > 0.0:
                dec = min(-diff, vec[i])
                vec[i] -= dec
                diff += dec
    return vec

import time
import numpy as np

# ------------------------------------------------------------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Scale and clip a vector so that sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    # residual fix
    if diff > 0:
        for i in range(len(vec)):
            if diff <= 1e-12:
                break
            if vec[i] < 1.0:
                inc = min(diff, 1.0 - vec[i])
                vec[i] += inc
                diff -= inc
    elif diff < 0:
        for i in range(len(vec)):
            if diff >= -1e-12:
                break
            if vec[i] > 0.0:
                dec = min(-diff, vec[i])
                vec[i] -= dec
                diff += dec
    return vec


def random_binary_vector(n, ones, rng):
    """Uniform random binary vector with exactly `ones` entries equal to 1."""
    vec = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=ones, replace=False)
    vec[idx] = 1
    return vec

def binary_sa_global(n, ones, dx, rng, deadline, init_bin=None):
    """
    Simulated‑annealing on binary vectors.
    Returns the best vector (as float) and its C5 value.
    """
    if init_bin is not None:
        h = init_bin.copy()
    else:
        h = random_binary_vector(n, ones, rng)

    cur_score = compute_c5_vec(h.astype(float), dx)
    best_h = h.copy()
    best_score = cur_score
    no_improve = 0
    start = time.time()
    total_allowed = max(deadline - start, 1e-12)

    while time.time() < deadline:
        # pick a random donor (1) and recipient (0)
        donors = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if donors.size == 0 or zeros.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(zeros)

        # propose swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5_vec(h.astype(float), dx)
        delta = new_score - cur_score

        # temperature decreasing linearly to ~0
        elapsed = time.time() - start
        fraction = elapsed / total_allowed
        temp = max(0.001, 0.02 * (1.0 - fraction))

        accept = (delta <= 0) or (rng.random() < np.exp(-delta / temp))
        if accept:
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            no_improve += 1

        # occasional restart if stagnating
        if no_improve >= 5000:
            h = random_binary_vector(n, ones, rng)
            cur_score = compute_c5_vec(h.astype(float), dx)
            no_improve = 0

    return best_h.astype(float), best_score

def continuous_refine(h, dx, target_sum, deadline, rng):
    """
    Local coordinate‑descent on continuous values while preserving the sum.
    Returns refined vector and its C5.
    """
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.05
    n = len(cur_h)

    while time.time() < deadline:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_sum(cand, target_sum)
        cand_val = compute_c5_vec(cand, dx)

        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.2, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.95

    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        init_h = None

    target_sum = n / 2.0
    dx = 2.0 / n
    ones = int(round(target_sum))

    best_h = None
    best_val = np.inf

    # start from provided construction if any
    if init_h is not None:
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = project_to_sum(init_h, target_sum)
        val = compute_c5_vec(init_h, dx)
        best_h = init_h
        best_val = val

    # --------------------------------------------------------
    # Phase 1: binary simulated annealing (majority of the budget)
    sa_deadline = start_time + budget_s * 0.75
    # initialise binary start from best_h if possible
    if best_h is not None:
        bin_start = np.zeros(n, dtype=int)
        bin_start[np.argsort(-best_h)[:ones]] = 1
    else:
        bin_start = None

    bin_h, bin_val = binary_sa_global(
        n, ones, dx, rng, sa_deadline, init_bin=bin_start
    )
    if bin_val < best_val - 1e-12:
        best_h = bin_h
        best_val = bin_val

    # early exit if target already reached
    if best_val < 0.3808:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_to_sum(best_h, target_sum)
        return best_h, best_val, n

    # --------------------------------------------------------
    # Phase 2: continuous refinement from the binary best
    time_left = budget_s - (time.time() - start_time)
    if time_left > 2.0:
        cont_deadline = time.time() + time_left
        refined_h, refined_val = continuous_refine(
            best_h.astype(float), dx, target_sum, cont_deadline, rng
        )
        if refined_val < best_val - 1e-12:
            best_h = refined_h
            best_val = refined_val

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
