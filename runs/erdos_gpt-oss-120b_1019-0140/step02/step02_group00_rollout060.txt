def compute_c5_fft(h):
    n = h.shape[0]
    N = 2*n
    h_pad = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    g_pad = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(n)]))
    corr = np.fft.irfft(h_pad * np.conj(g_pad), n=N)
    # Linear correlation from -(n-1) to (n-1) corresponds to corr[:2*n-1] (since rfft returns real part of complex).
    # np.fft.irfft yields length N. However conv length is 2*n - 1. The extra zero at end will be zero.
    # We'll take corr[:2*n-1]
    corr = corr[:2*n-1]
    # Multiply by dx: dx = 2.0 / n
    return np.max(corr) * (2.0 / n)

def local_search_binary(h0, time_left):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    # Precheck: maybe early stop if best_score already < 0.3808
    # Parameters for annealing:
    start_temp = 0.01
    end_temp = 0.0001
    # Determine total steps based on time budget; we compute approximated loops.
    # We'll just loop until time left < 0.5 sec.
    it = 0
    while time_left() > 0.5:
        it += 1
        # linear cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / 10000) # but we need to compute ratio?
        # Actually compute T as:
        # T = max(end_temp, start_temp * (0.99**it)) perhaps simpler.
        T = max(end_temp, start_temp * (0.99**it))
        # propose a swap
        # choose random position i from ones and random j from zeros
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swapping:
        h[i], h[j] = h[j], h[i]  # flip
        score = compute_c5_fft(h)
        delta = score - best_score
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            # accept, maybe update best
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            # reject => revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.optimize as opt

    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # choose n_small and n_large adaptively.
    n_small = 128  # moderate
    n_large = 512

    # Phase 1: SLSQP on small n
    # Build list of starting vectors.
    base_small = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        # downsample to n_small
        base_small.append(project_onto_box_sum(downsample(init_full, n_small), n_small/2))
    # add thue-morse

    # add random binary balanced

    # add random continuous
    for _ in range(3):
        v = rng.random(n_small)
        base_small.append(project_onto_box_sum(v, n_small/2))
    # Do SLSQP on each candidate, until time left low.
    best_h_small = None
    best_score_small = np.inf
    max_iter_slsqp = 800
    for cand in base_small:

        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=max_iter_slsqp)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()
    # Additional random starts while time permits
    while time_left() > 10:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small/2)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=500)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()
        if best_score_small <= 0.3808:
            break

    # Phase 2: Upsample to large n and binarize
    h_up = upsample(best_h_small, n_large)
    # convert to binary balanced using top values
    # choose n_large//2 biggest values
    indices = np.argpartition(-h_up, n_large//2)[:n_large//2]
    h_bin = np.zeros(n_large, dtype=float)
    h_bin[indices] = 1.0

    # compute score
    best_h_large = h_bin.copy()
    best_score_large = compute_c5_fft(best_h_large)

    # Phase 3: binary local search
    # used function local_search_binary
    if time_left() > 5:
        h_local, score_local = local_search_binary(best_h_large, time_left)
        if score_local < best_score_large:
            best_h_large, best_score_large = h_local, score_local

    # Phase 4: final gradient (fractional) refinement
    # Possibly we can refine the best solution as fractional via gradient.
    if time_left() > 10:
        # start from best_h_large (binary) but as float
        h0_frac = best_h_large.copy()
        # run momentum gradient descent (or Nesterov) for limited iterations
        est_iters = int(min(20000, max(5000, time_left() * 800)))
        h_ref, sc_ref = gradient_descent_momentum(h0_frac, n_iter=est_iters, step0=0.5, beta=0.9)
        if sc_ref < best_score_large:
            best_h_large = h_ref
            best_score_large = sc_ref

    # final project onto constraints
    best_h_large = project_onto_box_sum(best_h_large, best_h_large.size / 2)
    best_score_large = compute_c5_fft(best_h_large)

    # return
    return best_h_large, best_score_large, best_h_large.size

def local_search_binary(h0, rng, time_left_func, target_score=0.3808):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.shape[0]
    # We'll iterate until time is low
    iter_count = 0
    while time_left_func() > 1.0:
        iter_count += 1
        # Identify positions of ones and zeros
        ones = np.nonzero(h)[0]
        zeros = np.nonzero(1 - h)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        # number of trial swaps per iteration: maybe 5
        best_swap = None
        best_swap_score = best_score
        for _ in range(5):
            i = rng.integers(ones.size)
            j = rng.integers(zeros.size)
            idx_one = ones[i]
            idx_zero = zeros[j]
            # swap bits
            h[idx_one], h[idx_zero] = h[idx_zero], h[idx_one]
            sc = compute_c5(h)
            if sc < best_swap_score:
                best_swap_score = sc
                best_swap = (idx_one, idx_zero)
            # revert
            h[idx_one], h[idx_zero] = h[idx_zero], h[idx_one]
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            best_h = h.copy()
            best_score = best_swap_score
            if best_score <= target_score:
                break
        else:
            # random swap to escape
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

def local_search_binary(h0, rng, time_budget_func, target_score=0.3808):
    import numpy as np, time
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.shape[0]
    # compute schedule for number of trial swaps per iteration based on n
    trials = max(5, n // 40)  # about 12 for n=512
    while True:
        # compute remaining time
        remaining = time_budget_func()
        if remaining < 0.5:
            break
        # identify ones and zeros
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        # best swap in this iteration
        best_swap = None
        best_swap_score = best_score
        # try a few random swaps
        for _ in range(trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # swap
            h[i], h[j] = h[j], h[i]
            sc = compute_c5(h)
            if sc < best_swap_score:
                best_swap_score = sc
                best_swap = (i, j)
            # revert swap
            h[i], h[j] = h[j], h[i]
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            best_h = h.copy()
            best_score = best_swap_score
            if best_score <= target_score:
                break
        else:
            # random swap to escape
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = h[j], h[i]
        # continue
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """C5 using FFT‐based linear correlation (dx = 2/n)."""
    n = h.shape[0]
    N = 2 * n
    g_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    G = np.fft.rfft(np.concatenate([g_rev, np.zeros(n)]))
    conv = np.fft.irfft(H * G, n=N)
    conv = conv[:2 * n - 1]
    return np.max(conv) * (2.0 / n)

def compute_c5(h):
    return compute_c5_fft(h)

def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    """Linear interpolation from len(arr) down to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def to_balanced_binary(x):
    """Convert any vector to a balanced binary (exactly half ones)."""
    n = x.shape[0]
    half = n // 2
    idx = np.argpartition(-x, half)[:half]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Average gradient for the max‑correlation shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def local_search_binary(h0, rng, start_time, budget_s, target_score=0.3808):
    """Stochastic swap hill‑climbing on a balanced binary vector."""
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.shape[0]
    trials = max(5, n // 40)  # number of random swaps examined per iteration
    while True:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 0.5:
            break
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        best_swap = None
        best_swap_score = best_score
        for _ in range(trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = h[j], h[i]
            sc = compute_c5(h)
            if sc < best_swap_score:
                best_swap_score = sc
                best_swap = (i, j)
            h[i], h[j] = h[j], h[i]
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            best_h = h.copy()
            best_score = best_swap_score
            if best_score <= target_score:
                break
        else:
            # random perturbation if no improvement found
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    target = 0.3808
    n_target = 512  # final resolution (must be even)
    # ----- Initial candidate -----
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n_target:
            if init_full.size < n_target:
                init_up = upsample(init_full, n_target)
            else:
                init_up = downsample(init_full, n_target)
        else:
            init_up = init_full.copy()
        h0 = to_balanced_binary(init_up)
    else:
        h0 = to_balanced_binary(thue_morse(n_target).astype(float))
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    # ----- Quick random start (optional) -----
    if budget_s - (time.time() - start) > 30.0:
        cand = random_binary_balanced(n_target, rng)
        sc = compute_c5(cand)
        if sc < best_score:
            best_h, best_score = cand, sc
    # ----- Local binary search -----
    if budget_s - (time.time() - start) > 5.0:
        h_loc, sc_loc = local_search_binary(best_h, rng, start, budget_s, target_score=target)
        if sc_loc < best_score:

    # ----- Fractional gradient refinement -----
    if budget_s - (time.time() - start) > 10.0:

        est_iters = int(min(20000, max(5000, remaining * 800)))
        h_ref, sc_ref = gradient_descent_momentum(best_h, n_iter=est_iters,
                                                  step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref
    # ----- Final projection and score -----
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
