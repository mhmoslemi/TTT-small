def simulated_annealing_binary(initial_h, dx, target_sum, time_limit, rng):
    cur_h = initial_h.copy()
    cur_val = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    # Temperature schedule.
    # Let T0 = max(0.01, cur_val * 0.1) maybe.
    T0 = 0.01
    # Linear cooling to zero after max_iters or time.
    # Use time proportion: fraction = (time.time() - start_time) / (time_limit - start_time)
    start = time.time()
    while time.time() < time_limit:
        # compute progress
        elapsed = time.time() - start
        frac = elapsed / (time_limit - start)
        T = T0 * (1.0 - frac)  # linearly decreasing
        # random swap
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(~cur_h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = False
        cur_h[j] = True
        new_val = compute_c5(cur_h, dx)
        delta = new_val - cur_val
        if delta <= 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
            # keep swap
        else:
            # revert swap
            cur_h[i] = True
            cur_h[j] = False
    return best_h, best_val
```

def refine_continuous_from_binary(h_bin, dx, target_sum, time_limit, rng):
    # start with h_bin as float (0/1)
    h = h_bin.astype(float).copy()
    cur_val = compute_c5(h, dx)
    step = 0.1
    while time.time() < time_limit:
        # randomly pick i and j, i may give some mass, j may receive
        i = rng.integers(len(h))
        j = rng.integers(len(h))
        if i == j:
            continue
        # choose delta in [-step, step] but ensure keep within [0,1] and sum constant.
        # We'll transfer mass from i to j: delta positive => decrease i, increase j.
        max_decrease_i = min(step, h[i])
        max_increase_j = min(step, 1.0 - h[j])
        max_delta = min(max_decrease_i, max_increase_j)
        if max_delta <= 1e-12:
            continue
        delta = rng.uniform(0, max_delta)
        cand = h.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum unchanged
        cand_val = compute_c5(cand, dx)
        if cand_val < cur_val - 1e-12:
            h = cand
            cur_val = cand_val
            # optionally increase step size
            step = min(step * 1.1, 0.5)
        else:
            # maybe reduce step
            step = max(step * 0.9, 0.01)
    return h, cur_val
```

def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    current = np.sum(vec)
    diff = target_sum - current
    # distribute diff uniformly
    vec += diff / len(vec)
    # clip again
    vec = np.clip(vec, 0.0, 1.0)
    # now adjust residual iteratively
    diff = target_sum - np.sum(vec)
    if abs(diff) < 1e-10:
        return vec
    # adjust using sorted indices
    if diff > 0:
        # need to add diff: increase elements not at 1
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # need to subtract diff (diff negative)
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec
```

def simulated_annealing_binary(initial_h, dx, target_ones, time_limit, rng):
    cur_h = initial_h.copy()
    cur_val = compute_c5(cur_h.astype(float), dx)
    best_h = cur_h.copy()
    best_val = cur_val
    n = len(cur_h)
    # define schedule parameters
    start = time.time()
    # Base temperature determined from initial value
    T0 = 0.01  # initial temperature
    while time.time() < time_limit:
        elapsed = time.time() - start
        total = time_limit - start
        frac = elapsed / total if total > 0 else 1.0
        T = T0 * (1 - frac)
        # Choose random swap
        # To avoid overhead of computing indices each iteration, maintain arrays of indices; but small enough.
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(~cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = False
        cur_h[j] = True
        new_val = compute_c5(cur_h.astype(float), dx)
        delta = new_val - cur_val
        if delta <= 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = True
            cur_h[j] = False
    return best_h.astype(float), best_val

def generate_random_feasible(n, target_sum, rng):
    # draw uniform random in [0,1]
    x = rng.random(n)
    # scale to target sum proportionally
    total = np.sum(x)
    if total == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / total)
    # ensure within bounds
    x = np.clip(x, 0.0, 1.0)
    # adjust residual sum
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        # add or subtract diff equally across indices not at bound
        # simple loop
        indices = np.arange(n)
        for i in indices:
            if diff > 0 and x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return x

import time
import numpy as np

# ---------- Core utilities ----------
def compute_c5(h, dx):
    """Maximum overlap integral C5 for a given step function h."""
    # h is a 1‑D numpy array of floats in [0,1]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum, tol=1e-12):
    """Adjust vec (in [0,1]) to satisfy sum(vec) == target_sum exactly."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) <= tol:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if abs(diff) <= tol:
        return vec
    # greedy fill of residual
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= tol:
                break
    else:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -tol:
                break
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random vector in [0,1]^n with exact sum target_sum."""
    x = rng.random(n)
    total = np.sum(x)
    if total == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / total)
    x = np.clip(x, 0.0, 1.0)
    return project_to_sum(x, target_sum)


# ---------- Simulated annealing for continuous vectors ----------
def sa_continuous(start_h, dx, target_sum, time_limit, rng):
    """Simulated annealing on a continuous step function."""
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    n = len(cur_h)
    step = 0.2
    start = time.time()
    # base temperature; tuned empirically
    T0 = 0.015
    while time.time() < time_limit:
        elapsed = time.time() - start
        total = time_limit - start
        frac = elapsed / total if total > 0 else 1.0
        T = T0 * (1.0 - frac)  # linear cooling
        # choose donor (decrease) and recipient (increase)
        donor = rng.integers(n)
        recip = rng.integers(n)
        if donor == recip:
            continue
        max_delta = min(step, cur_h[donor], 1.0 - cur_h[recip])
        if max_delta <= 0:
            step = max(step * 0.5, 1e-3)
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[donor] -= delta
        cand[recip] += delta
        cand_val = compute_c5(cand, dx)
        dval = cand_val - cur_val
        if dval <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dval / T)):
            cur_h = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val
            step = min(step * 1.08, 0.5)
        else:
            step = max(step * 0.92, 1e-3)
    # final projection (should already be exact)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5(best_h, dx)
    return best_h, best_val


# ---------- Simulated annealing for binary vectors ----------
def sa_binary(start_h, dx, target_ones, time_limit, rng):
    """Simulated annealing on a binary step function (bool array)."""
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h.astype(float), dx)
    best_h = cur_h.copy()
    best_val = cur_val
    n = len(cur_h)
    start = time.time()
    T0 = 0.02
    while time.time() < time_limit:
        elapsed = time.time() - start
        total = time_limit - start
        frac = elapsed / total if total > 0 else 1.0
        T = T0 * (1.0 - frac)
        # pick a 1 and a 0 to swap
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(~cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur_h[i] = False
        cur_h[j] = True
        new_val = compute_c5(cur_h.astype(float), dx)
        dval = new_val - cur_val
        if dval <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dval / T)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = True
            cur_h[j] = False
    return best_h.astype(float), best_val

    if h_initial is not None:

        v0 = compute_c5(h0, dx)
        best_h = h0
        best_val = v0

    # ----- Continuous simulated annealing phase -----
    # Allocate roughly 70% of the budget to many short SA runs
    time_for_cont = budget_s * 0.7
    cont_end = start_time + time_for_cont
    while True:
        now = time.time()
        if now >= cont_end - 4.0:
            break
        # time slice for this SA run
        slice_time = min(4.0, cont_end - now - 0.5)
        h_start = generate_random_feasible(n, target_sum, rng)
        h_cand, v_cand = sa_continuous(h_start, dx, target_sum, now + slice_time, rng)
        if v_cand < best_val:
            best_val = v_cand
            best_h = h_cand.copy()
            if best_val < 0.3808:
                break

    # ----- Binary simulated annealing phase (if time permits) -----
    now = time.time()
    time_left = budget_s - (now - start_time)
    if time_left > 6.0:
        bin_end = now + time_left * 0.5
        used_initial_bin = False
        while time.time() < bin_end - 3.0:
            # choose start vector
            if h_initial is not None and not used_initial_bin:
                # binary version of the initial solution
                idx = np.argsort(-h_initial)[:target_ones]
                start_bin = np.zeros(n, dtype=bool)
                start_bin[idx] = True
                used_initial_bin = True
            else:
                idx = rng.choice(n, size=target_ones, replace=False)
                start_bin = np.zeros(n, dtype=bool)
                start_bin[idx] = True
            run_time = min(3.0, bin_end - time.time())
            h_bin_cand, v_bin_cand = sa_binary(start_bin, dx, target_ones, time.time() + run_time, rng)
            if v_bin_cand < best_val:
                best_val = v_bin_cand
                best_h = h_bin_cand.copy()
                if best_val < 0.3808:
                    break

    # ----- Final refinement (continuous) with remaining time -----
    now = time.time()
    remaining = budget_s - (now - start_time)
    if remaining > 2.0 and best_h is not None:
        h_ref, v_ref = sa_continuous(best_h, dx, target_sum, time.time() + remaining - 0.5, rng)
        if v_ref < best_val:
            best_h = h_ref
            best_val = v_ref

    # Ensure feasibility before returning
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5(best_h, dx)

    return best_h, best_val, n
