def project_sum_and_bounds(x, target):
    """Clip to [0,1] and adjust to achieve sum(target) exactly."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # distribute uniformly
    # allocate diff equally among all entries
    x += diff / len(x)
    # clip again
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # fix residual by iterating
    for i in range(len(x)):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
        else:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
    return x

        return x
    # distribute uniformly, then clip and fix residual
    x += diff / len(x)
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    for i in range(len(x)):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
        else:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
    return x

def local_optimize(h_start, target, time_budget, rng):
    """Gradient descent with soft‑max smoothing on a single start."""
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 200.0
    alpha0 = 0.8
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        # raise smoothing
        beta = min(max_beta, beta * 1.01)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / v.size
        else:
            w /= w_sum
        # gradient of soft‑max objective
        g = np.zeros_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                l = -k
                tail = slice(l, n)
                head = slice(0, n - l)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # preserve total mass
        g -= g.mean()
        # step‑size schedule
        alpha = alpha0 * max(0.01, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h = project_sum_and_bounds(h_new, target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_global = time.time()
    rng = np.random.default_rng(seed)

    # --- initial construction -------------------------------------------------
    if "initial_h_values" in globals():
        h0_initial = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0_initial)
        if n0 * 2 <= 800:
            n = n0 * 2
            h0 = np.repeat(h0_initial, 2)
        else:
            n = n0
            h0 = h0_initial.copy()
    else:
        n = 200
        h0 = rng.random(n)
        h0 = project_sum_and_bounds(h0, n / 2.0)

    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # --- few simple parametric seeds ----------------------------------------
    for a in np.linspace(0.1, 0.5, 9):
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h_cos = 0.5 + a * np.cos(np.pi * x)
        h_cos = np.clip(h_cos, 0.0, 1.0)
        h_cos = project_sum_and_bounds(h_cos, target)
        c5_cos = compute_c5(h_cos)
        if c5_cos < best_c5:
            best_c5 = c5_cos
            best_h = h_cos.copy()

    # --- main multi‑restart optimization --------------------------------------
    restart_counter = 0
    while True:
        elapsed_global = time.time() - start_global
        time_left = budget_s - elapsed_global
        if time_left < 2.0:
            break
        # decide start point
        if rng.random() < 0.2:
            h_start = rng.random(n)
            h_start = project_sum_and_bounds(h_start, target)
        else:
            scale = 0.05 * (0.7 ** restart_counter)
            h_start = best_h + rng.normal(scale=scale, size=n)
            h_start = project_sum_and_bounds(h_start, target)
        local_time = min(30.0, time_left * 0.5)
        h_local, c5_local = local_optimize(h_start, target, local_time, rng)
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_local.copy()
            restart_counter = 0
        else:
            restart_counter += 1
            if restart_counter >= 12:
                restart_counter = 0  # reset scale growth

    # --- final projection and result -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
