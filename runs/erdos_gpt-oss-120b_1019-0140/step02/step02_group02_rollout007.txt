        h = np.repeat(base, factor)
    else:
        n = 120
        h = random_start_vector(n, rng)

    target = n / 2.0
    adjust_sum(h, target)
    h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(best_h)

    # ----- main stochastic improvement loop -----
    while time.time() < deadline - 0.5:
        # current state evaluation
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_c5 = np.max(corr) * dx
        if cur_c5 < best_val:
            best_val = cur_c5
            best_h = h.copy()
            if best_val < 0.3808:
                break

        # identify worst shift
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        # targeted move (skip if shift zero)
        if k != 0:
            if k > 0:
                i_range = np.arange(0, n - k)
                i_target = i_range + k
            else:
                kk = -k
                i_range = np.arange(kk, n)
                i_target = i_range - kk
            contrib = h[i_range] * (1.0 - h[i_target])
            i_max_local = np.argmax(contrib)
            src = i_range[i_max_local]
            dst = i_target[i_max_local]
            max_delta = min(h[src], 1.0 - h[dst])
            if max_delta > 0:
                delta = max_delta * (rng.random() * 0.4 + 0.1)  # 10‑50 % of allowed
                h[src] -= delta
                h[dst] += delta
                new_corr = np.correlate(h, 1.0 - h, mode="full")
                new_c5 = np.max(new_corr) * dx
                if new_c5 < cur_c5:
                    cur_c5 = new_c5
                else:
                    # annealing acceptance
                    remaining = max(deadline - time.time(), 0.0)
                    T = (remaining / budget_s) * 0.008 + 1e-12
                    if np.exp(-(new_c5 - cur_c5) / T) > rng.random():
                        cur_c5 = new_c5
                    else:
                        h[dst] -= delta
                        h[src] += delta

        # occasional random transfer to escape local traps
        if rng.random() < 0.18:
            src_cands = np.where(h > 1e-12)[0]
            dst_cands = np.where(h < 1.0 - 1e-12)[0]
            if src_cands.size and dst_cands.size:
                src2 = rng.choice(src_cands)
                dst2 = rng.choice(dst_cands)
                max_delta2 = min(h[src2], 1.0 - h[dst2])
                if max_delta2 > 0:
                    delta2 = max_delta2 * rng.random() * 0.3
                    h[src2] -= delta2
                    h[dst2] += delta2
                    new_corr2 = np.correlate(h, 1.0 - h, mode="full")
                    new_c5_2 = np.max(new_corr2) * dx
                    if new_c5_2 < cur_c5:
                        cur_c5 = new_c5_2
                    else:
                        remaining = max(deadline - time.time(), 0.0)
                        T = (remaining / budget_s) * 0.008 + 1e-12
                        if np.exp(-(new_c5_2 - cur_c5) / T) > rng.random():
                            cur_c5 = new_c5_2
                        else:
                            h[dst2] -= delta2
                            h[src2] += delta2

    # ----- optional SLSQP refinement (if time permits) -----
    if time.time() < deadline - 2.0:
        # pack h and auxiliary variable t
        x0 = np.concatenate([best_h, [best_val]])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        constraints = [{"type": "eq", "fun": SumConstraint(n, target)}]
        for kk in range(-n + 1, n):
            if kk == 0:
                continue
            constraints.append({"type": "ineq", "fun": OverlapConstraint(kk, n, dx)})
        try:
            res = minimize(
                fun=objective_last,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 600, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                adjust_sum(h_opt, target)
                final_val = compute_c5(h_opt)
                if final_val < best_val:
                    best_h = h_opt
                    best_val = final_val
        except Exception:
            pass

    # ----- final safety enforcement -----
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum(best_h, target)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
