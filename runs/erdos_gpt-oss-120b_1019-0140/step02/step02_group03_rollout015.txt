def simulated_annealing(start_h, start_c5, time_limit):
    best_h = start_h.copy()
    best_c5 = start_c5
    cur_h = start_h.copy()
    cur_c5 = start_c5
    T = 0.002  # initial temperature
    decay = 0.9995  # per iteration factor
    # step size max delta gradually shrink
    max_delta = 0.2
    iter = 0
    while time.time() < time_limit:
        iter += 1
        # Choose two indices
        i, j = rng.choice(n, size=2, replace=False)
        # compute allowed delta range
        hi = cur_h[i]
        hj = cur_h[j]
        # for delta > 0: add to i, subtract from j
        delta_max = min(1.0 - hi, hj)  # ensure hi+delta <=1, hj-delta >=0
        delta_min = -min(hi, 1.0 - hj)  # ensure hi+delta >=0, hj-delta <=1
        if delta_max <= delta_min:
            continue  # no feasible delta; unlikely
        # choose delta with scale, maybe larger early
        scale = max_delta * (T / (T + 0.001))  # gradually shrink with T
        # sample delta uniformly within [delta_min, delta_max] scaled by factor?
        # Actually we can sample uniform and then clip to scaled magnitude
        delta = rng.uniform(delta_min, delta_max)
        # Optionally amplitude scaling:
        delta = np.clip(delta, -scale, scale)  # ensure not too big
        # If delta not feasible after clipping, continue
        if delta == 0.0:
            continue
        # Apply move
        cand_h = cur_h.copy()
        cand_h[i] += delta
        cand_h[j] -= delta
        # Compute C5
        cand_c5 = compute_c5(cand_h)
        # Acceptance
        if cand_c5 < cur_c5:
            cur_h = cand_h
            cur_c5 = cand_c5

                best_h = cand_h.copy()

        if iter % 1000 == 0:

Define projection to box sum:

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = hi - np.min(v) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = hi - np.min(v) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    def __init__(self, n, target):
        self.n = n
        self.target = target
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def objective_t(x):
    return x[-1]

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):

    return minimize(objective_t, x0, method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': maxiter,
                                              'ftol': 1e-9,
                                              'disp': False})

def simulated_annealing(start_h, start_c5, n, target_sum, time_limit, rng):
    cur_h = start_h.copy()
    cur_c5 = start_c5
    best_h = start_h.copy()
    best_c5 = start_c5
    T = 0.002
    decay = 0.99975
    max_delta = 0.2
    it = 0
    while time.time() < time_limit:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = cur_h[i], cur_h[j]

        if delta == 0.0:
