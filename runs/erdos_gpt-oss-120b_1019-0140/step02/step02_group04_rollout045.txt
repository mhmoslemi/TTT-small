def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # Distribute diff uniformly
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    # re-calc diff
    diff = target_sum - np.sum(vec)
    # If diff still nonzero (due to clipping), adjust greedily
    if abs(diff) > 1e-12:
        # sort indices by closeness to bounds? simpler: iterate
        if diff > 0:
            # need to add diff
            idx = np.where(vec < 1.0)[0]
            for i in idx:
                inc = min(diff, 1.0 - vec[i])
                vec[i] += inc
                diff -= inc
                if diff <= 1e-12:
                    break
        else:
            idx = np.where(vec > 0.0)[0]
            for i in idx:
                dec = min(-diff, vec[i])
                vec[i] -= dec
                diff += dec
                if diff >= -1e-12:
                    break
    return vec

import time
import numpy as np

def project_to_sum(vec, target):
    """Clip to [0,1] and adjust so that sum(vec) == target."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return vec
    # distribute diff uniformly
    vec += diff / vec.size
    vec = np.clip(vec, 0.0, 1.0)
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec

def generate_random_feasible(n, target, rng):
    """Random vector in [0,1]^n with exact sum=target."""
    x = rng.random(n)
    s = x.sum()
    if s == 0:
        x = np.full(n, target / n)
    else:
        x = x * (target / s)
    return project_to_sum(x, target)

def compute_c5(h, dx):
    """Maximum overlap integral for step values h, given spacing dx."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def binary_hill_climb(h_bin, cur_val, dx, end_time, rng):
    """Swap‑based hill climbing on a binary (0/1) vector."""
    best = h_bin.copy()
    best_val = cur_val
    n = best.size
    while time.time() < end_time:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best = cand
            best_val = cand_val
    return best.astype(float), best_val
