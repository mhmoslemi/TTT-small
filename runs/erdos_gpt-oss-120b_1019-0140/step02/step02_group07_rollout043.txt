def hill_climb_binary(h_bin, time_end):
    n = len(h_bin)
    while time.time() < time_end:
        # compute corr and max
        corr = np.correlate(h_bin.astype(float), 1 - h_bin.astype(float), mode='full')
        cur_max = corr.max()
        # compute contributions matrix
        contribs = compute_contributions(h_bin)
        ones = np.flatnonzero(h_bin)
        zeros = np.flatnonzero(1 - h_bin)
        best_impr = 0
        best_swap = None
        best_delta_full = None
        for i in ones:
            ci = contribs[i]
            for j in zeros:
                # delta_pos = ci - contribs[j]
                delta_pos = ci - contribs[j]
                # delta_full = ...
                delta_full = np.concatenate((delta_pos[::-1], [0], delta_pos))
                new_max = np.max(corr + delta_full)
                improvement = cur_max - new_max
                if improvement > best_impr + 1e-12:

        else:

    c5 = np.max(corr) * (2.0 / n)

def compute_c5(arr):
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return float(corr.max() * (2.0 / n))

def compute_c5_corr(arr):
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    c5 = float(corr.max() * (2.0 / n))
    return corr, c5

def compute_contributions(h):
    n = len(h)
    contribs = np.zeros((n, n-1), dtype=int)
    for idx in range(n):
        if idx > 0:
            contribs[idx, :idx] = h[idx-1::-1]
        if idx < n-1:
            contribs[idx, :n-idx-1] += h[idx+1:]
    return contribs

def hill_climb_binary(h_bin, time_end):
    n = len(h_bin)
    while time.time() < time_end:
        corr = np.correlate(h_bin.astype(float), 1 - h_bin.astype(float), mode='full')
        cur_max = corr.max()
        contribs = compute_contributions(h_bin)
        ones = np.flatnonzero(h_bin)
        zeros = np.flatnonzero(1 - h_bin)
        best_impr = 0.0
        best_i = -1
        best_j = -1
        best_delta_full = None
        for i in ones:
            ci = contribs[i]
            for j in zeros:
                delta_pos = ci - contribs[j]
                # Build full delta
                delta_full = np.concatenate((delta_pos[::-1], [0], delta_pos))
                new_max = np.max(corr + delta_full)
                improvement = cur_max - new_max
                if improvement > best_impr + 1e-12:
                    best_impr = improvement
                    best_i = i
                    best_j = j
                    best_delta_full = delta_full
        if best_impr > 1e-12:

            h_bin[best_i] = 0
            h_bin[best_j] = 1
            # Update corr using delta
            # This step optional; will recalc next iteration anyway.
            # Optionally break early if time low
        else:
            break
    # final compute
    final_corr, final_c5 = compute_c5_corr(h_bin.astype(float))

def adjust_binary_sum(seq, target_sum, rng):
    cur = int(seq.sum())
    diff = target_sum - cur
    while diff != 0:
        if diff > 0:
            # need to turn zero to one
            zeros = np.flatnonzero(1 - seq)
            if zeros.size == 0:

import time
import numpy as np

def compute_c5(arr):
    """C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated by correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return float(corr.max() * (2.0 / n))

def compute_c5_corr(arr):
    """Return correlation array and C5 value."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return corr, float(corr.max() * (2.0 / n))

def thue_morse_seq(n):
    """Thue‑Morse binary sequence."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    """Rudin‑Shapiro binary sequence (+1/‑1 → 1/0)."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def adjust_binary_sum(seq, target, rng):
    """Flip random bits so that sum(seq) == target."""
    cur = int(seq.sum())
    while cur != target:
        if cur < target:
            zeros = np.flatnonzero(1 - seq)
            if zeros.size == 0:
                break
            idx = rng.choice(zeros)
            seq[idx] = 1
            cur += 1
        else:
            ones = np.flatnonzero(seq)
            if ones.size == 0:
                break
            idx = rng.choice(ones)
            seq[idx] = 0
            cur -= 1
    return seq

def compute_contributions(h):
    """For each index i, contributions_i[k] = h[i+k] + h[i-k] (where defined)."""
    n = len(h)
    contribs = np.zeros((n, n - 1), dtype=int)
    for i in range(n):
        if i > 0:
            contribs[i, :i] = h[i - 1 :: -1]          # backward part
        if i < n - 1:
            contribs[i, : n - i - 1] += h[i + 1 :]   # forward part
    return contribs

def hill_climb_binary(h_bin, time_end, target_max_int):
    """Exhaustive steepest‑descent swap hill climbing for binary vectors."""
    n = len(h_bin)
    while time.time() < time_end:
        corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        cur_max = int(corr.max())
        if cur_max <= target_max_int:
            break
        contribs = compute_contributions(h_bin)
        ones = np.flatnonzero(h_bin)
        zeros = np.flatnonzero(1 - h_bin)
        best_impr = 0
        best_i = best_j = -1
        for i in ones:
            ci = contribs[i]
            for j in zeros:
                delta_pos = ci - contribs[j]                     # length n‑1
                delta_full = np.concatenate((delta_pos[::-1], [0], delta_pos))
                new_max = int((corr + delta_full).max())
                impr = cur_max - new_max
                if impr > best_impr:
                    best_impr = impr
                    best_i, best_j = i, j
        if best_impr > 0:
            h_bin[best_i] = 0
            h_bin[best_j] = 1
        else:
            break
    _, final_c5 = compute_c5_corr(h_bin.astype(float))
    return h_bin.astype(float), final_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    end_time = start_time + budget_s - 0.5        # safety margin
    rng = np.random.default_rng(seed)

    # determine problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80
    target_sum = n // 2
    target_max_int = int(np.floor(0.3808 * n / 2))   # max_corr <= 15 for n=80

    # build a collection of binary starting seeds
    seeds = []

    # 1. rounded initial construction (if provided)
    if 'initial_h_values' in globals():
        init_bin = (np.array(initial_h_values, dtype=float) >= 0.5).astype(int)
        init_bin = adjust_binary_sum(init_bin, target_sum, rng)
        seeds.append(init_bin)

    # 2. Thue‑Morse
    tm = thue_morse_seq(n)
    tm = adjust_binary_sum(tm, target_sum, rng)
    seeds.append(tm)

    # 3. Rudin‑Shapiro
    rs = rudin_shapiro_seq(n)
    rs = adjust_binary_sum(rs, target_sum, rng)
    seeds.append(rs)

    # 4. Alternating pattern
    alt = np.tile([1, 0], n // 2).astype(int)
    seeds.append(alt)

    # 5. Block pattern (half block)
    block = np.concatenate([np.ones(target_sum, dtype=int),
                            np.zeros(n - target_sum, dtype=int)])
    seeds.append(block)

    # 6. Several random half‑ones vectors
    for _ in range(5):
        rand = np.zeros(n, dtype=int)
        rand[rng.choice(n, target_sum, replace=False)] = 1
        seeds.append(rand)

    best_h = None
    best_c5 = np.inf

    # explore each seed with hill climbing
    for start in seeds:
        if time.time() >= end_time:
            break
        h0 = start.copy()
        # local time budget per seed (a few seconds, but hill climb stops early)
        local_end = min(end_time, time.time() + 8.0)
        h_opt, c5_opt = hill_climb_binary(h0, local_end, target_max_int)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        if best_c5 <= 0.3808:
            break

    # random perturbation + hill climb loop
    while time.time() < end_time - 0.2:
        if best_h is None:
            break
        # create a perturbed copy of the current best (swap a few bits)
        h_pert = best_h.astype(int).copy()
        swaps = rng.integers(1, 6)            # 1‑5 random swaps
        for _ in range(swaps):
            ones = np.flatnonzero(h_pert)
            zeros = np.flatnonzero(1 - h_pert)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_pert[i] = 0
            h_pert[j] = 1
        local_end = end_time
        h_opt, c5_opt = hill_climb_binary(h_pert, local_end, target_max_int)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        if best_c5 <= 0.3808:
            break

    # final safety check: enforce exact sum and bounds
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        diff = target_sum - np.sum(best_h)
        if abs(diff) > 1e-12:
            best_h += diff / n
            best_h = np.clip(best_h, 0.0, 1.0)
            # final tiny correction loop
            residual = target_sum - np.sum(best_h)
            for i in range(n):
                if abs(residual) < 1e-12:
                    break
                adj = np.clip(residual, -best_h[i], 1 - best_h[i])
                best_h[i] += adj
                residual = target_sum - np.sum(best_h)

        final_c5 = compute_c5(best_h)
        best_c5 = final_c5

    else:
        # fallback: simple half‑ones random vector
        best_h = np.zeros(n, dtype=float)
        best_h[rng.choice(n, target_sum, replace=False)] = 1.0
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
