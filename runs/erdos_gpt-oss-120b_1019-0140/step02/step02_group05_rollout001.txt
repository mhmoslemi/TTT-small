                    h[i], h[j] = h[j], h[i]
                else:
                    # small transfer
                    max_tr = min(h[i], 1.0 - h[j])
                    if max_tr <= 0:
                        max_tr = min(h[j], 1.0 - h[i])
                        if max_tr <= 0:
                            continue
                        donor, recv = j, i
                    else:
                        donor, recv = i, j
                    delta = rng.random() * max_tr
                    h[donor] -= delta
                    h[recv] += delta
            # re‑impose exact sum
            diff = target_sum - np.sum(h)
            if abs(diff) > 1e-12:
                free = np.where((h > 0) & (h < 1))[0]
                if free.size:
                    inc = diff / free.size
                    h[free] += inc
                    h = np.clip(h, 0.0, 1.0)
                diff = target_sum - np.sum(h)
                for i in range(n):
                    if abs(diff) < 1e-12:
                        break
                    if diff > 0 and h[i] < 1.0:
                        inc = min(diff, 1.0 - h[i])
                        h[i] += inc
                        diff -= inc
                    elif diff < 0 and h[i] > 0.0:
                        dec = min(-diff, h[i])
                        h[i] -= dec
                        diff += dec
                h = np.clip(h, 0.0, 1.0)

        cur_h = h.copy()
        cur_val = compute_c5(cur_h)

        # ----- inner simulated‑annealing moves -----------------------
        for _ in range(max_iters_per_run):
            if time.time() - start >= main_budget:
                break

            elapsed = time.time() - start
            T0 = 0.2
            T = T0 * max(0.0, 1.0 - elapsed / total_time)

            r = rng.random()
            # ----- swap move -----
            if r < 0.4:
                i, j = rng.choice(n, 2, replace=False)
                cand = cur_h.copy()
                cand[i], cand[j] = cand[j], cand[i]

            # ----- transfer move -----
            elif r < 0.7:
                i, j = rng.choice(n, 2, replace=False)
                max_tr = min(cur_h[i], 1.0 - cur_h[j])
                if max_tr <= 0:
                    max_tr = min(cur_h[j], 1.0 - cur_h[i])
                    if max_tr <= 0:
                        continue
                    donor, recv = j, i
                else:
                    donor, recv = i, j
                delta = rng.random() * max_tr
                cand = cur_h.copy()
                cand[donor] -= delta
                cand[recv] += delta

            # ----- gradient move on worst shift -----
            else:
                corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
                idx_max = int(np.argmax(corr))
                k = idx_max - (n - 1)  # shift index
                grad = np.zeros(n)
                if k >= 0:
                    grad[: n - k] += 1.0 - cur_h[k:]
                    grad[k:] -= cur_h[: n - k]
                else:
                    kk = -k
                    grad[kk:] += 1.0 - cur_h[: n - kk]
                    grad[: n - kk] -= cur_h[kk:]
                delta_vec = -grad
                max_step = 0.3
                mabs = np.max(np.abs(delta_vec))
                if mabs > max_step:
                    delta_vec = delta_vec * (max_step / mabs)
                cand = cur_h + delta_vec
                cand = np.clip(cand, 0.0, 1.0)
                # re‑impose sum
                diff = target_sum - np.sum(cand)
                if abs(diff) > 1e-12:
                    free = np.where((cand > 0) & (cand < 1))[0]
                    if free.size:
                        inc = diff / free.size
                        cand[free] += inc
                        cand = np.clip(cand, 0.0, 1.0)
                    diff = target_sum - np.sum(cand)
                    for i in range(n):
                        if abs(diff) < 1e-12:
                            break
                        if diff > 0 and cand[i] < 1.0:
                            add = min(diff, 1.0 - cand[i])
                            cand[i] += add
                            diff -= add
                        elif diff < 0 and cand[i] > 0.0:
                            sub = min(-diff, cand[i])
                            cand[i] -= sub
                            diff += sub
                    cand = np.clip(cand, 0.0, 1.0)

            new_val = compute_c5(cand)
            accept = False
            if new_val < cur_val:
                accept = True
            elif T > 0.0:
                prob = math.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True
            if accept:
                cur_h = cand
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = cand.copy()
                    if best_val < target_threshold:
                        break
        # ----- optional smoothing of current best -------------------
        if time.time() - start < main_budget - 1.0:
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm

        if best_val < target_threshold:
            break

    # ----- final smooth‑max refinement --------------------------------
    time_left = budget_s - (time.time() - start)
    if time_left > 5.0:
        alpha = 15.0  # smoothing parameter for log‑sum‑exp
        def smooth_obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            overlaps = corr * (2.0 / n)
            max_o = np.max(overlaps)
            lse = max_o + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (overlaps - max_o))))
            penalty = 1e6 * (np.sum(h) - target_sum) ** 2
            return lse + penalty

        bounds = [(0.0, 1.0)] * n
        maxfun = max(500, int(2000 * (time_left / budget_s)))
        try:
            res = minimize(
                smooth_obj,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxfun': maxfun, 'maxiter': maxfun, 'ftol': 1e-12, 'disp': False},
            )
            if res.success:
                h_opt = np.clip(res.x, 0.0, 1.0)
                # enforce exact sum
                diff = target_sum - np.sum(h_opt)
                if abs(diff) > 1e-9:
                    interior = np.where((h_opt > 0) & (h_opt < 1))[0]
                    if interior.size:
                        inc = diff / interior.size
                        h_opt[interior] += inc
                        h_opt = np.clip(h_opt, 0.0, 1.0)
                    diff = target_sum - np.sum(h_opt)
                    for i in range(n):
                        if abs(diff) < 1e-12:
                            break
                        if diff > 0 and h_opt[i] < 1.0:
                            add = min(diff, 1.0 - h_opt[i])
                            h_opt[i] += add
                            diff -= add
                        elif diff < 0 and h_opt[i] > 0.0:
                            sub = min(-diff, h_opt[i])
                            h_opt[i] -= sub
                            diff += sub
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # ----- final safety correction ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-9:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        diff = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and best_h[i] < 1.0:
                add = min(diff, 1.0 - best_h[i])
                best_h[i] += add
                diff -= add
            elif diff < 0 and best_h[i] > 0.0:
                sub = min(-diff, best_h[i])
                best_h[i] -= sub
                diff += sub
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, n
