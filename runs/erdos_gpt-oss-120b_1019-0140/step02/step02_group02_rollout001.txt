def get_worst_shift_info(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def get_worst_shift_info(h: np.ndarray):
    """Find shift k giving maximal overlap and contributions for that shift."""
    n = h.shape[0]

        prod = h[kk:] * (1.0 - h[: n - kk])

            if abs(diff) < 1e-12:
                break
            if diff > 0 and h0[i] < 1.0:
                inc = min(diff, 1.0 - h0[i])
                h0[i] += inc
                diff -= inc
            elif diff < 0 and h0[i] > 0.0:
                dec = min(-diff, h0[i])
                h0[i] -= dec
                diff += dec

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # --- repeated local annealing with occasional random restarts ---
    while time.time() < deadline - 2.0:
        # decide whether to start from scratch or perturb current best
        if rng.random() < 0.2:
            h_start = random_feasible_start(n, target_sum, rng)
        else:
            h_start = best_h.copy()
            for _ in range(5):
                src = rng.choice(np.where(h_start > 1e-12)[0])
                dst = rng.choice(np.where(h_start < 1.0 - 1e-12)[0])
                max_t = min(h_start[src], 1.0 - h_start[dst])
                if max_t > 0:
                    d = rng.random() * max_t
                    h_start[src] -= d
                    h_start[dst] += d

        cur_val = compute_c5(h_start)
        h_opt, val_opt = local_anneal(h_start, cur_val, deadline, rng)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # --- optional SLSQP refinement if time permits ---
    if time.time() < deadline - 1.0:
        dx = 2.0 / n
        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

        constraints = [{"type": "eq", "fun": sum_eq}]

        def make_overlap_con(k):
            def con(x, k=k):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[: n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[: n - kk])
                return x[-1] - overl * dx
            return con

        for kk in range(-n + 1, n):
            if kk == 0:
                continue
            constraints.append({"type": "ineq", "fun": make_overlap_con(kk)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        x0 = np.concatenate([best_h, [best_val]])
        time_left = deadline - time.time()
        maxiter = int(min(2000, max(200, time_left * 5)))
        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_ref = np.clip(res.x[:n], 0.0, 1.0)
                # enforce sum exactly again
                diff = target_sum - h_ref.sum()
                if abs(diff) > 1e-9:
                    h_ref += diff / n
                    h_ref = np.clip(h_ref, 0.0, 1.0)
                val = compute_c5(h_ref)
                if val < best_val:
                    best_val = val
                    best_h = h_ref.copy()
        except Exception:
            pass

    # --- final safety adjustments ---
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target_sum) > 1e-9:
        best_h += (target_sum - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
