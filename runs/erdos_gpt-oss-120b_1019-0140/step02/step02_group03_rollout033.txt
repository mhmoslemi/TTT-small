```
class RandomSwapStep:
    def __init__(self, rng, step_size_frac=0.1):
        self.rng = rng
        self.fraction = step_size_frac
    def __call__(self, x):
        n = x.shape[0]
        # choose some indices to adjust
        # simple: pick two indices i,j, move delta from i to j ensuring bounds.
        y = x.copy()
        for _ in range(int(self.fraction * n) + 1):
            i, j = self.rng.integers(0, n, size=2)
            if i == j: continue
            # Determine max amount we can move from i to j
            # Move delta in [-max_move, max_move] where max_move = min(y[i], 1-y[j])
            max_move = min(y[i], 1.0 - y[j])
            if max_move <= 0: continue
            delta = (self.rng.random() - 0.5) * 2 * max_move * 0.5  # random fraction up to half max_move
            # Actually we just want to transfer from i to j: delta >0 means reduce i, increase j.
            # Choose uniform in [0, max_move]
            delta = self.rng.random() * max_move
            y[i] -= delta
            y[j] += delta
        return y
```

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while enforcing sum(v)=target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0

        x = np.clip(v + mid, lo, hi)
        s = x.sum()

            low = mid
        else:
            high = mid
