def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.optimize
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)

    target_sum = n/2.0
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # allocate fractions
    time_budget = budget_s
    time_mid = start + time_budget * 0.45
    time_end = start + time_budget

    # Phase DE
    def de_obj(x):
        x = project_to_box_sum(x, target_sum)
        return compute_c5(x)

    # compute de_maxiter based on time
    de_bounds = [(0.0, 1.0)] * n

        return time.time() >= time_mid

            maxiter=200,
            popsize=10,

            disp=False,
        )

        de_res = None

    if de_res is not None and de_res.x is not None:

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type':'eq','fun': sum_con}]
    for oc in overlap_cons:
        constraints.append({'type':'ineq','fun': oc})
    x0 = np.concatenate([h_start, [t_start]])
    res = scipy.optimize.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds, constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res
```

import time
import numpy as np
import scipy.optimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0, iters=60):
    """Project v onto [lo,hi]^n while fixing the sum to target using binary search."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(iters):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    """Equality constraint sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500):
    """Run SLSQP to minimise t subject to constraints."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = scipy.optimize.minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res
