def project_to_sum_target(x, target):
    # clip x in [0,1]
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if np.isclose(s, target):
        return x
    if s > target:
        # decrease sum
        # Compute surplus = s - target
        surplus = s - target
        # Sort indices by x descending
        idx = np.argsort(-x)  # largest first
        for i in idx:
            if surplus <= 0:
                break
            # amount we can reduce from x[i] (current value) down to 0
            reducible = x[i]
            delta = min(reducible, surplus)
            x[i] -= delta
            surplus -= delta
        # after loop, may still have small surplus due to rounding, adjust tiny.
        # If still surplus > 1e-12, we can distribute uniformly (but should not happen)
        if surplus > 1e-12:
            # distribute evenly among all (avoid breaking bounds)
            # This is extremely unlikely; just subtract surplus/len(x) and clip again
            x -= surplus / len(x)
            x = np.clip(x, 0.0, 1.0)
    else:
        deficit = target - s
        # increase sum
        idx = np.argsort(x)  # smallest first
        for i in idx:
            if deficit <= 0:
                break
            # amount we can increase from x[i] up to 1
            increasable = 1.0 - x[i]
            delta = min(increasable, deficit)
            x[i] += delta
            deficit -= delta
        if deficit > 1e-12:
            x += deficit / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode="full")
    return np.max(v) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def softmax_weights(v, beta):
    vmax = np.max(v)
    w = np.exp(beta * (v - vmax))
    w_sum = np.sum(w)
    if w_sum == 0:
        return np.full_like(v, 1.0 / len(v))
    return w / w_sum

def gradient_softmax(h, beta):
    """Gradient of softmax‑smoothed max‑overlap."""
    n = len(h)
    v = np.correlate(h, 1.0 - h, mode="full")
    w = softmax_weights(v, beta)
    g = np.zeros_like(h)
    offset = n - 1
    for idx, weight in enumerate(w):
        if weight < 1e-16:
            continue
        shift = idx - offset
        if shift >= 0:
            tail = slice(0, n - shift)
            head = slice(shift, n)
        else:
            s = -shift
            tail = slice(s, n)
            head = slice(0, n - s)
        g[tail] += weight * (1.0 - h[head])
        g[head] -= weight * h[tail]
    # keep updates on the feasible hyperplane
    g -= np.mean(g)
    return g, np.max(v) * (2.0 / n)

def optimize_h_adam(h0, budget, seed):
    """Adam optimizer on the soft‑max smoothed objective."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v2 = np.zeros_like(h)
    t_iter = 0
    # smoothing schedule
    beta = 1.0
    beta_max = 200.0
    # learning‑rate schedule
    lr0 = 0.2
    no_improve = 0
    while time.time() - start < budget:
        g, cur_c5 = gradient_softmax(h, beta)
        t_iter += 1
        m = beta1 * m + (1 - beta1) * g
        v2 = beta2 * v2 + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v2 / (1 - beta2 ** t_iter)
        # linear decay of lr
        elapsed = time.time() - start
        lr = lr0 * max(0.01, 1.0 - elapsed / budget)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        # increase smoothing

        # random shake when stuck
        if no_improve >= 5000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            m.fill(0.0)
            v2.fill(0.0)
            t_iter = 0
            beta = 1.0
            no_improve = 0
    return best_h, best_c5
