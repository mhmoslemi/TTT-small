def cem_optimize(h0, n, target_sum, time_budget, rng):
    # parameters
    N_samples = 200   # maybe dynamic based on time
    elite_frac = 0.1
    learning_rate_mu = 0.7
    learning_rate_sigma = 0.7
    min_sigma = 0.01   # avoid collapse

    # initial distribution
    mu = h0.copy()
    sigma = np.full(n, 0.3)    # start wide but less than half range
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    while time.time() - start < time_budget:
        # generate samples
        N = N_samples
        # sample from N(mu, sigma^2)
        samples = rng.normal(loc=mu, scale=sigma, size=(N, n))
        # clip and project
        # Evaluate each
        c5_vals = np.empty(N, dtype=float)
        projected = np.empty((N, n), dtype=float)
        for i in range(N):
            s = np.clip(samples[i], 0.0, 1.0)
            s = project_sum_and_bounds(s, target_sum)
            projected[i] = s
            c5_vals[i] = compute_c5(s)
        # best overall
        min_idx = np.argmin(c5_vals)
        if c5_vals[min_idx] < best_c5:
            best_c5 = c5_vals[min_idx]
            best_h = projected[min_idx].copy()
        # elite selection
        elite_N = max(1, int(N * elite_frac))
        elite_idx = np.argpartition(c5_vals, elite_N)[:elite_N]

        # update mu, sigma using elite
        elite_mu = elite.mean(axis=0)
        elite_sigma = elite.std(axis=0)
        mu = (1 - learning_rate_mu) * mu + learning_rate_mu * elite_mu
        sigma = (1 - learning_rate_sigma) * sigma + learning_rate_sigma * elite_sigma

        # optionally reduce sigma over time
        # early stop if best_c5 < target
        if best_c5 < 0.38:

def gradient_refine(h0, time_budget, rng):
    if time_budget <= 0:
        return h0, compute_c5(h0)
    n = len(h0)
    target = n/2.0
    dx = 2.0/n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 5.0
    max_beta = 50.0
    alpha0 = 0.3
    no_improve = 0
    while time.time() - start < time_budget:
        # compute correlation vector v (without dx)
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # update beta
        if beta < max_beta:
            beta = min(max_beta, beta * 1.01)

        # softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / len(v)
        else:
            w = w / w_sum

        # compute gradient
        g = np.zeros(n)
        offset = n - 1
        for idx in range(len(v)):
            weight = w[idx]
            if weight == 0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # enforce zero sum direction
        g -= g.mean()

        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        # step
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)

        # accept if improves or random
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        # random shake if stuck
        if no_improve > 2000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    final_c5 = compute_c5(best_h)
    return best_h, final_c5

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximation."""
    dx = 2.0 / h.size
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # first global shift
        x += diff / x.size
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # fix remaining deficit/excess greedily
            for i in range(x.size):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def cem_optimize(h0: np.ndarray, n: int, target: float,
                 time_budget: float, rng: np.random.Generator):
    """Cross‑entropy method with diagonal Gaussian."""
    # parameters (tuned for speed vs diversity)
    N_samples = 200
    elite_frac = 0.1
    mu = h0.copy()
    sigma = np.full(n, 0.3)          # moderate spread
    min_sigma = 0.01
    lr_mu = 0.7
    lr_sigma = 0.7
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_budget:
        # generate samples
        samples = rng.normal(loc=mu, scale=sigma, size=(N_samples, n))
        proj = np.empty_like(samples)
        vals = np.empty(N_samples, dtype=float)
        for i in range(N_samples):
            s = np.clip(samples[i], 0.0, 1.0)
            s = project_sum_and_bounds(s, target)
            proj[i] = s
            vals[i] = compute_c5(s)
        # update best
        idx_best = np.argmin(vals)
        if vals[idx_best] < best_c5:
            best_c5 = vals[idx_best]
            best_h = proj[idx_best].copy()
        # elite selection
        elite_n = max(1, int(N_samples * elite_frac))
        elite_idx = np.argpartition(vals, elite_n)[:elite_n]
        elite = proj[elite_idx]
        elite_mu = elite.mean(axis=0)
        elite_sigma = elite.std(axis=0)
        mu = (1 - lr_mu) * mu + lr_mu * elite_mu
        sigma = (1 - lr_sigma) * sigma + lr_sigma * elite_sigma
        sigma = np.maximum(sigma, min_sigma)
        if best_c5 < 0.38:
            break
    return best_h, best_c5
