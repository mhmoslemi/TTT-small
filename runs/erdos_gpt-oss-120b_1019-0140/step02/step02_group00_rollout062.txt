    n = h.shape[0]
    m = 2*n - 1
    # FFT length we can set to next power of 2 for speed?

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the root is bracketed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation of arr to size new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def binary_balanced(n, rng):
    """Return a balanced binary vector of length n (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def binary_from_real(v, n):
    """Top n/2 entries of v become 1, rest 0."""
    half = n // 2
    idx = np.argpartition(v, -half)[-half:]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated via FFT."""
    n = h.shape[0]
    m = 2 * n - 1
    fh = np.fft.fft(h, m)
    f1mh = np.fft.fft(1.0 - h, m)
    corr = np.fft.ifft(fh * f1mh).real
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term w.r.t. h (averaged over all max shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term1[n - shift :] = 0.0
            term2[shift:] = h[: n - shift]
            term2[: shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term1[:s] = 0.0
            term2[: n - s] = h[s:]
            term2[n - s :] = 0.0
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def hill_climb_binary(h0, max_steps, rng, time_left):
    """Greedy hill‑climbing on a balanced binary vector."""
    h = h0.copy()
    curr_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = curr_score
    no_improve = 0
    for step in range(max_steps):
        if time_left() < 0.001:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        if new_score < curr_score:
            curr_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        if no_improve >= 2000:
            break
    return best_h, best_score
