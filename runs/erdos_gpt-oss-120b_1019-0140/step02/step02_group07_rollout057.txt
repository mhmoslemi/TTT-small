def repair_h(x):
    # Clip bounds
    h = np.clip(x, 0.0, 1.0)
    # Adjust sum
    cur = np.sum(h)
    diff = target_sum - cur
    # iterate adjusting free elements
    # free = indices where not at bound (0 or 1)
    # add equally diff/len(free) to those, then clip again; repeat few times
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size == 0:
            # if no free, just distribute to all with some small fudge
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
            continue
        add = diff / free.size
        h[free] += add
        # clip again
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    # final adjust by distributing residual equally
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        h += residual / n
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_hill_climb(start_bin, time_end, rng):
    n = CURRENT_N
    target = CURRENT_TARGET
    h_bin = start_bin.copy()
    cur_c5 = compute_c5(h_bin.astype(float))
    best_c5 = cur_c5
    best_bin = h_bin.copy()
    no_improve = 0
    # ensure sum correct: if sum != target, correct by flipping
    cur_sum = h_bin.sum()
    if cur_sum != target:
        # adjust: if too many ones, turn some to zero; else set zeros to one
        diff = int(target - cur_sum)
        if diff > 0:
            zeros = np.flatnonzero(h_bin == 0)
            choice = rng.choice(zeros, diff, replace=False)
            h_bin[choice] = 1
        elif diff < 0:
            ones = np.flatnonzero(h_bin == 1)
            choice = rng.choice(ones, -diff, replace=False)
            h_bin[choice] = 0
        cur_c5 = compute_c5(h_bin.astype(float))
        best_c5 = cur_c5
        best_bin = h_bin.copy()
    while time.time() < time_end and no_improve < 3000:
        ones = np.flatnonzero(h_bin == 1)
        zeros = np.flatnonzero(h_bin == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_bin[i] = 0
        h_bin[j] = 1
        cand_c5 = compute_c5(h_bin.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_bin = h_bin.copy()
            no_improve = 0
        else:
            # revert swap
            h_bin[i] = 1
            h_bin[j] = 0
            no_improve += 1
    return best_bin.astype(float), best_c5

import time
import numpy as np
from scipy.optimize import differential_evolution

# Global placeholders – will be set in run()
CURRENT_N = None
CURRENT_TARGET = None
CURRENT_BEST_H = None

def compute_c5(arr):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = CURRENT_N
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_h(x):
    """Clip to [0,1] and enforce ∑h = CURRENT_TARGET."""
    n = CURRENT_N
    target = CURRENT_TARGET
    h = np.clip(x, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            continue
        add = diff / free.size
        h[free] += add
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        h += residual / n
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_hill_climb(start_bin, time_end, rng):
    """Swap‑based hill climbing on binary vectors (preserves number of ones)."""
    n = CURRENT_N
    target = CURRENT_TARGET
    h_bin = start_bin.copy()
    # enforce exact number of ones
    cur_sum = h_bin.sum()
    if cur_sum != target:
        diff = int(target - cur_sum)
        if diff > 0:
            zeros = np.flatnonzero(h_bin == 0)
            h_bin[rng.choice(zeros, diff, replace=False)] = 1
        elif diff < 0:
            ones = np.flatnonzero(h_bin == 1)
            h_bin[rng.choice(ones, -diff, replace=False)] = 0
    cur_c5 = compute_c5(h_bin.astype(float))
    best_c5 = cur_c5
    best_bin = h_bin.copy()
    no_improve = 0
    while time.time() < time_end and no_improve < 3000:
        ones = np.flatnonzero(h_bin == 1)
        zeros = np.flatnonzero(h_bin == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_bin[i] = 0
        h_bin[j] = 1
        cand_c5 = compute_c5(h_bin.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_bin = h_bin.copy()
            no_improve = 0
        else:
            # revert swap
            h_bin[i] = 1
            h_bin[j] = 0
            no_improve += 1
    return best_bin.astype(float), best_c5

def local_continuous_descent(h, time_end, rng):
    """Simple pairwise transfer descent preserving the sum."""
    n = CURRENT_N
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    while time.time() < time_end:
        inc_idx = np.where(cur_h < 1.0 - 1e-12)[0]
        dec_idx = np.where(cur_h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1.0 - cur_h[i], cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * 0.5
        cur_h[i] += delta
        cur_h[j] -= delta
        new_c5 = compute_c5(cur_h)
        if new_c5 + 1e-12 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert move
            cur_h[i] -= delta
            cur_h[j] += delta
    return best_h, best_c5

def _de_obj(x):
    """Objective for differential evolution – repaired C5."""
    h = repair_h(x)
    return compute_c5(h)

def de_init(popsize, dim, rnd):
    """Initializer for DE – inject current best as first individual."""
    global CURRENT_BEST_H
    pop = rnd.random((popsize, dim))
    if CURRENT_BEST_H is not None:
        pop[0] = CURRENT_BEST_H
    return pop

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function (array), its C5 bound, and length."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    global CURRENT_N, CURRENT_TARGET, CURRENT_BEST_H
    CURRENT_N = n
    CURRENT_TARGET = n / 2.0
    best_h = repair_h(h0)
    best_c5 = compute_c5(best_h)
    CURRENT_BEST_H = best_h.copy()

    # Bounds for DE
    bounds = [(0.0, 1.0)] * n

    # Determine DE budget (short but enough)
    remaining = budget_s - (time.time() - start_time) - 0.5
    if remaining > 0.5:
        popsize = max(5, min(15, int(remaining * 2)))
        max_iter = max(1, min(30, int(remaining * 5)))
        try:
            result = differential_evolution(
                _de_obj,
                bounds,
                seed=rng.integers(0, 2**32 - 1),
                maxiter=max_iter,
                popsize=popsize,
                polish=False,
                init=de_init,
                disp=False,
            )
            cand_h = repair_h(result.x)
            cand_c5 = compute_c5(cand_h)
            if cand_c5 < best_c5:
                best_h = cand_h
                best_c5 = cand_c5
                CURRENT_BEST_H = best_h.copy()
        except Exception:
            pass

    # Continuous local refinement
    time_end = start_time + budget_s - 0.2
    best_h, best_c5 = local_continuous_descent(best_h, time_end, rng)
    CURRENT_BEST_H = best_h.copy()

    # Optional binary hill‑climb if time permits
    if time.time() < start_time + budget_s - 0.3:
        bin_start = (best_h >= 0.5).astype(int)
        final_h_bin, final_c5_bin = binary_hill_climb(
            bin_start, start_time + budget_s - 0.1, rng
        )
        final_h = repair_h(final_h_bin)
        final_c5 = compute_c5(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    # Final safety clip & exact‑sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = CURRENT_TARGET - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = CURRENT_TARGET - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1.0 - best_h[i])
            best_h[i] += adj
            residual = CURRENT_TARGET - np.sum(best_h)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
