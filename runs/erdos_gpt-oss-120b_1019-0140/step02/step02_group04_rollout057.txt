import time, numpy as np

def compute_c5_vec(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # distribute uniformly
    vec += diff / len(vec)
    # clip again
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if abs(diff) < 1e-12:
        return vec
    # fix remaining via greedy
    if diff > 0:
        # need to increase some values
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec

def mismatch_indices(h, k, n):
    if k >= 0:
        i_range = np.arange(0, n - k)
        hi = h[i_range]
        hj = h[i_range + k]
    else:
        k_abs = -k
        i_range = np.arange(k_abs, n)
        hi = h[i_range]
        hj = h[i_range - k_abs]
    mask_10 = (hi == 1) & (hj == 0)
    mask_01 = (hi == 0) & (hj == 1)
    idx_i = i_range[mask_10]
    idx_j = i_range[mask_01]
    return idx_i, idx_j

def binary_greedy(h, dx, target_sum, start, budget, rng):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5_vec(best_h.astype(float), dx)
    # allocate time fraction
    while time.time() - start < budget * 0.5:
        cur_val = compute_c5_vec(h.astype(float), dx)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= 0.3808:
                break
        # compute worst shift
        corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        idx_i, idx_j = mismatch_indices(h, k, n)
        if idx_i.size and idx_j.size:
            i = rng.choice(idx_i)
            j = rng.choice(idx_j)
            h[i], h[j] = h[j], h[i]
            continue
        # fallback random swap attempts
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        improved = False
        for _ in range(10):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = h[j], h[i]
            new_val = compute_c5_vec(h.astype(float), dx)
            if new_val < cur_val - 1e-12:
                improved = True
                break
            else:
                # revert
                h[i], h[j] = h[j], h[i]
        if not improved:
            # occasional random reset to avoid deadlock
            if rng.random() < 0.005:
                h = np.zeros(n, dtype=int)
                h[rng.choice(n, size=target_sum, replace=False)] = 1
    return best_h, best_val, h

def simulated_annealing(h, dx, target_sum, start, budget, rng):
    n = len(h)
    cur_h = h.astype(float)
    cur_val = compute_c5_vec(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    T0 = 0.02
    Tmin = 1e-6

        frac = elapsed / (budget * 0.9)
        T = max(Tmin, T0 * (1.0 - frac))
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(0.5, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:

import time
import numpy as np

def compute_c5_vec(h, dx):
    """Maximum overlap integral C₅ for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target_sum):
    """Clamp to [0,1] and adjust to satisfy exact sum constraint."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if abs(diff) < 1e-12:
        return vec
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec

def mismatch_indices(h, k, n):
    """Indices i where h[i]==1, h[i+k]==0 and vice‑versa for shift k."""
    if k >= 0:
        i_range = np.arange(0, n - k)
        hi = h[i_range]
        hj = h[i_range + k]
    else:
        k_abs = -k
        i_range = np.arange(k_abs, n)
        hi = h[i_range]
        hj = h[i_range - k_abs]
    mask_10 = (hi == 1) & (hj == 0)
    mask_01 = (hi == 0) & (hj == 1)
    idx_i = i_range[mask_10]
    idx_j = i_range[mask_01]
    return idx_i, idx_j

def binary_greedy(h, dx, target_sum, start, budget, rng):
    """Targeted swap hill‑climbing on a binary mask."""
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5_vec(best_h.astype(float), dx)
    while time.time() - start < budget * 0.5:
        cur_val = compute_c5_vec(h.astype(float), dx)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= 0.3808:
                break
        corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
        k = np.argmax(corr) - (n - 1)
        idx_i, idx_j = mismatch_indices(h, k, n)
        if idx_i.size and idx_j.size:
            i = rng.choice(idx_i)
            j = rng.choice(idx_j)
            h[i], h[j] = h[j], h[i]
            new_val = compute_c5_vec(h.astype(float), dx)
            if new_val <= cur_val + 1e-12:
                continue
            h[i], h[j] = h[j], h[i]  # revert if worsened
        else:
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
            improved = False
            for _ in range(10):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i], h[j] = h[j], h[i]
                new_val = compute_c5_vec(h.astype(float), dx)
                if new_val < cur_val - 1e-12:
                    improved = True
                    break
                h[i], h[j] = h[j], h[i]
            if not improved and rng.random() < 0.005:
                h = np.zeros(n, dtype=int)
                h[rng.choice(n, size=target_sum, replace=False)] = 1
    return best_h, best_val, h

def simulated_annealing(h, dx, target_sum, start, budget, rng):
    """Fractional mass‑transfer annealing starting from a binary mask."""
    cur_h = h.astype(float)
    cur_val = compute_c5_vec(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    T0 = 0.02
    Tmin = 1e-6
    while time.time() - start < budget * 0.9:

        T = max(Tmin, T0 * (1.0 - frac))

        cand[j] += delta
