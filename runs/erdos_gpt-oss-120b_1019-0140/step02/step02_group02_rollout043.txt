def continuous_anneal(h_start, best_val, start_time, deadline, rng):
    h = h_start.copy()
    cur_val = best_val
    best_h = h.copy()
    best = best_val
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        if best < 0.3808:
            break
    return best_h, best

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _build_shift_mat(h):
    n = h.shape[0]
    L = 2 * n - 1
    mat = np.zeros((n, L), dtype=int)
    for i in range(n):
        off = n - 1 - i
        mat[i, off:off + n] = h
    return mat

def _greedy_improve(h, deadline):
    n = h.shape[0]
    L = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = corr.max()
    target_corr = int(np.floor(0.3808 * n / 2 + 1e-12))
    while time.time() < deadline - 0.05:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        shift_mat = _build_shift_mat(h)
        best_new = max_corr
        best_i = best_j = None
        best_delta = None
        for i in ones:

                delta = row_i - shift_mat[j]
                new_max = np.max(corr + delta)

        if best_i is None:

    return h, max_corr, corr

    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i = i_range[np.argmax(prod)]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            remaining = deadline - (time.time() - start_time)
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        if best < 0.3808:
            break
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
    else:
        n = 80
        init_arr = None

    target = n // 2
    best_h = None
    best_c5 = np.inf

    # start from provided initial if any
    if init_arr is not None:
        h0 = _to_binary(init_arr, target, rng)
        h_g, max_corr, _ = _greedy_improve(h0.copy(), deadline)
        c5 = max_corr * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_g.astype(float)

    # random restarts
    while time.time() < deadline - 5.0:
        h0 = _random_binary(n, target, rng)
        h_g, max_corr, _ = _greedy_improve(h0.copy(), deadline)
        c5 = max_corr * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_g.astype(float)
            if best_c5 <= 0.3808:
                break

    # final continuous refinement if time permits
    if best_h is not None and time.time() < deadline - 0.5:
        val = compute_c5(best_h)
        best_h, best_c5 = _continuous_anneal(best_h, val, start, deadline, rng)

    # ensure feasibility
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[:target] = 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    target_sum = n / 2.0
    if abs(cur_sum - target_sum) > 1e-9:
        best_h += (target_sum - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h.astype(float), float(final_c5), n
