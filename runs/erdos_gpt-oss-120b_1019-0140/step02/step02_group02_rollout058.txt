def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine n based on presence of initial_h_values
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = init.shape[0]
        # Choose n = n0 * 2 if even? We'll double if n0 <= 100.
        if n0 < 120:
            factor = 2
        else:
            factor = 1
        n = n0 * factor
        if factor > 1:
            h = np.repeat(init, factor)
        else:
            h = init.copy()
    else:
        n = 120
        # generate random binary with sum n/2
        h = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0

    # Ensure constraints
    target_sum = n / 2.0
    # Clip and adjust sum
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Simple scaling: add uniform delta to free indices
        delta = (target_sum - cur_sum) / n
        h = np.clip(h + delta, 0.0, 1.0)
        # final correction by adjusting random free entries
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        attempts = 0
        while abs(diff) > 1e-12 and attempts < 1000:
            i = rng.integers(n)
            # adjust within bounds
            max_inc = 1.0 - h[i] if diff > 0 else h[i]
            inc = np.clip(diff, -max_inc, max_inc)
            h[i] += inc
            diff = target_sum - h.sum()
            attempts += 1

    best_h = h.copy()
    best_val = compute_c5(best_h)

    # Helper: projection
    def project(v):
        # Clip, then adjust sum
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        target = target_sum
        # Use simple iterative water-filling
        max_iter = 100
        for _ in range(max_iter):
            diff = target - s
            if abs(diff) < 1e-12:
                break
            # Identify free indices (not at bounds)
            free = np.where((v > 1e-12) & (v < 1 - 1e-12))[0]
            if free.size == 0:
                # distribute to all, ignoring bounds (rare)
                free = np.arange(len(v))
            add = diff / free.size
            # Add and clip again
            v[free] += add
            v = np.clip(v, 0.0, 1.0)
            s = v.sum()
        return v

    # Optimize loop
    iteration = 0
    while time.time() < deadline - 0.5:
        # Compute overlap correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_idx = np.argmax(corr)
        # shift k: index offset from -n+1
        k = k_idx - (n - 1)
        # If k == 0, ignore (should be zero), pick second max
        if k == 0:
            # find second largest
            sorted_idx = np.argsort(corr)[::-1]
            for idx in sorted_idx:
                k_test = idx - (n-1)
                if k_test != 0:
                    k = k_test
                    k_idx = idx
                    break
        # compute contributions p_i for this shift
        if k >= 0:
            i_range = np.arange(n - k)
            p = h[i_range] * (1.0 - h[i_range + k])
        else:
            kk = -k
            i_range = np.arange(kk, n)
            p = h[i_range] * (1.0 - h[i_range - kk])
        # choose i_max
        i_local = np.argmax(p)
        i = i_range[i_local]
        # choose partner j
        # candidate indices with h < 1
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0:
            cand = np.where(h > 1e-12)[0]  # fallback
        # Prefer indices where p for shift k is small
        if k >= 0:
            p_all = h * (1.0 - np.concatenate([h[k:], np.zeros(k)]))
        else:
            kk = -k
            p_all = h * (1.0 - np.concatenate([np.zeros(kk), h[:n-kk]]))
        # Among candidates, choose those with small p
        cand_p = p_all[cand]
        # choose j with probability weighted toward small p
        # Inverse weights
        inv_weights = 1.0 / (cand_p + 1e-6)
        prob = inv_weights / inv_weights.sum()
        j = rng.choice(cand, p=prob)

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            # just skip
            iteration += 1
            continue
        # propose delta
        delta = rng.random() * max_transfer
        # temperature schedule
        elapsed = time.time() - start
        remaining = deadline - start - elapsed
        # simple linear temperature
        T = remaining / (deadline - start)  # ratio
        # apply annealing acceptance
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # project to ensure constraints (should already be satisfied)
        # but keep clamped
        val_new = compute_c5(h_new)
        cur_val = compute_c5(h)
        # If improvement accept
        if val_new < cur_val:
            h = h_new
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()
                # early exit if target reached
                if best_val < 0.3808:
                    break
        else:
            # Accept with probability exp(-(val_new-cur_val)/T_scaled)
            # Scale temperature: maybe multiply by typical C5 magnitude
            # Use T_scaled = 0.01 * T?
            T_scaled = 0.005 + 0.02 * T  # small positive
            if np.exp(-(val_new - cur_val) / T_scaled) > rng.random():
                h = h_new
        # occasional gradient step
        if iteration % 25 == 0:
            # gradient descent on worst shift
            # compute gradient
            grad = np.zeros_like(h)
            if k >= 0:
                # i from 0 to n-k-1
                grad[: n - k] += 1.0 - h[k:]  # derivative from term h_i*(1-h_{i+k})
                grad[k:] -= h[: n - k]  # derivative from term -h_{i-k}*h_i? Wait earlier expression was -h_{i-k}
                # Actually for indices where i+k exists, gradient includes -(h_{i-k}) for i where i-k >=0 => i>=k
                # So implement:
                # For i>=k: grad[i] -= h[i - k]
                grad[k:] -= h[: n - k]
            else:
                kk = -k
                # For i from kk to n-1: grad[i] += 1 - h[i - kk]
                grad[kk:] += 1.0 - h[: n - kk]
                # For i from 0 to n-kk-1: grad[i] -= h[i + kk]
                grad[: n - kk] -= h[kk:]
            # gradient step size
            eta = 0.1 * (remaining / (deadline - start))  # decreasing
            h_candidate = h - eta * grad
            h_candidate = project(h_candidate)
            val_candidate = compute_c5(h_candidate)
            if val_candidate < cur_val:
                h = h_candidate
                if val_candidate < best_val:
                    best_val = val_candidate
                    best_h = h_candidate.copy()
                    if best_val < 0.3808:
                        break
        iteration += 1

    # Final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce sum exactly
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-9:
        # distribute diff across non-bound indices
        free = np.where((best_h > 1e-12) & (best_h < 1 - 1e-12))[0]
        if free.size == 0:
            free = np.arange(len(best_h))
        # distribute equally
        best_h[free] += diff / free.size
        best_h = np.clip(best_h, 0.0, 1.0)
    final_val = compute_c5(best_h)

    best_h_local = h.copy()

    n = len(h)

    # define project function uses n.

    # main loop:
    iteration = 0
    while time.time() < deadline - 0.1:
        # compute correlation and find worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)
        # if k==0, get second max
        if k == 0:
            # maybe find next
            sorted_idx = np.argsort(corr)[::-1]
            for idx in sorted_idx:
                k_test = idx - (n - 1)
                if k_test != 0:
                    k = k_test
                    break

        # compute contributions p for this shift
        if k >= 0:
            i_range = np.arange(n - k)
            p = h[i_range] * (1.0 - h[i_range + k])
        else:
            kk = -k
            i_range = np.arange(kk, n)
            p = h[i_range] * (1.0 - h[i_range - kk])
        # choose i_max
        i_local = np.argmax(p)
        i = i_range[i_local]

        # choose candidate j: among indices not at bound 1
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0:

        # compute p_all for shift k for weighting
        if k >= 0:

            h_shift = np.concatenate([h[k:], np.zeros(k)])

        inv_weights = 1.0 / (cand_p + 1e-12)
        prob = inv_weights / inv_weights.sum()

        h_new = h.copy()

        val_new = compute_c5(h_new)

import time

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    """Random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v / v.sum() * target
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    for i in rng.permutation(n):
        if abs(diff) < 1e-12:
            break
        max_inc = (1.0 - v[i]) if diff > 0 else v[i]
        inc = np.clip(diff, -max_inc, max_inc)
        v[i] += inc
        diff = target - v.sum()
    return v

def solve_lp_relaxation(n):
    """Linear‑programming relaxation using McCormick envelopes."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    h = cp.Variable(n)
    t = cp.Variable()
    num_pairs = n * (n - 1) // 2
    y = cp.Variable(num_pairs)
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2.0,
        t >= 0,
    ]
    idx = 0
    shift_idx = [[] for _ in range(n)]  # 0‑unused
    for k in range(1, n):
        for i in range(n - k):
            yi = y[idx]
            constraints += [
                yi <= h[i],
                yi <= h[i + k],
                yi >= h[i] + h[i + k] - 1,
                yi >= 0,
            ]
            shift_idx[k].append(idx)
            idx += 1
        # after building this shift we will add the overlap constraint later
    for k in range(1, n):
        if shift_idx[k]:
            constraints.append(cp.sum(cp.hstack([y[j] for j in shift_idx[k]]))
                               >= n / 2.0 - t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK, glpk={'msg_lev': 'GLP_MSG_OFF'})
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            return None, None
    if h.value is None:
        return None, None
    h_opt = np.array(h.value, dtype=float)
    t_val = t.value
    return h_opt, t_val
