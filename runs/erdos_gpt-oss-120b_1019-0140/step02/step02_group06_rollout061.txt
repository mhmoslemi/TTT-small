def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def initial_binary(n):
    # generate random binary of exactly n/2 ones
    arr = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, n//2, replace=False)
    arr[ones] = 1.0
    return arr

def binary_from_continuous(x):

    arr = np.zeros(n, dtype=np.float64)
    arr[idx] = 1.0
    return arr

def simulated_annealing(h0, time_budget, rng):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    start = time.time()
    if best_c5 > 0:
        T0 = best_c5 * 0.1
    else:
        T0 = 0.01
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(1e-8, T0 * (1.0 - elapsed / time_budget))
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        # pick indices
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        cand_c5 = compute_c5(h)
        delta = cand_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            curr_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # optional early stop: if best_c5 < desired threshold (0.3808) we can break
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

    start = time.time()

    n = len(h)
    while time.time() - start < time_budget:
        # if stuck for too long, break
        if no_improve > 20000:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h)
        if c5 < best_local_c5:
            best_local_c5 = c5

            no_improve = 0
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return best_local_h, best_local_c5

def gradient_descent_smoothing(h0, time_budget, rng):
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.2
    no_improve = 0
    while time.time() - start < time_budget:
        # compute correlation vector v
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # update beta
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        # compute softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / len(v)
        else:
            w /= w_sum

        # gradient
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            # h[head] part multiplied by -weight, h[tail] part +weight*(1-h[head])
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # zero-sum direction
        g -= np.mean(g)

        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        # update h
        h_new = h - alpha * g
        # project
        h_new = project_sum_and_bounds(h_new, target)
        # compute c5 for new
        c5_new = compute_c5(h_new)
        # accept if improvement or random small prob
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 3000:
            # random perturbation
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

        # early exit if best_c5 already good
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

def project_sum_and_bounds(x, target):
    # target is sum needed (n/2)
    x = np.clip(x, 0.0, 1.0)
    # simple heuristic: adjust globally
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # scale to meet target
        # compute adjustment per element
        x += diff / len(x)
        # clip again and adjust iteratively
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # if still not met, fix by greedy
        if abs(diff) > 1e-12:
            # adjust elements
            # if diff>0, increase some below 1; else decrease some above 0
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def binary_from_continuous(x):
    n = len(x)
    half = n // 2
    arr = np.zeros(n, dtype=np.float64)
    if half > 0:
        idx = np.argpartition(x, -half)[-half:]
        arr[idx] = 1.0
    return arr

def random_binary(n, rng):
    arr = np.zeros(n, dtype=np.float64)
    half = n // 2
    if half > 0:
        idx = rng.choice(n, half, replace=False)
        arr[idx] = 1.0
    return arr

def hill_climb_binary(h0, time_budget, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    start = time.time()
    no_improve = 0
    while time.time() - start < time_budget:
        if no_improve > 20000:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return best_h, best_c5

def gradient_descent_smoothing(h0, time_budget, rng):
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.2
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / len(v)
        else:
            w /= w_sum
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue

            else:

        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)

            h = h_new

            no_improve = 0

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, float(final_c5), n
