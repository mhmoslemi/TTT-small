def generate_random_feasible_v(target_sum, rng):
    x = rng.random(5)
    s = np.sum(x)
    if s == 0:
        x = np.full(5, target_sum / 5)
    else:
        x = x * (target_sum / s)
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # adjust residual sum similar to earlier function
    diff = target_sum - np.sum(x)
    # if diff positive, add to entries <1
    if diff > 1e-12:
        for i in np.where(x < 1)[0]:
            inc = min(diff, 1 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < -1e-12:
        for i in np.where(x > 0)[0]:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

import time
import numpy as np

def compute_c5_vec(h, dx):
    """Return max overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target_sum):
    """Adjust vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # distribute uniformly
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    # greedily fix any remaining discrepancy
    for i in range(len(vec)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine number of points
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 80  # default size (divisible by 5)

    # Ensure n is a multiple of 5 for the simple construction
    if n % 5 != 0:
        n = 80  # fallback to safe size
    block_len = n // 5
    target_sum = n / 2.0
    dx = 2.0 / n

    # Simple monotone block construction: values linearly spaced from 0 to 1
    v = np.linspace(0.0, 1.0, 5)          # [0, 0.25, 0.5, 0.75, 1]
    h = np.repeat(v, block_len)[:n]      # ensure exact length n
    h = project_to_sum(h, target_sum)    # enforce exact integral

    best_h = h
    best_val = compute_c5_vec(best_h, dx)

    # (optional) tiny local search to polish solution within remaining time
    time_left = budget_s - (time.time() - start_time)
    if time_left > 2.0:
        # simple coordinate descent on random indices
        cur_h = best_h.copy()
        cur_val = best_val
        step = 0.02
        while time.time() - start_time < budget_s - 0.5:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # propose moving mass from i to j
            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = rng.uniform(0.0, max_delta)
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = project_to_sum(cand, target_sum)
            cand_val = compute_c5_vec(cand, dx)
            if cand_val < cur_val - 1e-12:
                cur_h, cur_val = cand, cand_val
                step = min(step * 1.1, 0.5)
            else:
                step *= 0.99
        if cur_val < best_val:
            best_h, best_val = cur_h, cur_val

    return best_h, best_val, n
