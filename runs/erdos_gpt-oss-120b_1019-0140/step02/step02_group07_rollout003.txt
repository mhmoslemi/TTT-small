import time
import numpy as np
from scipy.optimize import differential_evolution

# ----- helper functions -------------------------------------------------

def compute_c5(arr):
    """Approximate C5 via full correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * (2.0 / n)


def repair_sum_bounds(x, target_sum):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    x = np.clip(x, 0.0, 1.0)
    residual = target_sum - np.sum(x)
    if residual > 1e-12:
        for i in range(len(x)):
            if residual <= 0:
                break
            avail = 1.0 - x[i]
            if avail > 0:
                delta = min(avail, residual)
                x[i] += delta
                residual -= delta
    elif residual < -1e-12:
        for i in range(len(x)):
            if residual >= 0:
                break
            avail = x[i]
            if avail > 0:
                delta = max(-avail, residual)   # residual is negative
                x[i] += delta
                residual -= delta
    return x


def objective_func(x, n, target_sum):
    """Objective for DE: repaired C5."""
    x = repair_sum_bounds(np.array(x, copy=False), target_sum)
    return compute_c5(x)


def de_callback(xk, convergence):
    """Stop DE when time is up or sufficiently good solution found."""
    global G_DE_END_TIME, G_TARGET_SUM, G_TARGET_CUTOFF, G_BEST_H, G_BEST_C5
    if time.time() >= G_DE_END_TIME:
        return True
    val = compute_c5(repair_sum_bounds(xk, G_TARGET_SUM))
    if val <= G_TARGET_CUTOFF:
        G_BEST_H = repair_sum_bounds(xk, G_TARGET_SUM)
        G_BEST_C5 = val
        return True
    return False


def thue_morse_seq(n):
    """Binary Thue‑Morse sequence."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                       dtype=float, count=n)

    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return ((a + 1) // 2).astype(float)


def local_search(start_h, start_c5, deadline, rng):
    """Simple greedy hill‑climbing with occasional random perturbations."""
    n = len(start_h)
    target = n / 2.0
    best_h = start_h.copy()
    best_c5 = start_c5
    no_improve = 0
    max_no_improve = 5000

    while time.time() < deadline:
        inc_idx = np.where(best_h < 1.0 - 1e-12)[0]
        dec_idx = np.where(best_h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1.0 - best_h[i], best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        cand = best_h.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best_h, best_c5 = cand, cand_c5
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # random perturbation preserving the sum
            noise = rng.normal(scale=0.02, size=n)
            cand = best_h + noise
            cand = repair_sum_bounds(cand, target)
            cand_c5 = compute_c5(cand)
            if cand_c5 + 1e-12 < best_c5:
                best_h, best_c5 = cand, cand_c5
            no_improve = 0

    return best_h, best_c5


# ----- main entry point ------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- determine problem size and initial candidates -----
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 80
    target_sum = n / 2.0

    candidates = []

    # 1. existing construction
    if "initial_h_values" in globals():
        candidates.append(np.array(initial_h_values, dtype=float))

    # 2. Thue‑Morse
    candidates.append(thue_morse_seq(n))

    # 3. Rudin‑Shapiro
    candidates.append(rudin_shapiro_seq(n))

    # 4. Random binary with exact half weight
    rand_bin = np.zeros(n, dtype=float)
    rand_idx = rng.choice(n, n // 2, replace=False)
    rand_bin[rand_idx] = 1.0
    candidates.append(rand_bin)

    # 5. Fully random real vector
    candidates.append(rng.random(n))

    # pick the best of the seeds
    best_h = None
    best_c5 = np.inf
    for cand in candidates:
        cand = repair_sum_bounds(cand, target_sum)
        c5 = compute_c5(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()

    # ----- global variables for DE callback -----
    global G_DE_END_TIME, G_TARGET_SUM, G_TARGET_CUTOFF, G_BEST_H, G_BEST_C5
    G_TARGET_SUM = target_sum
    G_TARGET_CUTOFF = 0.3808
    G_BEST_H = best_h.copy()
    G_BEST_C5 = best_c5

    # ----- DE phase (time‑limited) -----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    de_time = min(remaining * 0.6, 300.0)          # at most 5 min for DE
    G_DE_END_TIME = start_time + de_time

    bounds = [(0.0, 1.0)] * n
    try:
        result = differential_evolution(
            func=objective_func,
            bounds=bounds,
            args=(n, target_sum),
            strategy="best1bin",
            maxiter=2000,
            popsize=5,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=seed,
            callback=de_callback,
            polish=False,
            init="latinhypercube",
            updating="deferred",
        )
    except Exception:
        result = None

    # use DE result if available and better
    if result is not None and result.success:
        cand = repair_sum_bounds(result.x, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5
            G_BEST_H, G_BEST_C5 = best_h.copy(), best_c5

    # also consider any improvement captured by the callback
    if G_BEST_C5 < best_c5:
        best_h, best_c5 = G_BEST_H.copy(), G_BEST_C5

    # early exit if target already achieved
    if best_c5 <= G_TARGET_CUTOFF:
        best_h = repair_sum_bounds(best_h, target_sum)
        return best_h, best_c5, n

    # ----- local refinement (remaining time) -----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    local_time = max(0.0, remaining - 0.5)   # keep a small safety margin
    if local_time > 0.0:
        deadline = time.time() + local_time
        best_h, best_c5 = local_search(best_h, best_c5, deadline, rng)

    # final safety clipping and exact‑sum enforcement
    best_h = repair_sum_bounds(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
