def shift_array(arr, k):
    # shift by k: positive k shifts left (i.e., arr[i+k] -> position i)
    n = len(arr)
    if k >= 0:
        out = np.empty_like(arr)
        out[:n-k] = arr[k:]
        out[n-k:] = 0.0
    else:
        kk = -k
        out = np.empty_like(arr)
        out[kk:] = arr[:n-kk]
        out[:kk] = 0.0
    return out

def gradient_refine(h, time_left, start_time):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alpha = 0.1
    while time.time() - start_time < time_left:
        # Compute correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Mask zero shift
        mid = n - 1
        corr[mid] = -np.inf
        # find k_star
        idx = np.argmax(corr)
        k_star = idx - mid
        # compute shifted arrays
        fwd = shift_array(best_h, k_star)
        bwd = shift_array(best_h, -k_star)
        grad = dx * ((1 - fwd) - bwd)
        # Step
        cand = best_h - alpha * grad
        cand = project_to_box_sum(cand, n/2)
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5 - 1e-12:
            best_h, best_c5 = cand, cand_c5
            alpha *= 1.1  # increase step if progress
            # early termination if target reached
            if best_c5 < 0.3808:
                break
        else:
            alpha *= 0.5
            # if alpha gets too small, break early to avoid endless loops
            if alpha < 1e-6:
                break
    return best_h, best_c5
```

def shift_array(arr, k):
    n = len(arr)
    if k >= 0:
        out = np.empty_like(arr)
        out[:n-k] = arr[k:]
        out[n-k:] = 0.0
    else:
        kk = -k
        out = np.empty_like(arr)
        out[kk:] = arr[:n-kk]
        out[:kk] = 0.0
    return out
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper utilities ----------
def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0, iters=60):
    """Project onto [lo,hi]^n while fixing sum to target using a binary search on λ."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(iters):
        mid = 0.5 * (low + high)
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    lam = 0.5 * (low + high)
    return np.clip(v + lam, lo, hi)

# ---------- Constraint objects for SLSQP ----------
class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=1200):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False}
    )
    return res

# ---------- Shift helper ----------
def shift_array(arr, k):
    """Shift arr by k (positive => left). Out‑of‑range entries become 0."""
    n = arr.shape[0]
    out = np.empty_like(arr)
    if k >= 0:
        out[:n - k] = arr[k:]
        out[n - k:] = 0.0
    else:
        kk = -k
        out[kk:] = arr[:n - kk]
        out[:kk] = 0.0
    return out

# ---------- Gradient‑based refinement ----------
def gradient_refine(h, start_time, budget_s, target_sum):
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alpha = 0.1
    no_improve = 0
    rng = np.random.default_rng(int(time.time()))
    while time.time() - start_time < budget_s - 0.5:
        # correlation and active shift
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        mid = n - 1
        corr[mid] = -np.inf  # exclude zero shift if it is not the max
        idx = int(np.argmax(corr))
        k_star = idx - mid

        # gradient of the active overlap
        fwd = shift_array(best_h, k_star)
        bwd = shift_array(best_h, -k_star)
        grad = dx * ((1.0 - fwd) - bwd)

        cand = best_h - alpha * grad
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)

        if cand_c5 + 1e-12 < best_c5:
            best_h, best_c5 = cand, cand_c5
            alpha = min(alpha * 1.2, 1.0)
            no_improve = 0
            if best_c5 < 0.3808:
                break
        else:
            alpha *= 0.5
            no_improve += 1

        # occasional random perturbation to escape stagnation
        if no_improve >= 12:
            d = rng.normal(size=n)
            d -= d.mean()
            step = 0.05 * (rng.random() ** 2)
            cand = best_h + step * d
            cand = project_to_box_sum(cand, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_h, best_c5 = cand, cand_c5
                alpha = 0.1
                no_improve = 0
            else:
                no_improve = 0
                alpha *= 0.7
    return best_h, best_c5

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Initialise h ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---- Prepare constraints for SLSQP ----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # ---- Phase 1: Repeated SLSQP restarts (≈60% budget) ----
    phase1_deadline = start_time + budget_s * 0.6
    while time.time() < phase1_deadline:
        # Random feasible start
        h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, target_sum)
        t_start = compute_c5(h_start)

        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=1200)
        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)

        if c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break

    # ---- Phase 2: Gradient‑based subgradient refinement ----
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5.0 and best_c5 > 0.3808:
        best_h, best_c5 = gradient_refine(best_h, start_time, budget_s, target_sum)

    # ---- Final projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
