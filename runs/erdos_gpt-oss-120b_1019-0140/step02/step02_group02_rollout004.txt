def project_to_bounds_sum(v, target=..., max_iter=100, tol=1e-9):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - mid, 0.0, 1.0)

def project_to_capped_simplex(v, target):
    # Bisection to solve sum clip(v - lam, 0, 1) = target
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) < 1e-12:
            return w
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - mid, 0.0, 1.0)

def max_shift_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    # stable softmax
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / exp_vals.sum()

    grad = np.full_like(h, dx)  # base term sum(w)=1 => dx each.

    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the set {x | sum(x)=target, 0<=x_i<=1}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0          # enough to push all entries to 1
    hi = np.max(v)                # enough to push all entries to 0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) < 1e-12:
            return w
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - mid, 0.0, 1.0)

def max_shift_and_grad(h: np.ndarray):
    """
    Compute the current maximal overlap value and a subgradient of that value.
    Returns (max_val, gradient, shift_k).
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    max_val = corr[max_idx] * dx
    k = max_idx - (n - 1)       # shift causing the max
    # gradient of the active shift
    grad = np.full_like(h, dx)  # base contribution from the linear term
    if k >= 0:
        grad[: n - k] -= dx * h[k:]
        grad[k:]      -= dx * h[: n - k]
    else:
        kk = -k
        grad[kk:]     -= dx * h[: n - kk]
        grad[: n - kk] -= dx * h[kk:]
    return max_val, grad, k

def random_start_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random feasible vector (sum = n/2, 0<=h_i<=1)."""
    target = n / 2.0
    v = rng.random(n)                     # uniform in [0,1)
    v = np.clip(v, 0.0, 1.0)
    # shift to hit target sum roughly
    v += (target - v.sum()) / n
    v = np.clip(v, 0.0, 1.0)
    # fix any tiny residual
    residual = target - v.sum()
    it = 0
    while abs(residual) > 1e-12 and it < 1000:
        idx = rng.integers(0, n)
        if residual > 0:
            inc = min(residual, 1.0 - v[idx])
            v[idx] += inc
            residual -= inc
        else:
            dec = min(-residual, v[idx])
            v[idx] -= dec
            residual += dec
        it += 1
    return v

def local_optimize(h_start: np.ndarray, target_sum: float,
                   deadline: float, rng: np.random.Generator):
    """
    Simple projected (sub)gradient descent with backtracking and occasional noise.
    Stops when time runs out or a satisfactory bound is found.
    """
    n = h_start.shape[0]
    h = project_to_capped_simplex(h_start, target_sum)
    best_h = h.copy()
    best_val = compute_c5(best_h)

    step = 1.0           # initial step size
    min_step = 1e-5
    no_improve = 0

    while time.time() < deadline - 0.05:
        cur_val, grad, _ = max_shift_and_grad(h)

        # backtracking line search for an improvement in the true max value
        step_local = step
        improved = False
        while step_local > min_step:
            h_candidate = project_to_capped_simplex(h - step_local * grad, target_sum)
            cand_val = compute_c5(h_candidate)
            if cand_val < cur_val - 1e-12:
                improved = True
                break
            step_local *= 0.5

        if not improved:
            # try a small random perturbation to escape stagnation
            noise = rng.normal(scale=0.02, size=n)
            h_candidate = project_to_capped_simplex(h + noise, target_sum)
            cand_val = compute_c5(h_candidate)
            if cand_val < cur_val - 1e-12:
                improved = True

        if not improved:
            # no progress; shrink step and possibly give up
            step = max(step * 0.5, min_step)
            no_improve += 1
            if no_improve >= 60:
                break
            continue

        # accept the improvement
        h = h_candidate
        if cand_val < best_val:
            best_val = cand_val
            best_h = h.copy()
            if best_val < 0.3808:
                return best_h, best_val

        # adapt step size
        if cand_val < cur_val:
            step = min(step * 1.2, 2.0)
        else:
            step = max(step * 0.7, min_step)

        no_improve = 0

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # --- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_capped_simplex(h0, target)
    else:
        n = 80
        target = n / 2.0
        h0 = random_start_vector(n, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # --- first local optimisation from the provided (or random) start ---
    if time.time() < deadline - 5.0:
        h_opt, val_opt = local_optimize(best_h, target, deadline, rng)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt
            if best_val < 0.3808:
                final = compute_c5(best_h)
                return best_h.astype(float), float(final), n

    # --- multiple random restarts ---------------------------------------
    restart_limit = 30
    for _ in range(restart_limit):
        if time.time() >= deadline - 4.0:
            break
        h_start = random_start_vector(n, rng)
        sub_deadline = time.time() + min(30.0, deadline - time.time() - 2.0)
        h_opt, val_opt = local_optimize(h_start, target, sub_deadline, rng)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt
            if best_val < 0.3808:
                break

    # --- final safety projection and value --------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    total = best_h.sum()
    if abs(total - target) > 1e-9:
        best_h = project_to_capped_simplex(best_h, target)

    final_val = compute_c5(best_h)
    return best_h.astype(float), float(final_val), n
