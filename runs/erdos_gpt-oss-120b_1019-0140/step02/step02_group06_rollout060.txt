def project_to_feasible(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Compute current sum
    cur = x.sum()
    diff = target - cur
    # If diff is within tolerance, done
    if abs(diff) < 1e-10:
        return x
    # Now we need to adjust values while staying in [0,1].
    # Use a simple iterative approach: sort indices by (if diff>0) increasing x to fill; else decreasing x to reduce.
    # but diff small relative to n typically.
    # Approach: If diff>0, we need to increase total sum by diff. We can increase values that are not at 1.
    # Compute available capacity: caps = 1 - x
    # Similarly for diff<0: reduce values not at 0.
    # We'll distribute diff proportionally to capacities.
    # Use while loop to allocate.
    if diff > 0:
        # capacities
        caps = 1.0 - x
        total_cap = caps.sum()
        # If diff > total_cap due to rounding, set to total_cap.
        diff = min(diff, total_cap)
        # Proportional allocation
        while diff > 1e-12:
            caps = 1.0 - x
            total_cap = caps.sum()
            if total_cap < 1e-12:
                break
            # allocate proportionally but limited
            inc = diff * (caps / total_cap)
            # ensure not exceed caps
            inc = np.minimum(inc, caps)
            x += inc
            diff = target - x.sum()
    else:
        # diff < 0: need to reduce sum
        caps = x # amount we can reduce
        total_cap = caps.sum()
        diff = max(diff, -total_cap)
        while diff < -1e-12:
            caps = x
            total_cap = caps.sum()
            if total_cap < 1e-12:
                break
            dec = (-diff) * (caps / total_cap)
            dec = np.minimum(dec, caps)
            x -= dec

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for discretized h."""
    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode="full")
    return np.max(v) * dx

def project_box_sum(y, target, max_iter=60, tol=1e-12):
    """Project y onto [0,1]^n with sum == target using bisection on a shift λ."""
    lo, hi = -1.0, 1.0
    # expand bounds until feasible
    while np.clip(y + lo, 0.0, 1.0).sum() > target:
        lo *= 2.0
    while np.clip(y + hi, 0.0, 1.0).sum() < target:
        hi *= 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def gradient_step(h, beta, alpha, target):
    """One gradient descent step on the smooth max-overlap objective."""
    n = len(h)
    v = np.correlate(h, 1.0 - h, mode="full")
    vmax = v.max()
    w = np.exp(beta * (v - vmax))
    w /= w.sum()
    g = np.zeros_like(h)
    offset = n - 1
    for idx, weight in enumerate(w):
        if weight < 1e-15:
            continue
        k = idx - offset
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
        g[tail] += weight * (1.0 - h[head])
        g[head] -= weight * h[tail]
    # keep zero‑sum direction
    g -= np.mean(g)
    y = h - alpha * g
    return project_box_sum(y, target)

def local_search(h_start, time_budget, rng):
    """Greedy pair‑transfer local search respecting the sum constraint."""
    n = len(h_start)
    target = n / 2.0
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_delta)
        if delta <= 0.0:
            continue
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def upscale_initial(h_small, n_target):
    """Linear interpolation of a coarse h to a finer grid."""
    x_small = np.linspace(0.0, 2.0, len(h_small), endpoint=False)
    x_target = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_target = np.interp(x_target, x_small, h_small)
    h_target = np.clip(h_target, 0.0, 1.0)
    return project_box_sum(h_target, n_target / 2.0)

    rng = np.random.default_rng(seed)
    h = h0.copy()

    target = n / 2.0
    best_h = h.copy()
