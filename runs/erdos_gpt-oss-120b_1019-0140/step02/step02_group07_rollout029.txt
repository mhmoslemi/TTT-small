import time, math, numpy as np

def compute_c5(v):
    n = len(v)
    corr = np.correlate(v, 1 - v, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_capped_simplex(v, target):
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

def thue_morse_seq(n):
    # vectorized?
    # Use bit count parity
    idx = np.arange(n, dtype=np.int64)
    # compute parity of bit count
    # using bitwise xor? We can compute using built-in popcount if available. Not in numpy.
    # Simple loop for moderate n.
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    a = np.array([1], dtype=int)
    while a.size < n:
        # compute next iteration using vectorization
        # a_new = np.empty(2*a.size, dtype=int)
        # a_new[0::2] = a
        # a_new[1::2] = a * ((-1) ** np.arange(a.size))
        # But (-1)**k for k integer yields -1 if odd else 1.
        # So we can compute sign = 1 - 2*(np.arange(a.size)%2)
        sign = 1 - 2 * (np.arange(a.size) % 2)
        a_new = np.empty(2 * a.size, dtype=int)
        a_new[0::2] = a
        a_new[1::2] = a * sign
        a = a_new
    a = a[:n]
    return (a + 1) // 2

def adjust_to_target(v, target):
    # ensure sum exactly target via projection and small adjustment
    v = project_onto_capped_simplex(v, target)
    # The above should be close; adjust residual
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        v += diff / len(v)
        v = np.clip(v, 0.0, 1.0)
        # final correction
        residual = target - v.sum()
        for i in range(len(v)):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -v[i], 1 - v[i])
            v[i] += adj
            residual = target - v.sum()
    return v

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0

    # Build initial population
    pop_size = 30
    population = np.empty((pop_size, n), dtype=float)
    # Insert initial if exists
    if h0 is not None:
        population[0] = adjust_to_target(h0.copy(), target_sum)
        start_idx = 1
    else:
        start_idx = 0

    # Optionally add deterministic seeds Thue-Morse and Rudin-Shapiro
    deterministic_seeds = []
    tm = thue_morse_seq(n)
    deterministic_seeds.append(adjust_to_target(tm.astype(float), target_sum))
    rs = rudin_shapiro_seq(n)
    deterministic_seeds.append(adjust_to_target(rs.astype(float), target_sum))

    # Fill the rest with random feasible vectors
    for i in range(start_idx, pop_size):
        # generate random vector uniform, then project
        v = rng.random(n)
        v = adjust_to_target(v, target_sum)
        population[i] = v

    # Evaluate initial population
    c5_vals = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        c5_vals[i] = compute_c5(population[i])

    best_idx = int(np.argmin(c5_vals))
    best_h = population[best_idx].copy()
    best_c5 = c5_vals[best_idx]

    # parameters
    F = 0.8
    CR = 0.9

    # Main DE loop
    while time.time() - start_time < budget_s - 2.0:
        # one generation
        for i in range(pop_size):
            # select three distinct indices not equal to i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            # repair
            mutant = adjust_to_target(mutant, target_sum)
            # crossover
            cross_mask = rng.random(n) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n)] = True
            trial = np.where(cross_mask, mutant, population[i])
            trial = adjust_to_target(trial, target_sum)
            trial_c5 = compute_c5(trial)
            if trial_c5 < c5_vals[i]:
                population[i] = trial
                c5_vals[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
        # optional early exit if target met
        if best_c5 <= 0.3808:
            break

        # occasional injection of deterministic seeds (replace random individuals)
        if rng.random() < 0.1:
            # replace worst individual with a deterministic seed
            worst_idx = int(np.argmax(c5_vals))
            seed_vec = rng.choice(deterministic_seeds)
            population[worst_idx] = adjust_to_target(seed_vec.copy(), target_sum)
            c5_vals[worst_idx] = compute_c5(population[worst_idx])
            # update best if needed
            if c5_vals[worst_idx] < best_c5:
                best_c5 = c5_vals[worst_idx]
                best_h = population[worst_idx].copy()

    # Local binary hill climbing from rounded best
    # Ensure exactly half ones
    bin_h = (best_h >= 0.5).astype(float)
    # Fix sum if needed
    current_sum = bin_h.sum()
    if current_sum != target_sum:
        # adjust by flipping some entries
        diff = int(target_sum - current_sum)
        # if diff >0 need more ones; set some zeros to 1
        if diff > 0:
            zeros = np.flatnonzero(bin_h == 0.0)
            if zeros.size >= diff:
                idxs = rng.choice(zeros, size=diff, replace=False)
                bin_h[idxs] = 1.0
        else:
            ones = np.flatnonzero(bin_h == 1.0)
            if ones.size >= -diff:
                idxs = rng.choice(ones, size=-diff, replace=False)
                bin_h[idxs] = 0.0

    cur_c5 = compute_c5(bin_h)
    max_iter = 5000
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 0.5:
            break
        # pick a 1 and a 0 to swap
        ones = np.flatnonzero(bin_h == 1.0)
        zeros = np.flatnonzero(bin_h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = bin_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            bin_h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
        if best_c5 <= 0.3808:
            break

    # final repair and compute final C5
    best_h = adjust_to_target(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n

import time
import numpy as np

def compute_c5(v):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated by correlation."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_capped_simplex(v, target):
    """
    Project vector v onto the set {0≤x_i≤1, sum x_i = target}.
    Uses bisection on λ in x_i = clip(v_i‑λ,0,1).
    """
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)

def adjust_to_target(v, target):
    """Clip to [0,1] and enforce exact sum=target (tiny residual handled)."""
    v = project_onto_capped_simplex(v, target)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
        residual = target - v.sum()
        for i in range(v.size):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -v[i], 1.0 - v[i])
            v[i] += adj
            residual = target - v.sum()
    return v

def thue_morse_seq(n):
    """Thue‑Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)

def rudin_shapiro_seq(n):
    """Rudin‑Shapiro (+1/‑1 → 1/0) binary sequence of length n."""
    a = np.array([1], dtype=int)
    while a.size < n:
        sign = 1 - 2 * (np.arange(a.size) % 2)   # (-1)^k
        a_new = np.empty(2 * a.size, dtype=int)
        a_new[0::2] = a
        a_new[1::2] = a * sign
        a = a_new
    a = a[:n]
    return (a + 1) // 2
