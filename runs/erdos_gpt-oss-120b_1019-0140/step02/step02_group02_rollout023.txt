def compute_gradient(h, k):
    n = h.shape[0]
    g = np.ones(n)
    if k >= 0:
        g[:n-k] -= h[k:]  # h[i+k] for i in range(n-k)
        g[k:] -= h[:n-k]  # h[i-k] for i in range(k,n)
    else:
        kk = -k
        g[kk:] -= h[:n-kk]  # i+k = i-kk
        g[:n-kk] -= h[kk:]  # i-k = i+kk
    return g

def transfer_step(h, g, step_scale, rng):
    n = h.shape[0]
    donors = np.where((g > 0) & (h > 1e-12))[0]
    recipients = np.where((g < 0) & (h < 1 - 1e-12))[0]
    if donors.size == 0 or recipients.size == 0:
        return h, False
    # sort donors descending g and recipients ascending g.
    donor_order = donors[np.argsort(-g[donors])]
    recipient_order = recipients[np.argsort(g[recipients])]
    # limit number of transfers per call to avoid heavy loops
    max_transfers = min(50, donor_order.size, recipient_order.size)
    transferred = False
    d_idx = 0
    r_idx = 0
    for _ in range(max_transfers):
        if d_idx >= donor_order.size or r_idx >= recipient_order.size:
            break
        di = donor_order[d_idx]
        rj = recipient_order[r_idx]
        # compute capacity
        avail = h[di]
        cap = 1.0 - h[rj]
        max_delta = min(avail, cap)
        if max_delta < 1e-12:
            # move to next donor or recipient
            if avail < 1e-12:
                d_idx += 1
            if cap < 1e-12:
                r_idx += 1
            continue
        delta = max_delta * step_scale
        # ensure delta positive
        if delta < 1e-12:
            delta = max_delta * 0.5
        h[di] -= delta
        h[rj] += delta
        transferred = True
        # update if donor or recipient hits bound (within tolerance)
        if h[di] <= 1e-12:
            d_idx += 1
        if h[rj] >= 1 - 1e-12:
            r_idx += 1
    return h, transferred

def compute_gradient(h, k):
    n = h.shape[0]
    g = np.ones(n)
    if k >= 0:
        # subtract h_{i+k} for i to n-k-1
        g[:n - k] -= h[k:]
        # subtract h_{i-k} for i from k to n-1
        g[k:] -= h[:n - k]
    else:
        kk = -k
        # for i >= kk: i+k = i-kk
        g[kk:] -= h[:n - kk]
        # for i < n-kk: i-k = i+kk
        g[:n - kk] -= h[kk:]
    return g

def transfer_step(h, g, step_scale, rng):
    n = h.shape[0]
    donors = np.where((g > 0) & (h > 1e-12))[0]
    recipients = np.where((g < 0) & (h < 1 - 1e-12))[0]
    if donors.size == 0 or recipients.size == 0:
        return h, False
    # shuffle for randomness
    rng.shuffle(donors)
    rng.shuffle(recipients)
    transferred = False
    max_transfers = min(30, donors.size, recipients.size)
    d_idx = 0
    r_idx = 0
    for _ in range(max_transfers):
        while d_idx < donors.size and h[donors[d_idx]] <= 1e-12:
            d_idx += 1
        while r_idx < recipients.size and h[recipients[r_idx]] >= 1 - 1e-12:
            r_idx += 1
        if d_idx >= donors.size or r_idx >= recipients.size:
            break
        di = donors[d_idx]
        rj = recipients[r_idx]
        max_delta = min(h[di], 1.0 - h[rj])
        if max_delta <= 0:
            # shouldn't happen, but move indices
            if h[di] <= 1e-12:
                d_idx += 1
            if h[rj] >= 1 - 1e-12:
                r_idx += 1
            continue
        delta = max_delta * step_scale
        h[di] -= delta
        h[rj] += delta
        transferred = True
        # update if donors/recipients reach bound
        if h[di] <= 1e-12:
            d_idx += 1
        if h[rj] >= 1 - 1e-12:
            r_idx += 1
    return h, transferred

import time
import numpy as np

def compute_c5(h):
    """Return max_k ∫ h(x)(1−h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    """Continuous random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    # fix the sum by redistributing the residual
    cur = v.sum()
    diff = target - cur
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 500:
        if diff > 0:
            idxs = np.where(v < 1 - 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            idxs = np.where(v > 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        attempts += 1
    # final tiny uniform tweak if needed
    if abs(diff) > 1e-12:
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
    return v

def enforce_sum(v, target, rng):
    """Project v onto [0,1] with exact sum `target` (tiny adjustments)."""
    cur = v.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return v
    v += diff / v.size
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        idxs = np.arange(v.size)
        rng.shuffle(idxs)
        for i in idxs:
            if diff > 0 and v[i] < 1 - 1e-12:
                inc = min(diff, 1.0 - v[i])
                v[i] += inc
                diff -= inc
            elif diff < 0 and v[i] > 1e-12:
                dec = min(-diff, v[i])
                v[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return v

def compute_gradient(h, k):
    """Gradient of the worst‑shift overlap w.r.t. h (ignoring dx)."""
    n = h.shape[0]
    g = np.ones(n)
    if k >= 0:
        g[: n - k] -= h[k:]
        g[k:]       -= h[: n - k]
    else:
        kk = -k
        g[kk:]      -= h[: n - kk]
        g[: n - kk] -= h[kk:]
    return g

def transfer_step(h, g, step_scale, rng):
    """Transfer mass from donors (g>0) to recipients (g<0)."""
    n = h.shape[0]
    donors = np.where((g > 0) & (h > 1e-12))[0]
    recipients = np.where((g < 0) & (h < 1 - 1e-12))[0]
    if donors.size == 0 or recipients.size == 0:
        return h, False
    rng.shuffle(donors)
    rng.shuffle(recipients)
    transferred = False
    max_ops = min(30, donors.size, recipients.size)
    di = 0
    rj = 0
    for _ in range(max_ops):
        while di < donors.size and h[donors[di]] <= 1e-12:
            di += 1
        while rj < recipients.size and h[recipients[rj]] >= 1 - 1e-12:
            rj += 1
        if di >= donors.size or rj >= recipients.size:
            break
        src = donors[di]
        dst = recipients[rj]
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 0:
            if h[src] <= 1e-12:
                di += 1
            if h[dst] >= 1 - 1e-12:
                rj += 1
            continue
        delta = max_delta * step_scale
        h[src] -= delta
        h[dst] += delta
        transferred = True
        if h[src] <= 1e-12:
            di += 1
        if h[dst] >= 1 - 1e-12:
            rj += 1
    return h, transferred

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:
        n = 200                       # a reasonable size under 1 k
        target = n / 2.0
        if rng.random() < 0.5:
            h0 = random_start_vector(n, rng)
        else:
            v = np.zeros(n)
            v[rng.choice(n, n // 2, replace=False)] = 1.0
            h0 = v.astype(float)
        h0 = enforce_sum(h0, target, rng)

    current_h = h0.copy()
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    no_improve = 0

    while time.time() < deadline - 0.5:
        # current overlap
        corr = np.correlate(current_h, 1.0 - current_h, mode="full") * (2.0 / n)
        cur_val = corr.max()
        idx = corr.argmax()
        # update best if improved
        if cur_val < best_val:
            best_val = cur_val
            best_h = current_h.copy()
            no_improve = 0
            if best_val < 0.3808:
                break

        # gradient for worst shift
        k = idx - (n - 1)
        g = compute_gradient(current_h, k)

        # temperature & step size
        remaining = deadline - time.time()
        total = budget_s
        T = 0.02 * (remaining / total) ** 0.5
        step_scale = max(0.01, min(0.5, T * 5))

        # propose a move
        candidate = current_h.copy()
        candidate, did_transfer = transfer_step(candidate, g, step_scale, rng)
        if not did_transfer:
            i, j = rng.integers(n, size=2, replace=False)
            if candidate[i] > 1e-12 and candidate[j] < 1 - 1e-12:
                max_delta = min(candidate[i], 1.0 - candidate[j])
                delta = max_delta * step_scale * rng.random()
                candidate[i] -= delta
                candidate[j] += delta

        candidate = enforce_sum(candidate, target, rng)
        new_val = compute_c5(candidate)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            current_h = candidate
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = current_h.copy()
                no_improve = 0
                if best_val < 0.3808:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        # restart if stuck
        if no_improve >= 3000:
            if rng.random() < 0.5:
                h0 = random_start_vector(n, rng)
            else:
                v = np.zeros(n)
                v[rng.choice(n, n // 2, replace=False)] = 1.0
                h0 = v.astype(float)
            h0 = enforce_sum(h0, target, rng)
            current_h = h0.copy()
            no_improve = 0

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
