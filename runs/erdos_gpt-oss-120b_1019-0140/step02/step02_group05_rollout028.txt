def targeted_move(cur_h, kmax, rng):
    n = len(cur_h)
    # compute overlapping indices
    if kmax >= 0:
        idx_range = range(n - kmax)
        # pairs (i, i+kmax)
        pairs = [(i, i + kmax) for i in idx_range]
    else:
        kk = -kmax
        idx_range = range(n - kk)
        pairs = [(i + kk, i) for i in idx_range] # ensure i and i+kk order
    # from pairs select those where both > some epsilon
    candidates = [(i,j) for (i,j) in pairs if cur_h[i] > 1e-8 and cur_h[j] > 1e-8]
    if not candidates:
        # fallback to random transfer
        return None
    i,j = candidates[rng.integers(len(candidates))]
    # choose donor: whichever has larger value
    if cur_h[i] >= cur_h[j]:
        donor, partner = i, j
    else:
        donor, partner = j, i
    # choose recipient: index far from donor and partner; we can pick random
    # but avoid same as donor, keep within bounds
    rec = rng.integers(n)
    # ensure rec not same and h[rec] < 1
    # we will transfer delta = min(cur_h[donor], step)
    max_step = min(cur_h[donor], 0.5) # step size limit
    delta = rng.random() * max_step
    # Ensure recipient can receive delta without exceeding 1
    recv_allow = 1.0 - cur_h[rec]
    delta = min(delta, recv_allow)
    # Perform move
    new_h = cur_h.copy()
    new_h[donor] -= delta
    new_h[rec] += delta
    # No sum error because we transferred.
    return new_h

def gradient_move(cur_h, k, dx):
    n = len(cur_h)
    # compute sum of overlapping values at shifted positions (neigh)
    neigh = np.zeros(n)
    if k >= 0:
        # h[i] overlapped with h[i+k]
        neigh[: n - k] += cur_h[k:]          # neighbor at i from i+k side
        neigh[k:] += cur_h[: n - k]          # neighbor at i from i-k side
    else:
        kk = -k
        # shift negative: i-k = i+kk
        neigh[kk:] += cur_h[: n - kk]
        neigh[: n - kk] += cur_h[kk:]
    # gradient of overlap = dx*(1 - neigh)
    grad = dx * (1.0 - neigh)
    # enforce zero-sum: remove mean
    grad_centered = grad - np.mean(grad)
    # step direction = -grad_centered
    step = -grad_centered
    # limit magnitude
    max_step = 0.2
    max_abs = np.max(np.abs(step))
    if max_abs > max_step:
        step = step * (max_step / max_abs)
    cand = cur_h + step
    cand = np.clip(cand, 0.0, 1.0)
    # project sum
    cand = project_h(cand, target_sum)
    return cand
```

def targeted_move(cur_h, kmax, rng):
    n = len(cur_h)
    # generate list of overlapping pairs
    if kmax >= 0:
        i_vals = np.arange(n - kmax)
        j_vals = i_vals + kmax
    else:
        kk = -kmax
        i_vals = np.arange(kk, n)
        j_vals = i_vals - kk
    # find indices where both hi and hj > eps
    eps = 1e-8
    mask = (cur_h[i_vals] > eps) & (cur_h[j_vals] > eps)
    if not np.any(mask):
        return None
    # choose random pair among candidates
    idx = rng.choice(np.where(mask)[0])
    i = int(i_vals[idx])
    j = int(j_vals[idx])
    # donor is whichever larger value
    if cur_h[i] >= cur_h[j]:
        donor = i
    else:
        donor = j
    # choose recipient: random index not equal donor
    # ensure recipient can receive some mass
    for _ in range(10):
        rec = rng.integers(n)
        if rec != donor and cur_h[rec] < 1.0 - 1e-12:
            break
    else:
        # if cannot find, return None
        return None
    max_tr = min(cur_h[donor], 1.0 - cur_h[rec])
    if max_tr <= 0:
        return None
    delta = rng.random() * max_tr
    new_h = cur_h.copy()
    new_h[donor] -= delta
    new_h[rec] += delta
    return new_h
```

import time
import numpy as np

def compute_c5_and_shift(h, dx):
    """Return C5 and the shift (index offset) attaining the max."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = np.max(corr) * dx
    shift = int(np.argmax(corr) - (len(h) - 1))
    return val, shift

def project_h(vec, target):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target - cur
    if abs(diff) > 1e-12:
        free = np.where((vec > 0) & (vec < 1))[0]
        if free.size:
            inc = diff / free.size
            vec[free] += inc
            vec = np.clip(vec, 0.0, 1.0)
        cur = np.sum(vec)
        diff = target - cur
        for i in range(len(vec)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and vec[i] < 1.0:
                add = min(diff, 1.0 - vec[i])
                vec[i] += add
                diff -= add
            elif diff < 0 and vec[i] > 0.0:
                sub = min(-diff, vec[i])
                vec[i] -= sub
                diff += sub
        vec = np.clip(vec, 0.0, 1.0)
    return vec

def targeted_move(cur_h, kmax, rng):
    """Transfer a small amount from a high‑overlap pair to a free slot."""
    n = len(cur_h)
    if kmax >= 0:
        i_arr = np.arange(n - kmax)
        j_arr = i_arr + kmax
    else:
        kk = -kmax
        i_arr = np.arange(kk, n)
        j_arr = i_arr - kk
    eps = 1e-8
    mask = (cur_h[i_arr] > eps) & (cur_h[j_arr] > eps)
    if not np.any(mask):
        return None
    idx = rng.choice(np.where(mask)[0])
    i, j = int(i_arr[idx]), int(j_arr[idx])
    donor = i if cur_h[i] >= cur_h[j] else j
    # pick a receiver with room
    rec = None
    for _ in range(10):
        cand = rng.integers(n)
        if cand != donor and cur_h[cand] < 1.0 - 1e-12:
            rec = cand
            break
    if rec is None:
        return None
    max_tr = min(cur_h[donor], 1.0 - cur_h[rec])
    if max_tr <= 0:
        return None
    delta = rng.random() * max_tr
    new_h = cur_h.copy()
    new_h[donor] -= delta
    new_h[rec] += delta
    return new_h

def gradient_move(cur_h, k, dx, target):
    """Step opposite the gradient of the worst‑shift overlap."""
    n = len(cur_h)
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += cur_h[k:]
        neigh[k:] += cur_h[: n - k]
    else:
        kk = -k
        neigh[kk:] += cur_h[: n - kk]
        neigh[: n - kk] += cur_h[kk:]
    grad = dx * (1.0 - neigh)
    grad_centered = grad - np.mean(grad)
    step = -grad_centered
    max_step = 0.2
    mx = np.max(np.abs(step))
    if mx > max_step:
        step = step * (max_step / mx)
    cand = cur_h + step
    cand = np.clip(cand, 0.0, 1.0)
    cand = project_h(cand, target)
    return cand
