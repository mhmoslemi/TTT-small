def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = hi - np.min(v) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def grad_smooth_max(h, alpha):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    a = alpha * conv
    max_a = np.max(a)
    exp_a = np.exp(a - max_a)
    weights = exp_a / exp_a.sum()
    grad = np.zeros_like(h)
    for m, w in enumerate(weights):
        lag = m - (n - 1)
        wdx = w * dx
        if lag >= 0:
            L = lag
            end = n - L
            if end <= 0:

            i = slice(0, end)
            j = slice(L, n)
            grad[i] += wdx * (1.0 - h[j])
            grad[j] -= wdx * h[i]

        # optional local refinement with SLSQP
        if time.time() - start < budget_s - 0.5 and best_c5 > 0.3806:
            h_ref, c5_ref = slsqp_refine(best_h, best_c5)
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref
        if best_c5 < 0.3806:
            break
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
