def build_constraints(h_len):

    N = h_len

    dx = 2.0 / N
    target_sum = N / 2.0

    # sum equality
    def sum_con(x):
        return np.sum(x[:N]) - target_sum

    constraints = [{'type': 'eq', 'fun': sum_con}]

    # for each lag
    def make_overlap(k):
        def con(x, k=k):
            h = x[:N]
            if k >= 0:
                overl = np.dot(h[: N - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: N - kk])
            return x[-1] - overl * dx
        return con

    for k in range(-N + 1, N):
        if k == 0:
            continue
        constraints.append({'type': 'ineq', 'fun': make_overlap(k)})
    return constraints

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Discrete approximation of C5."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def worst_lag_and_contributions(h):
    """Return max value, lag k, index range and contributions for that lag."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)
    if k >= 0:
        i_range = np.arange(n - k)
    else:
        i_range = np.arange(-k, n)
    contrib = h[i_range] * (1.0 - h[i_range + k])
    max_val = corr[max_idx] * (2.0 / n)
    return max_val, k, i_range, contrib

def random_start_vector(n, rng):
    """Random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        v = np.clip(v + diff / n, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    inc = min(diff, 1.0 - v[i])
                    v[i] += inc
                    diff -= inc
                elif diff < 0 and v[i] > 0.0:
                    dec = min(-diff, v[i])
                    v[i] -= dec

def simulated_annealing(h_start, deadline, rng):
    """SA with directed moves on the worst lag."""
    h = h_start.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / max(1e-12, deadline - start)
        T = max(1e-6, (1 - frac) * 0.05)

        if rng.random() < 0.7:
            # directed move
            _, k, i_range, contrib = worst_lag_and_contributions(h)
            src_local = int(np.argmax(contrib))
            src = int(i_range[src_local])
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                continue
            # score for each candidate destination
            dests = cand
            term1 = np.zeros_like(dests, dtype=float)
            term2 = np.zeros_like(dests, dtype=float)
            valid1 = (dests + k >= 0) & (dests + k < n)
            term1[valid1] = 1.0 - h[dests[valid1] + k]
            valid2 = (dests - k >= 0) & (dests - k < n)
            term2[valid2] = h[dests[valid2] - k]
            scores = term1 - term2  # lower is better
            dest = int(dests[np.argmin(scores)])
            if dest == src:
                alt = np.setdiff1d(cand, np.array([src]), assume_unique=True)
                if alt.size == 0:
                    continue
                dest = int(rng.choice(alt))
        else:
            # random transfer
            src_candidates = np.where(h > 1e-12)[0]
            dst_candidates = np.where(h < 1.0 - 1e-12)[0]
            if src_candidates.size == 0 or dst_candidates.size == 0:
                break
            src = int(rng.choice(src_candidates))
            dest = int(rng.choice(dst_candidates))
            if src == dest:
                continue

        max_transfer = min(h[src], 1.0 - h[dest])
        if max_transfer <= 0:
            continue
        # occasionally move the whole transferable amount
        if rng.random() < 0.2:
            epsilon = max_transfer
        else:
            eps_factor = 0.5 * (1 - frac) + 0.01
            epsilon = rng.random() * max_transfer * eps_factor
        if epsilon <= 0:
            continue

        h[src] -= epsilon
        h[dest] += epsilon
        new_val = compute_c5(h)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            h[src] += epsilon
            h[dest] -= epsilon
    return best_h, best_val

        T = max(1e-6, (1 - frac) * 0.02)

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i], h[j] = h[j], h[i]  # swap

        new_val = compute_c5(h)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
    return best_h, best_val

def build_constraints(N):
    """Constraints for SLSQP: sum and per‑lag overlap ≤ t."""
    dx = 2.0 / N
    target = N / 2.0

    def sum_con(x):
        return np.sum(x[:N]) - target

    cons = [{"type": "eq", "fun": sum_con}]

    def make_overlap(k):
        def con(x, k=k):
            h = x[:N]
            if k >= 0:
                overl = np.dot(h[: N - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: N - kk])
            return x[-1] - overl * dx
        return con

    for k in range(-N + 1, N):
        if k == 0:
            continue
        cons.append({"type": "ineq", "fun": make_overlap(k)})
    return cons

def refine_with_slsqp(h_init, cur_val, deadline, rng):
    """One SLSQP refinement from a given start."""
    N = h_init.shape[0]
    cons = build_constraints(N)
    bounds = [(0.0, 1.0)] * N + [(0.0, None)]
    x0 = np.concatenate([h_init, [cur_val]])

    time_left = deadline - time.time()
    maxiter = int(min(5000, max(10, time_left * 10)))
    if maxiter < 10:
        return h_init, cur_val

    try:
        res = minimize(
            fun=lambda x: x[-1],
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-12, "disp": False},
        )
    except Exception:
        return h_init, cur_val

    if not res.success:
        return h_init, cur_val

    h_opt = np.clip(res.x[:N], 0.0, 1.0)
    target = N / 2.0
    cur_sum = h_opt.sum()
    if abs(cur_sum - target) > 1e-9:
        h_opt += (target - cur_sum) / N
        h_opt = np.clip(h_opt, 0.0, 1.0)
    val = compute_c5(h_opt)
    return h_opt, val

def to_binary(h):
    """Return a binary vector with exactly n/2 ones (top‑value rounding)."""
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(-h, k - 1)[:k]
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        cur = h0.sum()
        if abs(cur - target) > 1e-12:
            diff = target - cur
            h0 = np.clip(h0 + diff / n, 0.0, 1.0)
            cur = h0.sum()
            diff = target - cur
            if abs(diff) > 1e-12:
                idxs = np.arange(n)
                rng.shuffle(idxs)
                for i in idxs:
                    if abs(diff) < 1e-12:
                        break
                    if diff > 0 and h0[i] < 1.0:
                        inc = min(diff, 1.0 - h0[i])
                        h0[i] += inc
                        diff -= inc
                    elif diff < 0 and h0[i] > 0.0:
                        dec = min(-diff, h0[i])
                        h0[i] -= dec
                        diff += dec
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    h0 = np.clip(h0, 0.0, 1.0)
    target = n / 2.0
    cur = h0.sum()
    if abs(cur - target) > 1e-12:
        h0 += (target - cur) / n
        h0 = np.clip(h0, 0.0, 1.0)
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h0[i] < 1.0:
                    inc = min(diff, 1.0 - h0[i])
                    h0[i] += inc
                    diff -= inc
                elif diff < 0 and h0[i] > 0.0:
                    dec = min(-diff, h0[i])
                    h0[i] -= dec
                    diff += dec

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- multi‑phase search -----
    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        phase_time = min(30.0, remaining - 5.0)

        # start from either a fresh random vector or a perturbed best
        if rng.random() < 0.3:
            h_start = random_start_vector(n, rng)
        else:
            h_start = best_h.copy()
            src = rng.choice(np.where(h_start > 1e-12)[0])
            dst = rng.choice(np.where(h_start < 1.0 - 1e-12)[0])
            max_t = min(h_start[src], 1.0 - h_start[dst])
            if max_t > 0:
                eps = rng.random() * max_t * 0.1
                h_start[src] -= eps
                h_start[dst] += eps

        cand_h, cand_val = simulated_annealing(h_start, time.time() + phase_time, rng)
        if cand_val < best_val:
            best_val = cand_val
            best_h = cand_h.copy()
            if best_val <= 0.3808:
                break

    # ----- binary refinement (if time permits) -----
    if time.time() < deadline - 5.0:
        binary_start = to_binary(best_h)
        bin_h, bin_val = binary_annealing(binary_start, deadline - 5.0, rng)
        if bin_val < best_val:
            best_val = bin_val
            best_h = bin_h.copy()

    # ----- final SLSQP refinement (short window) -----
    if time.time() < deadline - 2.0:
        refined_h, refined_val = refine_with_slsqp(best_h, best_val, deadline, rng)
        if refined_val < best_val:
            best_val = refined_val
            best_h = refined_h.copy()

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h += (target - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
