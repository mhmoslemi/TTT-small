import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def random_vector(n, rng):
    # generate random float vector in [0,1] with exact sum n/2
    target = n/2.0
    # sample from Dirichlet then scale
    x = rng.random(n)
    # Use simple scaling: adjust sum using additive correction distributed uniformly
    # Ensure within [0,1]
    total=x.sum()
    diff = target - total
    # add diff/n to each, clip, then fix residual
    x = np.clip(x + diff/n, 0.0, 1.0)
    cur = x.sum()
    residual = target - cur
    # distribute residual greedily
    if abs(residual) > 1e-12:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for i in idxs:
            if abs(residual) < 1e-12:
                break
            if residual > 0 and x[i] < 1.0:
                inc = min(residual, 1.0 - x[i])
                x[i] += inc
                residual -= inc
            elif residual < 0 and x[i] > 0.0:
                dec = min(-residual, x[i])
                x[i] -= dec
                residual += dec
    return x
```
