def solve_lp(n):
    # Build data for constraints
    import numpy as np
    from scipy.optimize import linprog
    import scipy.sparse as sp

    N = n + n*(n-1) + 1
    idx_h = np.arange(n)
    idx_L = n + n*(n-1)
    # Compute mapping for w variables
    # Use list of tuples (i,k) -> idx
    w_pairs = []
    # precomputer shift ranges
    for k in range(-n+1, n):
        if k == 0: continue
        if k > 0:
            i_vals = np.arange(0, n - k)
        else:
            i_vals = np.arange(-k, n)  # because i+k >= 0 => i >= -k
        for i in i_vals:
            w_pairs.append((i,k))
    # w_pairs length = n*(n-1)
    n_w = len(w_pairs)
    # map: w_idx = n + t
    # bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n_w + [(0.0, None)]  # L >=0 (makes sense)
    # equality: sum h = n/2
    A_eq = sp.csr_matrix((1, N))
    A_eq[0, idx_h] = 1.0
    b_eq = np.array([n/2.0])
    # Now build inequalities rows
    # We'll accumulate row data vectors
    rows = []
    cols = []
    data = []
    b_ub = []

    # constraint type 1: w - h[i] <= 0
    for t, (i,k) in enumerate(w_pairs):
        w_idx = n + t
        # w - h_i <= 0
        rows.append(len(b_ub))
        cols.append(w_idx)
        data.append(1.0)
        rows.append(len(b_ub))
        cols.append(idx_h[i])
        data.append(-1.0)
        b_ub.append(0.0)

        # w - h_j <= 0
        j = i + k
        rows.append(len(b_ub))
        cols.append(w_idx)
        data.append(1.0)
        rows.append(len(b_ub))
        cols.append(idx_h[j])
        data.append(-1.0)
        b_ub.append(0.0)

        # -w + h_i + h_j <= 1
        rows.append(len(b_ub))
        cols.append(w_idx)
        data.append(-1.0)
        rows.append(len(b_ub))
        cols.append(idx_h[i])
        data.append(1.0)
        rows.append(len(b_ub))
        cols.append(idx_h[j])
        data.append(1.0)
        b_ub.append(1.0)

    # constraint type 4: L - sum_i w_{i,k} <= 0 for each shift k
    # we need to group w for each shift
    # map shift->list of w indices
    shift_to_widx = {}
    for t, (i,k) in enumerate(w_pairs):
        w_idx = n + t
        shift_to_widx.setdefault(k, []).append(w_idx)

    for k, w_idxs in shift_to_widx.items():
        # row: L - sum_{w in shift} w <= 0
        rows.append(len(b_ub))
        cols.append(idx_L)
        data.append(1.0)
        for w_idx in w_idxs:
            rows.append(len(b_ub))
            cols.append(w_idx)
            data.append(-1.0)
        b_ub.append(0.0)

    # Build A_ub sparse matrix
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(b_ub), N))
    # Solve
    res = linprog(c=np.concatenate([np.zeros(n + n_w), [-1.0]]), A_ub=A_ub, b_ub=np.array(b_ub),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    return res

def project_sum(h, target):
    # clip to [0,1]
    h = np.clip(h, 0, 1)
    cur = np.sum(h)
    diff = target - cur
    # distribute diff uniformly among free variables (0<h<1)
    free = np.where((h > 0) & (h < 1))[0]
    if free.size > 0:
        inc = diff / free.size
        h[free] += inc
        # clip again
        h = np.clip(h, 0, 1)
    # adjust residual diffs by scanning
    diff = target - np.sum(h)
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and h[i] < 1:
            add = min(diff, 1 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0, 1)

def compute_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    # compute weights softmax
    # to avoid overflow, subtract max
    exp_arg = alpha * overlaps
    maxexp = np.max(exp_arg)
    exp_shifted = np.exp(exp_arg - maxexp)
    weights = exp_shifted / np.sum(exp_shifted)
    # convolve weights with h
    conv = np.convolve(weights, h, mode='full')
    conv_rev = np.convolve(weights[::-1], h, mode='full')
    offset = n - 1
    # term arrays
    term = conv[offset:offset+n]
    term_rev = conv_rev[offset:offset+n]
    grad = 1.0 - term - term_rev
    return grad, overlaps.max()
```

def project_h(h, target):
    # clip and adjust sum as earlier
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        # add diff uniformly to interior entries where possible
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        # residual fix
        diff = target - np.sum(h)
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)
```

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        for i in range(h.size):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h
```

def worst_shift_info(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    max_val = corr[idx_max] * (2.0 / n)
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = 1.0 - neigh
    return grad, k, max_val
```

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discretized step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project onto [0,1] box and enforce exact sum `target_sum`."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        for i in range(h.size):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def worst_shift_info(h):
    """Return gradient for the shift attaining the current maximum overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)                     # shift index
    max_val = corr[idx] * (2.0 / n)       # current C5
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = 1.0 - neigh
    return grad, k, max_val

def smooth_and_fix(h, target_sum):
    """One Gaussian smoothing pass with re‑projection."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return project_h(sm, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- problem size -------------------------------------------------
    n = 200                        # number of discretisation points
    target_sum = n / 2.0

    # ---- initial construction -----------------------------------------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        m = init.size
        if m != n:
            xs = np.linspace(0.0, 2.0, m, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(xt, xs, init)
        else:
            h = init.copy()
    else:
        h = np.full(n, 0.5, dtype=float)

    h = project_h(h, target_sum)
    best_h = h.copy()
    best_val = compute_c5(best_h)

    # ---- main optimisation loop ---------------------------------------
    step = 0.2                     # initial step size for gradient moves
    T0 = 0.015                     # initial temperature for occasional uphill moves
    p_grad = 0.65                  # probability to use gradient move
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.94:
            break
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        # current worst shift information
        grad, k, cur_val = worst_shift_info(h)

        # ----- choose a move -----------------------------------------
        if rng.random() < p_grad:
            # deterministic gradient step
            cand = h - step * grad
        else:
            # random transfer move
            i, j = rng.integers(0, n, size=2)
            max_tr = min(h[i], 1.0 - h[j])
            if max_tr <= 0.0:
                continue
            delta = rng.random() * max_tr
            cand = h.copy()
            cand[i] -= delta
            cand[j] += delta

        # ----- enforce constraints ------------------------------------
        cand = project_h(cand, target_sum)
        cand_val = compute_c5(cand)

        # ----- acceptance test ----------------------------------------
        accept = False
        if cand_val < cur_val:
            accept = True
        elif T > 0.0:
            if rng.random() < np.exp(-(cand_val - cur_val) / T):
                accept = True

        if accept:
            h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break

        # ----- adapt step size ----------------------------------------
        step = max(step * 0.999, 1e-5)

    # ---- optional smoothing passes ------------------------------------
    while time.time() - start < budget_s - 5.0:
        improved = False
        for _ in range(12):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
                improved = True
            else:
                break
        if not improved:
            break

    # ---- final safety correction --------------------------------------
    best_h = project_h(np.clip(best_h, 0.0, 1.0), target_sum)
    final_val = compute_c5(best_h)
    if final_val < best_val:
        best_val = final_val

    return best_h, best_val, n
