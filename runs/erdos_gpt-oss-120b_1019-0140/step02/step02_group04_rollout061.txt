def project_to_sum(v, target):
    # v is vector, target sum.
    # Use bisection on lambda.
    lower = -1.0
    upper = 1.0
    for _ in range(50):
        lam = (lower + upper) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            upper = lam
        else:
            lower = lam
    return np.clip(v + (lower + upper) / 2.0, 0.0, 1.0)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(v, target):
    """Project vector onto the box [0,1] with exact sum = target using bisection."""
    lower, upper = -2.0, 2.0
    for _ in range(60):
        lam = (lower + upper) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            upper = lam
        else:
            lower = lam
    return np.clip(v + (lower + upper) * 0.5, 0.0, 1.0)

def generate_random_feasible(n, target, rng):
    """Uniform random feasible vector (values in [0,1] and exact sum)."""
    v = rng.random(n)
    return project_to_sum(v, target)

def logistic_h(a, b, xs):
    return 1.0 / (1.0 + np.exp(a * (xs - b)))

def sym_logistic_h(a, b, xs):
    d = np.abs(xs - 1.0)
    return 1.0 / (1.0 + np.exp(a * (d - b)))

def family_obj(params, h_func, xs, target, dx):
    a, b = params
    if a <= 0.0 or b < 0.0 or b > 2.0:
        return 1e6
    h = h_func(a, b, xs)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, target)
    return compute_c5_vec(h, dx)

def optimize_family(h_func, xs, target, dx, rng, trials=4):
    """Search best h from a two‑parameter family."""
    best_h, best_val = None, np.inf
    for _ in range(trials):
        a0 = rng.uniform(0.1, 15.0)
        b0 = rng.uniform(0.0, 2.0)
        res = opt.minimize(
            family_obj,
            x0=np.array([a0, b0]),
            args=(h_func, xs, target, dx),
            method="Nelder-Mead",
            options={"maxiter": 250, "xatol": 1e-6, "fatol": 1e-6, "disp": False},
        )
        if not res.success:
            continue
        a_opt, b_opt = res.x
        h = h_func(a_opt, b_opt, xs)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target)
        val = compute_c5_vec(h, dx)
        if val < best_val:
            best_val, best_h = val, h
    return best_h, best_val

def refine_local(start_h, dx, target, time_limit, global_start, rng):
    """Targeted coordinate descent with simple simulated annealing."""
    cur_h = start_h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    n = len(cur_h)
    eps = 1e-12
    step = 0.05
    init_temp = 0.02
    while time.time() < time_limit:
        # temperature based on remaining fraction of allocated window
        remaining = time_limit - time.time()
        if remaining <= 0:
            break
        frac = remaining / (time_limit - global_start)
        temp = max(1e-6, init_temp * frac)
        # current worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)  # shift in index units
        # decide move type
        if rng.random() < 0.6:
            success = False
            if k >= 0:
                mask = (cur_h[: n - k] > eps) & (cur_h[k:] < 1.0 - eps)
                idxs = np.where(mask)[0]
                if idxs.size > 0:
                    i = rng.choice(idxs)
                    j = i + k
                    success = True
            else:
                kk = -k
                mask = (cur_h[kk:] > eps) & (cur_h[: n - kk] < 1.0 - eps)
                idxs = np.where(mask)[0]
                if idxs.size > 0:
                    i = idxs[0] + kk
                    j = idxs[0]
                    success = True
            if not success:
                donors = np.where(cur_h > eps)[0]
                recipients = np.where(cur_h < 1.0 - eps)[0]
                if donors.size == 0 or recipients.size == 0:
                    break
                i = rng.choice(donors)
                j = rng.choice(recipients)
        else:
            donors = np.where(cur_h > eps)[0]
            recipients = np.where(cur_h < 1.0 - eps)[0]
            if donors.size == 0 or recipients.size == 0:
                break
            i = rng.choice(donors)
            j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            step *= 0.5
            continue
        delta = rng.random() * max_delta
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h, cur_val = cand, cand_val
            step = min(step * 1.1, 0.5)
        else:
            if rng.random() < np.exp(-(cand_val - cur_val) / temp):
                cur_h, cur_val = cand, cand_val
                step *= 0.99
            else:
                step *= 0.99
    cur_h = project_to_sum(cur_h, target)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def binary_hill_climb(bin_h, cur_val, dx, time_limit, rng):
    """Swap 0/1 entries to lower C5."""
    best = bin_h.copy()
    best_val = cur_val
    n = len(best)
    while time.time() < time_limit:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0

        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best, best_val = cand, cand_val
    return best.astype(float), best_val
