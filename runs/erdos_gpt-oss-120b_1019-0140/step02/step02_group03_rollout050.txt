def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto the box [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# ---------- Constraint classes ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k > 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=600):
    """Run a single SLSQP optimisation starting from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Initialise dimensionality and feasible start vector ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---- Phase 1: SLSQP restarts (≈10% of budget) ----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    phase1_deadline = start_time + budget_s * 0.1
    while time.time() < phase1_deadline:
        h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, target_sum)
        t_start = compute_c5(h_start)
        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=600)
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            if best_c5 < 0.3805:  # early stop if good enough
                break

    # ---- Phase 2: Projected gradient descent on the active shift ----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    base_step = 0.5
    grad_deadline = start_time + budget_s - 0.5
    while time.time() < grad_deadline:
        # compute correlation and identify the shift attaining the max
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        dx = 2.0 / n
        overlaps = corr * dx
        center = n - 1
        overlaps[center] = -np.inf  # ignore zero shift
        k_idx = int(np.argmax(overlaps))
        k = k_idx - center

        # gradient of the active overlap (subgradient of the max)
        grad = np.zeros_like(cur_h)
        if k > 0:
            grad[:n - k] += (1.0 - cur_h[k:])
            grad[k:] -= cur_h[:n - k]
        elif k < 0:
            kk = -k
            grad[kk:] += (1.0 - cur_h[:n - kk])
            grad[:n - kk] -= cur_h[kk:]
        else:  # k == 0 (should not happen due to -inf)
            grad = 1.0 - 2.0 * cur_h

        # step size that decays with elapsed time
        elapsed = time.time() - start_time
        step = base_step * max(0.001, 1.0 - elapsed / budget_s)

        new_h = cur_h - step * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)

        if new_c5 + 1e-12 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
            # slightly reduce step for finer search
            base_step *= 0.99
        else:
            # no improvement – shrink step
            base_step *= 0.9

        # occasional random restart to escape stagnation
        if base_step < 1e-4 or rng.random() < 0.001:
            cur_h = rng.random(n)
            cur_h = project_to_box_sum(cur_h, target_sum)
            cur_c5 = compute_c5(cur_h)
            base_step = 0.5

    # ---- Phase 3: Simple random-direction hill‑climbing for polishing ----
    polishing_deadline = start_time + budget_s - 0.1
    while time.time() < polishing_deadline:
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2)
        cand = cur_h + step * d
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        if rng.random() < 0.001:
            base_step *= 0.95

    # ---- Final projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
