import time
import numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # find bounds for lambda
    lam_low = -1.0
    lam_high = 1.0
    def sum_clip(lam):
        return np.clip(v - lam, lo, hi).sum()
    while sum_clip(lam_low) > target_sum:
        lam_low *= 2
    while sum_clip(lam_high) < target_sum:
        lam_high *= 2
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = sum_clip(lam_mid)
        if abs(s - target_sum) < 1e-12:
            break
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def max_corr_and_shift(h):
    """Return (max_corr, shift) where shift is integer offset where max occurs."""
    n = h.size
    C = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(C)
    idx = np.argmax(C)
    shift = idx - (n - 1)
    return max_corr, shift

def hill_climb(h0, rng, time_left):
    """Simple simulated‑annealing hill climbing on a binary (0/1) vector."""
    h = h0.copy()
    n = h.size
    max_corr, _ = max_corr_and_shift(h)
    best_h = h.copy()
    best_corr = max_corr
    total_iters = 20000
    for it in range(1, total_iters + 1):
        if time_left() < 0.2:
            break
        # recompute correlation and shift
        C = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(C)
        idx = np.argmax(C)
        shift = idx - (n - 1)

        # indices where (1,0) and (0,1) pairs for this shift
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            i_plus = i_range + shift
        else:
            s = -shift
            i_range = np.arange(s, n)
            i_plus = i_range - s
        h_i = h[i_range]
        h_ip = h[i_plus]
        pos_one_zero = i_range[(h_i == 1) & (h_ip == 0)]
        pos_zero_one = i_range[(h_i == 0) & (h_ip == 1)]

        if pos_one_zero.size and pos_zero_one.size:
            i = rng.choice(pos_one_zero)
            j = rng.choice(pos_zero_one)
        else:
            # fallback: random 1 and 0
            i = rng.choice(np.where(h == 1)[0])
            j = rng.choice(np.where(h == 0)[0])

        # propose swap
        h[i], h[j] = h[j], h[i]
        new_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))

        # simulated annealing acceptance
        if new_corr <= max_corr:
            accept = True
        else:
            # temperature schedule decreasing linearly
            T = max_corr * 0.05 * (1.0 - it / total_iters)
            if T > 1e-12:
                prob = np.exp(-(new_corr - max_corr) / T)
                accept = rng.random() < prob
            else:
                accept = False
        if not accept:
            # revert swap
            h[i], h[j] = h[j], h[i]
        else:
            max_corr = new_corr
            if new_corr < best_corr:
                best_corr = new_corr
                best_h = h.copy()
    return best_h, best_corr

def gradient_descent_momentum_float(h0, rng, time_left, max_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
    for it in range(1, max_iter + 1):
        if time_left() < 0.1:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)

        grad = np.zeros_like(h)
        for s in shifts:
            if s >= 0:
                grad[:n - s] += 1.0 - h[s:]
                grad[s:] -= h[:n - s]
            else:
                sneg = -s
                grad[sneg:] += 1.0 - h[:n - sneg]
                grad[:n - sneg] -= h[sneg:]
        grad /= len(shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = -1.0
    lam_high = 1.0
    def sum_clip(lam):
        return np.clip(v - lam, lo, hi).sum()
    while sum_clip(lam_low) > target_sum:
        lam_low *= 2
    while sum_clip(lam_high) < target_sum:
        lam_high *= 2
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = sum_clip(lam_mid)
        if abs(s - target_sum) < 1e-12:
            break
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def max_corr_and_shift(h):
    n = h.size
    C = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(C)
    idx = np.argmax(C)
    shift = idx - (n - 1)
    return max_corr, shift

def hill_climb(h0, rng, time_left):
    h = h0.copy()
    n = h.size
    max_corr, _ = max_corr_and_shift(h)
    best_h = h.copy()
    best_corr = max_corr
    total_iters = 20000
    for it in range(1, total_iters + 1):
        if time_left() < 0.2:
            break
        C = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(C)
        idx = np.argmax(C)
        shift = idx - (n - 1)

        if shift >= 0:
            i_range = np.arange(0, n - shift)
            i_plus = i_range + shift
        else:
            s = -shift
            i_range = np.arange(s, n)
            i_plus = i_range - s
        h_i = h[i_range]
        h_ip = h[i_plus]
        pos_one_zero = i_range[(h_i == 1) & (h_ip == 0)]
        pos_zero_one = i_range[(h_i == 0) & (h_ip == 1)]

        if pos_one_zero.size and pos_zero_one.size:
            i = rng.choice(pos_one_zero)
            j = rng.choice(pos_zero_one)
        else:
            i = rng.choice(np.where(h == 1)[0])
            j = rng.choice(np.where(h == 0)[0])

        h[i], h[j] = h[j], h[i]
        new_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))

        if new_corr <= max_corr:
            accept = True
        else:
            T = max_corr * 0.05 * (1.0 - it / total_iters)
            if T > 1e-12:
                prob = np.exp(-(new_corr - max_corr) / T)
                accept = rng.random() < prob
            else:
                accept = False
        if not accept:
            h[i], h[j] = h[j], h[i]
        else:
            max_corr = new_corr
            if new_corr < best_corr:
                best_corr = new_corr
                best_h = h.copy()
    return best_h, best_corr

def gradient_descent_momentum_float(h0, rng, time_left, max_iter=5000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
    for it in range(1, max_iter + 1):
        if time_left() < 0.1:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)

        grad = np.zeros_like(h)
        for s in shifts:
            if s >= 0:
                grad[:n - s] += 1.0 - h[s:]
                grad[s:] -= h[:n - s]
            else:
                sneg = -s
                grad[sneg:] += 1.0 - h[:n - sneg]
                grad[:n - sneg] -= h[sneg:]
        grad /= len(shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
