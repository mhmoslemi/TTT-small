def compute_c5(arr):
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

    rng = np.random.default_rng(seed)

        # ensure correct shape

            n = h.shape[0]
            # but we assume it is fine

    target_sum = n / 2.0
    cur_sum = np.sum(h)
    if abs(cur_sum - target_sum) > 1e-12:

        h += diff / n
        h = np.clip(h, 0.0, 1.0)

        for i in range(n):

    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)
