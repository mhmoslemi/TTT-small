            h = h_new
        # random shake if stuck
        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = len(init_h)
        target_sum = n / 2.0
        init_h = project_sum_and_bounds(init_h, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        init_h = rng.random(n)
        init_h = project_sum_and_bounds(init_h, target_sum)

    best_h = init_h.copy()
    best_c5 = compute_c5(best_h)

    # allocate time: 85 % for gradient descent restarts, rest for fine‑tuning
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    gd_budget = remaining * 0.85
    ft_budget = remaining - gd_budget

    # Gradient‑descent phase with multiple short restarts
    gd_start = time.time()
    while time.time() - gd_start < gd_budget:
        # choose start point
        if (time.time() - start) < gd_budget * 0.3 and 'initial_h_values' in globals():
            h0 = best_h.copy()
        else:
            h0 = rng.random(n)
            h0 = project_sum_and_bounds(h0, target_sum)

        # give each run a modest slice of the budget (max 5 s)
        sub_rem = gd_budget - (time.time() - gd_start)
        sub_budget = min(5.0, sub_rem)
        if sub_budget <= 0.0:
            break

        h_opt, c5_opt = gradient_descent_optimize(h0, sub_budget, rng, target_sum)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        # stop early if budget exhausted
        if time.time() - start > budget_s * 0.98:
            break

    # Fine‑tuning phase: small random perturbations with simulated‑annealing style acceptance
    ft_start = time.time()
    while time.time() - ft_start < ft_budget:
        # scale of perturbation decays over fine‑tuning time
        ft_elapsed = time.time() - ft_start
        scale = max(0.001, 0.1 * (1.0 - ft_elapsed / ft_budget))
        # propose perturbation on a few coordinates
        idx = rng.choice(n, size=min(5, n), replace=False)
        delta = rng.uniform(-scale, scale, size=idx.shape[0])
        h_new = best_h.copy()
        h_new[idx] += delta
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum_and_bounds(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            # occasional uphill acceptance
            prob = np.exp(-(c5_new - best_c5) / (scale + 1e-12))
            if rng.random() < prob * 0.001:
                best_h = h_new.copy()
                best_c5 = c5_new

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
