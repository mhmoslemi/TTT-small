def proj_box_sum(y, target):
    # binary search for lambda
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo+hi)/2
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo+hi)/2,0,1)
    # adjust small floating error by scaling diff equally across free variables
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 0) & (x < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            x[free] += inc
            x = np.clip(x,0,1)
        # fallback: adjust sequentially
        diff = target - x.sum()
        for i in range(len(x)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and x[i] < 1:
                add = min(diff, 1-x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
        x = np.clip(x,0,1)
    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a discretised step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(y, target):
    """
    Projection of y onto the set {x | 0<=x_i<=1, sum x = target}
    using a bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size:
            inc = diff / free.size
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0.0 and x[i] < 1.0:
                    add = min(diff, 1.0 - x[i])
                    x[i] += add
                    diff -= add
                elif diff < 0.0 and x[i] > 0.0:
                    sub = min(-diff, x[i])
                    x[i] -= sub
                    diff += sub
            x = np.clip(x, 0.0, 1.0)
    return x

            k = idx_max - (n - 1)

            # gradient of the active shift (zero‑sum direction)
            neigh = np.zeros(n, dtype=float)
            if k >= 0:
                neigh[: n - k] += h[k:]
                neigh[k:] += h[: n - k]
            else:
                kk = -k
                neigh[kk:] += h[: n - kk]
                neigh[: n - kk] += h[kk:]
            grad = dx * (1.0 - neigh)
            delta = -(grad - np.mean(grad))

            # limit step size to keep updates moderate
            max_step = 0.3
            mabs = np.max(np.abs(delta))
            if mabs > max_step:
                delta = delta * (max_step / mabs)

            cand = project_to_sum_box(h + delta, target_sum)

            cand_val = compute_c5(cand)
            # occasional uphill acceptance to escape plateaus
            accept = cand_val < cur_val or rng.random() < 0.001
            if accept:
                h = cand
            it += 1

        # optional smoothing of the best solution found so far
        if time.time() - start < budget_s - 5.0:
            for _ in range(5):
                sm = smooth_and_fix(best_h, target_sum)
                sm_val = compute_c5(sm)
                if sm_val + 1e-12 < best_val:
                    best_val = sm_val
                    best_h = sm
                else:
                    break

    # -----------------------------------------------------------
    # final refinement with SLSQP if time permits
    if time.time() - start < budget_s - 5.0:
        n = best_h.shape[0]
        dx = 2.0 / n
        target_sum = n / 2.0

        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

        constraints = [{"type": "eq", "fun": sum_eq}]
        def mk_ineq(shift):
            def fun(x):
                h = x[:n]
                if shift >= 0:
                    overl = np.dot(h[: n - shift], 1.0 - h[shift:])
                else:
                    kk = -shift
                    overl = np.dot(h[kk:], 1.0 - h[: n - kk])
                return x[-1] - overl * dx
            return fun
        for kk in range(-n + 1, n):
            if kk == 0:
                continue
            constraints.append({"type": "ineq", "fun": mk_ineq(kk)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 2000, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_to_sum_box(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum_box(best_h, target_sum)

    return best_h, best_val, n
