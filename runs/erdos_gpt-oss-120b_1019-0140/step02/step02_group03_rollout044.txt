def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine initial n and h0
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        # Ensure feasible
        h0 = project_to_box_sum(np.clip(h0,0.0,1.0), n/2.0)
    else:
        n = 80
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, n/2.0)

    best_h = h0
    best_c5 = compute_c5(best_h)

    # Phase 1: random restarts with SLSQP
    # allocate 70% of budget for this
    phase1_deadline = start_time + budget_s * 0.7
    # choose maxiter for SLSQP based on n perhaps
    slsqp_maxiter = 300 if n <= 100 else 200  # less iterations for larger n

    while time.time() < phase1_deadline:
        # random start
        h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, n/2.0)
        # run SLSQP from this start
        h_opt = slsqp_optimize(h_start, maxiter=slsqp_maxiter)
        if h_opt is None:

        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            # early exit if good enough
            if best_c5 <= 0.3808:
                break

    # Phase 2: Simulated annealing hill climbing
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # parameters
    base_step = 0.5
    # temperature schedule
    temp0 = 0.01
    temp_decay = 0.9995  # per iteration

    iter_start = time.time()
    # loop while time left
    while time.time() - start_time < budget_s - 1.0:
        # generate random direction with zero sum
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2)
        cand = cur_h + step * d
        cand = project_to_box_sum(cand, n/2.0)
        cand_c5 = compute_c5(cand)

        # acceptance criteria
        if cand_c5 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
        else:
            # annealing acceptance
            elapsed = time.time() - start_time
            # temperature decays with time
            temp = temp0 * (temp_decay ** (elapsed * 100))  # scaling
            # compute acceptance probability
            if rng.random() < np.exp(-(cand_c5 - cur_c5) / (temp + 1e-12)):
                cur_h = cand
                cur_c5 = cand_c5

        # update best
        if cur_c5 < best_c5:
            best_h = cur_h.copy()
            best_c5 = cur_c5
            # maybe lower step size for fine tuning

            # if solution good enough break

            h_refined = slsqp_optimize(best_h, maxiter=200)

                c5_refined = compute_c5(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    cur_h = best_h.copy()
                    cur_c5 = best_c5

        # shrink step occasionally

            base_step *= 0.9

    # final projection

    best_h = project_to_box_sum(best_h, n/2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def objective_t(x):
    return x[-1]

def eq_con(x, n):
    """Equality constraint: sum(h) == n/2."""
    return np.sum(x[:n]) - n / 2.0

def ineq_con(x, n):
    """Inequality constraints: t >= overlap(k) for all k."""
    h = x[:n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return x[-1] - corr

def slsqp_optimize(h0, n, maxiter=300):
    """Run SLSQP to minimize the max overlap t starting from h0."""
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {'type': 'eq', 'fun': eq_con, 'args': (n,)},
        {'type': 'ineq', 'fun': ineq_con, 'args': (n,)}
    ]
    res = minimize(
        fun=objective_t,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    if not res.success:
        return None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce exact sum after optimisation
    h_opt = project_to_box_sum(h_opt, n / 2.0)
    return h_opt

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Determine problem size and initial feasible vector ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_box_sum(h0, n / 2.0)
    else:
        n = 80
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---- Phase 1: many random SLSQP restarts ----
    phase1_deadline = start_time + budget_s * 0.7
    slsqp_maxiter = 300 if n <= 100 else 200

    while time.time() < phase1_deadline:
        # random feasible start
        h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, n / 2.0)

        h_opt = slsqp_optimize(h_start, n, maxiter=slsqp_maxiter)
        if h_opt is None:
            continue

        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            if best_c5 <= 0.3808:
                break

    # ---- Phase 2: Simulated‑annealing hill climbing ----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    base_step = 0.5
    temp0 = 0.01
    temp_decay = 0.9995

    while time.time() - start_time < budget_s - 1.0:
        # propose a zero‑mean perturbation
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2)
        cand = cur_h + step * d
        cand = project_to_box_sum(cand, n / 2.0)

        cand_c5 = compute_c5(cand)

        # acceptance rule
        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
        else:
            elapsed = time.time() - start_time
            temp = temp0 * (temp_decay ** (elapsed * 100))
            if rng.random() < np.exp(-(cand_c5 - cur_c5) / (temp + 1e-12)):

        # record best
