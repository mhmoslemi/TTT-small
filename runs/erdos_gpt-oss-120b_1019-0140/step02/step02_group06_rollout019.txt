    best_h = h.copy()

    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # compute gradient for max shift
        idx_max = np.argmax(v)
        k = idx_max - (len(h) - 1)  # offset = n-1
        g = np.zeros_like(h)

            head = slice(k, n)

            g[head] -= h[tail]

            tail = slice(k2, n)
            head = slice(0, n - k2)
            g[tail] += 1.0 - h[head]
            g[head] -= h[tail]

        g -= np.mean(g)

def project_sum_and_bounds(x, target):
    # Clip values to [0,1]
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # Simple proportional adjustment
    # If diff >0, we need to increase some entries not at 1.
    # If diff <0, we need to decrease some entries not at 0.
    # Create mask of eligible indices.
    if diff > 0:
        mask = x < 1.0
        # If no eligible, just return
        if not np.any(mask):

import time
import numpy as np

def compute_c5(h):
    """Return max overlap integral C5 for discretized h."""
    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full')
    return np.max(v) * dx

def project_sum_and_bounds(x, target):
    """Clip x to [0,1] and enforce sum(x) == target (exactly)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # first try a uniform correction
    x += diff / len(x)
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # fallback: distribute remaining diff greedily
    if diff > 0:
        # need to increase some entries
        for i in range(len(x)):
            if diff <= 0:
                break
            if x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
    else:
        # need to decrease some entries
        for i in range(len(x)):
            if diff >= 0:
                break
            if x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
    return np.clip(x, 0.0, 1.0)

def upsample_h(base, n):
    """Linearly upsample base array to length n."""
    n0 = len(base)
    if n0 == n:
        return base.copy()
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x0, base)

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0

            no_improve += 1
