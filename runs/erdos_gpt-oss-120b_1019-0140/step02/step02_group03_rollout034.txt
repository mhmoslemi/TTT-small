import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while forcing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    """Equality: sum(h) == target."""
    def __init__(self, n, target):
        self.n = n
        self.target = target
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx for a given shift k."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        t = x[-1]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return t - overlap * self.dx

def objective_t(x):
    """Objective for SLSQP: minimise the slack variable t."""
    return x[-1]

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500):
    """Run a single SLSQP optimisation from (h_start, t_start)."""

                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res
