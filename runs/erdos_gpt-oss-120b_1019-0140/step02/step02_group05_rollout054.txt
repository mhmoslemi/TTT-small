                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Clip to [0,1] and enforce exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
        # residual fix
        if abs(diff) > 1e-12:
            for i in range(len(h)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
    return np.clip(h, 0.0, 1.0)

def avg_gradient_worst_shifts(h):
    """Average gradient of all shifts attaining the maximal (unscaled) overlap."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    idxs = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n, dtype=float)
    for idx in idxs:
        k = idx - (n - 1)
        g = np.ones(n, dtype=float)
        if k >= 0:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[: n - kk] -= h[kk:]
            g[kk:] -= h[: n - kk]
        grad += g
    if idxs.size:
        grad /= idxs.size
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target_sum = n / 2.0
    cur_h = project_h(h0, target_sum)
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # ---- simulated annealing ---------------------------------------
    T0 = 0.05
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)
        r = rng.random()
        # swap move
        if r < 0.30:
            i, j = rng.choice(n, size=2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        # transfer move
        elif r < 0.60:
            i, j = rng.choice(n, size=2, replace=False)
            donor, recv = i, j
            max_tr = min(cur_h[donor], 1.0 - cur_h[recv])
            if max_tr <= 0:
                max_tr = min(cur_h[recv], 1.0 - cur_h[donor])
                if max_tr <= 0:
                    continue
                donor, recv = recv, donor
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        # gradient move
        elif r < 0.90:
            grad = avg_gradient_worst_shifts(cur_h)
            max_step = 0.20
            step = -grad
            mabs = np.max(np.abs(step))
            if mabs > max_step:
                step = step * (max_step / mabs)
            cand = project_h(cur_h + step, target_sum)
        # block swap
        else:
            max_len = min(6, n // 10)
            if max_len < 2:
                continue
            block_len = rng.integers(2, max_len + 1)
            start1 = rng.integers(0, n - block_len + 1)
            start2 = rng.integers(0, n - block_len + 1)
            if not (start2 + block_len <= start1 or start1 + block_len <= start2):
                continue
            cand = cur_h.copy()
            b1 = cand[start1:start1 + block_len].copy()
            b2 = cand[start2:start2 + block_len].copy()
            cand[start1:start1 + block_len] = b2
            cand[start2:start2 + block_len] = b1
            cand = np.clip(cand, 0.0, 1.0)

        new_val = compute_c5(cand)
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0:
            if rng.random() < np.exp(-(new_val - cur_val) / T):
                accept = True
        if accept:
            cur_h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                # continue searching even if we cross target

    # ---- smoothing passes -----------------------------------------
    if time.time() - start < budget_s - 5.0:
        kernel = np.array([0.25, 0.5, 0.25], dtype=float)
        for _ in range(20):
            sm = np.convolve(best_h, kernel, mode="same")
            sm = np.clip(sm, 0.0, 1.0)
            sm = project_h(sm, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
            else:
                break

    # ---- final refinement with SLSQP -------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 10.0:
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0

        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

        def make_ineq(k):
            def fun(x, k=k):
                h = x[:n]
                t = x[-1]
                if k >= 0:
                    overl = np.dot(h[: n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[: n - kk])
                return overl * dx - t
            return fun

        constraints = [{"type": "eq", "fun": sum_eq}]
        for k in range(-n + 1, n):
            if k == 0:
                continue
            constraints.append({"type": "ineq", "fun": make_ineq(k)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_h(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # final safety
    best_h = project_h(best_h, n / 2.0)
    best_val = compute_c5(best_h)
    return best_h, best_val, n
