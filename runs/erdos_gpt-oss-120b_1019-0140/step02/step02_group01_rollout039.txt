import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def binary_hill_climb_for_n(n, rng, budget, global_start):
    """Binary (0/1) hill‑climbing with swaps. Returns best (h, c5)."""
    start = time.time()
    M = n // 2
    best_h = None
    best_c5 = np.inf
    # each restart: random binary pattern, then swaps
    while time.time() - start < budget:
        # random binary init
        h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, M, replace=False)
        h[ones_idx] = 1.0
        cur_c5 = compute_c5(h)

        # maintain index lists for fast random swap selection
        ones_list = list(np.where(h == 1)[0])
        zeros_list = list(np.where(h == 0)[0])

        max_swaps = 2000
        swaps = 0
        # swap loop
        while swaps < max_swaps and time.time() - start < budget:
            if not ones_list or not zeros_list:
                break
            i_idx = rng.integers(len(ones_list))
            j_idx = rng.integers(len(zeros_list))
            i = ones_list[i_idx]
            j = zeros_list[j_idx]

            # perform swap
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)

            if new_c5 + 1e-12 < cur_c5:
                cur_c5 = new_c5
                # accept: update index lists
                ones_list[i_idx] = j
                zeros_list[j_idx] = i
                if cur_c5 < 0.3808:
                    break
            else:
                # revert swap
                h[i] = 1.0
                h[j] = 0.0
            swaps += 1

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            if best_c5 < 0.3808:
                break

        # early exit if remaining overall time low
        if time.time() - global_start > budget * 0.9:
            break

    return best_h, best_c5

def continuous_refine(h, rng, budget, start_time):
    """Small random mass transfers to further reduce C5."""
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    while time.time() - start < budget and time.time() - start_time < budget:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_transfer = min(1.0 - best_h[i], best_h[j])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        # apply transfer
        orig_i = best_h[i]
        orig_j = best_h[j]
        best_h[i] = orig_i + delta
        best_h[j] = orig_j - delta
        c5_new = compute_c5(best_h)
        if c5_new + 1e-12 < best_c5:
            best_c5 = c5_new
            if best_c5 < 0.3808:
                break
        else:
            # revert
            best_h[i] = orig_i
            best_h[j] = orig_j
    return best_h, best_c5

def _project_to_constraints(h, M):
    """Project onto 0 ≤ h ≤ 1 and sum h = M (simple iterative correction)."""
    for _ in range(10):
        cur = h.sum()
        diff = M - cur
        if abs(diff) < 1e-12:
            break
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h
