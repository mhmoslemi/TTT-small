import time
import numpy as np

def compute_c5(h):
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraints(h, target_sum):
    """Clip to [0,1] and enforce sum == target_sum (simple heuristic)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # distribute diff over non‑boundary entries; repeat a few times if needed
    for _ in range(15):
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # no interior points, distribute uniformly (will be clipped)
            inc = diff / n
            h += inc
            h = np.clip(h, 0.0, 1.0)
        else:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return h

def sa_optimize(h0, target_sum, budget, rng):
    """Simple simulated‑annealing optimizer."""
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    start = time.time()
    # move probabilities
    p_swap = 0.30
    p_transfer = 0.55
    p_smooth = 0.15
    T0 = 0.05  # initial temperature
    kernel = np.array([0.25, 0.5, 0.25])
    while True:
        elapsed = time.time() - start
        if elapsed >= budget:
            break
        T = T0 * max(0.0, 1.0 - elapsed / budget)
        r = rng.random()
        # ----- swap -----
        if r < p_swap:
            i, j = rng.choice(n, 2, replace=False)
            cand = h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        # ----- transfer -----
        elif r < p_swap + p_transfer:
            i, j = rng.choice(n, 2, replace=False)
            # direction that can move mass
            if h[i] > 0.0 and h[j] < 1.0:
                donor, recv = i, j
            elif h[j] > 0.0 and h[i] < 1.0:
                donor, recv = j, i
            else:
                continue
            max_tr = min(h[donor], 1.0 - h[recv])
            if max_tr <= 0:
                continue
            delta = rng.random() * max_tr
            cand = h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        # ----- smooth + noise -----
        else:
            cand = np.convolve(h, kernel, mode="same")
            noise = (rng.random(n) - 0.5) * 0.02
            cand += noise
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_onto_constraints(cand, target_sum)
        new_val = compute_c5(cand)
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                if best_val < 0.3808:

def gradient_refine(h0, target_sum, time_budget, rng):
    """Projected gradient descent on a soft‑max approximation of max overlap."""
    h = h0.copy()
    n = len(h)
    start = time.time()
    alpha = 15.0           # controls softness of the max
    lr = 0.1               # initial step size
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # autocorrelation of h with itself
        autocorr = np.correlate(h, h, mode="full")
        # soft‑max weights favouring small autocorrelation values
        a = -alpha * autocorr
        max_a = np.max(a)
        exp_a = np.exp(a - max_a)
        Z = np.sum(exp_a)
        w = exp_a / Z                      # size 2n‑1, sum(w)=1
        # gradient of the smooth objective
        grad = np.zeros_like(h)
        shift_center = n - 1
        for idx, weight in enumerate(w):
            k = idx - shift_center
            if k == 0:
                grad += weight * 2.0 * h
                continue
            if k > 0:
                # i from 0..n‑k‑1 contributes h[i+k]
                grad[: n - k] += weight * h[k:]
                # i from k..n‑1 contributes h[i‑k]
                grad[k:] += weight * h[: n - k]
            else:
                kk = -k
                grad[kk:] += weight * h[: n - kk]
                grad[: n - kk] += weight * h[kk:]
        # descent step
        cand = h - lr * grad
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_constraints(cand, target_sum)
        cur_val = compute_c5(h)
        new_val = compute_c5(cand)
        if new_val < cur_val:
            h = cand
            # optionally increase step size a little
            lr = min(lr * 1.05, 0.3)
        else:
            lr *= 0.5
            if lr < 1e-5:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the given time budget."""
    start_all = time.time()
    rng = np.random.default_rng(seed)

    # ---- choose grid size -------------------------------------------------
    n = 200                     # a modestly fine discretisation
    target_sum = n / 2.0        # ∑h = n/2 (so that integral equals 1)

    # ---- build initial candidates -----------------------------------------
    starts = []
    # use provided construction, interpolated to the new grid size
    if "initial_h_values" in globals():
        init0 = np.asarray(initial_h_values, dtype=float)
        n0 = init0.size
        xs0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        init_interp = np.interp(xs, xs0, init0)
        init_interp = project_onto_constraints(init_interp, target_sum)
        starts.append(init_interp)

    # add a few random starts
    for _ in range(3):
        v = rng.random(n)
        v = v / np.sum(v) * target_sum
        v = project_onto_constraints(v, target_sum)
        starts.append(v)

    best_h = None
    best_val = float('inf')
    elapsed_total = time.time() - start_all
    remaining_time = budget_s - elapsed_total
    # allocate ~90 % of the remaining time to SA passes
    time_for_sa = remaining_time * 0.9
    per_start = time_for_sa / max(1, len(starts))

    for h0 in starts:
        now = time.time()
        time_left = budget_s - (now - start_all)
        if time_left <= 5.0:
            break
        allow = min(per_start, time_left * 0.9)
        h_opt, val_opt = sa_optimize(h0, target_sum, allow, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # ---- final gradient‑based refinement ---------------------------------
    time_left = budget_s - (time.time() - start_all)
    if best_h is not None and time_left > 5.0:
        best_h = gradient_refine(best_h, target_sum, time_left, rng)
        best_val = compute_c5(best_h)

    # ---- final safety projection -----------------------------------------
    best_h = project_onto_constraints(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, n
