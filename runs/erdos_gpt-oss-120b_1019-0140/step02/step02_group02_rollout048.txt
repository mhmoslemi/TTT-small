def gradient_c5(h):
    n = h.shape[0]
    dx = 2./n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    G = np.zeros_like(h)
    if k >= 0:
        # slice h0 = h[:n - k], h1 = h[k:]
        h0 = h[:n - k]
        h1 = h[k:]
        G[:n - k] += (1.0 - h1) * dx
        G[k:] += -h0 * dx
    else:
        d = -k
        h0 = h[d:] # h_{i+d}
        h1 = h[:n - d] # h_i
        # G[d:] (i+d) gets (1 - h_i) term
        G[d:] += (1.0 - h1) * dx
        # G[:n-d] gets -h_{i+d}
        G[:n - d] += -h0 * dx
    # G now gradient of C(k) w.r.t. h (i.e., C(k) = c5). For max shift.
    # To ensure sum invariance, subtract mean.
    G -= G.mean()
    return G, k, corr[max_idx] * dx

def project_bounds_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Simple iterative water-filling
    # We'll adjust only those not at bounds.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase some entries not at 1
            free = np.where(h < 1 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        inc = diff / free.size
        # Apply inc limited by bounds
        if diff > 0:
            h[free] = np.minimum(h[free] + inc, 1.0)
        else:
            h[free] = np.maximum(h[free] + inc, 0.0)  # inc negative
        diff = target - h.sum()
    # final maybe clip again
    h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral C5 for a discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def gradient_c5(h):
    """
    Gradient of C5 (with respect to h) at the shift achieving the maximum.
    Returns (grad, shift, value).
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)            # shift corresponding to max
    grad = np.zeros_like(h)

    if k >= 0:
        # h[i] * (1 - h[i+k])
        front = h[: n - k]
        back = h[k:]
        grad[: n - k] += (1.0 - back) * dx
        grad[k:]      += -front * dx
    else:
        d = -k
        front = h[d:]                 # h[i+d]
        back = h[: n - d]             # h[i]
        grad[d:]      += (1.0 - back) * dx
        grad[: n - d] += -front * dx

    # Remove component along the all‑ones direction to keep the sum unchanged
    grad -= grad.mean()
    val = corr[max_idx] * dx
    return grad, k, val

def project_bounds_sum(h, target):
    """Clip h to [0,1] and adjust to satisfy sum(h)=target (simple water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # iterative redistribution
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        inc = diff / free.size
        if diff > 0:
            h[free] = np.minimum(h[free] + inc, 1.0)
        else:
            h[free] = np.maximum(h[free] + inc, 0.0)
        diff = target - h.sum()
    # any leftover tiny discrepancy is forced on a free element if possible
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - (1.0 if diff > 0 else 0.0)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def subgradient_descent(h_start, target, max_iters, start_alpha, deadline, rng):
    """
    Simple sub‑gradient descent on C5 (max overlap) with projection onto
    [0,1]^n and the equality sum constraint.
    """
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    n = h.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline - 0.5:
            break
        grad, _, cur_val = gradient_c5(h)
        # Linear decay of step size
        alpha = start_alpha * (1.0 - it / max_iters)
        if alpha < 1e-8:
            break
        # optional small random jitter (helps escape flat regions)
        if rng.random() < 0.02:
            grad += rng.normal(scale=0.001, size=n)
        h = h - alpha * grad
        h = project_bounds_sum(h, target)
        new_val = compute_c5(h)
        if new_val < best_val:
            best_val = new_val
            best_h = h.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

def random_local_improve(h, target, deadline, rng):
    """Pairwise mass‑transfer hill climbing."""
    best_h = h.copy()
    best_val = compute_c5(best_h)
    n = len(h)

    while time.time() < deadline - 0.5:
        src = np.where(best_h > 1e-12)[0]
        dst = np.where(best_h < 1.0 - 1e-12)[0]
        if src.size == 0 or dst.size == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:

        delta = max_transfer * rng.random() * 0.5
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        val = compute_c5(new_h)
        if val < best_val:
            best_val = val
            best_h = new_h
            if best_val < 0.3808:
                break
    return best_h, best_val
