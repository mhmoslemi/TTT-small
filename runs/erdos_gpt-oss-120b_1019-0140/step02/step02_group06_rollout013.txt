import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # simple uniform shift
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # greedy fill
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def gradient_descent_refine(h0, time_budget, rng):
    """Simple gradient descent with smooth max as previous approach."""
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 20.0
    alpha0 = 0.2
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # increase smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w /= w.sum()
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # zero-sum direction
        g -= np.mean(g)
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
    # final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5

def ga_optimize(initial_h, rng, time_budget):
    n = len(initial_h)
    target = n / 2.0
    POP_SIZE = 40
    ELITE_COUNT = 2
    TOURN_SIZE = 3
    MUT_RATE = 0.4
    MUT_STRENGTH = 0.05
    # initialize population
    population = []
    # include the provided initial_h (projected)
    h0 = project_sum_and_bounds(initial_h, target)
    population.append(h0)
    while len(population) < POP_SIZE:
        child = rng.uniform(0.0, 1.0, n)
        child = project_sum_and_bounds(child, target)
        population.append(child)
    # evaluation
    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    start = time.time()
    # track improvement streak
    stagnation = 0
    while time.time() - start < time_budget:
        # check time
        elapsed = time.time() - start
        remaining = time_budget - elapsed
        if remaining < 0.1:
            break
        # elite preservation
        elite_idx = np.argsort(fitness)[:ELITE_COUNT]
        new_pop = [population[i].copy() for i in elite_idx]
        # generate rest
        while len(new_pop) < POP_SIZE:
            # tournament selection for parent1
            cand = rng.integers(0, POP_SIZE, size=TOURN_SIZE)
            p1 = cand[np.argmin(fitness[cand])]
            # tournament selection for parent2
            cand = rng.integers(0, POP_SIZE, size=TOURN_SIZE)
            p2 = cand[np.argmin(fitness[cand])]
            parent1 = population[p1]
            parent2 = population[p2]
            # uniform crossover
            mask = rng.random(n) < 0.5
            child = np.where(mask, parent1, parent2)
            # mutation
            if rng.random() < MUT_RATE:
                child = child + rng.normal(0.0, MUT_STRENGTH, n)
                child = np.clip(child, 0.0, 1.0)
            child = project_sum_and_bounds(child, target)
            new_pop.append(child)
        population = new_pop
        fitness = np.array([compute_c5(ind) for ind in population])
        idx = np.argmin(fitness)
        if fitness[idx] + 1e-12 < best_c5:
            best_c5 = fitness[idx]
            best_h = population[idx].copy()
            stagnation = 0
        else:
            stagnation += 1
        # early stop if no improvement for many generations
        if stagnation >= 200:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    # Determine dimensionality
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
    else:
        # create a simple start: half ones, half zeros (integral satisfied)
        n = 80
        half = n // 2
        arr = np.zeros(n, dtype=float)
        arr[:half] = 1.0
        h_initial = arr
    n = len(h_initial)
    target = n / 2.0
    # allocate time between GA and gradient descent
    total_time = budget_s - (time.time() - start)
    # Give 80% to GA, 20% to refinement
    ga_time = total_time * 0.80
    refine_time = max(0.0, total_time - ga_time)
    # Run GA
    best_h, best_c5 = ga_optimize(h_initial, rng, ga_time)
    # Run local refinement if time left
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 0.1:
        best_h, best_c5 = gradient_descent_refine(best_h, remaining, rng)
    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral times dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # uniform adjustment
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # greedy fill/empty
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def gradient_descent_refine(h0, time_budget, rng):
    """Smooth‑max gradient descent (as in previous best program)."""
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 20.0
    alpha0 = 0.2
    no_improve = 0

    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w /= w.sum()

        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        g -= np.mean(g)  # keep total mass unchanged
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5

def ga_optimize(initial_h, rng, time_budget):
    """Simple genetic algorithm."""
    n = len(initial_h)
    target = n / 2.0
    POP_SIZE = 40
    ELITE_COUNT = 2
    TOURN_SIZE = 3
    MUT_RATE = 0.4
    MUT_STRENGTH = 0.05

    # --- initialize population ---
    population = []
    population.append(project_sum_and_bounds(initial_h, target))

    # deterministic seeds: block and alternating patterns
    block = np.zeros(n)
    block[:int(target)] = 1.0
    population.append(project_sum_and_bounds(block, target))

    alt = np.tile([1.0, 0.0], n // 2)
    population.append(project_sum_and_bounds(alt, target))

    while len(population) < POP_SIZE:
        cand = rng.uniform(0.0, 1.0, n)
        population.append(project_sum_and_bounds(cand, target))

    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    start = time.time()
    stagnation = 0

    # --- evolution loop ---
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        if time_budget - elapsed < 0.1:
            break

        elite_idx = np.argsort(fitness)[:ELITE_COUNT]
        new_pop = [population[i].copy() for i in elite_idx]

        while len(new_pop) < POP_SIZE:
            # tournament selection for parent 1
            cand = rng.integers(0, POP_SIZE, size=TOURN_SIZE)
            p1 = cand[np.argmin(fitness[cand])]
            # tournament selection for parent 2
            cand = rng.integers(0, POP_SIZE, size=TOURN_SIZE)
            p2 = cand[np.argmin(fitness[cand])]
            parent1 = population[p1]
            parent2 = population[p2]

            # uniform crossover
            mask = rng.random(n) < 0.5
            child = np.where(mask, parent1, parent2)

            # mutation
            if rng.random() < MUT_RATE:
                child = child + rng.normal(0.0, MUT_STRENGTH, n)
                child = np.clip(child, 0.0, 1.0)
            child = project_sum_and_bounds(child, target)

            new_pop.append(child)

        population = new_pop
        fitness = np.array([compute_c5(ind) for ind in population])
        idx = np.argmin(fitness)

        if fitness[idx] + 1e-12 < best_c5:
            best_c5 = fitness[idx]
            best_h = population[idx].copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation >= 200:
            break

    return best_h, best_c5
