    rng = np.random.default_rng(seed)
    start_time = time.time()

        return max(0.0, budget_s - (time.time() - start_time))

    n = 512 # may adjust to 256 if time limited. We'll set 512 but we can fallback if time left insufficient.

    # Initialize h.

    # Determine number of iterations based on time.
    # Approx iteration cost for n=512 maybe 0.001 seconds? We can't know. Use dynamic: while time_left > some threshold and iterations < max_iter.
    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # Set step0 = 1.0 perhaps, momentum = 0.9
    h = h0.copy()
    momentum = np.zeros_like(h)
    step0 = 1.0
    beta = 0.9
    it = 0
    # set max_iters maybe 50000.
    while time_left() > 20.0 and it < 50000:
        # compute correlation and find max indices
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it + 1)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, n/2.0)
        # compute score
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1

    # after gradient: optional polishing step using SLSQP from scipy.optimize if available.
    # Since we have SciPy, we could try a short SLSQP to tighten.
    try:
        from scipy.optimize import minimize
        x0 = np.concatenate([best_h, [best_score]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        def eq_sum(x):
            return np.sum(x[:n]) - n/2.0

            return t - corr

                       constraints=cons, options={'maxiter':200, 'ftol':1e-12, 'disp':False})
        if res.success:

            sc_opt = compute_c5(h_opt)
            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()
    except Exception:
        pass

    # Choose top n/2 indices.
    idx_desc = np.argsort(-best_h) # descending

    h_bin[idx_desc[:n//2]] = 1.0
    bin_score = compute_c5(h_bin)

        best_score = bin_score
        best_h = h_bin.copy()

        cur_bin[idx_desc[:n//2]] = 1.0

    cur_score = compute_c5(cur_bin)

    init_temp = 0.001 * best_score # scale
    temp = init_temp
    decay = 0.9999
    # loop while time left > small.
    while time_left() > 2.0:

        ones = np.flatnonzero(cur_bin == 1.0)
        zeros = np.flatnonzero(cur_bin == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_bin[i] = 0.0
        cur_bin[j] = 1.0
        new_score = compute_c5(cur_bin)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # acceptance prob
            if temp > 0:
                accept = rng.random() < np.exp(-delta / temp)
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_bin.copy()
        else:
            # revert
            cur_bin[i] = 1.0
            cur_bin[j] = 0.0
        temp *= decay
        # occasionally reset temperature if too low
        if temp < 1e-12:
            temp = init_temp

        # if we achieve target bound break
        if best_score <= 0.3808:
            break

    # final projection (ensure sum)
    best_h = project_onto_box_sum(best_h, n/2.0) # but if binary ensures sum
    best_score = compute_c5(best_h)

    return best_h, best_score, n

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced 0/1 vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max-correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # ------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------
    n = 512                     # number of sample points
    target_bound = 0.3808       # we stop if we reach this value

    # ------------------------------------------------------------
    # Initial solution
    # ------------------------------------------------------------
    init = None
    if "initial_h_values" in globals():
        try:
            init_full = np.asarray(initial_h_values, dtype=float)
            init = upsample(init_full, n)
        except Exception:
            init = None
    if init is None:
        init = rng.random(n)
    h = project_onto_box_sum(init, n / 2.0)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # ------------------------------------------------------------
    # Phase 1: Projected momentum gradient descent
    # ------------------------------------------------------------
    momentum = np.zeros_like(h)
    beta = 0.9
    step0 = 1.0
    it = 0
    last_improve = 0
    max_iters = 20000
    while time_left() > 30.0 and it < max_iters:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it + 1)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            last_improve = it
            if best_score <= target_bound:
                break
        h = h_new
        it += 1
        if it - last_improve > 2000:
            break

    # ------------------------------------------------------------
    # Optional short SLSQP refinement (if time permits)
    # ------------------------------------------------------------
    if time_left() > 5.0:
        try:
            from scipy.optimize import minimize
            x0 = np.concatenate([best_h, [best_score]])
            bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

            def eq_sum(x):
                return np.sum(x[:n]) - n / 2.0

            def ineq_fun(x):
                hh = x[:n]
                t = x[-1]
                corr = np.correlate(hh, 1.0 - hh, mode='full') * (2.0 / n)
                return t - corr

            cons = [{'type': 'eq', 'fun': eq_sum},
                    {'type': 'ineq', 'fun': ineq_fun}]

            res = minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 250, 'ftol': 1e-12, 'disp': False})
            if res.success:
                h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
                sc_opt = compute_c5(h_opt)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_h = h_opt.copy()
        except Exception:
            pass

    # ------------------------------------------------------------
    # Phase 2: Convert to binary and local search (simulated annealing)
    # ------------------------------------------------------------
    # Binary projection: top n/2 entries become 1
    idx_desc = np.argsort(-best_h)
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx_desc[:n // 2]] = 1.0
    cur_h = h_bin.copy()
    cur_score = compute_c5(cur_h)
    if cur_score < best_score:
        best_score = cur_score
        best_h = cur_h.copy()

    # Simulated annealing parameters
    init_temp = max(1e-5, 0.001 * best_score)
    temp = init_temp
    decay = 0.9997

    while time_left() > 2.0:
        # pick a random 1 and a random 0 to swap
        ones = np.flatnonzero(cur_h == 1.0)
        zeros = np.flatnonzero(cur_h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if temp > 0:
                accept = rng.random() < np.exp(-delta / temp)
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                if best_score <= target_bound:
                    break
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        temp = max(1e-12, temp * decay)

    # final projection (protect against numerical drift)
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
