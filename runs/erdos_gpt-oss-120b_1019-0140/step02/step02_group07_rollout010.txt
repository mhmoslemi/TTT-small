    init_temp = 1e-2
    final_temp = 1e-6
    total_iters = 500000

    it = 0
    while time.time() < time_end and it < total_iters:
        # choose random 1 and 0 indices
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1

        it += 1

def solve_milp(n, time_budget):
    import cvxpy as cp, numpy as np, time
    dx = 2.0 / n
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == n//2, t >= 0, t <= 1]
    for l in range(1, n):
        length = n - l
        i_idx = np.arange(length)
        z = cp.Variable(length)
        constraints += [
            z <= x[i_idx],
            z <= x[i_idx + l],
            z >= x[i_idx] + x[i_idx + l] - 1,
            z >= 0,
            cp.sum(z) >= t / dx

    prob = cp.Problem(cp.Maximize(t), constraints)

            return None, None, False

    x_val = np.round(x.value).astype(float)

    diff = n/2 - np.sum(x_val)

        # adjust by toggling nearest entries

def continuous_anneal(start_h, end_time, rng):
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 500000
    it = 0
    while time.time() < end_time and it < max_iter:
        # pick i where h < 1 and j where h > 0
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        # propose delta scaled by temperature
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = new_c5
        it += 1
    # final fix sum
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        # residual fix:
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_c5

import time
import math
import numpy as np

def compute_c5(arr):
    """Approximate C5 via discrete correlation."""
    a = np.asarray(arr, dtype=float)
    n = a.shape[0]
    corr = np.correlate(a, 1 - a, mode='full')
    return np.max(corr) * (2.0 / n)

def continuous_anneal(start_h, end_time, rng):
    """Simulated annealing on continuous values (preserving sum)."""
    n = len(start_h)
    target = n / 2.0
    h = start_h.copy()
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    init_temp, final_temp = 5e-3, 1e-7
    max_it = 500_000
    it = 0
    while time.time() < end_time and it < max_it:
        inc = np.where(h < 1 - 1e-12)[0]
        dec = np.where(h > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        T = init_temp * (1 - it / max_it) + final_temp * (it / max_it)
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new = compute_c5(h_new)
        dE = new - cur
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h, cur = h_new, new
            if new < best - 1e-12:
                best_h, best = h_new.copy(), new
        it += 1
    # exact sum correction
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target - np.sum(best_h)
    return best_h, best

def binary_anneal(start_vec, end_time, rng):
    """Simulated annealing on binary vectors (swap moves, preserving sum)."""
    n = len(start_vec)
    target = n // 2
    h = start_vec.copy()
    cur = compute_c5(h.astype(float))
    best_h, best = h.copy(), cur
    init_temp, final_temp = 1e-2, 1e-6
    max_it = 400_000
    it = 0
    while time.time() < end_time and it < max_it:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        dE = cand_c5 - cur
        T = init_temp * (1 - it / max_it) + final_temp * (it / max_it)

            h, cur = cand, cand_c5
            if cand_c5 < best - 1e-12:
                best_h, best = cand.copy(), cand_c5
        it += 1
    return best_h.astype(float), best
