def binary_local_search(h, time_limit, rng):
    # h is binary vector with sum = n/2
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    # we can compute min autocorr directly for speed: but compute_c5 fine.
    # We'll precompute indices of ones, zeros.
    one_idx = np.where(best_h == 1)[0]
    zero_idx = np.where(best_h == 0)[0]
    while time.time() < time_limit:
        # attempt a random swap
        i = rng.choice(one_idx)
        j = rng.choice(zero_idx)
        h_candidate = best_h.copy()
        h_candidate[i] = 0
        h_candidate[j] = 1
        score = compute_c5(h_candidate)
        if score < best_score - 1e-12:
            best_h = h_candidate
            best_score = score
            # update indices
            one_idx = np.where(best_h == 1)[0]
            zero_idx = np.where(best_h == 0)[0]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # set resolutions
    n_small = 128
    n_large = 256

    # candidate generation for small scale
    candidates = []

    # Add initial if exists
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # downsample or upsample to n_small as needed
        if init.shape[0] != n_small:
            init_small = downsample(init, n_small) if init.shape[0] > n_small else upsample(init, n_small)
        else:
            init_small = init.copy()
        init_small = project_onto_box_sum(init_small, n_small/2.0)
        candidates.append(init_small)

    # Add thue-morse binary pattern
    candidates.append(thue_morse(n_small))

    # Add few random binary balanced vectors
    for _ in range(4):

    for _ in range(4):

        candidates.append(project_onto_box_sum(v, n_small/2.0))

    # Keep best solution
    best_h_small = None
    best_score_small = float('inf')

    # Phase 1: SLSQP on small scale
    while candidates and time_left() > 8.0:
        cand = candidates.pop(0)
        max_iter = 600  # moderate iteration

        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=max_iter)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()

    # Additional random starts
    while time_left() > 6.0:

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation of arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    """Linear interpolation from a larger array to a smaller size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of the max-correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    """Run a short SLSQP on a small‑scale problem to improve a candidate."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def binary_local_search(h, end_time, rng):
    """Swap‑based local search on binary vectors (exact sum constraint)."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    one_idx = np.where(best_h == 1)[0]
    zero_idx = np.where(best_h == 0)[0]
    while time.time() < end_time:
        if one_idx.size == 0 or zero_idx.size == 0:
            break
        i = rng.choice(one_idx)
        j = rng.choice(zero_idx)
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        sc = compute_c5(cand)
        if sc < best_score - 1e-12:
            best_h = cand
            best_score = sc
            one_idx = np.where(best_h == 1)[0]
            zero_idx = np.where(best_h == 0)[0]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_small = 128
    n_large = 256

    # ----- Phase 1: generate small‑scale candidates -----
    candidates = []

    if 'initial_h_values' in globals():

        if init_full.shape[0] != n_small:

        init_small = project_onto_box_sum(init_small, n_small / 2.0)
