    """Maximum overlap integral C5 for a given (float) array h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Project vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


# ---------- Binary optimizer ----------
def optimize_binary_n(n, rng, time_limit):
    """Search for a binary (0/1) vector of length n with sum=n/2 minimizing C5."""
    dx = 2.0 / n
    half = n // 2
    # initial vector
    if "initial_h_values" in globals():
        hv = np.array(initial_h_values, dtype=float)
        if len(hv) == n:
            idx = np.argsort(-hv)[:half]
            cur_h = np.zeros(n, dtype=float)
            cur_h[idx] = 1.0
        else:
            cur_h = np.zeros(n, dtype=float)
            cur_h[:half] = 1.0
            rng.shuffle(cur_h)
    else:
        cur_h = np.zeros(n, dtype=float)
        cur_h[:half] = 1.0
        rng.shuffle(cur_h)

    cur_val = compute_c5_vec(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    iter_without_improve = 0
    restart_after = 200

    while time.time() < time_limit:
        # correlation to locate the worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        k_idx = np.argmax(corr)
        k_shift = k_idx - (n - 1)

        # overlapping indices for this shift
        if k_shift >= 0:
            i_range = np.arange(0, n - k_shift, dtype=int)
            i2_range = i_range + k_shift
        else:
            i_range = np.arange(-k_shift, n, dtype=int)
            i2_range = i_range + k_shift

        # positions that contribute to the current max (1 then 0)
        mask_mis = (cur_h[i_range] > 0.5) & (cur_h[i2_range] < 0.5)
        mismatches = i_range[mask_mis]

        # positions that would fix it if turned to 1 (0 then 1)
        mask_fix = (cur_h[i_range] < 0.5) & (cur_h[i2_range] > 0.5)
        potentials = i_range[mask_fix]

        moved = False
        if mismatches.size and potentials.size:
            i_swap = rng.choice(mismatches)
            j_swap = rng.choice(potentials)
            cur_h[i_swap] = 0.0
            cur_h[j_swap] = 1.0
            new_val = compute_c5_vec(cur_h, dx)
            if new_val < cur_val - 1e-12:
                cur_val = new_val
                iter_without_improve = 0
                moved = True
                if new_val < best_val - 1e-12:
                    best_h = cur_h.copy()
                    best_val = new_val
            else:
                # revert
                cur_h[i_swap] = 1.0
                cur_h[j_swap] = 0.0
                iter_without_improve += 1
        else:
            # random swap fallback
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
            if ones.size == 0 or zeros.size == 0:
                break
            i_swap = rng.choice(ones)
            j_swap = rng.choice(zeros)
            cur_h[i_swap] = 0.0
            cur_h[j_swap] = 1.0
            new_val = compute_c5_vec(cur_h, dx)
            if new_val < cur_val - 1e-12:
                cur_val = new_val
                iter_without_improve = 0
                moved = True
                if new_val < best_val - 1e-12:
                    best_h = cur_h.copy()
                    best_val = new_val
            else:
                cur_h[i_swap] = 1.0
                cur_h[j_swap] = 0.0
                iter_without_improve += 1

        if not moved and iter_without_improve >= restart_after:
            # random restart of current search
            cur_h = np.zeros(n, dtype=float)
            cur_h[:half] = 1.0
            rng.shuffle(cur_h)
            cur_val = compute_c5_vec(cur_h, dx)
            iter_without_improve = 0

    return best_h, best_val


# ---------- Continuous refinement ----------
def refine_continuous(cur_h, dx, target_sum, time_limit, rng):
    """Small random mass‑transfer refinement on a possibly fractional vector."""
    cur_h = cur_h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.05
    n = len(cur_h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Candidate problem sizes (even numbers)
    n_candidates = [80, 96, 112, 128]

    # Prefer the size of any provided initial construction
    if "initial_h_values" in globals():
        hv = np.array(initial_h_values, dtype=float)
        n0 = len(hv)
        if n0 not in n_candidates:
            n_candidates.insert(0, n0)
        else:
            n_candidates = [n0] + [n for n in n_candidates if n != n0]

    best_h = None
    best_val = float("inf")
    best_n = None

    # Loop over candidate discretizations
    for n in n_candidates:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 30.0:
            break
        per_n_budget = min(300.0, time_left * 0.5)
        time_limit = time.time() + per_n_budget

        h_n, val_n = optimize_binary_n(n, rng, time_limit)
        if val_n < best_val:
            best_h = h_n
            best_val = val_n
            best_n = n
        # early exit if we already beat target
        if best_val < 0.3808:
            break

    # If nothing was found, fall back to a simple construction
    if best_h is None:
        best_n = 80
        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
        best_val = compute_c5_vec(best_h, 2.0 / best_n)

    # Final continuous refinement with remaining time
    time_left = budget_s - (time.time() - start_time)
    if time_left > 10.0:
        dx = 2.0 / best_n
        target_sum = best_n / 2.0
        refine_limit = time.time() + time_left
        refined_h, refined_val = refine_continuous(best_h, dx, target_sum, refine_limit, rng)
        if refined_val < best_val:
            best_h, best_val = refined_h, refined_val

    # Ensure feasibility before returning
    dx = 2.0 / best_n
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, best_n
