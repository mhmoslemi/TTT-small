def search_binary(n, target_sum, time_budget, rng):
    start = time.time()
    best_h = None
    best_O = np.inf
    # Determine number of restarts based on time left
    # We'll run many restarts in loop:
    while time.time() - start < time_budget * 0.9:
        # Initialize h
        if best_h is not None and rng.random() < 0.2:
            # Start from best with small random perturbation: swap few pairs
            h = best_h.copy()
            # perform few random swaps to diversify
            for _ in range(10):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i] = 0
                h[j] = 1
        else:
            # random binary
            indices = rng.choice(n, size=int(target_sum), replace=False)
            h = np.zeros(n, dtype=int)
            h[indices] = 1

        # local hill climb
        # compute O_max of this start
        O_max = np.max(np.correlate(h, 1 - h, mode='full'))
        # keep best overall start
        if O_max < best_O:
            best_O = O_max
            best_h = h.copy()
            # early if already good enough
            if best_O <= 15:
                break

        # perform targeted swaps for some iteration budget
        # we set inner loop max steps based on remaining time portion, say 2000 per restart or until O_max <= 15
        inner_deadline = time.time() + min(0.5, time_budget - (time.time() - start))  # maybe limit each restart.
        # Actually we can allocate inner steps until time runs out.
        while time.time() < inner_deadline:
            # if O_max <= 15 break both loops via flag
            if O_max <= 15:
                best_h = h.copy()
                best_O = O_max
                break
            # compute O_counts
            O_counts = np.correlate(h, 1 - h, mode='full')
            O_max = O_counts.max()
            # get shifts with max O
            max_idxs = np.where(O_counts == O_max)[0]
            # choose random shift idx
            idx = rng.choice(max_idxs)
            k = idx - (n - 1)
            # get I and safe zeros
            # compute array of indices
            idx_arr = np.arange(n)
            j_arr = idx_arr + k
            valid = (j_arr >= 0) & (j_arr < n)
            # contributing ones
            contrib_mask = (h == 1) & valid & (h[j_arr[valid]] == 0)
            I = idx_arr[contrib_mask]
            # safe zeros
            zero_mask = (h == 0) & ((~valid) | (h[j_arr[valid]] == 1))
            Z = idx_arr[zero_mask]
            if I.size > 0 and Z.size > 0:
                i_sel = rng.choice(I)
                j_sel = rng.choice(Z)
                # swap
                h[i_sel] = 0
                h[j_sel] = 1
                # recompute O_max
                O_new = np.max(np.correlate(h, 1 - h, mode='full'))
                if O_new < O_max:
                    O_max = O_new
                else:
                    # revert if not improvement
                    h[i_sel] = 1
                    h[j_sel] = 0
                # update best
                if O_max < best_O:
                    best_O = O_max
                    best_h = h.copy()
                    if best_O <= 15:
                        break
                continue  # next iteration
            # else fallback random swap
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i_swap = rng.choice(ones_idx)
            j_swap = rng.choice(zeros_idx)
            # swap
            h[i_swap] = 0
            h[j_swap] = 1
            O_new = np.max(np.correlate(h, 1 - h, mode='full'))
            if O_new < O_max:
                O_max = O_new
                if O_max < best_O:
                    best_O = O_max
                    best_h = h.copy()
                    if best_O <= 15:
                        break
            else:
                # revert
                h[i_swap] = 1
                h[j_swap] = 0
            # loop again

        if best_O <= 15:
            break

    return best_h, best_O

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n
    half_n = int(round(target_sum))

    # initial best from initial_h or random
    if h0 is not None:
        # project to sum just in case
        h0 = np.clip(h0, 0.0, 1.0)
        # adjust sum to exact target via projection (as before)
        cur_sum = np.sum(h0)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # adjust uniformly
            h0 += diff / n
            h0 = np.clip(h0, 0.0, 1.0)
            # fix residual
            cur_sum = np.sum(h0)
            diff = target_sum - cur_sum
            # small adjustment loop
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h0[i] < 1.0:
                    inc = min(diff, 1.0 - h0[i])
                    h0[i] += inc
                    diff -= inc
                elif diff < 0 and h0[i] > 0.0:
                    dec = min(-diff, h0[i])
                    h0[i] -= dec
                    diff += dec
        best_h = h0.copy()
    else:
        best_h = generate_random_feasible(n, target_sum, rng)

    best_val = compute_c5_vec(best_h, dx)

    # Define improvement function
    def improve_continuous(h, deadline):
        """Targeted local refinement decreasing max overlap."""
        cur_h = h.copy()
        cur_val = compute_c5_vec(cur_h, dx)
        step = 0.1
        eps = 1e-12
        while time.time() < deadline:
            # recompute O_counts each iteration
            O_counts = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            O_max = O_counts.max()
            if O_max < cur_val - 1e-12:
                # update cur_val
                cur_val = O_max
            # early exit if very low
            if O_max <= 15:  # bound target
                break
            idx_max = O_counts.argmax()
            k = idx_max - (n - 1)  # shift relative to zero
            # compute donor and recipient masks
            idx_arr = np.arange(n)
            j_arr = idx_arr + k
            valid = (j_arr >= 0) & (j_arr < n)

            donor_mask = np.zeros(n, dtype=bool)
            recipient_mask = np.zeros(n, dtype=bool)

            # donors: h[i] > eps and (1 - h[j]) > eps for valid
            donor_mask[valid] = (cur_h[idx_arr[valid]] > eps) & ((1.0 - cur_h[j_arr[valid]]) > eps)
            donor_idx = np.where(donor_mask)[0]

            # recipients: h[i] < 1 - eps and (not valid or (1 - h[j]) < eps)
            recipient_mask = (cur_h < 1.0 - eps) & ((~valid) | ((1.0 - cur_h[j_arr[valid]]) < eps))
            recip_idx = np.where(recipient_mask)[0]

            if donor_idx.size == 0 or recip_idx.size == 0:
                # fallback random donor/recipient
                donor_idx = np.where(cur_h > eps)[0]
                recip_idx = np.where(cur_h < 1.0 - eps)[0]
                if donor_idx.size == 0 or recip_idx.size == 0:
                    break

            i = rng.choice(donor_idx)
            j = rng.choice(recip_idx)

            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                step *= 0.5
                continue
            delta = rng.uniform(0.0, max_delta)

            cand = cur_h.copy()
            cand[i] -= delta

import time
import numpy as np

def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target_sum):
    """Project vec onto the hyperplane sum(vec)==target_sum while staying in [0,1]."""
    v = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(v)
    if abs(diff) < 1e-12:
        return v
    v += diff / len(v)
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - np.sum(v)
    for i in range(len(v)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and v[i] < 1.0:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        elif diff < 0 and v[i] > 0.0:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    return v

def generate_random_feasible(n, target_sum, rng):
    """Generate a random vector in [0,1]^n with exact sum target_sum."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

def improve_continuous(h, dx, target_sum, deadline, rng):
    """Targeted local refinement decreasing the max overlap."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.1
    eps = 1e-12
    n = len(cur_h)
    idx_arr = np.arange(n)

    while time.time() < deadline:
        O_counts = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        O_max = O_counts.max()
        if O_max < cur_val - 1e-12:
            cur_val = O_max
        # stop if we already beat the target integer bound (15)
        if O_max <= 15:
            break
        idx_max = O_counts.argmax()
        k = idx_max - (n - 1)   # shift corresponding to the worst overlap

        j_arr = idx_arr + k
        valid = (j_arr >= 0) & (j_arr < n)

        # donors: positions that positively contribute to this overlap
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[valid] = (cur_h[idx_arr[valid]] > eps) & ((1.0 - cur_h[j_arr[valid]]) > eps)
        donor_idx = np.where(donor_mask)[0]

        # recipients: positions where adding mass does NOT increase the worst overlap
        recipient_mask = np.zeros(n, dtype=bool)
        # out‑of‑range indices are safe
        recipient_mask[~valid] = True
        # in‑range indices are safe if the partner already has value ≈1
        recipient_mask[valid] = (1.0 - cur_h[j_arr[valid]]) < eps
        # must also respect the box constraints
        recipient_mask &= (cur_h < 1.0 - eps)

        recip_idx = np.where(recipient_mask)[0]

        if donor_idx.size == 0 or recip_idx.size == 0:
            # fallback to any feasible donor / recipient
            donor_idx = np.where(cur_h > eps)[0]
            recip_idx = np.where(cur_h < 1.0 - eps)[0]
            if donor_idx.size == 0 or recip_idx.size == 0:
                break

        i = rng.choice(donor_idx)
        j = rng.choice(recip_idx)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)

        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)

        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.9

    return cur_h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # initialise best feasible vector
    if h0 is not None:
        best_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        best_h = generate_random_feasible(n, target_sum, rng)

    best_val = compute_c5_vec(best_h, dx)

    # main search – many short refinement phases with occasional restarts
    while time.time() - start_time < budget_s * 0.9:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        # allocate a small budget for this phase (max 5 s or 20 % of remaining)
        phase_time = min(5.0, remaining * 0.2)
        deadline = time.time() + phase_time

        # decide starting point for this phase
        if rng.random() < 0.3:
            h_start = generate_random_feasible(n, target_sum, rng)
        else:
            # perturb current best a little
            h_start = best_h.copy()
            for _ in range(3):
                i, j = rng.integers(n, size=2)
                if i == j:
                    continue
                delta = min(0.02, h_start[i], 1.0 - h_start[j])
                if delta > 0:
                    h_start[i] -= delta
                    h_start[j] += delta

        # improve
        h_ref, val_ref = improve_continuous(h_start, dx, target_sum, deadline, rng)

        if val_ref < best_val - 1e-12:
            best_h, best_val = h_ref, val_ref
            if best_val <= 15 * dx:   # achieved the integer target bound
                break

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    final_val = compute_c5_vec(best_h, dx)

    return best_h.astype(float), final_val, n
