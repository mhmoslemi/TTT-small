def max_shift_gradient(h):
    n = len(h)
    v = np.correlate(h, 1.0 - h, mode='full')
    max_val = v.max()
    # find all indices with max value (within tolerance)
    idxs = np.where(np.abs(v - max_val) < 1e-12)[0]
    idx = idxs[0]  # choose first
    offset = n - 1
    k = idx - offset
    g = np.zeros_like(h)
    if k >= 0:
        tail = slice(0, n - k)
        head = slice(k, n)
        g[tail] = 1.0 - h[head]
        g[head] -= h[tail]
    else:
        k2 = -k
        tail = slice(k2, n)
        head = slice(0, n - k2)
        # In this case, tail corresponds to i where i = i'+k (i' from head)
        g[tail] = 1.0 - h[head]
        g[head] -= h[tail]
    # Ensure zero-mean for sum constraint
    g -= np.mean(g)
    return g

    """Clip to [0,1] and adjust to achieve sum == target."""

    remaining = max(0.0, budget_s - (time.time() - start))
    stage1_budget = remaining * 0.7
    best_h, best_c5 = subgradient_optimize(h0, stage1_budget, seed + 1)

    # refinements with perturbed starts
    while time.time() - start < budget_s - 0.2:
        time_left = max(0.0, budget_s - (time.time() - start))
        sub_budget = min(time_left, 0.15 * remaining)
        pert = best_h + rng.normal(scale=0.02, size=n)
        pert = project_to_feasible(pert, target)
        h_opt, c5_opt = subgradient_optimize(pert, sub_budget,
                                             rng.integers(1, 2**31 - 1))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
