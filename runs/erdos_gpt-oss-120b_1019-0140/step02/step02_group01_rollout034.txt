def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    dx = 2.0 / n

    auto = np.correlate(h, h, mode='full')

    start = time.time()

    while time.time() - start < time_budget:
        iter += 1
        T = max(1e-6, 0.01 * (0.9995 ** iter))
        i, j = rng.choice(n, size=2, replace=False)
        # direction: transfer mass from higher to lower? Not needed, we choose random direction with delta positive to i.
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        # delta scale with T to adapt?
        delta = max_transfer * rng.random()
        # Save old values for later
        old_i = h[i]
        old_j = h[j]
        # Update auto in place:
        # loop over p != i,j
        for p in range(n):

            hp = h[p]
            idx_ip = p - i + n - 1
            idx_pi = i - p + n - 1
            auto[idx_ip] += delta * hp
            auto[idx_pi] += delta * hp
            idx_jp = p - j + n - 1
            idx_pj = j - p + n - 1
            auto[idx_jp] -= delta * hp
            auto[idx_pj] -= delta * hp
        # Handle diagonal i
        idx0 = n - 1
        auto[idx0] += 2 * old_i * delta + delta * delta
        # Handle diagonal j
        auto[idx0] += -2 * old_j * delta + delta * delta
        # Pair (i,j) entries:
        shift_ij = j - i
        idx_ij = shift_ij + n - 1
        idx_ji = -shift_ij + n - 1
        delta_pair = delta * (-old_i + old_j) - delta * delta

        auto[idx_ji] += delta_pair
        # Update h
        h[i] += delta
        h[j] -= delta
        # Compute new_min_auto quickly? Instead of scanning full array each iteration, we can compute min with np.min(auto) O(n). That's fine.
        min_auto = np.min(auto)
        new_c5 = (M - min_auto) * dx

def sa_optimize(h_init, n, time_budget, rng):
    M = n // 2
    dx = 2.0 / n
    h = h_init.copy()
    auto = np.correlate(h, h, mode='full')
    cur_c5 = (M - np.min(auto)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    it = 0
    while True:
        if time.time() - start > time_budget:
            break
        it += 1
        # temperature
        T = max(1e-6, 0.01 * (0.9995 ** it))
        i, j = rng.choice(n, size=2, replace=False)
        # choose direction of mass transfer: we can transfer from higher to lower to stay feasible
        # compute max allowed transfer
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random()
        old_i = h[i]
        old_j = h[j]
        auto_old = auto.copy()
        # update contributions
        # Loop over p != i,j
        for p in range(n):
            if p == i or p == j:
                continue
            hp = h[p]
            idx_ip = p - i + n - 1
            idx_pi = i - p + n - 1
            auto[idx_ip] += delta * hp
            auto[idx_pi] += delta * hp
            idx_jp = p - j + n - 1
            idx_pj = j - p + n - 1
            auto[idx_jp] -= delta * hp
            auto[idx_pj] -= delta * hp
        # diagonal i
        idx0 = n - 1
        auto[idx0] += 2.0 * old_i * delta + delta * delta
        # diagonal j
        auto[idx0] += -2.0 * old_j * delta + delta * delta
        # pair (i,j)
        shift_ij = j - i
        idx_ij = shift_ij + n - 1
        idx_ji = -shift_ij + n - 1
        delta_pair = delta * (-old_i + old_j) - delta * delta
        auto[idx_ij] += delta_pair
        auto[idx_ji] += delta_pair
        # update h
        h[i] = old_i + delta
        h[j] = old_j - delta
        # compute new C5
        min_auto = np.min(auto)
        new_c5 = (M - min_auto) * dx
        accept = False
        if new_c5 <= cur_c5:
            accept = True
        else:
            # acceptance probability
            if rng.random() < math.exp(-(new_c5 - cur_c5) / T):
                accept = True
        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                # optionally store best auto
        else:
            # revert
            h[i] = old_i
            h[j] = old_j
            auto = auto_old
        # occasional random perturbation if no improvement for many steps
        # not needed
    return best_h, best_c5
```

def repair_h(h, n, rng):
    M = n // 2
    # Clip
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur != 0:
        h = h * (M / cur)
    # After scaling, clip again and fix sum
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) > 1e-12:
        # adjust a coordinate not at bound
        if diff > 0:
            candidates = np.where(h < 1 - 1e-12)[0]
        else:
            candidates = np.where(h > 1e-12)[0]
        if len(candidates) == 0:
            i = rng.integers(n)
        else:
            i = rng.choice(candidates)
        h[i] = min(max(h[i] + diff, 0.0), 1.0)
    return h
```

    start_time = time.time()

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # Determine base n
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 != 0:
            n0 = n0 + 1 if n0 < 200 else n0 - 1

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 evaluation using full correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_h(h: np.ndarray, n: int, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and enforce exact sum = n/2."""
    M = n // 2
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur > 0:
        h = h * (M / cur)
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1.0 - 1e-12)[0]
        else:
            cand = np.where(h > 1e-12)[0]
        if cand.size == 0:
            idx = rng.integers(n)
        else:
            idx = rng.choice(cand)
        h[idx] = min(max(h[idx] + diff, 0.0), 1.0)
    return h


def random_feasible_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector with sum = n/2."""
    M = n // 2
    if rng.random() < 0.5:
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    else:
        h = rng.random(n)
        cur = np.sum(h)
        if cur > 0:
            h = h * (M / cur)
        h = np.clip(h, 0.0, 1.0)
        h = repair_h(h, n, rng)
    return h


def upscale_h(h_small: np.ndarray, N: int, rng: np.random.Generator) -> np.ndarray:
    """Interpolate a smaller vector to a larger resolution and repair."""
    n0 = h_small.shape[0]
    xs = np.linspace(0.0, 2.0, n0, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, N, endpoint=False)
    h_new = np.interp(xs_new, xs, h_small)
    h_new = repair_h(h_new, N, rng)
    return h_new


def sa_optimize(h_init: np.ndarray, n: int, time_budget: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simulated annealing with incremental auto‑correlation updates."""
    M = n // 2
    dx = 2.0 / n
    h = h_init.copy()
    auto = np.correlate(h, h, mode="full")   # linear auto‑correlation
    cur_c5 = (M - np.min(auto)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        it += 1
        T = max(1e-6, 0.01 * (0.9995 ** it))
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0.0:
            continue
        delta = max_transfer * rng.random()
        old_i = h[i]
        old_j = h[j]
        auto_old = auto.copy()

        # update contributions for all p != i,j
        for p in range(n):
            if p == i or p == j:
                continue
            hp = h[p]
            idx_ip = p - i + n - 1
            idx_pi = i - p + n - 1
            auto[idx_ip] += delta * hp
            auto[idx_pi] += delta * hp
            idx_jp = p - j + n - 1
            idx_pj = j - p + n - 1
            auto[idx_jp] -= delta * hp
            auto[idx_pj] -= delta * hp

        # diagonal updates
        idx0 = n - 1
        auto[idx0] += 2.0 * old_i * delta + delta * delta
        auto[idx0] += -2.0 * old_j * delta + delta * delta

        # pair (i,j) entries
        shift_ij = j - i
        idx_ij = shift_ij + n - 1
        idx_ji = -shift_ij + n - 1
        delta_pair = delta * (-old_i + old_j) - delta * delta
        auto[idx_ij] += delta_pair
        auto[idx_ji] += delta_pair

        # apply move
        h[i] = old_i + delta
        h[j] = old_j - delta

        new_c5 = (M - np.min(auto)) * dx
        accept = False
        if new_c5 <= cur_c5:
            accept = True
        else:
            if rng.random() < math.exp(-(new_c5 - cur_c5) / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = old_i
            h[j] = old_j
            auto = auto_old

    return best_h, best_c5


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # base resolution (use parent if present)
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 != 0:
            n0 += 1  # ensure even
    else:
        n0 = 80

    # candidate resolutions (even numbers, modest size)
    candidate_ns = [80, 96, 112, 128, 144, 160, 176, 192, 208, 224, 240, 256]
    if n0 in candidate_ns:
        candidate_ns.remove(n0)
    candidate_ns = [n0] + candidate_ns

    for n in candidate_ns:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5.0:
            break
        remaining = budget_s - elapsed

        # time allocation
        if n == n0:
            alloc = min(remaining * 0.6, 800.0)
        else:
            # spread remaining time over the rest
            remaining_ns = len(candidate_ns) - candidate_ns.index(n) - 1
            alloc = min(remaining * 0.4 / max(remaining_ns, 1), 150.0)

        # initialise h for this n
        if n == n0 and "initial_h_values" in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
            h0 = repair_h(h0, n, rng)
        elif best_h_global is not None:
            h0 = upscale_h(best_h_global, n, rng)
        else:
            h0 = random_feasible_h(n, rng)

        # simulated annealing
        h_opt, c5_opt = sa_optimize(h0, n, alloc, rng)

        # optional brief hill‑climbing on the local optimum
        hc_start = time.time()
        h_local = h_opt.copy()
        c5_local = c5_opt
        while time.time() - hc_start < min(alloc * 0.05, 5.0):
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(1.0 - h_local[i], h_local[j])
            if max_transfer <= 0.0:
                continue
            delta = max_transfer * rng.random()
            h_new = h_local.copy()
            h_new[i] += delta
            h_new[j] -= delta
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < c5_local:
                h_local = h_new
                c5_local = c5_new

        # update global best if improved
        if c5_local < best_c5_global:
            best_c5_global = c5_local
            best_h_global = h_local
            best_n_global = n
            if best_c5_global < 0.3808:
                break

    # final safety / fallback
    if best_h_global is None:
        best_n_global = n0
        best_h_global = np.full(best_n_global, 0.5)
        best_c5_global = compute_c5(best_h_global)
    else:
        # final feasibility enforcement
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        M = best_n_global // 2
        cur = np.sum(best_h_global)
        if abs(cur - M) > 1e-9:
            idx = rng.integers(best_n_global)
            best_h_global[idx] = min(max(best_h_global[idx] + (M - cur), 0.0), 1.0)
        best_c5_global = compute_c5(best_h_global)

    return best_h_global, best_c5_global, best_n_global
