def fix_sum_and_bounds(h, target_sum):
    # Clip first
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    # attempt to adjust in free entries
    max_iterations = 5
    for _ in range(max_iterations):
        if abs(diff) < 1e-12:
            break
        free_idx = np.where((h > 0) & (h < 1))[0]
        if free_idx.size > 0:
            # distribute diff equally among free entries
            inc = diff / free_idx.size

def transfer_move(h, target_sum, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # donor i, receiver j
    donor, recv = i, j
    max_tr = min(h[donor], 1.0 - h[recv])

        # try opposite
        max_tr = min(h[recv], 1.0 - h[donor])
        if max_tr <= 0:
            return h  # no move
        donor, recv = recv, donor
    delta = rng.random() * max_tr
    cand = h.copy()
    cand[donor] -= delta
    cand[recv] += delta
    # sum preserved
    return cand

def fix_sum_and_bounds(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # Adjust free entries
    free = np.where((h > 0) & (h < 1))[0]
    if free.size > 0:
        inc = diff / free.size
        h[free] = np.clip(h[free] + inc, 0.0, 1.0)
        cur = np.sum(h)
        diff = target_sum - cur
    # If still diff not zero, adjust iteratively
    if abs(diff) > 1e-12:
        # Add or subtract from entries in order
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)

import time
import math
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def fix_sum_and_bounds(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    free = np.where((h > 0) & (h < 1))[0]
    if free.size:
        inc = diff / free.size
        h[free] = np.clip(h[free] + inc, 0.0, 1.0)
        cur = h.sum()
        diff = target - cur
    # residual fix – walk through indices
    if abs(diff) > 1e-12:
        for i in range(h.size):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)

def swap_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    i, j = rng.choice(h.size, size=2, replace=False)
    cand = h.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand

def transfer_move(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    donor, recv = i, j
    max_tr = min(h[donor], 1.0 - h[recv])
    if max_tr <= 0:
        max_tr = min(h[recv], 1.0 - h[donor])
        if max_tr <= 0:
            return h
        donor, recv = recv, donor
    delta = rng.random() * max_tr
    cand = h.copy()
    cand[donor] -= delta
    cand[recv] += delta
    return cand

def gradient_move(h: np.ndarray, target: float) -> np.ndarray:
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    neigh = np.zeros(n, dtype=float)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = dx * (1.0 - neigh)
    grad_centered = grad - np.mean(grad)
    delta = -grad_centered
    mabs = np.max(np.abs(delta))
    if mabs > 0.2:
        delta = delta * (0.2 / mabs)
    cand = h + delta
    cand = np.clip(cand, 0.0, 1.0)
    cand = fix_sum_and_bounds(cand, target)
    return cand

def smoothing_move(h: np.ndarray, target: float) -> np.ndarray:
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    sm = fix_sum_and_bounds(sm, target)
    return sm
