def project_sum_and_bounds(x, target):
    x = np.clip(x,0.0,1.0)
    cur = x.sum()
    diff = target - cur
    # Simple distribution of diff equally
    if abs(diff)>1e-12:
        x += diff/len(x)
        x = np.clip(x,0.0,1.0)
        # after clip maybe still diff
        cur = x.sum()
        diff = target - cur
        # final fix iteratively
        i = 0
        while abs(diff)>1e-12 and i < 10*len(x):
            idx = i % len(x)
            if diff > 0 and x[idx] < 1.0:
                inc = min(diff, 1.0 - x[idx])
                x[idx] += inc
                diff -= inc
            elif diff < 0 and x[idx] > 0.0:
                dec = min(-diff, x[idx])
                x[idx] -= dec
                diff += dec
            i += 1
    return x

def gradient_descent(h0, time_budget, rng):
    h = h0.copy()
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 5.0
    max_beta = 80.0
    alpha = 0.2
    while time.time() - start < time_budget:
        # compute correlation
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0/n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Update smoothing beta slowly
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
        # softmax weighting
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / len(v)
        else:
            w /= w_sum
        # gradient
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-15:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # zero mean gradient ensures sum unchanged
        g -= np.mean(g)
        # step size schedule
        elapsed = time.time() - start
        # linearly decay
        alpha_cur = alpha * max(0.2, 1.0 - elapsed / time_budget)
        h_new = h - alpha_cur * g
        # project
        h_new = project_sum_and_bounds(h_new, target)
        # accept if improvement or random
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        else:
            # occasional random shake
            if rng.random() < 0.001:
                h = project_sum_and_bounds(h + rng.normal(scale=0.01, size=n), target)
    return best_h, best_c5

def simulated_annealing(h0, time_budget, rng):
    h = h0.copy()
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    T0 = 0.02
    Tmin = 1e-4
    decay = 0.9995
    T = T0
    it = 0
    while time.time() - start < time_budget:
        it += 1
        # propose move
        h_new = h.copy()
        # Choose move type
        if rng.random() < 0.8:
            # swap a 0 and a 1
            zeros = np.where(h_new <= 0.5)[0]  # include near zero
            ones = np.where(h_new >= 0.5)[0]
            if zeros.size > 0 and ones.size > 0:
                i = rng.choice(zeros)
                j = rng.choice(ones)
                # swap
                tmp = h_new[i]
                h_new[i] = h_new[j]
                h_new[j] = tmp
        else:
            # small perturb two indices
            i, j = rng.integers(0, n, size=2)
            delta = rng.normal(scale=0.03)
            hi = h_new[i] + delta
            hj = h_new[j] - delta
            hi = np.clip(hi, 0.0, 1.0)
            # adjust delta to actual used
            delta_adj = hi - h_new[i]
            hj = h_new[j] - delta_adj
            hj = np.clip(hj, 0.0, 1.0)
            h_new[i] = hi
            h_new[j] = hj
            # ensure sum
            h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        # acceptance
        if c5_new < cur_c5 or rng.random() < np.exp(-(c5_new - cur_c5) / max(T,1e-12)):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        # temperature decay
        T = max(Tmin, T * decay)
        # occasional random restart if stuck
        if it % 10000 == 0 and rng.random() < 0.1:
            h = np.zeros(n)
            h[:int(target)] = 1.0
            rng.shuffle(h)
            cur_c5 = compute_c5(h)
    return best_h, best_c5

def simulated_annealing(h0, time_budget, rng):
    h = h0.copy()
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    T = 0.02
    Tmin = 1e-4
    decay = 0.9995
    it = 0
    while time.time() - start < time_budget:
        it += 1
        # propose move
        h_new = h.copy()
        if rng.random() < 0.9:
            # swap two positions
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            # swap
            tmp = h_new[i]
            h_new[i] = h_new[j]
            h_new[j] = tmp
        else:
            # small perturbation
            i, j = rng.integers(0, n, size=2)
            delta = rng.normal(scale=0.05)
            hi = h_new[i] + delta
            hj = h_new[j] - delta
            hi = np.clip(hi, 0.0, 1.0)
            delta_adj = hi - h_new[i]
            hj = h_new[j] - delta_adj
            hj = np.clip(hj, 0.0, 1.0)
            h_new[i] = hi
            h_new[j] = hj
            h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        # acceptance
        if c5_new < cur_c5 or rng.random() < np.exp(-(c5_new - cur_c5) / max(T, 1e-12)):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        # temperature decay
        T = max(Tmin, T * decay)
        # occasional random restart
        if it % 20000 == 0 and rng.random() < 0.1:
            h = np.zeros(n)
            h[:int(target)] = 1.0
            rng.shuffle(h)
            cur_c5 = compute_c5(h)
    return best_h, best_c5
```

import time, numpy as np

def compute_c5(h):
    """Maximum overlap (scaled by dx) using full correlation."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target, tol=1e-12):
    """Clip to [0,1] and adjust to meet sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > tol:
            i = 0
            n = len(x)
            while abs(diff) > tol and i < 10 * n:
                idx = i % n
                if diff > 0 and x[idx] < 1.0:
                    inc = min(diff, 1.0 - x[idx])
                    x[idx] += inc
                    diff -= inc
                elif diff < 0 and x[idx] > 0.0:
                    dec = min(-diff, x[idx])
                    x[idx] -= dec
                    diff += dec
                i += 1
    return x

def binarize_h(h):
    """Return binary vector with same length and half entries set to 1."""
    n = len(h)
    target = n // 2
    idx = np.argpartition(-h, target)[:target]
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec

def gradient_descent(h0, time_budget, rng):
    """Smooth max-overlap gradient descent."""
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 5.0
    max_beta = 80.0
    alpha0 = 0.2
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / len(v)
        else:
            w /= w_sum
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        g -= np.mean(g)                # preserve total mass
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
    return best_h, best_c5

def simulated_annealing(h0, time_budget, rng):
    """Swap‑based annealing on (possibly binary) vectors."""
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    T = 0.02
    Tmin = 1e-4
    decay = 0.9995
    it = 0
    while time.time() - start < time_budget:
        it += 1
        h_new = h.copy()
        if rng.random() < 0.9:                     # swap two positions
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            tmp = h_new[i]
            h_new[i] = h_new[j]
            h_new[j] = tmp
        else:                                      # small perturbation
            i, j = rng.integers(0, n, size=2)
            delta = rng.normal(scale=0.05)
            hi = h_new[i] + delta
            hj = h_new[j] - delta
            hi = np.clip(hi, 0.0, 1.0)
            delta_adj = hi - h_new[i]
            hj = h_new[j] - delta_adj
            hj = np.clip(hj, 0.0, 1.0)
            h_new[i] = hi
            h_new[j] = hj
            h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5 or rng.random() < np.exp(-(c5_new - cur_c5) / max(T, 1e-12)):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        T = max(Tmin, T * decay)
        if it % 20000 == 0 and rng.random() < 0.1:
            h = np.zeros(n)
            h[:int(target)] = 1.0
            rng.shuffle(h)
            cur_c5 = compute_c5(h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 120
        h0 = np.zeros(n, dtype=float)
        h0[:n // 2] = 1.0
        rng.shuffle(h0)

    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    total_budget = max(0.0, budget_s - (time.time() - start))
    gd_budget = total_budget * 0.3
    h_gd, c5_gd = gradient_descent(h0, gd_budget, rng)

    # possibly improve by binarizing the GD result
    if n % 2 == 0:
        h_bin = binarize_h(h_gd)
        c5_bin = compute_c5(h_bin)
        h_start_sa = h_bin if c5_bin < c5_gd else h_gd
    else:
        h_start_sa = h_gd

    elapsed = time.time() - start
    remaining_budget = max(0.0, budget_s - elapsed)
    sa_budget = remaining_budget * 0.8
    h_sa, c5_sa = simulated_annealing(h_start_sa, sa_budget, rng)

    elapsed = time.time() - start
    final_budget = max(0.0, budget_s - elapsed)
    h_final, c5_final = gradient_descent(h_sa, final_budget, rng)

    candidates = [(h0, compute_c5(h0)), (h_gd, c5_gd), (h_sa, c5_sa), (h_final, c5_final)]
    if n % 2 == 0:
        candidates.append((h_bin, c5_bin))
    best_h, best_c5 = min(candidates, key=lambda t: t[1])

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
