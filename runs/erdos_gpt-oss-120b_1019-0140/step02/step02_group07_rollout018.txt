def continuous_anneal(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # parameters
    init_temp = 1e-2  # maybe higher
    final_temp = 1e-7
    max_iter = 500000  # large
    for it in range(max_iter):
        if time.time() >= time_end:
            break
        # compute max-lag
        c5, lag = compute_c5_and_maxlag(h)
        # select move type
        if rng.random() < 0.7:
            # targeted move using lag
            L = lag
            if L != 0:
                # choose i on domain
                if L > 0:
                    i = rng.integers(0, n - L)
                else:
                    i = rng.integers(-L, n)
                j = i + L
                # direction: move from i to j (decrease i, increase j)
                max_delta = min(h[i], 1 - h[j])
                if max_delta > 0:
                    delta = max_delta * rng.random() * 0.5
                else:
                    # fallback random pair
                    i = rng.integers(n)
                    j = rng.integers(n)
                    max_delta = min(h[i], 1 - h[j]) if i != j else 0
                    if max_delta == 0:
                        continue
                    delta = max_delta * rng.random() * 0.5
                # apply
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
            else:
                # fallback random
                i = rng.integers(n)
                j = rng.integers(n)
                max_delta = min(h[i], 1 - h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random() * 0.5
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
        else:
            # generic swap of two random indices
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(h[i], 1 - h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random() * 0.5
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta

        new_c5 = compute_c5(h_new)
        # compute temperature
        it_frac = it / max_iter
        T = init_temp * (1 - it_frac) + final_temp * it_frac
        dE = new_c5 - cur_c5
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    # final correction to exact sum:
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        # reproject to binary if needed? Not required
    return best_h, best_c5

import time
import math
import numpy as np
import cvxpy as cp

def compute_c5(arr):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_c5_and_maxlag(arr):
    """Return (C5, lag) where lag is the shift attaining the maximum."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full') * (2.0 / n)
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    return float(corr[idx]), int(lag)

def adjust_sum_to_target(arr, target):
    """Clip to [0,1] and flip entries to hit the exact integer sum `target`."""
    arr = np.clip(arr, 0.0, 1.0)
    cur = int(round(np.sum(arr)))
    diff = int(target - cur)
    if diff == 0:
        return arr
    if diff > 0:
        zeros = np.where(arr < 0.5)[0]
        if zeros.size < diff:
            zeros = np.arange(len(arr))
        arr[zeros[:diff]] = 1.0
    else:
        ones = np.where(arr > 0.5)[0]
        if ones.size < -diff:
            ones = np.arange(len(arr))
        arr[ones[: -diff]] = 0.0
    return arr

def continuous_anneal(start_h, time_end, rng):
    """Simulated annealing on a continuous vector preserving the sum."""
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    max_iter = 500000
    init_temp = 5e-3
    final_temp = 1e-7

    for it in range(max_iter):
        if time.time() >= time_end:
            break
        c5, lag = compute_c5_and_maxlag(h)
        # targeted move with probability 0.7
        if rng.random() < 0.7 and lag != 0:
            L = lag
            if L > 0:
                i = rng.integers(0, n - L)
            else:
                i = rng.integers(-L, n)
            j = i + L
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                # fallback generic move
                i = rng.integers(n)
                j = rng.integers(n)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0.0:
                    continue
            delta = max_delta * rng.random() * 0.5
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # generic transfer move
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            delta = max_delta * rng.random() * 0.5
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta

        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0.0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:

                best_h = h_new.copy()

    # final exact-sum correction

    return best_h, best_c5

def binary_hill_climb(start_bin, time_end, rng):
    """Hill climbing on binary vectors (swap one 0/1 pair)."""

    h = start_bin.copy()

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        init_h = None
    target_sum = n / 2.0

    best_h = None
    best_c5 = float('inf')

    # Use parent solution if available
    if init_h is not None:
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = adjust_sum_to_target(init_h, target_sum)
        best_h = init_h.copy()
        best_c5 = compute_c5(best_h)

    # ---------- MILP phase ----------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    milp_time = max(10.0, min(800.0, remaining * 0.7))

    try:
        h_var = cp.Variable(n, boolean=True)
        C_var = cp.Variable()
        constraints = [cp.sum(h_var) == target_sum]

        for k in range(1, n):  # positive lags only; lag 0 gives zero overlap
            i_idx = np.arange(0, n - k)
            j_idx = i_idx + k
            p = len(i_idx)
            if p == 0:
                continue
            y = cp.Variable(p, boolean=True)
            constraints += [
                y <= h_var[i_idx],
                y <= h_var[j_idx],
                y >= h_var[i_idx] + h_var[j_idx] - 1,
                C_var >= target_sum - cp.sum(y)
            ]

        constraints += [C_var >= 0]
        prob = cp.Problem(cp.Minimize(C_var), constraints)
        prob.solve(solver=cp.CBC, maxSeconds=milp_time, verbose=False)

        if h_var.value is not None:
            cand = np.rint(np.array(h_var.value)).astype(float)
            cand = adjust_sum_to_target(cand, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
    except Exception:
        pass  # MILP failed; fall back to heuristics

    # ---------- Heuristic refinement ----------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5.0:
        if best_h is not None:
            start_vec = best_h.copy()
        else:
            start_vec = np.zeros(n, dtype=float)
            start_vec[rng.choice(n, int(target_sum), replace=False)] = 1.0

        # Continuous annealing
        end_time = time.time() + remaining * 0.6
        cand_h, cand_c5 = continuous_anneal(start_vec, end_time, rng)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h.copy()

        # Binary hill climbing from rounded solution
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining > 2.0 and best_h is not None:
            bin_start = (best_h >= 0.5).astype(int)
            end_time = time.time() + remaining - 0.5
            final_h, final_c5 = binary_hill_climb(bin_start, end_time, rng)
            final_h = adjust_sum_to_target(final_h, target_sum)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()

    # Final safety enforcement
    if best_h is None:
        # fallback: simple balanced binary vector
        best_h = np.zeros(n, dtype=float)
        best_h[rng.choice(n, int(target_sum), replace=False)] = 1.0

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_to_target(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
