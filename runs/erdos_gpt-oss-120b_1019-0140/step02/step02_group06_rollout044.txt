def gradient_descent_refine(h0, time_budget, rng):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.2
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_budget:
        # Compute correlation vector
        v = np.correlate(h, 1.0 - h, mode='full')
        # Compute softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / len(v)
        else:
            w /= w_sum
        # Gradient
        g = np.zeros(n)
        offset = n - 1
        # vectorized loop for each shift? Use for loop as N small.
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
        # zero-mean to preserve sum
        g -= np.mean(g)
        # schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        # update
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # accept if improved or with small probability
        if c5_new < compute_c5(h) or rng.random() < 0.001:
            h = h_new
        # increase beta gradually
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target, tol=1e-12):
    """Clip to [0,1] and enforce exact sum == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        # first uniform correction
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # fallback iterative fix
        if abs(diff) > tol:
            for i in range(len(x)):
                if abs(diff) < tol:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def random_feasible_h(n, rng):
    """Generate a random feasible vector of length n."""
    x = rng.random(n)
    return project_sum_and_bounds(x, n / 2.0)

def neighbor_move(h, rng, max_eps=0.05):
    """Transfer a small mass epsilon between two indices, preserving sum."""
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    if rng.random() < 0.5:
        # move from i to j
        max_move = min(max_eps, h[i], 1.0 - h[j])
        if max_move <= 0:
            return h.copy()
        eps = rng.random() * max_move
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
    else:
        # move from j to i
        max_move = min(max_eps, h[j], 1.0 - h[i])
        if max_move <= 0:
            return h.copy()
        eps = rng.random() * max_move
        h_new = h.copy()
        h_new[j] -= eps
        h_new[i] += eps
    return h_new

def gradient_descent_refine(h0, time_budget, rng):
    """Smooth‑max gradient descent refinement (similar to prior approach)."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.2

    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / v.size
        else:
            w /= w_sum

        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]

        g -= np.mean(g)                    # preserve total mass
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        # accept if improvement or tiny random acceptance
        if c5_new < compute_c5(h) or rng.random() < 0.001:
            h = h_new

        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

    return best_h, best_c5

            refined_h, refined_c5 = gradient_descent_refine(best_h, min(remaining * 0.2, 30.0), rng)
            if refined_c5 < best_c5:
                best_h, best_c5 = refined_h, refined_c5
                cur_h, cur_c5 = refined_h, refined_c5

        iter_count += 1

    # Final polishing with remaining time, if any
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        best_h, best_c5 = gradient_descent_refine(best_h, remaining, rng)

    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
