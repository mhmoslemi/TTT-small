def improve_by_gradient(h, max_iters, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    while time condition and iter < max_iters:
        overlaps = compute_overlap_array(h)
        max_ov = overlaps.max()
        # find shifts achieving max within tolerance
        worst_shifts = np.where(np.abs(overlaps - max_ov) <= 1e-12)[0]  # indices of corr array
        # map to k values
        ks = worst_shifts - (n-1)
        # compute gradient contributions
        grad = np.zeros(n)
        for k in ks:
            if k >= 0:
                # i: 0..n-k-1
                i_range = slice(0, n - k)
                grad[i_range] += (1.0 - h[i_range + k])  # but need vector addition
                # in numpy:
                grad[:n-k] += (1.0 - h[k:])
            else:
                kk = -k
                grad[kk:] += (1.0 - h[:n-kk])
        # Determine source and dest
        # Source: index with largest grad (reducing h will reduce objective most)
        i_src = np.argmax(grad)
        i_dst = np.argmin(grad)
        if i_src == i_dst:
            break
        max_transfer = min(h[i_src], 1.0 - h[i_dst])
        if max_transfer <= 0:
            break
        # Transfer small step: maybe step = min(max_transfer, step_size)
        step = min(max_transfer, 0.02) # adjust step size
        # apply
        h_new = h.copy()
        h_new[i_src] -= step
        h_new[i_dst] += step
        # compute new C5
        new_c5 = compute_c5(h_new)
        if new_c5 + 1e-12 < max_ov:
            h = h_new
            # continue
        else:
            # maybe try random direction or smaller step
            # reduce step and retry?
            if step > 1e-4:
                # try half step
                step = step / 2
                h_new[i_src] = h[i_src] - step
                h_new[i_dst] = h[i_dst] + step
                new_c5 = compute_c5(h_new)
                if new_c5 + 1e-12 < max_ov:
                    h = h_new
            # else break
        iter += 1
    return h, compute_c5(h)

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx  # length 2n-1, k = index - (n-1)

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def generate_initial_h(n, M, rng):
    method = rng.integers(4)
    if method == 0:
        # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:
        # uniform random scaled
        h = rng.random(n)
        h = h / np.sum(h) * M
        h = np.clip(h, 0.0, 1.0)
        # adjust to exact sum
        cur = np.sum(h)
        diff = M - cur
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    elif method == 2:
        # linear decreasing shape
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = M - cur
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    else:
        # smooth sinusoidal shape
        x = np.arange(n)
        h = 0.5 + 0.5*np.sin(2*np.pi*x/n)
        # Scale to sum M and clip
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if cur > 0:
            h = h / cur * M
        else:
            h = np.full(n, M / n)
        # Clip again and adjust sum
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = M - cur
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def optimise_for_n(n, alloc_time, rng):
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    constraints = [{'type':'eq','fun':sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]
    bounds = [(0.0,1.0)]*n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf

    start = time.time()
    # use parent if available
    use_parent = False
    if 'initial_h_values' in globals():
        if len(initial_h_values) == n:
            use_parent = True
            parent_h = np.array(initial_h_values, dtype=float)
            # adjust sum
            cur = np.sum(parent_h)
            if abs(cur - M) > 1e-12:
                parent_h += (M - cur) / n
                parent_h = np.clip(parent_h, 0.0, 1.0)
            parent_c5 = compute_c5(parent_h)
            best_h, best_c5 = parent_h, parent_c5

    while True:
        elapsed = time.time() - start
        if elapsed > alloc_time - 0.1:
            break

        if use_parent:
            h0 = parent_h.copy()
            use_parent = False
        else:
            h0 = generate_initial_h(n, M, rng)

        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])

        try:
            res = minimize(fun=lambda x: x[-1],
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter':1000,'ftol':1e-9,'disp':False})
        except Exception:
            continue
        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur = np.sum(h_opt)
        if abs(cur - M) > 1e-9:
            h_opt += (M - cur) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)

        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

        if best_c5 < 0.3808:
            break

    return best_h, best_c5

def gradient_refine(h, rng, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    step_base = 0.02
    max_iters = 10000
    it = 0
    while time.time() - start_time < time_limit and it < max_iters:
        overlaps = compute_overlaps(h)
        max_ov = overlaps.max()
        # identify worst shifts (max overlap)
        ks_idx = np.where(np.abs(overlaps - max_ov) <= 1e-12)[0]
        ks = ks_idx - (n - 1)
        # compute gradient
        grad = np.zeros(n)
        for k in ks:
            if k >= 0:
                grad[:n - k] += (1.0 - h[k:])
            else:
                kk = -k
                grad[kk:] += (1.0 - h[:n - kk])
        # source: max grad, dest: min grad
        i_src = np.argmax(grad)
        i_dst = np.argmin(grad)
        if i_src == i_dst:
            break
        max_transfer = min(h[i_src], 1.0 - h[i_dst])
        if max_transfer <= 0:
            break
        step = min(step_base, max_transfer)
        # try improvement
        h_new = h.copy()
        h_new[i_src] -= step
        h_new[i_dst] += step
        new_c5 = compute_c5(h_new)
        if new_c5 + 1e-12 < max_ov:
            h = h_new
            continue
        # try smaller step
        step = step * 0.5
        if step < 1e-5:
            break
        h_new[i_src] = h[i_src] - step
        h_new[i_dst] = h[i_dst] + step
        new_c5 = compute_c5(h_new)
        if new_c5 + 1e-12 < max_ov:
            h = h_new
            continue
        # else break if cannot improve
        break
        it += 1
    return h, compute_c5(h)

def random_hill_climb(h, rng, time_limit, start_time, best_c5):
    n = len(h)
    while time.time() - start_time < time_limit:
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            h = h_new
            best_c5 = c5_new
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate n list: keep small for speed, include parent size if any
    candidate_ns = [80, 96, 112, 128, 144]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 not in candidate_ns:
            candidate_ns = [n0] + candidate_ns
        elif n0 % 2 == 0:
            candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        # time guard for outer loop
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        # allocate slice of time: max 250s or 30% of remaining whichever smaller
        alloc = min(250.0, remaining * 0.3)
        h_opt, c5_opt = optimise_for_n(n, alloc, rng)
        if h_opt is None:
            continue
        # gradient refinement
        time_for_refine = min(30.0, budget_s - (time.time() - start_time) - 5)
        h_opt, c5_opt = gradient_refine(h_opt, rng, time_for_refine, time.time())
        # random hill climb
        time_for_hc = min(30.0, budget_s - (time.time() - start_time) - 5)
        h_opt, c5_opt = random_hill_climb(h_opt, rng, time_for_hc, time.time(), c5_opt)

        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n
        if best_c5 < 0.3808:
            break

    if best_h is None:
        # fallback
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        # final projection to constraints
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = np.sum(best_h)
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def generate_initial_h(n, M, rng):
    method = rng.integers(4)
    if method == 0:
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:
        h = rng.random(n)
        h = h / np.sum(h) * M
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = M - cur
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    elif method == 2:
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = M - cur
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    else:
        x = np.arange(n)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * x / n)
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)

        else:

def optimise_for_n(n, alloc_time, rng):
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf
    start = time.time()
    use_parent = False
    if 'initial_h_values' in globals():
        if len(initial_h_values) == n:
            use_parent = True
            parent_h = np.array(initial_h_values, dtype=float)
            cur = np.sum(parent_h)
            if abs(cur - M) > 1e-12:
                parent_h += (M - cur) / n
                parent_h = np.clip(parent_h, 0.0, 1.0)
            parent_c5 = compute_c5(parent_h)
            best_h, best_c5 = parent_h, parent_c5

    while True:
        if time.time() - start > alloc_time - 0.1:
            break
        if use_parent:

            use_parent = False

            h_opt = np.clip(h_opt, 0.0, 1.0)

        if c5_opt + 1e-12 < best_c5:
            best_c5, best_h = c5_opt, h_opt
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

def gradient_refine(h, rng, time_limit, start_time):
    n = len(h)
    step_base = 0.01
    max_iters = 10000
    it = 0
    while time.time() - start_time < time_limit and it < max_iters:
        overlaps = compute_overlaps(h)
        max_ov = overlaps.max()
        ks_idx = np.where(np.abs(overlaps - max_ov) <= 1e-12)[0]
        ks = ks_idx - (n - 1)
        grad = np.zeros(n)
        for k in ks:
            if k >= 0:
                grad[:n - k] += (1.0 - h[k:])
            else:
                kk = -k
                grad[kk:] += (1.0 - h[:n - kk])
        i_src = np.argmax(grad)
        i_dst = np.argmin(grad)
        if i_src == i_dst:
            break
        max_transfer = min(h[i_src], 1.0 - h[i_dst])
        if max_transfer <= 0:
            break
        step = min(step_base, max_transfer)
        h_new = h.copy()
        h_new[i_src] -= step
        h_new[i_dst] += step
        new_c5 = compute_c5(h_new)
        if new_c5 + 1e-12 < max_ov:
            h = h_new
            it += 1
            continue
        step_half = step * 0.5
        if step_half < 1e-6:
            break
        h_new[i_src] = h[i_src] - step_half
        h_new[i_dst] = h[i_dst] + step_half
        new_c5 = compute_c5(h_new)
        if new_c5 + 1e-12 < max_ov:
            h = h_new
            it += 1
            continue
        break
    return h, compute_c5(h)

def random_hill_climb(h, rng, time_limit, start_time, best_c5):
    n = len(h)
    while time.time() - start_time < time_limit:
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            h = h_new
            best_c5 = c5_new
    return h, best_c5
