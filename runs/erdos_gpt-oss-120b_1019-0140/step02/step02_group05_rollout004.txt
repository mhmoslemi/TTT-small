import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Integral max overlap C5 for step function h on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum_and_bounds(h, target):
    """Clip to [0,1] and adjust to meet exactly target sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # distribute diff among free entries
    free = np.where((h > 0) & (h < 1))[0]
    if free.size:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = target - cur
    # residual fix by scanning
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0.0, 1.0)

def smooth_and_fix(h, target):
    """Gaussian-like smoothing (3‑point kernel) with sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return enforce_sum_and_bounds(sm, target)

def random_continuous(n, target, rng):
    """Generate random vector in [0,1] with exact sum target."""
    h = np.zeros(n)
    remaining = target
    for i in range(n-1):
        max_val = min(1.0, remaining)
        h[i] = rng.random() * max_val
        remaining -= h[i]
    h[-1] = remaining
    rng.shuffle(h)
    return h

def random_binary(n, target, rng):
    """Generate binary vector with exactly target ones (target integer)."""
    ones = int(round(target))
    h = np.zeros(n)
    idx = rng.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    n = 240                      # finer discretisation
    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- initial construction -----------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        if n % len(base) == 0:
            factor = n // len(base)
            init_h = np.repeat(base, factor)
        else:
            init_h = random_continuous(n, target_sum, rng)
    else:
        init_h = random_continuous(n, target_sum, rng)

    init_h = enforce_sum_and_bounds(init_h, target_sum)

    best_h = init_h.copy()
    best_val = compute_c5(best_h)

    # ----- multi‑start simulated annealing ------------------------------
    num_restarts = 6
    for restart in range(num_restarts):
        if restart == 0:
            h = init_h.copy()
        elif restart == 1:
            h = random_binary(n, target_sum, rng)
        else:
            h = random_continuous(n, target_sum, rng)

        cur_h = h.copy()
        cur_val = compute_c5(cur_h)
        if cur_val < best_val:
            best_val, best_h = cur_val, cur_h.copy()

        # annealing parameters
        while True:
            elapsed = time.time() - start
            if elapsed >= budget_s * 0.90:
                break
            T = max(0.0001, 0.2 * (1.0 - elapsed / budget_s))

            r = rng.random()
            # ----- smoothing move (5%) ---------------------------------
            if r < 0.05:
                cand = smooth_and_fix(cur_h, target_sum)
            # ----- swap move -------------------------------------------
            elif r < 0.40:
                i, j = rng.choice(n, size=2, replace=False)
                cand = cur_h.copy()
                cand[i], cand[j] = cand[j], cand[i]
            # ----- transfer move ----------------------------------------
            elif r < 0.75:
                i, j = rng.choice(n, size=2, replace=False)
                donor, recv = (i, j) if cur_h[i] > 0 and cur_h[j] < 1 else (j, i)
                if donor == recv:
                    continue
                max_tr = min(cur_h[donor], 1.0 - cur_h[recv])
                if max_tr <= 0:
                    continue
                delta = rng.random() * max_tr
                cand = cur_h.copy()
                cand[donor] -= delta
                cand[recv] += delta
                cand = enforce_sum_and_bounds(cand, target_sum)
            # ----- gradient move ----------------------------------------
            else:
                corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
                k = int(np.argmax(corr)) - (n - 1)
                neigh = np.zeros(n)
                if k >= 0:
                    neigh[: n - k] += cur_h[k:]
                    neigh[k:] += cur_h[: n - k]
                else:
                    kk = -k
                    neigh[kk:] += cur_h[: n - kk]

                grad = 1.0 - neigh

                else:
                    step = np.zeros_like(cur_h)
                cand = cur_h + step
                cand = np.clip(cand, 0.0, 1.0)

            # ----- acceptance -------------------------------------------
            if cand_val < cur_val:

            else:

            if accept:

        for _ in range(3):

                best_val, best_h = sm_val, sm.copy()
                cur_h, cur_val = sm, sm_val
            else:
                break

        # ----- final local refinement with SLSQP if time permits -------
        remaining = budget_s - (time.time() - start)
        if remaining > 20.0:
            # variable x = [h..., Y] where Y bound on C5
            def sum_eq(x):
                return np.sum(x[:n]) - target_sum

            def make_ineq(k):
                def fun(x):
                    hh = x[:n]
                    if k >= 0:
                        overl = np.dot(hh[: n - k], 1.0 - hh[k:])
                    else:
                        kk = -k
                        overl = np.dot(hh[kk:], 1.0 - hh[: n - kk])
                    return x[-1] - overl * dx
                return fun

            constraints = [{"type": "eq", "fun": sum_eq}]
            for kk in range(-n + 1, n):
                if kk == 0:
                    continue
                constraints.append({"type": "ineq", "fun": make_ineq(kk)})

            bounds = [(0.0, 1.0)] * n + [(0.0, None)]
            x0 = np.concatenate([best_h, [best_val]])

            try:
                res = minimize(
                    fun=lambda x: x[-1],
                    x0=x0,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=constraints,
                    options={"maxiter": 300, "ftol": 1e-8, "disp": False},
                )
                if res.success:
                    h_opt = enforce_sum_and_bounds(np.clip(res.x[:n], 0.0, 1.0), target_sum)
                    val_opt = compute_c5(h_opt)
                    if val_opt + 1e-12 < best_val:
                        best_val, best_h = val_opt, h_opt.copy()
            except Exception:
                pass

        if best_val < 0.3808:
            break

    # ----- final safety correction ---------------------------------------
    best_h = enforce_sum_and_bounds(best_h, target_sum)
    best_val = compute_c5(best_h)

    return best_h, best_val, n
