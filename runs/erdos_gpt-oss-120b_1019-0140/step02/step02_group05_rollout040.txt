def project_feasible(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # simple iterative correction
    for _ in range(5):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # after iteration sum may still not match due to clipping, do final fix:
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute diff to free entries
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
    # final adjust if needed
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        # adjust uniformly
        h += (target - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_c_k(h, shift):
    n = len(h)
    dx = 2.0 / n
    neigh = np.zeros(n)
    if shift >= 0:
        neigh[: n - shift] += h[shift:]
        neigh[shift:] += h[: n - shift]
    else:
        s = -shift
        neigh[s:] += h[: n - s]
        neigh[: n - s] += h[s:]
    grad = dx * (1.0 - neigh)
    return grad

def subgradient_step(h, target_sum, step_size):
    # compute worst shift
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (len(h) - 1)
    grad = gradient_c_k(h, shift)

    grad_centered = grad - np.mean(grad)
    # propose new h
    cand = h - step_size * grad_centered
    cand = project_feasible(cand, target_sum)

def project_feasible(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # adjust sum to target
    cur = v.sum()
    diff = target - cur
    # simple iterative scheme
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        # distribute diff equally among non-saturated
        free = np.where((v > 0) & (v < 1))[0]
        if free.size == 0:
            # all at bounds, just adjust all equally (could violate)
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
        else:
            inc = diff / free.size
            v[free] += inc
            v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
    # final correction if still off
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
    return v

def smooth_and_fix(h, target):
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode='same')
    sm = np.clip(sm, 0.0, 1.0)
    # adjust sum similarly
    cur = sm.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # add diff uniformly to free slots
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            sm[free] += diff / free.size
        # iterative clip fix
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
        sm = np.clip(sm, 0.0, 1.0)
    return sm

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    for _ in range(5):
        cur = v.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0) & (v < 1))[0]
        if free.size:
            inc = diff / free.size
            v[free] += inc
            v = np.clip(v, 0.0, 1.0)
        else:
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
    return v

def gradient_c_k(h, shift):
    n = len(h)
    dx = 2.0 / n
    neigh = np.zeros(n)
    if shift >= 0:
        neigh[: n - shift] += h[shift:]
        neigh[shift:] += h[: n - shift]
    else:
        s = -shift
        neigh[s:] += h[: n - s]
        neigh[: n - s] += h[s:]
    return dx * (1.0 - neigh)

def smooth_and_fix(h, target):
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    cur = sm.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            sm[free] += diff / free.size
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
        sm = np.clip(sm, 0.0, 1.0)
    return sm
