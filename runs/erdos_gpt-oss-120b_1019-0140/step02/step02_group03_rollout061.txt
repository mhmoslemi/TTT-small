import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def local_optimize(h, target, maxiter=400):
    n = h.shape[0]
    # minimize compute_c5
    res = minimize(compute_c5, h, method='SLSQP',
                   bounds=[(0.0,1.0)]*n,
                   constraints=[{'type':'eq', 'fun': lambda x: x.sum() - target}],
                   options={'maxiter': maxiter, 'ftol':1e-9, 'disp': False})
    if res.success:
        x = np.clip(res.x, 0.0, 1.0)
        x = project_to_box_sum(x, target)
        return x, compute_c5(x)
    else:
        # Return original
        return h, compute_c5(h)

    # GA parameters

    mutation_sigma = 0.05
    max_local_iter = 300
    # initialize population
    population = [h0]
    for _ in range(pop_size-1):

        cand = project_to_box_sum(cand, target)
        population.append(cand)
    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    # Main GA loop
    while time.time() - start < budget_s - 5:
        # tournament selection

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project vector v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def local_optimize(h, target, maxiter=300):
    """Refine a candidate using SLSQP to reduce C5."""
    n = h.shape[0]
    res = minimize(
        compute_c5,
        h,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    if res.success:
        x = np.clip(res.x, 0.0, 1.0)
        x = project_to_box_sum(x, target)
        return x, compute_c5(x)
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Determine problem size and initial feasible vector ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0
    best_c5 = compute_c5(best_h)

    # ---- Phase 0: random SLSQP restarts (exploratory) ----
    max_restarts = 30
    for _ in range(max_restarts):
        if time.time() - start_time > budget_s * 0.4:
            break
        cand = rng.random(n)
        cand = project_to_box_sum(cand, target_sum)
        cand_opt, cand_c5 = local_optimize(cand, target_sum, maxiter=200)
        if cand_c5 < best_c5:
            best_h = cand_opt
            best_c5 = cand_c5

    # ---- Genetic Algorithm parameters ----
    pop_size = 30
    tournament_k = 3
    mut_rate = 0.15
    mut_sigma = 0.05
    max_local_iter = 300

    # ---- Initialise population (include current best) ----
    population = [best_h]
    for _ in range(pop_size - 1):
        indiv = rng.random(n)
        indiv = project_to_box_sum(indiv, target_sum)
        population.append(indiv)

    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # ---- Helper for tournament selection ----
    def tournament():
        idxs = rng.choice(pop_size, tournament_k, replace=False)
        return idxs[np.argmin(fitness[idxs])]

    # ---- Main GA loop ----
    while time.time() - start_time < budget_s - 5.0:
        # Select two parents
        p1 = tournament()
        p2 = tournament()
        parent1 = population[p1]
        parent2 = population[p2]

        # Uniform crossover
        mask = rng.random(n) < 0.5
        child = np.where(mask, parent1, parent2)

        # Gaussian mutation
        mut_mask = rng.random(n) < mut_rate
        child[mut_mask] += rng.normal(scale=mut_sigma, size=mut_mask.sum())

        # Feasibility repair
        child = np.clip(child, 0.0, 1.0)
        child = project_to_box_sum(child, target_sum)

        # Evaluate
        child_f = compute_c5(child)

        # Local refinement if promising
        if child_f < best_c5 * 0.99 or rng.random() < 0.05:
            refined_child, refined_f = local_optimize(child, target_sum, maxiter=max_local_iter)
            if refined_f < child_f:
                child = refined_child
                child_f = refined_f

        # Replace worst individual if improvement
        worst_idx = np.argmax(fitness)
        if child_f < fitness[worst_idx]:
            population[worst_idx] = child
            fitness[worst_idx] = child_f
            if child_f < best_c5:
                best_c5 = child_f
                best_h = child.copy()
                if best_c5 < 0.3808:
                    break

        # Occasional refinement of the current best
        if rng.random() < 0.01:
            refined_best, refined_best_f = local_optimize(best_h, target_sum, maxiter=max_local_iter)
            if refined_best_f < best_c5:
                best_h = refined_best
                best_c5 = refined_best_f
                # Update its copy inside the population
                best_pop_idx = np.argmin(fitness)
                population[best_pop_idx] = best_h
                fitness[best_pop_idx] = best_c5

    # ---- Final safety projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
