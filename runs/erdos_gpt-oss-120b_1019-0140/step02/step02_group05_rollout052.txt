        cand = cur_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
    elif r < p_swap + p_transfer:
        # transfer
        i,j = rng.choice(n,2,replace=False)
        # direction determined randomly
        if rng.random() < 0.5:
            donor, recv = i, j
        else:
            donor, recv = j, i
        max_tr = min(cur_h[donor], 1.0 - cur_h[recv])
        if max_tr <= 0:
            continue
        delta = rng.random() * max_tr
        cand = cur_h.copy()
        cand[donor] -= delta
        cand[recv] += delta
    elif r < p_swap + p_transfer + p_smooth:
        cand = smooth_and_fix(cur_h, target_sum)
    else:
        # random perturb many indices
        cand = cur_h.copy()
        m = max(2, int(0.05*n))  # perturb 5% entries
        idx = rng.choice(n, size=m, replace=False)
        # add random delta in [-max_delta, max_delta]
        max_delta = 0.2
        deltas = (rng.random(m) - 0.5) * 2 * max_delta
        # apply and clip
        cand[idx] += deltas
        cand = np.clip(cand, 0.0, 1.0)
        # enforce sum
        cur_sum = np.sum(cand)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # distribute diff across all indices proportionally to free capacity
            free = np.where((cand > 0) & (cand < 1))[0]
            if free.size >0:
                inc = diff / free.size
                cand[free] += inc
                cand = np.clip(cand, 0.0, 1.0)
            # residual fix
            diff = target_sum - np.sum(cand)
            for ii in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and cand[ii] < 1.0:
                    add = min(diff, 1.0 - cand[ii])
                    cand[ii] += add
                    diff -= add

def enforce_sum(h, target):
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # distribute diff to free entries
    free = np.where((h > 0) & (h < 1))[0]
    if free.size > 0:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    # residual adjust
    cur = np.sum(h)
    diff = target - cur
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0.0, 1.0)

def enforce_sum(h, target):
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Distribute to free indices
    free = np.where((h > 0) & (h < 1))[0]
    if free.size:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    # Residual fix
    cur = np.sum(h)
    diff = target - cur
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0.0, 1.0)

    best_h = cur_h.copy()

    max_no_improve = 3000
    last_improve = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget * 0.95:
            break
        T = T0 * max(0.0, 1.0 - elapsed / budget)
        r = rng.random()
        # move probabilities
        if r < 0.2:  # swap
            i, j = rng.choice(n, 2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.2 + 0.4:  # transfer
            i, j = rng.choice(n, 2, replace=False)
            # direction random
            if rng.random() < 0.5:
                donor, recv = i, j
            else:
                donor, recv = j, i
            max_tr = min(cur_h[donor], 1.0 - cur_h[recv])
            if max_tr <= 0:
                continue
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        elif r < 0.2 + 0.4 + 0.1:  # smooth
            cand = smooth_and_fix(cur_h, target_sum, dx)
        elif r < 0.2 + 0.4 + 0.1 + 0.1:  # random perturb many

            max_delta = 0.2

import time
import numpy as np

def compute_c5(h, dx):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum(h, target):
    """Adjust h in-place (and return it) so that sum(h) == target while keeping 0≤h≤1."""
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # distribute diff among free entries
    free = np.where((h > 0) & (h < 1))[0]
    if free.size:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    # residual fix
    cur = np.sum(h)
    diff = target - cur
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0.0, 1.0)

def smooth_and_fix(h, target, dx):
    """One smoothing pass (Gaussian kernel) followed by sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    sm = enforce_sum(sm, target)
    return sm

def anneal(start_h, start_val, target, dx, budget, rng):
    """Simulated annealing for a given time budget (seconds)."""
    n = len(start_h)
    cur_h = start_h.copy()
    cur_val = start_val
    best_h = cur_h.copy()
    best_val = cur_val

    T0 = 0.03                     # initial temperature
    max_no_improve = 3000        # iterations without improvement before a kick
    last_improve = 0
    iter_count = 0
    start_time = time.time()

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget * 0.95:
            break
        T = T0 * max(0.0, 1.0 - elapsed / budget)
        r = rng.random()
        # ---- move selection ------------------------------------------------
        if r < 0.20:                       # swap two entries
            i, j = rng.choice(n, 2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.60:                     # transfer mass from i to j
            i, j = rng.choice(n, 2, replace=False)
            if rng.random() < 0.5:
                donor, recv = i, j
            else:
                donor, recv = j, i
            max_tr = min(cur_h[donor], 1.0 - cur_h[recv])
            if max_tr <= 0.0:
                continue
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        elif r < 0.70:                     # smoothing move
            cand = smooth_and_fix(cur_h, target, dx)
        elif r < 0.80:                     # perturb many entries
            cand = cur_h.copy()
            m = max(2, int(0.05 * n))
            idx = rng.choice(n, size=m, replace=False)
            max_delta = 0.2
            deltas = (rng.random(m) - 0.5) * 2.0 * max_delta
            cand[idx] += deltas
            cand = np.clip(cand, 0.0, 1.0)
            cand = enforce_sum(cand, target)
        else:                               # gradient move targeting worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = int(np.argmax(corr))
            k = idx_max - (n - 1)           # shift causing max overlap
            neigh = np.zeros(n)
            if k >= 0:
                neigh[: n - k] += cur_h[k:]
                neigh[k:] += cur_h[: n - k]
            else:
                kk = -k
                neigh[kk:] += cur_h[: n - kk]
                neigh[: n - kk] += cur_h[kk:]
            grad = dx * (1.0 - neigh)
            grad_centered = grad - np.mean(grad)   # zero‑sum direction
            delta_vec = -grad_centered
            mabs = np.max(np.abs(delta_vec))
            if mabs > 0.2:
                delta_vec = delta_vec * (0.2 / mabs)
            cand = cur_h + delta_vec
            cand = np.clip(cand, 0.0, 1.0)
            cand = enforce_sum(cand, target)

        # ---- evaluate ----------------------------------------------------
        new_val = compute_c5(cand, dx)
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                last_improve = iter_count
                if best_val < 0.3808:
                    break

        iter_count += 1
        # ---- kick if stagnating -----------------------------------------
        if iter_count - last_improve > max_no_improve:
            cur_h = best_h.copy()
            noise = rng.normal(scale=0.02, size=n)
            cur_h += noise
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = enforce_sum(cur_h, target)
            cur_val = compute_c5(cur_h, dx)
            last_improve = iter_count

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- problem size --------------------------------------------------
    n = 200                         # more degrees of freedom than the 80‑point baseline
    target = n / 2.0
    dx = 2.0 / n

    # ----- initial construction -----------------------------------------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if len(init) != n:
            # up‑sample by linear interpolation
            x_old = np.linspace(0.0, 2.0, len(init), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, init)
        else:
            h0 = init.copy()
    else:
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    h0 = enforce_sum(h0, target)
    best_h = h0.copy()
    best_val = compute_c5(best_h, dx)

    # ----- main simulated annealing phase -------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 30.0:
        # use the bulk of time for a thorough anneal
        best_h, best_val = anneal(best_h, best_val, target, dx, remaining * 0.85, rng)

    # ----- final refinement (short anneal) -----------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        best_h, best_val = anneal(best_h, best_val, target, dx, remaining * 0.95, rng)

    # ----- safety post‑processing ---------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = np.sum(best_h)
    if abs(cur_sum - target) > 1e-9:
        best_h = enforce_sum(best_h, target)

    return best_h, best_val, n
