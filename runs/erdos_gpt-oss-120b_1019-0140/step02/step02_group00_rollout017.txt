def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1) as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max‑correlation w.r.t. h for given active shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]       -= h[:n - s]
            grad[:n - s]   -= h[s:]
        grad += 1.0
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on C5."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    """Short SLSQP to improve a small‑scale candidate."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def random_perturb(h, rng, delta=0.05):
    """Add zero‑sum random noise of magnitude delta and re‑project."""
    n = h.size
    u = rng.uniform(-delta, delta, size=n)
    u -= np.mean(u)
    h_new = h + u
    return project_onto_box_sum(h_new, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_small = 128
    n_large = 512

    # --- Phase 1: small‑scale exploration with SLSQP ---
    base_small = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_small = upsample(init_full, n_small) if init_full.size != n_small else init_full.copy()
        init_small = project_onto_box_sum(init_small, n_small / 2.0)
        base_small.append(init_small)

    base_small.append(thue_morse(n_small))
    for _ in range(3):
        base_small.append(random_binary_balanced(n_small, rng))
    for _ in range(3):
        v = rng.random(n_small)
        base_small.append(project_onto_box_sum(v, n_small / 2.0))

    best_h_small = None
    best_score_small = float('inf')
    for cand in base_small:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=800)
        if sc_opt < best_score_small:
            best_score_small, best_h_small = sc_opt, h_opt.copy()

    while time_left() > 8.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=500)
        if sc_opt < best_score_small:
            best_score_small, best_h_small = sc_opt, h_opt.copy()

    if best_h_small is None:
        best_h_small = project_onto_box_sum(np.full(n_small, 0.5, dtype=float), n_small / 2.0)
        best_score_small = compute_c5(best_h_small)

    # --- Phase 2: upsample to larger grid ---
    h_large = upsample(best_h_small, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    best_h = h_large.copy()
    best_score = compute_c5(best_h)

    # --- Phase 3: momentum gradient refinement on large grid ---
    if time_left() > 5.0:
        remaining = time_left()
        # Rough estimate: ~0.0008 s per iteration on this size
        est_iters = int(min(80000, max(15000, remaining * 800)))
        h_ref, sc_ref = gradient_descent_momentum(best_h, n_iter=est_iters,
                                                  step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_h, best_score = h_ref.copy(), sc_ref

    # --- Phase 4: random local perturbations (hill climbing) ---
    while time_left() > 2.0:
        # perturbation magnitude shrinks as time runs out
        frac = max(0.01, time_left() / budget_s)
        delta = 0.1 * frac
        cand = random_perturb(best_h, rng, delta=delta)
        sc = compute_c5(cand)
        if sc < best_score:
            best_h, best_score = cand.copy(), sc

    # final safeguard projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
