def gradient_max_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    d = max_idx - (n - 1)  # shift as defined in correlation
    grad = np.zeros_like(h)
    if d >= 0:
        # i >= d: add 1
        grad[d:] += 1.0
        # subtract h[i-d] for i>=d
        grad[d:] -= h[:n - d]
        # subtract h[i+d] for i <= n-1-d
        grad[:n - d] -= h[d:]
    else:
        p = -d
        # i <= n-1-p: add 1
        grad[:n - p] += 1.0
        # subtract h[i+p] for i <= n-1-p
        grad[:n - p] -= h[p:]
        # subtract h[i-p] for i >= p
        grad[p:] -= h[:n - p]
    return grad, d

        grad[p:] -= h[:n - p]

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_feasible(v, target, max_iter=50):
    # Project vector v onto {x: 0<=x<=1, sum(x)=target}
    # Binary search on lambda: x = clip(v - lam, 0, 1)
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    # Tiny correction for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        idx = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if idx.size > 0:
            inc = diff / idx.size
            x[idx] += inc
            x = np.clip(x, 0.0, 1.0)
    return x

def gradient_max_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    d = max_idx - (n - 1)  # shift corresponding to max overlap
    grad = np.zeros_like(h)
    if d >= 0:
        # indices i >= d
        grad[d:] += 1.0
        grad[d:] -= h[:n - d]
        # indices i <= n-1-d
        grad[:n - d] -= h[d:]
    else:
        p = -d
        grad[:n - p] += 1.0
        grad[:n - p] -= h[p:]
        grad[p:] -= h[:n - p]
    return grad, d

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_feasible(v, target, max_iter=50):
    # Project onto the intersection of box [0,1]^n and sum(v)=target
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        idx = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if idx.size:
            inc = diff / idx.size
            x[idx] += inc
            x = np.clip(x, 0.0, 1.0)
    return x

def gradient_max_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    d = max_idx - (n - 1)          # shift giving the maximal overlap
    grad = np.zeros_like(h)
    if d >= 0:
        grad[d:] += 1.0
        grad[d:] -= h[: n - d]
        grad[: n - d] -= h[d:]
    else:
        p = -d
        grad[: n - p] += 1.0
        grad[: n - p] -= h[p:]
        grad[p:] -= h[: n - p]
    return grad, d

def pgd_optimize(h_start, target, deadline, rng, max_iters=2000):
    h = h_start.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    alpha = 0.5
    decay = 0.995
    for _ in range(max_iters):
        if time.time() >= deadline - 0.05:
            break
        grad, _ = gradient_max_shift(h)
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, target)
        new_val = compute_c5(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
            if best_val < 0.3808:
                break
        if new_val <= cur_val:
            h = h_new
            cur_val = new_val
        alpha *= decay
        if alpha < 1e-8:
            break
    return best_h, best_val

def random_feasible_vector(n, rng):
