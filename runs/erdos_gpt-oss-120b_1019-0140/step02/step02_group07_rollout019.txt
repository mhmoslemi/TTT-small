function targeted_hill_climb(h, time_end, rng):
  while time.time()<time_end:
    corr = np.correlate(h.astype(float), 1 - h.astype(float), mode='full')
    max_val = corr.max()
    if max_val * dx <= best_c5: # actually best_c5 is previous best maybe
    best found

import time
import numpy as np

def compute_c5(arr):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by correlation."""
    n = arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * (2.0 / n)

def continuous_anneal(start_h, end_time, rng):
    """Simulated annealing on continuous values while preserving the sum."""
    h = start_h.copy()
    n = h.size
    target = n / 2.0
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    init_temp, final_temp = 1e-2, 1e-7
    max_iter = 200_000
    it = 0
    while time.time() < end_time and it < max_iter:
        inc = np.where(h < 1.0 - 1e-12)[0]
        dec = np.where(h > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0.0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new = compute_c5(h_new)
        dE = new - cur
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0.0 or rng.random() < np.exp(-dE / (T + 1e-18)):
            h, cur = h_new, new
            if new < best:
                best, best_h = new, h_new.copy()
        it += 1
    # exact sum correction
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target - np.sum(best_h)
        for k in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[k], 1.0 - best_h[k])
            best_h[k] += adj
            residual = target - np.sum(best_h)
    return best_h, best

def binary_hill_climb(start_bin, end_time, rng):
    """Swap‑based hill climbing on binary vectors."""
    h = start_bin.astype(float).copy()
    best_h, best = h.copy(), compute_c5(h)
    while time.time() < end_time:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        cur = compute_c5(h)
        if cur < best - 1e-12:
            best, best_h = cur, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best

def get_conflict_sets(h, shift):
    """Return indices i (1→0) and j (0→1) that form opposite conflicts for a given shift."""
    n = h.size
    if shift == 0 or shift >= n:
        return np.array([], dtype=int), np.array([], dtype=int)
    seg = n - shift
    h1, h2 = h[:seg], h[shift:]
    A = np.where((h1 == 1.0) & (h2 == 0.0))[0]
    B = np.where((h1 == 0.0) & (h2 == 1.0))[0]
    return A, B

def targeted_hill_climb(start_bin, end_time, rng):
    """Hill climbing that targets the lag with maximal overlap."""
    h = start_bin.astype(float).copy()
    n = h.size
    best_h, best = h.copy(), compute_c5(h)
    while time.time() < end_time:
        corr = np.correlate(h, 1.0 - h, mode="full")
        lag = int(np.argmax(corr) - (n - 1))
        shift = abs(lag)
        A, B = get_conflict_sets(h, shift)
        performed = False
        if A.size and B.size:
            i = rng.choice(A)
            j = rng.choice(B)
            h[i], h[j] = 0.0, 1.0
            cur = compute_c5(h)
            if cur < best - 1e-12:
                best, best_h = cur, h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
            performed = True
        if not performed:
            # fall back to a random swap
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1.0 - h)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = 0.0, 1.0
            cur = compute_c5(h)
            if cur < best - 1e-12:
                best, best_h = cur, h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
        if best <= 0.3808:
            break
    return best_h, best
