def local_optimize(h, O, n):
    while True:
        O = np.correlate(h, 1 - h, mode='full')[n-1:]
        cur_max = O.max()
        k_max = np.argmax(O)
        # Compute neighbor sums for shift k_max:
        left_idx = np.arange(n) - k_max
        right_idx = np.arange(n) + k_max
        # mask for bounds
        mask_left = left_idx >= 0
        mask_right = right_idx < n
        neighbor_sum = np.zeros(n, dtype=int)
        neighbor_sum[mask_left] += h[left_idx[mask_left]]
        neighbor_sum[mask_right] += h[right_idx[mask_right]]
        # Now find candidate swap:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)

        min_val = None

                i_min = i

        # Continue loop to recompute O

def improve_until_stuck(h, rng):

    # Greedy improve using local optimum step until no improvement.

    while True:
        O = np.correlate(h, 1 - h, mode='full')[n-1:]
        cur_max = O.max()
        k_max = np.argmax(O)
        # compute neighbor sum for all positions at shift k_max
        left = np.arange(n) - k_max
        right = np.arange(n) + k_max
        neighbor_sum = np.zeros(n, dtype=int)
        mask_left = left >= 0
        neighbor_sum[mask_left] += h[left[mask_left]]
        mask_right = right < n
        neighbor_sum[mask_right] += h[right[mask_right]]
        # find candidate i (ones) with minimal neighbor_sum, j (zeros) with maximal neighbor_sum
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i_vals = neighbor_sum[ones]
        j_vals = neighbor_sum[zeros]
        i_min_idx = np.argmin(i_vals)
        j_max_idx = np.argmax(j_vals)
        i = ones[i_min_idx]
        j = zeros[j_max_idx]
        if neighbor_sum[i] >= neighbor_sum[j]:
            # no improving swap for this k_max
            break

        h[i] = 0
        h[j] = 1
        # Loop again
    # after loop compute final O and max
    O = np.correlate(h, 1 - h, mode='full')[n-1:]
    max_val = O.max()
    return h, max_val

import time, math, numpy as np
from scipy.optimize import differential_evolution

def compute_c5(arr):
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def adjust_binary_to_target(h, target_ones, rng):
    # h is binary array 0/1 but may have sum not equal target_ones
    cur = int(np.sum(h))
    diff = target_ones - cur
    if diff > 0:
        zeros = np.flatnonzero(1 - h)
        idx = rng.choice(zeros, diff, replace=False)
        h[idx] = 1
    elif diff < 0:
        ones = np.flatnonzero(h)
        idx = rng.choice(ones, -diff, replace=False)
        h[idx] = 0
    return h

def random_binary_vector(n, target_ones, rng):
    h = np.zeros(n, dtype=int)
    h[rng.choice(n, target_ones, replace=False)] = 1
    return h

def improve_until_stuck(h, rng):
    n = len(h)
    # h is binary numpy array int
    while True:
        O = np.correlate(h, 1 - h, mode='full')[n-1:]
        cur_max = O.max()
        k_max = int(np.argmax(O))
        # neighbor sum at distance k_max
        left = np.arange(n) - k_max
        right = np.arange(n) + k_max
        neigh = np.zeros(n, dtype=int)
        mask_left = left >= 0
        neigh[mask_left] += h[left[mask_left]]
        mask_right = right < n
        neigh[mask_right] += h[right[mask_right]]
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i_vals = neigh[ones]
        j_vals = neigh[zeros]
        i_idx = ones[np.argmin(i_vals)]
        j_idx = zeros[np.argmax(j_vals)]
        if neigh[i_idx] >= neigh[j_idx]:
            break
        # perform swap
        h[i_idx] = 0
        h[j_idx] = 1
        # continue loop
    O = np.correlate(h, 1 - h, mode='full')[n-1:]
    return h, O.max()

def binary_local_search(start_h, time_end, rng):
    # run greedy improvements with occasional random perturbation
    h = start_h.copy()
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1 - h, mode='full')[len(h)-1:])
    while time.time() < time_end:
        # try improvement
        h, cur_val = improve_until_stuck(h, rng)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # occasional random perturbation to escape
        if rng.random() < 0.1:
            # random swap of a few positions
            ones = np.flatnonzero(best_h)
            zeros = np.flatnonzero(1 - best_h)
            if ones.size and zeros.size:
                for _ in range(rng.integers(1, 4)):
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    best_h[i] = 0
                    best_h[j] = 1
        # early exit
        if best_val * (2.0 / len(h)) <= 0.3808:
            break
    return best_h, best_val

def de_optimize(initial_h, time_end, rng):
    n = len(initial_h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def repair(x):
        x = np.clip(x, 0.0, 1.0)
        sumx = np.sum(x)
        diff = target_sum - sumx
        x = x + diff / n
        x = np.clip(x, 0.0, 1.0)
        residual = target_sum - np.sum(x)
        if abs(residual) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for idx in idxs:
                if abs(residual) < 1e-12:
                    break
                adj = np.clip(residual, -x[idx], 1 - x[idx])
                x[idx] += adj
                residual = target_sum - np.sum(x)
        return x
    def objective(x):
        x = repair(x)
        return compute_c5(x)
    # Determine number of iterations based on remaining time
    remaining = time_end - time.time()
    # Use moderate settings
    maxiter = max(5, int(remaining // 0.5))
    popsize = max(5, int(n // 20))
    result = differential_evolution(objective, bounds, seed=int(rng.integers(1<<32)), maxiter=maxiter, popsize=popsize, polish=False, updating='deferred')
    x_best = repair(result.x)
    return x_best, compute_c5(x_best)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None
    target_ones = n // 2
    dx = 2.0 / n
    best_h = None
    best_raw = np.inf

    # Helper to update best
    def try_candidate(h_candidate):
        nonlocal best_h, best_raw
        c5 = compute_c5(h_candidate)
        raw = c5 / dx
        if raw < best_raw:
            best_raw = raw
            best_h = h_candidate.copy()
    # Start with initial continuous if available
    if h0 is not None:
        # ensure feasibility (sum*dx=1)
        cur_sum = np.sum(h0)
        target_sum = target_ones
        if abs(cur_sum - target_sum) > 1e-12:
            # repair by scaling difference
            diff = target_sum - cur_sum
            h0 = h0 + diff / n
            h0 = np.clip(h0, 0.0, 1.0)
            # residual fix
            residual = target_sum - np.sum(h0)
            for i in range(n):
                if abs(residual) < 1e-12:
                    break
                adj = np.clip(residual, -h0[i], 1 - h0[i])
                h0[i] += adj
                residual = target_sum - np.sum(h0)
        try_candidate(h0)

    # Phase 1: binary greedy search with random restarts
    while time.time() - start_time < budget_s * 0.5:
        # generate start: perturb best if exists, else random
        if best_h is not None and rng.random() < 0.7:
            start = best_h.copy()
            # make binary if not
            start_bin = (start >= 0.5).astype(int)
            # random few swaps
            for _ in range(rng.integers(1, 4)):
                ones = np.flatnonzero(start_bin)
                zeros = np.flatnonzero(1 - start_bin)
                if ones.size and zeros.size:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    start_bin[i] = 0
                    start_bin[j] = 1
        else:
            start_bin = random_binary_vector(n, target_ones, rng)
        # Local greedy improvement
        end_time = start_time + budget_s * 0.5
        # allocate limited time chunk for this run
        chunk = min(5.0, (budget_s - (time.time() - start_time)) * 0.4)
        h_opt, raw_opt = binary_local_search(start_bin, time.time() + chunk, rng)
        try_candidate(h_opt)
        if best_raw * dx <= 0.3808:
            break

    # Phase 2: differential evolution on continuous (if time left)
    if time.time() - start_time < budget_s * 0.9:
        remaining = start_time + budget_s - time.time()
        end_de = time.time() + remaining * 0.6
        # use current best as initial guess for DE (if binary)
        init = best_h if best_h is not None else random_binary_vector(n, target_ones, rng).astype(float)
        # DE cannot use initial guess directly, but we can start from random seeds
        # We'll just run DE once.
        h_de, c5_de = de_optimize(init, end_de, rng)
        try_candidate(h_de)
        # also try binary rounding of DE result and improve
        h_de_bin = (h_de >= 0.5).astype(int)
        # adjust sum if needed
        cur_sum = h_de_bin.sum()
        diff = target_ones - cur_sum
        if diff != 0:
            zeros = np.flatnonzero(1 - h_de_bin)
            ones = np.flatnonzero(h_de_bin)
            if diff > 0:
                idx = rng.choice(zeros, diff, replace=False)
                h_de_bin[idx] = 1
            else:
                idx = rng.choice(ones, -diff, replace=False)
                h_de_bin[idx] = 0
        # local improve
        h_bin_opt, raw_bin_opt = binary_local_search(h_de_bin, time.time() + (budget_s - (time.time() - start_time)) * 0.3, rng)
        try_candidate(h_bin_opt)

    # Ensure final best_h satisfies constraints, clip
    if best_h is None:
        # fallback: random binary
        best_h = random_binary_vector(n, target_ones, rng).astype(float)
        best_raw = compute_c5(best_h) / dx
    best_h = np.clip(best_h, 0.0, 1.0)
    # final sum correction (should hold)
    cur_sum = np.sum(best_h)
    target_sum = target_ones
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        best_h = best_h + diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        # residual fix
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n

import time, math, numpy as np
from scipy.optimize import differential_evolution

def compute_c5(arr):
    """C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated by correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_vector(n, ones, rng):
    v = np.zeros(n, dtype=int)
    v[rng.choice(n, ones, replace=False)] = 1
    return v

def improve_until_stuck(h, rng):
    """Greedy reduction of max overlap for binary h."""
    n = len(h)
    while True:
        O = np.correlate(h, 1 - h, mode='full')[n - 1 :]
        cur_max = O.max()
        k = int(np.argmax(O))
        left = np.arange(n) - k
        right = np.arange(n) + k
        neigh = np.zeros(n, dtype=int)
        mask = left >= 0
        neigh[mask] += h[left[mask]]
        mask = right < n
        neigh[mask] += h[right[mask]]
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(neigh[ones])]
        j = zeros[np.argmax(neigh[zeros])]
        if neigh[i] >= neigh[j]:
            break
        h[i] = 0
        h[j] = 1
    O = np.correlate(h, 1 - h, mode='full')[n - 1 :]
    return h, O.max()

def binary_local_search(start_h, time_end, rng):

    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    x = x + diff / len(x)
    x = np.clip(x, 0.0, 1.0)
    residual = target_sum - np.sum(x)
    if abs(residual) > 1e-12:

                break
            adj = np.clip(residual, -x[i], 1 - x[i])
            x[i] += adj
            residual = target_sum - np.sum(x)
    return x

def de_optimize(n, target_sum, time_end, rng):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        x = repair_vector(x, target_sum, rng)
        return compute_c5(x)
    remaining = max(0.0, time_end - time.time())
    maxiter = max(5, int(remaining // 1.0))

        obj,
        bounds,
        seed=seed,
        maxiter=maxiter,
        popsize=popsize,
        polish=False,
        updating='deferred',
    )
    x_best = repair_vector(result.x, target_sum, rng)
    return x_best, compute_c5(x_best)
