import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n=v.size
    if s <= lo*n:
        return np.full_like(v, lo)
    if s >= hi*n:
        return np.full_like(v, hi)
    lam_low = np.min(v)-hi
    lam_high = np.max(v)-lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - 10*hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10*lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high)*0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n=h.shape[0]
    corr=np.correlate(h,1.0-h,mode='full')
    return np.max(corr)*(2.0/n)

def upsample(arr, new_n):
    m=arr.shape[0]
    if m==new_n:
        return arr.copy()
    x_old=np.linspace(0,1,m,endpoint=False)
    x_new=np.linspace(0,1,new_n,endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)  # linear interp works both ways

def compute_gradients_for_shifts(h, shifts):
    n=h.size
    grad=np.zeros_like(h)
    for shift in shifts:
        if shift>=0:
            term1=np.zeros(n); term2=np.zeros(n)
            term1[:n-shift]=h[shift:]
            term2[shift:]=h[:n-shift]
        else:
            s=-shift
            term1=np.zeros(n); term2=np.zeros(n)
            term1[s:]=h[:n-s]
            term2[:n-s]=h[s:]
        grad+= (1.0 - term1 - term2)
    if shifts.size>0:
        grad/=shifts.size
    return grad

def gradient_descent_momentum_custom(h0, max_iter, time_left, rng, beta=0.9, step0=0.5):
    h=h0.copy()
    n=h.size; s=n/2.0
    best_h=h.copy(); best_score=compute_c5(h)
    momentum=np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left()<=0:
            break
        corr=np.correlate(h,1.0-h,mode='full')
        overlaps=corr*(2.0/n)
        max_val=overlaps.max()
        eps=max(1e-12, max_val*1e-6)
        shift_idxs=np.where(overlaps>=max_val-eps)[0]-(n-1)
        grad=compute_gradients_for_shifts(h, shift_idxs)
        momentum=beta*momentum+(1-beta)*grad
        eta=step0/np.sqrt(it)
        h_new=h-eta*momentum
        # occasional jitter
        if it%200==0 and time_left()>2.0:
            h_new+=rng.uniform(-0.01,0.01,size=n)
        h_new=project_onto_box_sum(h_new,s)
        sc=compute_c5(h_new)
        if sc<best_score:
            best_score=sc; best_h=h_new.copy()
        h=h_new
    return best_h,best_score

def thue_morse_seq(m):
    # generate 0/1 Thue-Morse
    i=np.arange(m)
    # compute parity of popcount using bitwise xor reduce: using builtin count bits?
    # Use vectorized popcount via bitwise operations on uint64.
    # Convert to uint64
    v=i.astype(np.uint64)
    # popcount using Kernighan's method vectorized might be slower; just use Python loop for small m.
    return np.fromiter(((bin(int(x)).count('1')&1) for x in i), dtype=float, count=m)

def slsqp_optimize_candidate(h0, max_iter=200):
    n=h0.shape[0]
    t0=compute_c5(h0)
    x0=np.concatenate([h0,[t0]])
    bounds=[(0.0,1.0)]*n+[(0.0,1.0)]
    def eq_sum(x):
        return np.sum(x[:n])-n/2.0
    def ineq_fun(x):
        h=x[:n]; t=x[-1]
        corr=np.correlate(h,1.0-h,mode='full')*(2.0/n)
        return t-corr
    cons=[{'type':'eq','fun':eq_sum},
          {'type':'ineq','fun':ineq_fun}]
    res=opt.minimize(lambda x:x[-1],x0,method='SLSQP',
                     bounds=bounds,constraints=cons,
                     options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    h_opt=res.x[:n]
    h_opt=project_onto_box_sum(h_opt,n/2.0)
    t_opt=compute_c5(h_opt)
    return h_opt,t_opt

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def gradient_descent_momentum_custom(h0, max_iter, time_left, rng,
                                     beta=0.9, step0=0.5):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * (2.0 / n)
        max_val = overlaps.max()
        eps = max(1e-12, max_val * 1e-6)
        shift_idxs = np.where(overlaps >= max_val - eps)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shift_idxs)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 200 == 0 and time_left() > 2.0:
            h_new += rng.uniform(-0.01, 0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse_seq(m):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(m)),
                       dtype=float, count=m)

def slsqp_optimize_candidate(h0, max_iter=200):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [
        {'type': 'eq', 'fun': eq_sum},
        {'type': 'ineq', 'fun': ineq_fun}
    ]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12,
                                'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    n_target = 512
    best_h = None
    best_score = float('inf')

    # seed candidates
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = upsample(init_full, n_target) if init_full.shape[0] != n_target else init_full.copy()
        cand = project_onto_box_sum(cand, n_target / 2.0)
        sc = compute_c5(cand)
        if sc < best_score:
            best_score, best_h = sc, cand.copy()

    tm = project_onto_box_sum(thue_morse_seq(n_target), n_target / 2.0)
    sc = compute_c5(tm)
    if sc < best_score:
        best_score, best_h = sc, tm.copy()

    sin_seed = (np.sin(2 * np.pi * np.arange(n_target) / n_target) + 1) / 2.0
    sin_seed = project_onto_box_sum(sin_seed, n_target / 2.0)
    sc = compute_c5(sin_seed)
    if sc < best_score:
        best_score, best_h = sc, sin_seed.copy()

    # main random restarts
    while time_left() > 6.0:
        v = rng.random(n_target)
        h0 = project_onto_box_sum(v, n_target / 2.0)
        # estimate feasible iterations (rough 0.001 s per iter)
        max_iter = int(min(8000, max(2000, time_left() * 400)))
        step0 = rng.uniform(0.3, 0.7)
        h_opt, sc_opt = gradient_descent_momentum_custom(
            h0, max_iter, time_left, rng, beta=0.9, step0=step0)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= 0.3808:
                break

    # optional refinement on a smaller grid
    if best_h is not None and time_left() > 20.0:
        n_small = 256
        h_small = downsample(best_h, n_small)
        h_small = project_onto_box_sum(h_small, n_small / 2.0)
        h_ref, _ = slsqp_optimize_candidate(h_small, max_iter=300)
        h_up = upsample(h_ref, n_target)
        h_up = project_onto_box_sum(h_up, n_target / 2.0)
        sc_up = compute_c5(h_up)
        if sc_up < best_score:
            best_score, best_h = sc_up, h_up.copy()

    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = np.full(n_target, 0.5)
        best_h = project_onto_box_sum(best_h, n_target / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
