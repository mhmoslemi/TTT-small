def compute_c5_bin(b):
    # b: np.array int or float binary
    n = b.shape[0]
    corr = np.correlate(b, b, mode='full')
    dot = corr[n:]  # indices n to 2n-2 inclusive for shifts 1..n-1
    min_dot = np.min(dot)  # integer
    return 1.0 - (2.0 / n) * min_dot
```

def correlation_fft(b):
    # compute full correlation of b with b.
    # Use FFT with length 2*n for zero padding to avoid circular convolution.
    m = 1
    while m < 2*n:
        m <<= 1
    # Not necessary to compute next power of two; np.fft works fine.
    fb = np.fft.rfft(b, n=2*n)
    corr = np.fft.irfft(fb * np.conj(fb), n=2*n)
    # The linear convolution result length 2*n - 1.
    # For auto-correlation (b with itself) this yields values for k from -(n-1) to n-1.
    # For shift k >=0, dot_k = corr[n-1 + k]
    return corr
```

def subgrad_descent(n, seed, budget_s):
    rng = np.random.default_rng(seed)
    dx = 2.0/n
    target_sum = n/2.0
    # initial h random feasible
    h = generate_random_feasible(n, target_sum, rng)
    best_h = h.copy()
    best_val = compute_c5_vec(h, dx)
    start = time.time()
    # run until time
    iteration = 0
    while time.time() - start < budget_s:
        # compute f_k for all k
        # we can compute overlaps via convolution: overl = np.correlate(h, 1-h, mode='full')[n-1:]
        # Actually full includes shift zero at index n-1; for k positive, overl_k = sum_i h[i] * (1 - h[i+k]) = same as correlation.
        corr = np.correlate(h, 1 - h, mode='full')
        # get overlaps for k = 1..n-1: indices n..2n-2
        overlaps = corr[n:]
        # find worst shift (max overlap)
        k = np.argmax(overlaps) + 1  # shift value
        # compute gradient for this k
        g = np.zeros_like(h)
        # for i from 0..n-1-k: add (1 - h[i+k])
        g[:n-k] += (1 - h[k:])
        # for i from k..n-1: subtract h[i-k]
        g[k:] -= h[:n-k]
        # compute step size; maybe use diminishing step: alpha = 0.1 * (target_sum) / (iteration+1)**0.5
        alpha = 0.5 / np.sqrt(iteration + 1)  # adjust scaling
        h_new = h - alpha * g
        # project into [0,1] and sum constraint
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        # evaluate
        val = compute_c5_vec(h_new, dx)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h_new.copy()
            # optional: break if below target
            if best_val < 0.3808:
                break
        h = h_new
        iteration += 1
    return best_h, best_val

def compute_c5_vec(h, dx):
    """Return max overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Adjust vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible continuous vector (values in [0,1] and exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x.fill(target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    # fix residual sum difference
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def refine_continuous(h, dx, target_sum, time_limit, start_time, rng):
    """Local coordinate descent transferring mass between entries."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.1
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.9
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val


# ---------- Constraint classes ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        return np.sum(x[: self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overl * self.dx


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if "initial_h_values" in globals():
        init_vec = np.array(initial_h_values, dtype=float)
        n0 = len(init_vec)
        desired_n = max(n0, 120)
        if desired_n != n0:
            # linear interpolation to enlarge the vector
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, desired_n, endpoint=False)
            h0 = np.interp(x_new, x_old, init_vec)
        else:
            h0 = init_vec
    else:
        desired_n = 120
        h0 = None

    n = desired_n
    target_sum = n / 2.0
    dx = 2.0 / n

    # Initialise best solution
    if h0 is not None:
        best_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        best_h = generate_random_feasible(n, target_sum, rng)
    best_val = compute_c5_vec(best_h, dx)

    # Build constraints for SLSQP
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{"type": "eq", "fun": sum_con}]
    constraints += [{"type": "ineq", "fun": oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # Main optimisation loop (SLSQP + refinement)
    slsqp_deadline = start_time + budget_s * 0.80
    while time.time() < slsqp_deadline:
        # Random start; occasionally reuse current best as start
        if h0 is not None and rng.random() < 0.1:
            h_start = best_h.copy()
        else:
            h_start = generate_random_feasible(n, target_sum, rng)
        t_start = compute_c5_vec(h_start, dx)
        x0 = np.concatenate([h_start, [t_start]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 1500, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_sum(h_opt, target_sum)

        # Local refinement
        refine_deadline = start_time + budget_s - 2.0
        h_ref, val_ref = refine_continuous(
            h_opt, dx, target_sum, refine_deadline, start_time, rng
        )
        if val_ref < best_val - 1e-12:
            best_h, best_val = h_ref, val_ref
            if best_val < 0.3808:
                break

    # Final refinement with any remaining time
    if time.time() < start_time + budget_s - 1.0:
        final_deadline = start_time + budget_s
        best_h, best_val = refine_continuous(
            best_h, dx, target_sum, final_deadline, start_time, rng
        )

    # Ensure feasibility before returning
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
