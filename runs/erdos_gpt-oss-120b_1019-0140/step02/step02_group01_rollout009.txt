def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)

def simulated_annealing(initial_h, time_limit, rng):
    n = len(initial_h)
    M = n//2
    cur_h = np.clip(initial_h, 0.0, 1.0)
    # Repair sum (if needed)
    cur_sum = cur_h.sum()
    if abs(cur_sum - M) > 1e-9:
        cur_h += (M - cur_sum) / n
        cur_h = np.clip(cur_h, 0.0, 1.0)
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    # define parameters
    T_initial = 0.3
    T_final = 1e-4
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # progress fraction
        frac = elapsed / time_limit
        T = T_initial * (1 - frac) + T_final * frac
        # propose move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # direction flag
        direction = rng.integers(2)  # 0: i->j, 1: j->i
        if direction == 0:
            # moving from i to j
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])

            new_h[i] -= delta
            new_h[j] += delta

def random_feasible_h(n, M, rng):
    h = np.zeros(n)
    remaining = M
    indices = rng.permutation(n)
    for idx in indices:
        if remaining <= 0:
            break
        val = min(1.0, rng.random() * remaining)  # assign random up to remaining
        h[idx] = val
        remaining -= val
    # If any remaining > 0 after loop (unlikely unless random gave zero), fill first index
    if remaining > 0:
        for idx in indices:
            inc = min(1.0 - h[idx], remaining)
            h[idx] += inc
            remaining -= inc
            if remaining <= 1e-12:
                break
    return h

import time, math
import numpy as np

# ---------- Core evaluation ----------
def compute_c5(h):
    """Maximum overlap integral C5 for discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # convolution via FFT (full correlation = conv of h and reversed (1-h))
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, L)
    G = np.fft.rfft((1.0 - h)[::-1], L)
    conv = np.fft.irfft(H * G, L)[: 2 * n - 1]
    return dx * np.max(conv)


# ---------- Feasible vector generators ----------
def random_feasible_h(n, M, rng):
    """Binary feasible vector with exactly M entries equal to 1."""
    h = np.zeros(n)
    if M > 0:
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    return h


def upscale_h(h0, n):
    """Scale a parent vector h0 (length n0) to a new size n by linear interpolation."""
    n0 = len(h0)
    pos_new = np.linspace(0.0, 2.0, n, endpoint=False)
    pos_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    h_new = np.interp(pos_new, pos_old, h0)
    M = n // 2
    cur = np.sum(h_new)
    if cur > 0:
        h_new *= (M / cur)
    h_new = np.clip(h_new, 0.0, 1.0)
    cur = np.sum(h_new)
    if abs(cur - M) > 1e-9:
        h_new += (M - cur) / n
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new


# ---------- Simulated annealing ----------
def anneal(h_start, M, time_limit, rng):
    """Pairwise-mass-transfer simulated annealing with occasional random restarts."""
    n = h_start.shape[0]
    cur_h = np.clip(h_start, 0.0, 1.0)
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    T_initial, T_final = 0.4, 1e-5
    nosucc = 0

    while time.time() - start < time_limit:
        frac = (time.time() - start) / time_limit
        T = T_initial * (1 - frac) + T_final * frac

        # propose a pairwise transfer
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if rng.random() < 0.5:
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr <= 0:
                continue
            delta = rng.random() * max_tr
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            max_tr = min(cur_h[j], 1.0 - cur_h[i])
            if max_tr <= 0:
                continue
            delta = rng.random() * max_tr
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        new_val = compute_c5(new_h)
        dE = new_val - cur_val
        if dE < 0 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            cur_h, cur_val = new_h, new_val
            nosucc = 0
            if cur_val < best_val:

        else:

        # restart if stagnating
        if nosucc > 2000:
            cur_h = random_feasible_h(n, M, rng)

            nosucc = 0
