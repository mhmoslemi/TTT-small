def project_feasible(v, target_sum):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Adjust sum to target by distributing deficit/excess uniformly and re-clipping iteratively
    # Use simple loop up to max attempts
    diff = target_sum - v.sum()
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Determine indices that can be increased (if diff>0) or decreased (diff<0)
        if diff > 0:
            inc_idxs = np.where(v < 1.0)[0]
            if inc_idxs.size == 0:
                break
            # Add equally amount to all inc_idxs up to remaining diff
            add = diff / inc_idxs.size
            # Clip addition to capacity of each index: max possible is 1-v[i]
            cap = 1.0 - v[inc_idxs]
            add_clipped = np.minimum(add, cap)
            # Actually we may over-add if diff smaller than capacity sum; we can spread proportionally
            # Simpler: add to each idx a share limited by capacity; handle leftover.
            inc = np.minimum(add, cap)
            v[inc_idxs] += inc
        else:
            dec_idxs = np.where(v > 0.0)[0]
            if dec_idxs.size == 0:
                break
            sub = diff / dec_idxs.size  # diff negative
            cap = v[dec_idxs]  # max we can subtract
            sub_clipped = np.maximum(sub, -cap)
            dec = np.maximum(sub, -cap)
            v[dec_idxs] += dec  # note sub neg
        diff = target_sum - v.sum()
    # If still off a little, adjust using random indices.
    if abs(diff) > 1e-9:
        idxs = np.arange(len(v))
        rng = np.random.default_rng()
        rng.shuffle(idxs)
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and v[i] < 1.0:
                inc = min(diff, 1.0 - v[i])
                v[i] += inc
                diff -= inc
            elif diff < 0 and v[i] > 0.0:
                dec = min(-diff, v[i])
                v[i] -= dec
                diff += dec
    return v

def repair(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # distribute diff equally
        inc = diff / len(v)
        v += inc
        # clip again
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
    if abs(diff) > 1e-9:
        # small residual fix by random adjustments
        idxs = np.arange(len(v))
        rng = np.random.default_rng()
        rng.shuffle(idxs)
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and v[i] < 1.0:
                inc = min(diff, 1.0 - v[i])
                v[i] += inc
                diff -= inc
            elif diff < 0 and v[i] > 0.0:
                dec = min(-diff, v[i])
                v[i] -= dec
                diff += dec
    return v

Define compute_c5:

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate returns length 2n-1, maximum is required
    # We can compute full correlation
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

Define repair_vector:

def repair_vector(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # iterative adjustment to satisfy sum exactly
    diff = target - v.sum()
    max_iter = 100
    n = v.shape[0]
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        inc = diff / n
        v += inc
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    if abs(diff) > 1e-9:
        # final small residual fix by random walk
        rng = np.random.default_rng()
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and v[i] < 1.0:
                inc = min(diff, 1.0 - v[i])
                v[i] += inc
                diff -= inc
            elif diff < 0 and v[i] > 0.0:
                dec = min(-diff, v[i])
                v[i] -= dec
                diff += dec
    return v

def gradient_descent(h_start, target, deadline, max_iters=5000):
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(h)
    it = 0

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = rng.random(n)  # uniform random
    target = n / 2.0

    # Ensure feasibility
    h0 = repair_vector(h0, target)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Pre-create constraints for SLSQP (used later)
    # define equality sum constraint
    def sum_con(x, target_sum=target):
        return np.sum(x[:n]) - target_sum
    constraints = [{'type': 'eq', 'fun': sum_con}]
    # inequality constraints for each shift
    def make_ineq(k):
        def con(x, k=k):
            h = x[:n]
            # compute overl = sum_i h_i * (1 - h_{i+k}) for shift k
            if k >= 0:
                overl = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])
            # t is last variable
            return x[-1] - overl * (2.0 / n)
        return con
    for kk in range(-n + 1, n):
        if kk == 0:
            continue
        constraints.append({'type': 'ineq', 'fun': make_ineq(kk)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # Outer loop: random restarts + gradient descent
    # We'll allocate some time for gradient search; after half of total time, do final SLSQP.
    # Compute split: maybe use 70% time for GD, 30% for SLSQP refine.
    gd_deadline = start + budget_s * 0.7
    while time.time() < gd_deadline:
        # Choose start: random or best
        if rng.random() < 0.3:
            # random start
            h_start = rng.random(n)
            h_start = repair_vector(h_start, target)
        else:
            h_start = best_h.copy()
        # Perform gradient descent
        # allocate internal deadline for each run, e.g., time slice of 2 seconds
        slice_deadline = min(gd_deadline, time.time() + 2.0)
        h_opt, val_opt = gradient_descent(h_start, target, slice_deadline, max_iters=3000)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # Final refinement with SLSQP if time remains
    if time.time() < deadline - 0.5:
        # Use best_h as start
        x0 = np.concatenate([best_h, [best_val]])
        try:
            res = minimize(fun=lambda x: x[-1],
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = repair_vector(h_opt, target)
                val_opt = compute_c5(h_opt)
                if val_opt < best_val:
                    best_val = val_opt
                    best_h = h_opt.copy()
        except Exception:
            pass

    # Ensure final feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_vector(best_h, target)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated discretely."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def repair_vector(v, target):
    """Clip to [0,1] and adjust to achieve exact sum `target`."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    n = v.shape[0]
    # iterative equal redistribution
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        inc = diff / n
        v += inc
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    if abs(diff) > 1e-9:
        rng = np.random.default_rng()
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and v[i] < 1.0:
                inc = min(diff, 1.0 - v[i])
                v[i] += inc
                diff -= inc
            elif diff < 0 and v[i] > 0.0:
                dec = min(-diff, v[i])
                v[i] -= dec
                diff += dec
    return v

def gradient_descent(h_start, target, deadline, rng, max_iters=3000):
    """Projected gradient descent on the worst shift."""
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(h)
    it = 0
    while time.time() < deadline - 0.001 and it < max_iters:
        cur_val = compute_c5(h)
        # correlation, ignore zero shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        mid = n - 1
        corr[mid] = -np.inf
        idx_max = np.argmax(corr)
        k = idx_max - mid

        # subgradient of the worst shift
        g = np.ones(n)
        if k >= 0:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] -= h[: n - kk]
            g[: n - kk] -= h[kk:]

        # diminishing step size
        eta0 = 0.2
        eta = eta0 * (1.0 - it / max_iters)
        h_new = h - eta * g
        h_new = repair_vector(h_new, target)

        val_new = compute_c5(h_new)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        # accept improvement or occasional uphill move
        if val_new < cur_val or rng.random() < 0.01:
            h = h_new
        if best_val < 0.3808:
            break
        it += 1
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # ---- initialise h ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = repair_vector(h0, target_sum)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ---- build SLSQP constraints (used once at the end) ----
    def sum_con(x, target=target_sum):
        return np.sum(x[:n]) - target

    constraints = [{"type": "eq", "fun": sum_con}]

    def make_overlap(k):
        def con(x, k=k):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])
            return x[-1] - overl * (2.0 / n)
        return con

    for kk in range(-n + 1, n):
        if kk == 0:
            continue
        constraints.append({"type": "ineq", "fun": make_overlap(kk)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # ---- gradient descent with random restarts ----
    gd_deadline = start_time + budget_s * 0.7
    while time.time() < gd_deadline:
        # choose start point
        if rng.random() < 0.3:
            h_start = repair_vector(rng.random(n), target_sum)
        else:
            h_start = best_h.copy()
        slice_deadline = min(gd_deadline, time.time() + 2.0)
        h_candidate, val_candidate = gradient_descent(
            h_start, target_sum, slice_deadline, rng, max_iters=3000
        )
        if val_candidate < best_val:
            best_val = val_candidate
            best_h = h_candidate.copy()
            if best_val < 0.3808:
                break

    # ---- final SLSQP refinement if time permits ----
    if time.time() < deadline - 0.5:
        x0 = np.concatenate([best_h, [best_val]])
        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 3000, "ftol": 1e-10, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = repair_vector(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt < best_val:
                    best_val = val_opt
                    best_h = h_opt.copy()
        except Exception:
            pass

    # ---- final safety clip and sum enforcement ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_vector(best_h, target_sum)
    final_val = compute_c5(best_h)
    return best_h.astype(float), float(final_val), n
