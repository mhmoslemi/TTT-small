import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Return max overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Adjust vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # distribute uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    # fix remaining diff greedily
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible continuous vector (values in [0,1] and exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def gradient_descent_refine(h, dx, target_sum, time_limit):
    """Local refinement using gradient steps aimed at the worst overlap shifts."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    n = len(cur_h)
    step = 0.05          # initial max entry change
    min_step = 0.001
    decay = 0.995        # factor to shrink step when no improvement

    while time.time() < time_limit and step > min_step:
        conv = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        conv_max = conv.max()
        # consider all shifts whose overlap is within a tiny tolerance of the max
        eps = 1e-12
        active_idx = np.where(conv >= conv_max - eps)[0] - (n - 1)

        d = np.zeros_like(cur_h)
        for k in active_idx:
            if k > 0:
                kk = k
                d[: n - kk] += cur_h[kk:]
                d[kk:] += cur_h[: n - kk]
            elif k < 0:
                kk = -k
                d[kk:] += cur_h[: n - kk]
                d[: n - kk] += cur_h[kk:]
            else:  # k == 0
                d += 2.0 * cur_h

        d_max = np.max(np.abs(d))
        if d_max == 0:
            break

        delta = step * (d / d_max)
        cand = cur_h + delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_to_sum(cand, target_sum)
        cand_val = compute_c5_vec(cand, dx)

        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step / decay, 0.1)   # modest increase on success
        else:
            step = max(step * decay, min_step)

        if cur_val < 0.3808:
            break

    return cur_h, cur_val


def binary_hill_climb(h_bin, cur_val, dx, time_limit, rng):
    """Swap‑based hill climbing on a binary mask."""
    best_bin = h_bin.copy()
    best_val = cur_val
    n = len(best_bin)
    while time.time() < time_limit:
        ones = np.flatnonzero(best_bin)
        zeros = np.flatnonzero(1 - best_bin)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_bin.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best_bin = cand
            best_val = cand_val
    return best_bin.astype(float), best_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- problem size & initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 80
        h0 = generate_random_feasible(n0, n0 / 2.0, rng)

    # we upscale the grid to give the optimizer more freedom
    factor = 3
    n = n0 * factor
    h_initial = np.repeat(h0, factor)
    target_sum = n / 2.0
    dx = 2.0 / n
    # ensure feasibility after up‑sampling
    h_initial = project_to_sum(np.clip(h_initial, 0.0, 1.0), target_sum)

    best_h = h_initial.copy()
    best_val = compute_c5_vec(best_h, dx)

    # ---- stochastic multi‑restart gradient descent ----
    while time.time() - start_time < budget_s * 0.85:
        # time remaining for this trial
        remaining = budget_s - (time.time() - start_time)
        trial_time = min(5.0, remaining * 0.5)
        if trial_time <= 0.5:
            break

        # choose a start point: either a perturbed current best or a fresh random feasible
        if rng.random() < 0.3:
            h_start = generate_random_feasible(n, target_sum, rng)
        else:
            h_start = best_h.copy()
            h_start += (rng.random(n) - 0.5) * 0.1
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = project_to_sum(h_start, target_sum)

        # local refinement
        time_limit = time.time() + trial_time
        h_ref, val_ref = gradient_descent_refine(h_start, dx, target_sum, time_limit)

        if val_ref < best_val - 1e-12:
            best_h, best_val = h_ref, val_ref
            if best_val < 0.3808:
                break

    # ---- binary rounding and hill climbing (if time permits) ----
    time_left = budget_s - (time.time() - start_time)
    if time_left > 8.0:
        # round to a binary vector with exactly target_sum ones
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argpartition(-best_h, int(target_sum))[: int(target_sum)]
        binary_h[top_idx] = 1
        bin_val = compute_c5_vec(binary_h.astype(float), dx)
        # hill climb
        bin_h, bin_val = binary_hill_climb(
            binary_h, bin_val, dx, time.time() + time_left - 0.5, rng
        )
        if bin_val < best_val - 1e-12:
            best_h = bin_h
            best_val = bin_val

    # ---- final safety enforcement ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
