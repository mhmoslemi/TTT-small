import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj_and_grad(h, beta):
    """Smooth approximation of max_k f_k and its gradient."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx                     # length 2*n-1
    max_f = np.max(f)
    # softmax weights
    exps = np.exp(beta * (f - max_f))
    Z = np.sum(exps)
    w = exps / Z                      # sum to 1
    w_rev = w[::-1]

    # convolutions to build gradient
    conv1 = np.convolve(w_rev, 1.0 - h, mode="full")
    conv2 = np.convolve(w, h, mode="full")
    start = n - 1
    end = start + n
    grad = dx * (conv1[start:end] - conv2[start:end])
    smooth_val = max_f + np.log(Z) / beta
    return smooth_val, grad

def project_to_sum_and_bounds(x, target):
    """Project x onto the box [0,1] while enforcing sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(100):
        cur = np.sum(x)
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = np.count_nonzero(free)
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction if needed
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        idxs = np.where(diff > 0, np.where(x < 1.0)[0], np.where(x > 0.0)[0])
        for idx in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - x[idx])
                x[idx] += add
                diff -= add
            else:
                sub = min(-diff, x[idx])
                x[idx] -= sub
                diff += sub
    return x

def smooth_and_fix(h, target):
    """Gentle Gaussian smoothing with sum projection."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    sm = project_to_sum_and_bounds(sm, target)
    return sm

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise problem size and target integral
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = project_to_sum_and_bounds(h0, target)
        init_vectors = [h0]
    else:
        n = 80
        target = n / 2.0
        init_vectors = []

    # add several random starts
    while len(init_vectors) < 9:
        rnd = rng.random(n)
        rnd = project_to_sum_and_bounds(rnd, target)
        init_vectors.append(rnd)

    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # main multi‑start gradient descent on smoothed max
    for h_start in init_vectors:
        if time.time() - start_time > budget_s * 0.95:
            break
        h = h_start.copy()
        cur_val = compute_c5(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        beta = 8.0
        stall = 0
        it = 0
        max_it = 3000
        while True:
            if time.time() - start_time > budget_s * 0.95:
                break
            if it >= max_it:
                break
            smooth_val, grad = smooth_obj_and_grad(h, beta)
            # backtracking line search
            alpha = 1.0
            improved = False
            for _ in range(5):
                cand = h - alpha * grad
                cand = project_to_sum_and_bounds(cand, target)
                cand_val = compute_c5(cand)
                if cand_val + 1e-12 < cur_val:
                    h = cand
                    cur_val = cand_val
                    improved = True
                    if cur_val < best_val:
                        best_val = cur_val
                        best_h = h.copy()
                        if best_val < 0.3808:
                            return best_h, best_val, n
                    break
                else:
                    alpha *= 0.5
            if not improved:
                # occasional random tweak
                if stall < 3:
                    noise = rng.normal(scale=0.05, size=n)
                    cand = h + noise
                    cand = project_to_sum_and_bounds(cand, target)
                    cand_val = compute_c5(cand)
                    if cand_val < cur_val:
                        h = cand
                        cur_val = cand_val
                        improved = True
                        stall = 0
                        if cur_val < best_val:
                            best_val = cur_val
                            best_h = h.copy()
                            if best_val < 0.3808:
                                return best_h, best_val, n
                if not improved:
                    stall += 1
                    if stall >= 10:
                        break
            else:
                stall = 0
                if beta < 30.0:
                    beta = min(beta + 2.0, 30.0)
            it += 1

    # ------------------------------------------------------------------
    # optional gentle smoothing passes
    while time.time() - start_time < budget_s - 5.0:
        cand = smooth_and_fix(best_h, target)
        cand_val = compute_c5(cand)
        if cand_val + 1e-12 < best_val:
            best_val = cand_val
            best_h = cand
        else:
            break

    # ------------------------------------------------------------------
    # final precise refinement with SLSQP (if time permits)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 6.0:
        dx = 2.0 / n

        def sum_eq(x):
            return np.sum(x[:n]) - target

        def make_ineq(k):
            def fun(x):
                h = x[:n]
                if k >= 0:
                    overlap = np.dot(h[: n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overlap = np.dot(h[kk:], 1.0 - h[: n - kk])
                return x[-1] - overlap * dx
            return fun

        constraints = [{"type": "eq", "fun": sum_eq}]
        for kk in range(-n + 1, n):
            if kk == 0:
                continue
            constraints.append({"type": "ineq", "fun": make_ineq(kk)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_to_sum_and_bounds(h_opt, target)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # ------------------------------------------------------------------
    # final sanity projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum_and_bounds(best_h, target)

    return best_h, best_val, n
