import cvxpy as cp

def slp_optimize(h0, target_sum, max_iter, time_end):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(max_iter):
        if time.time() >= time_end:
            break
        # compute constraints linear approximations
        # variable
        h_var = cp.Variable(n)
        t = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        # precompute for each shift
        for k in range(-n+1, n):
            if k == 0:
                continue
            # compute c_k0 and gradient g_k (shifts)
            # compute overlap index linear approx:
            # c_k = sum_i h[i]*h[i+k]
            # gradient g_k[i] = h[i+k] + h[i-k] (0 if out-of-range)
            c0 = 0.0
            grad = np.zeros(n)
            if k >= 0:
                # i from 0..n-1-k for product
                # compute c0 as sum_i h[i]*h[i+k]
                c0 = np.dot(h[:n-k], h[k:])
                # gradient:
                grad[:n-k] += h[k:]  # derivative of product w.r.t. h[i] (i in 0..n-k-1)
                grad[k:] += h[:n-k]   # derivative w.r.t. h[i+k] (i in k..n-1)
            else:
                kk = -k
                c0 = np.dot(h[kk:], h[:n-kk])
                grad[kk:] += h[:n-kk]
                grad[:n-kk] += h[kk:]
            # linear constraint: c0 + grad.T @ (h_var - h) >= t
            constraints.append(c0 + grad @ (h_var - h) >= t)
        obj = cp.Maximize(t)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            break
        h_new = h_var.value
        # clip to numeric bounds
        h_new = np.clip(h_new, 0, 1)
        # enforce sum exactly maybe adjust if small
        cur = np.sum(h_new)
        if abs(cur - target_sum) > 1e-8:

        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        # update for next iteration
        h = h_new
    return best_h, best_val

def project_to_sum(h, target_sum):
   # Ensure within [0,1] and sum = target_sum.
   h = np.clip(h, 0, 1)
   diff = target_sum - np.sum(h)
   if abs(diff) < 1e-12:
        return h
   # find indices where 0 < h < 1
   free = np.where((h > 0) & (h < 1))[0]
   if free.size > 0:
        inc = diff / free.size
        h[free] += inc
        # clip again
        h = np.clip(h, 0, 1)
        diff = target_sum - np.sum(h)
   # residual adjust
   for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and h[i] < 1:
            add = min(diff, 1 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
   return np.clip(h, 0, 1)

import time
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_sum(h, target):
    """Clip to [0,1] and adjust to meet exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    free = np.where((h > 0) & (h < 1))[0]
    if free.size:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
    # residual fix
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0.0, 1.0)

def slp_optimize(h0, target_sum, max_iter, time_end):
    """Successive linear programming: maximize min autocorrelation."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    for _ in range(max_iter):
        if time.time() >= time_end:
            break
        # build LP
        h_var = cp.Variable(n)
        t = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        for k in range(-n + 1, n):
            if k == 0:
                continue
            # compute current autocorrelation and gradient
            if k >= 0:
                c0 = np.dot(h[:n - k], h[k:])
                grad = np.zeros(n)
                grad[:n - k] += h[k:]
                grad[k:] += h[:n - k]
            else:
                kk = -k
                c0 = np.dot(h[kk:], h[:n - kk])
                grad = np.zeros(n)
                grad[kk:] += h[:n - kk]
                grad[:n - kk] += h[kk:]
            # linear lower bound on autocorrelation
            constraints.append(c0 + grad @ (h_var - h) >= t)
        prob = cp.Problem(cp.Maximize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=400)
        except Exception:
            break
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            break
        h_new = np.array(h_var.value).astype(float)
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce exact sum (tiny numerical drift)
        cur = np.sum(h_new)
        if abs(cur - target_sum) > 1e-9:
            h_new += (target_sum - cur) / n
            h_new = np.clip(h_new, 0.0, 1.0)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val

    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- generate diverse starting vectors -----

        h_up = np.interp(x_new, x_old, h0)
        candidates.append(project_to_sum(h_up, target_sum))

    t = np.arange(n)
    candidates.append(project_to_sum(0.5 + 0.5 * np.cos(2 * np.pi * t / n), target_sum))

        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

                if k >= 0:

                    kk = -k

        cur = np.sum(best_h)
        if abs(cur - target_sum) > 1e-9:
            best_h += (target_sum - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)
