    start = time.time()
    n = 256  # could be changed to 256 for speed

    # function definitions as nested? But we need top-level functions only, not closures. So define all functions top-level.

    # Build initial_h:

        # upsample to n
        # using linear interpolation
        x_old = np.linspace(0, 1, init_full.shape[0], endpoint=False)
        x_new = np.linspace(0, 1, n, endpoint=False)
        init_interp = np.interp(x_new, x_old, init_full)
        # binarize: >0.5
        init_bin = (init_interp > 0.5).astype(np.int8)
        # adjust to exact n/2 ones
        ones_needed = n//2

        elif cur_sum < ones_needed:

import time
import numpy as np

# ---------- Helper functions ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    hf = h.astype(float, copy=False)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h, dtype=float)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=float)
            term2 = np.zeros(n, dtype=float)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=float)
            term2 = np.zeros(n, dtype=float)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.astype(float, copy=True)
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


# ---------- Main optimization routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    n = 256                     # grid size (even, sum = n/2)
    target = 0.3808

    # ---- Initial candidate ----
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        h_bin = (upsample(init_full, n) > 0.5).astype(np.int8)
        need = n // 2
        cur = h_bin.sum()
        if cur > need:
            excess = cur - need
            idx = rng.choice(np.where(h_bin == 1)[0], size=excess, replace=False)
            h_bin[idx] = 0
        elif cur < need:
            shortage = need - cur
            idx = rng.choice(np.where(h_bin == 0)[0], size=shortage, replace=False)
            h_bin[idx] = 1
        h = h_bin.copy()
    else:
        h = random_binary_balanced(n, rng)

    # ---- Record best ----
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score

    # ---- Sets for fast sampling ----
    ones_set = set(np.where(h == 1)[0])
    zeros_set = set(np.where(h == 0)[0])

    # ---- Simulated annealing on binary vectors ----
    max_sa_iters = 200000
    T0 = 0.01  # initial temperature
    sa_iter = 0

    while sa_iter < max_sa_iters and time_left() > 0.6:
        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))

        # propose swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5(h)
        delta = new_score - current_score

        # temperature schedule (linear cooling)
        T = T0 * max(0.0, 1.0 - sa_iter / max_sa_iters)
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            ones_set.remove(i)
            zeros_set.add(i)
            zeros_set.remove(j)
            ones_set.add(j)
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0

        sa_iter += 1
        if best_score <= target:
            break

    # ---- Continuous refinement (momentum gradient descent) ----
    remaining = time_left()
    if remaining > 2.0:
        # heuristic: ~0.001 s per iteration on this size
        n_iter = int(min(20000, max(5000, remaining * 800)))
        h_float = best_h.astype(float, copy=False)
        h_ref, score_ref = gradient_descent_momentum(h_float, n_iter=n_iter,
                                                    step0=0.5, beta=0.9)
        if score_ref < best_score:
            best_score = score_ref
            best_h = h_ref.copy()

    # final projection to guarantee constraints
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
