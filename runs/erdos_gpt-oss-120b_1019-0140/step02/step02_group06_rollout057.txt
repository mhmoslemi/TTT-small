def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # compute convolution via FFT
    # we need h * (1-h) reversed.
    a = np.fft.rfft(h, n=2*n)
    b = np.fft.rfft((1-h)[::-1], n=2*n) # reversed
    conv = np.fft.irfft(a * b, n=2*n)
    # conv length = 2*n
    # relevant part: from 0 to 2*n-2 (full correlation)
    # The full correlation returned by np.correlate(h, 1-h, mode='full') gives indices -n+1 to n-1.
    # Using convolution of h and reversed (1-h) gives index shift where zero shift corresponds to index n-1.
    # So max correlation is max(conv[:2*n-1]) I think.
    corr = conv[:2*n-1]
    return np.max(corr) * dx

    hi = np.max(x)  # shift large positive > max(x) makes all x_i - lambda <=0 => sum 0

    # Actually at lo = min(x) - 1, each x_i - lo >= 1 => clip to 1 => sum = n.

def project_feasible(x, target, max_iter=60):
    # vect x, target float
    # bound for lambda
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    # final projection
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # check sum difference
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        # adjust among non-bound elements
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        count = np.count_nonzero(mask)
        if count > 0:
            adjustment = diff / count
            proj[mask] += adjustment
            proj = np.clip(proj, 0.0, 1.0)
    # final diff may still small
    return proj

import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    H = np.fft.rfft(h, n=2*n)
    B = np.fft.rfft((1.0 - h)[::-1], n=2*n)
    conv = np.fft.irfft(H * B, n=2*n)
    corr = conv[:2*n-1]
    return np.max(corr) * dx

def project_feasible(x, target, max_iter=60):
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        cnt = np.count_nonzero(mask)

            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # convolution via FFT (h * reversed(1-h))
    H = np.fft.rfft(h, n=2 * n)
    B = np.fft.rfft((1.0 - h)[::-1], n=2 * n)
    conv = np.fft.irfft(H * B, n=2 * n)
    corr = conv[: 2 * n - 1]
    return np.max(corr) * dx


def project_feasible(x, target, max_iter=60):
    """
    Project vector x onto the set {0 <= y <= 1, sum(y) = target}.
    Uses bisection on the shift lambda such that y = clip(x - lambda,0,1).
    """
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            y[mask] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
    return y


def simulated_annealing(h0, target, time_budget, rng):
    """Simple SA that respects the sum constraint via projection."""
    n = h0.shape[0]
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = h0.copy()
    cur_c5 = best_c5
    steps_without_improve = 0
    start = time.time()
    temp_start = 0.005
    temp_end = 0.0005

    while time.time() - start < time_budget:
        t = (time.time() - start) / time_budget
        temp = temp_start * (1 - t) + temp_end * t
        sigma = temp
        proposal = cur_h + rng.normal(scale=sigma, size=n)
        proposal = project_feasible(proposal, target)
        prop_c5 = compute_c5(proposal)
        delta = prop_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            cur_h = proposal
            cur_c5 = prop_c5
            steps_without_improve = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            steps_without_improve += 1

        if steps_without_improve >= 2000:
            cur_h = best_h + rng.normal(scale=0.1, size=n)
            cur_h = project_feasible(cur_h, target)
            cur_c5 = compute_c5(cur_h)
            steps_without_improve = 0

    return best_h, best_c5

                h0 = rng.random(n)
                h0 = project_feasible(h0, n / 2.0)

            best_h_n, best_c5_n = simulated_annealing(h0, n / 2.0, slice_time, rng)

            if best_c5_n < best_c5_overall:
                best_c5_overall = best_c5_n
                best_h_overall = best_h_n.copy()
                best_n = n

            # early exit if we beat the target bound
            if best_c5_overall <= 0.3808:
                break
        if best_c5_overall <= 0.3808:
            break

    # fallback if nothing was produced
    if best_h_overall is None:
        if "initial_h_values" in globals():
            h = np.array(initial_h_values, dtype=float).copy()
            n = len(h)
        else:
            n = 80
            h = rng.random(n)
        h = project_feasible(h, n / 2.0)
        c5 = compute_c5(h)
        return h, c5, n

    # final safety projection and evaluation
    best_h_overall = np.clip(best_h_overall, 0.0, 1.0)
    best_h_overall = project_feasible(best_h_overall, best_n / 2.0)
    final_c5 = compute_c5(best_h_overall)
    return best_h_overall, final_c5, best_n
