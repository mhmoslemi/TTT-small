def project_to_feasible(x, target):
    # clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # compute diff
    cur = x.sum()
    diff = target - cur
    if abs(diff) <= 1e-12:
        return x
    # We need to distribute diff without violating bounds.
    # The following simple algorithm: while |diff| > tolerance:
    # find indices where adding diff would not exceed bounds.
    # but diff small relative to n. We can distribute evenly then clip.
    # We'll adjust by adding diff/n to all, then clip again, and repeat.
    # That's similar to previous code but simpler; iterate a few times.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # Determine allowed inc for each element
        if diff > 0:
            # capacity to increase
            capacity = 1.0 - x
            total_capacity = capacity.sum()
            if total_capacity == 0:
                break
            inc = diff * (capacity / total_capacity)
            # add inc
            x += inc
        else:
            # capacity to decrease (i.e., x > 0)
            capacity = x
            total_capacity = capacity.sum()
            if total_capacity == 0:
                break
            dec = diff * (capacity / total_capacity)  # diff negative
            x += dec  # subtract as dec negative
        # clip for numeric safety
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
    return x

def project_feasible(x, target):
    # Clip to bounds
    x = np.clip(x, 0.0, 1.0)
    # Compute diff
    diff = target - x.sum()
    if abs(diff) <= 1e-12:
        return x
    # We'll adjust by distributing diff proportionally to capacity
    # Determine capacity: if diff >0, can increase elements which are <1
    # else decrease elements >0
    # We'll iterate until diff small
    for _ in range(10):
        if diff > 0:
            cap = 1.0 - x
            total_cap = cap.sum()
            if total_cap <= 0:
                break
            # allocate proportionally
            inc = diff * (cap / total_cap)
            x += inc
        else:
            cap = x
            total_cap = cap.sum()
            if total_cap <= 0:
                break
            dec = diff * (cap / total_cap)  # diff negative
            x += dec
        # Clip again due to floating errors
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) <= 1e-12:
            break
    # After iterations diff maybe still small, adjust by small epsilon
    if abs(diff) > 1e-9:
        # Force adjust by adding diff/n to all, then clip
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
    # Finally enforce sum exactly using final correction: if still diff, adjust first element.
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust a single element within bounds
        for i in range(len(x)):
            if (diff > 0 and x[i] < 1) or (diff < 0 and x[i] > 0):
                # compute possible adjustment limited by bound
                max_adj = (1 - x[i]) if diff > 0 else -x[i]
                adj = np.sign(diff) * min(abs(diff), abs(max_adj))
                x[i] += adj
                diff -= adj
                if abs(diff) <= 1e-12:
                    break
    return x

def simulated_annealing_continuous(h0, target, time_budget, rng):
    n = len(h0)
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    # Parameters
    T0 = 0.02
    # scale for perturbation start
    scale0 = 0.05
    # minimal scale
    min_scale = 0.001
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget
        # Temperature linear schedule down
        T = T0 * max(0.001, (1 - frac))
        # Scale decays with T
        scale = max(min_scale, scale0 * (1 - frac))
        # generate new candidate
        new_h = perturb_continuous(cur_h, scale, rng, target)
        new_c5 = compute_c5(new_h)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        # optional occasional random restart
        if rng.random() < 0.0005:
            # random new point
            rnd = rng.random(n)
            rnd = project_feasible(rnd, target)
            cur_h = rnd
            cur_c5 = compute_c5(cur_h)

Implement perturb_continuous:

def perturb_continuous(h, scale, rng, target):
    n = len(h)
    i, j = rng.integers(n, size=2)
    while i == j:
        j = rng.integers(n)
    delta = rng.normal(scale=scale)
    # adjust
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    # project to feasible
    h_new = project_feasible(h_new, target)
    return h_new

def simulated_annealing_binary(h0, target, time_budget, rng):

        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        T = T0 * max(0.001, (1 - frac))

        zeros = np.flatnonzero(1 - cur_h)

def grad_descent_smooth(h0, target, time_budget, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 1.0
    max_beta = 200.0
    alpha = 0.2
    momentum = 0.9
    vel = np.zeros_like(h)
    # schedule param
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # compute correlation v
        v_raw = np.correlate(h, 1.0 - h, mode='full')
        c5 = v_raw.max() * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # update beta slowly
        # Each iteration multiplicative factor 1.002 maybe
        beta = min(max_beta, beta * 1.0015)
        # compute softmax weights for v_raw
        vmax = v_raw.max()
        # To avoid overflow, shift
        exp_arg = beta * (v_raw - vmax)
        # Clip exp_arg to avoid overflow?
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w = w / w_sum
        # compute gradient
        g = np.zeros(n)
        offset = n - 1
        # vectorized contributions: We can compute using convolution? Might be heavy.
        # We'll implement loop for each shift where weight > threshold.
        # To improve speed, only consider weights > 1e-12
        # We'll fetch indices where weight > 0 to loop
        idxs = np.where(w > 1e-12)[0]
        for idx in idxs:
            weight = w[idx]
            k = idx - offset
            if k >= 0:
                # tail = 0:n-k, head = k:n
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            # tail part corresponds to h[i] term
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # Zero-mean to preserve sum: but after projection we handle sum.
        # Use momentum
        vel = momentum * vel + (1 - momentum) * g
        # step size decay: alpha linearly decreasing
        # fraction done
        frac = elapsed / time_budget
        alpha_cur = alpha * max(0.1, 1 - frac)
        # update
        h_new = h - alpha_cur * vel
        # project to feasible
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (scaled by dx)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max()) * dx

def project_feasible(x, target, tol=1e-12):
    """Clip to [0,1] and enforce sum(x) == target (iterative water‑filling)."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) <= tol:
        return x
    for _ in range(10):
        if abs(diff) <= tol:
            break
        if diff > 0:
            cap = 1.0 - x
            total = cap.sum()
            if total == 0:
                break
            inc = diff * (cap / total)
            x += inc
        else:
            cap = x
            total = cap.sum()
            if total == 0:
                break
            dec = diff * (cap / total)   # diff is negative
            x += dec
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    if abs(diff) > tol:
        for i in range(len(x)):
            if diff > 0 and x[i] < 1.0:
                adj = min(diff, 1.0 - x[i])
                x[i] += adj
                diff -= adj
            elif diff < 0 and x[i] > 0.0:
                adj = min(-diff, x[i])
                x[i] -= adj
                diff += adj
            if abs(diff) <= tol:
                break
    return x

def perturb_continuous(h, scale, rng, target):
    """Random two‑index move preserving total mass, then project."""
    n = h.shape[0]
    i, j = rng.integers(n, size=2)
    while i == j:
        j = rng.integers(n)
    delta = rng.normal(scale=scale)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return project_feasible(h_new, target)

def anneal_continuous(h0, target, time_budget, rng):
    """Simulated annealing on the continuous domain."""
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    T0 = 0.10            # initial temperature
    scale0 = 0.10        # initial perturbation scale
    min_scale = 0.001
    no_improve = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget
        T = T0 * max(0.001, 1.0 - frac)   # linear cooling
        scale = max(min_scale, scale0 * (1.0 - frac))
        new_h = perturb_continuous(cur_h, scale, rng, target)
        new_c5 = compute_c5(new_h)
        delta = new_c5 - cur_c5
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # occasional random restart if stuck
        if no_improve > 20000:
            cur_h = project_feasible(rng.random(n), target)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0
        # occasional large shake
        if rng.random() < 0.001:
            cur_h = project_feasible(cur_h + rng.normal(scale=0.1, size=n), target)
            cur_c5 = compute_c5(cur_h)
    return best_h, best_c5

def anneal_binary(bin_h, target_int, time_budget, rng):
    """Binary simulated annealing (swap 1↔0)."""
    n = bin_h.shape[0]
    cur_h = bin_h.copy()
    cur_c5 = compute_c5(cur_h.astype(float))
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    T0 = 0.10
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget
        T = T0 * max(0.001, 1.0 - frac)
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c5 = compute_c5(new_h.astype(float))
        delta = new_c5 - cur_c5
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        # occasional global reshuffle
        if rng.random() < 0.001:
            perm = rng.permutation(n)
            reshuf = np.zeros(n, dtype=int)
            reshuf[perm[:target_int]] = 1
            cur_h = reshuf
            cur_c5 = compute_c5(cur_h.astype(float))
    return best_h.astype(float), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        h_old = np.asarray(initial_h_values, dtype=float).copy()
        n0 = h_old.size
        up_factor = 2
        n = n0 * up_factor
        if n > 1000:

            up_factor = 1
        if up_factor > 1:
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h_old)
        else:
            h0 = h_old.copy()
    else:
        n = 100
        h0 = rng.random(n)

    target = n / 2.0
    h0 = project_feasible(h0, target)

    # ----- continuous simulated annealing ------------------------------------
    elapsed = time.time() - start_total
    remaining = max(0.0, budget_s - elapsed)

    remaining = max(0.0, budget_s - elapsed)
    if n % 2 == 0 and remaining > 0.0:

    # ----- final projection and output ----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
