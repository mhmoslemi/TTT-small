def smooth_max_obj(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    max_approx = (1.0/alpha) * np.log(np.sum(np.exp(alpha*overlaps)))
    return max_approx

def smooth_max_grad(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    exp_vals = np.exp(alpha*overlaps)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    n = len(h)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if wk == 0.0:
            continue
        if k >= 0:
            i1 = np.arange(0, n - k)
            grad[i1] += dx * wk * (1.0 - h[i1 + k])
            i2 = np.arange(k, n)

            grad[i_range] -= dx * wk * h[i_range + kk]
            grad[i_range + kk] += dx * wk * (1.0 - h[i_range])
    return grad

import time
import numpy as np
import scipy.optimize as opt

_dx = None
_alpha = None
_target_sum = None


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1 - h(x+k))dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_and_fix(h, target_sum):
    """Gaussian smoothing (kernel [0.25,0.5,0.25]) with exact sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    cur = np.sum(sm)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            inc = diff / free.size
            sm[free] += inc
            sm = np.clip(sm, 0.0, 1.0)
        diff = target_sum - np.sum(sm)
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
    return np.clip(sm, 0.0, 1.0)


def sum_eq_global(x):
    """Equality constraint: sum(x) == target_sum (global)."""
    return np.sum(x) - _target_sum


def slsqp_obj(x):
    """Smooth approximation of max overlap."""
    h = x
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * _dx
    exp_vals = np.exp(_alpha * overlaps)
    return (1.0 / _alpha) * np.log(np.sum(exp_vals))


def slsqp_grad(x):
    """Gradient of the smooth max objective."""
    h = x
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * _dx
    exp_vals = np.exp(_alpha * overlaps)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp  # weights for each shift
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if wk == 0.0:
            continue
        if k >= 0:
            i1 = np.arange(0, n - k)
            grad[i1] += _dx * wk * (1.0 - h[i1 + k])
            i2 = np.arange(k, n)
            grad[i2] -= _dx * wk * h[i2 - k]
        else:
            kk = -k
            ir = np.arange(0, n - kk)
            grad[ir] -= _dx * wk * h[ir + kk]
            grad[ir + kk] += _dx * wk * (1.0 - h[ir])
    return grad


def _binary_from_h(h, target_sum):
    """Return a binary vector with exactly target_sum ones, preferring larger entries."""
    idx = np.argsort(-h)
    bin_vec = np.zeros_like(h)
    bin_vec[idx[:int(target_sum)]] = 1.0
    return bin_vec


def _adjust_sum(v, target_sum):
    """Project vector onto the hyperplane sum(v)=target_sum while staying in [0,1]."""
    cur = np.sum(v)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return np.clip(v, 0.0, 1.0)
    v = v + diff / len(v)
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    diff = target_sum - cur
    for i in range(len(v)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and v[i] < 1.0:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        elif diff < 0 and v[i] > 0.0:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    return np.clip(v, 0.0, 1.0)
