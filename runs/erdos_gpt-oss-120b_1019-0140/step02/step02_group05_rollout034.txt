def subgradient_optimize(initial_h, max_iters, time_deadline):
    h = initial_h.copy()
    best_local_h = h.copy()

    for it in range(max_iters):

def run(...):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        initial_h = None

    dx = 2.0 / n
    target_sum = n / 2.0

    # function to make feasible vector from arbitrary vector
    def make_feasible(v):
        v = np.clip(v, 0.0, 1.0)
        return project_onto_capped_simplex(v, target_sum)

    # collect starting vectors
    starts = []
    if initial_h is not None:
        starts.append(make_feasible(initial_h))
    # add binary random start
    for _ in range(4):
        # generate binary half ones
        h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, n//2, replace=False)
        h[ones_idx] = 1.0
        starts.append(h)
    # add uniform random start projected
    for _ in range(3):
        h = rng.random(n)
        starts.append(make_feasible(h))
    # add smoothed uniform start
    for _ in range(2):
        h = rng.random(n)
        for _ in range(3):
            h = np.convolve(h, np.array([0.25,0.5,0.25]), mode='same')
        starts.append(make_feasible(h))

    best_h = None
    best_val = np.inf

    # loop over starts
    for init_h in starts:
        # check time
        if time.time() - start_time >= budget_s * 0.93:
            break
        # local optimization
        # compute number of iterations based on remaining time per start
        remaining = budget_s - (time.time() - start_time)
        # allocate maybe min(20000, remaining*200?) Choose some figure.
        max_iters = int(min(20000, remaining * 50))  # assume ~50 iterations per second?
        # subgradient
        local_h, local_val = subgradient_optimize(init_h, max_iters, time.time() + remaining * 0.9, best_val)
        if local_val < best_val:
            best_val = local_val
            best_h = local_h
            if best_val < 0.3808:
                break

    # final smoothing passes (apply smooth_and_fix like earlier)
    if best_h is not None and time.time() - start_time < budget_s - 10:
        for _ in range(5):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
            else:
                break

    # final SLSQP refinement if time remains
    if best_h is not None and time.time() - start_time < budget_s - 20:
        # use constraints method for each shift
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0
        # define objective: the C5 bound variable
        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

        def overlap_k(x, k):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[:n-k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[:n-kk])
            return overl * dx

        constraints = [{'type': 'eq', 'fun': sum_eq}]
        # inequalities: x[-1] >= overlap_k for each k (except k=0)
        # We'll treat x[-1] as variable representing the bound.
        for k in range(-n+1, n):
            if k == 0:
                continue
            def make_ineq(k):
                return lambda x, k=k: x[-1] - overlap_k(x, k)
            constraints.append({'type': 'ineq', 'fun': make_ineq(k)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(fun=lambda x: x[-1],
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                # reproject onto sum
                cur = np.sum(h_opt)
                if abs(cur - target_sum) > 1e-9:
                    h_opt = project_onto_capped_simplex(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt < best_val - 1e-12:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # final safety correction
    if best_h is None:
        # fallback: use uniform feasible
        best_h = make_feasible(np.full(n, 0.5))
        best_val = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce sum exactly
    cur = np.sum(best_h)
    if abs(cur - target_sum) > 1e-9:

def project_onto_capped_simplex(v, target_sum):
    # Projection onto {x: 0 <= x <= 1, sum x = target_sum}
    # Bisection on Lagrange multiplier lambda
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode='same')
    sm = np.clip(sm, 0.0, 1.0)
    cur = sm.sum()
    diff = target_sum - cur

        # distribute diff among interior points
        free = np.where((sm > 0.0) & (sm < 1.0))[0]
        if free.size:
            inc = diff / free.size
            sm[free] += inc

        max_iters = int(min(25000, remaining * 30))
        local_h, local_val = subgradient_optimize(
            init_h, max_iters, time.time() + remaining * 0.85, target_sum
        )
        if local_val < best_val:
            best_val = local_val
            best_h = local_h
            if best_val < 0.3808:
                break

    # optional smoothing refinements
    if best_h is not None and time.time() - start_time < budget_s - 8:
        for _ in range(6):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val, best_h = sm_val, sm
            else:
                break

    # final SLSQP refinement (inequality constraints for each shift)
    if best_h is not None and time.time() - start_time < budget_s - 20:
        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

        def overlap_k(x, k):
            h = x[:n]
            if k >= 0:
                ov = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                ov = np.dot(h[kk:], 1.0 - h[: n - kk])
            return ov * (2.0 / n)

        constraints = [{"type": "eq", "fun": sum_eq}]
        for k in range(-n + 1, n):
            if k == 0:
                continue
            constraints.append(
                {"type": "ineq", "fun": lambda x, kk=k: x[-1] - overlap_k(x, kk)}
            )

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_onto_capped_simplex(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val, best_h = val_opt, h_opt
        except Exception:
            pass

    # fallback (should never happen)
    if best_h is None:
        best_h = make_feasible(np.full(n, 0.5))
        best_val = compute_c5(best_h)

    # final safety clipping and exact sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target_sum) > 1e-9:
        best_h = project_onto_capped_simplex(best_h, target_sum)

    return best_h, best_val, n
