def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # candidate n values: choose list, ensure even.
    candidate_ns = [80, 96, 112, 128, 144, 176, 200]

    # include parent size first if present:
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    # For each n:
    for n in candidate_ns:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
        # allocate time: maybe allocate min(200, remaining_time * 0.6)
        alloc = min(200.0, remaining_time * 0.6)
        h_opt, c5_opt = optimise_for_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global = h_opt
            best_c5_global = c5_opt
            best_n_global = n
            if best_c5_global < 0.3808:
                break

    # after exploring n's, do hill climbing on best solution.
    if best_h_global is not None:
        n = best_n_global
        M = n // 2
        # hill climbing loop until time
        while time.time() - start_time < budget_s - 0.5:
            i, j = rng.choice(n, 2, replace=False)
            # move a random delta <= min(1-h_i, h_j)
            max_transfer = min(1.0 - best_h_global[i], best_h_global[j])
            if max_transfer <= 0:
                # try opposite direction
                max_transfer = min(best_h_global[i], 1.0 - best_h_global[j])
                if max_transfer <= 0:
                    continue
                delta = rng.random() * max_transfer
                h_new = best_h_global.copy()
                h_new[i] -= delta
                h_new[j] += delta
            else:
                delta = rng.random() * max_transfer
                h_new = best_h_global.copy()
                h_new[i] += delta
                h_new[j] -= delta
            # after move, ensure sum still M? It should preserve because we moved mass from j to i.
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < best_c5_global:
                best_h_global = h_new
                best_c5_global = c5_new

    # final safety
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)

        M = n // 2
        cur = best_h_global.sum()
        if abs(cur - M) > 1e-9:
            best_h_global = project_feasible(best_h_global, M)  # ensure sum precisely
        # re-evaluate c5
        best_c5_global = compute_c5(best_h_global)

    return best_h_global, best_c5_global, n

def project_feasible(y, M):
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h = np.clip(y + mid, 0.0, 1.0)
        s = h.sum()
        if s < M:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo+hi)/2.0, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(y, M):
    """Project onto {0<=x_i<=1, sum x_i = M} via a simple water‑filling."""
    lo = -np.max(y) - 1.0   # ensures all zero
    hi = 1.0 - np.min(y) + 1.0   # ensures all one
    for _ in range(50):
        mid = (lo + hi) * 0.5
        h = np.clip(y + mid, 0.0, 1.0)
        s = h.sum()
        if s < M:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def generate_plateau_shape(n, a):
    """Left‑aligned shape: height 1 on [0,a], then linear descent to 0 at 2‑a."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.zeros(n)
    if a >= 1.0 - 1e-12:
        h[x <= 1.0] = 1.0
    else:
        mask1 = x <= a
        mask2 = (x > a) & (x <= 2.0 - a)
        h[mask1] = 1.0
        den = 2.0 - 2.0 * a
        if den > 0:
            h[mask2] = 1.0 - (x[mask2] - a) / den
    return project_feasible(h, n // 2)


def generate_initial_h(n, M, rng):
    """Random feasible start: binary with prob 0.2, else projected uniform."""
    if rng.random() < 0.2:
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
        return h
    y = rng.random(n)
    return project_feasible(y, M)


# ---------- constraints ----------
class SumConstraint:
    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        return np.sum(x[: self.n]) - self.n / 2.0


class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[: self.n - k], 1.0 - h[k :])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overl * self.dx


def optimise_for_n(n, time_budget, rng):
    """Run SLSQP restarts on size n within a budget."""
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{"type": "eq", "fun": sum_con}]
    constraints += [{"type": "ineq", "fun": oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf
    start = time.time()

    # initial candidates from plateau shapes
    a_grid = np.linspace(0.0, 0.99, 21)
    plateau_cands = []
    for a in a_grid:
        h = generate_plateau_shape(n, a)
        c5 = compute_c5(h)
        plateau_cands.append((c5, h))
    plateau_cands.sort(key=lambda p: p[0])
    plateau_start = [h for _, h in plateau_cands[:3]]

    # include parent if available
    parent_h = None
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        ph = np.array(initial_h_values, dtype=float)
        cur = ph.sum()
        if abs(cur - M) > 1e-12:
            ph = project_feasible(ph, M)
        parent_h = ph

    use_parent = parent_h is not None
    while True:
        if time.time() - start > time_budget - 0.1:
            break

        # choose start vector
        if use_parent:
            h0 = parent_h.copy()
            use_parent = False
        elif plateau_start:
            h0 = plateau_start.pop(0)
            # after using, refill later if empty
            if not plateau_start:
                # regenerate a fresh random plateau candidate
                a = rng.random()
                h0 = generate_plateau_shape(n, a)
        else:
            h0 = generate_initial_h(n, M, rng)

        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 800, "ftol": 1e-10, "disp": False},
            )
        except Exception:
            continue

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur = h_opt.sum()
        if abs(cur - M) > 1e-9:
            h_opt = project_feasible(h_opt, M)

        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

        if best_c5 < 0.3808:
            break

    return best_h, best_c5


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    candidate_ns = [80, 96, 112, 128, 144, 176, 200]

    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        alloc = min(200.0, remaining * 0.6)
        h_opt, c5_opt = optimise_for_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global = h_opt
            best_c5_global = c5_opt
            best_n_global = n
            if best_c5_global < 0.3808:
                break

    # local pairwise mass transfer refinements
    if best_h_global is not None:
        n = best_n_global
        while time.time() - start_time < budget_s - 0.5:
            i, j = rng.choice(n, 2, replace=False)
            # attempt move from j to i
            max_transfer = min(1.0 - best_h_global[i], best_h_global[j])
            if max_transfer <= 0:
                max_transfer = min(best_h_global[i], 1.0 - best_h_global[j])
                if max_transfer <= 0:
                    continue
                delta = rng.random() * max_transfer
                h_new = best_h_global.copy()
                h_new[i] -= delta
                h_new[j] += delta
            else:
                delta = rng.random() * max_transfer
                h_new = best_h_global.copy()
                h_new[i] += delta
                h_new[j] -= delta
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < best_c5_global:
                best_h_global = h_new
                best_c5_global = c5_new

    # final safety checks
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)

        cur = best_h_global.sum()
        if abs(cur - M) > 1e-9:
            best_h_global = project_feasible(best_h_global, M)
        best_c5_global = compute_c5(best_h_global)

    return best_h_global, best_c5_global, n
