def propose_move(h, rng, max_step):

    while j == i:
        j = rng.integers(n)
    # ensure i is donor: h[i] > epsilon, j recipient: h[j] < 1 - epsilon
    # Choose i donor and j recipient accordingly; we can randomly decide orientation.

        # pick differently: attempt few times
        # fallback: skip move
        return None
    delta_max = min(max_step, h[i], 1 - h[j])

    return new_h, delta, i, j

import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Return max overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Adjust vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # Distribute the remaining difference uniformly
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    # Fix any leftover by a greedy pass
    for i in range(len(vec)):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0.0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible vector (values in [0,1] and exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    # fix any residual sum error
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def refine_continuous(h, dx, target_sum, time_limit, rng):
    """Local deterministic refinement: transfer small mass between entries."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.05
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recips = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recips.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recips)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            step *= 0.5
            continue
        delta = rng.random() * max_delta
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val


def simulated_annealing(h_start, dx, target_sum, start_time, deadline, budget_s, rng):
    """Continuous SA that preserves sum and box constraints."""
    h = h_start.copy()
    cur_val = compute_c5_vec(h, dx)
    best_h = h.copy()
    best_val = cur_val

    n = len(h)
    # annealing schedule parameters
    T_init, T_min = 0.02, 1e-7
    step_init, step_min = 0.2, 1e-5
    stagnation = 0
    max_stagnation = 5000

    while time.time() < deadline:
        # progress fraction (1 -> 0)
        remain_frac = max(0.0, deadline - time.time()) / (deadline - start_time)

        T = T_min + (T_init - T_min) * remain_frac
        max_step = step_min + (step_init - step_min) * remain_frac

        # pick donor and recipient
        i = rng.integers(n)
        if h[i] <= 1e-12:
            continue
        j = rng.integers(n)
        if h[j] >= 1.0 - 1e-12:
            continue

        delta = rng.random() * min(max_step, h[i], 1.0 - h[j])
        if delta <= 0.0:
            continue

        h[i] -= delta
        h[j] += delta
        new_val = compute_c5_vec(h, dx)
        dE = new_val - cur_val

        if dE < -1e-12:
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_h = h.copy()
                best_val = cur_val
                stagnation = 0
            else:
                stagnation += 1
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                cur_val = new_val
                stagnation = 0
            else:
                # revert move
                h[i] += delta
                h[j] -= delta
                stagnation += 1

        if stagnation > max_stagnation:
            # random restart preserving feasibility
            h = generate_random_feasible(n, target_sum, rng)
            cur_val = compute_c5_vec(h, dx)
            stagnation = 0

        # early exit if we already beat the target bound
        if best_val <= 0.3808 + 1e-6:
            break

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin
    rng = np.random.default_rng(seed)

    # problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        target_sum = n / 2.0
        h0 = project_to_sum(h0, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = generate_random_feasible(n, target_sum, rng)

    dx = 2.0 / n

    # initial evaluation
    cur_val = compute_c5_vec(h0, dx)
    best_h = h0.copy()
    best_val = cur_val

    # main SA phase (takes ~80% of the budget)
    sa_deadline = start_time + budget_s * 0.80
    if sa_deadline > deadline:
        sa_deadline = deadline
    h_sa, val_sa = simulated_annealing(
        best_h, dx, target_sum, start_time, sa_deadline, budget_s, rng
    )
    if val_sa < best_val:
        best_h, best_val = h_sa, val_sa

    # final deterministic refinement (remaining time)
    refine_deadline = deadline
    if refine_deadline > time.time():
        h_ref, val_ref = refine_continuous(
            best_h, dx, target_sum, refine_deadline, rng
        )
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
