def project_to_sum_one(h, target):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute diff uniformly
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # if still diff, adjust greedily
        cur = h.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # adjust each element sequentially
            for i in range(len(h)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    inc = min(diff, 1.0 - h[i])
                    h[i] += inc
                    diff -= inc
                elif diff < 0 and h[i] > 0.0:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
    return h

def gradient_refine(h0, time_budget, rng):
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.2
    while time.time() - start < time_budget:
        # correlation
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        # update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # increase beta
        if beta < max_beta:
            beta = min(max_beta, beta * 1.002)  # slow increase
        # softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # zero-mean
        g -= np.mean(g)
        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        # project to [0,1] and sum
        h_new = project_sum_and_bounds(h_new, target)
        # Evaluate new c5
        c5_new = compute_c5(h_new)
        # accept if improved or small random acceptance
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        # else keep h unchanged
    return best_h, best_c5

    return best_h, final_c5, n
