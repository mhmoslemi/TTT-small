def simulated_annealing(h_start, time_budget, rng):
    import time, numpy as np
    n = len(h_start)
    dx = 2.0 / n
    best_h = h_start.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    # wish to have many restarts: We can run multiple independent runs within budget.
    # We'll maintain a loop that restarts after certain number of iterations without improvement.
    # Simpler: keep a single run with random jumps.

    # temperature schedule: linear from T0 to 0
    T0 = 0.02  # initial temperature; may adjust.

    # Compute pairwise indexes pre? Not needed.

    # Variation: small delta moves for general values; also binary swap moves.
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # compute temperature
        T = T0 * (1 - elapsed / time_budget)
        if T <= 0:
            T = 1e-12

        # With probability p_big do binary swap move else do small delta transfer.
        # p_big = 0.2
        if rng.random() < 0.2:
            # binary swap: pick i with h_i > 1e-8 (donor) and j with h_j < 1-1e-8 (receiver)
            # ensure we don't pick same.
            # If many values are integer, we may effectively swap.
            # generate candidate indices
            donors = np.where(cur_h > 1e-8)[0]
            receivers = np.where(cur_h < 1 - 1e-8)[0]
            if donors.size == 0 or receivers.size == 0:
                continue
            i = rng.choice(donors)
            j = rng.choice(receivers)
            # delta = max possible transfer: min(cur_h[i], 1 - cur_h[j])
            max_delta = min(cur_h[i], 1 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
            # apply
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # small transfer: pick i,j distinct random
            i, j = rng.choice(n, 2, replace=False)
            # attempt to move delta from j to i (increase i, decrease j)
            max_delta = min(1 - cur_h[i], cur_h[j])
            if max_delta <= 0:
                # try opposite direction
                max_delta = min(cur_h[i], 1 - cur_h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random()
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            else:
                delta = max_delta * rng.random()
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta

        # compute new value

def simulate_annealing_one_run(h_init, time_limit, rng):
    import time, numpy as np
    n = len(h_init)
    dx = 2.0 / n
    best_h = h_init.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    T0 = 0.02

    # Precompute lists of feasible donors and receivers each iteration.
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        remaining = time_limit - elapsed
        # compute temperature linear schedule
        T = T0 * (1 - elapsed / time_limit)
        if T <= 0:
            T = 1e-12
        # Select move
        # 30% chance of swap with max delta
        if rng.random() < 0.3:
            # start with donors indices >0 and receivers <1
            donors = np.where(cur_h > 1e-12)[0]
            receivers = np.where(cur_h < 1 - 1e-12)[0]
            if donors.size == 0 or receivers.size == 0:
                continue
            i = rng.choice(donors)
            j = rng.choice(receivers)
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # small random transfer between any two distinct indices
            i, j = rng.choice(n, 2, replace=False)
            # Choose direction that allows transfer
            max_delta = min(1.0 - cur_h[i], cur_h[j])
            if max_delta > 0:
                delta = max_delta * rng.random()
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
            else:
                max_delta = min(cur_h[i], 1.0 - cur_h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random()
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta

        # Evaluate new_h
        new_val = compute_c5(new_h)

        # Acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            cur_h = new_h
            cur_val = new_val
            # Update global best
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = new_h.copy()

        # Reproject sum if drift
        if abs(cur_h.sum() - n/2) > 1e-8:
            cur_h += ((n/2) - cur_h.sum()) / n
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_val = compute_c5(cur_h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()

    return best_h, best_val

        M = n // 2
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
            cur = h0.sum()
            if abs(cur - M) > 1e-12:
                h0 += (M - cur) / n
                h0 = np.clip(h0, 0.0, 1.0)
        else:
            # start with linear shape; later restarts may be random
            h0 = np.linspace(1.0, 0.0, n)

        best_h_n = None
        best_c5_n = np.inf
        n_start = time.time()
        while True:
            elapsed_n = time.time() - n_start
            if elapsed_n >= alloc_n:
                break
            remaining_n = alloc_n - elapsed_n
            inner_time = min(5.0, remaining_n)

            if best_h_n is None:
                if rng.random() < 0.5:
                    start_h = np.linspace(1.0, 0.0, n)
                else:
                    start_h = np.zeros(n)
                    idx = rng.choice(n, M, replace=False)
                    start_h[idx] = 1.0
            else:
                start_h = best_h_n.copy()
                for _ in range(3):
                    i, j = rng.choice(n, 2, replace=False)
                    max_delta = min(start_h[i], 1 - start_h[j])
                    if max_delta > 0:
                        delta = max_delta * rng.random()
                        start_h[i] -= delta
                        start_h[j] += delta
                cur = start_h.sum()
                if abs(cur - M) > 1e-9:
                    start_h += (M - cur) / n
                    start_h = np.clip(start_h, 0.0, 1.0)

            h_local, c5_local = simulated_annealing_one_run(start_h, inner_time, rng)
            if c5_local < best_c5_n:
                best_c5_n = c5_local
                best_h_n = h_local.copy()
            if best_c5_n < 0.3808:
                break

        if best_h_n is not None and best_c5_n < best_c5_global:
            best_c5_global = best_c5_n
            best_h_global = best_h_n.copy()
            best_n_global = n
        if best_c5_global < 0.3808:
            break

    if best_h_global is not None:
        n = best_n_global
        M = n // 2
        while time.time() - start < budget_s - 0.5:
            i, j = rng.choice(n, 2, replace=False)
            max_delta = min(1.0 - best_h_global[i], best_h_global[j])
            if max_delta > 0:
                delta = max_delta * rng.random()
                new_h = best_h_global.copy()
                new_h[i] += delta
                new_h[j] -= delta
            else:
                max_delta = min(best_h_global[i], 1.0 - best_h_global[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random()
                new_h = best_h_global.copy()
                new_h[i] -= delta
                new_h[j] += delta
            new_val = compute_c5(new_h)
            if new_val + 1e-12 < best_c5_global:
                best_h_global = new_h
                best_c5_global = new_val
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        cur = best_h_global.sum()
        if abs(cur - M) > 1e-9:
            best_h_global += (M - cur) / n
            best_h_global = np.clip(best_h_global, 0.0, 1.0)
    else:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n

    return best_h_global, best_c5_global, best_n_global
