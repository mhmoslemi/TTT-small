def simulated_annealing(h_init, time_limit, start_temp, end_temp, max_delta=0.1):
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = len(h_init)
    rng = np.random.default_rng()
    # compute temperature schedule based on elapsed time
    start_time = time.time()
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        # temperature linearly decreasing
        t = start_temp + (end_temp - start_temp) * (elapsed / time_limit)
        # generate neighbor
        # pick i where cur_h[i] > 0 and j where cur_h[j] < 1
        # keep i != j
        i = rng.integers(0, n)
        # ensure we have room to adjust
        if cur_h[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if cur_h[j] >= 1 - 1e-12:
            continue
        # allowable delta
        max_delta_ij = min(max_delta, cur_h[i], 1.0 - cur_h[j])
        # also maybe allow small negative delta (swap direction) by also picking delta negative? but we already assign i as donor.
        delta = rng.random() * max_delta_ij
        # apply move
        cur_h[i] -= delta
        cur_h[j] += delta
        # compute score
        new_score = compute_c5(cur_h)
        # acceptance
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(t, 1e-12)):
            cur_score = new_score
            # accept
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert move
            cur_h[i] += delta
            cur_h[j] -= delta
        # optionally adjust max_delta based on temperature? we keep constant.
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h_start, time_limit, start_temp, end_temp, max_delta=0.1):
    rng = np.random.default_rng()
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = h_start.shape[0]
    start_time = time.time()
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        # linear temperature schedule
        t = start_temp + (end_temp - start_temp) * (elapsed / time_limit)
        # propose a move: transfer delta from i to j
        i = rng.integers(0, n)
        if cur_h[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if cur_h[j] >= 1.0 - 1e-12:
            continue
        max_delta_ij = min(max_delta, cur_h[i], 1.0 - cur_h[j])
        if max_delta_ij <= 0.0:
            continue
        delta = rng.random() * max_delta_ij
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = compute_c5(cur_h)
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(t, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    n_target = 256
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = upsample(init_full, n_target)
        cand = project_onto_box_sum(cand, n_target / 2.0)
        candidates.append(cand)

    # Thue‑Morse based start
    tm_small = thue_morse(64)
    cand = upsample(tm_small, n_target)
    cand = project_onto_box_sum(cand, n_target / 2.0)
    candidates.append(cand)

    # Random binary starts with small noise
    for _ in range(3):
        h_bin = random_binary_balanced(n_target, rng).astype(float)
        h_noisy = h_bin + rng.normal(scale=0.08, size=n_target)
        h_noisy = np.clip(h_noisy, 0.0, 1.0)
        h_noisy = project_onto_box_sum(h_noisy, n_target / 2.0)
        candidates.append(h_noisy)

    best_h = None
    best_score = float('inf')

    # Quick gradient refinements for each candidate
    for cand in candidates:
        if time_left() < 8.0:
            break
        h_opt, sc_opt = gradient_descent_momentum(cand, n_iter=2000, step0=0.5, beta=0.9)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_target, 0.5, dtype=float), n_target / 2.0)
        best_score = compute_c5(best_h)

    # Longer gradient descent using half of the remaining budget
    remaining = time_left()
    if remaining > 5.0:
        gd_time = remaining * 0.5
        # estimate iteration count assuming ~0.001 s per iteration
        iters = max(1000, int(gd_time / 0.001))
        h_ref, sc_ref = gradient_descent_momentum(best_h, n_iter=iters, step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # Simulated annealing on whatever time is left
    remaining = time_left()
    if remaining > 5.0:
        start_temp = max(0.001, best_score * 0.2)
        end_temp = start_temp * 0.01
        h_sa, sc_sa = simulated_annealing(best_h, time_limit=remaining, start_temp=start_temp, end_temp=end_temp, max_delta=0.15)
        if sc_sa < best_score:
            best_score, best_h = sc_sa, h_sa.copy()

    best_h = project_onto_box_sum(best_h, n_target / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n_target

        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h_start, time_limit, start_temp, end_temp,
                        max_delta=0.1, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = h_start.shape[0]
    start_time = time.time()
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        t = start_temp + (end_temp - start_temp) * (elapsed / time_limit)
        i = rng.integers(0, n)
        if cur_h[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if cur_h[j] >= 1.0 - 1e-12:
            continue
        max_d = min(max_delta, cur_h[i], 1.0 - cur_h[j])
        if max_d <= 0.0:
            continue
        delta = rng.random() * max_d
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = compute_c5(cur_h)
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(t, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    n_target = 256
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = upsample(init_full, n_target)
        cand = project_onto_box_sum(cand, n_target / 2.0)
        candidates.append(cand)

    # Thue–Morse seed
    tm_small = thue_morse(64)
    cand = upsample(tm_small, n_target)
    cand = project_onto_box_sum(cand, n_target / 2.0)
    candidates.append(cand)

    # Random binary + noise seeds
    for _ in range(3):
        h_bin = random_binary_balanced(n_target, rng).astype(float)
        h_noisy = h_bin + rng.normal(scale=0.08, size=n_target)
        h_noisy = np.clip(h_noisy, 0.0, 1.0)
        h_noisy = project_onto_box_sum(h_noisy, n_target / 2.0)
        candidates.append(h_noisy)

    best_h = None
    best_score = float('inf')

    # Brief momentum refinement for each starter
    for cand in candidates:
        if time_left() < 8.0:
            break
        h_opt, sc_opt = gradient_descent_momentum(cand, n_iter=2000,
                                                  step0=0.5, beta=0.9)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_target, 0.5, dtype=float),
                                      n_target / 2.0)
        best_score = compute_c5(best_h)

    # Extended momentum phase (≈ half remaining budget)
    remaining = time_left()
    if remaining > 5.0:
        gd_time = remaining * 0.5
        iters = max(1000, int(gd_time / 0.001))
        h_ref, sc_ref = gradient_descent_momentum(best_h, n_iter=iters,
                                                  step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # Simulated annealing on the rest of the budget
    remaining = time_left()
    if remaining > 5.0:
        start_temp = max(0.001, best_score * 0.2)
        end_temp = start_temp * 0.01
        h_sa, sc_sa = simulated_annealing(best_h, time_limit=remaining,
                                          start_temp=start_temp,
                                          end_temp=end_temp,
                                          max_delta=0.15, rng=rng)
        if sc_sa < best_score:
            best_score, best_h = sc_sa, h_sa.copy()

    # Final polishing gradient step if any time left
    remaining = time_left()
    if remaining > 2.0:
        iters = max(500, int(remaining / 0.001))
        h_fin, sc_fin = gradient_descent_momentum(best_h, n_iter=iters,
                                                  step0=0.3, beta=0.9)
        if sc_fin < best_score:
            best_score, best_h = sc_fin, h_fin.copy()

    best_h = project_onto_box_sum(best_h, n_target / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_target
