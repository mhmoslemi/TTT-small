def run(seed=42, budget_s=1100):
    start = time.time()
    rng = np.random.default_rng(seed)
    best_global_c5 = np.inf
    best_global_h = None
    best_global_n = None
    candidate_ns = [80, 96, 112, 128, 144]
    # optionally include initial size if initial_h_values available
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                # ensure it's first
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]
    # For each n data
    for n in candidate_ns:
        remaining = budget_s - (time.time() - start)
        if remaining <= 5:
            break
        # allocate time for this n: maybe max 0.5 * remaining, cap at 250 sec
        allowed = min(250.0, remaining * 0.5)
        # run optimization: returns best h and c5 for this n
        h_best, c5_best = optimize_continuous_h(n, rng, allowed)

def simulated_annealing(h0, time_limit, rng):
    n = len(h0)
    M = n/2
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    # schedule parameters
    T0 = 0.02  # start temperature
    Tmin = 1e-5
    # compute total steps approx = int((time_limit)/(time per iteration) )
    # but we don't know; just loop while time remains
    step = 0
    while time.time() - start < time_limit:
        # determine temperature as decreasing exponential
        t_frac = (time.time() - start) / time_limit
        T = T0 * (1 - t_frac) + Tmin * t_frac
        # propose move
        i, j = rng.choice(n, 2, replace=False)
        # choose direction randomly
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_transfer = min(cur_h[donor], 1.0 - cur_h[receiver])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        # apply
        new_h = cur_h.copy()
        new_h[donor] -= delta
        new_h[receiver] += delta
        new_c5 = compute_c5(new_h)
        dE = new_c5 - cur_c5
        if dE <= 0 or rng.random() < np.exp(-dE / T):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        step += 1
    return best_h, best_c5

    if cur < M:
        deficit = M - cur
        # fill deficits to ones with smallest values
        for i in np.argsort(h):
            inc = min(deficit, 1.0 - h[i])
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:

        for i in np.argsort(-h):

            excess -= dec

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def generate_initial_h(n, M, rng):
    """Random feasible h (0<=h<=1, sum=M)."""
    h = rng.random(n)
    h = h / np.sum(h) * M
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur < M:
        deficit = M - cur
        for i in np.argsort(h):
            inc = min(deficit, 1.0 - h[i])
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    elif cur > M:
        excess = cur - M
        for i in np.argsort(-h):
            dec = min(excess, h[i])
            h[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    return h

def compute_grad_for_shift(h, k):
    """Gradient of the overlap for a particular shift k."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            grad[i] += dx * (1.0 - h[i + k])
            grad[i + k] -= dx * h[i]
    else:
        kk = -k
        for i in range(kk, n):
            grad[i] += dx * (1.0 - h[i - kk])
            grad[i - kk] -= dx * h[i]
    return grad

def project_to_feasible(y, M):
    """
    Projection of y onto the set {z: 0<=z_i<=1, sum(z)=M}
    using a bisection on the threshold tau where z_i = clip(y_i - tau,0,1).
    """
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        z = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(z)
        if s > M:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = M - np.sum(z)
    if abs(diff) > 1e-12:
        mask = (z > 1e-12) & (z < 1.0 - 1e-12)
        if np.any(mask):
            adj = diff / np.sum(mask)
            z[mask] += adj
            z = np.clip(z, 0.0, 1.0)
    return z

def gradient_descent_refine(h0, max_iter, time_limit, rng):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = len(h0)
    M = n / 2.0
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    for _ in range(max_iter):
        if time.time() - start > time_limit:
            break
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        center = n - 1
        corr[center] = -np.inf

        grad = compute_grad_for_shift(h, k)
