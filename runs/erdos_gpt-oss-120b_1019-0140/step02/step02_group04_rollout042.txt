import time, math, numpy as np

def compute_cost(h):
    return np.max(np.correlate(h, 1 - h, mode='full'))

def greedy_init(n, target_ones):
    h = np.zeros(n, dtype=int)
    for _ in range(target_ones):
        zeros = np.where(h == 0)[0]
        best_i = None
        best_c = None
        for i in zeros:
            h[i] = 1
            c = compute_cost(h)
            h[i] = 0
            if best_c is None or c < best_c:
                best_c = c
                best_i = i
        h[best_i] = 1
    return h

def local_optimize(h):
    cur_cost = compute_cost(h)
    n = len(h)
    improved = True
    while improved:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_cost = compute_cost(h)
                if new_cost < cur_cost - 1e-12:
                    cur_cost = new_cost
                    improved = True
                    break
                else:
                    h[i] = 1
                    h[j] = 0
            if improved:
                break
    return h, cur_cost

import time, math, numpy as np

def _cost(h: np.ndarray) -> float:
    """Maximum overlap count for binary vector h (without dx scaling)."""
    return np.max(np.correlate(h, 1 - h, mode="full"))

def _greedy_init(n: int, target_ones: int) -> np.ndarray:
    """Build a binary vector by adding ones one at a time with minimal current max overlap."""
    h = np.zeros(n, dtype=int)
    for _ in range(target_ones):
        zeros = np.where(h == 0)[0]
        best_i = zeros[0]
        best_c = None
        for i in zeros:
            h[i] = 1
            c = _cost(h)
            h[i] = 0
            if best_c is None or c < best_c:
                best_c = c
                best_i = i
        h[best_i] = 1
    return h

def _local_optimize(h: np.ndarray):
    """Deterministic hill‑climb: swap only if it strictly reduces the max overlap."""
    cur = _cost(h)
    n = len(h)
    improved = True
    while improved:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new = _cost(h)
                if new < cur - 1e-12:
                    cur = new
                    improved = True
                    break
                h[i] = 1
                h[j] = 0
            if improved:
                break
    return h, cur

def _solve_n(n: int, rng: np.random.Generator, time_limit: float):
    """Search for a binary vector of length n with weight n/2 lowering max overlap."""
    target_ones = n // 2
    dx = 2.0 / n
    threshold = math.floor(0.3808 * n / 2)          # integer O_k bound needed
    best_h = None
    best_cost = float("inf")
    start = time.time()

    # possible binary start from the provided continuous construction
    init_from_parent = None
    if "initial_h_values" in globals():
        ih = np.array(initial_h_values, dtype=float)
        if ih.size == n:
            idx = np.argsort(-ih)[:target_ones]
            bh = np.zeros(n, dtype=int)
            bh[idx] = 1
            init_from_parent = bh

    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        if remaining < 2.0:
            break

        # initial vector for this run
        if best_h is not None:
            h = best_h.copy()
            for _ in range(5):                     # small perturbation
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i], h[j] = 0, 1
        elif init_from_parent is not None:
            h = init_from_parent.copy()
            init_from_parent = None                 # use only once
        else:
            h = np.zeros(n, dtype=int)
            perm = rng.permutation(n)
            h[perm[:target_ones]] = 1

        cur_cost = _cost(h)
        if cur_cost < best_cost:
            best_cost = cur_cost
            best_h = h.copy()
            if best_cost <= threshold:
                return best_h.astype(float), best_cost * dx, n

        # ----- simulated annealing on this start -----
        sa_time = min(remaining * 0.9, 12.0)          # cap each SA block
        sa_start = time.time()
        T0 = cur_cost * 2.0 + 1.0
        Tf = 1e-4
        while time.time() - sa_start < sa_time:
            prog = (time.time() - sa_start) / sa_time
            T = T0 * (1 - prog) + Tf * prog

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

            h[i] = 0
            h[j] = 1
            new_cost = _cost(h)
            delta = new_cost - cur_cost
            accept = delta <= 0 or rng.random() < math.exp(-delta / T)
            if accept:
                cur_cost = new_cost
                if new_cost < best_cost:
                    best_cost = new_cost
                    best_h = h.copy()
                    if best_cost <= threshold:
                        return best_h.astype(float), best_cost * dx, n
            else:
                h[i] = 1
                h[j] = 0

        # optional local improvement of the global best before the next restart
        if best_h is not None and time.time() - start < time_limit - 1.0:
            hb, cb = _local_optimize(best_h.copy())
            if cb < best_cost:
                best_cost = cb
                best_h = hb.copy()
                if best_cost <= threshold:
                    return best_h.astype(float), best_cost * dx, n

    # fallback: if nothing found, use greedy construction
    if best_h is None:
        best_h = _greedy_init(n, target_ones)
        best_cost = _cost(best_h)

    # final polish
    if time.time() - start < time_limit - 0.5:
        hb, cb = _local_optimize(best_h.copy())
        if cb < best_cost:
            best_h, best_cost = hb, cb

    return best_h.astype(float), best_cost * dx, n

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    start_total = time.time()
    rng = np.random.default_rng(seed)

    # try a smaller discretisation first (n=64)
    time_n64 = budget_s * 0.45
    h64, c5_64, n64 = _solve_n(64, rng, time_n64)

    # if already good enough, return immediately
    if c5_64 <= 0.3808:
        return h64, float(c5_64), int(n64)

    # use the remaining budget for n=80 (the size of the parent solution)
    remaining = budget_s - (time.time() - start_total)
    if remaining > 5.0:
        h80, c5_80, n80 = _solve_n(80, rng, remaining)
        # pick the better bound
        if c5_80 < c5_64:
            best_h, best_c5, best_n = h80, c5_80, n80
        else:
            best_h, best_c5, best_n = h64, c5_64, n64
    else:
        best_h, best_c5, best_n = h64, c5_64, n64

    # enforce exact sum constraint (possible tiny floating errors)
    target = best_n / 2
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h = (best_h > 0.5).astype(float)
        cur = best_h.sum()
        diff = int(round(target - cur))
        if diff > 0:
            zero_idx = np.where(best_h < 0.5)[0]
            best_h[zero_idx[:diff]] = 1.0
        elif diff < 0:
            one_idx = np.where(best_h > 0.5)[0]
            best_h[one_idx[: -diff]] = 0.0

    # final exact evaluation
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * (2.0 / best_n)
    return best_h, float(final_c5), int(best_n)
