def simulated_annealing_binary(n, rng, time_budget, start_h=None):
    start = time.time()
    h = start_h if start_h is not None else random_binary_balanced(n, rng)
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T0 = 0.01
    T_end = 1e-6
    total = time_budget
    iter = 0
    while True:
        elapsed = time.time() - start
        remaining = time_budget - elapsed
        if remaining <= 0.1:
            break
        # temperature schedule
        T = T0 * (T_end/T0) ** (elapsed/total)
        # propose swap
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            # if same parity, continue
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        iter += 1
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

# ----- basic utilities -------------------------------------------------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation of arr (length m) to length new_n on [0,2)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def binary_from_continuous(v, n):
    """Return binary vector of length n with exactly n/2 ones, keeping the largest entries."""
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


# ----- gradient descent with momentum -----------------------------------
def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


# ----- simulated annealing on binary vectors ---------------------------
def simulated_annealing_binary(n, rng, max_time, start_h=None):
    """Simple simulated annealing that swaps a 1 with a 0 while preserving the sum."""
    start = time.time()
    if start_h is None:
        h = random_binary_balanced(n, rng)
    else:
        h = start_h.astype(float).copy()
        # assure exact balance
        h = project_onto_box_sum(h, n / 2.0)
        h = np.rint(h)  # force to 0/1 after projection
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    T0 = 0.02
    T_end = 1e-6
    total = max_time

    while True:
        elapsed = time.time() - start
        if elapsed >= total:
            break
        # temperature schedule
        prog = elapsed / total
        T = T0 * (T_end / T0) ** prog
        # propose a swap of opposite values
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_score


# ----- SLSQP refinement (continuous) ------------------------------------
def slsqp_optimize_continuous(h0, max_iter=200):
    """SLSQP minimizing the epigraph formulation of C5."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr  # should be >=0 for all shifts

    cons = [
        {'type': 'eq', 'fun': eq_sum},
        {'type': 'ineq', 'fun': ineq_fun}
    ]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
