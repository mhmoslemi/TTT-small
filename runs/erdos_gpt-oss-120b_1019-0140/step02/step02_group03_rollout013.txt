        h0 = project_to_box_sum(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Parameters
    # start temperature and step size
    T0 = 5e-4  # acceptable
    step0 = 0.5
    # schedule
    iter_count = 0
    # maybe set max_iter based on time

    while time.time() - start < budget_s - 5.0:
        # compute progress fraction
        elapsed = time.time() - start
        fraction = elapsed / (budget_s - 5.0)
        # linearly decay temperature and step
        T = T0 * (1 - fraction) + 1e-6
        step = step0 * (1 - fraction)**2 + 1e-3   # ensures minimal step

        # propose candidate
        d = rng.normal(size=n)
        d -= d.mean()
        candidate = cur_h + step * d
        candidate = project_to_box_sum(candidate, target)

        cand_c5 = compute_c5(candidate)

        delta = cand_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_h = candidate
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = cand_c5

        # occasional reheat: with small probability, increase temperature step
        if rng.random() < 0.001:
            # add random perturbation to current approaching best drop
            cur_h = best_h + step * rng.normal(size=n) * 3.0
            cur_h = project_to_box_sum(cur_h, target)
            cur_c5 = compute_c5(cur_h)

        iter_count += 1

    # optional final local refinement via SLSQP
    # define function objective that returns max overlap
    def objective(x):
        h = x
        # compute c5
        return compute_c5(h)

    # constraints: sum == target
    cons = {'type': 'eq', 'fun': lambda x: x.sum() - target}
    bounds = [(0.0, 1.0)] * n

    # Use best_h as initial guess
    try:

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise feasible vector -----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = project_to_box_sum(h0, target)
    else:
        n = 80
        target = n / 2.0
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ----- simulated annealing -----
    T0 = 1e-3          # initial temperature
    T_end = 1e-8       # final temperature
    step0 = 0.5        # initial step size
    step_min = 1e-4    # final step size
    no_imp = 0         # counter for iterations without improvement

    while time.time() - start < budget_s - 2.0:
        elapsed = time.time() - start
        frac = elapsed / (budget_s - 2.0)  # 0..1

        # linear cooling and step decay
        T = T0 * (1 - frac) + T_end * frac
        step = step0 * (1 - frac) + step_min * frac

        # propose a zero‑mean Gaussian perturbation
        d = rng.normal(size=n)
        d -= d.mean()
        cand = cur_h + step * d
        cand = project_to_box_sum(cand, target)

        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = cand
            cur_c5 = cand_c5
            no_imp = 0
            if cand_c5 < best_c5 - 1e-12:
                best_h = cand.copy()
                best_c5 = cand_c5
        else:
            no_imp += 1

        # occasional re‑heat from the best solution
        if rng.random() < 0.001 or no_imp > 20000:
            cur_h = best_h + 3.0 * step * rng.normal(size=n)
            cur_h = project_to_box_sum(cur_h, target)
            cur_c5 = compute_c5(cur_h)
            no_imp = 0

    # ----- local refinement with SLSQP -----
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: x.sum() - target}
    try:
        res = minimize(
            fun=lambda x: compute_c5(x),
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[cons],
            options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            refined = np.clip(res.x, 0.0, 1.0)
            refined = project_to_box_sum(refined, target)
            refined_c5 = compute_c5(refined)
            if refined_c5 < best_c5:
                best_h = refined
                best_c5 = refined_c5
    except Exception:
        pass

    # ----- final safety projection -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
