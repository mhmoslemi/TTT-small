def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
```

def generate_initial_h(n, M, rng):
    # method random binary or uniform or linear
    method = rng.integers(3)
    if method == 0:
        # binary random selection of M ones
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:
        # uniform random then adjust sum to M
        h = rng.random(n)
        # scale to sum M
        h = h / h.sum() * M
        # clip to [0,1]; adjust iteratively
        # Ensure no overflow beyond 1
        clip = np.clip(h, 0, 1)
        deficit = M - clip.sum()
        # if deficit > 0, add to smallest entries
        if deficit > 1e-12:
            # sorted indices by current value
            order = np.argsort(clip)
            for i in order:
                inc = min(deficit, 1 - clip[i])
                clip[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif deficit < -1e-12:
            excess = -deficit
            order = np.argsort(-clip)
            for i in order:
                dec = min(excess, clip[i])
                clip[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
        h = clip
    else:
        # linear decreasing shape clipped to sum M
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0, 1)
        cur = h.sum()
        if cur < M - 1e-12:
            deficit = M - cur
            order = np.argsort(h)
            for i in order:
                inc = min(deficit, 1 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M + 1e-12:
            excess = cur - M
            order = np.argsort(-h)
            for i in order:
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    return h

import time
import math
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discretised h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def generate_initial_h(n, M, rng):
    """Create a feasible h (0<=h<=1, sum==M)."""
    method = rng.integers(3)
    if method == 0:  # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:  # uniform random then adjust
        h = rng.random(n)
        h = h / h.sum() * M
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if cur < M - 1e-12:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M + 1e-12:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    else:  # linear decreasing shape then adjust
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if cur < M - 1e-12:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M + 1e-12:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    return h

def optimise_for_n(n, time_budget, rng):
    """Simulated‑annealing style optimisation for a given resolution."""
    M = n // 2
    # initialise from parent if possible
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        h0 = np.array(initial_h_values, dtype=float)
        cur = h0.sum()
        if abs(cur - M) > 1e-12:
            h0 += (M - cur) / n
            h0 = np.clip(h0, 0.0, 1.0)
        best_h = h0.copy()
        best_c5 = compute_c5(best_h)
    else:
        best_h = generate_initial_h(n, M, rng)
        best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5

    start = time.time()
    end = start + time_budget
    T0 = 1e-3                 # initial temperature
    step_scale0 = 0.2         # initial move magnitude factor

    while time.time() < end - 0.01:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * max(0.0, 1.0 - frac)
        step_scale = max(0.01, step_scale0 * max(0.0, 1.0 - frac))

        move_type = rng.integers(3)   # 0: transfer, 1: swap, 2: random restart
        if move_type == 0:
            src, dst = rng.choice(n, 2, replace=False)
            max_delta = min(current_h[src], 1.0 - current_h[dst])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta * step_scale
            if delta <= 0.0:
                continue
            new_h = current_h.copy()
            new_h[src] -= delta
            new_h[dst] += delta
        elif move_type == 1:
            i, j = rng.choice(n, 2, replace=False)
            new_h = current_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
        else:
            new_h = generate_initial_h(n, M, rng)

        new_c5 = compute_c5(new_h)

        if new_c5 < best_c5 - 1e-12:
            best_h, best_c5 = new_h, new_c5
            current_h, current_c5 = new_h, new_c5
            continue
        if new_c5 < current_c5:
            current_h, current_c5 = new_h, new_c5
            continue
        if T > 0.0:
            prob = math.exp(-(new_c5 - current_c5) / T)
            if rng.random() < prob:
                current_h, current_c5 = new_h, new_c5
    return best_h, best_c5

    start_time = time.time()
    rng = np.random.default_rng(seed)

        remaining = budget_s - (time.time() - start_time)
        alloc = min(250.0, remaining * 0.5)
        h_opt, c5_opt = optimise_for_n(n, alloc, rng)
        if c5_opt < best_c5_global:

            best_h_global = h_opt
            best_n_global = n
            if best_c5_global < 0.3808:
                break

    if best_h_global is None:
        n = 80
        best_n_global = n
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        M = n // 2
        cur = best_h_global.sum()
        if abs(cur - M) > 1e-9:
            best_h_global += (M - cur) / n
            best_h_global = np.clip(best_h_global, 0.0, 1.0)

    return best_h_global, best_c5_global, n
