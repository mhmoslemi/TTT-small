def project_to_sum(h, target):
    h = np.clip(h, 0, 1)
    cur = np.sum(h)
    diff = target - cur
    eps = 1e-12
    if abs(diff) < eps:
        return h
    # If diff>0, need to increase some entries.
    if diff > 0:
        # capacity per index = 1 - h[i]
        capacity = 1 - h
        total = np.sum(capacity)
        if total < eps:
            # no capacity
            return h
        # allocate portion proportionally
        inc = diff * capacity / total
        h += inc
        # clip again
        h = np.clip(h, 0, 1)
        # recompute diff
        cur = np.sum(h)
        diff = target - cur
    else:
        # diff<0, need to decrease
        capacity = h
        total = np.sum(capacity)
        if total < eps:
            return h
        dec = -diff * capacity / total
        h -= dec
        h = np.clip(h, 0, 1)
        cur = np.sum(h)
        diff = target - cur
    # Might still not be exact due to rounding; apply small iterative adjustments limited loops
    # While loop for iteration up to, say, 5 corrections.
    for _ in range(5):
        if abs(diff) < eps:
            break
        if diff > 0:
            idx = np.where(h < 1 - eps)[0]
            if idx.size == 0: break
            # add equally to those indices
            inc = diff / idx.size
            inc = np.minimum(inc, 1 - h[idx])
            h[idx] += inc
        else:
            idx = np.where(h > eps)[0]
            if idx.size == 0: break
            dec = -diff / idx.size
            dec = np.minimum(dec, h[idx])
            h[idx] -= dec
        cur = np.sum(h)
        diff = target - cur
    return h

def project_to_sum(vec, target_sum, max_iter=100):
    # Clip to [0,1].
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = v.sum()
        diff = target_sum - cur_sum
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            # capacity to increase
            cap = 1.0 - v
            total_cap = cap.sum()
            if total_cap < 1e-12:
                break
            inc = diff * cap / total_cap
            v += inc
            v = np.clip(v, 0.0, 1.0)
        else:
            cap = v
            total_cap = cap.sum()
            if total_cap < 1e-12:
                break
            dec = (-diff) * cap / total_cap
            v -= dec
            v = np.clip(v, 0.0, 1.0)
    # Ensure sum exact by maybe adjusting smallest slack
    # compute final diff
    cur_sum = v.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-10:
        # adjust few entries
        if diff > 0:
            idx = np.where(v < 1 - 1e-12)[0]
            if idx.size > 0:
                # add equally as possible

                dec = -diff / idx.size

                v[idx] -= dec
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v

def mutate_h(h, rng, target_sum, max_step=0.2):
    n = len(h)
    cand = h.copy()
    # number of transfers
    for _ in range(rng.integers(1, 4)):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        # max delta: from i (donor) to j (recipient)
        max_delta = min(cand[i], 1.0 - cand[j])
        if max_delta <= 0:

        delta = rng.random() * max_delta * max_step
        cand[i] -= delta
        cand[j] += delta
    # optionally add small Gaussian noise
    if rng.random() < 0.3:

        cand += rng.normal(scale=sigma, size=n)
        cand = np.clip(cand, 0.0, 1.0)
    # project to sum
    cand = project_to_sum(cand, target_sum)
    return cand

import time
import numpy as np

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum(v, target):
    # Ensure within bounds
    v = np.clip(v, 0.0, 1.0)
    eps = 1e-12
    for _ in range(20):
        cur = v.sum()
        diff = target - cur
        if abs(diff) <= eps:
            break
        if diff > 0:
            mask = v < 1.0 - eps
            if not np.any(mask):
                break
            inc = diff / np.count_nonzero(mask)
            # limited by capacity
            inc_vec = np.minimum(inc, 1.0 - v[mask])
            v[mask] += inc_vec
        else:
            mask = v > eps
            if not np.any(mask):
                break
            dec = -diff / np.count_nonzero(mask)
            dec_vec = np.minimum(dec, v[mask])
            v[mask] -= dec_vec
    # final adjust residual
    cur = v.sum()
    diff = target - cur
    if abs(diff) > eps:
        if diff > 0:
            idx = np.where(v < 1.0 - eps)[0]
            if idx.size:
                add = min(diff, 1.0 - v[idx[0]])
                v[idx[0]] += add
        else:
            idx = np.where(v > eps)[0]
            if idx.size:
                sub = min(-diff, v[idx[0]])
                v[idx[0]] -= sub
    return v

def generate_random_feasible(n, target, rng):

    x = x * (target / x.sum())

    return x

        ref_h, ref_val = refine_local(best_h, dx, target, start + budget_s, rng)
        if ref_val < best_val:
            best_h, best_val = ref_h, ref_val

    # ---------- optional binary hill‑climb ----------
    time_left = budget_s - (time.time() - start)
    if time_left > 2.0:
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argpartition(best_h, -int(target))[-int(target) :]
        binary_h[top_idx] = 1
        bin_val = compute_c5(binary_h.astype(float), dx)
        bin_h, bin_val = binary_hill_climb(binary_h, bin_val, dx, rng, start + budget_s)
        if bin_val < best_val:
            best_h = bin_h.astype(float)
            best_val = bin_val

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target)
    best_val = compute_c5(best_h, dx)

    return best_h, best_val, n
