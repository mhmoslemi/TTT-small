def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type':'eq','fun': sum_con}]
    constraints += [{'type':'ineq','fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(lambda x: x[-1], x0=x0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res
```

def compute_gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = hi - np.min(v) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def compute_gradient_for_shift(h, k):
    """Gradient of the overlap O_k = sum h_i (1-h_{i+k}) w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        nk = n - k
        if nk > 0:
            grad[:nk] += 1.0 - h[k:]
            grad[k:] -= h[:nk]
    else:
        kk = -k
        nk = n - kk
        if nk > 0:
            grad[kk:] += 1.0 - h[:nk]
            grad[:nk] -= h[kk:]
    return grad

class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k)*dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500):
    """Run SLSQP from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Initialise h
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 80
        h = rng.random(n)
    target_sum = n / 2.0
    h = project_to_box_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient descent on the worst shift
    phase1_deadline = start_time + budget_s * 0.6
    step_size = 0.5
    step_decay = 0.9995
    min_step = 1e-5
    no_improve = 0

    while time.time() < phase1_deadline:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        grad = compute_gradient_for_shift(best_h, k)
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            direction = grad
        else:
            direction = grad / norm
        cand = best_h - step_size * direction
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)

        if cand_c5 + 1e-12 < best_c5:
            best_h = cand
            best_c5 = cand_c5
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000 and rng.random() < 0.5:
            # random escape
            pert = (rng.random(n) - 0.5) * step_size * 10.0
            cand2 = best_h + pert
            cand2 = project_to_box_sum(cand2, target_sum)
            cand2_c5 = compute_c5(cand2)
            if cand2_c5 < best_c5:
                best_h = cand2
                best_c5 = cand2_c5
                no_improve = 0

        step_size = max(step_size * step_decay, min_step)
        if best_c5 < 0.3808:

    # Constraints for SLSQP
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k_shift, n) for k_shift in range(-n + 1, n)]

                best_c5 = c5_opt

    # Ensure feasibility and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
