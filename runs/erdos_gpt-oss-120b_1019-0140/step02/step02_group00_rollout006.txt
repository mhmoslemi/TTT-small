def adam_optimize(h0, n_iter, step0=0.1, beta1=0.9, beta2=0.999, eps=1e-8):

   h = h0.copy()
   m = np.zeros_like(h)
   v = np.zeros_like(h)
   best_h = h.copy()
   best_score = compute_c5(h)
   for t in range(1, n_iter+1):
       grad = compute_gradients_for_shifts(h, get_shifts_of_max(h)) # we can compute inside.
       m = beta1*m + (1-beta1)*grad
       v = beta2*v + (1-beta2)*(grad**2)
       m_hat = m/(1-beta1**t)
       v_hat = v/(1-beta2**t)
       step = step0 * m_hat / (np.sqrt(v_hat) + eps)
       h_new = h - step
       # projection onto box sum
       h_new = project_onto_box_sum(h_new, h_new.size/2.0)
       score = compute_c5(h_new)
       if score < best_score:
           best_score = score
           best_h = h_new.copy()
       h = h_new
   return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # function f(lam) = sum clip(v - lam)
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def adam_optimize(h0, n_iter, step0=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter + 1):
        grad = compute_gradient(h)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, h_new.size / 2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # Return binary Thue-Morse of length n as float array
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def to_balanced_binary(h):
    n = h.size
    # break ties arbitrarily by argsort
    idx_desc = np.argsort(-h)
    bin_h = np.zeros_like(h)
    bin_h[idx_desc[:n // 2]] = 1.0
    return bin_h

def simulated_annealing_binary(h0, rng, time_limit):
    # h0 must be balanced binary (sum = n/2)
    start = time.time()
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.size
    # temperature schedule parameters
    T0 = 0.02
    Tf = 1e-5
    steps = 0
    while time.time() - start < time_limit:
        zero_idx = np.where(cur_h == 0)[0]
        one_idx = np.where(cur_h == 1)[0]
        if zero_idx.size == 0 or one_idx.size == 0:
            break
        i = rng.choice(zero_idx)
        j = rng.choice(one_idx)
        new_h = cur_h.copy()
        new_h[i] = 1.0
        new_h[j] = 0.0
        new_score = compute_c5(new_h)
        delta = new_score - cur_score
        # exponential cooling
        T = max(T0 * (0.9995 ** steps), Tf)
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        steps += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)
    # target resolution (even)
    n_target = 256
    # Prepare initial candidates
    candidates = []
    # 1. use parent construction if supplied
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = upsample(init_full, n_target)
        cand = project_onto_box_sum(cand, n_target / 2.0)
        candidates.append(cand)
    # 2. Thue-Morse
    candidates.append(project_onto_box_sum(thue_morse(n_target), n_target / 2.0))
    # 3. Random balanced binaries
    for _ in range(3):
        candidates.append(random_binary_balanced(n_target, rng))
    # 4. Uniform 0.5 vector
    candidates.append(project_onto_box_sum(np.full(n_target, 0.5, dtype=float), n_target / 2.0))
    best_h = None
    best_score = float('inf')
    # Evaluate each and optionally refine with Adam
    for cand in candidates:
        if time_left() < 5:
            break
        cur_score = compute_c5(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_h = cand.copy()
        # allocate iterations based on remaining time
        # each Adam iteration ~0.001s for n=256; be conservative
        max_iter = int(min(5000, max(500, time_left() * 300)))
        if max_iter <= 0:
            continue
        h_opt, sc_opt = adam_optimize(cand, n_iter=max_iter, step0=0.1)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
    # If time permits, run simulated annealing from the best balanced binary
    if best_h is not None and time_left() > 20:
        # Create balanced binary version of best_h
        bin_start = to_balanced_binary(best_h)
        # Ensure balanced
        bin_start = project_onto_box_sum(bin_start, n_target / 2.0)  # will keep integer but safe
        # Run SA
        sa_time = min(time_left() - 5, 120)  # leave a bit for final refinement
        h_sa, sc_sa = simulated_annealing_binary(bin_start, rng, sa_time)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa.copy()
        # Final continuous refinement
        if time_left() > 5:
            # a few thousand Adam steps
            refine_iters = int(min(8000, max(1000, time_left() * 400)))
            h_ref, sc_ref = adam_optimize(best_h, n_iter=refine_iters, step0=0.05)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref.copy()
    # Final projection (guard against numerical drift)
    best_h = project_onto_box_sum(best_h, n_target / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_target
