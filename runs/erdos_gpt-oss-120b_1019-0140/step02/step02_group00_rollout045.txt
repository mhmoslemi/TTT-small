- compute_c5(h):
   n = len(h)
   # zero-pad to length 2*n-1 for full correlation via FFT
   H = np.fft.rfft(h, n=2*n-1)
   G = np.fft.rfft(1 - h, n=2*n-1)

   dx = 2.0 / n
   c5 = max_corr * dx
   return c5

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    sz = 2 * n - 1
    H = np.fft.rfft(h, n=sz)
    G = np.fft.rfft(1.0 - h, n=sz)
    corr = np.fft.irfft(H * G, n=sz)
    return np.max(corr) * (2.0 / n)


def get_shifts_of_max_corr(h):
    """Return list of integer shifts attaining the maximal correlation."""
    n = h.shape[0]
    sz = 2 * n - 1
    H = np.fft.rfft(h, n=sz)
    G = np.fft.rfft(1.0 - h, n=sz)
    corr = np.fft.irfft(H * G, n=sz)
    max_corr = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= max_corr - eps)[0]
    return idxes - (n - 1)


def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max correlation w.r.t. h (average over shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            s = -k
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] -= h[s:]
    if shifts:
        grad /= len(shifts)
    return grad


def upsample(arr, new_n):

    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    return upsample(arr, new_n)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    vec = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    vec[idx] = 1.0
    return vec


def slsqp_optimize_candidate(h0, max_iter=500):
    """Short SLSQP to minimize max correlation."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        sz = 2 * n - 1
        H = np.fft.rfft(h, n=sz)
        G = np.fft.rfft(1.0 - h, n=sz)
        corr = np.fft.irfft(H * G, n=sz) * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def gradient_descent_adam(h0, max_iters, step_size,
                         beta1=0.9, beta2=0.999, eps=1e-8):
    """Projected Adam optimizer for the max‑correlation objective."""
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iters + 1):
        shifts = get_shifts_of_max_corr(h)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h - step, target_sum)

        cur = compute_c5(h)
        if cur < best_score:
            best_score = cur
            best_h = h.copy()
    return best_h, best_score


def hill_climb_binary_swaps(h_bin, time_left, rng):

    n = h_bin.size
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    ones = np.flatnonzero(best_h == 1.0)
    zeros = np.flatnonzero(best_h == 0.0)

            best_score = sc
            # update index arrays
            ones = np.flatnonzero(best_h == 1.0)
            zeros = np.flatnonzero(best_h == 0.0)
    return best_h, best_score

    h_bin = np.zeros_like(best_h)
    h_bin[order[:n // 2]] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_h, best_score = h_bin, sc_bin

    if time_left() > 5.0:
        h_sw, sc_sw = hill_climb_binary_swaps(best_h, time_left, rng)
        if sc_sw < best_score:
            best_h, best_score = h_sw, sc_sw

    # ---- Random binary search for any missed improvement ----
    while time_left() > 2.0:
        h_rand = random_binary_balanced(n_large, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_h, best_score = h_rand, sc_rand
        if best_score <= 0.3808:
            break

    # ---- Optional upsample to finer grid if time permits ----
    if time_left() > 10.0:
        n_fine = 512
        h_fine = upsample(best_h, n_fine)
        h_fine = project_onto_box_sum(h_fine, n_fine / 2.0)
        # short Adam on fine grid
        est_iters = int(min(20000, max(2000, time_left() * 1500)))
        h_fine_opt, sc_fine_opt = gradient_descent_adam(
            h_fine, max_iters=est_iters, step_size=0.15)
        if sc_fine_opt < best_score:
            best_h, best_score = h_fine_opt, sc_fine_opt

    # final projection safeguard
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
