def compute_overlap_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def apply_transfer(overlap, h, src, dst, delta):
    n = h.shape[0]
    # src contributions (decrease)
    start_src = n - 1 - src
    # since we need original h values before change for factor (1-h)
    overlap[start_src:start_src + n] -= delta * (1.0 - h)

    # dst contributions (increase)

    h[src] -= delta
    h[dst] += delta

        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)

    if abs(cur - target) > 1e-12:
        h += (target - cur) / n
        np.clip(h, 0.0, 1.0, out=h)
        # fix remaining diff
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # greedy adjust
            indices = np.arange(n)
            rng.shuffle(indices)
            for i in indices:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:

                    h[i] += inc
                    diff -= inc
                elif diff < 0 and h[i] > 0.0:

                    h[i] -= dec
                    diff += dec

            dst_candidates = np.where(h < 1.0 - eps)[0]

    return np.correlate(h, 1.0 - h, mode='full')

    n = h.shape[0]
    # Decrease src
    start_src = n - 1 - src
    overlap[start_src:start_src + n] -= delta * (1.0 - h)
    overlap[src:src + n] += delta * h
    # Increase dst

    h[src] -= delta
    h[dst] += delta

        # propose a move
        if k == 0:
            src_cand = np.where(h > eps)[0]
            dst_cand = np.where(h < 1.0 - eps)[0]
            if src_cand.size == 0 or dst_cand.size == 0:
                break
            src = int(rng.choice(src_cand))
            dst = int(rng.choice(dst_cand))
        else:
            if k > 0:
                i_range = np.arange(n - k)
                prod = h[i_range] * (1.0 - h[i_range + k])
            else:
                i_range = np.arange(-k, n)
                prod = h[i_range] * (1.0 - h[i_range + k])
            i_local = int(prod.argmax())
            src = int(i_range[i_local])
            dst = int(src + k)

            # fallback if chosen pair cannot move
            if not (0 <= dst < n) or h[src] <= eps or h[dst] >= 1.0 - eps:
                src_cand = np.where(h > eps)[0]
                dst_cand = np.where(h < 1.0 - eps)[0]
                if src_cand.size == 0 or dst_cand.size == 0:
                    break
                src = int(rng.choice(src_cand))
                dst = int(rng.choice(dst_cand))

        if src == dst:
            continue

        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= eps:
            continue
        delta = max_transfer * (0.1 + 0.9 * rng.random())

        # save state for possible revert
        overlap_before = overlap.copy()
        cur_raw_before = cur_raw

        # perform move
        apply_transfer(overlap, h, src, dst, delta)
        new_raw = overlap.max()

        # accept if not worse, otherwise Metropolis
        if new_raw <= cur_raw_before + 1e-15:
            cur_raw = new_raw
            if new_raw < best_raw - 1e-12:
                best_raw = new_raw
                best_h = h.copy()
        else:
            elapsed = time.time() - start
            T = max(1e-12, T0 * (1.0 - elapsed / budget_s))
            prob = np.exp((cur_raw_before - new_raw) / T)
            if prob > rng.random():
                cur_raw = new_raw
                if new_raw < best_raw - 1e-12:
                    best_raw = new_raw
                    best_h = h.copy()
            else:
                # revert
                h[src] += delta
                h[dst] -= delta
                overlap[:] = overlap_before
                cur_raw = cur_raw_before

        # early exit if we already beat the target bound
        if best_raw * (2.0 / n) < 0.3808:
            break

    # ----- final sanitation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h += (target - cur) / n
        np.clip(best_h, 0.0, 1.0, out=best_h)
        diff = target - best_h.sum()
        if abs(diff) > 1e-12:
            order = np.arange(n)
            rng.shuffle(order)
            for i in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and best_h[i] < 1.0:
                    inc = min(diff, 1.0 - best_h[i])
                    best_h[i] += inc
                    diff -= inc
                elif diff < 0 and best_h[i] > 0.0:
                    dec = min(-diff, best_h[i])
                    best_h[i] -= dec
                    diff += dec

    # recompute C5 from the final vector to avoid drift
    final_raw = compute_overlap_array(best_h).max()
    final_c5 = final_raw * (2.0 / n)

    return best_h.astype(float), float(final_c5), int(n)
