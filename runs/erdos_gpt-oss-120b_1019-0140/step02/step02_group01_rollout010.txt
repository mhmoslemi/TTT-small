def gradient_of_overlap(h, k):
    n = h.shape[0]; dx = 2.0 / n
    g = np.zeros_like(h)
    if k >= 0:
        # forward term
        g[:n-k] = dx * (1.0 - h[k:])   # note h[k:] length n-k
        # backward term
        g[k:] -= dx * h[:n-k]          # h[:n-k] is h[i-k] for i = k..n-1
    else:
        kk = -k
        # forward term (1 - h[i - kk]) for i >= kk
        g[kk:] = dx * (1.0 - h[:n-kk]) # h[:n-kk] corresponds to h[i-kk]
        # backward term -h[i+kk] for i <= n-1-kk
        g[:n-kk] -= dx * h[kk:]        # h[kk:] is h[i+kk] for i=0..n-1-kk
    return g
```

def project_onto_capped_simplex(v, M):
    # Solve min ||x-v||^2 s.t. 0 <= x_i <= 1, sum x_i = M
    # This is projection onto intersection of box and hyperplane.
    # We can use bisection for lambda: x_i = min(max(v_i - lambda, 0), 1)
    # Find lambda such that sum x_i = M.
    # Since sum of clip function is decreasing in lambda (monotonic), can binary search.
    lo = -1.0
    hi = 2.0
    # extend bounds: v_i - lambda > 1 => lambda < v_i - 1 => lower bound may be up to max(v_i) - 1.
    # We can find min and max needed: lambda in [min(v_i)-1, max(v_i)].
    lo = min(v) - 1.0
    hi = max(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > M:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # due to numeric errors, adjust to meet sum exactly by distributing small residual.
    residual = M - x.sum()
    if abs(residual) > 1e-12:
        # adjust using free variables (those not at bounds) if any
        free = (x > 1e-12) & (x < 1-1e-12)
        if free.any():
            # distribute residual equally (small) among free vars
            x[free] += residual / free.sum()
            # reclip just in case
            x = np.clip(x, 0.0, 1.0)
    # final clip ensure sum close
    return x
```

import time
import numpy as np

# ---------- helpers ----------
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def compute_c5_and_k(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * dx, idx - (n - 1)

def gradient_of_overlap(h, k):
    n = h.shape[0]
    dx = 2.0 / n
    g = np.zeros_like(h)
    if k >= 0:
        g[:n - k] = dx * (1.0 - h[k:])
        g[k:] -= dx * h[:n - k]
    else:
        kk = -k
        g[kk:] = dx * (1.0 - h[:n - kk])
        g[:n - kk] -= dx * h[kk:]
    return g

def project_onto_capped_simplex(v, M):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > M:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    residual = M - x.sum()
    if abs(residual) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += residual / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def random_feasible(n, rng, M):
    v = rng.random(n)
    v = v / v.sum() * M
    return project_onto_capped_simplex(v, M)

def get_initial_h(n):
    init = globals().get("initial_h_values")
    if init is not None and len(init) == n:
        return np.array(init, dtype=float)
    return None

def optimise_n(n, rng, time_budget, global_best):
    start = time.time()
    M = n // 2
    best_h = None
    best_c5 = np.inf
    init_h = get_initial_h(n)
    while time.time() - start < time_budget - 2.0:
        if init_h is not None and best_h is None:
            h = project_onto_capped_simplex(init_h, M)
        else:
            h = random_feasible(n, rng, M)
        alpha = 0.05 + 0.05 * rng.random()
        decay = 0.995
        for it in range(500):
            if time.time() - start >= time_budget - 2.0:
                break
            c5, k = compute_c5_and_k(h)
            if c5 + 1e-12 < best_c5:
                best_c5, best_h = c5, h.copy()
                if best_c5 < global_best - 1e-12:
                    return best_h, best_c5
            g = gradient_of_overlap(h, k)
            h = project_onto_capped_simplex(h - alpha * g, M)
            alpha *= decay
            if (it + 1) % 100 == 0:
                h = project_onto_capped_simplex(h + rng.normal(scale=0.004, size=n), M)
    return best_h, best_c5

def refine_local(h, rng, time_left):
    start = time.time()
    n = h.shape[0]
    M = n // 2
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    while time.time() - start < time_left:
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - best_h[i], best_h[j])
        if max_transfer <= 0.0:

        delta = rng.random() * max_transfer
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:

    return best_h, best_c5

        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    time_left = budget_s - (time.time() - start) - 0.5
    if time_left > 0.0:

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    M = best_n // 2
    cur = best_h.sum()
    if abs(cur - M) > 1e-9:
        best_h = project_onto_capped_simplex(best_h, M)

    return best_h, best_c5, best_n
