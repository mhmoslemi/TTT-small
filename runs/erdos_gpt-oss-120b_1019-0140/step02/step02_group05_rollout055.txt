def enforce_sum_and_clip(h, target_sum):
    h = np.clip(h, 0., 1.)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # distribute diff equally among free positions
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0., 1.)
        # residual adjust sequentially
        diff = target_sum - np.sum(h)
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0., 1.)
    return h

def gradient_step(h, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = np.argmax(corr)
    k = idx_max - (n - 1)
    # compute gradient of overlap for shift k
    grad = np.zeros(n)
    if k >= 0:
        # for i from 0 to n-1-k, term d/dh_i is dx*(1 - h_{i+k})
        grad[: n - k] = dx * (1.0 - h[k:])
        # for i from k to n-1, term d/dh_i is -dx*h_{i-k}
        grad[k:] += -dx * h[: n - k]
    else:
        kk = -k
        # i from kk to n-1: d/dh_i from first part h_i*(1 - h_{i+ k}) where i+k = i - kk
        grad[kk:] = dx * (1.0 - h[: n - kk])
        # i from 0 to n-kk-1: second part -dx*h_{i+kk}
        grad[: n - kk] += -dx * h[kk:]
    # We aim to reduce overlap, so move opposite direction of gradient.
    # Project onto sum-zero direction: subtract mean.
    grad_centered = grad - np.mean(grad)
    # step size adaptively limited
    step = -grad_centered
    max_step = 0.3  # limit max absolute step
    mabs = np.max(np.abs(step))
    if mabs > max_step:
        step = step * (max_step / mabs)
    cand = h + step
    cand = np.clip(cand, 0., 1.)
    cand = enforce_sum_and_clip(cand, target_sum)
    return cand

def perturb_move(h, target_sum, max_delta=0.1):
    n = len(h)
    idx = np.random.randint(n)
    delta = (np.random.rand()*2 - 1) * max_delta  # uniform [-max_delta,max_delta]

def enforce_sum_and_clip(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if np.abs(diff) > 1e-12:
        # adjust using free positions
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        # residual fix
        diff = target_sum - np.sum(h)
        # distribute sequentially
        for i in range(len(h)):
            if np.abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral C5 for a discretized h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum_and_clip(h, target_sum):
    """Clip to [0,1] and adjust to meet exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if np.abs(diff) > 1e-12:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        for i in range(len(h)):
            if np.abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_once(h):
    """One pass of 3‑point averaging."""
    kernel = np.array([0.25, 0.5, 0.25])
    return np.convolve(h, kernel, mode="same")

def generate_random_h(rng, n, target_sum):
    v = rng.random(n)
    v = v * (target_sum / np.sum(v))
    v = np.clip(v, 0.0, 1.0)
    return enforce_sum_and_clip(v, target_sum)

def perturb_move(rng, h, target_sum, max_delta=0.1):
    n = len(h)
    idx = rng.integers(n)
    delta = (rng.random() * 2.0 - 1.0) * max_delta
    cand = h.copy()
    cand[idx] = np.clip(cand[idx] + delta, 0.0, 1.0)
    return enforce_sum_and_clip(cand, target_sum)

def gradient_step(h, target_sum):
    """Move opposite to gradient of the worst shift."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += dx * (1.0 - h[k:])
        grad[k:]     += -dx * h[: n - k]
    else:
        kk = -k
        grad[kk:]    += dx * (1.0 - h[: n - kk])
        grad[: n - kk] += -dx * h[kk:]
    # zero‑sum direction
    grad -= np.mean(grad)
    step = -grad
    max_step = 0.3
    mabs = np.max(np.abs(step))
    if mabs > max_step:
        step = step * (max_step / mabs)
    cand = h + step
    cand = np.clip(cand, 0.0, 1.0)
    return enforce_sum_and_clip(cand, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initial shape -------------------------------------------------
    desired_n = 200
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n_init = len(init)
        if n_init == desired_n:
            h = init.copy()
        else:
            x_old = np.linspace(0.0, 2.0, n_init, endpoint=False)
            x_new = np.linspace(0.0, 2.0, desired_n, endpoint=False)
            h = np.interp(x_new, x_old, init)
    else:
        h = np.zeros(desired_n, dtype=float)

    n = desired_n
    target_sum = n / 2.0
    h = enforce_sum_and_clip(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    T0 = 0.6                # initial temperature
    no_improve = 0
    restart_interval = 2500

    # ----- simulated annealing -----------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.94:
            break
        # restart if stagnated
        if no_improve >= restart_interval:
            cur_h = generate_random_h(rng, n, target_sum)
            cur_val = compute_c5(cur_h)
            no_improve = 0

        T = T0 * max(0.0, 1.0 - elapsed / (budget_s * 0.94))

        r = rng.random()
        if r < 0.4:               # gradient move
            cand = gradient_step(cur_h, target_sum)
        elif r < 0.7:             # transfer move
            i, j = rng.choice(n, 2, replace=False)
            donor, recv = i, j
            max_tr = min(cur_h[donor], 1.0 - cur_h[recv])
            if max_tr <= 0:
                max_tr = min(cur_h[recv], 1.0 - cur_h[donor])
                if max_tr <= 0:
                    continue
                donor, recv = recv, donor
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
            cand = enforce_sum_and_clip(cand, target_sum)
        elif r < 0.85:            # perturb move
            cand = perturb_move(rng, cur_h, target_sum, max_delta=0.12)
        else:                     # swap move (preserves sum)
            i, j = rng.choice(n, 2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]

        cand_val = compute_c5(cand)
        accept = False
        if cand_val < cur_val:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(cand_val - cur_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h = cand
            cur_val = cand_val
            no_improve = 0
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break
        else:
            no_improve += 1

    # ----- smoothing passes ---------------------------------------------
    time_left = budget_s - (time.time() - start)
    if time_left > 2.0:
        max_pass = int(min(300, time_left / 0.02))
        for _ in range(max_pass):
            sm = smooth_once(best_h)
            sm = enforce_sum_and_clip(sm, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
            else:
                break

    # ----- final correction ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_and_clip(best_h, target_sum)

    return best_h, best_val, n
