def compute_c5(arr, n):
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def enforce_constraints(arr, target_sum):
    n = len(arr)
    arr = np.clip(arr, 0.0, 1.0)
    diff = target_sum - np.sum(arr)
    if abs(diff) > 1e-12:
        arr += diff / n
        arr = np.clip(arr, 0.0, 1.0)
        residual = target_sum - np.sum(arr)
        if abs(residual) > 1e-12:
            if residual > 0:
                slack = 1.0 - arr
                if np.any(slack > 0):
                    idx = np.argmax(slack)
                    add = min(residual, slack[idx])
                    arr[idx] += add
            else:
                slack = arr
                if np.any(slack > 0):
                    idx = np.argmax(slack) # largest value for decreasing
                    sub = min(-residual, arr[idx])
                    arr[idx] -= sub
            arr = np.clip(arr, 0.0, 1.0)
    return arr

import time
import math
import numpy as np
from scipy.optimize import differential_evolution

# global variables for DE time callback
DE_START_TIME = 0.0
DE_BUDGET_S = 0.0

def compute_c5(arr, n):
    """Approximate C5 from discrete sample array."""
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def enforce_constraints(arr, target_sum):
    """Clip to [0,1] and enforce exact sum target_sum."""
    n = len(arr)
    arr = np.clip(arr, 0.0, 1.0)
    diff = target_sum - np.sum(arr)
    if abs(diff) > 1e-12:
        arr += diff / n
        arr = np.clip(arr, 0.0, 1.0)
        residual = target_sum - np.sum(arr)
        if abs(residual) > 1e-12:
            if residual > 0:
                slack = 1.0 - arr
                if np.any(slack > 0):
                    idx = np.argmax(slack)
                    add = min(residual, slack[idx])
                    arr[idx] += add
            else:
                slack = arr
                if np.any(slack > 0):
                    idx = np.argmax(slack)  # largest value to decrease
                    sub = min(-residual, arr[idx])
                    arr[idx] -= sub
            arr = np.clip(arr, 0.0, 1.0)
    return arr

def thue_morse_binary(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def rudin_shapiro_binary(n):
    """Binary Rudin–Shapiro sequence of length n (0/1)."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = a[:n]
    arr = np.array(a, dtype=int)
    return ((arr + 1) // 2).astype(float)

def de_objective(x, target_sum, n):
    """Objective for differential evolution."""
    h = enforce_constraints(x, target_sum)
    return compute_c5(h, n)

def de_callback(xk, convergence):
    """Stop DE when time budget is nearly exhausted."""
    return (time.time() - DE_START_TIME) > (DE_BUDGET_S - 2.0)

def local_refine_continuous(h, target_sum, time_end, rng):
    """Greedy pairwise transfer moves while time permits."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    step = 0.02
    while time.time() < time_end and step > 1e-5:
        improved = False
        for _ in range(200):
            i, j = rng.choice(n, 2, replace=False)
            max_inc = min(step, 1.0 - best_h[i])
            max_dec = min(step, best_h[j])
            if max_inc <= 0 or max_dec <= 0:
                continue

            cand = best_h.copy()
            cand[i] += delta
            cand[j] -= delta
            cand_c5 = compute_c5(cand, n)
            if cand_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cand, cand_c5
                improved = True
                break
            # opposite direction

            max_dec2 = min(step, best_h[i])
            if max_inc2 <= 0 or max_dec2 <= 0:
                continue
            delta2 = min(max_inc2, max_dec2)

            cand2[j] += delta2

            cand2_c5 = compute_c5(cand2, n)
            if cand2_c5 < best_c5 - 1e-12:

                improved = True
                break
        if not improved:
            step *= 0.5
    best_h = enforce_constraints(best_h, target_sum)
    return best_h, compute_c5(best_h, n)

def simulated_annealing_continuous(h_init, target_sum, time_end, rng):
    """Simple continuous SA keeping the sum constant by transfers."""
    n = len(h_init)
    h = h_init.copy()
    cur_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 200000
    it = 0
    while time.time() < time_end and it < max_iter:
        i, j = rng.choice(n, 2, replace=False)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new, n)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:

        it += 1
    best_h = enforce_constraints(best_h, target_sum)
    return best_h, compute_c5(best_h, n)
