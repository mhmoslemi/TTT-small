def smooth_min_gradient(h, beta):
    n = h.shape[0]
    corr = np.correlate(h, h, mode='full')
    # corr array length 2n-1; shift index k = idx - (n-1)
    O = corr
    # Shift indices range -(n-1)...(n-1)
    # Compute weights: w = exp(-beta * (O - O_max)) to avoid underflow
    O_max = O.max()
    w = np.exp(-beta * (O - O_max))
    p = w / w.sum()
    grad = np.zeros_like(h)
    # iterate over shifts using slices
    for idx, weight in enumerate(p):
        if weight == 0:
            continue
        shift = idx - (n - 1)
        if shift == 0:
            # gradient of O_0 = sum_i h_i^2 ; derivative 2*h_i
            grad += weight * 2.0 * h
            continue
        if shift > 0:
            i = slice(0, n - shift)
            j = slice(shift, n)
            grad[i] += weight * h[j]
            grad[j] += weight * h[i]
        else:
            s = -shift
            i = slice(s, n)
            j = slice(0, n - s)
            grad[i] += weight * h[j]
            grad[j] += weight * h[i]
    return grad

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]  # t can be up to 1 (max correlation)
    def eq_sum(x): return np.sum(x[:n]) - n/2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type':'eq','fun':eq_sum},
            {'type':'ineq','fun':ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    # reproject onto constraints, just in case:
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def smooth_max_gradient(h, beta):
    """
    Gradient of the smooth max of F_s = corr_s = sum_i h_i(1-h_{i+s})
    where s ranges over all integer shifts (full mode). Implements
    ∇ L = Σ_s p_s ∇F_s with p_s = exp(beta*F_s)/Σ_t exp(beta*F_t).
    """
    n = h.size
    F = np.correlate(h, 1.0 - h, mode='full')               # length 2n-1
    maxF = F.max()
    w = np.exp(beta * (F - maxF))                           # shift for stability
    p = w / w.sum()

    grad = np.zeros_like(h)

    # iterate over all shifts
    for idx, weight in enumerate(p):
        if weight == 0.0:
            continue
        shift = idx - (n - 1)          # shift value: -(n-1) .. (n-1)
        if shift >= 0:
            # left part: i = 0 .. n-shift-1  (contribute 1 - h[i+shift])
            l = n - shift
            grad[0:l] += weight * (1.0 - h[shift:n])
            # right part: i = shift .. n-1 (contribute -h[i-shift])
            grad[shift:n] += weight * (-h[0:l])
        else:
            s = -shift
            l = n - s
            # i = s .. n-1  (contribute 1 - h[i-s])
            grad[s:n] += weight * (1.0 - h[0:l])
            # i = 0 .. n-s-1 (contribute -h[i+s])
            grad[0:l] += weight * (-h[s:n])
    return grad

def slsqp_optimize_candidate(h0, max_iter=500):
    """Short SLSQP refinement of a candidate."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [
        {'type': 'eq', 'fun': eq_sum},
        {'type': 'ineq', 'fun': ineq_fun}
    ]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    # target resolution
    n = 256

    # --- Initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        h = upsample(init_full, n)
    else:
        h = rng.random(n)

    h = project_onto_box_sum(h, n / 2.0)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # --- Smooth‑max gradient descent --------------------------------
    momentum = np.zeros_like(h)
    it = 0
    beta_start = 5.0
    beta_max = 60.0
    step0 = 0.5

    while time_left() > 1.0:
        it += 1
        # ramp beta linearly over first 3000 iterations
        if it <= 3000:
            beta = beta_start + (beta_max - beta_start) * (it / 3000.0)
        else:
            beta = beta_max

        grad = smooth_max_gradient(h, beta)
        momentum = 0.9 * momentum + grad
        eta = step0 / np.sqrt(it)
        h = h - eta * momentum
        h = project_onto_box_sum(h, n / 2.0)

        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            if best_score <= 0.3808:
                # good enough – keep improving but give a small slack
                pass

    # --- Optional SLSQP refinement ----------------------------------
    if time_left() > 15.0:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=400)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # --- Random binary search (if time left) -----------------------
    while time_left() > 1.0:
        h_rand = random_binary_balanced(n, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
            if best_score <= 0.3808:
                break

    # final safe projection
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
