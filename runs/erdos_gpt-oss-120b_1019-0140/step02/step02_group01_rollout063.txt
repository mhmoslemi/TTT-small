def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate n's list
    candidate_ns = [80, 96, 112, 128, 144, 160, 176, 192]

    # place initial parent vector if exists
    if 'initial_h_values' in globals():
        n_parent = len(initial_h_values)
        if n_parent % 2 == 0 and n_parent not in candidate_ns:
            candidate_ns.insert(0, n_parent)
        else:
            # maybe separate approach
            pass

    for n in candidate_ns:
        if time.time() - start > budget_s - 1.0:
            break
        # Initialize h
        # if parent size matches n, start with that
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h = np.array(initial_h_values, dtype=float)
            # ensure sum exactly n/2
            cur = np.sum(h)
            target = n / 2.0
            if abs(cur - target) > 1e-12:
                h += (target - cur) / n
                np.clip(h, 0.0, 1.0, out=h)
        else:
            # linear decreasing shape from 1 to 0
            h = np.linspace(1.0, 0.0, n)
            # adjust to sum to n/2
            cur = np.sum(h)
            target = n / 2.0
            # scaling factor
            scaling = target / cur
            h = h * scaling
            # clip, adjust again if needed
            np.clip(h, 0.0, 1.0, out=h)
            cur = np.sum(h)
            if abs(cur - target) > 1e-12:
                h += (target - cur) / n
                np.clip(h, 0.0, 1.0, out=h)

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

        # Simulated annealing / hill climbing on this n
        # Set up temperature schedule
        # Let's use exponential cooling: temp_start ~0.1 maybe
        temp_start = 0.02  # initial temperature
        temp_end = 1e-5
        # Decide iterations based on time budget
        # We'll try many moves; compute approx iteration count per n
        max_iters = 50000  # arbitrary large; break if time out
        for it in range(max_iters):
            if time.time() - start > budget_s - 0.5:
                break
            # temperature decay
            # linear dec (or exponential)
            frac = it / max_iters
            temp = temp_start * (1 - frac) + temp_end * frac

            # propose move
            i, j = rng.integers(0, n, size=2, endpoint=False)
            if i == j:
                continue
            # compute max mass that can be moved from j to i
            max_transfer = min(1.0 - h[i], h[j])
            if max_transfer <= 0:
                continue
            # propose delta
            delta = rng.random() * max_transfer
            # create new h (copy)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            # compute c5
            c5_new = compute_c5(h_new)
            # Acceptance
            if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / max(temp, 1e-12)):
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                    best_n = n
                # early exit if good enough
                if best_c5 < 0.3808:
                    break
        if best_c5 < 0.3808:
            break

    # final fallback if no solution
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        # final enforcement of constraints
        np.clip(best_h, 0.0, 1.0, out=best_h)
        cur = np.sum(best_h)
        target = best_n / 2.0
        if abs(cur - target) > 1e-9:
            best_h += (target - cur) / best_n
            np.clip(best_h, 0.0, 1.0, out=best_h)

    return best_h, best_c5, best_n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function achieving a low C₅ bound."""
    start = time.time()

    # Choose resolution. Prefer the length of a possible parent solution,
    # otherwise take a modest size (200 points) that stays well below 1 000.
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 200
    if n % 2 != 0:                     # ensure even number of points
        n += 1

    dx = 2.0 / n
    i = np.arange(n)
    # Mid‑point sampling of the symmetric V‑shape |x‑1|
    x = (i + 0.5) * dx
    h = np.abs(x - 1.0)

    # Normalise exactly to ∫₀² h = 1
    cur = np.sum(h) * dx
    if abs(cur - 1.0) > 1e-14:
        h = h / cur

    # Enforce 0 ≤ h ≤ 1 (already true, but robust against rounding)
    np.clip(h, 0.0, 1.0, out=h)

    # Enforce the discrete sum constraint ∑h = n/2  (within tiny tolerance)
    target = n / 2.0
    tot = np.sum(h)
    if abs(tot - target) > 1e-12:
        h += (target - tot) / n
        np.clip(h, 0.0, 1.0, out=h)

    # Evaluate the bound
    c5 = compute_c5(h)

    # Return the solution (h, bound, number of points)
    return h, c5, n
