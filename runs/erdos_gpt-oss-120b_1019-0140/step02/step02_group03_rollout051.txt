def solve_milp_binary(n, time_limit, rng):
    # Build MILP with binary h, integer t.
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n/2,
                   t >= 0,
                   t <= n]
    # For each shift
    for k in range(1, n):
        # positive
        idx = np.arange(n - k)
        hi = h[idx]
        hj = h[idx + k]
        y = cp.Variable(n - k, boolean=True)
        constraints += [y <= hi,
                        y <= 1 - hj,
                        y >= hi - hj,
                        cp.sum(y) <= t]
        # negative shift
        hi2 = h[idx + k]
        hj2 = h[idx]
        y2 = cp.Variable(n - k, boolean=True)
        constraints += [y2 <= hi2,
                        y2 <= 1 - hj2,
                        y2 >= hi2 - hj2,
                        cp.sum(y2) <= t]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None
    h_val = np.round(h.value).astype(float)  # rounding to 0/1
    t_val = int(round(t.value))
    return h_val, t_val

def adjust_to_sum(vec, target_ones, rng):
    current = int(vec.sum())
    diff = target_ones - current
    if diff > 0:
        zeros = np.where(vec == 0)[0]
        idx = rng.choice(zeros, size=diff, replace=False)
        vec[idx] = 1
    elif diff < 0:
        ones = np.where(vec == 1)[0]
        idx = rng.choice(ones, size=-diff, replace=False)
        vec[idx] = 0
    return vec

    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def adjust_to_sum(vec, target_ones, rng):
    """Force a binary vector to have exactly target_ones ones."""
    cur = int(vec.sum())
    diff = target_ones - cur
    if diff > 0:
        zeros = np.where(vec == 0)[0]
        if zeros.size >= diff:
            idx = rng.choice(zeros, size=diff, replace=False)
            vec[idx] = 1
    elif diff < 0:
        ones = np.where(vec == 1)[0]
        if ones.size >= -diff:
            idx = rng.choice(ones, size=-diff, replace=False)
            vec[idx] = 0
    return vec

def random_binary_vec(n, target_ones, rng):
    """Generate a random binary vector with exactly target_ones ones."""
    arr = np.zeros(n, dtype=float)
    arr[:target_ones] = 1.0
    rng.shuffle(arr)
    return arr

# ---------- MILP solver ----------
def solve_milp_binary(n, time_limit):
    """Attempt to solve the binary MILP for the Erdős overlap problem."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)

    constraints = [
        cp.sum(h) == n / 2,
        t >= 0,
        t <= n
    ]

    for k in range(1, n):
        idx = np.arange(n - k)
        hi = h[idx]
        hj = h[idx + k]
        y = cp.Variable(n - k, boolean=True)
        constraints += [
            y <= hi,
            y <= 1 - hj,
            y >= hi - hj,
            cp.sum(y) <= t
        ]

        # negative shift -k
        hi2 = h[idx + k]
        hj2 = h[idx]
        y2 = cp.Variable(n - k, boolean=True)
        constraints += [
            y2 <= hi2,
            y2 <= 1 - hj2,
            y2 >= hi2 - hj2,
            cp.sum(y2) <= t
        ]

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return None, None

    # Accept any feasible solution (including USER_LIMIT)
    if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.USER_LIMIT):
        if h.value is not None and t.value is not None:
            h_val = np.rint(np.clip(h.value, 0, 1)).astype(float)
            # ensure exact sum
            h_val = adjust_to_sum(h_val, n // 2, np.random.default_rng())
            t_val = int(round(t.value))
            return h_val, t_val
    return None, None

# ---------- Greedy binary improvement ----------
def greedy_improve_binary(start_h, rng, max_iters=2000):
    """Perform greedy swaps until no further improvement."""
    h = start_h.copy()
    best_c5 = compute_c5(h)
    n = h.shape[0]
    improved = True
    it = 0
    while improved and it < max_iters:
        improved = False
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            for j in zeros:
                cand = h.copy()
                cand[i] = 0.0
                cand[j] = 1.0
                cand_c5 = compute_c5(cand)
                if cand_c5 + 1e-12 < best_c5:
                    h = cand
                    best_c5 = cand_c5
                    improved = True
                    break
            if improved:
                break
    return h, best_c5

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.size
    else:
        n = 80
        init_h = None

    target_ones = n // 2
    best_h = None
    best_c5 = float('inf')

    # ---- Phase 1: MILP (binary) ---------------------------------
    time_elapsed = time.time() - start_time
    remaining = max(budget_s - time_elapsed, 0.0)
    milp_time = min(remaining * 0.5, 250.0)  # allocate up to 250 s
    if milp_time > 5.0:  # give it a chance only if we have time
        milp_h, milp_t = solve_milp_binary(n, milp_time)
        if milp_h is not None:
            c5_val = compute_c5(milp_h)
            best_h = milp_h
            best_c5 = c5_val
            # Early exit if already good enough
            if best_c5 <= 0.3808:
                return best_h, best_c5, n

    # ---- Phase 2: Greedy random restarts ------------------------
    # Prepare a starting binary vector (either from init or random)
    def prepare_start():
        if init_h is not None:
            bin_vec = np.where(init_h >= 0.5, 1.0, 0.0)
            bin_vec = adjust_to_sum(bin_vec, target_ones, rng)
            return bin_vec
        return random_binary_vec(n, target_ones, rng)

    # Main loop of restarts
    while True:
        time_elapsed = time.time() - start_time
        remaining = budget_s - time_elapsed
        if remaining < 1.0:
            break

        # Allocate a modest chunk for one restart
        chunk = min(5.0, remaining)

        # Generate start and improve greedily
        start_vec = prepare_start()
        h_opt, c5_opt = greedy_improve_binary(start_vec, rng, max_iters=5000)

        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            if best_c5 <= 0.3808:
                break

        # If we are close to the target, do a short randomized hill‑climb
        if best_c5 < 0.390:
            # perform a few random swaps allowing occasional uphill moves
            cur = best_h.copy()
            cur_c5 = best_c5
            inner_start = time.time()
            while time.time() - inner_start < chunk:
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                cand = cur.copy()
                cand[i] = 0.0
                cand[j] = 1.0
                cand_c5 = compute_c5(cand)
                # simple simulated‑annealing acceptance
                if cand_c5 < cur_c5 or rng.random() < 0.001:
                    cur = cand
                    cur_c5 = cand_c5
                    if cur_c5 + 1e-12 < best_c5:
                        best_h = cur.copy()
                        best_c5 = cur_c5
                        if best_c5 <= 0.3808:
                            break
            # after inner climb, keep best found
            if best_c5 <= 0.3808:
                break

    # Final safety: ensure feasibility
    if best_h is None:
        # fallback to a simple feasible vector
        best_h = random_binary_vec(n, target_ones, rng)
        best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_to_sum(best_h, target_ones, rng)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
