import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while enforcing sum(v)=target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def _initial_binary(pop_size, n, target_ones, rng):
    """Generate pop_size binary vectors of length n with exactly target_ones ones."""
    pop = np.zeros((pop_size, n), dtype=np.uint8)
    for i in range(pop_size):
        idx = rng.choice(n, target_ones, replace=False)
        pop[i, idx] = 1
    return pop

def _crossover_and_mutate(p1, p2, target_ones, rng):
    """Uniform crossover of two parents and repair to exact number of ones; include mutation."""
    n = p1.shape[0]
    mask = rng.random(n) < 0.5
    child = np.where(mask, p1, p2).astype(np.uint8)
    ones = int(child.sum())
    if ones > target_ones:
        idx_one = np.flatnonzero(child)
        to_flip = rng.choice(idx_one, ones - target_ones, replace=False)
        child[to_flip] = 0
    elif ones < target_ones:
        idx_zero = np.flatnonzero(child == 0)
        to_flip = rng.choice(idx_zero, target_ones - ones, replace=False)
        child[to_flip] = 1
    # mutation: swap a 1 with a 0 with prob 0.2
    if rng.random() < 0.2:
        idx_one = np.flatnonzero(child)
        idx_zero = np.flatnonzero(child == 0)
        if idx_one.size and idx_zero.size:
            i1 = rng.choice(idx_one)
            i0 = rng.choice(idx_zero)
            child[i1] = 0
            child[i0] = 1
    return child

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    target_ones = int(round(target_sum))

    # Initialise best solution with the provided one (if any)
    best_h = None
    best_c5 = np.inf
    if h0 is not None:
        best_h = h0.copy()
        best_c5 = compute_c5(best_h)

    # ---------- Genetic Algorithm (binary) ----------
    pop_size = 200
    elite_size = 2
    pop = _initial_binary(pop_size, n, target_ones, rng)
    fitness = np.empty(pop_size, dtype=np.float64)
    for i in range(pop_size):
        fitness[i] = compute_c5(pop[i])

    # Keep track of global best
    idx_best = int(np.argmin(fitness))
    if fitness[idx_best] < best_c5:
        best_c5 = float(fitness[idx_best])
        best_h = pop[idx_best].astype(np.float64)

    ga_deadline = start_time + budget_s * 0.75
    while time.time() < ga_deadline:
        # Elitism
        elite_idx = np.argsort(fitness)[:elite_size]
        new_pop = np.empty_like(pop)
        new_fitness = np.empty_like(fitness)
        new_pop[:elite_size] = pop[elite_idx]
        new_fitness[:elite_size] = fitness[elite_idx]

        # Generate offspring
        for i in range(elite_size, pop_size):
            # tournament selection (size 3)
            t1 = rng.choice(pop_size, 3, replace=False)
            p1_idx = t1[np.argmin(fitness[t1])]
            t2 = rng.choice(pop_size, 3, replace=False)
            p2_idx = t2[np.argmin(fitness[t2])]

            child = _crossover_and_mutate(pop[p1_idx], pop[p2_idx], target_ones, rng)
            c5 = compute_c5(child)
            new_pop[i] = child
            new_fitness[i] = c5

            # Update global best
            if c5 < best_c5:
                best_c5 = float(c5)
                best_h = child.astype(np.float64)
                if best_c5 <= 0.3808:
                    ga_deadline = time.time()  # break outer loop early
        pop = new_pop
        fitness = new_fitness

    # ---------- Continuous refinement ----------
    # Use the best binary solution as starting point (may be None if initial provided)
    if best_h is None:
        # fall back to a feasible random start
        cur_h = rng.random(n)
        cur_h = project_to_box_sum(cur_h, target_sum)
    else:
        cur_h = best_h.copy()
    cur_c5 = compute_c5(cur_h)

    # keep a record of the best encountered during refinement
    best_h_ref = cur_h.copy()
    best_c5_ref = cur_c5

    step_base = 0.5
    ref_deadline = start_time + budget_s - 0.5
    while time.time() < ref_deadline:
        # random zero‑mean direction
        d = rng.normal(size=n)
        d -= d.mean()

        cand = cur_h + step * d
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_to_box_sum(cand, target_sum)

        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5_ref - 1e-12:
                best_h_ref = cur_h.copy()
                best_c5_ref = cur_c5
                if best_c5_ref <= 0.3808:
                    break
        # occasional step size reduction
        if rng.random() < 0.001:
            step_base *= 0.9

    # Final safety projection
    best_h = best_h_ref
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), n
