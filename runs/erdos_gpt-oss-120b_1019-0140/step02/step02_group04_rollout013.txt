import time
import numpy as np

def compute_c5_vec(h, dx):
    """Return max overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target_sum):
    """Project vec onto the simplex {x in [0,1]^n, sum(x)=target_sum}."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # distribute uniformly, then fix any remaining overflow/underflow
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec

def generate_random_feasible(n, target_sum, rng):
    """Generate a random vector in [0,1]^n with exact sum target_sum."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    # fix rounding errors
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    # final projection ensures exact sum
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # --- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None
    target_sum = n / 2.0
    dx = 2.0 / n

    # --- initial feasible vector ---------------------------------------
    if h0 is not None:
        cur_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        cur_h = generate_random_feasible(n, target_sum, rng)
    cur_val = compute_c5_vec(cur_h, dx)
    best_h, best_val = cur_h.copy(), cur_val

    # --- quick greedy refinement (few seconds) ------------------------
    early_deadline = start_time + budget_s * 0.05
    cur_h, cur_val = greedy_descent(cur_h, dx, target_sum, rng, early_deadline)
    if cur_val < best_val:
        best_h, best_val = cur_h.copy(), cur_val

    # --- simulated annealing -----------------------------------------
    sa_end = start_time + budget_s - 8.0  # reserve time for final phases
    while time.time() < sa_end:
        # compute current worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        # decide move type
        if rng.random() < 0.75:
            # targeted move using bad contributions
            bad = compute_bad(cur_h, k)
            donors = np.where(cur_h > 1e-12)[0]
            recips = np.where(cur_h < 1.0 - 1e-12)[0]
            if donors.size == 0 or recips.size == 0:
                continue
            donor = donors[np.argmax(bad[donors])]
            recip = recips[np.argmin(bad[recips])]
        else:
            # pure random pair
            donors = np.where(cur_h > 1e-12)[0]
            recips = np.where(cur_h < 1.0 - 1e-12)[0]
            if donors.size == 0 or recips.size == 0:
                continue
            donor = rng.choice(donors)
            recip = rng.choice(recips)

        max_delta = min(0.2, cur_h[donor], 1.0 - cur_h[recip])
        if max_delta <= 0:
            continue
        delta = rng.uniform(0.0, max_delta)
        new_h = cur_h.copy()
        new_h[donor] -= delta
        new_h[recip] += delta
        new_val = compute_c5_vec(new_h, dx)

        # temperature schedule
        progress = (time.time() - start_time) / (budget_s - 8.0)
        T = (cur_val) * (0.5 * (1.0 - progress) + 1e-4)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
                if best_val < 0.3808:
                    break

    # --- optional binary hill‑climbing --------------------------------
    time_left = budget_s - (time.time() - start_time)
    if time_left > 12.0:
        # round to binary by selecting largest entries
        binary_h = np.zeros(n, dtype=float)
        top_idx = np.argpartition(-best_h, int(target_sum))[:int(target_sum)]
        binary_h[top_idx] = 1.0
        bin_val = compute_c5_vec(binary_h, dx)

        # swaps
        hill_end = time.time() + time_left - 5.0
        cur_bin_h, cur_bin_val = binary_h.copy(), bin_val
        while time.time() < hill_end:
            ones = np.flatnonzero(cur_bin_h)
            zeros = np.flatnonzero(1 - cur_bin_h)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cand = cur_bin_h.copy()
            cand[i] = 0.0
            cand[j] = 1.0
            cand_val = compute_c5_vec(cand, dx)
            if cand_val < cur_bin_val:
                cur_bin_h, cur_bin_val = cand, cand_val
        if cur_bin_val < best_val:
            best_h, best_val = cur_bin_h, cur_bin_val

    # --- final safety projection ---------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
