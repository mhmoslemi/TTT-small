                elif diff < 0 and v[i] > 0.0:
                    delta = min(-diff, v[i])
                    v[i] -= delta
                    diff += delta
    return v

def deterministic_improve(h, deadline, rng):
    n = h.shape[0]
    dx = 2.0 / n
    best_val = compute_c5(h)
    no_improve = 0
    max_no_improve = 2000
    ar = np.arange(n)
    while time.time() < deadline - 0.01:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * dx
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        k = max_idx - (n - 1)
        if k == 0:
            ones = np.where(h > 1e-12)[0]
            zeros = np.where(h < 1.0 - 1e-12)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            delta = min(h[i], 1.0 - h[j]) * rng.random()
            if delta <= 0:
                continue
            h[i] -= delta
            h[j] += delta
            continue
        if k > 0:
            i_range = np.arange(0, n - k)
            p = h[i_range] * (1.0 - h[i_range + k])
            i_local = np.argmax(p)
            i = i_range[i_local]
            j_cand = np.where((h < 1.0 - 1e-12) & (ar >= n - k))[0]
            if j_cand.size == 0:
                j_cand = np.where(h < 1.0 - 1e-12)[0]
            j = rng.choice(j_cand)
        else:
            kk = -k
            i_range = np.arange(kk, n)
            p = h[i_range] * (1.0 - h[i_range - kk])
            i_local = np.argmax(p)
            i = i_range[i_local]
            j_cand = np.where((h < 1.0 - 1e-12) & (ar < kk))[0]
            if j_cand.size == 0:
                j_cand = np.where(h < 1.0 - 1e-12)[0]
            j = rng.choice(j_cand)
        delta = min(h[i], 1.0 - h[j]) * rng.random()
        if delta <= 0:
            continue
        h[i] -= delta
        h[j] += delta
    return h

def simulated_anneal(h, cur_best_val, start, deadline, rng):
    n = h.shape[0]
    cur_val = cur_best_val
    best_h = h.copy()
    best_val = cur_best_val
    while time.time() < deadline - 0.1:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random()
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val <= 0.3808:
                    break
        else:
            T = max((deadline - time.time()) * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    n = 200
    candidates = []

    if 'initial_h_values' in globals():
        v0 = np.array(initial_h_values, dtype=float)
        n0 = v0.shape[0]
        if n0 != n:
            factor = n // n0
            extra = n - factor * n0
            v_rep = np.repeat(v0, factor)
            if extra > 0:
                v_rep = np.concatenate([v_rep, v0[:extra]])
            v0 = v_rep
        candidates.append(v0)

    best_h = None
    best_val = np.inf

    for cand in candidates:
        h = np.clip(cand, 0.0, 1.0)
        target = n / 2.0
        cur = h.sum()
        if abs(cur - target) > 1e-9:
            h = np.clip(h + (target - cur) / n, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                idxs = np.arange(n)
                rng.shuffle(idxs)
                for i in idxs:
                    if abs(diff) < 1e-12:
                        break
                    if diff > 0 and h[i] < 1.0:
                        inc = min(diff, 1.0 - h[i])
                        h[i] += inc
                        diff -= inc
                    elif diff < 0 and h[i] > 0.0:
                        dec = min(-diff, h[i])
                        h[i] -= dec
                        diff += dec
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            if best_val <= 0.3808:
                return best_h.astype(float), best_val, n

    restarts = 20
    for _ in range(restarts):
        if time.time() > deadline - 5:
            break
        h = random_start_vector(n, rng)
        h = deterministic_improve(h, deadline, rng)
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            if best_val <= 0.3808:
                break
        if time.time() < deadline - 1:
            best_h, best_val = simulated_anneal(best_h, best_val, start, deadline, rng)
            if best_val <= 0.3808:
                break

    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h = np.clip(best_h + (target - cur) / n, 0.0, 1.0)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
