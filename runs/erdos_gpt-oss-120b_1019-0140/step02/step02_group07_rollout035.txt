def project_to_bounds_and_sum(x, target, tol=1e-9, max_iter=100):
    # Clip first? Actually we incorporate box constraints directly via proj(x_i+λ).
    # start with λ low=-1, high=1
    low = -1.0
    high = 1.0
    for iter in range(max_iter):
        mid = (low + high) / 2
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            high = mid
        else:
            low = mid
    # return best approx
    return np.clip(x + (low + high)/2, 0.0, 1.0)

import time
import math
import numpy as np

def compute_c5(arr):
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    # generate Thue-Morse sequence of length n (binary 0/1)
    # using parity of bits count
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    # generate Rudin-Shapiro sequence: start with +1, produce sequence of +/-1, map to 0/1
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return ((a + 1) // 2)

def stochastic_targeted_hill_climb(start_h, target_sum, time_end, rng):
    """Hill climbing that targets the lag with max overlap, with occasional random moves."""
    h = start_h.copy()
    n = len(h)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    while time.time() < time_end and no_improve < 3000:
        # find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        # compute score p_i based on lag
        p = np.zeros(n)
        if lag >= 0:
            # forward shift
            upto = n - lag
            p[:upto] += h[lag:]          # h[i+lag]
            p[lag:] += h[:upto]          # h[i-lag]
        else:
            l = -lag
            upto = n - l
            p[l:] += h[:upto]            # h[i-lag]
            p[:upto] += h[l:]            # h[i+lag]
        # occasionally random move
        if rng.random() < 0.15:
            inc_candidates = np.where(h < 1 - 1e-12)[0]
            dec_candidates = np.where(h > 1e-12)[0]
            if inc_candidates.size == 0 or dec_candidates.size == 0:
                break
            i = rng.choice(inc_candidates)
            j = rng.choice(dec_candidates)
        else:
            inc_candidates = np.where(h < 1 - 1e-12)[0]
            dec_candidates = np.where(h > 1e-12)[0]
            if inc_candidates.size == 0 or dec_candidates.size == 0:
                break
            i = inc_candidates[np.argmax(p[inc_candidates])]
            j = dec_candidates[np.argmin(p[dec_candidates])]
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            no_improve += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        if new_c5 + 1e-12 < cur_c5:
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def binary_hill_climb(start_bin, time_end, rng):

def compute_c5(arr):
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)

def rudin_shapiro_seq(n):
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def stochastic_targeted_hill_climb(start_h, target_sum, time_end, rng):
    """Hill climbing that focuses on the worst lag, with occasional random swaps."""
    h = start_h.copy()

            p[lag:] += h[:upto]

def binary_hill_climb(start_bin, time_end, rng):
    """Swap-based hill climbing on a binary vector."""
    h = start_bin.copy()

    cur_c5 = compute_c5(h.astype(float))
    best_c5, best_h = cur_c5, h.copy()
    no_improve = 0
    while time.time() < time_end and no_improve < 3000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
