def gradient_of_max_shift(h):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_val = corr.max()
    eps = 1e-9
    active_idxs = np.where(corr >= max_val - eps)[0]
    grad = np.zeros_like(h)
    for idx in active_idxs:
        k = idx - (n - 1)  # shift k
        if k >= 0:
            # overlapping region length L = n - k
            # contribution from terms h[i] * (1 - h[i + k])
            # gradient w.r.t. h[i] = (1 - h[i + k]) for i=0..L-1
            # gradient w.r.t. h[i + k] = -h[i] for i=0..L-1
            L = n - k
            # vectorized:
            grad[:L] += one_minus_h[k:k+L]   # (1-h[i+k]) for i from 0..L-1
            grad[k:k+L] -= h[:L]           # -h[i] for those positions
        else:
            # k < 0, let kk = -k
            kk = -k
            L = n - kk
            # i from 0..L-1 with shift negative: h[i+kk] * (1 - h[i])
            # gradient w.r.t. h[i+kk] = (1 - h[i])
            # gradient w.r.t. h[i] = -h[i+kk]
            grad[kk:kk+L] += one_minus_h[:L]  # (1 - h[i])
            grad[:L] -= h[kk:kk+L]           # -h[i+kk]
    # Since objective is max_corr*dx, gradient scaling factor dx
    dx = 2.0 / n
    grad *= dx
    return grad
```

def slsqp_refine(h, maxiter=200):
    n = len(h)
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    # variable t
    x0 = np.concatenate([h, [compute_c5(h)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if res.success:
        h_opt = np.clip(res.x[:n], 0., 1.)
        h_opt = project_to_box_sum(h_opt, n/2)
        return h_opt, compute_c5(h_opt)
    else:
        return h, compute_c5(h)

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- Utility functions -------------------

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target via a binary search."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)


def gradient_of_max_shift(h):
    """Subgradient of max_k overlap (before scaling by dx)."""
    n = h.shape[0]
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_val = corr.max()
    eps = 1e-12
    active = np.where(corr >= max_val - eps)[0]
    grad = np.zeros_like(h)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            L = n - k
            grad[:L] += one_minus_h[k:k + L]
            grad[k:k + L] -= h[:L]
        else:
            kk = -k
            L = n - kk
            grad[kk:kk + L] += one_minus_h[:L]
            grad[:L] -= h[kk:kk + L]
    grad *= (2.0 / n)  # scale by dx
    return grad


# ------------------- Constraint classes -------------------

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx


def objective_t(x):
    """Objective for SLSQP: minimise t (last variable)."""
    return x[-1]


def slsqp_refine(h, maxiter=200):
    """Local refinement via SLSQP on the max-overlap formulation."""
    n = h.shape[0]
    target = n / 2.0
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    res = minimize(objective_t, x0, method='SLSQP',
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-8, 'disp': False})
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, target)
        return h_opt, compute_c5(h_opt)
    else:
        # fallback: return original
        return h, compute_c5(h)

    # ----- Initialise problem size and feasible start -----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 80
        # Triangular shape: peak 1 at centre, linearly to 0 at edges
        i = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        dist = np.abs(i - center)
        h0 = np.maximum(0.0, 1.0 - dist / center)
        # h0 already satisfies sum ≈ n/2, but enforce exactly
    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----- Parameters for the search -----
    base_step = 0.12          # initial step scale
    p_random = 0.22           # probability of random direction move
    safety = 5.0              # seconds reserved for finalisation
    last_improve_time = start_time

    # ----- Main optimisation loop -----
    while time.time() - start_time < budget_s - safety:
        # Choose move type
        if rng.random() < p_random:
            # Random zero‑mean direction
            d = rng.normal(size=n)
            d -= d.mean()
            step = base_step * (rng.random() ** 2)
            cand = best_h + step * d
        else:
            # Gradient descent on the active max shift
            grad = gradient_of_max_shift(best_h)
            step = base_step * (rng.random() ** 2)
            cand = best_h - step * grad

        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)

        # Accept if improvement
        if cand_c5 + 1e-12 < best_c5:
            best_h = cand
            best_c5 = cand_c5
            last_improve_time = time.time()
            # Occasionally attempt a local SLSQP refinement
            if time.time() - start_time < budget_s - safety - 2.0:
                refined_h, refined_c5 = slsqp_refine(best_h, maxiter=150)
                if refined_c5 + 1e-12 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        # Gradual step size reduction
        if rng.random() < 0.015:
            base_step *= 0.88

        # If stagnation for a while, inject a random perturbation
        if time.time() - last_improve_time > 35.0:
            # Restart from current best with a small random jiggle
            jitter = rng.normal(scale=0.04, size=n)
            cand = project_to_box_sum(best_h + jitter, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 + 1e-12 < best_c5:
                best_h = cand
                best_c5 = cand_c5
                last_improve_time = time.time()
            # also shrink step a bit after stagnation
            base_step = max(base_step, 0.02)

    # ----- Final safety projection -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
