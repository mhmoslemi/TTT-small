import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    # Euclidean projection onto box + sum constraint
    # Use method of Lagrange multiplier with binary search (like earlier)
    n=v.size
    if s <= lo*n:
        return np.full(n, lo, dtype=v.dtype)
    if s >= hi*n:
        return np.full(n, hi, dtype=v.dtype)
    # initial bounds for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # define function
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure f(lam_low) >= s >= f(lam_high) (monotonically decreasing in l)
    # Actually f decreasing as l increases, because subtract bigger lambda.
    # We need f(lam_low) >= s >= f(lam_high)
    # Expand if needed
    while f(lam_low) < s:
        lam_low = lam_low - (hi - lo) * 2  # expand lower
    while f(lam_high) > s:
        lam_high = lam_high + (hi - lo) * 2  # expand higher
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def max_shift_gradient(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    # epsilon tolerance
    eps = 1e-12
    idxes = np.where(corr >= max_c - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    # For each shift compute contributions
    for shift in shifts:
        if shift >= 0:
            # term1: (1 - h[i+shift]) for i = 0..n-shift-1

def adam_optimize(h0, max_iter, init_lr, time_left_func):

    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter+1):
        if time_left_func() <= 0.1:
            break
        # compute gradient
        n_corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr_val = np.max(n_corr)
        idxes = np.where(n_corr >= max_corr_val - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto { lo <= x_i <= hi, sum x_i = s } """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure bounds enclose solution
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 2.0
    while f(lam_high) > s:
        lam_high += (hi - lo) * 2.0

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller to a larger array."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Linear interpolation from a larger to a smaller array."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Binary Thue–Morse sequence of length n, values 0.0 or 1.0."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_c5(h):
    """Maximum overlap integral approximated on the current grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of the max‑correlation term for given integer shifts."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # contribution 1 - h[i+shift]
            grad[:n - shift] += 1.0 - h[shift:]
            # contribution -h[i-shift]
            grad[shift:] += -h[:n - shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] += -h[s:]
    grad /= len(shifts)
    return grad


def slsqp_optimize_candidate(h0, max_iter=500):
    """Refine a candidate with SLSQP on the lifted (h, t) formulation."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [
        {'type': 'eq', 'fun': eq_sum},
        {'type': 'ineq', 'fun': ineq_fun}
    ]

    try:
        res = opt.minimize(
            lambda x: x[-1],
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
        )
    except Exception:
        return h0, compute_c5(h0)

    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def adam_optimize(h0, max_iter, init_lr, time_left_func, rng):
    """Adam‑style projected optimization on the max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iter + 1):
        if time_left_func() <= 0.0:
            break

        # correlation and shifts achieving the maximum
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)

        grad = compute_gradients_for_shifts(h, shifts)

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)

        # occasional stochastic perturbation to escape plateaus
        if it % 4000 == 0:
            h_new = project_onto_box_sum(h_new + rng.normal(scale=0.015, size=n), s)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

        init_full = np.asarray(initial_h_values, dtype=float)
        candidates.append(project_onto_box_sum(downsample(init_full, n_small), n_small / 2.0))

    candidates.append(thue_morse(n_small))
    candidates.append(random_binary_balanced(n_small, rng))
    candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    best_h = None
    best_score = np.inf

    for cand in candidates:
        if time_left() < 2.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=400)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # random small restarts while budget permits
    while time_left() > 5.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=250)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5, dtype=float), n_small / 2.0)
        best_score = compute_c5(best_h)

    # ---- Phase 2: Upsample to larger grid ----
    h_large = upsample(best_h, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    best_h = h_large.copy()
    best_score = compute_c5(best_h)

    # ---- Phase 3: Adam‑style refinement on the large grid ----
    remaining = time_left()
    if remaining > 5.0:
        # generous iteration budget scaled to remaining time
        max_iter = int(min(50000, max(0, remaining * 2000)))
        if max_iter >= 100:
            h_opt, sc_opt = adam_optimize(best_h, max_iter, init_lr=0.5,
                                          time_left_func=time_left, rng=rng)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()

    # ---- Phase 4: Final SLSQP polish on the large grid ----
    if time_left() > 5.0:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=300)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # final safeguard projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
