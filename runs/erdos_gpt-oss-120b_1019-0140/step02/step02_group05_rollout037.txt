def max_overlap_gradient(h, dx):
    n = len(h)
    O = np.correlate(h, 1-h, mode='full')
    O = O
    max_val = np.max(O)
    # get indices where O==max
    idxs = np.where(O == max_val)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)  # shift
        if k >= 0:
            # i = 0..n-1-k
            grad[:n-k] += (1 - h[k:]) * dx
            grad[k:] += -h[:n-k] * dx
        else:
            kk = -k
            # i = kk..n-1
            grad[kk:] += (1 - h[:n-kk]) * dx
            grad[:n-kk] += -h[kk:] * dx
    grad /= len(idxs)  # average
    return grad

def project_onto_constraints(h, target):
    h = np.clip(h, 0.0, 1.0)
    current = np.sum(h)
    diff = target - current
    if abs(diff) < 1e-12:
        return h
    # Identify free indices where h not at bound
    free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
    if free.size > 0:
        inc = diff / free.size
        h[free] += inc
        # clip again
        h = np.clip(h, 0.0, 1.0)
        # recalc diff
        diff = target - np.sum(h)
    # If still diff, adjust iteratively across all indices
    if abs(diff) > 1e-12:
        # distribute greedily
        for i in range(len(h)):
            if abs(diff) < 1e-14:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h
```

                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def max_overlap_gradient(h, dx):
    """Subgradient of the maximal overlap O_k (without dx factor)."""
    n = len(h)
    O = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(O)
    idxs = np.where(O == max_val)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)          # shift corresponding to this index
        if k >= 0:
            grad[:n - k] += (1.0 - h[k:]) * dx
            grad[k:]     += -h[:n - k] * dx
        else:
            kk = -k
            grad[kk:]    += (1.0 - h[:n - kk]) * dx
            grad[:n - kk] += -h[kk:] * dx
    grad /= len(idxs)
    return grad

def smooth_and_fix(h, target):
    """One Gaussian smoothing pass with exact-sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    cur = np.sum(sm)
    diff = target - cur
    if abs(diff) > 1e-12:
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            inc = diff / free.size
            sm[free] += inc
            sm = np.clip(sm, 0.0, 1.0)
        diff = target - np.sum(sm)
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
        sm = np.clip(sm, 0.0, 1.0)
    return sm

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- choose discretisation size -----
    n = 120                     # more points give finer control
    dx = 2.0 / n
    target_sum = n / 2.0

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h0_small = np.array(initial_h_values, dtype=float)
        n_small = len(h0_small)
        # linear up‑sampling to the chosen n
        x_small = np.linspace(0.0, 2.0, n_small, endpoint=False)
        x_new   = np.linspace(0.0, 2.0, n,    endpoint=False)
        h = np.interp(x_new, x_small, h0_small)
    else:
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    h = project_onto_constraints(h, target_sum)
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    # ----- subgradient descent on the max overlap -----
    step0 = 0.6
    decay = 2e-4
    max_iter = 300000
    for it in range(max_iter):
        if time.time() - start >= budget_s * 0.75:
            break
        grad = max_overlap_gradient(h, dx)
        step = step0 / (1.0 + it * decay)
        cand = h - step * grad
        cand = project_onto_constraints(cand, target_sum)
        cand_val = compute_c5(cand)
        if cand_val < cur_val:
            h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break
        else:
            # occasional random transfer to escape plateaus
            if rng.random() < 0.001:
                i, j = rng.choice(n, 2, replace=False)
                max_tr = min(h[i], 1.0 - h[j])
                if max_tr <= 0:
                    max_tr = min(h[j], 1.0 - h[i])
                    if max_tr <= 0:
                        continue
                    donor, recv = j, i
                else:
                    donor, recv = i, j
                delta = rng.random() * max_tr * 0.2
                cand2 = h.copy()
                cand2[donor] -= delta
                cand2[recv] += delta
                cand2 = project_onto_constraints(cand2, target_sum)
                val2 = compute_c5(cand2)
                if val2 < cur_val:
                    h = cand2
                    cur_val = val2
                    if val2 < best_val:
                        best_val = val2
                        best_h = cand2.copy()
                        if best_val < 0.3808:
                            break

    # ----- simulated‑annealing refinement -----
    cur_h = best_h.copy()
    cur_val = best_val
    T0 = 0.008
    while time.time() - start < budget_s - 4.0:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)
        i, j = rng.choice(n, 2, replace=False)
        max_tr = min(cur_h[i], 1.0 - cur_h[j])
        if max_tr <= 0:
            max_tr = min(cur_h[j], 1.0 - cur_h[i])
            if max_tr <= 0:
                continue
            donor, recv = j, i
        else:
            donor, recv = i, j
        delta = rng.random() * max_tr * 0.5
        cand = cur_h.copy()
        cand[donor] -= delta
        cand[recv] += delta
        cand = project_onto_constraints(cand, target_sum)
        cand_val = compute_c5(cand)
        if cand_val < cur_val:
            cur_h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break
        elif T > 0:
            prob = np.exp(-(cand_val - cur_val) / T)
            if rng.random() < prob:
                cur_h = cand
                cur_val = cand_val

    # ----- optional smoothing passes -----
    if time.time() - start < budget_s - 2.0:
        for _ in range(5):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
            else:
                break

    # ----- final local refinement with SLSQP (if time permits) -----
    if time.time() - start < budget_s - 6.0:
        # build constraints: one equality for the sum, inequalities for each shift
        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

        def make_ineq(k):
            def fun(x):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[: n - k], 1.0 - h[k:]) * dx
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[: n - kk]) * dx
                return x[-1] - overl
            return fun

        constraints = [{"type": "eq", "fun": sum_eq}]
        for kk in range(-n + 1, n):
            if kk == 0:
                continue
            constraints.append({"type": "ineq", "fun": make_ineq(kk)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                cur = np.sum(h_opt)
                if abs(cur - target_sum) > 1e-9:
                    h_opt += (target_sum - cur) / n
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # ----- final safety projection -----
    best_h = project_onto_constraints(best_h, target_sum)
    best_val = compute_c5(best_h)

    return best_h, best_val, n
