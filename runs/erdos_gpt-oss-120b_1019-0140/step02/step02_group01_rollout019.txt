def anneal_opt(n, rng, time_budget):
    M = n // 2
    # initial h
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        # adjust sum if needed
        cur_sum = h.sum()
        if abs(cur_sum - M) > 1e-9:
            # shift uniformly as before
            h += (M - cur_sum) / n
            np.clip(h, 0, 1, out=h)
    else:
        # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    # optionally add small jitter.
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    start = time.time()
    T = 0.01
    decay = 0.9995
    it = 0
    no_improve = 0
    while time.time() - start < time_budget:
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        # transfer amount
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        # apply move virtually
        h[i] += delta
        h[j] -= delta
        # compute
        new_c5 = compute_c5(h)
        delta_c5 = new_c5 - c5
        accept = False
        if delta_c5 <= 0 or rng.random() < np.exp(-delta_c5 / T):
            accept = True
        if accept:
            c5 = new_c5
            # accept new state: already updated h
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert move
            h[i] -= delta
            h[j] += delta
            # c5 unchanged
            no_improve += 1
        T = max(T * decay, 1e-6)
        it += 1
        # restart if stagnation
        if no_improve > 5000:
            # restart with new random h
            h = np.zeros(n)
            idx = rng.choice(n, M, replace=False)
            h[idx] = 1.0
            c5 = compute_c5(h)
            no_improve = 0
            # reset temperature?
            T = 0.01
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # Candidate n's
    candidate_ns = [80, 96, 112, 128]  # small enough.

    # If parent construction exists, try its size first
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                # ensure ordering first
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    # Allocate time per n gradually: maybe proportionally.
    # We'll loop over candidates, break if overall time near end.
    for n in candidate_ns:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        remaining = budget_s - elapsed
        # allocate at most 300 seconds per n, but also use half of remaining time.
        alloc = min(300.0, remaining * 0.5)

        # Run annealing for this n
        h_opt, c5_opt = anneal_opt(n, rng, alloc)

        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global = h_opt
            best_c5_global = c5_opt
            best_n_global = n

            if best_c5_global < 0.3808:
                break

    # In case nothing found:
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n

    # Final check, ensure feasibility
    n = best_n_global
    M = n // 2
    best_h_global = np.clip(best_h_global, 0., 1.)
    cur = best_h_global.sum()
    if abs(cur - M) > 1e-9:
        # Adjust uniformly and clip to [0,1]
        best_h_global += (M - cur) / n
        best_h_global = np.clip(best_h_global, 0., 1.)
        # If still off, adjust using simple iterative fix
        cur = best_h_global.sum()
        if abs(cur - M) > 1e-9:
            # simple loop: add to those <1, subtract from those >0.
            # Using while loop with tolerance:
            deficit = M - cur
            # If deficit >0 (need increase)
            if deficit > 0:
                # fill up available capacity
                cap = np.sum(1.0 - best_h_global)
                if cap > 0:
                    inc = min(deficit, cap)
                    # distribute proportionally to capacity
                    best_h_global += inc * (1.0 - best_h_global) / cap
            else:
                # need decrease
                excess = -deficit
                cap = np.sum(best_h_global)
                if cap > 0:
                    dec = min(excess, cap)
                    best_h_global -= dec * best_h_global / cap
            # final clip again
            best_h_global = np.clip(best_h_global, 0., 1.)

    return best_h_global, best_c5_global, n

    if method == 0:

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum(h, M):
    """Adjust h to satisfy sum(h)=M while staying in [0,1]."""
    cur = h.sum()
    if abs(cur - M) > 1e-12:
        h += (M - cur) / h.shape[0]
        np.clip(h, 0.0, 1.0, out=h)
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            h += (M - cur) / h.shape[0]
            np.clip(h, 0.0, 1.0, out=h)
    return h

def generate_initial_h(n, M, rng):
    """Return a feasible h (0≤h≤1, sum=M) using a simple random scheme."""
    method = rng.integers(3)
    if method == 0:  # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:  # uniform random then project
        h = rng.random(n)
        h = h / h.sum() * M
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    else:  # linear decreasing shape
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    return enforce_sum(h, M)

def anneal_opt(n, rng, time_budget):
    """Simulated‑annealing search for a low C₅ at resolution n."""
    M = n // 2
    # initial feasible point
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        h = enforce_sum(h, M)
        np.clip(h, 0.0, 1.0, out=h)
    else:
        h = generate_initial_h(n, M, rng)

    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    start = time.time()
    T = 0.01
    decay = 0.9995
    stagn = 0

    while time.time() - start < time_budget:
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] += delta
        h[j] -= delta
        new_c5 = compute_c5(h)
        dC = new_c5 - cur_c5
        if dC <= 0 or rng.random() < np.exp(-dC / T):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                stagn = 0
            else:
                stagn += 1
        else:
            h[i] -= delta
            h[j] += delta
            stagn += 1

        T = max(T * decay, 1e-7)

        if stagn > 8000:
            # restart from a fresh random feasible point
            h = generate_initial_h(n, M, rng)
            cur_c5 = compute_c5(h)
            stagn = 0
            T = 0.01

    return best_h, best_c5

def gradient_refine(h, rng, time_budget):
    """Deterministic refinement using gradient of the currently maximal shift."""
    n = h.shape[0]
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    while time.time() - start < time_budget:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_idx = int(np.argmax(corr))
        k = k_idx - (n - 1)

        grad = np.zeros(n)
        if k >= 0:
            grad[: n - k] += (1.0 - best_h[k:])
            grad[k:] -= best_h[: n - k]
        else:
            kk = -k
            grad[kk:] += (1.0 - best_h[: n - kk])
            grad[: n - kk] -= best_h[kk:]

        i = int(np.argmin(grad))
        j = int(np.argmax(grad))
        if i == j:
            # fallback to random pair
            i, j = rng.choice(n, 2, replace=False)

        max_transfer = min(1.0 - best_h[i], best_h[j])
        if max_transfer <= 0:
            # fallback to random pair
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(1.0 - best_h[i], best_h[j])
            if max_transfer <= 0:
                continue

        delta = rng.random() * max_transfer
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            best_h = h_new
            best_c5 = c5_new
            if best_c5 < 0.3808:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # candidate discretisations (even numbers, modest size)
    candidate_ns = [80, 96, 112, 128]

    # prioritize parent size if available
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        alloc = min(250.0, remaining * 0.5)  # time slice for this resolution
        h_opt, c5_opt = anneal_opt(n, rng, alloc)
        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global = h_opt
            best_c5_global = c5_opt
            best_n_global = n
            if best_c5_global < 0.3808:
                break

    # final gradient‑based polish on the best candidate found
    if best_h_global is not None:
        time_left = budget_s - (time.time() - start_time)
        if time_left > 5.0:
            best_h_global, best_c5_global = gradient_refine(best_h_global, rng, time_left)

    # safety fallback
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        best_h_global = enforce_sum(best_h_global, n // 2)

    return best_h_global, best_c5_global, n
