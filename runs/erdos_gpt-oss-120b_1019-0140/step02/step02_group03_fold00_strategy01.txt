        h_fine_f = project_to_unit_sum(h_fine_f)
        # Refine with local search

            local_result = minimize(fine_objective, h_fine_f, method='Nelder-Mead',

            h_final = local_result.x
        except:

        overlap_result = compute_fine_overlap(h_final)
        return h_final, overlap_result
