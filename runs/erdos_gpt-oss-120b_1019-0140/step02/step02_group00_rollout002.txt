def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = target_sum}."""
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)

    # binary search on lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # enlarge interval if needed
    if f(lam_low) < target_sum:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target_sum:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Approximate C5 on a uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation to larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Linear interpolation to smaller size."""
    return upsample(arr, new_n)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def thue_morse_binary(n):
    """Thue–Morse binary sequence (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max correlation term averaged over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def slsqp_optimize_candidate(h0, max_iter=500):
    """Run a short SLSQP to improve a candidate."""
    n = h0.shape[0]
    target_sum = n / 2.0
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - target_sum

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr  # elementwise >=0

    cons = [
        {'type': 'eq', 'fun': eq_sum},
        {'type': 'ineq', 'fun': ineq_fun}
    ]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, target_sum)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    n = 256
    target_sum = n / 2.0
    best_h = None
    best_score = float('inf')

    def consider(h):
        nonlocal best_h, best_score
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # ----- Seed candidates -----
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n:
            if init_full.size < n:
                h0 = upsample(init_full, n)
            else:
                h0 = downsample(init_full, n)
        else:
            h0 = init_full.copy()
        h0 = project_onto_box_sum(h0, target_sum)
        consider(h0)

        # Binary version of the initial
        binary = (h0 >= 0.5).astype(float)
        cur_ones = int(binary.sum())
        need = int(target_sum) - cur_ones
        if need > 0:
            zeros = np.where(binary == 0)[0]
            flip = rng.choice(zeros, size=need, replace=False)
            binary[flip] = 1.0
        elif need < 0:
            ones = np.where(binary == 1)[0]
            flip = rng.choice(ones, size=-need, replace=False)
            binary[flip] = 0.0
        consider(binary)

    # Thue‑Morse binary
    consider(thue_morse_binary(n))

    # ----- Random binary hill‑climbing restarts -----
    while time_left() > 5.0:
        h_cur = random_binary_balanced(n, rng)
        cur_score = compute_c5(h_cur)

        # allocate a short time slice for this climb
        climb_budget = min(2.0, time_left() - 3.0)
        climb_start = time.time()
        local_best_h = h_cur
        local_best_sc = cur_score

        while time.time() - climb_start < climb_budget:
            # pick a random 0 and a random 1 to swap
            i = rng.integers(n)
            vi = h_cur[i]
            j = rng.integers(n)
            while h_cur[j] == vi:
                j = rng.integers(n)
            # swap
            h_new = h_cur.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            new_sc = compute_c5(h_new)

            if new_sc < cur_score:
                h_cur = h_new
                cur_score = new_sc
                if new_sc < local_best_sc:
                    local_best_sc = new_sc
                    local_best_h = h_new.copy()
            else:
                # occasional uphill move (simulated annealing)
                elapsed = time.time() - climb_start
                frac = elapsed / climb_budget
                T = 0.01 * (1.0 - frac)
                if T > 0 and rng.random() < np.exp(-(new_sc - cur_score) / T):
                    h_cur = h_new
                    cur_score = new_sc

        consider(local_best_h)
        if best_score <= 0.3808:
            break

    # ----- Continuous refinement via SLSQP -----
    if best_h is not None and time_left() > 10.0:
        h_float = best_h.astype(float).copy()
        h_opt, sc_opt = slsqp_optimize_candidate(h_float, max_iter=400)
        if sc_opt < best_score:
            best_h = h_opt
            best_score = sc_opt

    # ----- Momentum gradient descent -----
    if best_h is not None and time_left() > 2.0:
        remaining = time_left()
        # Rough iteration budget: ~0.0005 s per iteration on 256‑dim problem
        est_iters = int(min(20000, max(5000, remaining * 1500)))
        h_ref, sc_ref = gradient_descent_momentum(best_h.astype(float), n_iter=est_iters,
                                                  step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_h = h_ref
            best_score = sc_ref

    # final projection and score
    if best_h is None:
        best_h = random_binary_balanced(n, rng).astype(float)
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
