def project_to_bounds_and_sum(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    # iterative distribution
    # Keep adjusting entries that are not at bounds
    while True:
        s = np.sum(x)
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (x > 1e-12) & (x < 1 - 1e-12)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            # if diff still not zero, adjust all equally ignoring bounds (small violation)
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
            break
        adjustment = diff / nfree
        # apply adjustment to free entries
        x[free] += adjustment
        # clip again
        x = np.clip(x, 0.0, 1.0)
    return x

def project_capped_simplex(y, target_sum):
    # goal: find x = argmin ||x - y||^2 s.t. sum(x) = target_sum, 0 <= x <= 1
    # solve via projection with Lagrange multiplier λ, x_i = clip(y_i - λ, 0, 1)
    # Use bisection for λ.
    lo = -1e6
    hi = 1e6
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # small adjustment for numeric error
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        # distribute diff uniformly across entries not at bounds
        free = (x > 1e-12) & (x < 1 - 1e-12)
        nfree = np.count_nonzero(free)
        if nfree > 0:
            x[free] += diff / nfree
            x = np.clip(x, 0.0, 1.0)
    return x

import time
import math
import numpy as np

def compute_c5(arr):
    """Approximate C5 via full correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def project_capped_simplex(y, target_sum):
    """
    Project y onto the intersection of the box [0,1]^n and the hyperplane sum(x)=target_sum.
    Solves: min||x-y||^2 s.t. 0<=x_i<=1, sum(x)=target_sum.
    """
    lo, hi = -1e6, 1e6
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        nfree = np.count_nonzero(free)
        if nfree:
            x[free] += diff / nfree
            x = np.clip(x, 0.0, 1.0)
    return x

def binary_hill_climb(start_bin, time_deadline, rng):
    """
    Simple hill climbing on binary vectors with exact half‑ones constraint.
    Swaps one 1 and one 0 if the move improves C5.
    """
    n = len(start_bin)
    h = start_bin.copy()
    best_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    no_improve = 0
    while time.time() < time_deadline and no_improve < 3000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5

def continuous_anneal(start_h, time_end, rng):
    """
    Simulated annealing on the continuous feasible set.
    Maintains the exact sum by swapping mass between two coordinates.
    """
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    init_temp = 1e-2
    final_temp = 1e-7
    max_iter = 300000
    it = 0
    while time.time() < time_end and it < max_iter:
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5

        it += 1

        best_h += diff / n

        for i in range(n):

                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_c5
