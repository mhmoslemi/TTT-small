def targeted_move(cur_h, k, step, rng):

    n = len(cur_h)

    if k >=0:

        max_i = n - k - 1

        if max_i < 0: return None

        donors_idx = np.where(cur_h[:n-k] > 0)[0]

        if donors_idx.size ==0: return None

        i = rng.choice(donors_idx)

import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5(h, dx):
    """Maximum overlap integral C5 for step values h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Adjust vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # distribute uniformly, then fix leftovers
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible vector (values in [0,1] with exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def random_move(cur_h, step, rng):
    """Transfer a random amount of mass from a donor to a recipient."""
    donors = np.where(cur_h > 1e-12)[0]
    recipients = np.where(cur_h < 1.0 - 1e-12)[0]
    if donors.size == 0 or recipients.size == 0:
        return None
    i = rng.choice(donors)
    j = rng.choice(recipients)
    max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
    if max_delta <= 1e-14:
        return None
    delta = rng.uniform(0.0, max_delta)
    new_h = cur_h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def targeted_move(cur_h, k, step, rng):
    """Transfer mass along the worst shift direction to reduce its overlap."""
    n = len(cur_h)
    if k >= 0:
        if k >= n:
            return None
        donors = np.where(cur_h[: n - k] > 1e-12)[0]
        if donors.size == 0:
            return None
        i = rng.choice(donors)
        j = i + k
        if cur_h[j] >= 1.0 - 1e-12:
            return None
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
    else:
        shift = -k
        if shift >= n:
            return None
        donors = np.where(cur_h[shift:] > 1e-12)[0] + shift
        if donors.size == 0:
            return None
        i = rng.choice(donors)
        j = i + k  # i - shift
        if cur_h[j] >= 1.0 - 1e-12:
            return None
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
    if max_delta <= 1e-14:
        return None
    delta = rng.uniform(0.0, max_delta)
    new_h = cur_h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    dx = 2.0 / n
    target_sum = n / 2.0

    # initial feasible solution
    if h0 is not None:
        cur_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        cur_h = generate_random_feasible(n, target_sum, rng)

    cur_val = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val

    # parameters for the stochastic search
    step = 0.2                     # initial mass-transfer step size
    min_step = 0.001
    step_decay = 0.9995
    temp_initial = 0.006           # initial temperature for Metropolis acceptance
    target_ratio = 0.98            # leave a small margin before hard deadline

    # main simulated‑annealing loop
    while time.time() - start_time < budget_s * target_ratio:
        # current worst shift
        corr_raw = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        k_idx = int(np.argmax(corr_raw))
        k = k_idx - (n - 1)            # shift corresponding to max overlap
        # propose a move
        if rng.random() < 0.7:
            cand = targeted_move(cur_h, k, step, rng)
            if cand is None:
                cand = random_move(cur_h, step, rng)
        else:
            cand = random_move(cur_h, step, rng)

        if cand is None:
            # nothing feasible; just decay step and continue
            step = max(min_step, step * step_decay)
            continue

        cand_val = compute_c5(cand, dx)

        # annealing acceptance
        improve = cand_val < cur_val - 1e-12
        elapsed = time.time() - start_time
        temp = max(1e-12, temp_initial * (1.0 - elapsed / budget_s))
        if improve:
            accept = True
        else:
            prob = np.exp(-(cand_val - cur_val) / temp)
            accept = rng.random() < prob

        if accept:
            cur_h = cand
            cur_val = cand_val
            # adapt step size slightly upwards when we improve
            step = min(0.5, step * 1.01)
        else:
            # decay step size on rejection
            step = max(min_step, step * step_decay)

        # track global best
        if cur_val < best_val - 1e-12:
            best_h = cur_h.copy()
            best_val = cur_val
            # early exit if we beat the target bound
            if best_val < 0.3808:
                break

    # final safety projection and re‑evaluation
    best_h = project_to_sum(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_val = compute_c5(best_h, dx)

    return best_h, best_val, n
