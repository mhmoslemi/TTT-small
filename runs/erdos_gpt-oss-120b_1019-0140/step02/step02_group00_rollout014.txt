def upsample(arr, new_n):
    """Linearly interpolate arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C₅ = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def sinusoid_candidate(n, rng):
    """Small sinusoidal perturbation, then projected onto constraints."""
    freq = rng.integers(1, max(2, n // 4))
    amp = rng.uniform(0.1, 0.5)
    t = np.arange(n) / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * t)
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, n / 2.0)
    return h


def gradient_for_max_shifts(h, shifts):
    """Gradient of max‑correlation term w.r.t. h (sum of sub‑gradients)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            # forward contribution
            grad[:n - s] -= h[s:]
            # backward contribution
            grad[s:] -= h[:n - s]
        else:
            s2 = -s
            grad[s2:] -= h[:n - s2]
            grad[:n - s2] -= h[s2:]
    grad += len(shifts)  # the constant +1 term for each shift
    return grad


def refine_candidate(h0, max_iters, n, target_sum):
    """Projected sub‑gradient descent with momentum."""
    h = h0.copy()
    momentum = np.zeros_like(h)
    beta = 0.9
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iters + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = gradient_for_max_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = 0.5 / np.sqrt(it)
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512                     # target resolution (must be < 1000 for speed)
    target_sum = n / 2.0

    # ---------- build initial candidate pool ----------
    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init, n)
        init_up = project_onto_box_sum(init_up, target_sum)
        candidates.append(init_up)

    # Thue–Morse binary pattern
    tm = thue_morse(n).astype(float)
    candidates.append(project_onto_box_sum(tm, target_sum))

    # Random continuous vectors
    for _ in range(4):
        v = rng.random(n)
        v = project_onto_box_sum(v, target_sum)
        candidates.append(v)

    # Random binary balanced vectors
    for _ in range(4):
        candidates.append(random_binary_balanced(n, rng))

    # Sinusoidal‑based candidates
    for _ in range(3):
        candidates.append(sinusoid_candidate(n, rng))

    best_h = None
    best_score = float('inf')

    # ---------- refine each candidate ----------
    for cand in candidates:
        remaining = budget_s - (time.time() - start)
        if remaining < 5.0:
            break
        max_iters = max(500, min(5000, int(remaining * 4000)))
        h_ref, sc_ref = refine_candidate(cand, max_iters, n, target_sum)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # ---------- local perturbation search ----------
    while True:
        remaining = budget_s - (time.time() - start)
        if remaining < 2.0:
            break
        if best_h is None:
            h0 = project_onto_box_sum(rng.random(n), target_sum)
        else:
            h0 = best_h.copy()

        i, j = rng.choice(n, size=2, replace=False)
        eps = rng.uniform(0.0, 0.02)
        if h0[i] - eps >= 0.0 and h0[j] + eps <= 1.0:
            h0[i] -= eps
            h0[j] += eps
        else:
            h0 = h0 + rng.normal(scale=0.01, size=n)
        h0 = project_onto_box_sum(h0, target_sum)

        max_iters = max(200, int(min(2000, remaining * 2500)))
        h_ref, sc_ref = refine_candidate(h0, max_iters, n, target_sum)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # ---------- final safeguard ----------
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), target_sum)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, target_sum)
        best_score = compute_c5(best_h)

    return best_h, best_score, n
