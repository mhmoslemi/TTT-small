def mismatched_one_indices(h, lag):
    n = h.shape[0]
    if lag >= 0:
        # indices where i+lag >= n -> out-of-range => mismatch if h[i]==1
        out_start = n - lag
        out = np.arange(out_start, n) if lag > 0 else np.arange(0, n)  # if lag=0, out_start=n -> empty; but lag=0 mismatches always zero because (1 - h_i) equals 0 when h_i=1
        # out mismatch condition:
        out_mis = out[h[out]==1]   # indices of ones in out region
        
        # range indices
        rng = np.arange(0, n - lag)
        if rng.size > 0:
            # mismatches where h[i]==1 and h[i+lag]==0
            mis_mask = (h[rng] == 1) & (h[rng + lag] == 0)
            rng_mis = rng[mis_mask]
        else:
            rng_mis = np.array([], dtype=int)
        return np.concatenate((out_mis, rng_mis))
    else:
        d = -lag
        # out-of-range left side: i < d
        out = np.arange(0, d)
        out_mis = out[h[out]==1]
        # range: i >= d
        rng = np.arange(d, n)
        if rng.size > 0:
            mis_mask = (h[rng] == 1) & (h[rng - d] == 0)
            rng_mis = rng[mis_mask]
        else:
            rng_mis = np.array([], dtype=int)
        return np.concatenate((out_mis, rng_mis))

def subgradient_step(h, target_sum, step):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    lag = np.argmax(corr) - (n - 1)
    g = np.ones(n)
    if lag >= 0:
        if lag < n:
            g[:n - lag] -= h[lag:]
            g[lag:] -= h[:n - lag]
    else:
        d = -lag
        g[d:] -= h[:n - d]
        g[:n - d] -= h[d:]
    h_new = h - step * g
    h_new = np.clip(h_new, 0.0, 1.0)

    h_new = make_feasible(h_new, target)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def make_feasible(arr, target):
    arr = np.clip(arr, 0.0, 1.0)
    diff = target - arr.sum()
    if abs(diff) > 1e-12:
        arr = arr + diff / arr.shape[0]
        arr = np.clip(arr, 0.0, 1.0)
        residual = target - arr.sum()
        if abs(residual) > 1e-12:
            for i in range(arr.shape[0]):
                if abs(residual) < 1e-12:
                    break
                adj = np.clip(residual, -arr[i], 1 - arr[i])
                arr[i] += adj
                residual = target - arr.sum()
    return arr

def binary_hill_climb(start_bin, end_time, rng):
    n = start_bin.shape[0]
    bin_vec = start_bin.copy()
    best_h = bin_vec.astype(float)
    best_c5 = compute_c5(best_h)
    no_imp = 0
    while time.time() < end_time and no_imp < 2000:
        ones = np.flatnonzero(bin_vec)
        zeros = np.flatnonzero(1 - bin_vec)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = bin_vec.copy()
        cand[i] = 0
        cand[j] = 1
        cand_h = cand.astype(float)
        cand_c5 = compute_c5(cand_h)
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5

def subgradient_step(h, target, step):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    lag = np.argmax(corr) - (n - 1)

            g[lag:] -= h[:n - lag]

        g[d:] -= h[:n - d]
        g[:n - d] -= h[d:]
    h_new = h - step * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, target)
    return h_new
