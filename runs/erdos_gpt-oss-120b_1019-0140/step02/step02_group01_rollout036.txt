def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    v = 1.0 - h
    # reversed v for correlation
    vr = v[::-1]
    L = 1
    while L < 2*n:
        L <<= 1
    # compute rfft
    H = np.fft.rfft(h, n=L)
    V = np.fft.rfft(vr, n=L)
    corr = np.fft.irfft(H * V, n=L)
    corr = corr[:2*n-1]  # relevant part
    return np.max(corr) * dx

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------

def compute_c5(h):
    """Maximum overlap integral for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    v = 1.0 - h
    # correlation via FFT (convolution with reversed v)
    vr = v[::-1]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=L)
    V = np.fft.rfft(vr, n=L)
    corr = np.fft.irfft(H * V, n=L)
    corr = corr[: 2 * n - 1]          # keep the relevant part
    return np.max(corr) * dx


def generate_random_feasible_h(n, rng):
    """Binary feasible vector (0/1) with exactly n/2 ones."""
    m = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, m, replace=False)
    h[idx] = 1.0
    return h


def project_to_sum(h, target):
    """Clip to [0,1] and adjust to achieve exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase some entries
        order = np.argsort(h)                # smallest first
        for i in order:
            inc = min(diff, 1.0 - h[i])
            if inc <= 0:
                continue
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # need to decrease some entries
        order = np.argsort(-h)               # largest first
        for i in order:
            dec = min(-diff, h[i])
            if dec <= 0:
                continue
            h[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return h


def anneal_h(start_h, total_time, rng):
    """Simulated annealing on the mass‑transfer space."""
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5

    t0 = time.time()
    deadline = t0 + total_time
    T0 = 0.02                      # initial temperature

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - t0
        frac = elapsed / total_time
        T = T0 * (1.0 - frac)

        # propose a transfer of mass from j to i
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(1.0 - cur[i], cur[j])
        if max_delta <= 0.0:
            continue

        # sometimes move all possible mass, sometimes a random fraction
        if rng.random() < 0.2:
            delta = max_delta
        else:
            delta = rng.random() * max_delta

        cur[i] += delta
        cur[j] -= delta

        new_c5 = compute_c5(cur)
        dC = new_c5 - cur_c5

        if dC < 0.0:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            # accept with probability exp(-dC/T)
            if T > 0.0 and rng.random() < np.exp(-dC / (T + 1e-16)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best = cur.copy()
            else:
                # revert move
                cur[i] -= delta
                cur[j] += delta

    return best, best_c5
