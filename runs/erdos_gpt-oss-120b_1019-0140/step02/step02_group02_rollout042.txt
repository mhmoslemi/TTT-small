def project_onto_feasible(h, target):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust to match sum
    max_iter = 10
    for _ in range(max_iter):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if len(idx) == 0:
                break
            inc = diff / len(idx)
            # ensure not exceed 1
            incs = np.minimum(inc, 1.0 - h[idx])
            h[idx] += incs
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            dec = (-diff) / len(idx)
            decs = np.minimum(dec, h[idx])
            h[idx] -= decs
    # final clip again
    h = np.clip(h, 0.0, 1.0)
    return h
```

### Projection onto box & sum hyperplane

Problem: minimize ||x - v||^2 subject to 0 ≤ x ≤ 1 and sum x = target.

Standard solution: Use water-filling algorithm with clipping at bounds. Implementation:

1. Start with v = np.clip(v, 0, 1).
2. Compute diff = target - sum(v). While |diff| > tolerance:
   - Determine set of free variables where 0 < v_i < 1 (or tolerance).
   - If diff >0, need to increase; add delta = diff / len(free) to each free variable, but limited by upper bound. So for each i, inc = min(delta, 1 - v_i). For those hitting bound, remove from free set.
   - If diff <0, similarly decrease with lower bound.

We can implement iterative loop.

Implementation:

```
def project_onto_feasible(v, target, tol=1e-12, max_iter=1000):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) <= tol:
        return v
    for _ in range(max_iter):
        # identify free indices where adjusting won't violate bounds
        if diff > 0:
            free = np.where(v < 1.0 - tol)[0]
            if len(free) == 0:
                break
            inc = diff / len(free)
            # cap inc per index
            incs = np.minimum(inc, 1.0 - v[free])
            v[free] += incs
        else:
            free = np.where(v > tol)[0]
            if len(free) == 0:
                break
            dec = (-diff) / len(free)
            decs = np.minimum(dec, v[free])
            v[free] -= decs
        diff = target - v.sum()
        if abs(diff) <= tol:
            break
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v
```

import time
import numpy as np
from scipy.optimize import minimize

def _dx(n):
    return 2.0 / n

def compute_c5(h):
    n = h.shape[0]
    dx = _dx(n)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def compute_C_array(h):
    n = h.shape[0]
    dx = _dx(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    C = corr * dx
    return C[n:]  # shifts k=1..n-1

def gradient_Ck(h, k):
    """gradient of C(k) = dx*sum_i h_i(1-h_{i+k}) w.r.t. h."""
    n = h.shape[0]
    g = np.ones(n)
    if k < n:
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    return g * _dx(n)

def project_onto_feasible(v, target, tol=1e-12, max_iter=200):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) <= tol:
        return v
    for _ in range(max_iter):
        if diff > 0:
            free = np.where(v < 1.0 - tol)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            incs = np.minimum(inc, 1.0 - v[free])
            v[free] += incs
        else:
            free = np.where(v > tol)[0]
            if free.size == 0:
                break
            dec = (-diff) / free.size
            decs = np.minimum(dec, v[free])
            v[free] -= decs
        diff = target - v.sum()
        if abs(diff) <= tol:
            break
    return np.clip(v, 0.0, 1.0)

def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_onto_feasible(v, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()

    rng = np.random.default_rng(seed)

    else:
        n = 80
        target = n / 2.0
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    dx = _dx(n)
    target_sum = n / 2.0
    sum_con = lambda x: np.sum(x[:n]) - target_sum
    constraints = [{"type": "eq", "fun": sum_con}]

            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])
            return x[-1] - overl * dx
        return con
    for kk in range(-n + 1, n):
        if kk == 0:
            continue

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        except Exception:

        h_opt = np.clip(res.x[:n], 0.0, 1.0)

        val = compute_c5(h_opt)

            if best_val < 0.3808:

    h = best_h.copy()
