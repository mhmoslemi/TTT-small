    n = a.shape[0]

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

def gradient_descent_softmax(h0, max_iter, step0, alpha0, alpha_inc, beta):
    h = h0.copy()
    n = h.shape[0]
    s_target = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    alpha = alpha0
    for it in range(1, max_iter + 1):
        grad, corr = softmax_max_gradient(h, alpha)
        # momentum update
        momentum = beta * momentum + (1.-beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        alpha *= alpha_inc
    return best_h, best_score

import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def full_corr_fft(a, b):
    """Convolution a * b using FFT (both real arrays). Returns full result."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(np.pad(a, (0, size - n)))
    B = np.fft.rfft(np.pad(b, (0, size - m)))
    conv = np.fft.irfft(A * B, size)[:n + m - 1]
    return conv


def correlation_full(a, b):
    """Full correlation np.correlate(a, b, mode='full') via FFT."""
    return full_corr_fft(a, b[::-1])


def compute_c5(h):
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    corr = correlation_full(h, 1.0 - h)
    return np.max(corr) * (2.0 / n)


def softmax_max_gradient(h, alpha):
    """Gradient of smooth max of the correlation using softmax weights."""
    corr = correlation_full(h, 1.0 - h)           # length 2n-1
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)            # softmax weights
    k = w + w[::-1]                            # kernel for gradient
    conv = full_corr_fft(h, k)                 # length 3n-2
    start = k.shape[0] - 1                     # =2n-2
    conv_slice = conv[start:start + h.shape[0]]
    grad = 1.0 - conv_slice
    return grad, corr


def gradient_descent_softmax(h0, max_iter, step0=0.5, beta=0.9,
                            alpha0=5.0, alpha_inc=1.001):
    """Projected gradient descent on smoothed max objective."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    alpha = alpha0
    for it in range(1, max_iter + 1):
        grad, _ = softmax_max_gradient(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        alpha *= alpha_inc
    return best_h, best_score


def upsample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones (n assumed even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

    n = 256
    target_sum = n / 2.0

    best_h = None
    best_score = float('inf')

    # ----- generate start candidates -----
    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        cand = upsample(init, n)
        cand = project_onto_box_sum(cand, target_sum)
        candidates.append(cand)

    candidates.append(thue_morse(n))
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(3):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, target_sum))

    # ----- gradient descent on each candidate -----
    for cand in candidates:
        if time_left() < 8.0:
            break
        # allocate iterations based on remaining time
        max_iters = int(min(20000, max(2000, time_left() * 80)))
        h_opt, sc_opt = gradient_descent_softmax(cand, max_iters,
                                                 step0=0.5, beta=0.9,
                                                 alpha0=5.0, alpha_inc=1.001)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # if nothing found, fall back to uniform
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5, dtype=float), target_sum)
        best_score = compute_c5(best_h)

    # ----- binary refinement -----
    if time_left() > 2.0:
        # binary projection of the best continuous vector
        bin_h = np.zeros_like(best_h)
        order = np.argsort(-best_h)  # descending
        bin_h[order[:n // 2]] = 1.0
        bin_score = compute_c5(bin_h)
        if bin_score < best_score:
            best_score, best_h = bin_score, bin_h.copy()

        # hill‑climbing swaps on binary vector
        cur_h = best_h.copy()
        cur_score = best_score
        idx_one = np.flatnonzero(cur_h > 0.5)
        idx_zero = np.flatnonzero(cur_h < 0.5)
        if idx_one.size and idx_zero.size:
            while time_left() > 0.6:
                i = rng.choice(idx_one)
                j = rng.choice(idx_zero)
                # propose swap
                cur_h[i], cur_h[j] = 0.0, 1.0
                new_score = compute_c5(cur_h)
                if new_score < cur_score:
                    cur_score = new_score
                    # update index sets
                    idx_one = idx_one[idx_one != i]
                    idx_one = np.append(idx_one, j)
                    idx_zero = idx_zero[idx_zero != j]
                    idx_zero = np.append(idx_zero, i)
                    if cur_score < best_score:
                        best_score, best_h = cur_score, cur_h.copy()
                    if best_score <= 0.3808:
                        break
                else:
                    # revert swap
                    cur_h[i], cur_h[j] = 1.0, 0.0

    # final safeguard projection
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
