        # distribute diff uniformly then clip

        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        # final fix
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                max_adj = 1.0 - v[i] if diff > 0 else v[i]
