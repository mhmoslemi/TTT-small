import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def _hill_climb(initial_h: np.ndarray, budget: float, rng: np.random.Generator):
    """
    Directed mass‑transfer hill climbing.
    Transfers a small amount of mass from a low‑ns position to a high‑ns one,
    where ns encodes the contribution to the currently worst shift.
    """
    h = initial_h.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)

    start = time.time()
    step = 0.2                # initial transfer magnitude

    while time.time() - start < budget:
        # Correlation (unscaled) – scale only for final C5 evaluation
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)          # shift offset producing the max overlap

        # neighbor sum for this shift
        ns = np.zeros(n)
        if k >= 0:
            if k < n:
                ns[:n - k] += h[k:]
                ns[k:] += h[:n - k]
        else:
            kk = -k
            if kk < n:
                ns[kk:] += h[:n - kk]
                ns[:n - kk] += h[kk:]

        # masks for feasible increase / decrease
        inc_mask = h < 1.0 - 1e-12
        dec_mask = h > 1e-12
        if not inc_mask.any() or not dec_mask.any():
            break

        # preferred indices
        ns_inc = np.where(inc_mask, ns, -np.inf)
        i_inc = int(np.argmax(ns_inc))
        ns_dec = np.where(dec_mask, ns, np.inf)
        i_dec = int(np.argmin(ns_dec))

        # if the chosen pair is not promising, fall back to random pair
        if ns[i_inc] <= ns[i_dec] + 1e-12:
            i_inc = int(rng.choice(np.where(inc_mask)[0]))
            i_dec = int(rng.choice(np.where(dec_mask)[0]))

        delta = min(step, 1.0 - h[i_inc], h[i_dec])
        if delta <= 1e-12:
            step *= 0.5
            continue

        # apply transfer
        h[i_inc] += delta
        h[i_dec] -= delta

        new_c5 = compute_c5(h)
        if new_c5 + 1e-12 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            # slowly decay step (keep exploring)
            if step < 0.5:
                step *= 1.001
        else:
            # revert and shrink step
            h[i_inc] -= delta
            h[i_dec] += delta
            step *= 0.5
            if step < 1e-6:
                break

    return best_h, best_c5


def _initial_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Simple deterministic feasible start (linear decreasing, mass corrected)."""
    M = n // 2
    lin = np.linspace(1.0, 0.0, n)
    scaling = M / np.sum(lin)
    h = lin * scaling
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) > 1e-12:
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # candidate sample counts (even numbers)
    candidates = [80, 96, 112, 128, 144, 160]

    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            # ensure the parent size is tried first
            candidates = [n0] + [n for n in candidates if n != n0]

    for n in candidates:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        remaining = budget_s - elapsed

        # allocate time for this resolution
        alloc = min(250.0, remaining * 0.5)

        # starting vector
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
            M = n // 2
            cur = np.sum(h0)
            if abs(cur - M) > 1e-12:
                h0 += (M - cur) / n
                h0 = np.clip(h0, 0.0, 1.0)
        else:
            h0 = _initial_h(n, rng)

        # evaluate initial point
        c5_0 = compute_c5(h0)
        if c5_0 < best_c5_global:
            best_c5_global = c5_0
            best_h_global = h0.copy()
            best_n_global = n
            if best_c5_global < 0.3808:
                break

        # main hill‑climb for this n
        h_opt, c5_opt = _hill_climb(h0, alloc, rng)
        if c5_opt < best_c5_global:
            best_c5_global = c5_opt
            best_h_global = h_opt.copy()
            best_n_global = n
            if best_c5_global < 0.3808:
                break

        # additional short restarts while time permits
        while time.time() - start_time < budget_s - 2:
            rem = budget_s - (time.time() - start_time)
            if rem < 10:
                break
            # small budget per restart
            sub_alloc = min(30.0, rem * 0.4)

            # generate a random feasible start
            M = n // 2
            mode = rng.integers(3)
            if mode == 0:                     # binary start
                h_start = np.zeros(n)
                idx = rng.choice(n, M, replace=False)
                h_start[idx] = 1.0
            elif mode == 1:                   # normalized random start
                h_start = rng.random(n)
                h_start = h_start / np.sum(h_start) * M
                h_start = np.clip(h_start, 0.0, 1.0)
                cur = np.sum(h_start)
                if cur < M:
                    deficit = M - cur
                    order = np.argsort(h_start)
                    for i in order:
                        inc = min(deficit, 1.0 - h_start[i])
                        h_start[i] += inc
                        deficit -= inc
                        if deficit <= 1e-12:
                            break
                elif cur > M:
                    excess = cur - M
                    order = np.argsort(-h_start)
                    for i in order:
                        dec = min(excess, h_start[i])
                        h_start[i] -= dec
                        excess -= dec
                        if excess <= 1e-12:
                            break
            else:                             # noisy linear shape
                h_start = _initial_h(n, rng)
                h_start += rng.normal(scale=0.05, size=n)
                h_start = np.clip(h_start, 0.0, 1.0)
                cur = np.sum(h_start)
                if abs(cur - M) > 1e-12:
                    h_start += (M - cur) / n
                    h_start = np.clip(h_start, 0.0, 1.0)

            h_opt2, c5_opt2 = _hill_climb(h_start, sub_alloc, rng)
            if c5_opt2 < best_c5_global:
                best_c5_global = c5_opt2
                best_h_global = h_opt2.copy()
                best_n_global = n
                if best_c5_global < 0.3808:
                    break
        if best_c5_global < 0.3808:
            break

    # fallback if nothing succeeded
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        M = n // 2
        cur = np.sum(best_h_global)
        if abs(cur - M) > 1e-9:
            best_h_global += (M - cur) / n
            best_h_global = np.clip(best_h_global, 0.0, 1.0)
        best_c5_global = compute_c5(best_h_global)

    return best_h_global, best_c5_global, n
