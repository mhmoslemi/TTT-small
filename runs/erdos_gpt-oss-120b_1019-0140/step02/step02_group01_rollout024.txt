class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n/2.0
class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def slqp_optimize(n, time_budget, rng, init_h=None):
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    constraints = [{'type':'eq','fun':sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]
    bounds = [(0.0,1.0)]*n + [(0.0,None)]
    best_h = None
    best_c5 = np.inf
    start = time.time()
    # if init_h provided, ensure sum correct
    if init_h is not None:
        h0 = init_h.copy()
        cur_sum = np.sum(h0)
        if abs(cur_sum - M) > 1e-12:
            h0 += (M - cur_sum) / n
            h0 = np.clip(h0, 0.0, 1.0)
        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])
        # single attempt
        try:
            res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP', bounds=bounds,
                            constraints=constraints, options={'maxiter':500,'ftol':1e-10,'disp':False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                cur = np.sum(h_opt)
                if abs(cur - M) > 1e-9:
                    h_opt += (M - cur) / n
                    h_opt = np.clip(h_opt,0.0,1.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
        except Exception:
            pass
    # additional restarts with random init
    while time.time() - start < time_budget - 0.1:
        # random init - binary
        h0 = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h0[idx] = 1.0
        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])
        try:
            res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP', bounds=bounds,
                            constraints=constraints, options={'maxiter':500,'ftol':1e-10,'disp':False})
            if not res.success:
                continue
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            cur = np.sum(h_opt)
            if abs(cur - M) > 1e-9:
                h_opt += (M - cur) / n
                h_opt = np.clip(h_opt,0.0,1.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
            if best_c5 < 0.3808:
                break
        except Exception:
            continue
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper classes ----------
class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx for a given shift k."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ---------- Objective utilities ----------
def compute_c5(h):
    """Maximum overlap integral for a discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_shift(h):
    """Return (C5, shift) where shift gives the maximal overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    c5 = corr[idx] * dx
    shift = idx - (n - 1)          # shift = k (positive -> right)
    return c5, shift

# ---------- SLSQP optimisation (lightweight) ----------
def slqp_optimize(n, time_budget, rng, init_h=None):
    """Run a few SLSQP restarts for size n within time_budget seconds."""
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf
    start = time.time()

    # use supplied init_h first, if any
    if init_h is not None:
        h0 = np.asarray(init_h, dtype=float).copy()
        cur = np.sum(h0)
        if abs(cur - M) > 1e-12:
            h0 += (M - cur) / n
            h0 = np.clip(h0, 0.0, 1.0)
        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])
        try:
            res = minimize(fun=lambda x: x[-1],
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,

            if res.success:

                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
        except Exception:
            pass

    # additional binary restarts

def refine_h(h, time_limit, rng, target=0.3808):
    """Hill‑climbing using the worst shift to guide mass transfers."""
    n = h.shape[0]
    dx = 2.0 / n
    cur_c5, cur_k = compute_c5_and_shift(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()

    while time.time() - start < time_limit:

        cur_c5, k = compute_c5_and_shift(h)
        if k >= 0:

        else:

            others = idxs[idxs != i]

        else:
