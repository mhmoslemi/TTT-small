def slsqp_optimize(h_start, n, deadline):
    dx = 2.0 / n
    target_sum = n / 2.0
    def sum_con(x): return np.sum(x[:n]) - target_sum
    constraints = [{"type": "eq", "fun": sum_con}]
    def make_overlap(k):
        def con(x, k=k):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])
            return x[-1] - overl * dx
        return con
    for kk in range(-n + 1, n):
        if kk == 0:
            continue
        constraints.append({"type": "ineq", "fun": make_overlap(kk)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5(h_start, dx)
    x0 = np.concatenate([h_start, [t0]])
    # set maxiter maybe 500
    try:
        res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol':1e-6, 'disp': False})
    except Exception:
        return h_start, compute_c5(h_start,dx)
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # ensure sum constraint
        cur = h_opt.sum()
        if abs(cur - target_sum) > 1e-9:
            h_opt += (target_sum - cur) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        val = compute_c5(h_opt, dx)
        return h_opt, val
    else:
        return h_start, compute_c5(h_start,dx)

import time
import numpy as np
from scipy.optimize import minimize  # imported only for possible future use, not a lambda


def compute_c5(h, dx=None):
    """Maximum overlap integral approximated by discrete full correlation."""
    n = h.shape[0]
    if dx is None:
        dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def random_start_vector(n, rng):
    """Random vector in [0,1]^n with exact sum n/2 via mass‑transfer fix‑up."""
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    # simple iterative correction
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(v < 1.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            delta = min(diff, 1.0 - v[i])
            v[i] += delta
            diff -= delta
        else:
            idx = np.where(v > 0.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            delta = min(-diff, v[i])
            v[i] -= delta
            diff += delta
    return v


def generate_bump(n):
    """Cosine bump: 0 at ends, 1 at centre, average 0.5 → exact sum n/2."""
    idx = np.arange(n)
    angle = np.pi * (2.0 * idx / (n - 1) - 1.0)   # -π … π
    h = 0.5 + 0.5 * np.cos(angle)
    # tiny sum correction for floating errors
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h


def _max_corr_info(h):
    """Return (lag k, max correlation value, correlation array)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_corr = corr[max_idx]
    k = max_idx - (n - 1)
    return k, max_corr, corr


def _product_contributions(h, k):
    """Return (indices array, product contributions) for lag k."""
    n = h.shape[0]
    if k >= 0:
        i_range = np.arange(n - k)
        prod = h[i_range] * (1.0 - h[i_range + k])
    else:
        kk = -k
        i_range = np.arange(kk, n)
        prod = h[i_range] * (1.0 - h[i_range - kk])
    return i_range, prod


def local_search(h_start, n, deadline, rng):
    """Pairwise mass‑transfer descent with simple simulated‑annealing fallback."""
    dx = 2.0 / n
    target_sum = n / 2.0
    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    max_no_improve = 2000
    no_improve = 0
    while time.time() < deadline - 0.2:
        k, max_corr, _ = _max_corr_info(cur_h)
        i_range, prod = _product_contributions(cur_h, k)
        # source = index with largest product, destination = smallest
        i_src = i_range[int(np.argmax(prod))]
        i_dst = i_range[int(np.argmin(prod))]
        # ensure feasible
        if cur_h[i_src] <= 1e-12 or cur_h[i_dst] >= 1.0 - 1e-12:
            src_cand = np.where(cur_h > 1e-12)[0]
            dst_cand = np.where(cur_h < 1.0 - 1e-12)[0]
            if src_cand.size == 0 or dst_cand.size == 0:
                break
            i_src = rng.choice(src_cand)
            i_dst = rng.choice(dst_cand)
        max_delta = min(cur_h[i_src], 1.0 - cur_h[i_dst])
        if max_delta <= 0:
            break
        delta = max_delta * (0.1 + 0.2 * rng.random())
        new_h = cur_h.copy()
        new_h[i_src] -= delta
        new_h[i_dst] += delta
        new_val = compute_c5(new_h, dx)
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            # simulated‑annealing acceptance
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_h, cur_val = new_h, new_val
            no_improve += 1
        if best_val < 0.3808:
            break
        if no_improve > max_no_improve:
            break
    # final projection onto exact sum (tiny drift possible)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        best_h = np.clip(best_h + diff / n, 0.0, 1.0)
    return best_h, best_val


def best_rotation_opt(h, n, deadline):
    """Search all cyclic shifts for the smallest C5 (early‑exit on timeout)."""
    dx = 2.0 / n
    best_h = h
    best_val = compute_c5(h, dx)
    for s in range(1, n):
        if time.time() >= deadline - 0.05:
            break
        h_rot = np.roll(h, s)
        val = compute_c5(h_rot, dx)
        if val < best_val:
            best_val, best_h = val, h_rot
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    best_h = None
    best_val = np.inf
    best_n = None

    # candidate sample sizes (keep modest for speed)
    candidate_ns = [80, 120, 160, 200]

    for n in candidate_ns:
        if time.time() >= deadline - 5.0:
            break
        dx = 2.0 / n
        target_sum = n / 2.0

        # collect starting vectors
        start_vectors = []

        # deterministic cosine bump
        start_vectors.append(generate_bump(n))

        # parent construction if compatible
        if "initial_h_values" in globals():
            init_arr = np.array(initial_h_values, dtype=float)
            if init_arr.shape[0] == n:
                h0 = init_arr.copy()
                # enforce bounds & sum
                h0 = np.clip(h0, 0.0, 1.0)
                cur = h0.sum()
                if abs(cur - target_sum) > 1e-12:
                    h0 = np.clip(h0 + (target_sum - cur) / n, 0.0, 1.0)
                start_vectors.append(h0)

        # a few random starts
        for _ in range(4):
            if time.time() >= deadline - 5.0:
                break
            start_vectors.append(random_start_vector(n, rng))

        for h0 in start_vectors:
            if time.time() >= deadline - 5.0:
                break
            h0 = np.clip(h0, 0.0, 1.0)
            cur = h0.sum()
            if abs(cur - target_sum) > 1e-12:
                h0 = np.clip(h0 + (target_sum - cur) / n, 0.0, 1.0)

            # local descent
            h_opt, val_opt = local_search(h0, n, deadline, rng)

            # rotation refinement
            h_opt, val_opt = best_rotation_opt(h_opt, n, deadline)

            if val_opt < best_val:
                best_val, best_h, best_n = val_opt, h_opt.copy(), n
                if best_val < 0.3808:
                    break
        if best_val < 0.3808:
            break

    # fallback: uniform 0.5 if nothing found
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5, dtype=float)
        best_val = compute_c5(best_h)

    # final safety projection onto exact constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_n / 2.0
    cur = best_h.sum()
    if abs(cur - target_sum) > 1e-9:
        # distribute tiny discrepancy uniformly and re‑clip
        best_h = np.clip(best_h + (target_sum - cur) / best_n, 0.0, 1.0)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, best_n
