def repair(h):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    target = n / 2.0
    diff = target - total
    # Iterate while diff not negligible
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass: find index with h < 1
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h

import time
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(h):
    """Maximum overlap integral approximation via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def repair_vector(v):
    """Project vector onto the feasible set: 0≤v≤1 and sum(v)=len(v)/2."""
    v = np.clip(v, 0.0, 1.0)
    n = v.shape[0]
    target = n / 2.0
    total = v.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        # add to smallest entries first
        idx = np.argsort(v)
        for i in idx:
            if diff <= 0:
                break
            cap = 1.0 - v[i]
            add = min(cap, diff)
            if add > 0:
                v[i] += add
                diff -= add
    else:
        diff = -diff
        # remove from largest entries first
        idx = np.argsort(-v)
        for i in idx:
            if diff <= 0:
                break
            cap = v[i]
            sub = min(cap, diff)
            if sub > 0:
                v[i] -= sub
                diff -= sub
    return v

def de_objective(x):
    """Objective for differential evolution: C5 after repairing sum constraint."""
    h = repair_vector(np.array(x, copy=True))
    return compute_c5(h)

def anneal(h_start, start_time, deadline, rng):
    """Simple simulated annealing on the feasible set."""
    h = h_start.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    total_time = deadline - start_time
    while time.time() < deadline - 0.01:
        # current worst shift and its contributions
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        # choose donor index
        if rng.random() < 0.7:
            donor_local = np.argmax(prod)
            donor = i_range[donor_local]
        else:
            donors = np.where(h > 1e-12)[0]
            if donors.size == 0:
                break
            donor = rng.choice(donors)
        # choose receiver index
        receivers = np.where(h < 1.0 - 1e-12)[0]
        if receivers.size == 0:
            break
        receiver = rng.choice(receivers)
        if donor == receiver:
            continue
        max_transfer = min(h[donor], 1.0 - h[receiver])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[donor] -= delta
        h[receiver] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            remaining = deadline - time.time()
            T = max(0.01 * remaining / total_time, 1e-12)
            if np.exp(-(new_val - cur_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[donor] += delta
                h[receiver] -= delta
        # optional early stop when a very good bound is reached
        if best_val < 0.3808:
            break
    return best_h, best_val
