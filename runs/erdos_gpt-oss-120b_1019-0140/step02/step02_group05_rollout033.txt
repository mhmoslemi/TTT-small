                            overl = 0.0
                        else:
                            overl = np.dot(h[: n - k], 1.0 - h[k:])
                    else:
                        kk = -k
                        if n - kk <= 0:
                            overl = 0.0
                        else:
                            overl = np.dot(h[kk:], 1.0 - h[: n - kk])
                    return x[-1] - overl * (2.0 / n)
                return fun

            x0 = np.concatenate([best_h, [best_val]])
            cons = [{"type": "eq", "fun": sum_eq}]
            for kk in range(-n + 1, n):
                if kk == 0:
                    continue
                cons.append({"type": "ineq", "fun": make_ineq(kk)})
            bounds = [(0.0, 1.0)] * n + [(0.0, None)]

            res = minimize(fun=lambda x: x[-1],
                           x0=x0,
                           method="SLSQP",
                           bounds=bounds,
                           constraints=cons,
                           options={"maxiter": 300, "ftol": 1e-9, "disp": False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_to_sum_and_bounds(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    best_h = project_to_sum_and_bounds(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, n
