def adam_subgradient(h0, n_iter, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    s = h.size/2.0
    for t in range(1, n_iter+1):
        grad = compute_gradients_for_shifts(h, get_max_shifts(h))
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = lr*m_hat/(np.sqrt(v_hat)+eps)
        h = h - step
        h = project_onto_box_sum(h, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score
```

def hill_climb_binary(h0, rng, time_limit):
    h = h0.copy()
    score = compute_c5(h)
    best_h = h.copy()
    best_score = score
    n = len(h)
    # indices of zeros and ones
    zeros = np.flatnonzero(h == 0)
    ones = np.flatnonzero(h == 1)
    # loop
    while time.time() < time_limit:
        # choose random zero and one:
        if len(zeros)==0 or len(ones)==0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        # swap
        h[i], h[j] = 1, 0
        new_score = compute_c5(h)
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            # update lists
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
        else:
            # revert
            h[i], h[j] = 0, 1
        # maybe after certain iterations with no improvement break
    return best_h, best_score
```

def hill_climb_binary(h0, rng, end_time):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    zeros = np.flatnonzero(h == 0)
    ones = np.flatnonzero(h == 1)
    no_improve = 0
    max_no_improve = 2000
    while time.time() < end_time and no_improve < max_no_improve:
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        # swap
        h[i] = 1.0
        h[j] = 0.0
        new_score = compute_c5(h)
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            # update zero/one lists
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            no_improve = 0
        else:
            # revert
            h[i] = 0.0
            h[j] = 1.0
            # update counts maybe unchanged
            no_improve += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.shape[0]

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]  # t variable lower bound zero, no upper bound
    # Actually upper bound for t not needed.

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        # correlation scaled
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr  # all elements >=0 constraint t >= corr for each shift

import time
import numpy as np
import scipy.optimize as opt

# ----- Helper functions -------------------------------------------------

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure bracketing
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = (lam_low + lam_high) * 0.5
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Linear interpolation from a larger array to a smaller size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        term1 = np.zeros(n, dtype=h.dtype)
        term2 = np.zeros(n, dtype=h.dtype)
        if shift >= 0:
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts) if shifts else 1.0
    return grad


def slsqp_optimize_candidate(h0, max_iter=500):
    """Run a short SLSQP to improve a candidate."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr  # elementwise; SLSQP expects >=0

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def hill_climb_binary(h0, rng, end_time):
    """Swap pairs of 0/1 to reduce C5; respects exact balance."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    zeros = np.flatnonzero(h == 0)
    ones = np.flatnonzero(h == 1)
    no_improve = 0
    max_no_improve = 2000
    while time.time() < end_time and no_improve < max_no_improve:
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        # perform swap
        h[i] = 1.0
        h[j] = 0.0
        new_score = compute_c5(h)
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            no_improve = 0
        else:
            # revert
            h[i] = 0.0
            h[j] = 1.0
            no_improve += 1
    return best_h, best_score


def adam_optimize(h0, max_seconds, lr=0.2):
    """Projected Adam‑style sub‑gradient descent on C5."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    start = time.time()
    while time.time() - start < max_seconds:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = h - step
        h = project_onto_box_sum(h, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            if best_score <= 0.3808:
                break
    return best_h, best_score
