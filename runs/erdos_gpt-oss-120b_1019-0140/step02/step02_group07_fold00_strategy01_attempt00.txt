import numpy as np

def compute_c5(arr):
    n = len(arr)
    if n == 0:
        return 0.0
    corr = np.correlate(arr, 1 - arr, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def thue_morse_seq(n):
    seq = []
    seq.append(0)
    while len(seq) < n:
        next_bits = []
        for bit in seq:
            next_bits.append(bit)
            bit = 1 - bit
            next_bits.append(bit)
        seq.extend(next_bits)
    return np.array(seq[:n])

def rudin_shapiro_seq(n):
    seq = [1]
    while len(seq) < n:
        for idx, val in enumerate(seq):
            seq.append(val)
            if (idx+1) % 2 == 0:
                val *= -1
                seq.append(val)
    return np.array(seq[:n])

def repair_vector(x, target_sum):
    vec = x.astype(float)
    while True:
        s = np.sum(vec)
        if abs(s - target_sum) < 1e-5:
            return vec
        if s < target_sum:
            idxs = np.flatnonzero(vec < 1)[np.random.permutation(len(vec))[:int(target_sum - s)]]
            vec[idxs] = 1.0
        else:
            idxs = np.flatnonzero(vec > 0)[np.random.permutation(len(vec))[:int(s - target_sum)]]
            vec[idxs] = 0.0
    return vec

def adjust_parent(parent, target_sum):

    s = np.sum(vec)
    needed = target_sum - s
    if abs(needed) < 1e-5:
        return vec

        idxs = np.flatnonzero(vec < 1.0)
        if len(idxs) == 0:
            return vec
        add_idx = np.random.choice(idxs, int(needed), replace=False)
        vec[add_idx] += (1.0 - vec[add_idx])
    else:
        idxs = np.flatnonzero(vec > 0.0)

            return vec
        add_idx = np.random.choice(idxs, int(-needed), replace=False)
