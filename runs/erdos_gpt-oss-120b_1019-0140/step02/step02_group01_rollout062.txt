    lo = -2.0
    hi = 2.0

        lam = (lo + hi) / 2.0

def gradient_step(h, beta):
    n = len(h)
    # compute full correlation
    corr = np.correlate(h, h, mode='full')
    # extract S for non-zero shifts
    S_neg = corr[:n-1]      # shifts -n+1 .. -1
    S_pos = corr[n:]        # shifts 1 .. n-1
    S_vals = np.concatenate((S_neg, S_pos))
    # compute weights
    exp_arg = -beta * S_vals
    max_e = np.max(exp_arg)
    w = np.exp(exp_arg - max_e)
    w_sum = w.sum()
    if w_sum == 0:
        prob = np.full_like(w, 1.0 / w.size)
    else:
        prob = w / w_sum
    grad = np.zeros_like(h)
    # loops over shifts
    # negative shifts:
    # for idx, s = -n+1 + idx
    # but we can treat both same using same code for positive
    # For negative shift s = -t (t>0):
    # derived earlier: contributions: for j >= t: grad[j] += prob[idx] * h[j - t]; for j <= n-1 - t: grad[j] += prob[idx] * h[j + t]
    # We'll run loop for s in range(1,n): use prob for negative and positive separately.
    # However, we can just loop over shift magnitude t=1..n-1 with combined weight prob_neg and prob_pos.
    # Let's separate:
    prob_neg = prob[:n-1]
    prob_pos = prob[n-1:]
    for t in range(1, n):
        w_neg = prob_neg[t-1]
        w_pos = prob_pos[t-1]
        # for negative shift -t:
        if w_neg > 0:
            # add contributions
            # j >= t:
            grad[t:] += w_neg * h[:n-t]   # h[j - t]
            # j <= n-1 - t:
            grad[:n-t] += w_neg * h[t:]   # h[j + t]
        # for positive shift +t:
        if w_pos > 0:
            # same pattern as earlier
            grad[:n-t] += w_pos * h[t:]   # h[j + t]
            grad[t:] += w_pos * h[:n-t]   # h[j - t]
    return grad

def project_box_sum(v, target_sum):
    # Clip to [0,1] after shifting via lambda
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    proj = np.clip(v - lam, 0.0, 1.0)
    # Ensure final sum close to target; minor adjustment if needed
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-8:
        # distribute small amount across non-saturated entries
        mask = (proj > 0.0) & (proj < 1.0)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def subgradient_overlap(h, k):
    """
    Compute gradient of O_k = dx * (P_k - S_k) w.r.t h.
    Returns vector grad.
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if k > 0:
        t = k
        # indices j 0..n-1
        # portion from P_k:
        grad[:n - t] += dx  # j <= n-1-t
        # portion from -S_k:

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """Maximum overlap integral for a step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_box_sum(v, target):
    """Project vector v onto [0,1]^n with exact sum = target (box simplex)."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    proj = np.clip(v - lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-8:
        mask = (proj > 0.0) & (proj < 1.0)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def subgradient_overlap(h, k):
    """
    Subgradient of the overlap O_k = dx * (P_k - S_k) w.r.t. h,
    where P_k is the sum of h over the overlapping region and
    S_k = sum_i h_i h_{i+k} (linear correlation).  k != 0.
    """
    n = h.size
    dx = 2.0 / n
    g = np.zeros_like(h)
    if k > 0:
        t = k
        # prefix contribution
        g[: n - t] += dx
        # -h_{j+k} term
        g[: n - t] -= dx * h[t:]
        # -h_{j-k} term
        g[t:] -= dx * h[: n - t]
    elif k < 0:
        t = -k
        g[t:] += dx
        g[t:] -= dx * h[: n - t]
        g[: n - t] -= dx * h[t:]
    return g


# ---------- optimisation for a single resolution ----------
def optimise_n(n, rng, time_budget):
    M = n // 2

    # initialise h
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, M)
    else:
        h = project_box_sum(rng.random(n), M)

    # initial evaluation
    n_mid = n - 1
    dx = 2.0 / n
    corr1 = np.correlate(h, 1.0 - h, mode="full")
    Ovals = corr1 * dx
    Ovals[n_mid] = -np.inf
    cur_c5 = Ovals.max()
    best_h, best_c5 = h.copy(), cur_c5

    start = time.time()
    it = 0
    eta0 = 0.5  # initial step scale

    while time.time() - start < time_budget - 0.1:
        it += 1
        # worst shift
        idx = int(np.argmax(Ovals))
        k_star = idx - n_mid

        # subgradient descent step
        grad = subgradient_overlap(h, k_star)
        eta = eta0 / np.sqrt(it)
        h_new = h - eta * grad

        # occasional jitter
        if it % 200 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_box_sum(h_new, M)

        # recompute overlaps
        corr1 = np.correlate(h_new, 1.0 - h_new, mode="full")
        Onew = corr1 * dx
        Onew[n_mid] = -np.inf
        new_c5 = Onew.max()

        # accept if improved, otherwise occasional uphill move
        if new_c5 < cur_c5 - 1e-12 or rng.random() < 0.01:
            h, cur_c5, Ovals = h_new, new_c5, Onew
            it = 0  # reset step adaptation after progress

        # keep global best
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()
            if best_c5 < 0.3808:
                break

        # restart if stuck
        if it > 5000:
            h = project_box_sum(rng.random(n), M)
            corr1 = np.correlate(h, 1.0 - h, mode="full")
            Ovals = corr1 * dx
            Ovals[n_mid] = -np.inf
            cur_c5 = Ovals.max()
            it = 0

    return best_h, best_c5


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h, best_c5, best_n = None, np.inf, None

    # candidate sample sizes (even)
    cand_ns = [80, 96, 112, 128, 144]

    # prioritize parent size if present
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in cand_ns:
                cand_ns.insert(0, n0)
            else:
                cand_ns = [n0] + [n for n in cand_ns if n != n0]

    for n in cand_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        alloc = min(250.0, remaining * 0.5)
        h_opt, c5_opt = optimise_n(n, rng, alloc)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # ----- local pairwise refinement (optional) -----
    if best_h is not None:
        n = best_n
        M = n // 2
        cur = best_c5
        while time.time() - start_time < budget_s - 0.5:
            i, j = rng.choice(n, 2, replace=False)
            max_move = min(1.0 - best_h[i], best_h[j])
            if max_move <= 0:
                continue
            delta = rng.random() * max_move * 0.5
            h_new = best_h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < cur:
                best_h, cur = h_new, c5_new
                best_c5 = c5_new
                if cur < 0.3808:
                    break

    # ----- fallback if everything failed -----
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    M = best_n // 2
    if abs(best_h.sum() - M) > 1e-9:
        best_h = project_box_sum(best_h, M)
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
