def binary_anneal(start_bin, time_end, rng):
    n = len(start_bin)
    h = start_bin.copy()
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    target_sum = n // 2
    # schedule: linear decrease temperature from init_temp to final_temp over iterations.
    init_temp = 1e-2 (maybe)
    final_temp = 1e-6
    iter = 0
    max_iter = 200000
    while time.time() < time_end and iter < max_iter:
        # propose swapping
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size==0 or zeros.size==0: break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_c5 = compute_c5(h_new.astype(float))
        dE = new_c5 - cur_c5
        T = init_temp * (1 - iter/max_iter) + final_temp * (iter/max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE/T):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        iter += 1
    return best_h.astype(float), best_c5

def solve_milp_binary(n, time_limit):
    import cvxpy as cp
    import numpy as np
    target_sum = n // 2
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == target_sum,
                   t >= 0, t <= 1]
    for l in range(1, n):
        dim = n - l
        y = cp.Variable(dim, boolean=True)
        constraints += [y <= x[:dim],
                        y <= x[l:],
                        y >= x[:dim] + x[l:] - 1]
        constraints.append(cp.sum(y) + target_sum * t >= target_sum)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, **{'seconds': time_limit})
    except Exception:
        return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None
    x_val = np.rint(x.value).astype(int)
    # correct sum if necessary
    cur_sum = x_val.sum()
    if cur_sum != target_sum:
        diff = target_sum - cur_sum
        zero_idxs = np.where(x_val == 0)[0]
        one_idxs = np.where(x_val == 1)[0]
        if diff > 0:
            # flip zeros to ones
            flip = np.random.choice(zero_idxs, diff, replace=False)
            x_val[flip] = 1
        elif diff < 0:
            flip = np.random.choice(one_idxs, -diff, replace=False)
            x_val[flip] = 0
    h = x_val.astype(float)
    c5 = compute_c5(h)
    return h, c5

    h = start_bin.copy()
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    target_sum = n // 2
    # schedule parameters
    init_temp = 5e-3
    final_temp = 1e-7

import time
import math
import numpy as np

def compute_c5(arr):
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated by discrete correlation."""
    n = len(arr)
    # correlation of h with its complement (zero padding)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Thue‑Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)

def rudin_shapiro_seq(n):
    """Rudin‑Shapiro binary sequence (+1/−1 → 1/0) of length n."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def binary_anneal(start_bin, time_end, rng):
    """Simulated annealing on binary vectors (swap moves)."""
    n = len(start_bin)
    h = start_bin.copy()
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    max_iter = 300000
    it = 0
    init_temp = 5e-3
    final_temp = 1e-7
    while time.time() < time_end and it < max_iter:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_c5 = compute_c5(h_new.astype(float))
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    return best_h.astype(float), best_c5

def solve_milp_binary(n, time_limit, rng):
    """Attempt exact MILP solution for binary h using cvxpy + CBC."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    target_sum = n // 2
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == target_sum,
                   t >= 0,
                   t <= 1]
    for l in range(1, n):
        dim = n - l
        y = cp.Variable(dim, boolean=True)
        constraints += [y <= x[:dim],
                        y <= x[l:],
                        y >= x[:dim] + x[l:] - 1]
        constraints.append(cp.sum(y) + target_sum * t >= target_sum)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, **{'seconds': time_limit})
    except Exception:
        return None, None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None
    xv = np.rint(x.value).astype(int)
    cur_sum = int(xv.sum())
    if cur_sum != target_sum:
        diff = target_sum - cur_sum
        if diff > 0:
            zeros = np.flatnonzero(xv == 0)
            if zeros.size >= diff:
                flip = rng.choice(zeros, diff, replace=False)
                xv[flip] = 1
        elif diff < 0:
            ones = np.flatnonzero(xv == 1)
            diff = -diff
            if ones.size >= diff:
                flip = rng.choice(ones, diff, replace=False)
                xv[flip] = 0
    h = xv.astype(float)
    return h, compute_c5(h)
