def random_binary_vector(n, rng):
    h = np.zeros(n, dtype=int)
    ones_idx = rng.choice(n, n//2, replace=False)
    h[ones_idx] = 1
    return h

def binary_hill_climb(start_bin, time_end, rng):
    n = len(start_bin)
    best_h = start_bin.copy()
    best_c5 = compute_c5(best_h.astype(float))
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5
```

import time
import numpy as np

def compute_c5(arr):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)


def repair(arr, target_sum):
    """Clip to [0,1] and enforce exact sum = target_sum."""
    arr = np.clip(arr, 0.0, 1.0)
    cur = np.sum(arr)
    if abs(cur - target_sum) > 1e-12:
        diff = target_sum - cur
        arr = arr + diff / len(arr)
        arr = np.clip(arr, 0.0, 1.0)
        residual = target_sum - np.sum(arr)
        for i in range(len(arr)):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -arr[i], 1 - arr[i])
            arr[i] += adj
            residual = target_sum - np.sum(arr)
    return arr


def random_binary(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    h[rng.choice(n, n // 2, replace=False)] = 1
    return h


def hill_climb(start_bin, time_end, rng):
    """Swap‑based hill climb on binary vectors; preserves number of ones."""
    best_h = start_bin.copy()
    best_c5 = compute_c5(best_h.astype(float))
    no_improve = 0
    while time.time() < time_end and no_improve < 5000:
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5


def rudin_shapiro_seq(m):
    """Rudin–Shapiro binary sequence of length m (values 0/1)."""
    a = np.array([1], dtype=int)
    while len(a) < m:
        idx = np.arange(len(a))
        a_even = a
        a_odd = a * ((-1) ** idx)
        a = np.concatenate([a_even, a_odd])
    a = a[:m]
    return (a + 1) // 2


def generate_seeds(n, rng):
    """A few deterministic balanced binary seeds."""
    seeds = []
    seeds.append(random_binary(n, rng).astype(float))

    tm = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int)
    diff = n // 2 - np.sum(tm)
    if diff != 0:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for idx in idxs:
            if diff == 0:
                break
            if diff > 0 and tm[idx] == 0:
                tm[idx] = 1
                diff -= 1
            elif diff < 0 and tm[idx] == 1:
                tm[idx] = 0
                diff += 1
    seeds.append(tm.astype(float))

    rs = rudin_shapiro_seq(n)
    diff = n // 2 - np.sum(rs)
    if diff != 0:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for idx in idxs:
            if diff == 0:
                break
            if diff > 0 and rs[idx] == 0:
                rs[idx] = 1
                diff -= 1
            elif diff < 0 and rs[idx] == 1:
                rs[idx] = 0
                diff += 1
    seeds.append(rs.astype(float))

    return seeds
