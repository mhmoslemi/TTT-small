def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def slsqp_refine(h, best_c5):
    n = len(h)
    # define objective
    def obj(x):
        # x only h part
        return compute_c5(x)
    # constraints: sum(x) = target_sum
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
    bounds = [(0.0,1.0)] * n
    # use minimize, with initial x = h
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=[cons], options={'maxiter':2000,'ftol':1e-9})
    if res.success:
        h_opt = np.clip(res.x, 0.0, 1.0)
        # enforce sum exactly
        h_opt = project_to_box_sum(h_opt, n/2)
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt
    else:
        return h, best_c5

def slsqp_optimize_h(h, time_left):
    n = len(h); target_sum = n/2

    def objective(x):
        return compute_c5(x)

    constraints = [{'type':'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0,1.0)]*n

    # set maxiter maybe 1000 or based on remaining time
    maxiter = int(5000 * time_left / budget_s) # approximate
    # Ensure at least some min
    maxiter = max(200, maxiter)

    res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol':1e-9, 'disp':False})

        c5_opt = compute_c5(h_opt)

def reduce_max_overlap(h, target_overlap, rng, max_iters=None):
    n = len(h)
    # h is binary numpy array (float 0/1).
    # Continue until max_overlap <= target_overlap or no improvement.
    # max_iters optionally.
    for it in range(max_iters if max_iters is not None else 100000):
        # compute corr
        corr = np.correlate(h, 1 - h, mode='full')
        # max value and index
        max_idx = np.argmax(corr)
        max_val = int(corr[max_idx])  # integer
        if max_val <= target_overlap:
            break
        shift = max_idx - (n - 1)
        # compute rolled version
        rolled = np.roll(h, -shift)
        S = np.where((h == 1) & (rolled == 0))[0]

            j = rng.choice(np.where(h == 0)[0])
        # swap
        h[i], h[j] = h[j], h[i]  # swap values
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.size
    else:
        n = 80
    target_sum = n // 2 if n % 2 == 0 else int(n/2) # but spec sum = n/2 exactly maybe n always even.

    # define target overlap count based on desired C5 bound 0.3808
    dx = 2.0 / n
    target_c5_bound = 0.3808
    target_overlap = int(np.floor(target_c5_bound / dx))

    # Initialize best solution
    best_h = None
    best_c5 = np.inf

    # If initial_h_values exists, convert to binary: use top half as 1
    if 'initial_h_values' in globals():
        # compute binary version
        idx_sorted = np.argsort(init_h)[::-1]
        binary_h = np.zeros_like(init_h)
        binary_h[idx_sorted[:target_sum]] = 1.0
        c5_val = compute_c5(binary_h)
        best_h = binary_h
        best_c5 = c5_val

    # Phase 1: binary random + reduction
    binary_deadline = start_time + budget_s * 0.6
    while time.time() < binary_deadline:
        # generate random binary vector
        h = np.zeros(n, dtype=float)
        indices = rng.choice(n, size=target_sum, replace=False)
        h[indices] = 1.0
        # maybe quick check evaluate c5, skip if already > best? Actually we want better.
        # Reduce max overlap
        h_reduced = reduce_max_overlap(h, target_overlap, rng)
        c5_val = compute_c5(h_reduced)
        if c5_val < best_c5 - 1e-12:

            best_c5 = c5_val
            # early break if already at target bound (or lower)
            if best_c5 <= target_c5_bound:
                # but we want strictly smaller? okay
                break

    # Phase 2: Continuous refinement via SLSQP
    # Use remaining time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        # Perform maybe up to N attempts
        attempts = 5
        # allocate time per attempt proportionally
        for attempt in range(attempts):
            if time.time() - start_time >= budget_s - 2:
                break
            # Starting point: perturb best_h slightly
            h0 = best_h.copy()
            # Add small Gaussian noise and then project to box/sum
            noise = rng.normal(scale=0.1, size=n)  # maybe moderate noise
            h0 = h0 + noise
            # Clip and project
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = project_to_box_sum(h0, target_sum)
            # Run SLSQP
            # compute maxiter based on remaining time
            time_left = budget_s - (time.time() - start_time)
            # allocate maxiter ~ 2000 * time_left / budget_s
            maxiter = max(200, int(2000 * time_left / budget_s))
            # define constraints
            def sum_constraint(x):
                return np.sum(x) - target_sum
            cons = {'type': 'eq', 'fun': sum_constraint}
            bounds = [(0.0, 1.0)] * n
            res = minimize(lambda x: compute_c5(x), h0, method='SLSQP',
                           bounds=bounds, constraints=[cons],
                           options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
            if res.success:
                h_opt = np.clip(res.x, 0.0, 1.0)
                # enforce exact sum
                h_opt = project_to_box_sum(h_opt, target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5 - 1e-12:

                    best_c5 = c5_opt
                    if best_c5 <= target_c5_bound:
                        break
    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def reduce_max_overlap(h, target_overlap, rng, max_iters=5000):
    n = h.shape[0]
    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = int(corr[max_idx])
        if max_val <= target_overlap:
            break
        shift = max_idx - (n - 1)
        rolled = np.roll(h, -shift)
        S = np.where((h == 1) & (rolled == 0))[0]
        T = np.where((h == 0) & (rolled == 1))[0]
        if S.size and T.size:
            i = rng.choice(S)
            j = rng.choice(T)
        else:
            # fallback random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # swap values
        h[i], h[j] = h[j], h[i]
    return h

        h_bin[idx] = 1.0
        c5_val = compute_c5(h_bin)
        best_h = h_bin

    binary_deadline = start_time + budget_s * 0.6

    while time.time() < binary_deadline:

        c5_val = compute_c5(h)
