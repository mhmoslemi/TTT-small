import time
import numpy as np
from scipy.optimize import minimize

# ---------- Constraints ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ---------- Helpers ----------
def compute_c5(h):
    """Maximum overlap integral for a given discretised h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def generate_initial_h(n, M, rng):
    """Feasible h (0≤h≤1, sum=M) with simple random constructions."""
    method = rng.integers(3)
    if method == 0:      # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:    # Dirichlet‑like continuous
        h = rng.random(n)
        h = h / np.sum(h) * M
        np.clip(h, 0.0, 1.0, out=h)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    else:                # linear decreasing shape
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        np.clip(h, 0.0, 1.0, out=h)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    return h

def project_sum(h, target):
    """Adjust h so that its sum equals target while staying in [0,1]."""
    cur = np.sum(h)
    if abs(cur - target) > 1e-12:
        h += (target - cur) / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

# ---------- Main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # decide discretisation size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        if n % 2 != 0:
            # ensure even for integer half‑mass; keep original n otherwise
            pass
    else:
        n = 80
    M = n / 2.0          # exact target sum (may be fractional)

    # initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_sum(h0, M)
    else:
        h0 = np.full(n, 0.5)

    # ---------- SLSQP local optimisation ----------
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf

    slsqp_budget = min(0.15 * budget_s, budget_s - 30.0)
    slsqp_start = time.time()
    while time.time() - slsqp_start < slsqp_budget - 0.2:
        # random start for each restart
        h_start = generate_initial_h(n, int(np.round(M)), rng)
        h_start = project_sum(h_start, M)

        t0 = compute_c5(h_start)
        x0 = np.concatenate([h_start, [t0]])

        try:
            res = minimize(fun=lambda x: x[-1],
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 800, 'ftol': 1e-10, 'disp': False})
        except Exception:
            continue
        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_sum(h_opt, M)
        c5_opt = compute_c5(h_opt)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            if best_c5 < 0.3808:
                break

    if best_h is None:
        best_h = h0
        best_c5 = compute_c5(best_h)

    # ---------- Simulated annealing refinement ----------
    cur_h = best_h.copy()
    cur_c5 = best_c5
    best_local_h = cur_h.copy()
    best_local_c5 = cur_c5

    # temperature schedule parameters
    T_start = 0.02
    T_end = 0.0005

    while time.time() - start_time < budget_s - 0.5:
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        T = T_start * (1 - frac) + T_end * frac

        # occasional smoothing move
        if rng.random() < 0.015:
            smooth_h = 0.5 * cur_h + 0.25 * np.roll(cur_h, 1) + 0.25 * np.roll(cur_h, -1)
            smooth_h = np.clip(smooth_h, 0.0, 1.0)
            smooth_h = project_sum(smooth_h, M)
            new_h = smooth_h
        else:
            i, j = rng.choice(n, 2, replace=False)
            inc, dec = (i, j) if rng.random() < 0.5 else (j, i)
            max_inc = min(1.0 - cur_h[inc], cur_h[dec])
            if max_inc <= 0:
                continue
            # step size scales with temperature (larger early)
            delta = rng.random() * max_inc * (0.5 + 0.5 * (1 - frac))
            new_h = cur_h.copy()
            new_h[inc] += delta
            new_h[dec] -= delta

        new_c5 = compute_c5(new_h)
        diff = new_c5 - cur_c5
        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-14)):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_local_c5:
                best_local_c5 = new_c5
                best_local_h = new_h.copy()
                if best_local_c5 < 0.3808:
                    break

    # ---------- Final greedy pairwise refinement ----------
    steps = [0.1, 0.05, 0.02, 0.01, 0.005]
    for step in steps:
        for _ in range(12 * n):
            if time.time() - start_time > budget_s - 0.2:
                break
            i, j = rng.choice(n, 2, replace=False)
            inc, dec = (i, j) if rng.random() < 0.5 else (j, i)
            max_inc = min(1.0 - best_local_h[inc], best_local_h[dec])
            if max_inc <= 0:
                continue
            delta = step * max_inc
            trial_h = best_local_h.copy()
            trial_h[inc] += delta
            trial_h[dec] -= delta
            trial_c5 = compute_c5(trial_h)
            if trial_c5 + 1e-12 < best_local_c5:
                best_local_h = trial_h
                best_local_c5 = trial_c5

    # enforce constraints one last time
    final_h = np.clip(best_local_h, 0.0, 1.0)
    final_h = project_sum(final_h, M)
    final_c5 = compute_c5(final_h)

    return final_h, float(final_c5), n
