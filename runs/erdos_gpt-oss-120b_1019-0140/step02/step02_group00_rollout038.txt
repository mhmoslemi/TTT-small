import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets contain solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def binary_hill_climb(h0, time_limit, rng):
    """
    Targeted hill‑climbing / simulated‑annealing on a balanced binary vector.
    Returns the best binary vector found and its C5 value.
    """
    n = h0.size
    current = h0.copy()
    current_score = compute_c5(current)
    best = current.copy()
    best_score = current_score

    T = 0.02          # initial temperature
    cooling = 0.9995
    min_T = 1e-6
    shift_vals = np.arange(-n + 1, n)

    while time.time() < time_limit:
        # correlation of h with itself
        corr_self = np.correlate(current.astype(float), current.astype(float), mode='full')
        mask = shift_vals != 0
        min_corr = np.min(corr_self[mask])
        worst_idxs = np.where((corr_self == min_corr) & mask)[0]
        s = shift_vals[rng.choice(worst_idxs)]

        # overlapping region for this shift
        if s >= 0:
            start, end = 0, n - s
        else:
            start, end = -s, n
        if end - start < 2:
            continue

        idx_range = np.arange(start, end)
        j_range = idx_range + s
        hi = current[idx_range]
        hj = current[j_range]

        A = idx_range[(hi == 1) & (hj == 0)]   # 1‑0 pattern
        B = idx_range[(hi == 0) & (hj == 1)]   # 0‑1 pattern

        if A.size > 0 and B.size > 0:
            i_swap = rng.choice(A)
            j_swap = rng.choice(B)
        else:
            # fallback: any random 1‑0 swap
            ones = np.where(current == 1)[0]
            zeros = np.where(current == 0)[0]
            i_swap = rng.choice(ones)
            j_swap = rng.choice(zeros)

        cand = current.copy()
        cand[i_swap] = 0
        cand[j_swap] = 1
        cand_score = compute_c5(cand)

        if cand_score < current_score - 1e-12:
            accept = True
        else:
            delta = cand_score - current_score
            accept = (T > 0) and (rng.random() < np.exp(-delta / T))

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                if best_score <= 0.3808:
                    break

        T = max(min_T, T * cooling)

    return best, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    # target resolution
    n = 256
    # ----- build initial binary candidate -----
    if 'initial_h_values' in globals():
        init_small = np.asarray(initial_h_values, dtype=float)
        m = init_small.size
        if n % m == 0:
            factor = n // m
            # binarise the small vector by top‑half values
            top = np.argpartition(-init_small, m // 2 - 1)[: m // 2]
            small_bin = np.zeros(m, dtype=int)
            small_bin[top] = 1
            h0_bin = np.repeat(small_bin, factor)
        else:
            x_old = np.linspace(0, 1, m, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            inter = np.interp(x_new, x_old, init_small)
            top = np.argpartition(-inter, n // 2 - 1)[: n // 2]
            h0_bin = np.zeros(n, dtype=int)
            h0_bin[top] = 1
    else:
        h0_bin = np.zeros(n, dtype=int)
        h0_bin[rng.choice(n, size=n // 2, replace=False)] = 1

    # ----- Phase 1: binary hill‑climbing (≈30 % of budget) -----
    phase1_end = time.time() + time_left() * 0.30
    best_bin, best_score = binary_hill_climb(h0_bin, phase1_end, rng)

    # ----- Phase 2: gradient‑descent refinement on the real domain -----
    remaining = time_left()
    if remaining > 5.0:
        # allocate the rest (but keep a small safety margin)
        iters = int(max(2000, min(50000, remaining * 800)))
        h_float = best_bin.astype(float)
        refined, refined_score = gradient_descent_momentum(h_float, n_iter=iters,
                                                          step0=0.5, beta=0.9)
        final_score = compute_c5(refined)
        if final_score < best_score:
            best_bin = refined
            best_score = final_score

    # final feasibility projection and score
    best_bin = project_onto_box_sum(best_bin, n / 2.0)
    best_score = compute_c5(best_bin)

    return best_bin.astype(float), best_score, n
