def random_start_vector(n, rng):
    v = rng.random(n)
    # Ensure sum = target via scaling: multiply by target / sum(v)
    target = n / 2.0
    total = v.sum()
    v = v * (target / total)
    # Clip to [0,1].
    v = np.clip(v, 0.0, 1.0)
    # Fix small sum error via loop
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute diff over entries with slack.
        # We'll increment/decrement earliest entries that aren't at bound.
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and v[i] < 1.0:
                inc = min(diff, 1.0 - v[i])
                v[i] += inc
                diff -= inc
            elif diff < 0 and v[i] > 0.0:
                dec = min(-diff, v[i])
                v[i] -= dec
                diff += dec
    return v

def hill_climb(h_start, deadline, rng):
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    # stagnation counter
    no_improve_iters = 0
    while time.time() < deadline - 0.1:
        # compute correlation and find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # compute sum_nb
        sum_nb = compute_sum_nb(h, k)
        # compute epsilon and slack
        eps_i = h
        slack_j = 1 - h
        # compute delta matrix
        D = sum_nb[:, None] - sum_nb[None, :]
        eps_mat = np.minimum(eps_i[:, None], slack_j[None, :])
        # exclude diagonal
        np.fill_diagonal(eps_mat, 0.0)
        # also perhaps only consider eps_mat > 0
        # compute delta = eps_mat * D
        delta_mat = eps_mat * D
        # we want negative delta (reduce overlap)
        # find minimal delta
        min_delta = np.min(delta_mat)
        if min_delta >= -1e-12:
            # no improving pair found, break or random perturb
            # try random perturb
            # if we have stagnation > threshold, break
            no_improve_iters += 1
            if no_improve_iters > 20:
                break
            # random small move: choose i with h[i]>1e-4, j with h[j] < 1-1e-4, epsilon small fraction
            i_candidates = np.where(h > 1e-4)[0]
            j_candidates = np.where(h < 1 - 1e-4)[0]
            if len(i_candidates)==0 or len(j_candidates)==0:
                break
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
            eps = min(h[i], 1 - h[j], 0.01)  # small step
            h[i] -= eps
            h[j] += eps
            # compute new val
            new_val = compute_c5(h)
            if new_val < cur_val:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
                no_improve_iters = 0
            else:
                # revert
                h[i] += eps
                h[j] -= eps
            continue
        # get best i,j
        idx_pair = np.argmin(delta_mat)
        i, j = np.unravel_index(idx_pair, (n, n))
        eps = eps_mat[i, j]
        if eps <= 0:
            # Should not happen
            break
        # apply move
        h[i] -= eps
        h[j] += eps
        # maybe due to floating error clip
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        new_val = compute_c5(h)
        if new_val <= cur_val + 1e-12:
            cur_val = new_val

                best_val = new_val

            h[j] -= eps
            # no improvement, break or try other?
            no_improve_iters += 1
            if no_improve_iters > 20:
                break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)
    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        cur = h0.sum()
        if abs(cur - target) > 1e-12:
            # Adjust to meet sum exactly by scaling then fixing
            h0 = h0 * (target / cur)
            # Clip & fix
            h0 = np.clip(h0, 0.0, 1.0)
            # Fix sum by simple loop
            cur = h0.sum()
            diff = target - cur
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h0[i] < 1.0:
                    inc = min(diff, 1.0 - h0[i])
                    h0[i] += inc
                    diff -= inc
                elif diff < 0 and h0[i] > 0.0:
                    dec = min(-diff, h0[i])
                    h0[i] -= dec
                    diff += dec
    else:
        n = 80
        target = n / 2.0
        h0 = random_start_vector(n, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Hill climb from initial
    best_h, best_val = hill_climb(best_h, deadline, rng)

    # Random restarts
    # while time left, generate random start vector, apply hill_climb, update best
    while time.time() < deadline - 0.5:
        # generate new start
        h_start = random_start_vector(n, rng)
        # optional: start with some jitter near best
        # We'll just use random start, but can also perturb best occasionally
        # Apply hill climb with reduced time budget (e.g., 0.2 sec)
        sub_deadline = time.time() + 0.5  # limited per hill climb
        h_opt, val_opt = hill_climb(h_start, min(deadline, sub_deadline), rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # Ensure final vector respects constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        # adjust first element to correct sum
        best_h[0] += target - cur
        best_h[0] = np.clip(best_h[0], 0.0, 1.0)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n

def compute_sum_nb(h, k):
    """
    For a given shift k (integer), compute array sigma_i = h_{i+k} + h_{i-k},
    where out-of-range indices contribute 0.
    """
    n = h.shape[0]
    sigma = np.zeros(n)
    if k >= 0:
        # forward term: h[i+k] for i from 0 to n-k-1
        if k < n:
            sigma[:n - k] = h[k:]
            # backward term: h[i-k] for i from k to n-1 => indices i-k from 0 to n-k-1
            sigma[k:] += h[:n - k]
    else:
        kk = -k
        if kk < n:
            # forward term: h[i+ k] = h[i-kk] for i>=kk
            sigma[kk:] = h[:n - kk]
            # backward term: h[i - k] = h[i+kk] for i from 0 to n-kk-1
            sigma[:n - kk] += h[kk:]
    return sigma

def hill_climb(h_start, deadline, rng):
    h = h_start.copy()
    n = h.shape[0]
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    no_improve = 0
    max_no_improve = 30
    iter_cnt = 0
    max_iter = 10000
    while time.time() < deadline - 0.01 and iter_cnt < max_iter:
        iter_cnt += 1
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        sigma = compute_sum_nb(h, k)
        # compute eps and slack
        eps_i = h
        slack_j = 1.0 - h
        # compute D and eps_mat
        D = sigma[:, None] - sigma[None, :]
        eps_mat = np.minimum(eps_i[:, None], slack_j[None, :])
        # zero diagonal
        np.fill_diagonal(eps_mat, 0.0)
        # compute delta
        delta = eps_mat * D
        min_delta = np.min(delta)
        if min_delta >= -1e-12:
            # no improving swap
            no_improve += 1
            if no_improve >= max_no_improve:
                break
            # random small perturbation
            i_candidates = np.where(h > 1e-8)[0]
            j_candidates = np.where(h < 1 - 1e-8)[0]
            if i_candidates.size == 0 or j_candidates.size == 0:
                break
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
            eps = min(h[i], 1.0 - h[j], 0.01)
            if eps <= 0:
                continue
            h[i] -= eps
            h[j] += eps
            new_val = compute_c5(h)
            if new_val < cur_val:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
                no_improve = 0
            else:
                h[i] += eps
                h[j] -= eps
            continue
        # find best pair
        idx_pair = np.argmin(delta)
        i, j = np.unravel_index(idx_pair, (n, n))
        eps = eps_mat[i, j]
        if eps <= 0:
            # shouldn't happen
            break
        # perform move
        h[i] -= eps
        h[j] += eps
        # clip for safety
        if h[i] < 0.0:
            h[i] = 0.0
        elif h[i] > 1.0:
            h[i] = 1.0
        if h[j] < 0.0:
            h[j] = 0.0
        elif h[j] > 1.0:
            h[j] = 1.0
        new_val = compute_c5(h)
        if new_val <= cur_val + 1e-12:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
            no_improve = 0
        else:
            # revert
            h[i] += eps
            h[j] -= eps
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    return best_h, best_val

import time
import numpy as np

def compute_c5(h):
    """
    Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation.
    """
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def compute_sum_nb(h, k):
    """
    For a given integer shift k, compute sigma_i = h_{i+k} + h_{i-k},
    where out‑of‑range indices contribute 0.
    """
    n = h.shape[0]
    sigma = np.zeros(n)
    if k >= 0:
        if k < n:
            sigma[: n - k] = h[k:]
            sigma[k:] += h[: n - k]
    else:
        kk = -k
        if kk < n:
            sigma[kk:] = h[: n - kk]
            sigma[: n - kk] += h[kk:]
    return sigma

def random_start_vector(n, rng):
    """
    Produce a random feasible vector with sum n/2 and entries in [0,1].
    Half the time it returns a binary vector.
    """
    if rng.random() < 0.5:
        v = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        v[idx] = 1.0
        return v
    v = rng.random(n)
    target = n / 2.0
    total = v.sum()
    if total > 0:
        v = v * (target / total)
    else:
        v = np.full(n, target / n)
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        v[0] = np.clip(v[0] + diff, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(1, n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    inc = min(diff, 1.0 - v[i])
                    v[i] += inc
                    diff -= inc
                elif diff < 0 and v[i] > 0.0:
                    dec = min(-diff, v[i])
                    v[i] -= dec
                    diff += dec
    return v

def hill_climb(h_start, deadline, rng):
    """
    Greedy pairwise mass‑transfer search guided by the worst shift.
    """
    h = h_start.copy()
    n = h.shape[0]
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    no_improve = 0
    max_no_improve = 30
    it = 0
    max_iter = 10000
    while time.time() < deadline - 0.01 and it < max_iter:
        it += 1
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        sigma = compute_sum_nb(h, k)

        eps_i = h                     # amount we can take from i
        slack_j = 1.0 - h             # amount we can add to j

        D = sigma[:, None] - sigma[None, :]          # sigma_i - sigma_j
        eps_mat = np.minimum(eps_i[:, None], slack_j[None, :])
        np.fill_diagonal(eps_mat, 0.0)               # ignore i=j
        delta = eps_mat * D                           # change in worst overlap

        min_delta = np.min(delta)
        if min_delta >= -1e-12:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
            # small random perturbation
            i_cand = np.where(h > 1e-8)[0]
            j_cand = np.where(h < 1 - 1e-8)[0]
            if i_cand.size == 0 or j_cand.size == 0:
                break
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
            eps = min(h[i], 1.0 - h[j], 0.01)
            if eps <= 0:
                continue
            h[i] -= eps
            h[j] += eps
            new_val = compute_c5(h)
            if new_val < cur_val:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
                no_improve = 0
            else:
                h[i] += eps
                h[j] -= eps
            continue

        # best improving pair
        idx_pair = np.argmin(delta)
        i, j = np.unravel_index(idx_pair, (n, n))
        eps = eps_mat[i, j]
        if eps <= 0:
            break

        h[i] -= eps
        h[j] += eps
        # safety clipping
        h[i] = min(max(h[i], 0.0), 1.0)
        h[j] = min(max(h[j], 0.0), 1.0)

        new_val = compute_c5(h)
        if new_val <= cur_val + 1e-12:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
            no_improve = 0
        else:
            # revert
            h[i] += eps
            h[j] -= eps
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        cur = h0.sum()
        if abs(cur - target) > 1e-12:
            if cur > 0:
                h0 = h0 * (target / cur)
            else:
                h0 = np.full(n, target / n)
            h0 = np.clip(h0, 0.0, 1.0)
            cur = h0.sum()
            diff = target - cur
            if abs(diff) > 1e-12:
                for i in range(n):
                    if abs(diff) < 1e-12:
                        break
                    if diff > 0 and h0[i] < 1.0:
                        inc = min(diff, 1.0 - h0[i])
                        h0[i] += inc
                        diff -= inc
                    elif diff < 0 and h0[i] > 0.0:
                        dec = min(-diff, h0[i])
                        h0[i] -= dec
                        diff += dec
    else:
        n = 80
        h0 = random_start_vector(n, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # local optimisation from the initial candidate
    best_h, best_val = hill_climb(best_h, deadline, rng)

    # additional random restarts
    while time.time() < deadline - 0.5:
        h_start = random_start_vector(n, rng)
        sub_deadline = min(deadline, time.time() + 0.5)
        h_opt, val_opt = hill_climb(h_start, sub_deadline, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # final feasibility correction
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h[0] = np.clip(best_h[0] + (target - cur), 0.0, 1.0)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
