def hill_climb_binary(h, rng, time_limit):
    start = time.time()
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    cur = h.copy()
    cur_val = best_val
    # indices
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    iter = 0
    while time.time() - start < time_limit:
        iter += 1
        # temperature schedule: linear cooling
        temp = max(1e-4, 0.5 * (1 - (time.time() - start) / time_limit))
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur[i], cur[j] = 0.0, 1.0
        new_val = compute_c5(cur)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            # accept
            cur_val = new_val
            # update indices
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
        else:
            # revert swap
            cur[i], cur[j] = 1.0, 0.0
        # maybe periodically recompute indices instead of each iteration for speed, but okay
    return best, best_val
```

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def thue_morse_binary(n):
    # returns binary array of length n (0/1) for power-of-two n
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def hill_climb_binary(start_h, rng, time_limit):
    """Binary (0/1) hill climbing with swaps, returns best (h, c5)."""
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    while time.time() - start < time_limit:
        # pick random i with value 1 and j with value 0
        i = rng.integers(n)
        if cur[i] != 1.0:
            continue
        j = rng.integers(n)
        if cur[j] != 0.0:
            continue
        # swap
        cur[i] = 0.0
        cur[j] = 1.0
        new_val = compute_c5(cur)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # revert
            cur[i] = 1.0
            cur[j] = 0.0
    return best, best_val

def random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_and_fix(arr, n, rng):
    """Threshold to 0/1 and enforce exactly n/2 ones."""
    h = (np.array(arr, dtype=float) >= 0.5).astype(float)
    cur = int(np.sum(h))
    target = n // 2
    if cur > target:
        ones = np.where(h == 1.0)[0]
        rng.shuffle(ones)
        h[ones[:cur - target]] = 0.0
    elif cur < target:
        zeros = np.where(h == 0.0)[0]
        rng.shuffle(zeros)
        h[zeros[:target - cur]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate discretisations (even)
    candidate_ns = [80, 96, 112, 128, 144, 160, 176, 192, 208, 224, 240, 256]

    # prioritize parent size if available
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 not in candidate_ns:
            candidate_ns.insert(0, n0)
        elif n0 % 2 == 0:
            candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        # allocate up to 120 s or half of remaining time for this n
        alloc = min(120.0, remaining * 0.5)
        init_sols = []

        # parent solution if size matches
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            init_sols.append(binarize_and_fix(initial_h_values, n, rng))

        # Thue‑Morse for powers of two
        if n & (n - 1) == 0:
            init_sols.append(thue_morse_binary(n))

        # a few random binary starts
        for _ in range(2):
            init_sols.append(random_binary(n, rng))

        # evaluate each entry once and keep the best as starting point
        per_init_time = alloc / max(1, len(init_sols))
        best_h_n = None
        best_c5_n = np.inf

        for h0 in init_sols:
            # quick evaluation
            val0 = compute_c5(h0)
            if val0 < best_c5_n:
                best_c5_n = val0
                best_h_n = h0.copy()
            # hill climbing
            h_opt, c5_opt = hill_climb_binary(h0, rng, per_init_time)
            if c5_opt < best_c5_n:
                best_c5_n = c5_opt
                best_h_n = h_opt.copy()

        if best_h_n is not None and best_c5_n < best_c5:
            best_c5 = best_c5_n
            best_h = best_h_n.copy()
            best_n = n
            if best_c5 < 0.3808:
                break

    # final continuous fine‑tuning (mass‑transfer) if time permits
    if best_h is not None:
        n = best_n
        M = n // 2
        h = best_h.copy()
        cur_val = best_c5
        while time.time() - start_time < budget_s - 1.0:
            i = rng.integers(n)
            j = rng.integers(n)

            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta

    cur = np.sum(best_h)
    target = best_n // 2
    if abs(cur - target) > 1e-9:
        best_h += (target - cur) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
