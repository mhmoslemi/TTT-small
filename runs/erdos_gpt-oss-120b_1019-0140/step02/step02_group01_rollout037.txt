import time
import numpy as np

# ---------- Utility ----------
def compute_c5(h):
    """Maximum overlap integral for a given (continuous) h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def random_binary(n, m, rng):
    """Binary vector of length n with exactly m ones."""
    h = np.zeros(n, dtype=np.int8)
    h[rng.choice(n, m, replace=False)] = 1
    return h

# ---------- Simulated Annealing on binary vectors ----------
def simulated_annealing_h(n, m, rng, time_budget, init_h):
    """Binary SA that tries to minimise C5 within time_budget seconds."""
    h = init_h.copy()
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode="full")
    cur_c5 = O.max() * dx
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    steps_since_best = 0

    while time.time() - start < time_budget - 0.01:
        # choose a shift attaining the current maximum overlap
        max_val = O.max()
        max_idxs = np.where(O >= max_val - 1e-12)[0]
        s_idx = rng.choice(max_idxs)
        s = s_idx - (n - 1)

        # construct shifted view for this s
        if s >= 0:
            h_shift = np.concatenate((h[s:], np.zeros(s, dtype=np.int8)))
        else:
            sneg = -s
            h_shift = np.concatenate((np.zeros(sneg, dtype=np.int8), h[: n - sneg]))

        type10 = np.where((h == 1) & (h_shift == 0))[0]
        type01 = np.where((h == 0) & (h_shift == 1))[0]

        if type10.size and type01.size:
            i = rng.choice(type10)
            j = rng.choice(type01)
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # trial swap
        h_trial = h.copy()
        h_trial[i] = 0
        h_trial[j] = 1
        O_trial = np.correlate(h_trial, 1 - h_trial, mode="full")
        c5_trial = O_trial.max() * dx
        delta = c5_trial - cur_c5

        # temperature schedule (linear cooling)
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T0 = 1e-3
        T = T0 * max(0.0, 1.0 - frac)

        accept = False
        if delta < -1e-12:
            accept = True
        elif T > 0.0:
            prob = np.exp(-delta / T) if delta > 0 else 1.0
            if rng.random() < prob:
                accept = True

        if accept:
            h = h_trial
            O = O_trial
            cur_c5 = c5_trial
            steps_since_best = 0
            if cur_c5 + 1e-12 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            steps_since_best += 1

        # occasional restart to avoid stagnation
        if steps_since_best > 20000:
            h = random_binary(n, m, rng)
            O = np.correlate(h, 1 - h, mode="full")
            cur_c5 = O.max() * dx
            steps_since_best = 0

        if best_c5 < 0.3808:
            break

    return best_h, best_c5

# ---------- Continuous fine‑tuning ----------
def continuous_refine(h, rng, time_budget):
    """Small mass‑transfer hill‑climbing on a (float) h."""
    n = len(h)
    h = h.astype(float)
    cur_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_budget - 0.01:
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < cur_c5:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < 0.3808:
                break
    return h, cur_c5

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate discretisations (even numbers)
    candidate_ns = [80, 96, 112, 128, 144, 160, 192, 224, 256]

    # prioritize parent size if available
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_total > budget_s - 5.0:
            break
        remaining = budget_s - (time.time() - start_total)
        # allocate up to 150 s or half of the remaining budget, whichever is smaller
        alloc = min(150.0, remaining * 0.5)

        m = n // 2  # required sum of h

        # initialise starting binary vector
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            parent = np.array(initial_h_values, dtype=np.float64)
            # binary projection: keep the largest m entries as 1
            idx_sorted = np.argsort(parent)[::-1]
            h0 = np.zeros(n, dtype=np.int8)
            h0[idx_sorted[:m]] = 1
        else:
            h0 = random_binary(n, m, rng)

        # binary simulated annealing
        h_opt, c5_opt = simulated_annealing_h(n, m, rng, alloc, h0)

        # continuous refinement (quick, using remaining time slice)
        time_left = budget_s - (time.time() - start_total)
        if time_left > 0.5:
            # give at most 30 s for refinement
            refine_budget = min(30.0, time_left * 0.5)
            h_refined, c5_refined = continuous_refine(h_opt.astype(float), rng, refine_budget)
            if c5_refined + 1e-12 < c5_opt:
                h_opt, c5_opt = h_refined, c5_refined

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
            if best_c5 < 0.3808:
                break

    # final safety fallback
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)

    # final constraint enforcement (clip, sum correction)
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    target_sum = best_n // 2
    if cur_sum != target_sum:
        diff = target_sum - cur_sum
        if diff > 0:
            zero_idx = np.where(best_h == 0)[0]
            if zero_idx.size >= diff:
                best_h[rng.choice(zero_idx, diff, replace=False)] = 1.0
        else:
            one_idx = np.where(best_h == 1)[0]
            if one_idx.size >= -diff:
                best_h[rng.choice(one_idx, -diff, replace=False)] = 0.0

    return best_h.astype(float), float(best_c5), int(best_n)
