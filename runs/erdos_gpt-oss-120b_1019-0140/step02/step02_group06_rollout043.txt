import time
import numpy as np

def compute_c5(h):
    """Return the overlap integral C5 for array h."""
    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    return float(v.max()) * dx

def max_overlap(h):
    """Maximum count of h[i]*(1-h[i+k]) over all shifts."""
    return int(np.max(np.correlate(h, 1 - h, mode="full")))

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise problem size and binary vector -----
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        target = n // 2
        # take the largest entries as ones to respect the mass constraint
        idx = np.argsort(-init)
        h = np.zeros(n, dtype=np.int8)
        h[idx[:target]] = 1
    else:
        n = 80
        target = n // 2
        perm = rng.permutation(n)
        h = np.zeros(n, dtype=np.int8)
        h[perm[:target]] = 1

    # ----- helper variables -----
    best_h = h.copy()
    best_max = max_overlap(best_h)
    target_max = 15          # target count giving C5 = 0.375
    T0 = 2.0                 # initial temperature for simulated annealing
    iteration = 0
    max_iter = 500000

    # ----- main simulated‑annealing loop -----
    while time.time() - start < budget_s and iteration < max_iter:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        iteration += 1
        T = max(0.001, T0 * (1.0 - elapsed / budget_s))

        # current correlation and shift of the worst overlap
        corr = np.correlate(h, 1 - h, mode="full")
        cur_max = int(corr.max())
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
            if best_max <= target_max:
                break
        max_idx = int(corr.argmax())
        shift = max_idx - (n - 1)   # can be negative

        # ----- propose a targeted swap -----
        if shift >= 0:
            tail = h[: n - shift]
            head = h[shift:]
        else:
            k2 = -shift
            tail = h[k2:]
            head = h[: n - k2]
        mism = np.where((tail == 1) & (head == 0))[0]

        if mism.size > 0:
            if shift >= 0:
                one_candidates = mism            # original positions of the 1s
            else:
                one_candidates = mism - shift   # map back to original indices
            i = int(rng.choice(one_candidates))
        else:
            # fallback: any 1
            i = int(rng.choice(np.flatnonzero(h == 1)))

        zero_positions = np.flatnonzero(h == 0)
        if zero_positions.size == 0:
            break  # should never happen

        # try to avoid swapping with the paired zero that caused the mismatch
        paired = None
        if shift >= 0 and i + shift < n:
            paired = i + shift
        elif shift < 0 and i + shift >= 0:
            paired = i + shift
        cand_zeros = zero_positions[zero_positions != i]
        if paired is not None:
            cand_zeros = cand_zeros[cand_zeros != paired]
        if cand_zeros.size == 0:
            cand_zeros = zero_positions[zero_positions != i]
        j = int(rng.choice(cand_zeros))

        # evaluate the swap
        h_candidate = h.copy()
        h_candidate[i], h_candidate[j] = h_candidate[j], h_candidate[i]
        new_max = max_overlap(h_candidate)
        delta = new_max - cur_max
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            h = h_candidate
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                if best_max <= target_max:
                    break

        # occasional random perturbation to escape plateaus
        if iteration % 5000 == 0:
            for _ in range(2):
                i1 = int(rng.choice(np.flatnonzero(h == 1)))
                j1 = int(rng.choice(np.flatnonzero(h == 0)))
                h[i1], h[j1] = h[j1], h[i1]

    # ----- final result -----
    final_c5 = compute_c5(best_h.astype(float))
    best_h_float = best_h.astype(float)
    return best_h_float, final_c5, n
