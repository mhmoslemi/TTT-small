def binary_greedy(h, target_corr, deadline, rng):
    n = len(h)
    w = h.sum()
    best_h = h.copy()
    best_corr = np.min(np.correlate(h, h, mode='full'))
    # Record best C5 also
    best_val = (w - best_corr) * (2.0 / n)
    # set move count
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, h, mode='full')
        min_corr = corr.min()
        # if we achieve target, break
        if min_corr >= target_corr:
            best_h = h.copy()
            best_corr = min_corr
            best_val = (w - min_corr) * (2.0 / n)
            break
        # find shift(s) achieving min_corr
        ks = np.where(corr == min_corr)[0]
        idx = rng.choice(ks)
        k = idx - (n - 1)
        # compute shifted h
        shifted = np.take(h, (np.arange(n) + k) % n)
        # find source i: h=1, shifted=0
        src = np.where((h == 1) & (shifted == 0))[0]
        # find destination j: h=0, shifted=1
        dst = np.where((h == 0) & (shifted == 1))[0]
        if src.size == 0 or dst.size == 0:
            # cannot improve for this shift, random swap for exploration
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # perform swap
            h[i] = 0
            h[j] = 1
            # evaluate new min_corr
            new_corr = np.min(np.correlate(h, h, mode='full'))
            if new_corr >= min_corr:
                # accept
                if new_corr > best_corr:
                    best_corr = new_corr
                    best_h = h.copy()
                continue
            else:
                # revert
                h[i] = 1
                h[j] = 0
                # maybe break
                continue
        # choose random pair
        i = rng.choice(src)
        j = rng.choice(dst)
        # swap
        h[i] = 0
        h[j] = 1
        new_corr = np.min(np.correlate(h, h, mode='full'))
        if new_corr >= min_corr:
            # accept
            if new_corr > best_corr:
                best_corr = new_corr
                best_h = h.copy()
        else:
            # revert
            h[i] = 1

        # maybe optionally break if best_corr meets target

import time, numpy as np, cvxpy as cp

def compute_c5(h):
    n = h.shape[0]; dx=2.0/n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    target = n/2.0
    v = rng.random(n)
    v = v - v.mean() + target/n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for i in idxs:
            if abs(diff) < 1e-12: break
            if diff > 0 and v[i] < 1.0:
                inc = min(diff, 1.0 - v[i])
                v[i] += inc; diff -= inc
            elif diff < 0 and v[i] > 0.0:
                dec = min(-diff, v[i])
                v[i] -= dec; diff += dec
    return v

def adjust_sum_to_target(h, target):

def binary_greedy(h_bin, w, target_corr, deadline, rng):
    n = h_bin.shape[0]
    h = h_bin.copy()
    best_h = h.copy()
    best_corr = np.min(np.correlate(h, h, mode='full'))
    best_val = (w - best_corr) * (2.0 / n)
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, h, mode='full')
        min_corr = corr.min()
        if min_corr >= target_corr:
            best_h = h.copy()
            best_corr = min_corr
            best_val = (w - min_corr) * (2.0 / n)
            break
        # choose a shift achieving min_corr
        ks = np.where(corr == min_corr)[0]
        idx = rng.choice(ks)
        k = idx - (n - 1)
        shifted = np.take(h, (np.arange(n) + k) % n)
        src = np.where((h == 1) & (shifted == 0))[0]
        dst = np.where((h == 0) & (shifted == 1))[0]
        if src.size == 0 or dst.size == 0:
            # random swap as fallback
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = rng.choice(src)
            j = rng.choice(dst)
        # swap
        h[i] = 0
        h[j] = 1
        new_corr = np.min(np.correlate(h, h, mode='full'))
        if new_corr >= min_corr:
            if new_corr > best_corr:
                best_corr = new_corr
                best_h = h.copy()
                best_val = (w - new_corr) * (2.0 / n)
        else:
            # revert
            h[i] = 1
            h[j] = 0
        # early stop if good enough
        if best_val < 0.3808:
            break
    return best_h, best_val

def continuous_anneal(h_start, best_val, start_time, deadline, rng):

    h = h_start.copy()
    cur_val = best_val
    best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # contribution array (as before)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # candidate destination
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:

            elapsed = time.time() - start_time

                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        if best < 0.3808:
            break
    return best_h, best

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and v[i] < 1.0:
                inc = min(diff, 1.0 - v[i])
                v[i] += inc
                diff -= inc
            elif diff < 0 and v[i] > 0.0:
                dec = min(-diff, v[i])
                v[i] -= dec
                diff += dec
    return v

def adjust_sum_to_target(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    n = h.shape[0]
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        idxs = np.arange(n)
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
    return h

def ccp_optimize(h0, w, deadline, rng, max_iter=200):
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(best_h)
    while time.time() < deadline - 0.5 and max_iter > 0:
        max_iter -= 1
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == w,
            t_var >= 0,
            t_var <= w,
        ]
        for k in range(1, n):
            grad = np.roll(h, -k) + np.roll(h, k)
            fk0 = np.dot(h, np.roll(h, k))
            cons.append(grad @ h_var >= t_var + fk0)
        prob = cp.Problem(cp.Maximize(t_var), cons)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=2000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(h_var.value, 0, 1)
        h_new = adjust_sum_to_target(h_new, w)
        val_new = compute_c5(h_new)
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_h = h_new.copy()
            if best_val < 0.3808:
                break
        h = h_new
    return best_h, best_val

def binary_greedy(h_bin, w, target_corr, deadline, rng):
    n = h_bin.shape[0]
    h = h_bin.copy()
    best_h = h.copy()
    best_corr = np.min(np.correlate(h, h, mode="full"))
    best_val = (w - best_corr) * (2.0 / n)
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, h, mode="full")
        min_corr = corr.min()
        if min_corr >= target_corr:
            best_h = h.copy()
            best_corr = min_corr
            best_val = (w - min_corr) * (2.0 / n)
            break
        ks = np.where(corr == min_corr)[0]
        idx = rng.choice(ks)
        k = idx - (n - 1)
        shifted = np.take(h, (np.arange(n) + k) % n)
        src = np.where((h == 1) & (shifted == 0))[0]
        dst = np.where((h == 0) & (shifted == 1))[0]
        if src.size == 0 or dst.size == 0:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = rng.choice(src)
            j = rng.choice(dst)
        h[i] = 0
        h[j] = 1
        new_corr = np.min(np.correlate(h, h, mode="full"))
        if new_corr >= min_corr:
            if new_corr > best_corr:
                best_corr = new_corr
                best_h = h.copy()
                best_val = (w - new_corr) * (2.0 / n)
        else:
            h[i] = 1
            h[j] = 0
        if best_val < 0.3808:
            break
    return best_h, best_val

def continuous_anneal(h_start, best_val, start_time, deadline, rng):
    h = h_start.copy()
    cur_val = best_val
    best_h = h.copy()
    best = best_val
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        if best < 0.3808:
            break
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():

    # Phase 1: successive linearization (CCP)

        if val_ccp < best_val:

    bin_deadline = deadline - 50.0
    if time.time() < bin_deadline:

        cur_w = h_bin.sum()
        diff = int(target - cur_w)
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            idx = rng.choice(zeros, diff, replace=False)
            h_bin[idx] = 1
        elif diff < 0:
            ones = np.where(h_bin == 1)[0]
            idx = rng.choice(ones, -diff, replace=False)
            h_bin[idx] = 0
        target_corr = 25  # enough to beat the 0.3808 bound for n=80

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_to_target(best_h, target)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
