def project_to_feasible(h):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    # Use a simple approach: add diff/n to all entries, then clip and repeat.
    # But better: use water-filling algorithm: adjust interior values.
    # Use while loop with tolerance.
    for _ in range(10):
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # adjust uniformly all entries
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def compute_c5(h):

import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # iterate to fix sum
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # adjust uniformly
            inc = diff / h.shape[0]
            h += inc
            h = np.clip(h, 0.0, 1.0)
        else:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
    return h

def upsample_array(arr, new_n):
    old_n = arr.shape[0]
    if new_n == old_n:
        return arr.copy()

    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-12:

            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
