    cur_h = start_h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.size
    # temperature schedule parameters

        j = rng.integers(n-1)

        # swap

                best_h = cur_h.copy()

            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    # initial bounds for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    return upsample(arr, new_n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def periodic_binary_balanced(n, period):
    """Binary pattern with period `period` containing period//2 ones per period."""
    if period <= 0 or n % period != 0:
        raise ValueError("period must divide n")
    ones = period // 2
    pattern = np.concatenate([np.ones(ones), np.zeros(period - ones)])
    return np.tile(pattern, n // period).astype(float)

def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_sa_optimize(start_h, rng, time_left):
    """Simulated annealing on binary vectors preserving the exact sum."""
    cur_h = start_h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.size
    it = 0
    max_it = 2000000  # safety cap
    T0 = 0.02
    T_end = 1e-5

    while time_left() > 0.5 and it < max_it:
        # linear temperature schedule
        T = T0 * (1.0 - it / max_it) + T_end * (it / max_it)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # problem sizes
    n_small = 256
    n_target = 512

    best_h_small = None
    best_score_small = np.inf

    # --- use parent construction if available ---
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_small = downsample(init_full, n_small)
        init_small = project_onto_box_sum(init_small, n_small / 2.0)
        # convert to binary by keeping the largest n/2 entries
        idx = np.argpartition(init_small, -n_small // 2)[-n_small // 2:]
        bin_start = np.zeros(n_small, dtype=float)
        bin_start[idx] = 1.0
        h_opt, sc_opt = binary_sa_optimize(bin_start, rng, time_left)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()

    # --- deterministic periodic seeds ---
    for period in (8, 16, 32):
        if time_left() <= 60:
            break
        cand = periodic_binary_balanced(n_small, period)
        h_opt, sc_opt = binary_sa_optimize(cand, rng, time_left)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()

    # --- random binary SA restarts ---
    while time_left() > 60:
        cand = random_binary_balanced(n_small, rng)
        h_opt, sc_opt = binary_sa_optimize(cand, rng, time_left)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()
        if best_score_small <= 0.3808:
            break

    if best_h_small is None:
        best_h_small = random_binary_balanced(n_small, rng)
        best_score_small = compute_c5(best_h_small)

    # --- upsample to target resolution ---
    h_big = upsample(best_h_small, n_target)
    h_big = project_onto_box_sum(h_big, n_target / 2.0)

    # --- momentum gradient refinement ---
    remaining = time_left()
    if remaining > 10.0:
        est_iters = int(min(50000, max(5000, remaining * 800)))
        h_ref, sc_ref = gradient_descent_momentum(h_big, n_iter=est_iters,
                                                  step0=0.5, beta=0.9)
        if sc_ref < best_score_small:
            best_h = h_ref
            best_score = sc_ref
        else:
            best_h = h_big
            best_score = compute_c5(h_big)
    else:
        best_h = h_big
        best_score = compute_c5(h_big)

    # --- final continuous SA fine‑tuning ---
    remaining = time_left()
    if remaining > 3.0:
        h_sa, sc_sa = continuous_sa_optimize(best_h, rng, time_left)
        if sc_sa < best_score:
            best_h = h_sa
            best_score = sc_sa

    # guard final feasibility
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
