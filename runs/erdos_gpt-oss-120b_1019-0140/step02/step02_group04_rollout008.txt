import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for step function values h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Scale and clip vec so that sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random vector in [0,1]^n with exact sum target_sum."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def random_binary(n, rng):
    """Return a binary vector with exactly n/2 ones."""
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


# ---------- Continuous Simulated Annealing ----------
def continuous_sa(h0, dx, target_sum, start_time, budget_s, rng):
    """Stochastic search on continuous [0,1] vectors preserving sum."""
    n = len(h0)
    cur_h = h0.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    T0 = 0.02
    total = budget_s
    while time.time() < start_time + budget_s:
        # pick donor and recipient
        i = rng.integers(n)
        if cur_h[i] <= 1e-9:
            continue
        j = rng.integers(n)
        if cur_h[j] >= 1.0 - 1e-9:
            continue
        max_delta = min(cur_h[i], 1.0 - cur_h[j], 0.1)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # propose
        cur_h[i] -= delta
        cur_h[j] += delta
        new_val = compute_c5_vec(cur_h, dx)
        dval = new_val - cur_val
        remaining = max(start_time + total - time.time(), 0.0)
        T = T0 * remaining / total
        accept = False
        if dval < 0:
            accept = True
        elif T > 1e-12 and rng.random() < np.exp(-dval / T):
            accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)
    return best_h, best_val


# ---------- Binary Simulated Annealing ----------
def binary_sa_hill_climb(initial_h, dx, start_time, budget_s, rng):
    """SA on binary vectors (0/1) with exact half density."""
    n = len(initial_h)
    cur_h = initial_h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    ones = set(np.flatnonzero(cur_h).tolist())
    zeros = set(range(n)) - ones
    T0 = 0.015
    total = budget_s
    max_no_imp = 5000
    no_imp = 0
    while time.time() < start_time + budget_s - 0.1:
        if not ones or not zeros:
            break
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = compute_c5_vec(cur_h, dx)
        dval = new_val - cur_val
        remaining = max(start_time + total - time.time(), 0.0)
        T = T0 * remaining / total
        accept = False
        if dval < 0:
            accept = True
        elif T > 1e-12 and rng.random() < np.exp(-dval / T):
            accept = True
        if accept:
            cur_val = new_val
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_imp += 1
        if no_imp >= max_no_imp:
            # perturb best a little and restart local search
            cur_h = best_h.copy()
            cur_val = best_val
            ones = set(np.flatnonzero(cur_h).tolist())
            zeros = set(range(n)) - ones
            # perform a handful of random swaps to escape
            for _ in range(5):
                if not ones or not zeros:
                    break
                i = rng.choice(list(ones))
                j = rng.choice(list(zeros))
                cur_h[i] = 0.0
                cur_h[j] = 1.0
                ones.remove(i)
                zeros.add(i)
                zeros.remove(j)
                ones.add(j)
            cur_val = compute_c5_vec(cur_h, dx)
            no_imp = 0
    return best_h, best_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        init_cont = np.array(initial_h_values, dtype=float)
        n = init_cont.shape[0]
    else:
        n = 100  # even dimension fallback
        init_cont = None

    dx = 2.0 / n
    target_sum = n / 2.0

    # Continuous SA (≈30% of budget)
    cont_budget = budget_s * 0.3
    if init_cont is not None:
        h0 = np.clip(init_cont, 0.0, 1.0)
        h0 = project_to_sum(h0, target_sum)
    else:
        h0 = generate_random_feasible(n, target_sum, rng)

    best_cont, best_cont_val = continuous_sa(
        h0, dx, target_sum, start_time, cont_budget, rng
    )

    # Convert to binary
    half = n // 2
    top_idx = np.argpartition(-best_cont, half)[:half]
    binary_h = np.zeros(n, dtype=float)
    binary_h[top_idx] = 1.0
    best_bin_val = compute_c5_vec(binary_h, dx)

    # Binary SA using remaining time
    time_used = time.time() - start_time
    remaining_budget = max(budget_s - time_used, 0.0)
    if remaining_budget > 1.0:
        best_h, best_val = binary_sa_hill_climb(
            binary_h, dx, start_time, remaining_budget, rng
        )
        # keep the better of the two binary results
        if best_bin_val < best_val:
            best_h, best_val = binary_h, best_bin_val
    else:
        best_h, best_val = binary_h, best_bin_val

    # Final safety projection (should be already feasible)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
