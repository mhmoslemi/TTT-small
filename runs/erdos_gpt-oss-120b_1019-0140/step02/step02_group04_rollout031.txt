def simulated_annealing_binary(h, dx, target_sum, time_budget, start_time, rng):

    cur = h.copy()
    cur_val = compute_c5_vec(cur.astype(float), dx)
    best = cur.copy()
    best_val = cur_val
    # Determine temperature schedule
    # Set initial T relative to cur_val (maybe 0.01)
    # Compute max steps based on time or while loop:
    while time.time() - start_time < time_budget:
        # propose swap
        ones = np.where(cur==1)[0]
        zeros = np.where(cur==0)[0]
        # if there are none, break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap:
        cur[i] = 0
        cur[j] = 1
        new_val = compute_c5_vec(cur.astype(float), dx)
        delta = new_val - cur_val
        # compute temperature based on elapsed proportion
        elapsed = time.time() - start_time
        # linear cooling: T = T0 * (1 - elapsed / time_budget)
        T0 = cur_val * 0.1  # 10% of initial objective maybe.
        T = T0 * max(0.0, 1.0 - elapsed / time_budget)
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.uniform() < np.exp(-delta / T)
        if accept:
            cur_val = new_val
            # update best if better
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
                # early exit if < target bound
                if best_val < 0.3808:
                    break
        else:
            # revert swap
            cur[i] = 1
            cur[j] = 0
        # maybe occasional random restart if many iterations without improvement
        # track no_improve count.
    return best, best_val

def binary_simulated_annealing(h_init, dx, target_sum, time_limit, start_time, rng):
    n = len(h_init)
    cur = h_init.copy()
    cur_val = compute_c5_vec(cur.astype(float), dx)
    best = cur.copy()
    best_val = cur_val
    # Temperature schedule
    T0 = cur_val * 0.05  # maybe 0.05 * cur_val
    # Have a counter for stagnation
    no_improve_counter = 0
    # Could maintain iteration count
    while time.time() < time_limit:
        # propose swap
        # pick two distinct indices i,j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if cur[i] == cur[j]:
            continue
        # swap values
        cur[i], cur[j] = cur[j], cur[i]
        new_val = compute_c5_vec(cur.astype(float), dx)
        delta = new_val - cur_val
        elapsed = time.time() - start_time
        # cooling: linear
        T = max(1e-12, T0 * (1.0 - (elapsed / (time_limit - start_time))))
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.uniform() < np.exp(-delta / T)
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
                no_improve_counter = 0
                # early termination if below target
                if best_val < 0.3808:
                    break
            else:
                no_improve_counter = 0
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]
            no_improve_counter += 1
        # occasional random restart after many non-improvements
        if no_improve_counter >= 5000:
            # restart current to random binary configuration
            cur = generate_random_binary(n, target_sum, rng)
            cur_val = compute_c5_vec(cur.astype(float), dx)
            no_improve_counter = 0
    return best.astype(float), best_val

def refine_continuous(h, dx, target_sum, time_limit, start_time, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.02 # initial transfer mass
    n = len(h)
    while time.time() < time_limit:
        # pick donor and recipient
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    # final projection
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def solve_binary_milp(n, target_sum, dx, time_limit, start_time, rng):
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    # Build problem
    h = cp.Variable(n, boolean=True)
    # Upper triangular matrix of w_ij
    W = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == target_sum]
    # Set lower triangle zero to avoid double counting
    for i in range(n):
        for j in range(i):
            constraints.append(W[i, j] == 0)
    # linking constraints for upper triangle
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(W[i, j] <= h[i])
            constraints.append(W[i, j] <= h[j])
            constraints.append(W[i, j] >= h[i] + h[j] - 1)
    # Overlap constraints for each shift k (1..n-1)
    for k in range(1, n):
        # sum over i=0..n-1-k of W[i, i+k]
        overlap = cp.sum(cp.hstack([W[i, i+k] for i in range(n - k)]))
        constraints.append(t >= (target_sum - overlap) * dx)
    # objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    # Set time limit if supported
    # Try CBC, then GLPK_MI
    sol = None
    sol_val = None
    for solver in [cp.CBC, cp.GLPK_MI]:
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
            h_val = np.rint(h.value).astype(int)  # round
            # ensure sum equals target (maybe adjust)
            if h_val.sum() != target_sum:
                # adjust via simple flip
                diff = target_sum - h_val.sum()
                if diff > 0:
                    zeros = np.where(h_val == 0)[0]
                    idx = rng.choice(zeros, size=diff, replace=False)
                    h_val[idx] = 1
                elif diff < 0:
                    ones = np.where(h_val == 1)[0]
                    idx = rng.choice(ones, size=-diff, replace=False)
                    h_val[idx] = 0
            c5 = compute_c5_vec(h_val.astype(float), dx)
            sol = h_val.astype(float)
            sol_val = c5
            break
    return sol, sol_val

def binary_simulated_annealing(h_init, dx, target_sum, time_limit, start_time, rng):
    cur = h_init.copy()
    cur_val = compute_c5_vec(cur.astype(float), dx)
    best = cur.copy()
    best_val = cur_val
    n = len(cur)
    # Temperature schedule
    # Determine initial temperature based on initial value
    # we set T0 = cur_val * 0.1 maybe
    T0 = cur_val * 0.1
    # Initialize counters
    no_improve_counter = 0
    while time.time() < time_limit:
        # generate swap
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # if same bits, skip
        if cur[i] == cur[j]:
            continue
        # apply swap
        cur[i], cur[j] = cur[j], cur[i]
        new_val = compute_c5_vec(cur.astype(float), dx)
        delta = new_val - cur_val
        elapsed = time.time() - start_time
        # Compute temperature
        T = max(1e-12, T0 * (1.0 - (elapsed / (time_limit - start_time))))
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.uniform() < np.exp(-delta / T)
        if accept:
            cur_val = new_val
            # update best
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
                no_improve_counter = 0
                if best_val < 0.3808:
                    break
            else:
                no_improve_counter = 0
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]
            no_improve_counter += 1
        # random restart condition
        if no_improve_counter >= 5000:
            cur = generate_random_binary(n, target_sum, rng)
            cur_val = compute_c5_vec(cur.astype(float), dx)
            no_improve_counter = 0
    return best.astype(float), best_val

def refine_continuous(h, dx, target_sum, time_limit, start_time, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.02
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    # final projection
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    # Greedy correction
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec

import time
import numpy as np


def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Adjust vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_binary(n, target_sum, rng):
    """Random binary vector with exactly target_sum ones."""
    idx = rng.choice(n, size=target_sum, replace=False)
    vec = np.zeros(n, dtype=int)
    vec[idx] = 1
    return vec


def binary_simulated_annealing(h_init, dx, target_sum, time_limit, start_time, rng):
    """Simulated annealing on a binary vector (0/1)."""
    cur = h_init.copy()
    cur_val = compute_c5_vec(cur.astype(float), dx)
    best = cur.copy()
    best_val = cur_val
    n = len(cur)
    T0 = cur_val * 0.1  # initial temperature
    no_improve = 0

    while time.time() < time_limit:
        # propose a swap between a 1 and a 0
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if cur[i] == cur[j]:
            continue
        # perform swap
        cur[i], cur[j] = cur[j], cur[i]
        new_val = compute_c5_vec(cur.astype(float), dx)
        delta = new_val - cur_val
        elapsed = time.time() - start_time
        T = max(1e-12, T0 * (1.0 - (elapsed / (time_limit - start_time))))
        accept = delta < 0 or (rng.random() < np.exp(-delta / T))
        if accept:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best = cur.copy()
                no_improve = 0
                if best_val < 0.3808:
                    break
            else:
                no_improve = 0
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]
            no_improve += 1

        if no_improve >= 5000:
            cur = generate_random_binary(n, target_sum, rng)
            cur_val = compute_c5_vec(cur.astype(float), dx)
            no_improve = 0

    return best.astype(float), best_val


def refine_continuous(h, dx, target_sum, time_limit, start_time, rng):
    """Local coordinate descent moving small mass between entries."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.02
    n = len(h)

    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99

    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val
