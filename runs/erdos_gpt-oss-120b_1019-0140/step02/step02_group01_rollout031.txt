def generate_smooth_h(n, M, rng):
    raw = rng.random(n)
    kernel = np.ones(5)/5
    smooth = np.convolve(raw, kernel, mode='same')
    # normalize to sum M
    smooth = smooth / np.sum(smooth) * M
    smooth = np.clip(smooth, 0, 1)
    # adjust sum if needed similar to earlier
    cur = np.sum(smooth)
    if cur < M:
        deficit = M - cur
        for i in np.argsort(smooth):
            inc = min(deficit, 1.0 - smooth[i])
            smooth[i] += inc
            deficit -= inc
            if deficit < 1e-12:
                break
    elif cur > M:
        excess = cur - M
        for i in np.argsort(-smooth):
            dec = min(excess, smooth[i])
            smooth[i] -= dec
            excess -= dec
            if excess < 1e-12:
                break
    return smooth

def simulated_annealing_refine(h, c5, time_budget, rng):

   start = time.time()

   n = len(h)
   M = n // 2
   best_h = h.copy()
   best_c5 = c5

   # parameters
   T0 = 0.01  # initial temperature
   decay = 0.9995  # per iteration

   temp = T0

   while time.time() - start < time_budget:
       # choose random i,j
       i, j = rng.choice(n, 2, replace=False)
       # compute max transferable
       max_transfer = min(1.0 - best_h[i], best_h[j])
       if max_transfer <= 0:
           continue
       # choose transfer amount with some randomness
       delta = rng.random() * max_transfer
       # maybe also allow reverse transfer: move from i to j
       # do both directions? we can random sign
       # but we pick direction as above
       new_h = best_h.copy()
       new_h[i] += delta
       new_h[j] -= delta
       # to maintain sum OK automatically
       # compute c5
       new_c5 = compute_c5(new_h)
       # acceptance
       if new_c5 < best_c5:
           best_h = new_h
           best_c5 = new_c5
           # maybe reset temperature?
       else:
           # Accept with probability exp(-(new_c5 - best_c5) / temp)
           if rng.random() < math.exp(-(new_c5 - best_c5) / temp):
               best_h = new_h
               best_c5 = new_c5
       # update temperature
       temp *= decay
       # ensure temp not too low
       if temp < 1e-8:
           temp = 1e-8
   return best_h, best_c5

        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)

        if k != 0:
            if k > 0:
                src_candidates = np.where((best_h[:n - k] > 1e-12) & (best_h[k:] < 1.0 - 1e-12))[0]
                if src_candidates.size > 0:
                    i = rng.choice(src_candidates)
                    j = i + k
                else:
                    i, j = rng.choice(n, 2, replace=False)
            else:
                kp = -k
                src_candidates = np.where((best_h[kp:] > 1e-12) & (best_h[:n - kp] < 1.0 - 1e-12))[0]
                if src_candidates.size > 0:
                    i = rng.choice(src_candidates) + kp
                    j = i - kp
                else:
                    i, j = rng.choice(n, 2, replace=False)
        else:
            i, j = rng.choice(n, 2, replace=False)
        # determine transfer direction: we will always move from i (source) to j (dest) as decreasing i/increasing j
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 or rng.random() < math.exp(-(c5_new - best_c5) / max(T, 1e-12)):
            best_h = h_new
            best_c5 = c5_new
        T *= decay
        if T < 1e-9:
            T = 1e-9
    return best_h, best_c5

class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""

        self.n = n
        self.dx = 2.0 / n

                        j = i + k
                        max_transfer = min(best_h[i], 1.0 - best_h[j])
                        delta = rng.random() * max_transfer
                        h_new = best_h.copy()
                        h_new[i] -= delta
                        h_new[j] += delta
                else:  # k < 0
                    kp = -k
                    src = np.where((best_h[kp:] > 1e-12) &
                                   (best_h[:n - kp] < 1.0 - 1e-12))[0]
                    if src.size == 0:
                        i, j = rng.choice(n, 2, replace=False)
                        max_transfer = min(1.0 - best_h[i], best_h[j])
                        if max_transfer <= 0:
                            continue
                        delta = rng.random() * max_transfer
                        h_new = best_h.copy()
                        h_new[i] += delta
                        h_new[j] -= delta
                    else:
                        i = rng.choice(src) + kp
                        j = i - kp
                        max_transfer = min(best_h[i], 1.0 - best_h[j])
                        delta = rng.random() * max_transfer
                        h_new = best_h.copy()
                        h_new[i] -= delta
                        h_new[j] += delta
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < best_c5:
                best_h = h_new
                best_c5 = c5_new
                if best_c5 < 0.3808:
                    break
    return best_h, best_c5

# ---------- Simulated annealing refinement ----------
def simulated_annealing_refine(h, c5, time_budget, rng):
    start = time.time()
    best_h = h.copy()
    best_c5 = c5
    n = len(h)
    T = 0.005
    decay = 0.9998
    while time.time() - start < time_budget:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)
        if k != 0:
            if k > 0:
                src = np.where((best_h[:n - k] > 1e-12) &
                               (best_h[k:] < 1.0 - 1e-12))[0]
                if src.size:
                    i = rng.choice(src)
                    j = i + k
                else:
                    i, j = rng.choice(n, 2, replace=False)
            else:
                kp = -k
                src = np.where((best_h[kp:] > 1e-12) &
                               (best_h[:n - kp] < 1.0 - 1e-12))[0]
                if src.size:
                    i = rng.choice(src) + kp
                    j = i - kp
                else:
                    i, j = rng.choice(n, 2, replace=False)
        else:
            i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 or rng.random() < math.exp(-(c5_new - best_c5) / max(T, 1e-12)):
            best_h = h_new
            best_c5 = c5_new
        T *= decay
        if T < 1e-9:
            T = 1e-9
    return best_h, best_c5

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    # Determine initial resolution
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values, dtype=float)
        n0 = parent_h.size
        M0 = n0 // 2
        cur = np.sum(parent_h)
        if abs(cur - M0) > 1e-12:
            parent_h += (M0 - cur) / n0
            parent_h = np.clip(parent_h, 0.0, 1.0)
        best_h_global = parent_h.copy()
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n0
    else:
        n0 = 80

    # Candidate resolutions (ascending)
    candidate_ns = [n0, 96, 112, 128, 144, 160, 176, 192, 224, 256]
    candidate_ns = [n for n in candidate_ns if n % 2 == 0]
    candidate_ns = sorted(set(candidate_ns))

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        # generous allocation for each stage, but never exceed half of remaining
        alloc = min(300.0, remaining * 0.5)

        # Build list of initial seeds for this n
        seed_h_list = []

        if n == best_n_global and best_h_global is not None:
            seed_h_list.append(best_h_global.copy())

        # Interpolation from previous best (if coarser)
        if best_h_global is not None and best_n_global is not None and best_n_global < n:
            x_old = np.linspace(0, 2, best_n_global, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            interp = np.interp(x_new, x_old, best_h_global)
            # enforce sum exactly
            M = n // 2
            interp = np.clip(interp, 0.0, 1.0)
            cur = np.sum(interp)
            if cur > 0:
                interp = interp * (M / cur)
            cur = np.sum(interp)
            if cur < M:
                deficit = M - cur
                for i in np.argsort(interp):
                    inc = min(deficit, 1.0 - interp[i])
                    interp[i] += inc
                    deficit -= inc
                    if deficit <= 1e-12:
                        break
            elif cur > M:
                excess = cur - M
                for i in np.argsort(-interp):
                    dec = min(excess, interp[i])
                    interp[i] -= dec
                    excess -= dec
                    if excess <= 1e-12:
                        break
            seed_h_list.append(interp)

        # a couple of random seeds
        for _ in range(2):
            seed_h_list.append(generate_initial_h(n, n // 2, rng))

        h_opt, c5_opt = optimise_with_seeds(n, alloc, rng, seed_h_list)

        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global = h_opt
            best_c5_global = c5_opt
            best_n_global = n
            if best_c5_global < 0.3808:
                break

    # ----- final simulated‑annealing refinement -----
    if best_h_global is not None:
        remaining = budget_s - (time.time() - start_time)
        if remaining > 1.0:
            best_h_global, best_c5_global = simulated_annealing_refine(
                best_h_global, best_c5_global, remaining, rng)

    # ----- safety net & final projection -----
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        M = n // 2
        cur = np.sum(best_h_global)
        if abs(cur - M) > 1e-9:
            best_h_global += (M - cur) / n
            best_h_global = np.clip(best_h_global, 0.0, 1.0)

    return best_h_global, best_c5_global, n
