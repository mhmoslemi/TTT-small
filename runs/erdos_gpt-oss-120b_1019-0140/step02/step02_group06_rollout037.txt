def perturb_h(h, rng, max_delta):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    delta = rng.uniform(-max_delta, max_delta)
    # try to add delta to i, subtract delta from j
    hi = h[i] + delta
    hj = h[j] - delta
    # If out of bounds, we can clip or reject; we can adjust delta to satisfy bounds:
    # Limit delta such that hi in [0,1] and hj in [0,1].
    lo = max(-h[i], h[j] - 1) # min delta to not go below 0 on i and above 1 on j
    hi_delta = min(1 - h[i], h[j]) # max delta to not exceed 1 on i and go below 0 on j
    # Our random delta must be between lo and hi_delta, also bounded by max_delta
    delta_min = max(-max_delta, lo)
    delta_max = min(max_delta, hi_delta)
    if delta_max < delta_min:
        # Can't adjust; return original
        return h.copy()
    delta = rng.uniform(delta_min, delta_max)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # Values remain in [0,1] and sum preserved exactly.
    return new_h

import time, numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def perturb_two_indices(h, rng, max_delta):
    n = len(h)
    # Find two indices where we can shift delta while staying in bounds
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    # Determine feasible delta interval
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    # Intersect with [-max_delta, max_delta]
    lo = max(lo, -max_delta)
    hi = min(hi, max_delta)
    if lo >= hi:
        # cannot move, return original copy
        return h
    delta = rng.uniform(lo, hi)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def project_sum_and_bounds(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Calculate current sum
    cur = x.sum()
    diff = target - cur
    # Adjust uniformly; because after clipping there may still be interior slack
    if abs(diff) > 1e-12:
        # Add diff equally to all entries, then clip again
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        # After this, sum may still be off due to clipping; distribute remaining diff iteratively
        cur = x.sum()
        diff = target - cur
        # Simple loop to distribute remaining diff over entries not at bounds
        for i in range(len(x)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
    return x

def gradient_optimize(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 80.0
    alpha0 = 0.3
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # increase beta
        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)
        # softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / v.size
        else:
            w = w / w_sum
        # gradient
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-15:
                continue
            k = idx - offset
            if k >= 0:
                tail_slice = slice(0, n - k)
                head_slice = slice(k, n)
            else:
                k2 = -k
                tail_slice = slice(k2, n)
                head_slice = slice(0, n - k2)
            g[tail_slice] += weight * (1.0 - h[head_slice])
            g[head_slice] -= weight * h[tail_slice]
        # zero-sum
        g -= np.mean(g)
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        # acceptance
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
    return best_h, best_c5

        max_idx = int(np.argmax(v))
        k = max_idx - (n - 1)
        # skip zero shift
        if k == 0:

            i = rng.integers(0, n)
            delta = rng.uniform(-0.02, 0.02)
            # attempt apply via two-index move

            # adjust using bounding logic
            # compute feasible delta based on bounds (like earlier)
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            lo = max(lo, -0.02)
            hi = min(hi, 0.02)
            if hi > lo:
                dd = rng.uniform(lo, hi)
                h[i] += dd
                h[j] -= dd

        if k > 0:
            i = rng.integers(0, n - k)
            j = i + k

            i = rng.integers(-k, n)
            j = i + k

            max_delta = min(h[j], 1.0 - h[i])

            # move from j to i
            delta = rng.uniform(0, max_delta) * 0.5
            new_h = h.copy()
            new_h[j] -= delta
            new_h[i] += delta
        else:
            delta = rng.uniform(0, max_delta) * 0.5
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        new_c5 = compute_c5(new_h)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def gradient_optimize(h0, time_budget, seed):

    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)

    dx = 2.0 / n

    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(v))
        k = max_idx - (n - 1)

        if k == 0:
            # random tiny two‑index shuffle to escape possible plateaus
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            lo = max(lo, -0.02)
            hi = min(hi, 0.02)
            if hi > lo:
                d = rng.uniform(lo, hi)
                h[i] += d
                h[j] -= d
            continue

        # pick a random feasible pair (i, j=i+k) inside the overlapping region
        if k > 0:
            i = rng.integers(0, n - k)
            j = i + k
        else:
            i = rng.integers(-k, n)
            j = i + k

        # try moving mass from i to j
        max_delta = min(h[i], 1.0 - h[j])
        direction = -1  # -1 means i → j
        if max_delta <= 0:
            # maybe opposite direction works
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 0:
                continue
            direction = 1   # j → i
        # take a modest fraction to allow fine tuning
        delta = rng.random() * max_delta * 0.1
        new_h = h.copy()
        if direction == -1:
            new_h[i] -= delta
            new_h[j] += delta
        else:
            new_h[i] += delta
            new_h[j] -= delta

        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
            h = new_h

        # occasional random shake to avoid stagnation
        if rng.random() < 0.001:
            h = project_sum_and_bounds(h + rng.normal(scale=0.01, size=n), n / 2.0)
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        h_initial = np.array(initial_h_values, dtype=float).copy()
        n = len(h_initial)
    else:
        n = 80
        # start from a balanced binary pattern
        h_initial = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=n // 2, replace=False)
        h_initial[ones] = 1.0

    target = n / 2.0
    h_initial = project_sum_and_bounds(h_initial, target)

    # allocate time for the three phases
    total_elapsed = time.time() - start
    remaining = max(0.0, budget_s - total_elapsed)

    # Phase 1: gradient descent (≈50% of remaining time)
    t1 = remaining * 0.5
    best_h, best_c5 = gradient_optimize(h_initial, t1, seed + 100)
    remaining = max(0.0, budget_s - (time.time() - start))

    # Phase 2: pairwise hill climbing (≈35% of remaining time)
    t2 = remaining * 0.7
    h2, c5_2 = pair_hill_climb(best_h, t2, seed + 200)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2
    remaining = max(0.0, budget_s - (time.time() - start))

    # Phase 3: final short gradient refinement (≈remaining time)
    if remaining > 0.0:
        h3, c5_3 = gradient_optimize(best_h, remaining, seed + 300)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
