import time
import math
import numpy as np

def compute_c5(arr):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated by full correlation."""
    n = arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Thue‑Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)

def rudin_shapiro_seq(n):
    """Rudin‑Shapiro binary sequence (+1/‑1 → 1/0) of length n."""
    a = np.array([1], dtype=int)
    while a.size < n:
        b = np.empty(a.size * 2, dtype=int)
        b[0::2] = a
        signs = np.where(np.arange(a.size) % 2 == 0, 1, -1)
        b[1::2] = a * signs
        a = b
    return (a[:n] + 1) // 2

def random_binary(n, rng):
    """Return a float array of length n with exactly n//2 ones."""
    x = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    x[idx] = 1.0
    return x

def adjust_binary(vec, target_sum, rng):
    """Force a {0,1} vector to have the exact target sum."""
    v = vec.astype(int)
    cur = int(v.sum())
    if cur > target_sum:
        idx = np.where(v == 1)[0]
        rng.shuffle(idx)
        v[idx[:cur - target_sum]] = 0
    elif cur < target_sum:
        idx = np.where(v == 0)[0]
        rng.shuffle(idx)
        v[idx[:target_sum - cur]] = 1
    return v.astype(float)

def binary_simulated_anneal(h_start, start_time, alloc, rng):
    """Simulated annealing on binary vectors using swaps."""
    h = h_start.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    T0 = 1e-2      # initial temperature
    Tf = 1e-7      # final temperature
    end_time = start_time + alloc

    while time.time() < end_time:
        elapsed = time.time() - start_time
        frac = elapsed / alloc
        T = T0 * (1.0 - frac) + Tf * frac

        # pick a 1‑index and a 0‑index
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)

        # propose swap
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        dE = new_c5 - cur_c5

        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-dE / (T + 1e-20)):
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0

    return best_h, best_c5

def binary_hill_climb(h_start, end_time, rng):
    """Deterministic hill climbing (only accept improving swaps)."""
    h = h_start.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    while time.time() < end_time:
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)

        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        if new_c5 + 1e-12 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- problem size ----------------------------------------------------
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        init_float = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        init_float = None
    target_sum = n // 2

    # ----- build seed pool -------------------------------------------------
    seeds = []

    if init_float is not None:
        bin0 = (init_float >= 0.5).astype(float)
        bin0 = adjust_binary(bin0, target_sum, rng)
        seeds.append(bin0)

    tm = thue_morse_seq(n).astype(float)
    tm = adjust_binary(tm, target_sum, rng)
    seeds.append(tm)

    rs = rudin_shapiro_seq(n).astype(float)
    rs = adjust_binary(rs, target_sum, rng)
    seeds.append(rs)

    for _ in range(4):
        seeds.append(random_binary(n, rng))

    # ----- evaluate seeds ---------------------------------------------------
    best_h = None
    best_c5 = np.inf
    for s in seeds:
        c = compute_c5(s)
        if c < best_c5:
            best_c5 = c
            best_h = s.copy()

    # ----- main stochastic search -------------------------------------------
    margin = 0.2
    while time.time() < start_time + budget_s - margin:
        remaining = start_time + budget_s - margin - time.time()
        alloc = min(8.0, remaining)          # short SA runs
        # choose a start vector
        if rng.random() < 0.5 and best_h is not None:
            start_vec = best_h.copy()
            # apply a few random swaps to perturb
            for _ in range(rng.integers(1, 4)):
                i = rng.integers(n)
                while start_vec[i] < 0.5:
                    i = rng.integers(n)
                j = rng.integers(n)
                while start_vec[j] > 0.5:
                    j = rng.integers(n)
                start_vec[i], start_vec[j] = 0.0, 1.0
        else:
            start_vec = seeds[rng.integers(len(seeds))]

        cand_h, cand_c5 = binary_simulated_anneal(start_vec, time.time(), alloc, rng)

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h.copy()
            if best_c5 <= 0.3808:
                break

    # ----- final deterministic refinement -----------------------------------
    time_left = start_time + budget_s - 0.1 - time.time()
    if time_left > 0.1 and best_h is not None:
        final_h, final_c5 = binary_hill_climb(best_h.copy(),
                                              time.time() + time_left,
                                              rng)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    # ----- safety fix -------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        # final exactness pass
        cur_sum = best_h.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            if cur_sum > target_sum:
                idx = np.where(best_h > 0.5)[0]
                rng.shuffle(idx)
                for i in idx:
                    best_h[i] = 0.0
                    cur_sum -= 1.0
                    if abs(cur_sum - target_sum) < 1e-9:
                        break
            else:
                idx = np.where(best_h < 0.5)[0]
                rng.shuffle(idx)
                for i in idx:
                    best_h[i] = 1.0
                    cur_sum += 1.0
                    if abs(cur_sum - target_sum) < 1e-9:
                        break

    return best_h, float(best_c5), n
