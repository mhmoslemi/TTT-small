def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 80
        h = rng.random(n)

    target_sum = n/2.0
    h = project_to_box_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    # Simulated annealing parameters
    T0 = 0.005   # initial temperature
    # step size parameters
    max_step = 0.5
    min_step = 0.001

    # Determine stop time for main SA loop
    # Reserve some time for final SLSQP (say 10% of budget)
    final_slsqp_time = min(30.0, budget_s * 0.1)  # at most 30 seconds
    sa_deadline = start + budget_s - final_slsqp_time

    # SA loop
    while time.time() < sa_deadline:
        # generate zero-mean random direction
        d = rng.normal(size=n)
        d -= d.mean()
        # sample step magnitude
        step = (max_step - min_step) * (1.0 - (time.time() - start) / budget_s) + min_step
        # use random power to bias smaller steps
        step = step * (rng.random() ** 2)

        cand = cur_h + step * d
        # enforce box constraints
        cand = np.clip(cand, 0.0, 1.0)
        # enforce sum exactly
        cand = project_to_box_sum(cand, target_sum)

        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5

        # temperature decays linearly
        T = T0 * max(0.0, 1.0 - (time.time() - start) / budget_s)

        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis
            if T > 0:
                if rng.random() < np.exp(-delta / T):
                    accept = True
        if accept:
            cur_h = cand
            cur_c5 = cand_c5

            if cur_c5 + 1e-12 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                # if we achieve target < 0.3808, we can break early
                if best_c5 < 0.3808:
                    # break early but still try to refine maybe
                    # break out of SA loop
                    break

    # Final refinement with SLSQP
    # Setup constraints
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]

    # If we have some time left, run SLSQP with decreasing maxiter
    # We'll loop while time remains > some threshold
    while time.time() < start + budget_s - 1.0:
        # Use current best_h as start, compute t_start
        h_start = best_h.copy()
        t_start = compute_c5(h_start)

        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            # enforce sum again (small deviations)
            if abs(h_opt.sum() - target_sum) > 1e-9:
                h_opt = project_to_box_sum(h_opt, target_sum)

            c5_opt = compute_c5(h_opt)
            if c5_opt + 1e-12 < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                # break early if very good
                if best_c5 < 0.3808:
                    # optional break if remaining time small
                    pass

        # Optionally reduce maxiter for subsequent tries
        # if time allows more iterations
        # break if time limited
        if time.time() > start + budget_s - 2.0:
            break

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------
# Utility functions
# -------------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while enforcing sum(v)=target via water‑filling binary search."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# -------------------------------------------------------
# Constraint classes for SLSQP
# -------------------------------------------------------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k)*dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

# -------------------------------------------------------
# SLSQP refinement routine
# -------------------------------------------------------
def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]  # t >= 0
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

# -------------------------------------------------------
# Main entry point
# -------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ---- simulated annealing (pairwise transfer moves) ----
    total_budget = budget_s
    sa_fraction = 0.85                # allocate 85% of time to SA
    sa_deadline = start_time + total_budget * sa_fraction

    T0 = 5e-3                         # initial temperature
    max_step_factor = 0.5
    min_step_factor = 0.02

    while time.time() < sa_deadline:
        # choose two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # random direction: move from i to j or opposite
        if rng.random() < 0.5:
            # i -> j
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            direction = 1
        else:
            # j -> i
            max_delta = min(cur_h[j], 1.0 - cur_h[i])
            direction = -1
        if max_delta <= 0.0:
            continue

        # annealing schedule
        elapsed = time.time() - start_time
        frac = elapsed / total_budget
        T = T0 * max(0.0, 1.0 - frac)

        # step factor decreasing linearly
        step_factor = (max_step_factor - min_step_factor) * max(0.0, 1.0 - frac) + min_step_factor

        delta = step_factor * max_delta * rng.random()
        cand = cur_h.copy()
        if direction == 1:
            cand[i] -= delta
            cand[j] += delta
        else:
            cand[i] += delta
            cand[j] -= delta

        # evaluate
        cand_c5 = compute_c5(cand)
        dC = cand_c5 - cur_c5

        accept = False
        if dC < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-dC / T):
                accept = True

        if accept:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 + 1e-12 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                if best_c5 < 0.3808:
                    # good enough; we can stop SA early
                    break

    # ---- final SLSQP refinement ----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # allocate remaining time (at least a few seconds)
    while time.time() < start_time + total_budget - 1.0:
        t_start = compute_c5(best_h)
        res = slsqp_optimize(best_h, t_start, sum_con, overlap_cons, maxiter=600)
        if not res.success:

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # enforce exact sum (numeric drift)
        if abs(h_opt.sum() - target_sum) > 1e-9:
            h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            if best_c5 < 0.3808:
                break
        # if time is low, stop
        if time.time() > start_time + total_budget - 0.5:
            break

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
