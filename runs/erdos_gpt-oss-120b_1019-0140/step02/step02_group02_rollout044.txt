def anneal(h, start_time, deadline, rng):
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_h = h.copy()
    cur_val = best_val
    while time.time() < deadline-0.1:
        # compute corr
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute product contributions
        if k >= 0:
            prod = cur_h[:n-k] * (1.0 - cur_h[k:])
            i_local = np.argmax(prod)
            i = i_local
            j = i + k
        else:
            kk = -k
            prod = cur_h[kk:] * (1.0 - cur_h[:n-kk])
            i_local = np.argmax(prod)
            i = i_local + kk
            j = i_local
        # attempt transfer from i to j
        delta_max = min(cur_h[i], 1.0 - cur_h[j])
        if delta_max > 1e-12:
            # random fraction
            delta = delta_max * rng.random()
            cur_h[i] -= delta
            cur_h[j] += delta
            new_val = compute_c5(cur_h)
            if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / max(1e-12, (deadline - time.time()) * 1e-3)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = cur_h.copy()
                continue
            else:
                # revert
                cur_h[i] += delta
                cur_h[j] -= delta
        # fallback random transfer between any src (h>0) and dst (h<1)
        src_candidates = np.where(cur_h > 1e-12)[0]
        dst_candidates = np.where(cur_h < 1.0 - 1e-12)[0]
        if len(src_candidates)==0 or len(dst_candidates)==0:
            break
        i = rng.choice(src_candidates)
        j = rng.choice(dst_candidates)
        delta_max = min(cur_h[i], 1.0 - cur_h[j])
        if delta_max < 1e-12:
            continue
        delta = delta_max * rng.random()
        cur_h[i] -= delta
        cur_h[j] += delta
        new_val = compute_c5(cur_h)
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / max(1e-12, (deadline - time.time()) * 1e-3)):

            cur_h[i] += delta
            cur_h[j] -= delta

    return best_h, best_val

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    # same as before
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def enforce_sum(h, target):
    # adjust using simple method
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # distribute diff equally (could break bounds)
    h = h + diff / len(h)
    # clip
    h = np.clip(h, 0.0, 1.0)
    # after clip may still deviate; perform iterative fix:
    cur = h.sum()
    diff = target - cur
    # fix by adding/subtracting small amount to first few entries
    i = 0
    while abs(diff) > 1e-12 and i < len(h):
        adj = np.clip(diff, -h[i], 1.0 - h[i])
        h[i] += adj
        diff = target - h.sum()
        i += 1
    return h

def anneal_once(h_start, target, deadline, rng):
    h = h_start.copy()
    n = len(h)
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    # simple temperature schedule based on remaining time
    while time.time() < deadline - 0.1:
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute contributions
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_local = np.argmax(prod)
            i = i_local
            j = i + k
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_local = np.argmax(prod)
            i = i_local + kk
            j = i_local
        # try targeted transfer
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max > 1e-12:
            delta = delta_max * rng.random()
            h[i] -= delta
            h[j] += delta
            new_val = compute_c5(h)
            accept = False
            if new_val < cur_val:
                accept = True
            else:
                # temperature based acceptance
                T = (deadline - time.time()) * 0.001
                if np.exp((cur_val - new_val) / max(T, 1e-12)) > rng.random():
                    accept = True
            if accept:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
                continue
            else:
                # revert
                h[i] += delta
                h[j] -= delta
        # fallback random transfer
        src = np.where(h > 1e-12)[0]
        dst = np.where(h < 1.0 - 1e-12)[0]
        if src.size == 0 or dst.size == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 1e-12:
            continue
        delta = delta_max * rng.random()
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            T = (deadline - time.time()) * 0.001
            if np.exp((cur_val - new_val) / max(T, 1e-12)) > rng.random():
                accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
        # early exit
        if best_val < 0.3808:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)
    # get n, construct initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = enforce_sum(h0, target)
    else:
        n = 80
        target = n / 2.0
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n//2, replace=False)
        h0[idx] = 1.0
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    # outer loop: multiple annealing runs
    while time.time() < deadline - 1.0:
        # create start point: copy best and perturb with few random swaps
        h_start = best_h.copy()
        # perform some random swaps (binary) maybe 5 swaps
        # convert to binary representation? If best_h has fractional, we need to keep continuity.
        # We'll just add small random perturbations: random pair of indices i,j modify small delta
        for _ in range(5):
            i, j = rng.choice(n, 2, replace=False)
            # transfer small amount
            delta_max = min(h_start[i], 1.0 - h_start[j])
            if delta_max > 0:
                delta = delta_max * rng.random()
                h_start[i] -= delta
                h_start[j] += delta
        # enforce sum just in case
        h_start = enforce_sum(h_start, target)
        # run annealing for a limited time slice, e.g., 5 seconds or remaining/10
        remaining = deadline - time.time()
        slice_time = min(5.0, remaining * 0.5)
        if slice_time < 0.2:
            break
        end_slice = time.time() + slice_time
        h_candidate, val_candidate = anneal_once(h_start, target, end_slice, rng)
        if val_candidate < best_val:
            best_val = val_candidate
            best_h = h_candidate.copy()
            if best_val < 0.3808:
                break
        # optional occasional SLSQP refinement
        if rng.random() < 0.2:
            # perform SLSQP starting from best_h
            x0 = best_h.copy()
            bounds = [(0.0, 1.0)] * n
            # equality constraint sum = target
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
            try:
                res = minimize(lambda x: compute_c5(x), x0=x0, method='SLSQP',
                               bounds=bounds, constraints=[constraints],
                               options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
                if res.success:
                    h_s = np.clip(res.x, 0.0, 1.0)
                    h_s = enforce_sum(h_s, target)
                    val_s = compute_c5(h_s)
                    if val_s < best_val:
                        best_val = val_s
                        best_h = h_s.copy()
                        if best_val < 0.3808:
                            break
            except Exception:
                pass
    # final safety clip and sum
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(h, target):
    """Adjust h in-place so that sum(h) == target while keeping 0<=h<=1."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # distribute diff uniformly, then clip and fix residuals iteratively
    h += diff / h.size
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    i = 0
    while abs(diff) > 1e-12 and i < h.size:
        adj = np.clip(diff, -h[i], 1.0 - h[i])
        h[i] += adj
        diff = target - h.sum()
        i += 1
    return h

def anneal_once(h_start, target, deadline, rng):
    """Simulated annealing with targeted mass transfers."""
    h = h_start.copy()
    n = h.size
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    while time.time() < deadline - 0.1:
        # worst shift and its most contributing index
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_loc = np.argmax(prod)
            i = i_loc
            j = i + k
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_loc = np.argmax(prod)
            i = i_loc + kk
            j = i_loc

        # targeted transfer from i (decrease) to j (increase)
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max > 1e-12:
            delta = delta_max * rng.random()
            h[i] -= delta
            h[j] += delta
            new_val = compute_c5(h)
            accept = False
            if new_val < cur_val:
                accept = True
            else:
                T = max((deadline - time.time()) * 0.001, 1e-12)
                if np.exp((cur_val - new_val) / T) > rng.random():
                    accept = True
            if accept:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
                continue
            else:
                h[i] += delta
                h[j] -= delta

        # fallback random transfer
        src = np.where(h > 1e-12)[0]
        dst = np.where(h < 1.0 - 1e-12)[0]
        if src.size == 0 or dst.size == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 1e-12:
            continue
        delta = delta_max * rng.random()
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            T = max((deadline - time.time()) * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta

        if best_val < 0.3808:
            break

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
    target = n / 2.0
    h0 = enforce_sum(h0, target)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- main search loop -----
    while time.time() < deadline - 1.0:
        # decide start point: either perturb best or fresh random binary
        if rng.random() < 0.2:
            h_start = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h_start[idx] = 1.0
        else:
            h_start = best_h.copy()
            # small random mass shuffles
            for _ in range(5):
                i, j = rng.choice(n, 2, replace=False)
                delta_max = min(h_start[i], 1.0 - h_start[j])
                if delta_max > 0:
                    delta = delta_max * rng.random()
                    h_start[i] -= delta
                    h_start[j] += delta
        h_start = enforce_sum(h_start, target)

        # allocate a time slice for this annealing run
        remaining = deadline - time.time()
        slice_time = min(5.0, remaining * 0.5)
        if slice_time < 0.2:
            break
        slice_deadline = time.time() + slice_time

        cand_h, cand_val = anneal_once(h_start, target, slice_deadline, rng)
        if cand_val < best_val:
            best_val, best_h = cand_val, cand_h.copy()
            if best_val < 0.3808:
                break

        # occasional SLSQP refinement
        if rng.random() < 0.15:
            x0 = best_h.copy()
            bounds = [(0.0, 1.0)] * n
            constraints = {"type": "eq", "fun": lambda x: np.sum(x) - target}
            try:
                res = minimize(
                    fun=lambda x: compute_c5(x),
                    x0=x0,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=[constraints],
                    options={"maxiter": 600, "ftol": 1e-9, "disp": False},
                )
                if res.success:
                    h_opt = enforce_sum(np.clip(res.x, 0.0, 1.0), target)
                    val_opt = compute_c5(h_opt)
                    if val_opt < best_val:
                        best_val, best_h = val_opt, h_opt.copy()
                        if best_val < 0.3808:
                            break
            except Exception:
                pass

    # final safety adjustments
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), target)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
