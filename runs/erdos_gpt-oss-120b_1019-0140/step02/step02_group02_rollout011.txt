def hill_climb(h, max_iters, rng):
    best = h.copy()
    best_val = compute_c5(best)
    n = len(h)
    no_improve = 0
    for it in range(max_iters):
        # compute worst shift
        corr = np.correlate(best, 1 - best, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute contributions
        if k >= 0:
            left = best[: n - k]
            right = best[k:]
            prod = left * (1 - right)
            i_range = np.arange(n - k)
        else:
            kk = -k
            left = best[kk:]
            right = best[: n - kk]
            prod = left * (1 - right)
            i_range = np.arange(kk, n)
        # pick source index with largest contribution (i_max)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # source must have h[i] > 0
        if best[i] <= 1e-12:
            continue
        # partner values for shift k
        if k >= 0:
            partner_i = best[i + k] if i + k < n else 0
        else:
            partner_i = best[i + k] if i + k >= 0 else 0
        # find candidate destinations j
        # compute partner values at shift k for all j that overlap
        if k >= 0:
            # overlapping j range: 0..n-k-1
            overlapping = np.arange(n - k)
        else:
            overlapping = np.arange(-k, n)
        # candidate j where h[j] < 1
        cand = overlapping[best[overlapping] < 1 - 1e-12]
        if len(cand) == 0:
            # random j not overlapping
            cand = np.where(best < 1 - 1e-12)[0]
        # evaluate each candidate's expected new max after moving a small delta
        # Instead of full evaluation for each, just pick j where partner_j is maximal
        # partner_j = best[j + k] if overlapping etc
        if k >= 0:
            partner_vals = best[cand + k]  # safe because cand <= n-k-1
        else:
            partner_vals = best[cand + k]
        # select j with largest partner value (most likely to reduce)
        j = cand[np.argmax(partner_vals)]

        # compute max delta reachable
        max_delta = min(best[i], 1.0 - best[j])
        if max_delta <= 0:
            continue
        # choose delta = fraction of max_delta
        delta = max_delta * rng.random() * 0.5 + max_delta * 0.01  # ensure some movement
        # apply move
        candidate = best.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # ensure sum unchanged
        val = compute_c5(candidate)
        if val < best_val - 1e-9:
            best = candidate
            best_val = val
            no_improve = 0
            if best_val < 0.3808:
                break
        else:
            # maybe accept some uphill with low probability (annealing)
            # compute temperature based on iteration
            T = max(0.001, (max_iters - it) / max_iters)
            if rng.random() < np.exp((best_val - val) / T):
                best = candidate
                best_val = val
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 1000:
            break

    return best, best_val

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated discretely."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _enforce_sum(h, target):
    """Clip to [0,1] and adjust to meet exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur

        h += diff / h.size

    return h
