def compute_Os(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def max_overlap_info(h):
    O = compute_Os(h)
    max_idx = np.argmax(O)
    return O[max_idx], max_idx - (len(h)-1)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact discretized C5 value for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_and_fix(h, target_sum):
    """Mild Gaussian smoothing followed by exact sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    cur = np.sum(sm)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            inc = diff / free.size
            sm[free] += inc
            sm = np.clip(sm, 0.0, 1.0)
        diff = target_sum - np.sum(sm)
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
        sm = np.clip(sm, 0.0, 1.0)
    return sm

def compute_Os(h):
    """Array of overlap integrals O_k for all integer shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx  # length 2n-1, shift index = k + (n-1)

def gradient_smooth_max(h, alpha):
    """
    Gradient of the smooth-maximum of O_k with smoothing parameter alpha.
    Returns (grad, O) where O is the array of O_k.
    """
    n = len(h)
    O = compute_Os(h)                     # shape (2n-1,)
    maxO = np.max(O)
    # stabilized weights for log-sum-exp
    w = np.exp(alpha * (O - maxO))
    Z = w.sum()
    w /= Z
    shifts = np.arange(-(n - 1), n)       # length 2n-1, aligns with O
    grad = np.zeros(n)
    for wk, k in zip(w, shifts):
        if wk == 0.0:
            continue
        if k >= 0:
            # forward contribution (1 - h_{j+k})
            grad[: n - k] += wk * (1.0 - h[k:])
            # backward contribution -h_{j-k}
            grad[k:] += wk * (-h[: n - k])
        else:
            kk = -k
            # forward contribution (1 - h_{j-kk})
            grad[kk:] += wk * (1.0 - h[: n - kk])
            # backward contribution -h_{j+kk}
            grad[: n - kk] += wk * (-h[kk:])
    return grad, O

def project_onto_feasible(h, target_sum):
    """Clip to [0,1] and enforce exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        interior = np.where((h > 0.0) & (h < 1.0))[0]
        if interior.size:
            inc = diff / interior.size
            h[interior] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
        if abs(diff) > 1e-12:
            for i in range(len(h)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_descent_optimize(h0, target_sum, time_limit, rng):
    """
    Local optimization by gradient descent on a smooth approximation of max O_k.
    time_limit is a per-call budget in seconds.
    """
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    # hyper‑parameters
    alpha = 12.0               # smoothing of max
    step = 0.15                # initial step size
    decay = 0.998
    min_step = 0.001
    it = 0
    start = time.time()
    while time.time() - start < time_limit * 0.95:
        grad, _ = gradient_smooth_max(h, alpha)
        cand = h - step * grad
        cand = project_onto_feasible(cand, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val, best_h = val, cand.copy()
        # accept if improvement, else keep old point (simple descent)
        if val <= compute_c5(h):
            h = cand
        else:
            # reject and reduce step
            step = max(step * 0.7, min_step)
        step = max(step * decay, min_step)
        it += 1
        if it % 250 == 0:
            # occasional tiny random perturbation to avoid stagnation
            noise = (rng.random(n) - 0.5) * 0.01
            h = project_onto_feasible(h + noise, target_sum)
        if best_val < 0.3808:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = rng.random(n)
        target_sum = n / 2.0
        h0 = h0 * (target_sum / np.sum(h0))
        h0 = project_onto_feasible(h0, target_sum)

    target_sum = n / 2.0
    h0 = project_onto_feasible(h0, target_sum)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ---- primary gradient descent ------------------------------------
    # allocate ~60 % of total budget to the first run
    primary_budget = budget_s * 0.60
    h_opt, val_opt = gradient_descent_optimize(best_h, target_sum, primary_budget, rng)
    if val_opt < best_val:
        best_h, best_val = h_opt, val_opt
        if best_val < 0.3808:
            return best_h, best_val, n

    # ---- multi‑start random restarts --------------------------------
    while time.time() - start < budget_s * 0.90:
        remaining = budget_s - (time.time() - start)
        # each restart gets at most 30 s (or whatever fits)
        alloc = min(30.0, remaining * 0.5)
        if alloc < 5.0:
            break
        # random feasible start
        h_rand = rng.random(n)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        h_rand = project_onto_feasible(h_rand, target_sum)
        h_opt2, val_opt2 = gradient_descent_optimize(h_rand, target_sum, alloc, rng)
        if val_opt2 < best_val:
            best_h, best_val = h_opt2, val_opt2
            if best_val < 0.3808:
                break

    # ---- optional smoothing passes ----------------------------------
    if time.time() - start < budget_s - 5.0:
        for _ in range(5):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_h, best_val = sm, sm_val
            else:
                break

    # ---- final local refinement using smooth‑max L‑BFGS‑B ------------
    if time.time() - start < budget_s - 10.0:
        alpha_obj = 20.0
        penalty = 1e4
        def objective(x):
            O = compute_Os(x)
            smooth = np.log(np.sum(np.exp(alpha_obj * O))) / alpha_obj
            return smooth + penalty * (np.sum(x) - target_sum) ** 2
        bounds = [(0.0, 1.0)] * n
        x0 = best_h.copy()
        remaining = budget_s - (time.time() - start)
        try:
            res = minimize(
                objective,
                x0,
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 500, "ftol": 1e-12, "gtol": 1e-12, "maxcor": 10, "disp": False},
            )
            if res.success:
                cand = project_onto_feasible(res.x, target_sum)
                cand_val = compute_c5(cand)
                if cand_val < best_val:
                    best_h, best_val = cand, cand_val
        except Exception:
            pass

    # ---- final safety correction ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = np.sum(best_h)
    if abs(cur - target_sum) > 1e-9:
        best_h += (target_sum - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, n
