def simulated_annealing(h, dx, target_sum, budget_s, start_time, rng):

    n = len(h); N2 = 2*n-1

    O = np.correlate(h, 1.0 - h, mode='full')
    cur_val = np.max(O) * dx

    best_h = h.copy(); best_val = cur_val

    accept_count = 0; iter_count = 0; last_improve_iter = 0

    T_start = 0.05
    T_end = 1e-5

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.9: break
        # sample i, j
        # Randomly choose donor (>= delta) and recipient (<1)
        donors = np.where(h > 1e-12)[0]
        recipients = np.where(h < 1 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        if i == j: continue
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            continue
        # sample delta; bias more to small values for fine search
        # maybe use beta distribution: delta = rng.random()**2 * max_delta
        delta = (rng.random()**2) * max_delta
        # compute candidate max
        max_candidate = -np.inf
        # compute inc contributions on the fly
        for s in range(-n+1, n):
            idx = s + n - 1
            inc_val = 0.0
            ip = i + s
            if 0 <= ip < n:
                inc_val += 1.0 - h[ip]
            im = i - s
            if 0 <= im < n:
                inc_val -= h[im]
            jp = j + s
            if 0 <= jp < n:
                inc_val -= (1.0 - h[jp])
            jm = j - s
            if 0 <= jm < n:
                inc_val += h[jm]
            cand = O[idx] + delta * inc_val
            if cand > max_candidate:
                max_candidate = cand
        new_val = max_candidate * dx
        # acceptance
        if new_val < cur_val - 1e-12:
            accept = True
        else:
            T = T_start * (1 - (elapsed / (budget_s * 0.9))) + T_end * (elapsed / (budget_s * 0.9))
            prob = math.exp(-(new_val - cur_val) / T) if T > 0 else 0.0
            accept = rng.random() < prob
        if accept:
            # apply update: recompute inc and update O in place

def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # Add uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    # Recompute diff
    diff = target_sum - np.sum(vec)
    if abs(diff) < 1e-12:
        return vec
    # Greedy fix
    if diff > 0:
        # need to increase some entries
        for i in range(len(vec)):
            if diff <= 0:
                break
            inc = min(diff, 1.0 - vec[i])
            if inc > 0:
                vec[i] += inc
                diff -= inc
    else:
        # need to decrease
        diff = -diff
        for i in range(len(vec)):
            if diff <= 0:
                break
            dec = min(diff, vec[i])
            if dec > 0:
                vec[i] -= dec
                diff -= dec
    return vec

def generate_random_feasible(n, target_sum, rng):
    x = rng.random(n)
    total = np.sum(x)
    if total == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / total)
    x = np.clip(x, 0.0, 1.0)
    # fix residual
    diff = target_sum - np.sum(x)
    if diff > 0:
        # increase entries not at 1
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        diff = -diff
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(diff, x[i])
            x[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    return x

def simulated_annealing(h, dx, target_sum, budget_s, start_time, rng):
    import math
    n = len(h)
    N2 = 2 * n - 1
    O = np.correlate(h, 1.0 - h, mode='full')
    cur_val = np.max(O) * dx
    best_h = h.copy()
    best_val = cur_val
    accept_count = 0
    iter_count = 0
    last_improve = 0
    T_start = 0.05
    T_end = 1e-5
    # pre-allocate inc_vals
    inc_vals = np.empty(N2, dtype=float)
    # Precompute index shift range
    shift_range = list(range(-n + 1, n))
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.9:
            break
        # select donor (h>0) and recipient (h<1)
        # donors:
        donors = np.where(h > 1e-12)[0]
        recipients = np.where(h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        if i == j:
            continue
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            continue
        # sample delta biased towards smaller values
        delta = (rng.random() ** 2) * max_delta
        if delta < 1e-14:
            continue
        # compute inc_vals and candidate max
        max_candidate = -np.inf
        # We'll compute inc for each shift
        for idx_shift, s in enumerate(shift_range):
            # s ranges
            inc = 0.0
            ip = i + s
            if 0 <= ip < n:
                inc += 1.0 - h[ip]
            im = i - s
            if 0 <= im < n:
                inc -= h[im]
            jp = j + s
            if 0 <= jp < n:
                inc -= (1.0 - h[jp])
            jm = j - s
            if 0 <= jm < n:
                inc += h[jm]
            inc_vals[idx_shift] = inc
            cand = O[idx_shift] + delta * inc
            if cand > max_candidate:
                max_candidate = cand
        new_val = max_candidate * dx
        # acceptance
        accept = False
        if new_val < cur_val - 1e-12:
            accept = True
        else:
            fraction = elapsed / (budget_s * 0.9)
            T = T_start * (1.0 - fraction) + T_end * fraction
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                accept = rng.random() < prob
        if accept:
            # update O
            O += delta * inc_vals
            # update h
            h[i] += delta
            h[j] -= delta
            cur_val = new_val
            accept_count += 1
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = h.copy()
                last_improve = iter_count
        iter_count += 1
        # occasional recompute
        if accept_count % 5000 == 0 and accept_count > 0:
            O = np.correlate(h, 1.0 - h, mode='full')
            cur_val = np.max(O) * dx
        # stagnation restart
        if iter_count - last_improve > 200000:
            # restart from best_h perturbed
            h = best_h.copy()
            # add small random noise while preserving sum and bounds
            # perform a few random transfers
            for _ in range(10):
                donors = np.where(h > 1e-12)[0]
                recipients = np.where(h < 1.0 - 1e-12)[0]
                if donors.size == 0 or recipients.size == 0:
                    break
                i2 = rng.choice(donors)
                j2 = rng.choice(recipients)
                max_delta2 = min(1.0 - h[i2], h[j2])
                if max_delta2 <= 0:
                    continue
                delta2 = (rng.random() ** 2) * max_delta2
                h[i2] += delta2
                h[j2] -= delta2
            O = np.correlate(h, 1.0 - h, mode='full')
            cur_val = np.max(O) * dx
            last_improve = iter_count
    return best_h, best_val

import time
import math
import numpy as np

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Clip to [0,1] and adjust to achieve exact sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if abs(diff) < 1e-12:
        return vec
    if diff > 0:
        # need to increase some entries
        for i in range(len(vec)):
            if diff <= 0:
                break
            inc = min(diff, 1.0 - vec[i])
            if inc > 0:
                vec[i] += inc
                diff -= inc
    else:
        # need to decrease some entries
        diff = -diff
        for i in range(len(vec)):
            if diff <= 0:
                break
            dec = min(diff, vec[i])
            if dec > 0:
                vec[i] -= dec
                diff -= dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Random vector in [0,1]^n with exact sum = target_sum."""
    x = rng.random(n)
    tot = np.sum(x)
    if tot == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / tot)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        diff = -diff
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(diff, x[i])
            x[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    return x

def simulated_annealing(h, dx, target_sum, budget_s, start_time, rng):
    """Projected SA moving mass between two entries."""
    n = len(h)
    N2 = 2 * n - 1
    O = np.correlate(h, 1.0 - h, mode="full")
    cur_val = np.max(O) * dx
    best_h = h.copy()
    best_val = cur_val
    accept_cnt = 0
    it = 0
    last_improve = 0
    T_start = 0.05
    T_end = 1e-5
    inc_vals = np.empty(N2, dtype=float)
    shift_range = list(range(-n + 1, n))

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.9:
            break

        donors = np.where(h > 1e-12)[0]
        recipients = np.where(h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        if i == j:
            continue
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            continue
        delta = (rng.random() ** 2) * max_delta
        if delta < 1e-14:
            continue

        # compute impact for every shift and candidate max
        max_candidate = -np.inf
        for idx_shift, s in enumerate(shift_range):
            inc = 0.0
            ip = i + s
            if 0 <= ip < n:
                inc += 1.0 - h[ip]
            im = i - s
            if 0 <= im < n:
                inc -= h[im]
            jp = j + s
            if 0 <= jp < n:
                inc -= (1.0 - h[jp])
            jm = j - s
            if 0 <= jm < n:
                inc += h[jm]
            inc_vals[idx_shift] = inc
            cand = O[idx_shift] + delta * inc
            if cand > max_candidate:
                max_candidate = cand
        new_val = max_candidate * dx

        # acceptance test
        accept = False
        if new_val < cur_val - 1e-12:
            accept = True
        else:
            frac = elapsed / (budget_s * 0.9)
            T = T_start * (1.0 - frac) + T_end * frac
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                accept = rng.random() < prob

        if accept:
            O += delta * inc_vals
            h[i] += delta
            h[j] -= delta
            cur_val = new_val
            accept_cnt += 1
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = h.copy()
                last_improve = it

        it += 1

        # occasional rebuild to curb drift
        if accept_cnt and accept_cnt % 5000 == 0:
            O = np.correlate(h, 1.0 - h, mode="full")
            cur_val = np.max(O) * dx

        # restart from best if stagnant too long
        if it - last_improve > 200_000:
            h = best_h.copy()
            # small random perturbation
            for _ in range(8):
                donors = np.where(h > 1e-12)[0]
                recipients = np.where(h < 1.0 - 1e-12)[0]
                if donors.size == 0 or recipients.size == 0:
                    break
                i2 = rng.choice(donors)
                j2 = rng.choice(recipients)
                max_d2 = min(1.0 - h[i2], h[j2])
                if max_d2 <= 0:
                    continue
                d2 = (rng.random() ** 2) * max_d2
                h[i2] += d2
                h[j2] -= d2
            O = np.correlate(h, 1.0 - h, mode="full")
            cur_val = np.max(O) * dx
            last_improve = it

    return best_h, best_val


def binary_hill_climb_from(h_float, dx, target_sum, budget_s, start_time, rng):
    """Greedy swap hill‑climbing on a binary rounding of h_float."""
    n = len(h_float)
    k = int(round(target_sum))
    binary_h = np.zeros(n, dtype=float)
    top_idx = np.argpartition(-h_float, k)[:k]
    binary_h[top_idx] = 1.0

    O = np.correlate(binary_h, 1.0 - binary_h, mode="full")
    cur_val = np.max(O) * dx
    best_h = binary_h.copy()
    best_val = cur_val

    while True:
        if time.time() - start_time >= budget_s * 0.95:
            break
        ones = np.flatnonzero(binary_h > 0.5)
        zeros = np.flatnonzero(binary_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand_h = binary_h.copy()
        cand_h[i] = 0.0
        cand_h[j] = 1.0
        O_cand = np.correlate(cand_h, 1.0 - cand_h, mode="full")
        val_cand = np.max(O_cand) * dx
        if val_cand < cur_val - 1e-12:
            binary_h = cand_h
            O = O_cand
            cur_val = val_cand
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = binary_h.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)

        cur_h = generate_random_feasible(n, target_sum, rng)

    # --- simulated annealing ---
    best_h, best_val = simulated_annealing(
        cur_h, dx, target_sum, budget_s, start_time, rng
    )

    # --- optional binary refinement ---
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5.0:
        bin_h, bin_val = binary_hill_climb_from(
            best_h, dx, target_sum, budget_s, start_time, rng
        )
        if bin_val < best_val - 1e-12:
            best_h, best_val = bin_h, bin_val

    # --- final safety enforcement ---
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
