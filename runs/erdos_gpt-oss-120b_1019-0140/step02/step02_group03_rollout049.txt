import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """
    Project vector v onto the hypercube [lo,hi]^n while forcing its sum to target.
    Uses a binary search on a scalar offset λ so that sum(clip(v+λ, lo, hi)) == target.
    """
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(v + lam, lo, hi)

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x) - self.n / 2.0

def obj_c5(x):
    """Objective for local refinement: C5 of vector x."""
    return compute_c5(x)

def local_refine(h):
    """One-shot SLSQP refinement of h respecting box and sum constraints."""
    n = h.shape[0]
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': SumConstraint(n)}
    res = minimize(
        fun=obj_c5,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=[cons],
        options={'maxiter': 400, 'ftol': 1e-9, 'disp': False}
    )
    if res.success:
        h_opt = project_to_box_sum(res.x, n / 2.0)
        return h_opt, compute_c5(h_opt)
    else:
        # fallback: just enforce feasible sum
        h_feas = project_to_box_sum(h, n / 2.0)
        return h_feas, compute_c5(h_feas)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        # golden-ratio binary seed
        alpha = (np.sqrt(5) - 1.0) / 2.0
        idx = np.arange(n)
        h0 = ((idx * alpha) % 1.0 < 0.5).astype(float)
        # adjust exact sum if needed
        target_sum = n / 2.0
        cur_sum = h0.sum()
        while cur_sum != target_sum:
            if cur_sum > target_sum:
                ones = np.where(h0 > 0.5)[0]
                if ones.size == 0:
                    break
                j = rng.choice(ones)
                h0[j] = 0.0
            else:
                zeros = np.where(h0 < 0.5)[0]
                if zeros.size == 0:
                    break
                j = rng.choice(zeros)
                h0[j] = 1.0
            cur_sum = h0.sum()
        h0 = project_to_box_sum(h0, target_sum)

    # ensure feasibility
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_box_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ---- simulated annealing / gradient descent phase ----
    phase1_end = start_time + budget_s * 0.85
    while time.time() < phase1_end:
        elapsed = time.time() - start_time
        total_phase = phase1_end - start_time
        # step scale slowly shrinks
        step_factor = max(0.05, 1.0 - elapsed / total_phase)
        # temperature for occasional uphill moves
        T = max(1e-8, 0.001 * (1.0 - elapsed / total_phase))

        # decide direction: gradient-like (70%) or random (30%)
        if rng.random() < 0.7:
            # gradient direction based on worst self-overlap
            corr_self = np.correlate(cur_h, cur_h, mode='full')
            center = len(corr_self) // 2
            corr_self[center] = np.inf   # ignore zero shift
            min_idx = int(np.argmin(corr_self))
            k_star = min_idx - center
            dir_vec = np.zeros_like(cur_h)
            if k_star >= 0:
                if k_star < n:
                    dir_vec[:n - k_star] = cur_h[k_star:]
                    dir_vec[k_star:] += cur_h[:n - k_star]
            else:
                k = -k_star
                if k < n:
                    dir_vec[k:] = cur_h[:n - k]
                    dir_vec[:n - k] += cur_h[k:]
        else:
            # random direction with zero mean
            dir_vec = rng.normal(size=n)
            dir_vec -= dir_vec.mean()

        # random step magnitude
        step = step_factor * (rng.random() ** 2)
        cand = cur_h + step * dir_vec
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_to_box_sum(cand, n / 2.0)

        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
