def upsample_h(initial_h, n_target):
    src = np.asarray(initial_h, dtype=float)
    n_src = src.shape[0]
    # coordinates: src positions 0..n_src-1
    x_target = np.linspace(0, n_src - 1, n_target)
    interp = np.interp(x_target, np.arange(n_src), src)
    # project to target sum:
    target_sum = n_target / 2.0
    interp = project_sum_and_bounds(interp, target_sum)
    return interp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full')
    return np.max(v) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and adjust to have sum exactly target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # spread uniformly
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # fix residual by greedy scan
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def upsample_h(src, n_target):
    src = np.asarray(src, dtype=float)
    n_src = src.shape[0]
    # interpolate linearly
    xs = np.linspace(0, n_src - 1, n_target)
    interp = np.interp(xs, np.arange(n_src), src)
    target_sum = n_target / 2.0
    return project_sum_and_bounds(interp, target_sum)

    interp = np.interp(xs, np.arange(n_src), src)

def binary_hill_climb(h_bin, time_budget, rng):
    """Simulated‑annealing style swaps on a binary (0/1) vector with exact sum."""
    n = h_bin.shape[0]
    cur_h = h_bin.copy()
    cur_c5 = compute_c5(cur_h)
    start = time.time()
    T0 = 0.3
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        T = T0 * max(0.01, 1.0 - elapsed / time_budget)
        i_one = rng.integers(n)
        while cur_h[i_one] == 0:
            i_one = rng.integers(n)
        i_zero = rng.integers(n)
        while cur_h[i_zero] == 1:
            i_zero = rng.integers(n)
        cur_h[i_one] = 0.0
        cur_h[i_zero] = 1.0
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        accept = delta < 0 or rng.random() < np.exp(-delta / T)
        if accept:
            cur_c5 = new_c5
        else:
            cur_h[i_one] = 1.0
            cur_h[i_zero] = 0.0
    return cur_h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 200                                 # discretization size (<1 k)
    target_sum = n / 2.0
    pop = 12                                # population size for continuous search
    h_pop = np.empty((pop, n), dtype=float)
    velocities = np.zeros_like(h_pop)

    # initialise population
    if 'initial_h_values' in globals():
        h_pop[0] = upsample_h(initial_h_values, n)
    else:
        h_pop[0] = project_sum_and_bounds(rng.random(n), target_sum)
    for i in range(1, pop):
        h_pop[i] = project_sum_and_bounds(rng.random(n), target_sum)

    best_h = None
    best_c5 = np.inf
    dx = 2.0 / n
    iter_count = 0
    no_improve = 0

    # ---------- continuous optimisation ----------
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s * 0.6:      # reserve time for binary refinement
            break
        alpha = 0.15 * max(0.02, 1.0 - elapsed / budget_s)
        momentum = 0.9
        for i in range(pop):
            h = h_pop[i]
            v = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(v) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            idx_max = np.argmax(v)
            k = idx_max - (n - 1)
            g = np.zeros(n, dtype=float)
            if k >= 0:
                end = n - k
                g[:end] += h[k:]
                g[k:] += h[:end]
            else:
                kk = -k
                end = n - kk
                g[kk:] += h[:end]
                g[:end] += h[kk:]

            velocities[i] = momentum * velocities[i] + (1.0 - momentum) * g
            h_new = h + alpha * velocities[i]
            if iter_count % 30 == 0:
                noise = 0.02 * max(0.1, 1.0 - elapsed / budget_s)
                h_new += rng.normal(scale=noise, size=n)
            h_pop[i] = project_sum_and_bounds(h_new, target_sum)

        iter_count += 1

        # rejuvenate half the population periodically
        if iter_count % 80 == 0:
            c5_vals = np.empty(pop)
            for i in range(pop):
                c5_vals[i] = np.max(np.correlate(h_pop[i], 1.0 - h_pop[i], mode='full')) * dx
            order = np.argsort(c5_vals)
            half = pop // 2
            for idx in order[half:]:
                pert = best_h + rng.normal(scale=0.05, size=n)
                h_pop[idx] = project_sum_and_bounds(pert, target_sum)
                velocities[idx] = np.zeros(n)

        if no_improve > 20000:
            for i in range(pop):
                h_pop[i] = project_sum_and_bounds(best_h + rng.normal(scale=0.1, size=n), target_sum)
                velocities[i] = np.zeros(n)
            no_improve = 0

    # optional local refinement with SLSQP (if SciPy is available)
    try:
        from scipy.optimize import minimize, LinearConstraint, Bounds
        def obj(x):
            return np.max(np.correlate(x, 1.0 - x, mode='full')) * dx
        cons = LinearConstraint(np.ones((1, n)), [target_sum], [target_sum])
        bnds = Bounds(0.0, 1.0)
        res = minimize(obj, best_h, method='SLSQP', constraints=[cons],
                       bounds=bnds, options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
        if res.success:
            cand = project_sum_and_bounds(res.x, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_h, best_c5 = cand, cand_c5
    except Exception:
        pass

    # ---------- binary refinement ----------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        # start from top‑half rounding of best continuous solution
        order = np.argsort(-best_h)
        binary_start = np.zeros(n, dtype=float)
        binary_start[order[:n // 2]] = 1.0
        bin_h, bin_c5 = binary_hill_climb(binary_start, remaining, rng)
        if bin_c5 < best_c5:
            best_h, best_c5 = bin_h, bin_c5

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
