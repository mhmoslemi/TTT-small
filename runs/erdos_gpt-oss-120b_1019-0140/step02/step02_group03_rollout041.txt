def milp_optimize(n, target_sum, time_limit):
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == target_sum, t >= 0]
    # optional bound t <= target_sum
    constraints.append(t <= target_sum)
    for k in range(1, n):
        m = n - k
        y = cp.Variable(m, boolean=True)
        constraints.append(y <= x[:m])
        constraints.append(y <= x[k:])
        constraints.append(y >= x[:m] + x[k:] - 1)
        # D_k
        Dk = cp.sum(x[:m]) - cp.sum(y)
        constraints.append(Dk <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    # solve
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, max_seconds=time_limit, verbose=False)
        except Exception:
            return None, None
    if x.value is None:
        return None, None
    h = np.array(x.value).astype(float)

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def milp_optimize(n, target_sum, time_limit=None):
    """Binary MILP: minimize the worst D_k = sum_i x_i(1-x_{i+k}) over all shifts."""
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [
        cp.sum(x) == target_sum,
        t >= 0,
        t <= target_sum
    ]
    # For each positive shift create auxiliary variables and constraints
    for k in range(1, n):
        m = n - k
        y = cp.Variable(m, boolean=True)           # y_i = x_i * x_{i+k}
        constraints.append(y <= x[:m])
        constraints.append(y <= x[k:])
        constraints.append(y >= x[:m] + x[k:] - 1)
        Dpos = cp.sum(x[:m]) - cp.sum(y)            # sum_i x_i (1 - x_{i+k})
        Dneg = cp.sum(x[k:]) - cp.sum(y)            # sum_i x_{i+k} (1 - x_i)
        constraints.append(Dpos <= t)
        constraints.append(Dneg <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    solve_args = {"solver": "CBC", "warm_start": True, "verbose": False}
    if time_limit is not None:
        solve_args["max_seconds"] = time_limit
    try:
        prob.solve(**solve_args)
    except Exception:
        # fallback to GLPK_MI
        solve_args = {"solver": "GLPK_MI", "warm_start": True, "verbose": False}
        if time_limit is not None:
            solve_args["max_seconds"] = time_limit
        prob.solve(**solve_args)
    if x.value is None:
        return None, None
    h = np.array(x.value).astype(float)
    t_val = float(t.value) if t.value is not None else None
    return h, t_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine dimension
    if "initial_h_values" in globals():
        n = int(np.asarray(initial_h_values).size)
    else:
        n = 80
    target_sum = n / 2.0

    # Initialise best solution
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        h0 = project_to_box_sum(h0, target_sum)
    else:
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # --- MILP phase ----------------------------------------------------
    elapsed = time.time() - start_time
    time_remaining = budget_s - elapsed
    if time_remaining > 30.0:
        # give the MILP at most half of the remaining budget, capped at 300 s
        milp_time = min(time_remaining * 0.5, 300.0)
        h_bin, t_val = milp_optimize(n, target_sum, time_limit=milp_time)
        if h_bin is not None:
            # Force binary values and exact sum
            h_bin = np.rint(h_bin)
            h_bin = project_to_box_sum(h_bin, target_sum)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5 - 1e-12:
                best_h = h_bin.copy()
                best_c5 = c5_bin

    # --- Local search / simulated annealing ----------------------------
    cur_h = best_h.copy()
    cur_c5 = best_c5
    base_step = 0.5
    # detect binary flag (exact 0/1 within tolerance)
    binary_flag = np.all(np.logical_or(cur_h < 1e-6, cur_h > 1 - 1e-6))

    while time.time() - start_time < budget_s - 0.5:
        elapsed = time.time() - start_time
        time_frac = elapsed / budget_s
        temperature = max(0.001, (1.0 - time_frac) * 0.02)

        if binary_flag:
            idx_one = np.where(cur_h > 0.5)[0]
            idx_zero = np.where(cur_h <= 0.5)[0]
            if idx_one.size and idx_zero.size:
                i = rng.choice(idx_one)
                j = rng.choice(idx_zero)
                cand = cur_h.copy()
                cand[i] = 0.0
                cand[j] = 1.0
            else:
                # fallback to a small random perturbation
                d = rng.normal(size=n)
                d -= d.mean()
                step = base_step * (rng.random() ** 2)
                cand = cur_h + step * d
                cand = project_to_box_sum(cand, target_sum)
        else:
            d = rng.normal(size=n)
            d -= d.mean()
            step = base_step * (rng.random() ** 2)
            cand = cur_h + step * d
            cand = project_to_box_sum(cand, target_sum)

        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5
        if delta < -1e-12 or rng.random() < np.exp(-delta / temperature):
            cur_h = cand
            cur_c5 = cand_c5
            if not binary_flag:
                binary_flag = np.all(np.logical_or(cur_h < 1e-6, cur_h > 1 - 1e-6))
            if cur_c5 < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = cur_c5

        if rng.random() < 0.005:
            base_step *= 0.9

    # Final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
