import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0, max_iter=40):
    """Project v onto [lo,hi]^n while enforcing sum == target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# ---------- Constraints for SLSQP ----------
class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{'type': 'eq', 'fun': sum_con}]
    cons += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

# ---------- Main ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine dimension and an optional initial vector
    if 'initial_h_values' in globals():
        h_initial = np.asarray(initial_h_values, dtype=float)
        n = h_initial.size
    else:
        n = 80
        h_initial = None

    target_sum = n / 2.0

    # Constraints for SLSQP (shared)
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # ---------- Differential Evolution ----------
    popsize = 30
    pop = np.empty((popsize, n), dtype=float)
    for i in range(popsize):
        v = rng.random(n)
        pop[i] = project_to_box_sum(v, target_sum)
    if h_initial is not None:
        pop[0] = project_to_box_sum(h_initial, target_sum)

    pop_f = np.empty(popsize, dtype=float)
    for i in range(popsize):
        pop_f[i] = compute_c5(pop[i])

    best_idx = int(np.argmin(pop_f))
    best_h = pop[best_idx].copy()
    best_c5 = pop_f[best_idx]

    F = 0.5
    CR = 0.9
    de_deadline = start_time + budget_s * 0.7
    while time.time() < de_deadline:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project_to_box_sum(trial, target_sum)

            trial_f = compute_c5(trial)
            if trial_f < pop_f[i] - 1e-12:
                pop[i] = trial
                pop_f[i] = trial_f
                if trial_f < best_c5 - 1e-12:
                    best_h = trial.copy()
                    best_c5 = trial_f
                    if best_c5 < 0.3808:
                        de_deadline = time.time()  # break early
                        break
        # early break if deadline already passed
        if time.time() >= de_deadline:
            break

    # ---------- Local SLSQP refinement ----------
    def slsqp_refine(h_start):
        t_start = compute_c5(h_start)
        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800)
        if not res.success:
            return None, None
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur_sum = h_opt.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt

    # one run from current best, then a couple of random restarts
    for attempt in range(3):
        if time.time() - start_time > budget_s - 0.5:
            break
        if attempt == 0:
            h_start = best_h
        else:
            h_start = rng.random(n)
            h_start = project_to_box_sum(h_start, target_sum)
        h_opt, c5_opt = slsqp_refine(h_start)
        if h_opt is not None and c5_opt < best_c5 - 1e-12:
            best_h = h_opt
            best_c5 = c5_opt
            if best_c5 < 0.3808:
                break

    # ---------- Simple hill-climbing ----------
    cur_h = best_h.copy()
    cur_c5 = best_c5
    base_step = 0.5
    while time.time() - start_time < budget_s - 0.2:
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2)
        cand = cur_h + step * d
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 < cur_c5 - 1e-12:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                if best_c5 < 0.3808:
                    break
        if rng.random() < 0.001:
            base_step *= 0.9

    # ---------- Final projection ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
