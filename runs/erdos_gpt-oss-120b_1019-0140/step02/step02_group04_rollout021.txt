    cur_val = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val

import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5(h, dx):
    """Maximum overlap integral for a step function h."""
    # np.correlate gives full convolution (cross‑correlation) of h and (1‑h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # uniform correction, then greedy fix any residual
    vec += diff / vec.size
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - vec.sum()
    n = vec.size
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random vector in [0,1] with exact sum target_sum."""
    x = rng.random(n)
    s = x.sum()
    if s > 0:
        x = x * (target_sum / s)
    else:
        x = np.full(n, target_sum / n)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    # greedy fix residual
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        attempts += 1
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
        else:
            idx = np.where(x > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
    return x


# ---------- Local search ----------
def local_search(start_h, dx, target_sum, time_limit, rng):
    """
    Simple hill‑climbing / simulated‑annealing style search.
    Moves consist of transferring a small amount of mass between two positions.
    """
    cur = start_h.copy()
    cur_val = compute_c5(cur, dx)
    best = cur.copy()
    best_val = cur_val
    n = len(cur)
    step = 0.1          # maximal transfer in a single move
    min_step = 1e-4
    # temperature schedule for occasional uphill acceptance
    T0 = 5e-3
    T_end = 1e-6

    start = time.time()
    it = 0
    while time.time() < time_limit:
        it += 1
        # schedule temperature
        elapsed = time.time() - start
        frac = min(1.0, elapsed / (time_limit - start + 1e-12))
        T = T0 * (T_end / T0) ** frac

        # pick move type
        if rng.random() < 0.7:
            # random donor / recipient
            donors = np.where(cur > 1e-12)[0]
            recipients = np.where(cur < 1.0 - 1e-12)[0]
            if donors.size == 0 or recipients.size == 0:
                break
            i = rng.choice(donors)
            j = rng.choice(recipients)
        else:
            # attempt smoothing move targeting the worst shift
            corr = np.correlate(cur, 1.0 - cur, mode="full")
            max_idx = int(np.argmax(corr))
            k = max_idx - (n - 1)       # shift value
            # try a few random i’s that admit a partner
            found = False
            for _ in range(5):
                i = rng.integers(n)
                if cur[i] <= 1e-12:
                    continue
                j = i + k
                if 0 <= j < n and cur[j] < 1.0 - 1e-12:
                    found = True
                    break
            if not found:
                # fall back to random move
                donors = np.where(cur > 1e-12)[0]
                recipients = np.where(cur < 1.0 - 1e-12)[0]
                if donors.size == 0 or recipients.size == 0:
                    break
                i = rng.choice(donors)
                j = rng.choice(recipients)

        max_delta = min(step, cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5(cand, dx)

        if cand_val < cur_val - 1e-12:
            cur = cand
            cur_val = cand_val
            step = min(step * 1.05, 0.3)          # grow a bit after success
            if cand_val < best_val:
                best = cand.copy()
                best_val = cand_val
                if best_val < 0.3808:
                    break
        else:
            # uphill acceptance according to temperature
            if rng.random() < np.exp(-(cand_val - cur_val) / (T + 1e-12)):
                cur = cand
                cur_val = cand_val
            # shrink step slowly on failure
            step = max(step * 0.99, min_step)

    return best, best_val


def binary_hill_climb(h_bin, cur_val, dx, time_limit):
    """
    Simple swap hill‑climbing on a binary vector.
    h_bin is an int array of 0/1 with exactly target_sum ones.
    """
    best = h_bin.copy()
    best_val = cur_val
    n = len(h_bin)
    while time.time() < time_limit:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best = cand
            best_val = cand_val
            if best_val < 0.3808:
                break
    return best.astype(float), best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # Initialise best solution
    if h0 is not None:
        best_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        best_h = generate_random_feasible(n, target_sum, rng)
    best_val = compute_c5(best_h, dx)

    # ---- multi‑restart local search (random starts) ----
    # Allocate roughly 60 % of time for independent restarts
    restart_time_budget = budget_s * 0.6
    while time.time() - start_time < restart_time_budget:
        # each restart gets ~5 s (or the remaining budget)
        remaining = start_time + restart_time_budget - time.time()
        if remaining < 2.0:
            break
        per_restart = min(5.0, remaining)
        h_start = generate_random_feasible(n, target_sum, rng)
        h_ref, val_ref = local_search(h_start, dx, target_sum,
                                      time.time() + per_restart, rng)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref
            if best_val < 0.3808:
                break

    # ---- extended refinement from the current best ----
    time_left = budget_s - (time.time() - start_time) - 1.0  # keep 1 s margin
    if time_left > 0.5 and best_val > 0.3808:
        best_h, best_val = local_search(best_h, dx, target_sum,
                                        time.time() + time_left, rng)

    # ---- optional binary hill‑climbing phase ----
    time_left = budget_s - (time.time() - start_time) - 0.5
    if time_left > 0.5:
        # round to binary by taking the largest entries
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argsort(-best_h)[:int(target_sum)]

    # ---- final safety enforcement ----
    best_h = project_to_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_val = compute_c5(best_h, dx)

    return best_h, best_val, n
