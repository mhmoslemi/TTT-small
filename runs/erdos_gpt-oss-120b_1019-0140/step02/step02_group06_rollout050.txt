import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral (scaled by dx) for a discretized h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_to_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the intersection of the box [0,1]^n and the hyperplane sum(x)=target.
    Uses a simple bisection on the shift λ such that sum(clip(x+λ)) = target.
    """
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + lo, 0.0, 1.0)

def _random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum exactly target."""
    return _project_to_sum(rng.random(n), target)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.size
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0  # because ∫h dx = 1 ⇒ sum(h)=n/2

    # Initialise a feasible point
    if h0 is None:
        cur_h = _random_feasible(n, target_sum, rng)
    else:
        h0 = np.clip(h0, 0.0, 1.0)
        cur_h = _project_to_sum(h0, target_sum)

    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # Simulated annealing parameters
    max_temp = 5e-3          # initial temperature (roughly the scale of C5 differences)
    min_temp = 1e-6
    restart_after = 50000   # steps without improvement → restart
    shake_after = 200000    # steps without improvement → global perturbation
    steps_without_improve = 0

    # Main optimisation loop
    while time.time() - start_time < budget_s:
        elapsed = time.time() - start_time
        # Linear cooling schedule
        temp = max(min_temp, max_temp * (1.0 - elapsed / budget_s))

        # Propose a move: transfer a small amount from one coordinate to another
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # Choose direction randomly
        if rng.random() < 0.5:
            # move from i to j
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer * (temp / max_temp)
            if delta == 0.0:
                continue
            cur_h[i] -= delta
            cur_h[j] += delta
            direction = 1
        else:
            # move from j to i
            max_transfer = min(cur_h[j], 1.0 - cur_h[i])
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer * (temp / max_temp)
            if delta == 0.0:
                continue
            cur_h[j] -= delta
            cur_h[i] += delta
            direction = -1

        new_c5 = compute_c5(cur_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / temp):
            cur_c5 = new_c5
            steps_without_improve = 0
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert the move
            if direction == 1:
                cur_h[i] += delta
                cur_h[j] -= delta
            else:
                cur_h[j] += delta
                cur_h[i] -= delta
            steps_without_improve += 1

        # Periodic restart if stuck
        if steps_without_improve >= restart_after:
            cur_h = _random_feasible(n, target_sum, rng)
            cur_c5 = compute_c5(cur_h)
            steps_without_improve = 0

        # Occasionally shake the whole solution
        if steps_without_improve >= shake_after:
            cur_h = _project_to_sum(cur_h + rng.normal(scale=0.05, size=n), target_sum)
            cur_c5 = compute_c5(cur_h)
            steps_without_improve = 0

    # Final safety projection (ensures exact feasibility)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
