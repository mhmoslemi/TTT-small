def binary_anneal(n, target, rng, time_budget):
    # target integer (target_ones)
    # start with random binary
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=target, replace=False)] = 1.0
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    # cooling schedule based on elapsed time fraction
    # We'll define T0 = 0.5, Tf = 1e-4
    last_improve = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max(1e-4, 0.5 * (1 - frac))  # linear cooling
        # Choose swap
        idx1 = rng.choice(np.flatnonzero(h > 0.5))
        idx2 = rng.choice(np.flatnonzero(h < 0.5))
        # swap
        h[idx1], h[idx2] = 0, 1  # set them accordingly
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                last_improve = 0
            else:
                last_improve += 1
        else:
            # revert swap
            h[idx1], idx2 # revert: set back original
            # but we need original values: they were idx1 had 1, idx2 had 0 before swap.
            h[idx1] = 1.0
            h[idx2] = 0.0
            # cur_c5 unchanged

        # optional restart if stuck

            # restart randomly
            h.fill(0)
            h[rng.choice(n, size=target, replace=False)] = 1.0

def continuous_optimize(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n/2
    dx = 2.0/n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 1.0
    max_beta = 15.0
    alpha0 = 0.15
    no_improve = 0
    while time.time() - start < time_budget:
        # compute correlation vector
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max()*dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # update beta gradually
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        # compute softmax weights
        # stable: subtract max
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        # avoid overflow, w large sums; but fine for moderate beta.
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w /= w_sum

        # compute gradient
        g = np.zeros(n)
        offset = n - 1
        # vectorized approach: For each shift s, weight w_s, compute contributions
        # loops over shifts
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            s = idx - offset
            if s >= 0:
                tail = slice(0, n - s)
                head = slice(s, n)
            else:
                s2 = -s
                tail = slice(s2, n)
                head = slice(0, n - s2)
            # gradient wrt tail positions: + weight * (1 - h[head])
            g[tail] += weight * (1.0 - h[head])
            # gradient wrt head positions: - weight * h[tail]
            g[head] -= weight * h[tail]

        # enforce zero-sum: subtract mean
        g -= np.mean(g)

        # step size schedule: linear decay based on elapsed time

def binary_anneal(n, target_ones, rng, time_budget):
    # target_ones is int, target sum * 1 (since h values are 0 or 1)
    # We will keep h as float array.
    h = np.zeros(n, dtype=float)
    # random initial placement of target ones
    idx = rng.choice(n, size=target_ones, replace=False)
    h[idx] = 1.0
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # maintain lists of indices
    ones_list = list(idx)
    zeros_list = [i for i in range(n) if h[i] == 0.0]
    start = time.time()
    no_improve = 0
    # pre-calc default T schedule variables
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        # Temperature decreasing linearly from T0=0.5 to Tf=1e-4
        T = 0.5 * (1.0 - frac) + 1e-4
        # pick random indices from lists
        i_pos = rng.integers(len(ones_list))
        j_pos = rng.integers(len(zeros_list))
        i = ones_list[i_pos]
        j = zeros_list[j_pos]
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            # to avoid overflow if delta large, take min exponent
            # but delta positive always < maybe 2.0, okay
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # keep swap
            cur_c5 = new_c5
            # update index lists
            ones_list[i_pos] = j
            zeros_list[j_pos] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        # restart if stuck
        if no_improve > 20000:
            # random restart
            h.fill(0.0)
            idx2 = rng.choice(n, size=target_ones, replace=False)
            h[idx2] = 1.0
            cur_c5 = compute_c5(h)
            ones_list = list(idx2)
            zeros_list = [i for i in range(n) if h[i] == 0.0]
            no_improve = 0
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (discretized)."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (iterative repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute uniformly first
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # greedy fix remaining discrepancy
        if abs(diff) > 1e-12:
            inc = diff > 0
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if inc and x[i] < 1.0:
                    d = min(diff, 1.0 - x[i])
                    x[i] += d
                    diff -= d
                elif not inc and x[i] > 0.0:
                    d = min(-diff, x[i])
                    x[i] -= d
                    diff += d
    return x

def continuous_optimize(h0, time_budget, seed):
    """Gradient descent on a soft‑max approximation of max‑overlap."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 1.0
    max_beta = 15.0
    alpha0 = 0.15
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.ones_like(w) / w.size
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            s = idx - offset
            if s >= 0:
                tail = slice(0, n - s)
                head = slice(s, n)
            else:
                s2 = -s
                tail = slice(s2, n)
                head = slice(0, n - s2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        g -= np.mean(g)  # zero‑sum direction
        elapsed = time.time() - start
        alpha = alpha0 * max(0.05, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        if no_improve > 5000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.03, size=n), target)
            no_improve = 0
    return best_h, best_c5

            cand_j_rel = np.where((tail_vals == 0) & (head_vals == 1))[0]
            if contrib_i_rel.size and cand_j_rel.size:
                i = rng.choice(contrib_i_rel)
                j = rng.choice(cand_j_rel)
                found = True
        else:
            s2 = -s
            tail = slice(s2, n)
            head = slice(0, n - s2)
            tail_vals = h[tail]
            head_vals = h[head]
            contrib_i_rel = np.where((tail_vals == 1) & (head_vals == 0))[0]
            cand_j_rel = np.where((tail_vals == 0) & (head_vals == 1))[0]
            if contrib_i_rel.size and cand_j_rel.size:
                i = s2 + rng.choice(contrib_i_rel)
                j = s2 + rng.choice(cand_j_rel)
                found = True
        if not found:
            ones_arr = np.where(h == 1)[0]
            zeros_arr = np.where(h == 0)[0]
            i = rng.choice(ones_arr)
            j = rng.choice(zeros_arr)
        # propose swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        elif rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        if no_improve > 20000:
            # random restart
            h.fill(0.0)
            idx = rng.choice(n, size=target_ones, replace=False)
            h[idx] = 1.0
            cur_c5 = compute_c5(h)
            no_improve = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 120  # even size for exact half‑mass
        h0 = rng.random(n)
    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)

    # Phase 1: continuous gradient descent (up to ~250 s)
    grad_time = min(250.0, remaining * 0.25)
    if grad_time > 5.0:
        cont_h, cont_c5 = continuous_optimize(h0, grad_time, seed + 1)
        best_h, best_c5 = cont_h, cont_c5
        remaining -= grad_time
    else:
        best_h, best_c5 = h0, compute_c5(h0)

    # Phase 2: binary simulated annealing for the rest (reserve 5 s for final tweak)
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    if remaining > 6.0:
        bin_time = remaining - 5.0
        bin_h, bin_c5 = binary_anneal(n, int(target), rng, bin_time)
        if bin_c5 < best_c5:
            best_h, best_c5 = bin_h, bin_c5
        remaining = max(0.0, budget_s - (time.time() - start))

    # Phase 3: optional final continuous refinement with any left time
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    if remaining > 1.0:
        fin_h, fin_c5 = continuous_optimize(best_h, remaining, seed + 100)
        if fin_c5 < best_c5:
            best_h, best_c5 = fin_h, fin_c5

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
