def local_anneal(h_start, cur_best, start_time, deadline, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_val = cur_best
    cur_val = cur_best
    n = len(h)
    while time.time() < deadline - 0.1:

        i = rng.integers(n)
        j = rng.integers(n)
        if i==j:
            continue
        # Transfer small delta = rng.random()*max_transfer
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * 0.5  # limit delta
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        # Accept if better, else maybe accept with prob
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # simple Metropolis with temperature decreasing
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            # Temperature scaling
            T = remaining / (deadline - start_time)  # fraction 0..1
            prob = np.exp((cur_val - new_val) / (1e-3 + T))
            if prob > rng.random():
                cur_val = new_val
            else:
                # revert
                h[i] += delta
                h[j] -= delta
        if best_val < 0.3808:
            break
    return best_h, best_val

def project_to_bounds_and_sum(h, target):

        return h
    # Distribute diff uniformly to not bound positions
    # Simple loop:
    for _ in range(10):
        # indices that not at bound
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size == 0:
            # none free, distribute to all but limited?
            # fallback: add equally and clip again
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-12:
                break
            continue
        # distribute equally among free
        delta = diff / free.size
        # Clip per free element
        h[free] += delta

        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)

            break
    return h

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def enforce_sum(vec, target):
    cur = np.sum(vec)
    diff = target - cur
    if abs(diff) < 1e-12:
        return vec
    # Adjust first entry
    vec[0] = np.clip(vec[0] + diff/2, 0.0, 1.0)
    vec[1] = np.clip(vec[1] + diff/2, 0.0, 1.0)
    # Recompute diff
    cur = np.sum(vec)
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute to any free positions
        free = np.where((vec > 1e-12) & (vec < 1.0 - 1e-12))[0]
        if free.size:
            delta = diff / free.size
            for i in free:
                vec[i] = np.clip(vec[i] + delta, 0.0, 1.0)
    # final clip
    vec = np.clip(vec, 0.0, 1.0)
    return vec

def local_anneal(h_start, cur_best, start_time, deadline, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_val = cur_best
    cur_val = cur_best
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # contribution product for this shift
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # choose j that can receive mass
        candidates = np.where(h < 1.0 - 1e-12)[0]
        if candidates.size == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:

            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(candidates)
        max_transfer = min(h[i], 1.0 - h[j])

        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta

import time
import numpy as np

def compute_c5(h):
    """Exact C5 for a discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(vec, target):
    """Project vec onto [0,1] with exact sum == target (within tolerance)."""
    v = np.clip(vec, 0.0, 1.0)
    cur = np.sum(v)
    diff = target - cur
    if abs(diff) < 1e-12:
        return v
    # try to fix by adjusting first two entries
    v[0] = np.clip(v[0] + diff / 2.0, 0.0, 1.0)
    v[1] = np.clip(v[1] + diff / 2.0, 0.0, 1.0)
    cur = np.sum(v)
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute remaining diff among interior free slots
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size:
            delta = diff / free.size
            v[free] = np.clip(v[free] + delta, 0.0, 1.0)
    return np.clip(v, 0.0, 1.0)

def local_anneal(h_start, cur_best, start_time, deadline, rng):
    """Targeted continuous annealing (same idea as the previous implementation)."""
    h = h_start.copy()
    best_h = h.copy()
    best_val = cur_best
    cur_val = cur_best

        k = max_idx - (n - 1)

            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            remaining = deadline - time.time()
            total = deadline - start_time
            T = max(0.1 * remaining / total, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        if best_val < 0.3808:
            break
    return best_h, best_val

def binary_hill_climb(h_bin, cur_best, start_time, deadline, rng):
    """Swap‑based simulated annealing on a binary vector (weight = n/2)."""
    n = h_bin.shape[0]
    cur_val = cur_best
    best_h = h_bin.copy()
    best_val = cur_best
    # maintain sets of indices for fast sampling
    ones = set(np.where(h_bin == 1)[0])
    zeros = set(np.where(h_bin == 0)[0])
    while time.time() < deadline - 0.1:
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # perform swap
        h_bin[i] = 0
        h_bin[j] = 1
        new_val = compute_c5(h_bin)
        if new_val < cur_val:
            cur_val = new_val
            ones.remove(i)
            zeros.add(i)
            ones.add(j)
            zeros.remove(j)
            if new_val < best_val:
                best_val = new_val
                best_h = h_bin.copy()
                if best_val < 0.3808:
                    break
        else:
            # Metropolis acceptance
            remaining = deadline - time.time()
            total = deadline - start_time
            T = max(0.1 * remaining / total, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
                ones.remove(i)
                zeros.add(i)
                ones.add(j)
                zeros.remove(j)
            else:
                # revert
                h_bin[i] = 1
                h_bin[j] = 0
        # occasional reshuffle of sets for safety
        if rng.random() < 0.01:
            ones = set(np.where(h_bin == 1)[0])
            zeros = set(np.where(h_bin == 0)[0])
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 200
        v = rng.random(n)
        v = v - v.mean() + 0.5
        v = np.clip(v, 0.0, 1.0)
        target = n / 2.0
        cur = v.sum()
        diff = target - cur
        v[0] = np.clip(v[0] + diff / 2.0, 0.0, 1.0)
        v[1] = np.clip(v[1] + diff / 2.0, 0.0, 1.0)
        h0 = v

    target = n / 2.0
    h = enforce_sum(h0, target)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    # ----- smoothed gradient descent ------------------------------------
    dx = 2.0 / n
    beta = 5.0
    max_iter = 12000
    eval_every = max(1, max_iter // 30)
    for it in range(max_iter):
        if time.time() > deadline - 5.0:
            break
        raw = np.correlate(h, 1.0 - h, mode="full")
        max_raw = np.max(raw)
        exp_term = np.exp(beta * (raw - max_raw))
        Z = np.sum(exp_term)
        w = exp_term / Z  # softmax weights
        grad = np.ones(n)
        for idx, wk in enumerate(w):
            if wk == 0.0:
                continue
            k = idx - (n - 1)
            if k >= 0:
                grad[: n - k] -= wk * h[k:]
                grad[k:]     -= wk * h[: n - k]
            else:
                kk = -k
                grad[kk:]    -= wk * h[: n - kk]
                grad[: n-kk] -= wk * h[kk:]
        grad = grad * dx
        step = 0.5 / np.sqrt(it + 1)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - np.sum(h_new)
        if abs(diff) > 1e-12:
            h_new += diff / n
            h_new = np.clip(h_new, 0.0, 1.0)
            if abs(np.sum(h_new) - target) > 1e-9:
                h_new = enforce_sum(h_new, target)
        h = h_new
        if it % eval_every == 0:
            cur_val = compute_c5(h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        if (it + 1) % 1000 == 0 and beta < 30:
            beta *= 1.5

    # ----- local continuous annealing -----------------------------------
    if time.time() < deadline - 1.0:
        best_h, best_val = local_anneal(best_h, best_val, start, deadline, rng)

    # ----- attempt binary improvement ------------------------------------
    # Round to binary, then adjust count to exact target
    h_bin = (best_h >= 0.5).astype(float)
    cur_ones = int(np.round(np.sum(h_bin)))
    need = int(target) - cur_ones
    if need != 0:
        # use original values as tie‑breaker
        order = np.argsort(best_h) if need > 0 else np.argsort(-best_h)
        for idx in order:
            if need == 0:
                break
            if need > 0 and h_bin[idx] == 0:
                h_bin[idx] = 1.0
                need -= 1
            elif need < 0 and h_bin[idx] == 1:
                h_bin[idx] = 0.0
                need += 1
    h_bin = enforce_sum(h_bin, target)  # safety clip
    bin_val = compute_c5(h_bin)
    if bin_val < best_val:
        best_h, best_val = h_bin, bin_val
    # binary hill‑climb if time permits
    if time.time() < deadline - 1.0:
        best_h, best_val = binary_hill_climb(best_h.copy(), best_val, start, deadline, rng)

    # ----- final correction & return ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
