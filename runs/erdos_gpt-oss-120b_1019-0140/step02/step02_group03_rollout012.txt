import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # binary search for lambda
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=2000):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type':'eq','fun':sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=lambda x: x[-1],
                   x0=x0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter':maxiter,'ftol':1e-12,'disp':False})
    return res

def overlap_gradient(k, h):
    n = h.shape[0]
    if k >= 0:
        # forward shift
        fwd = np.concatenate((h[k:], np.zeros(k, dtype=h.dtype)))
        bwd = np.concatenate((np.zeros(k, dtype=h.dtype), h[:n - k]))
    else:
        kk = -k
        fwd = np.concatenate((np.zeros(kk, dtype=h.dtype), h[:n - kk]))
        bwd = np.concatenate((h[kk:], np.zeros(kk, dtype=h.dtype)))
    return 1.0 - fwd - bwd

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    # prepare constraints for SLSQP (if we use it)
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    # Phase 1: SLSQP restarts (≈30% time)
    phase1_deadline = start_time + budget_s * 0.3
    while time.time() < phase1_deadline:
        # random feasible start
        h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, target_sum)
        t_start = compute_c5(h_start)
        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=1500)
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # enforce exact sum
        cur_sum = h_opt.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            # early stop if good enough
            if best_c5 < 0.3806:
                break
    # Phase 2: Simulated annealing / gradient descent
    cur_h = best_h.copy()
    cur_c5 = best_c5
    # SA parameters
    T = 0.01
    sigma = 0.2
    decay_T = 0.9995
    decay_sigma = 0.9997
    min_T = 1e-9
    min_sigma = 1e-5
    # acceptance stats
    it = 0
    last_improve_iter = 0
    # SA loop until time
    while True:
        now = time.time()
        if now - start_time > budget_s - 0.5:
            break
        # propose new candidate
        cand = cur_h + sigma * rng.standard_normal(n)
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, min_T)):
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 + 1e-12 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                last_improve_iter = it
                if best_c5 < 0.3805:
                    break
        # update temperature and step
        T = max(min_T, T * decay_T)
        sigma = max(min_sigma, sigma * decay_sigma)
        it += 1
        # occasional restart if stagnated
        if it - last_improve_iter > 200000:
            # reinitialize from best with a random perturbation
            cur_h = best_h + sigma * rng.standard_normal(n)
            cur_h = project_to_box_sum(cur_h, target_sum)
            cur_c5 = compute_c5(cur_h)
            last_improve_iter = it
    # Phase 3: simple projected gradient descent on active lag
    # ensure we have some time left
    grad_deadline = start_time + budget_s - 0.2
    step_size = 0.05
    while time.time() < grad_deadline:
        # find lag with max overlap
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        k = np.argmax(corr) - (n - 1)  # convert index to lag
        grad = overlap_gradient(k, cur_h)
        # move opposite direction
        cand = cur_h - step_size * grad
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = cur_c5
        # decay step size
        step_size *= 0.9999
        if step_size < 1e-6:
            break
    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

        else:
            high = mid
