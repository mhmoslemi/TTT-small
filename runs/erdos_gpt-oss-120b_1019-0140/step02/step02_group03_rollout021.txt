def compute_gradient(h, k, dx):
    n = h.shape[0]
    grad = np.full(n, dx)
    if k >= 0:
        # subtract h_{i+k}
        if k < n:
            grad[:n-k] -= h[k:] * dx
            # subtract h_{i-k}
            grad[k:] -= h[:n-k] * dx
    else:
        kk = -k
        if kk < n:
            grad[:n-kk] -= h[kk:] * dx
            grad[kk:] -= h[:n-kk] * dx
    return grad

def project_to_box_sum(v, target, lo=0., hi=1.):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):

        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.full(n, dx)
    if k >= 0:
        if k < n:
            grad[:n - k] -= h[k:] * dx
            grad[k:] -= h[:n - k] * dx
    else:
        kk = -k
        if kk < n:
            grad[:n - kk] -= h[kk:] * dx
            grad[kk:] -= h[:n - kk] * dx
    return grad

def worst_shifts_and_grad(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    max_idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
    grads = np.zeros_like(h)
    for idx in max_idxs:
        k = idx - (n - 1)
        grads += gradient_for_shift(h, k)
    grads /= max(1, len(max_idxs))
    return grads, max_val

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Subgradient of overlap for a given shift k (includes dx factor)."""
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.full(n, dx)
    if k >= 0:
        if k < n:
            grad[:n - k] -= h[k:] * dx
            grad[k:] -= h[:n - k] * dx
    else:
        kk = -k
        if kk < n:
            grad[:n - kk] -= h[kk:] * dx
            grad[kk:] -= h[:n - kk] * dx
    return grad

def worst_shifts_and_grad(h):
    """Average gradient over all shifts attaining the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # tolerate tiny numerical noise
    active = np.where(np.abs(corr - max_val) < 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in active:
        k = idx - (n - 1)
        grad += gradient_for_shift(h, k)
    grad /= max(1, len(active))
    return grad, max_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target = n / 2.0
    h0 = project_to_box_sum(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    init_step = 0.5
    min_step = 1e-6
    max_step = 2.0

    # outer restart loop
    while time.time() - start < budget_s - 0.5:
        # start from a perturbed version of the current best
        cur_h = best_h + 0.1 * (rng.random(n) - 0.5)
        cur_h = project_to_box_sum(cur_h, target)
        cur_c5 = compute_c5(cur_h)

        step = init_step
        no_improve = 0

        # inner projected subgradient descent
        while time.time() - start < budget_s - 0.2:
            grad, _ = worst_shifts_and_grad(cur_h)
            cand_h = project_to_box_sum(cur_h - step * grad, target)
            cand_c5 = compute_c5(cand_h)

            if cand_c5 + 1e-12 < cur_c5:
                cur_h, cur_c5 = cand_h, cand_c5
                no_improve = 0
                step = min(step * 1.05, max_step)
                if cur_c5 + 1e-12 < best_c5:
                    best_h, best_c5 = cur_h.copy(), cur_c5
                    if best_c5 < 0.3805:
                        break
                continue

            # occasional random perturbation when gradient step fails
            if rng.random() < 0.05:
                d = rng.normal(size=n)
                d -= d.mean()
                cand_h2 = project_to_box_sum(cur_h + step * d, target)
                cand_c5_2 = compute_c5(cand_h2)
                if cand_c5_2 + 1e-12 < cur_c5:
                    cur_h, cur_c5 = cand_h2, cand_c5_2
                    step = max(step * 0.9, min_step)
                    continue

            # reduce step size after failure
            step = max(step * 0.7, min_step)
            no_improve += 1
            if step <= min_step or no_improve > 3000:
                break

        if best_c5 < 0.3805:
            break

        # occasional full random restart to diversify search
        if rng.random() < 0.1:
            rand_h = rng.random(n)
            rand_h = project_to_box_sum(rand_h, target)
            rand_c5 = compute_c5(rand_h)
            if rand_c5 < best_c5:
                best_h, best_c5 = rand_h, rand_c5

    # final safe projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
