def project_to_bounds_and_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # final fix for rounding errors
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly among non-bound entries
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
        else:
            proj += diff / len(proj)
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def simple_anneal(h_start, cur_val, start_time, deadline, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    while time.time() < deadline - 0.05:
        # find worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute contributions to that shift
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # choose destination j with capacity
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # metropolis with temperature scaling
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        if best_val < 0.3808:
            break
    return best_h, best_val
```

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral C5 for discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _project_to_bounds_and_sum(v, target):
    """Project v onto [0,1]^n with sum(v)=target (water-filling)."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
        else:
            proj += diff / proj.size
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def random_start_vector(n, rng):
    """Uniform random start respecting sum=h_target."""
    v = rng.random(n)
    return _project_to_bounds_and_sum(v, n / 2.0)

def gradient_descent_fixed(h_init, target, dx, max_iter, rng):
    """Projected sub‑gradient descent on max_k overlap."""
    h = h_init.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 5.0
    decay = 0.9995
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        O = corr * dx
        idx = np.argmax(O)
        val = O[idx]
        if val < best_val:
            best_val = val
            best_h = h.copy()
        k = idx - (n - 1)
        grad = np.zeros_like(h)
        if k >= 0:
            i = np.arange(n - k)
            grad[i] += (1.0 - h[i + k]) * dx
            grad[i + k] -= h[i] * dx
        else:
            kk = -k
            i = np.arange(kk, n)
            grad[i] += (1.0 - h[i - kk]) * dx
            grad[i - kk] -= h[i] * dx
        h = h - step * grad
        h = _project_to_bounds_and_sum(h, target)
        step *= decay
        if it % 2000 == 0 and it > 0:
            noise = rng.normal(scale=0.01, size=n)
            h = _project_to_bounds_and_sum(h + noise, target)
    return best_h, best_val

def simple_anneal(h_start, cur_val, start_time, deadline, rng):
    """Lightweight simulated annealing on h."""
    h = h_start.copy()
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    while time.time() < deadline - 0.05:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        if best_val < 0.3808:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
    target_sum = n / 2.0
    h0 = _project_to_bounds_and_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    dx = 2.0 / n

    # primary gradient descent on the initial candidate
    h_opt, val_opt = gradient_descent_fixed(best_h, target_sum, dx, max_iter=20000, rng=rng)
    if val_opt < best_val:
        best_val, best_h = val_opt, h_opt
        if best_val <= 0.3808:
            return best_h.astype(float), best_val, n

    # multiple short gradient runs with random starts
    while time.time() < deadline - 5.0:
        h_start = random_start_vector(n, rng)
        h_opt, val_opt = gradient_descent_fixed(h_start, target_sum, dx, max_iter=5000, rng=rng)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt
            if best_val <= 0.3808:
                break

    # final annealing refinement if time permits
    if time.time() < deadline - 0.2:
        best_h, best_val = simple_anneal(best_h, best_val, start, deadline, rng)

    # final projection and evaluation
    best_h = _project_to_bounds_and_sum(best_h, target_sum)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
