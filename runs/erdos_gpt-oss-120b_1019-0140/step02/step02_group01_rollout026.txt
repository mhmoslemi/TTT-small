import time
import numpy as np
from numpy.lib.stride_tricks import sliding_window_view

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k0 = n - 1
    max_val = np.max(np.concatenate((corr[:k0], corr[k0 + 1 :])))
    return max_val * dx

def project_onto_box_simplex(v, M):
    lo, hi = -1.0, 2.0
    def s(lam):
        return np.sum(np.clip(v - lam, 0.0, 1.0))
    while s(lo) < M:
        lo -= 1.0
    while s(hi) > M:
        hi += 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if s(mid) > M:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def compute_gradient(h, alpha):
    n = h.shape[0]
    g = 1.0 - h
    O = np.correlate(h, g, mode='full')
    O[n - 1] = -1e9  # ignore zero shift
    O_max = O.max()
    exp_arg = alpha * (O - O_max)
    p = np.exp(exp_arg)
    p_norm = p / p.sum()
    windows = sliding_window_view(p_norm, n)      # shape (n, n)
    grad = windows[::-1] @ g
    return grad

def random_feasible(n, M, rng):
    method = rng.integers(3)
    if method == 0:  # binary
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:  # continuous random
        h = rng.random(n)
        h = h / h.sum() * M
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_simplex(h, M)
    else:  # decreasing linear shape
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_simplex(h, M)
    return h

def optimise_n(n, time_budget, rng):
    M = n // 2
    best_h = None
    best_c5 = np.inf
    start = time.time()
    start_list = []
    if 'initial_h_values' in globals():
        if len(initial_h_values) == n:
            start_list.append(np.array(initial_h_values, dtype=float))
    while True:
        if time.time() - start > time_budget - 0.2:
            break
        if start_list:
            h0 = start_list.pop(0)
        else:
            h0 = random_feasible(n, M, rng)
        c5 = compute_c5(h0)
        if c5 < best_c5:
            best_c5, best_h = c5, h0.copy()
            if best_c5 < 0.3808:
                return best_h, best_c5
        # inner gradient descent
        inner_budget = min(5.0, time_budget - (time.time() - start) - 0.5)
        inner_start = time.time()
        h = h0.copy()
        eta0 = 0.5
        alpha = 5.0
        it = 0
        while time.time() - inner_start < inner_budget:
            it += 1
            grad = compute_gradient(h, alpha)
            eta = eta0 / np.sqrt(it)
            h_new = h - eta * grad
            h_new = project_onto_box_simplex(h_new, M)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new.copy()
                if best_c5 < 0.3808:
                    return best_h, best_c5
            h = h_new
            if it % 50 == 0:
                alpha = min(alpha * 1.1, 100.0)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None
    cand_ns = [80, 96, 112, 128, 144]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in cand_ns:

            else:
                cand_ns = [n0] + [n for n in cand_ns if n != n0]
    for n in cand_ns:
        if time.time() - start_time > budget_s - 5.0:
            break
        remaining = budget_s - (time.time() - start_time)
        alloc = min(200.0, remaining * 0.7)
        h_opt, c5_opt = optimise_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global, best_c5_global, best_n_global = h_opt, c5_opt, n
            if best_c5_global < 0.3808:
                break
    # final local refinement
    if best_h_global is not None:
        n = best_n_global
        M = n // 2
        while time.time() - start_time < budget_s - 0.5:
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(1.0 - best_h_global[i], best_h_global[j])
            if max_transfer <= 0:

            delta = rng.random() * max_transfer
            h_new = best_h_global.copy()
            h_new[i] += delta
            h_new[j] -= delta
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < best_c5_global:
                best_h_global, best_c5_global = h_new, c5_new
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        best_h_global = project_onto_box_simplex(best_h_global, n // 2)
        best_c5_global = compute_c5(best_h_global)
    else:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    return best_h_global, best_c5_global, best_n_global
