def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    # Determine n and initialize h0

        n = len(h0)
    else:
        n = 80
        # Use random binary half ones
        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, n//2, replace=False)] = 1.0

    # Ensure feasibility
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = np.sum(h0)
    if abs(cur_sum - target_sum) > 1e-12:
        # simple fix
        diff = target_sum - cur_sum
        h0 += diff / n
        h0 = np.clip(h0, 0.0, 1.0)
        # additional fix if still off
        residual = target_sum - np.sum(h0)
        if abs(residual) > 1e-12:
            # adjust by adding to some indices
            idxs = rng.permutation(n)
            for i in idxs:
                if abs(residual) < 1e-12:
                    break
                # compute possible adjustment
                adj = np.clip(residual, -h0[i], 1 - h0[i])
                h0[i] += adj
                residual = target_sum - np.sum(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Generate seeds list
    seeds = []

    # Random binary half ones

import time
import math
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(arr):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Generate Thue‑Morse binary sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    """Generate Rudin‑Shapiro binary sequence (0/1) of length n."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)
            nxt.append(v * (1 if (k % 2 == 0) else -1))
        a = nxt
    a = np.array(a[:n], dtype=int)
    return (a + 1) // 2

def _make_feasible(v, target_sum):
    """Clip to [0,1] and adjust to meet exact sum target_sum."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target_sum) > 1e-12:
        diff = target_sum - cur
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
        residual = target_sum - np.sum(v)
        # linear sweep to fix any leftover tiny error
        for i in range(v.size):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -v[i], 1.0 - v[i])
            v[i] += adj
            residual = target_sum - np.sum(v)
    return v

def continuous_anneal(start_h, end_time, rng):
    """Simulated annealing moving a delta between two indices (sum preserved)."""
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    # ensure exact sum before starting
    h = _make_feasible(h, target_sum)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t0 = time.time()
    T_start = 0.02
    T_final = 1e-6
    while time.time() < end_time:
        # exponential schedule based on elapsed fraction
        frac = (time.time() - t0) / max(end_time - t0, 1e-12)
        T = T_start * (T_final / T_start) ** frac
        # choose indices that can be increased / decreased
        inc_idx = np.where(h < 1.0 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        if dE <= 0.0 or rng.random() < math.exp(-dE / (T + 1e-30)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        # early exit if we already beat the target
        if best_c5 <= 0.3808:
            break
    return best_h, best_c5

def binary_hill_climb(start_bin, end_time, rng):
    """Greedy hill climbing on binary vectors (swap moves)."""
    h = start_bin.astype(int).copy()
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    while time.time() < end_time and no_improve < 4000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    target_sum = n / 2.0
    h0 = _make_feasible(h0, target_sum)

    # Build a few diverse seeds
    seeds = [h0]
    # random binary half‑ones
    bin_rand = np.zeros(n, dtype=float)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    seeds.append(_make_feasible(bin_rand, target_sum))
    # Thue‑Morse
    tm = thue_morse_seq(n).astype(float)
    seeds.append(_make_feasible(tm, target_sum))
    # Rudin‑Shapiro
    rs = rudin_shapiro_seq(n).astype(float)
    seeds.append(_make_feasible(rs, target_sum))
    # a couple of random continuous vectors
    for _ in range(2):
        cont = rng.random(n)
        cont = cont / np.sum(cont) * target_sum
        seeds.append(_make_feasible(cont, target_sum))

    # Pick the best seed by evaluating C5 quickly
    best_h = None
    best_c5 = float('inf')
    for s in seeds:
        c5 = compute_c5(s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = s.copy()

    # Allocate most of the remaining budget to a long annealing run
    remaining = budget_s - (time.time() - start_time)
    # reserve ~5 s for final refinement
    anneal_time = max(0.0, remaining - 5.0)
    if anneal_time > 0.1:
        cand_h, cand_c5 = continuous_anneal(best_h, time.time() + anneal_time, rng)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h, cand_c5

    # Optional: a short differential‑evolution burst if time permits
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 8.0:
        de_bounds = [(0.0, 1.0)] * n
        # simple penalty for sum violation
        penalty = 100.0
        def de_obj(x):
            x = np.clip(x, 0.0, 1.0)
            s = np.sum(x)
            pen = penalty * abs(s - target_sum)
            return compute_c5(x) + pen
        de_maxiter = max(1, int(remaining // 0.05))  # rough scaling
        de_res = differential_evolution(
            de_obj,
            de_bounds,
            seed=rng,
            maxiter=de_maxiter,
            popsize=5,
            polish=False,
            updating='deferred',
            workers=1,
        )
        x = _make_feasible(de_res.x, target_sum)
        de_c5 = compute_c5(x)
        if de_c5 < best_c5:
            best_h, best_c5 = x, de_c5

    # Final binary hill‑climbing refinement
    if time.time() - start_time < budget_s - 2.0:
        bin_start = (best_h >= 0.5).astype(int)
        final_h, final_c5 = binary_hill_climb(
            bin_start,
            time.time() + (budget_s - (time.time() - start_time)) - 0.5,
            rng,
        )
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5

    # Final safety clip and exact sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _make_feasible(best_h, target_sum)

    return best_h, best_c5, n
