def gradient_optimize(h0, time_budget, rng):
    h = h0.copy()
    n = len(h)
    target = n/2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 5.0
    max_beta = 80.0
    alpha0 = 0.5
    no_improve = 0
    while time.time() - start < time_budget:
        # compute correlation raw vector
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase beta gradually
        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)  # faster increase

        # compute softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w /= w_sum

        # gradient
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                kk = -k
                tail = slice(kk, n)
                head = slice(0, n - kk)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # enforce zero-sum direction
        g -= g.mean()

        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.01, 1.0 - elapsed / time_budget)

        h_new = h - alpha * g
        # project to bounds and sum
        h_new = project_sum_and_bounds(h_new, target)

        # compute new c5
        new_c5 = compute_c5(h_new)
        # accept if improves or with small probability
        if new_c5 < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 2000:
            # random shake
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    # final safety projection
    h = np.clip(h, 0.0, 1.0)
    h = project_sum_and_bounds(h, target)
    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_h = h.copy()
        best_c5 = final_c5
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap value multiplied by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def gradient_optimize(h0, time_budget, rng):
    """Smooth‑max gradient descent."""
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 5.0
    max_beta = 80.0
    alpha0 = 0.5
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # increase smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w /= w_sum
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                kk = -k
                tail = slice(kk, n)
                head = slice(0, n - kk)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        g -= g.mean()
        elapsed = time.time() - start
        alpha = alpha0 * max(0.01, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5 or rng.random() < 0.001:
            h = h_new
        if no_improve > 2000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
    # final safety projection
    h = np.clip(h, 0.0, 1.0)
    h = project_sum_and_bounds(h, target)
    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_h = h.copy()
        best_c5 = final_c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
