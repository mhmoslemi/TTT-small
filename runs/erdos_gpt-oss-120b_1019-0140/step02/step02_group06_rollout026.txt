def optimize_h(h0, total_time, rng):
    Initialize:
        h = h0.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        n = len(h)
        target = n/2.0
        start = time.time()
        # parameters
        beta = 1.0
        beta_max = 200.0
        beta_mult = 1.02
        alpha0 = 0.1
        # momentum
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 0
        no_improve_iter = 0
        # later use restarts:
        while time.time() - start < total_time:
            # compute correlation
            v_corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = v_corr.max() * (2.0 / n)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve_iter = 0
            else:
                no_improve_iter += 1

            # update beta
            if beta < beta_max:
                beta = min(beta_max, beta * beta_mult)

            # compute softmax weights
            vmax = v_corr.max()
            # To avoid overflow: subtract vmax
            w = np.exp(beta * (v_corr - vmax))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum

            # compute gradient g
            g = np.zeros_like(h)
            offset = n - 1
            for idx, weight in enumerate(w):
                if weight == 0:
                    continue
                shift = idx - offset
                if shift >= 0:
                    tail = slice(0, n - shift)
                    head = slice(shift, n)
                    # tail indices i: i + shift = i + shift -> head index
                    # g[tail] += weight * (1 - h[head])
                    # g[head] -= weight * h[tail]
                    g[tail] += weight * (1.0 - h[head])
                    g[head] -= weight * h[tail]
                else:
                    s2 = -shift
                    tail = slice(s2, n)
                    head = slice(0, n - s2)
                    g[tail] += weight * (1.0 - h[head])
                    g[head] -= weight * h[tail]

            # ensure gradient direction sums to zero (preserves total mass)
            g -= np.mean(g)

            # Adam update
            t += 1
            m = 0.9 * m + (1 - 0.9) * g
            v = 0.999 * v + (1 - 0.999) * (g * g)
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            alpha = alpha0 * (1.0 / (np.sqrt(v_hat) + 1e-8))
            # step: we can use elementwise: h_new = h - alpha * m_hat
            step = alpha * m_hat
            h_new = h - step
            # project
            h_new = project_sum_and_bounds_bisection(h_new, target)
            # accept if better (or random chance)
            c5_new = compute_c5(h_new)
            if c5_new < c5 or rng.random() < 0.001:
                h = h_new
                # reset momentum?
                # Possibly keep momentum, as new h may have.
            # random restart if stuck
            if no_improve_iter > 2000:
                # random perturbation around best and reinitialize m,v
                perturb = rng.normal(scale=0.03, size=n)
                h = project_sum_and_bounds_bisection(best_h + perturb, target)
                m.fill(0); v.fill(0); t=0
                no_improve_iter = 0

            # optionally, after some time, commit to new restart from best with random perturb

                # Do a fresh random start

def project_sum_and_bounds_bisection(x, target, tol=1e-9, max_iter=100):
    # Clip to [0,1] initial for readability may help.
    # We search lambda such that sum(clamp(x + lambda,0,1)) = target.
    low = -np.max(x) - 1.0  # ensure all zero
    high = 1.0 - np.min(x) + 1.0  # ensure all one
    for _ in range(max_iter):
        mid = (low+high)/2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s < target:
            low = mid
        else:
            high = mid
    # return final
    return np.clip(x + (low+high)/2.0, 0.0, 1.0)

def project_sum_and_bounds(x, target, tol=1e-12, max_iter=60):
    # Using bisection on lambda to enforce sum
    # Clip to [0,1] after adding lambda, to avoid saturating.
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s < target:
            low = mid
        else:
            high = mid
    # final
    y = np.clip(x + (low+high)*0.5, 0.0, 1.0)
    # after final clip, adjust small diff by adding to first non-saturated element
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # distribute diff linearly among free elements
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
        else:
            # all at bounds, just adjust first
            y[0] = np.clip(y[0] + diff, 0.0, 1.0)
    return y

def optimize_h(h0, time_budget, rng, max_iters=None):
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    beta_max = 150.0
    beta_mult = 1.001
    alpha0 = 0.2 # initial step
    m = np.zeros_like(h)  # momentum
    momentum = 0.6
    no_improve = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase smoothing
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mult)

        # compute softmax weights
        vmax = corr.max()
        # to avoid overflow
        w = np.exp(beta * (corr - vmax))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient of smooth objective
        g = np.zeros_like(h)
        offset = n - 1
        # Loop over shifts
        # we can speed up by using np.where w > small threshold
        # (maybe not required)
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            shift = idx - offset
            if shift >= 0:
                tail = slice(0, n - shift)
                head = slice(shift, n)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
            else:
                s2 = -shift
                tail = slice(s2, n)
                head = slice(0, n - s2)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]

        # ensure zero-mean to keep sum constraint
        g -= g.mean()

        # momentum update
        m = momentum * m + (1 - momentum) * g
        # step size decay
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        # update
        h_new = h - alpha * m
        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        # random shake if stuck
        if no_improve > 2500:
            # add small gaussian noise to best
            h = project_sum_and_bounds(best_h + rng.normal(scale=0.02, size=n), target)
            m.fill(0.0)
            no_improve = 0

    return best_h, best_c5

                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
            else:
                s2 = -shift
                tail = slice(s2, n)
                head = slice(0, n - s2)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]

        # keep sum unchanged (zero‑mean direction)
        g -= g.mean()

        # momentum update
        m = momentum_coeff * m + (1.0 - momentum_coeff) * g

        # step‑size schedule
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        # take step and project
        h_new = h - alpha * m
        h_new = project_sum_and_bounds(h_new, target)

        # accept if improved or by tiny probability
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        # random shake if stagnated
        if no_improve > 2500:
            h = project_sum_and_bounds(best_h + rng.normal(scale=0.02, size=n), target)
            m.fill(0.0)
            no_improve = 0

    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (continuous) for a discrete step function."""
    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode="full")
    return np.max(v) * dx

def project_sum_and_bounds(x, target, tol=1e-12, max_iter=10):
    """
    Clip to [0,1] and adjust to satisfy sum(x) == target.
    Uses a simple water‑filling scheme with a few iterations.
    """
    y = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        s = y.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            adj = diff / free.sum()
            y[free] += adj
            y = np.clip(y, 0.0, 1.0)
        else:
            # all variables at a bound – distribute uniformly and re‑clip
            y += diff / len(y)
            y = np.clip(y, 0.0, 1.0)
    return y

def optimize_h(h0, time_budget, rng):
    """Gradient descent with momentum on a soft‑max approximation of C₅."""
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start = time.time()
    beta = 2.0           # smoothing parameter for soft‑max
    beta_max = 500.0
    beta_mult = 1.001
    alpha0 = 0.2         # base step size
    momentum_coeff = 0.6
    m = np.zeros_like(h)
    no_improve = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = corr.max() * dx

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if beta < beta_max:
            beta = min(beta_max, beta * beta_mult)

        vmax = corr.max()
        w = np.exp(beta * (corr - vmax))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        g = np.zeros_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            shift = idx - offset
            if shift >= 0:
                tail = slice(0, n - shift)
                head = slice(shift, n)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]
            else:
                s2 = -shift
                tail = slice(s2, n)
                head = slice(0, n - s2)
                g[tail] += weight * (1.0 - h[head])
                g[head] -= weight * h[tail]

        g -= g.mean()                     # keep updates zero‑mean (preserve total mass)

        m = momentum_coeff * m + (1.0 - momentum_coeff) * g
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * m
        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 2500:
            h = project_sum_and_bounds(best_h + rng.normal(scale=0.02, size=n), target)
            m.fill(0.0)
            no_improve = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.time()
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
