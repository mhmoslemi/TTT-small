def max_shift_and_gradient(h, dx):
    n = len(h)
    # compute correlation of h with 1 - h
    corr = np.correlate(h, 1 - h, mode='full')
    # find index of max value
    idx_max = np.argmax(corr)
    k = idx_max - (n - 1)
    # compute neighbor contributions (neigh) as sum of h_i shifted by k:
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    # gradient = dx * (1 - neigh)
    grad = dx * (1.0 - neigh)
    return k, grad
```

def repair(h, target_sum, eps=1e-12):
    # Clip to bounds
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # Try to distribute diff
    for _ in range(10):
        if abs(diff) < eps:
            break
        free = (h > 0.0) & (h < 1.0)
        free_cnt = free.sum()
        if free_cnt > 0:
            inc = diff / free_cnt
            h[free] += inc
            # Clip again
            h = np.clip(h, 0.0, 1.0)
            cur = h.sum()
            diff = target_sum - cur
        else:
            # No free variables, distribute uniformly (may saturate)
            inc = diff / h.size
            h += inc
            h = np.clip(h, 0.0, 1.0)
            cur = h.sum()
            diff = target_sum - cur
    # Final fix with loop to ensure exactness
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > eps:
        # Adjust elementwise
        for i in range(h.size):
            if abs(diff) < eps:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    # Ensure final clipping
    h = np.clip(h, 0.0, 1.0)
    return h
```

def max_shift_and_gradient(h, dx):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = np.argmax(corr)
    k = idx_max - (n - 1)  # shift corresponding to max overlap
    # compute neighbor contributions
    neigh = np.zeros(n, dtype=h.dtype)
    if k >= 0:
        # positions where both h[i] and h[i+k] exist
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = dx * (1.0 - neigh)
    return k, grad
```

def gradient_descent_step(h, target_sum, dx, step_size):
    # compute gradient for worst shift
    _, grad = max_shift_and_gradient(h, dx)
    # center gradient to preserve sum (zero mean)
    grad_centered = grad - np.mean(grad)
    cand = h - step_size * grad_centered
    cand = repair(cand, target_sum)
    return cand

            break
        free = (h > 0.0) & (h < 1.0)
        cnt = int(free.sum())
        if cnt:
            inc = diff / cnt
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            cur = h.sum()
            diff = target_sum - cur
        else:
            inc = diff / h.size
            h += inc
            h = np.clip(h, 0.0, 1.0)
            cur = h.sum()
            diff = target_sum - cur
    # final elementwise correction if needed
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > eps:
        for i in range(h.size):
            if abs(diff) < eps:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)

def max_shift_and_gradient(h, dx):
    """Return shift achieving maximal overlap and its gradient."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)          # shift corresponding to max overlap
    neigh = np.zeros(n, dtype=h.dtype)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = dx * (1.0 - neigh)      # ∂f_k/∂h_i
    return k, grad

def gradient_descent_step(h, target_sum, dx, step_size):
    """One gradient‑descent move on the active shift."""
    _, grad = max_shift_and_gradient(h, dx)
    grad_centered = grad - np.mean(grad)   # keep zero‑sum direction
    cand = h - step_size * grad_centered
    return repair(cand, target_sum)

def smooth_and_fix(h, target_sum):
    """Gaussian‑type smoothing followed by repair."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=h.dtype)
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return repair(sm, target_sum)

def optimize_h_with_budget(h0, n, time_limit, rng):
    """Gradient‑descent + occasional perturbations within a time budget."""
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    step_size = 0.5
    step_decay = 0.9995
    min_step = 1e-5
    pert_prob = 0.01

    while True:
        if time.time() - start >= time_limit:
            break

        cand = gradient_descent_step(h, target_sum, dx, step_size)
        cand_val = compute_c5(cand)

        if cand_val < cur_val or rng.random() < 0.01:
            h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()

        if rng.random() < pert_prob:
            pert = h.copy()
            subset = rng.choice(n, size=max(1, n // 10), replace=False)
            noise = (rng.random(subset.size) - 0.5) * 0.2   # [-0.1,0.1]
            pert[subset] += noise
            pert = repair(pert, target_sum)
            pert_val = compute_c5(pert)
            if pert_val < best_val:
                best_val = pert_val
                best_h = pert.copy()
            if pert_val < cur_val:
                h = pert
                cur_val = pert_val

        step_size *= step_decay
        if step_size < min_step:
            step_size = 0.5

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # ----- base construction -------------------------------------------------
    if "initial_h_values" in globals():
        base_h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 80
        base_h0 = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        base_h0[idx] = 1.0
    n0 = base_h0.shape[0]
    target_sum0 = n0 / 2.0
    base_h0 = repair(base_h0, target_sum0)

    best_h_global = None
    best_val_global = float("inf")
    best_n_global = None

    # ----- try several up‑sampling factors -----------------------------------
    factors = [1, 2, 3]  # n = n0 * factor
    for i, factor in enumerate(factors):
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        n = n0 * factor
        target_sum = n / 2.0

        if factor == 1:
            h0 = base_h0.copy()
        else:
            h0 = np.repeat(base_h0, factor)
            h0 = repair(h0, target_sum)

        # time slice for this factor
        time_slice = remaining / (len(factors) - i)
        time_slice = min(time_slice, 300.0)  # cap per factor

        h_opt, val_opt = optimize_h_with_budget(h0, n, time_slice, rng)

        if val_opt < best_val_global:
            best_val_global = val_opt
            best_h_global = h_opt.copy()
            best_n_global = n

        if best_val_global < 0.3808:
            break

    # ----- optional final smoothing -----------------------------------------
    if best_h_global is not None:
        remaining = budget_s - (time.time() - start_total)
        if remaining > 5.0:
            n = best_n_global
            target_sum = n / 2.0
            h = best_h_global.copy()
            for _ in range(5):
                sm = smooth_and_fix(h, target_sum)
                sm_val = compute_c5(sm)
                if sm_val + 1e-12 < best_val_global:
                    best_val_global = sm_val
                    best_h_global = sm.copy()
                    h = sm
                else:
                    break

    # ----- safety final repair ---------------------------------------------
    if best_h_global is None:
        best_h_global = base_h0.copy()
        best_val_global = compute_c5(best_h_global)
        best_n_global = n0
    else:
        best_h_global = repair(best_h_global, best_n_global / 2.0)

    return best_h_global, best_val_global, best_n_global
