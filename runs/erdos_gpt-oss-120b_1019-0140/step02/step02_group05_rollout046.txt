def project_to_constraints(h, target_sum):
    # clip
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    diff = target_sum - total
    # while loop for diff adjustment
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # no free, distribute to all with capacity
            # adjust by moving some towards bounds
            # fallback: adjust proportionally
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
            continue
        inc = diff / free.size
        h[free] += inc
        # clip again
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    # final correction: distribute residual greedily
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        indices = np.arange(len(h))
        for i in indices:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h
```

def gradient_step(h, target_sum, max_step=0.5):
    # compute correlation for all lags
    corr = np.correlate(h, h, mode='full')
    idx_min = np.argmin(corr)
    shift = idx_min - (len(h) - 1)
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        g[:n-shift] += h[shift:]
        g[shift:] += h[:n-shift]
    else:
        k = -shift
        g[k:] += h[:n-k]
        g[:n-k] += h[k:]
    # direction to increase correlation
    d = g - np.mean(g)  # zero sum direction
    # limit step size
    max_norm = np.max(np.abs(d))
    if max_norm == 0:
        return h
    # compute max allowable alpha to keep in [0,1] after step
    # h_new = h + alpha * d
    # handle sign
    # compute alphas for each i
    with np.errstate(divide='ignore', invalid='ignore'):
        pos = d > 0
        neg = d < 0
        alpha_pos = np.where(pos, (1.0 - h) / d, np.inf)
        alpha_neg = np.where(neg, -h / d, np.inf)
        alpha_max = np.min(np.concatenate([alpha_pos[pos], alpha_neg[neg]]))
    # if alpha_max < 0 or inf, set to large
    if not np.isfinite(alpha_max):
        alpha_max = np.inf
    # choose step size
    alpha = min(alpha_max * 0.9, max_step)  # 0.9 factor for margin
    # if alpha_max too small, fallback to small constant
    if alpha <= 0:
        alpha = min(max_step, 0.01)
    h_new = h + alpha * d
    h_new = np.clip(h_new, 0.0, 1.0)
    # enforce sum
    h_new = project_to_constraints(h_new, target_sum)
    return h_new

import time
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraints(h, target_sum):
    """Clip h to [0,1] and enforce sum(h)=target_sum by simple water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    # distribute diff among interior (0<h<1) points
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # no free variables – spread uniformly (will be re‑clipped)
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
            continue
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    # final residual fix (greedy)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_step(h, target_sum, max_step=0.5):
    """Perform a projected gradient step that raises the worst autocorrelation."""
    n = len(h)
    # raw self‑correlation (no dx factor needed for direction)
    corr = np.correlate(h, h, mode="full")
    idx_min = int(np.argmin(corr))
    shift = idx_min - (n - 1)            # lag giving minimal correlation
    g = np.zeros_like(h)
    if shift >= 0:
        g[: n - shift] += h[shift:]
        g[shift:]     += h[: n - shift]
    else:
        k = -shift
        g[k:]   += h[: n - k]
        g[: n - k] += h[k:]
    # zero‑sum direction (preserve total mass)
    d = g - np.mean(g)
    if np.allclose(d, 0):
        return h
    # maximal allowed step to stay inside [0,1]
    pos = d > 0
    neg = d < 0
    alphas = []
    if np.any(pos):
        alphas.append(((1.0 - h)[pos] / d[pos]).min())
    if np.any(neg):
        alphas.append((-h[neg] / d[neg]).min())
    alpha_max = min(alphas) if alphas else np.inf
    # pick a safe step size
    alpha = min(alpha_max * 0.9, max_step) if np.isfinite(alpha_max) else max_step
    if alpha <= 0:
        alpha = max_step * 0.1
    h_new = h + alpha * d
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_constraints(h_new, target_sum)
    return h_new

def smooth_and_fix(h, target_sum):
    """A light Gaussian smoothing pass followed by projection."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return project_to_constraints(sm, target_sum)

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    # -------- initialise grid ----------
