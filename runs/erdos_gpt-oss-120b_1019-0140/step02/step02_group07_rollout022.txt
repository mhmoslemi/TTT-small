import time, math, numpy as np, cvxpy as cp

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def solve_lp(n, time_limit):
    # Build and solve LP, with time limit considered by solver? Not easily.
    # We'll just call solve.
    h = cp.Variable(n)
    t = cp.Variable((n, n))
    T = cp.Variable()
    # constraints
    constraints = []
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(cp.sum(h) == n/2)
    # McCormick
    H_col = cp.reshape(h, (n, 1))
    H_row = cp.reshape(h, (1, n))
    constraints.append(t >= 0)
    constraints.append(t <= H_col)
    constraints.append(t <= H_row)
    constraints.append(t >= H_col + H_row - 1)
    # shift constraints
    for k in range(1, n):
        i_idx = np.arange(n - k)
        j_idx = i_idx + k
        # This indexing will create expression vectors
        sub = h[i_idx] - t[i_idx, j_idx]
        constraints.append(cp.sum(sub) <= T)
    constraints.append(T >= 0)
    prob = cp.Problem(cp.Minimize(T), constraints)
    # Solve with ECOS within time limit: set max_iters
    # ECOS param "max_iters". We'll allocate large e.g., 500000.
    # Compute approximate time budget for solver: remaining = time_limit - (time.time() - start_time)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500000, abstol=1e-8, reltol=1e-8)
    except Exception:
        return None, None
    if h.value is None or T.value is None:
        return None, None
    h_val = np.array(h.value).flatten()
    T_val = float(T.value)
    return h_val, T_val

def continuous_anneal(start_h, time_end, rng):
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # schedule parameters
    init_temp = 1e-2  # maybe larger initial
    final_temp = 1e-7
    max_iter = 200000
    it = 0
    while time.time() < time_end and it < max_iter:
        # random inc/dec selection
        inc_candidates = np.where(h < 1 - 1e-12)[0]
        dec_candidates = np.where(h > 1e-12)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            break
        i = rng.choice(inc_candidates)
        j = rng.choice(dec_candidates)
        max_delta = min(1 - h[i], h[j])
        # temperature
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        # propose delta: scaled by temperature for exploration
        # sample uniform in [0, max_delta]
        delta = max_delta * rng.random()
        # acceptance
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-15)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # final correction for sum
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

def adjust_to_feasibility(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    diff = target - sum_h
    if abs(diff) < 1e-12:
        return h
    # Simple water-filling style
    # Repeat up to few passes
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase some entries below 1
            inc = np.where(h < 1 - 1e-12)[0]
            if inc.size == 0:
                break
            delta = diff / inc.size
            delta = min(delta, 1 - np.max(h[inc]))  # ensure not exceed 1
            # add delta to all inc entries within bounds
            h[inc] = np.minimum(h[inc] + delta, 1.0)
        else:
            # need to decrease some entries above 0
            dec = np.where(h > 1e-12)[0]
            if dec.size == 0:
                break
            delta = diff / dec.size  # diff negative
            # subtract absolute
            h[dec] = np.maximum(h[dec] + delta, 0.0)  # delta negative
        diff = target - np.sum(h)
    # final small adjustment
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # adjust single element
        for i in range(n):
            adj = np.clip(residual, -h[i], 1 - h[i])
            h[i] += adj
            residual = target - np.sum(h)
            if abs(residual) < 1e-12:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    # Determine n
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:

    # Compute time budget for solving LP
    time_elapsed = time.time() - start_time
    remaining = budget_s - time_elapsed - 0.5  # keep margin
    # Try LP

        h_feas = adjust_to_feasibility(h_lp)
        c5_lp = compute_c5(h_feas)
        best_h = h_feas
        best_c5 = c5_lp
        # If good enough, return early
        if best_c5 <= 0.3808:
            return best_h, best_c5, n
    else:
        # fallback initial h binary random
        # use global initial if exists else random half ones
        if 'initial_h_values' in globals():
            h0 = np.array(initial_h_values, dtype=float)
        else:
            h0 = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h0[idx] = 1.0
        h0 = adjust_to_feasibility(h0)
        best_h = h0
        best_c5 = compute_c5(best_h)
    # Continue with local refinement using annealing
    while time.time() - start_time < budget_s - 0.3:
        # allocate for a short anneal segment (e.g., 5 seconds)
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 0.5:
            break
        end_time = time.time() + min(5.0, remaining * 0.5)

        if best_c5 <= 0.3808:
            break
    # final feasibility ensure
    best_h = adjust_to_feasibility(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

import time
import math
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated by discrete correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_to_feasibility(h):
    """Clip to [0,1] and enforce sum(h)=len(h)/2 with a simple water‑filling."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            inc = np.where(h < 1.0 - 1e-12)[0]
            if inc.size == 0:
                break
            delta = diff / inc.size
            delta = np.minimum(delta, 1.0 - h[inc])
            h[inc] += delta
        else:
            dec = np.where(h > 1e-12)[0]
            if dec.size == 0:
                break
            delta = diff / dec.size
            delta = np.maximum(delta, -h[dec])
            h[dec] += delta
    # final tiny fix on a single entry if needed
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        for i in range(n):
            adj = np.clip(residual, -h[i], 1.0 - h[i])
            h[i] += adj
            residual = target - np.sum(h)
            if abs(residual) < 1e-12:
                break
    return h

def solve_lp(n, time_limit):
    """Convex relaxation: minimize max_k sum_i h_i(1‑h_{i+k}) with McCormick bounds."""
    h = cp.Variable(n)
    t = cp.Variable((n, n))          # surrogate for product h_i*h_j
    T = cp.Variable()                # upper bound on all shifts

    # basic box and sum constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2.0,
        T >= 0
    ]

    # McCormick envelope for product
    Hcol = cp.reshape(h, (n, 1))
    Hrow = cp.reshape(h, (1, n))
    constraints += [
        t >= 0,
        t <= Hcol,
        t <= Hrow,
        t >= Hcol + Hrow - 1
    ]

    # shift constraints: for each k>0,
    #   sum_{i=0}^{n-1-k} (h_i - t_{i,i+k}) <= T
    for k in range(1, n):
        i_idx = np.arange(n - k, dtype=int)
        j_idx = i_idx + k
        sub = h[i_idx] - t[i_idx, j_idx]
        constraints.append(cp.sum(sub) <= T)

    prob = cp.Problem(cp.Minimize(T), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500000, abstol=1e-8, reltol=1e-8)
    except Exception:
        return None, None

    if h.value is None or T.value is None:
        return None, None
    h_val = np.array(h.value).flatten()
    T_val = float(T.value)
    return h_val, T_val

def continuous_anneal(start_h, time_end, rng):
    """Simple simulated annealing preserving the total integral."""
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    init_temp = 1e-2
    final_temp = 1e-7
    max_iter = 300000
    it = 0
    while time.time() < time_end and it < max_iter:
        inc = np.where(h < 1.0 - 1e-12)[0]
        dec = np.where(h > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        # temperature schedule (linear)
        Ttemp = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        if dE <= 0 or rng.random() < math.exp(-dE / (Ttemp + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1

    # final sum correction
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # choose problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80

    # 1️⃣ Convex relaxation (LP) – tries to get a good fractional seed
    remaining = budget_s - (time.time() - start_time) - 0.5

        h_lp = adjust_to_feasibility(h_lp)
        best_h = h_lp
        best_c5 = compute_c5(best_h)
    else:
        # fall back to a simple binary start if LP failed
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values, dtype=float)
        else:
            best_h = np.zeros(n, dtype=float)
            best_h[rng.choice(n, n // 2, replace=False)] = 1.0
        best_h = adjust_to_feasibility(best_h)
        best_c5 = compute_c5(best_h)

    target_c5 = 0.3808
    # 2️⃣ Local refinement with simulated annealing, multiple short runs
    while time.time() - start_time < budget_s - 0.3:
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 0.5:

        else:
            start_vec = best_h.copy()

        if best_c5 <= target_c5:
            break

    best_h = adjust_to_feasibility(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
