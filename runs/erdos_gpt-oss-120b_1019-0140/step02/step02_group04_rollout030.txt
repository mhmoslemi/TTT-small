def compute_c5_fft(h):
    n = len(h)
    g = 1.0 - h
    L = 2*n - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g, n=L)

def compute_c5_and_info(h, dx):
    n = h.shape[0]
    g = 1.0 - h
    L = 2*n - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    k_star = max_idx - (n - 1)
    return max_val * dx, k_star
