def pair_update(h, k_star):
    n = len(h)
    if k_star == 0:
        return h
    # compute slices:
    if k_star > 0:
        tail = slice(0, n - k_star)
        head = slice(k_star, n)
    else:
        k2 = -k_star
        tail = slice(k2, n)
        head = slice(0, n - k2)
    h_tail = h[tail]
    h_head = h[head]
    diff = h_head - h_tail
    # Find index with largest positive diff (head > tail)
    max_idx = np.argmax(diff)  # returns 0 if all diff <=0
    max_diff = diff[max_idx]
    if max_diff <= 1e-8:
        # Nothing to improve; return unchanged
        return h
    # Determine amount to transfer
    epsilon = min(0.05, max_diff/2.0)  # at most half of diff
    # Ensure we stay within [0,1]
    # head value at that index
    head_val = h_head[max_idx]
    tail_val = h_tail[max_idx]
    epsilon = min(epsilon, head_val)  # can't reduce head below 0
    epsilon = min(epsilon, 1.0 - tail_val)  # can't increase tail above 1
    if epsilon <= 0:
        return h
    # Apply transfer
    # Indices in actual h
    if k_star > 0:
        i_tail = max_idx
        i_head = max_idx + k_star
    else:
        i_tail = max_idx + (-k_star)
        i_head = max_idx
    h[i_tail] += epsilon
    h[i_head] -= epsilon
    return h

def project_onto_sum_and_bounds(x, target, max_iter=100):
    # Clip after shift
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            low = mid
        else:
            high = mid
        if high - low < 1e-12:
            break
    y = np.clip(x + (low + high) / 2.0, 0.0, 1.0)
    # Ensure exact sum via small correction if needed
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # Distribute diff to free variables not at bounds
        free = (y > 0.0 + 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            # Clip again
            y = np.clip(y, 0.0, 1.0)
        else:
            # fallback: adjust any index
            i = np.argmin(np.abs(y - 0.5))
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral (with dx factor)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_onto_sum_and_bounds(x: np.ndarray, target: float, max_iter: int = 60) -> np.ndarray:
    """
    Project x onto the box [0,1]^n while enforcing sum(x) == target.
    Uses a simple bisection on the shift lambda such that sum(clip(x+lambda)) == target.
    """
    # Ensure we start from a copy
    x = np.asarray(x, dtype=float).copy()
    # Bisection bounds for lambda
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            low = mid
        else:
            high = mid
        if high - low < 1e-13:
            break
    y = np.clip(x + (low + high) * 0.5, 0.0, 1.0)
    # Tiny correction if needed
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(y - 0.5))
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_onto_sum_and_bounds(h0, target_sum)

    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)

    # ----- phase 1: smooth gradient descent (soft‑max) ---------------
    grad_budget = budget_s * 0.55
    beta = 2.0
    max_beta = 20.0
    base_lr = 0.2
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    while time.time() - start < grad_budget:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        cur_c5 = corr.max() * (2.0 / n)
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h_cur.copy()

        # increase smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        # soft‑max weights
        vmax = corr.max()
        w = np.exp(beta * (corr - vmax))
        ws = w.sum()
        if ws == 0.0:
            break
        w /= ws

        # gradient of the smooth objective
        g = np.zeros(n)
        offset = n - 1
        for idx, wi in enumerate(w):
            if wi < 1e-12:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += wi * (1.0 - h_cur[head])
            g[head] -= wi * h_cur[tail]

        # zero‑sum direction (preserve total mass)
        g -= np.mean(g)

        # Adam step
        it += 1
        lr = base_lr * max(0.1, 1.0 - (time.time() - start) / budget_s)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        h_new = h_cur - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_sum_and_bounds(h_new, target_sum)

        new_c5 = compute_c5(h_new)
        if new_c5 < cur_c5 or rng.random() < 0.001:
            h_cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()

        # occasional shake to escape plateaus
        if it % 2000 == 0:

            h_cur = project_onto_sum_and_bounds(h_cur + shake, target_sum)

    # ----- phase 2: targeted pairwise improvements -------------------
    while time.time() - start < budget_s:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        cur_c5 = corr.max() * (2.0 / n)

        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h_cur.copy()

        # shift attaining the maximum overlap
        k_star = np.argmax(corr) - (n - 1)
        improved = False

        if k_star != 0:
            if k_star > 0:
                tail_slice = slice(0, n - k_star)
                head_slice = slice(k_star, n)
            else:
                k2 = -k_star
                tail_slice = slice(k2, n)
                head_slice = slice(0, n - k2)

            tail_vals = h_cur[tail_slice]
            head_vals = h_cur[head_slice]
            diff = head_vals - tail_vals
            max_diff = diff.max()
            if max_diff > 1e-8:
                idx = diff.argmax()
                eps_transfer = min(0.05, max_diff / 2.0)

                if k_star > 0:
                    i_tail = idx
                    i_head = idx + k_star
                else:
                    i_tail = idx + (-k_star)
                    i_head = idx

                eps_transfer = min(eps_transfer, h_cur[i_head])          # keep head ≥ 0
                eps_transfer = min(eps_transfer, 1.0 - h_cur[i_tail])    # keep tail ≤ 1

                if eps_transfer > 1e-12:
                    candidate = h_cur.copy()
                    candidate[i_tail] += eps_transfer
                    candidate[i_head] -= eps_transfer
                    cand_c5 = compute_c5(candidate)
                    if cand_c5 < cur_c5 - 1e-12:
                        h_cur = candidate
                        cur_c5 = cand_c5
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h_cur.copy()

        if not improved:
            # small random perturbation as fallback
            shake = rng.normal(scale=0.01, size=n)
            candidate = project_onto_sum_and_bounds(h_cur + shake, target_sum)
            cand_c5 = compute_c5(candidate)
            if cand_c5 < cur_c5:
                h_cur = candidate
                cur_c5 = cand_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_cur.copy()

    # ----- final safety projection ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
