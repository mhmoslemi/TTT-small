def enforce_sum_and_bounds(h, target_sum):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # compute sum diff
    diff = target_sum - np.sum(h)
    # if diff is small, return
    if abs(diff) < 1e-12:
        return h
    # distribute diff uniformly then clip again
    # loop at most few times
    for _ in range(10):
        # add uniform offset
        h += diff / len(h)
        # clip again
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return h

import time
import numpy as np

def compute_c5(arr):
    """Maximum overlap integral approximated by correlation."""
    n = len(arr)
    return np.max(np.correlate(arr, 1 - arr, mode='full')) * (2.0 / n)

def upsample_h(old_h, new_n):
    """Linear interpolation of old_h (defined on [0,2]) to new_n points."""
    old_n = len(old_h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)

def enforce_sum_and_bounds(h, target_sum):
    """Clip to [0,1] and enforce exact sum by uniform adjustments."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    # simple water-filling loop
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def gradient_step(h, beta, step):
    """One projected gradient descent step on the smooth-maximum of correlations."""
    n = len(h)
    # correlation (scaled)
    corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    max_corr = np.max(corr)
    # smooth weights (log-sum-exp trick)
    exp_vals = np.exp(beta * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)  # sum to 1
    # gradient accumulation
    grad = np.zeros_like(h)
    for shift in range(-n + 1, n):
        wk = w[shift + n - 1]
        if wk == 0.0:
            continue
        # periodic rolls (circular approximation)
        plus = np.roll(h, shift)
        minus = np.roll(h, -shift)
        grad += wk * (1.0 - plus - minus)
    grad = (2.0 / n) * grad
    # descent step
    h_new = h - step * grad
    h_new = enforce_sum_and_bounds(h_new, n / 2.0)
    return h_new, max_corr

def binary_hill_climb(start_bin, end_time, rng):
    """Swap-based hill climbing on binary vectors; keeps sum = n/2."""
    n = len(start_bin)
    h = start_bin.copy()
    best_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    no_improve = 0
    while time.time() < end_time and no_improve < 2000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5
