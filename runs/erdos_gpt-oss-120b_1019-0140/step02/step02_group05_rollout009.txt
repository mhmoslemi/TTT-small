def smooth_max_gradient(h, target_sum, alpha):
    # returns gradient for smoothed max objective.
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)  # shape (2n-1,)

def random_initial(rng, n, binary=True):
    if binary:
        h = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
    else:
        # random uniform, then adjust sum to target
        h = rng.random(n)
        # clip to [0,1] you already have
        cur_sum = np.sum(h)
        target_sum = n/2.0
        # use simple projection: shift to adjust sum, then clip and fix residual
        diff = target_sum - cur_sum
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = target_sum - cur
        # distribute over free entries
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        # residual loop
        diff = target_sum - np.sum(h)
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

_n_global = None  # will be set to problem size inside run


def compute_c5(h):
    """Maximum overlap C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_to_constraints(h, target_sum):
    """Clip to [0,1] and enforce exact sum by redistributing the residual."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)


def smooth_and_fix(h, target_sum):
    """Gaussian smoothing (kernel [0.25,0.5,0.25]) followed by projection."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    sm = project_to_constraints(sm, target_sum)
    return sm


def random_initial(rng, n, binary=True):
    """Generate a random feasible h (binary or continuous)."""
    target = n / 2.0
    if binary:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    else:
        h = rng.random(n)
        h = project_to_constraints(h, target)
    return h


def smooth_max_gradient(h, alpha):
    """Gradient of the smooth max (log-sum-exp) of the overlap values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)  # weights sum to 1

    grad = np.zeros_like(h)
    shift_range = range(-n + 1, n)
    for idx, k in enumerate(shift_range):
        wk = w[idx]
        if wk == 0:
            continue
        if k >= 0:
            if n - k > 0:
                grad[: n - k] += wk * dx * (1.0 - h[k:])
                grad[k:] -= wk * dx * h[: n - k]
        else:
            m = -k
            if n - m > 0:
                grad[m:] += wk * dx * (1.0 - h[: n - m])
                grad[: n - m] -= wk * dx * h[m:]
    return grad


def objective(x):
    """Objective for SLSQP: the auxiliary variable t."""
    return x[-1]


def eq_sum_constraint(x):
    """Equality: sum(h) = n/2."""
    n = _n_global
    return np.sum(x[:n]) - n / 2.0


def ineq_constraints(x):
    """Inequalities: t >= overlap for every non‑zero shift."""
    n = _n_global
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    diff = t - corr
    diff[n - 1] = 1e6  # ignore the zero shift
    return diff


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = random_initial(rng, n, binary=True)

    global _n_global
    _n_global = n
    target_sum = n / 2.0
    h0 = project_to_constraints(h0, target_sum)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # multi‑start projected gradient descent with smooth‑max surrogate
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.85:
            break

        h = random_initial(rng, n, binary=rng.random() < 0.5)
        h = project_to_constraints(h, target_sum)

        eta0 = 0.5
        max_iter = 2000

        for t in range(max_iter):

                break
            if t % 200 == 0 and t > 0:

            eta = eta0 / np.sqrt(t + 1)

                    if best_val < 0.3808:

            if sm_val + 1e-12 < best_val:

                best_h = sm
            else:

    remaining = budget_s - (time.time() - start)

        x0 = np.concatenate([best_h, [best_val]])

        try:

                if abs(cur - target_sum) > 1e-9:
                    h_opt += (target_sum - cur) / n
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = np.sum(best_h)
    if abs(cur - target_sum) > 1e-9:
        best_h += (target_sum - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)
