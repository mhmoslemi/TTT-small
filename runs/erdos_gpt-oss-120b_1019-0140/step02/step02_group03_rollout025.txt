### project_to_box_sum

```
def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # Binary search λ
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)
```

Works.

def gradient_max_overlap(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_index = int(np.argmax(corr))
    k = max_index - (n - 1)
    g = np.zeros_like(h)
    if k >= 0:
        # term1: for i in [0, n-k)
        g[:n - k] += 1.0 - h[k:]
        # term2: for i in [k, n)
        g[k:] -= h[:n - k]
    else:
        kk = -k
        # term1: for i in [kk, n)
        g[kk:] += 1.0 - h[:n - kk]
        # term2: for i in [0, n-kk)
        g[:n - kk] -= h[kk:]
    return k, g, corr[max_index]
```

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_max_overlap(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)
    g = np.zeros_like(h)
    if k >= 0:
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    else:
        kk = -k
        g[kk:] += 1.0 - h[:n - kk]
        g[:n - kk] -= h[kk:]
    return k, g, corr[max_idx]

def objective_t(x):
    return x[-1]

class SumConstraint:

        self.target = n / 2.0
