def compute_c5(h):

    # binary search for shift lambda such that sum(clip(v+lambda)) == target
    low = -2.0
    high = 2.0

        x = np.clip(v + mid, lo, hi)

            low = mid
        else:

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(v, target, lo=0.0, hi=1.0, iters=25):
    """Project v onto [lo,hi]^n while enforcing sum(v)=target via binary search."""
    low, high = -2.0, 2.0
    for _ in range(iters):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def max_corr_and_lag(h):
    """Return (max correlation value, lag index) for vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    val = corr[idx] * (2.0 / n)
    return val, lag

def gradient_for_lag(h, lag):
    """Gradient of the (unscaled) correlation at a given lag."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if lag >= 0:
        k = lag
        if k < n:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:] -= h[:n - k]
    else:
        k = -lag
        if k < n:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # --- initialise h --------------------------------------------------------
    if 'initial_h_values' in globals():

        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)

    target = n / 2.0
    h0 = project_to_sum_box(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # --- Phase 1: simulated annealing (global exploration) -------------------
    base_step = 0.2
    T = 0.01
    cooling = 0.9995

    while True:
        if time.time() - start > budget_s * 0.85:
            break

        # random unit direction with zero mean
        d = rng.normal(size=n)
        d -= d.mean()

        if norm > 0:

        step = base_step * (rng.random() ** 2)
        cand = cur_h + step * d
