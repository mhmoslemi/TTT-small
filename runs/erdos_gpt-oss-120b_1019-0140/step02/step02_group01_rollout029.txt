def simulated_annealing(h, budget, rng):
    n = len(h)
    M = n//2
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val
    start = time.time()
    # temperature schedule: start with T0 = some fraction of best_val maybe
    T0 = best_val * 0.1
    # number of iterations based on remaining time
    while time.time() - start < budget:
        # propose move: choose i and j distinct
        i, j = rng.choice(n, 2, replace=False)
        # propose delta: random fraction of max transferable
        max_transfer = min(1.0 - cur_h[i], cur_h[j])
        # also consider negative direction? We'll also allow moving from i to j; we can either move from j to i or i to j randomly.
        if max_transfer <= 0:

            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = cur_h.copy()
            new_h[j] += delta
            new_h[i] -= delta
        else:
            delta = rng.random() * max_transfer
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        # acceptance
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val)/temp):

        # update temperature gradually

def slsqp_refine(h0, M, rng, time_allow):
    n = len(h0)
    sum_con = lambda x: np.sum(x[:n]) - M
    # constraints for each shift as inequality: t >= overlap(k)*dx
    # compute overlaps using dot product.
    # We'll prepare list of functions
    dx = 2.0 / n
    constraints = [{'type': 'eq', 'fun': sum_con}]
    # Precompute indices for each k for speed? Not essential.
    for k in range(-n+1, n):
        if k == 0:
            continue
        def make_fun(k):
            def fun(x):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overl * dx
            return fun
        constraints.append({'type': 'ineq', 'fun': make_fun(k)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    # Run minimize with maxiter limited; use SLSQP may take many iterations but we can set maxiter based on time_allow.
    # We cannot know iterations vs time precisely; set options to moderate.
    try:
        res = minimize(fun=lambda x: x[-1],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 400, 'ftol': 1e-9})
    except Exception:
        return h0
    if not res.success:
        return h0
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = np.sum(h_opt)
    if abs(cur - M) > 1e-8:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt

def simulated_annealing(h_start, time_budget, rng):
    n = len(h_start)
    M = n // 2
    best_h = h_start.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    # temperature schedule: start T proportional to best_val*0.02 maybe.
    T0 = best_val * 0.02 + 1e-4
    # We'll compute total allowed iters based on time: approximate each iteration takes maybe ~0.001 sec
    # We'll just loop while time left.
    while time.time() - start < time_budget:
        # propose move
        i, j = rng.choice(n, 2, replace=False)
        # Random direction: move from j to i or i to j
        if rng.random() < 0.5:
            max_transfer = min(1.0 - cur_h[i], cur_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            max_transfer = min(1.0 - cur_h[j], cur_h[i])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = cur_h.copy()
            new_h[j] += delta
            new_h[i] -= delta
        new_val = compute_c5(new_h)
        # acceptance
        elapsed = time.time() - start
        # temperature decreasing linearly
        T = T0 * max(0.01, 1 - elapsed / time_budget)
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h
                best_val = new_val
                # optional early stop if good enough
                if best_val < 0.3808:
                    break
    return best_h, best_val

def compute_c5(h):
    """Maximum overlap integral for a discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def generate_initial_h(n, M, rng):
    """Produce a feasible h with sum = M (M = n/2 for even n)."""
    method = rng.integers(3)
    if method == 0:                       # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:                     # uniform random then project
        h = rng.random(n)
        h = h / np.sum(h) * M
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc

                h[i] -= dec

        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
