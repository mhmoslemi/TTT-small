def adam_optimize(h0, max_iters, learning_rate=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters+1):
        # compute active shifts
        cvals = np.correlate(h, 1-h, mode='full')
        max_c = np.max(cvals)
        idxes = np.where(cvals >= max_c - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)  # gradient of max c is derived
        # Update Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        # project
        h = project_onto_box_sum(h, h.size/2)
        # evaluate
        score = compute_c5(h)
        if score < best_score:
            best_score, best_h = score, h.copy()
        # early stop if time low
        if t % 100 == 0 and time_left() < 5:
            break
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    grad = np.zeros_like(h)
    for shift in shifts:
        grad += 1 - np.roll(h, shift) - np.roll(h, -shift)
    grad /= len(shifts) if len(shifts) > 0 else 1
    return grad

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # broaden interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Linear interpolation from a larger array to a smaller size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Return Thue‑Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n//2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of max‑correlation w.r.t. h for given active shifts."""
    grad = np.zeros_like(h)
    for shift in shifts:
        # grad_i = 1 - h_{i+shift} - h_{i-shift}
        grad += 1.0 - np.roll(h, -shift) - np.roll(h, shift)
    if shifts.size > 0:
        grad /= shifts.size
    return grad


def slsqp_optimize_candidate(h0, max_iter=300):
    """Refine a candidate using SLSQP (minimise max correlation)."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def adam_optimize(h0, max_iters, learning_rate, time_left):
    """Projected Adam optimiser for the max‑correlation objective."""
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_h = h.copy()
    best_score = compute_c5(h)

    n = h.size
    for t in range(1, max_iters + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = learning_rate * m_hat / (np.sqrt(v_hat) + eps)

        h = h - step
        h = project_onto_box_sum(h, n / 2.0)

        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()

        if time_left() < 0.5:
            break
    return best_h, best_score


def binary_hill_climb(h, best_score, rng, time_left):
    """Improve a binary solution by swapping a 1 with a 0."""
    n = h.size
    ks = np.arange(-(n - 1), n, dtype=int)          # shift indices
    cur_c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = cur_c.max()
    best_h = h.copy()
    while time_left() > 0.2:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = rng.choice(ones)
        b = rng.choice(zeros)

        a_plus = (ks + a) % n
        a_minus = (ks - a) % n
        b_plus = (ks + b) % n
        b_minus = (ks - b) % n

        dC = h[a_plus] + h[a_minus] - h[b_plus] - h[b_minus]
        new_c = cur_c + dC
        new_max = new_c.max()

        if new_max < cur_max - 1e-12:
            # accept swap
            h[a] = 0.0
            h[b] = 1.0
            cur_c = new_c
            cur_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_h = h.copy()
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    best_h = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Stage 1: build and improve small‑scale candidates (n=64)
    # ------------------------------------------------------------------
    n_small = 64
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        candidates.append(project_onto_box_sum(downsample(init_full, n_small),
                                              n_small / 2.0))

    candidates.append(thue_morse(n_small))
    candidates.append(random_binary_balanced(n_small, rng))

    for _ in range(3):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    for cand in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=300)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5, dtype=float),
                                      n_small / 2.0)
        best_score = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Stage 2: upscale to larger resolution (n=256)
    # ------------------------------------------------------------------
    n_large = 256
    h_up = upsample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large / 2.0)
    sc_up = compute_c5(h_up)
    if sc_up < best_score:
        best_h, best_score = h_up, sc_up

    # ------------------------------------------------------------------
    # Stage 3: continuous refinement with Adam (projected)
    # ------------------------------------------------------------------
    remaining = time_left()
    if remaining > 5.0:
        # generous iteration limit, but stop early if out of time
        max_iters = int(min(200000, remaining * 1000))
        h_ref, sc_ref = adam_optimize(best_h, max_iters,
                                       learning_rate=0.05,
                                       time_left=time_left)
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref

    # ------------------------------------------------------------------
    # Stage 4: binarise and perform combinatorial hill‑climbing
    # ------------------------------------------------------------------
    half = best_h.size // 2
    binary_h = np.zeros_like(best_h)
    idx_top = np.argpartition(best_h, -half)[-half:]
    binary_h[idx_top] = 1.0
    sc_bin = compute_c5(binary_h)
    if sc_bin < best_score:
        best_h, best_score = binary_h, sc_bin

    if time_left() > 2.0:
        best_h, best_score = binary_hill_climb(best_h, best_score,
                                               rng, time_left)

    # final projection (safety) and score recomputation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
