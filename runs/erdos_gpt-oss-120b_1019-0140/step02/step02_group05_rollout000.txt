def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    tol = 1e-12
    max_iter = 1000
    it = 0
    while abs(diff) > tol and it < max_iter:
        free = np.where(((diff > 0) & (h < 1.0)) | ((diff < 0) & (h > 0.0)))[0]
        if free.size == 0:
            break
        inc = diff / free.size
        # limit inc so we don't cross bounds
        if diff > 0:
            max_inc = np.min(1.0 - h[free])
            inc = min(inc, max_inc)
        else:
            max_inc = np.min(h[free])
            inc = max(inc, -max_inc)
        h[free] += inc
        # clip just in case
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        it += 1
    # final possible small residual fix
    diff = target_sum - np.sum(h)
    if abs(diff) > tol:
        # adjust any index that can accommodate
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) <= tol:
                break
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy import optimize

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx on a uniform grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Clip to [0,1] and adjust to meet the exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    tol = 1e-12
    it = 0
    max_it = 500
    while abs(diff) > tol and it < max_it:
        free = np.where(((diff > 0) & (h < 1.0)) | ((diff < 0) & (h > 0.0)))[0]
        if free.size == 0:
            break
        inc = diff / free.size
        if diff > 0:
            cap = np.min(1.0 - h[free])
            inc = min(inc, cap)
        else:
            cap = np.min(h[free])
            inc = max(inc, -cap)
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        it += 1
    # final tiny adjustment if needed
    diff = target_sum - np.sum(h)
    if abs(diff) > tol:
        for i in range(h.size):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) <= tol:
                break
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_and_fix(h, target_sum):
    """Gaussian smoothing (kernel [0.25,0.5,0.25]) followed by projection."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    sm = project_h(sm, target_sum)
    return sm

    rng = np.random.default_rng(seed)

    # ----- main simulated‑annealing loop -----------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:

        T = T0 * (1.0 - elapsed / budget_s) ** 2

        r = rng.random()
        if r < p_swap:

            i, j = rng.choice(n, size=2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
