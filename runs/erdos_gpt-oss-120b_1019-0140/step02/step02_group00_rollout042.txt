def hillclimb_binary(b, time_limit, rng):
    """
    b: 1D numpy array of 0/1 float, length n, sum n/2.
    Returns improved b (maybe same) and min_overlap.
    """
    n = b.shape[0]
    # compute overlaps for all shifts (including 0)
    # Using np.correlate: correlation of b with b.
    corr = np.correlate(b, b, mode='full')
    # shift values:
    overlaps = corr[n-1:]  # length n, index 0 shift0.
    # Ensure integer type:
    overlaps = overlaps.astype(int)

    best_min = np.min(overlaps[1:])  # exclude shift0
    best_b = b.copy()
    start = time.time()
    # Precompute list of indices for ones and zeros for faster selection? We'll recompute each iteration? Could keep arrays but update each swap.
    # We'll maintain sets of ones and zeros as python lists for random selection.
    # However we need to update them after each accepted swap.
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    # function to recompute min after a swap using delta for each shift
    # We'll compute delta array of length n using vectorized operations.

    while time.time() - start < time_limit:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # compute delta:
        idx = np.arange(n)
        ip = (i + idx) % n
        im = (i - idx) % n
        jp = (j + idx) % n
        jm = (j - idx) % n
        # contributions from i (before) = b[ip] + b[im]; after = 0+0
        # contributions from j (before) = 0; after = b[jp]+b[jm]
        delta = (b[jp] + b[jm]) - (b[ip] + b[im])
        delta[0] = 0  # ensure shift0 unchanged

        new_overlaps = overlaps + delta
        new_min = np.min(new_overlaps[1:])  # ignoring shift0

        if new_min > best_min:
            # accept
            # update b
            b[i] = 0.0
            b[j] = 1.0
            # update ones and zeros arrays
            # swap i from ones to zeros, j from zeros to ones
            # easier: recompute ones and zeros after swap
            # but more efficient: update arrays manually:
            # remove i from ones, add i to zeros; remove j from zeros, add j to ones.
            # We'll just recompute for simplicity, cost negligible.
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            overlaps = new_overlaps
            best_min = new_min
            best_b = b.copy()
            # Early exit if min overlap reaches target.
            # We'll let external code handle termination.
        else:
            # maybe accept with small probability? skip
            pass

def hillclimb_binary(b, time_limit, rng):
    n = b.shape[0]
    # compute overlaps
    corr = np.correlate(b, b, mode='full')
    overlaps = corr[n-1:].astype(int)  # shift 0 to n-1
    best_min = np.min(overlaps[1:])
    best_b = b.copy()
    start = time.time()
    while time.time() - start < time_limit:
        # pick random swap
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        idx = np.arange(n)
        ip = (i + idx) % n
        im = (i - idx) % n
        jp = (j + idx) % n
        jm = (j - idx) % n
        delta = (b[jp] + b[jm]) - (b[ip] + b[im])

        new_overlaps = overlaps + delta
        new_min = np.min(new_overlaps[1:])

            # accept
            b[i] = 0.0
            b[j] = 1.0
            overlaps = new_overlaps
            best_min = new_min
            best_b = b.copy()
            # early break if best_min >= target_min (we pass target_min as global maybe)
        # else reject; continue
    return best_b, best_min

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def linear_resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def rudin_shapiro_binary(n):
    # binary 0/1 based on parity of popcount of i & (i>>1)
    res = np.empty(n, dtype=float)
    for i in range(n):
        pc = int((i & (i >> 1)).bit_count())
        res[i] = 1.0 if (pc & 1) else 0.0
    return res

def get_min_overlap(b):
    n = b.shape[0]
    corr = np.correlate(b, b, mode='full')
    # shifts 1..n-1 correspond to indices n-1+1 .. n-1+n-1
    overlaps = corr[n: n + n - 1]  # length n-1
    # convert to int
    return int(np.min(overlaps))

def hillclimb_binary(b, time_limit, rng, target_min):
    n = b.shape[0]
    corr = np.correlate(b, b, mode='full')
    overlaps = corr[n-1:].astype(int)   # shift 0..n-1
    best_min = int(np.min(overlaps[1:]))
    best_b = b.copy()
    start = time.time()
    idx_arr = np.arange(n, dtype=int)
    while time.time() - start < time_limit:
        # pick random swap (1->0 and 0->1)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta for each shift
        ip = (i + idx_arr) % n
        im = (i - idx_arr) % n
        jp = (j + idx_arr) % n
        jm = (j - idx_arr) % n
        delta = (b[jp] + b[jm]) - (b[ip] + b[im])
        delta[0] = 0  # shift 0 unchanged
        new_overlaps = overlaps + delta
        new_min = int(np.min(new_overlaps[1:]))
        # accept move if improvement, or with small prob if equal
        if new_min > best_min or (new_min == best_min and rng.random() < 0.02):
            b[i] = 0.0
            b[j] = 1.0
            overlaps = new_overlaps
            best_min = new_min
            best_b = b.copy()
            if best_min >= target_min:
                break
        # else reject and continue
    return best_b, best_min

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo≤x_i≤hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on a uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):

def linear_resample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def rudin_shapiro_binary(n):
    """Binary Rudin–Shapiro sequence (0/1)."""
    res = np.empty(n, dtype=float)
    for i in range(n):
        pc = int((i & (i >> 1)).bit_count())
        res[i] = 1.0 if (pc & 1) else 0.0
    return res

def get_min_overlap(b):
    """Return min_{k≠0} Σ_i b_i b_{i+k} for a binary vector."""
    n = b.shape[0]
    corr = np.correlate(b, b, mode='full')
    # shifts 1..n-1 correspond to indices n..2n-2
    overlaps = corr[n: 2 * n - 1]
    return int(np.min(overlaps))

def hillclimb_binary(b, time_limit, rng, target_min):
    """Random swap hill‑climb on a balanced binary vector."""
    n = b.shape[0]
    corr = np.correlate(b, b, mode='full')
    overlaps = corr[n - 1:].astype(int)          # shifts 0..n‑1
    best_min = int(np.min(overlaps[1:]))
    best_b = b.copy()
    start = time.time()
    idx_arr = np.arange(n, dtype=int)
    while time.time() - start < time_limit:
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        ip = (i + idx_arr) % n
        im = (i - idx_arr) % n
        jp = (j + idx_arr) % n
        jm = (j - idx_arr) % n
        delta = (b[jp] + b[jm]) - (b[ip] + b[im])
        delta[0] = 0                         # shift 0 stays constant
        new_overlaps = overlaps + delta
        new_min = int(np.min(new_overlaps[1:]))
        # Accept improving moves, or occasional neutral moves
        if new_min > best_min or (new_min == best_min and rng.random() < 0.02):
            b[i] = 0.0
            b[j] = 1.0
            overlaps = new_overlaps
            best_min = new_min
            best_b = b.copy()
            if best_min >= target_min:
                break
    return best_b, best_min

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
