def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 10.0  # ensure below min possible to get all lo
    high = hi - np.min(v) + 10.0
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low+high)/2.0, lo, hi)

def slsqp_optimize(h_start, t_start, sum_con, overlap_constraints, maxiter=500):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)]*n + [(0.0, None)]
    x0 = np.concatenate([h_start, [t_start]])
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_constraints]
    res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res

class SumConstraint:
    def __init__(self, n, target):
        self.n=n; self.target=target
    def __call__(self,x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n, dx):
        self.k=k; self.n=n; self.dx=dx
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 10.0
    high = hi - np.min(v) + 10.0
    for _ in range(50):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    def __init__(self, n, target):
        self.n = n
        self.target = target
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

    return x[-1]

    sum_con = SumConstraint(n, target_sum)
    overlap_cons = [OverlapConstraint(k, n, dx) for k in range(-n + 1, n) if k != 0]

    # Phase 1: random sampling
    random_deadline = start_time + budget_s * 0.1
    while time.time() < random_deadline:
        cand = project_to_box_sum(rng.random(n), target_sum)
        c5 = compute_c5(cand)
        if c5 < best_c5:
            best_h, best_c5 = cand, c5

    # Phase 2: SLSQP restarts
    slsqp_deadline = start_time + budget_s * 0.85
    maxiter = 800
    while time.time() < slsqp_deadline:
        if rng.random() < 0.2:
            v = best_h + 0.2 * (rng.random(n) - 0.5)
        else:
            v = rng.random(n)
        h_start = project_to_box_sum(v, target_sum)
        t_start = compute_c5(h_start)

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        constraints = [{'type': 'eq', 'fun': sum_con}]
        constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]

        res = minimize(
            fun=objective_t,
            x0=np.concatenate([h_start, [t_start]]),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
        )
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-12:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break

    # Phase 3: simulated annealing hill climbing
    cur_h = best_h.copy()
    cur_c5 = best_c5
    base_step = 0.2
    T = 0.01
    cooling = 0.995
    min_T = 1e-6
    while time.time() - start_time < budget_s - 0.5:
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2)
        cand = cur_h + step * d
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c5 = cand, cand_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5
                if best_c5 < 0.3808:
                    break
        T = max(T * cooling, min_T)
        if rng.random() < 0.001:
            base_step *= 0.9

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
