        # Ensure T not zero.

            if elapsed > budget_s - 5:   # leave some margin
                break

            ones = np.flatnonzero(cur_bin)
            zeros = np.flatnonzero(1 - cur_bin)
            if len(ones)==0 or len(zeros)==0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

            # Create new candidate by swapping
            cand = cur_bin.copy()
            cand[i] = 0
            cand[j] = 1

            cand_val = compute_c5_vec(cand.astype(float), dx)

            delta = cand_val - cur_val
            # temperature schedule
            frac = elapsed / (budget_s - 5)
            T = 0.01 * (0.1 ** frac)  # decays from 0.01 to 0.001 approx.

            accept = False
            if delta < 0:
                accept = True
            else:
                if T > 0:
                    if rng.random() < np.exp(-delta / T):
                        accept = True
            if accept:
                cur_bin = cand
                cur_val = cand_val
                if cur_val < best_val:
                    best_val = cur_val

                    # Early stop if under target

def refine_continuous_simple(h, dx, target_sum, budget_s, start_time, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.1
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 2:
            break
        # pick donor with h>0
        donors = np.where(cur_h > 1e-9)[0]
        recipients = np.where(cur_h < 1.0 - 1e-9)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.2, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.9
    # Ensure feasibility (sum)
    cur_h = np.clip(cur_h, 0.0, 1.0)
    cur_sum = np.sum(cur_h)
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # distribute diff uniformly while staying in bounds
        cur_h += diff / len(cur_h)
        cur_h = np.clip(cur_h, 0.0, 1.0)
        # If still diff, adjust greedily
        cur_sum = np.sum(cur_h)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # adjust top entries
            order = np.argsort(-cur_h) if diff > 0 else np.argsort(cur_h)
            for idx in order:
                if diff <= 0: break
                if diff > 0 and cur_h[idx] < 1.0:
                    inc = min(diff, 1.0 - cur_h[idx])
                    cur_h[idx] += inc
                    diff -= inc
                elif diff < 0 and cur_h[idx] > 0.0:
                    dec = min(-diff, cur_h[idx])
                    cur_h[idx] -= dec
                    diff += dec
                if abs(diff) < 1e-12:
                    break
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # Initial best solution
    if h0 is not None:
        best_h = np.clip(h0, 0.0, 1.0)
        # project to exact sum
        cur_sum = np.sum(best_h)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            best_h += diff / n
            best_h = np.clip(best_h, 0.0, 1.0)
        best_val = compute_c5_vec(best_h, dx)
    else:

        best_h = rng.random(n)
        best_h = best_h * (target_sum / best_h.sum())

        cur_sum = np.sum(best_h)
        diff = target_sum - cur_sum

            best_h += diff / n

def simulated_annealing_binary(bin_h, dx, budget_remaining, start_time, rng):

    cur_bin = bin_h.copy()

    n = len(bin_h)

    end_time = start_time + budget_remaining

        ones = np.flatnonzero(cur_bin)

def refine_continuous_simple(h, dx, target_sum, time_limit, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.1
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.2, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.9
    # final ensure sum
    cur_h = np.clip(cur_h, 0.0, 1.0)
    cur_sum = np.sum(cur_h)
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        cur_h += diff / n
        cur_h = np.clip(cur_h, 0.0, 1.0)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

import time, numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def refine_continuous_simple(h, target_sum, time_limit, rng):
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    step = 0.1
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recips = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recips.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recips)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5(cand)
        if cand_val < cur_val - 1e-12:
            cur_h, cur_val = cand, cand_val
            step = min(step * 1.2, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.9
    # enforce sum
    cur_h = np.clip(cur_h, 0.0, 1.0)
    diff = target_sum - cur_h.sum()
    if abs(diff) > 1e-12:
        cur_h += diff / n
        cur_h = np.clip(cur_h, 0.0, 1.0)
    return cur_h, compute_c5(cur_h)

def simulated_annealing_binary(bin_h, time_limit, start_time, rng):
    best_bin = bin_h.copy()
    best_val = compute_c5(bin_h.astype(float))
    cur_bin, cur_val = bin_h.copy(), best_val
    n = len(bin_h)
    T0, Tmin = 0.01, 1e-6
    end_time = start_time + time_limit
    while time.time() < end_time:
        ones = np.flatnonzero(cur_bin)
        zeros = np.flatnonzero(1 - cur_bin)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur_bin.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand.astype(float))
        delta = cand_val - cur_val
        elapsed = time.time() - start_time
        T = max(T0 * np.exp(-5.0 * elapsed / (end_time - start_time)), Tmin)
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_bin, cur_val = cand, cand_val
            if cur_val < best_val:
                best_bin, best_val = cur_bin.copy(), cur_val
                if best_val < 0.3808:
                    break
    return best_bin, best_val

def legendre_search(start_time, budget_s, rng):
    best_seq, best_val = None, np.inf
    time_limit = start_time + min(10.0, budget_s * 0.05)
    # simple prime generator up to 199
    for n in range(31, 200, 2):
        # primality test
        is_prime = True
        if n % 3 == 0 and n != 3:
            is_prime = False
        else:
            i, w = 5, 2
            while i * i <= n:
                if n % i == 0:
                    is_prime = False
                    break
                i += w
                w = 6 - w
        if not is_prime:
            continue
        seq = np.zeros(n, dtype=float)
        for i in range(1, n):
            if pow(i, (n - 1) // 2, n) == 1:
                seq[i] = 1.0
        seq[0] = n / 2.0 - seq.sum()  # adds 0.5 for odd n
        cur_val = compute_c5(seq)
        if cur_val < best_val:
            best_seq, best_val = seq.copy(), cur_val
        if time.time() > time_limit:
            break
    return best_seq, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # start from parent if present

        cur_sum = best_h.sum()
        target = len(best_h) / 2.0

        if abs(diff) > 1e-12:
            best_h += diff / len(best_h)

    else:
        best_h = None

    if best_h is None:

        target = n / 2.0

        diff = target - best_h.sum()
        if abs(diff) > 1e-12:
            best_h += diff / n
            best_h = np.clip(best_h, 0.0, 1.0)

    # update parameters after any improvement
    n = len(best_h)
    target_sum = n / 2.0

    # random sampling phase (~15% budget)
    phase_end = start + budget_s * 0.15

        x = x * (target_sum / x.sum())
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - x.sum()
        if abs(diff) > 1e-12:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
        val = compute_c5(x)
        if val < best_val:
            best_h, best_val = x, val
            if best_val < 0.3808:
                break

    # continuous refinement (20% of remaining)
    remaining = budget_s - (time.time() - start)
    if remaining > 10.0:

        refined_h, refined_val = refine_continuous_simple(best_h, target_sum, lim, rng)
        if refined_val < best_val:
            best_h, best_val = refined_h, refined_val
            if best_val < 0.3808:
                return best_h, best_val, len(best_h)

    # binary simulated annealing (if even length)
    if n % 2 == 0:
        half = n // 2
        bin_h = np.zeros(n, dtype=int)

        # allocate remaining time
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            bin_h, bin_val = simulated_annealing_binary(bin_h, remaining, start, rng)
            if bin_val < best_val:
                best_h = bin_h.astype(float)
                best_val = bin_val
                if best_val < 0.3808:
                    return best_h, best_val, len(best_h)

    # final short continuous tweak
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        lim = time.time() + remaining
        best_h, best_val = refine_continuous_simple(best_h, target_sum, lim, rng)

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        # tiny corrective pass
        cur = best_h.sum()
        diff = target_sum - cur
        if abs(diff) > 1e-12:
            idxs = np.argsort(-best_h) if diff > 0 else np.argsort(best_h)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and best_h[i] < 1.0:
                    inc = min(diff, 1.0 - best_h[i])
                    best_h[i] += inc
                    diff -= inc
                elif diff < 0 and best_h[i] > 0.0:
                    dec = min(-diff, best_h[i])
                    best_h[i] -= dec
                    diff += dec
    best_val = compute_c5(best_h)
    return best_h, best_val, len(best_h)
