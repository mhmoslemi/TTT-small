def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        # term1: (1 - h[i+k]) for i in [0, n-k-1]
        grad[:n-k] += 1 - h[k:]
        # term2: -h[i-k] for i in [k, n-1]
        grad[k:] -= h[:n-k]
    else:
        kk = -k
        grad[kk:] += 1 - h[:n-kk]
        grad[:n-kk] -= h[kk:]
    return grad

def gradient_overlap_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    # find indices where value within tolerance
    # tolerance relative small
    idxs = np.where(np.isclose(corr, max_val, atol=1e-12))[0]
    n = h.shape[0]
    shifts = idxs - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        grad += gradient_for_shift(h, k)
    # average
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while enforcing sum(v)=target via a scalar shift."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def _gradient_for_shift(h, k):
    """Gradient of overlap_k = sum_i h[i]*(1-h[i+k]) w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_overlap_max(h):
    """Sub‑gradient of C5 (max overlap) at h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    idxs = np.where(np.isclose(corr, max_val, atol=1e-12))[0]
    shifts = idxs - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        grad += _gradient_for_shift(h, int(k))
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': sum_cons}] + overlap_cons,
                           options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
            if res.success:
                cand_h = np.clip(res.x[:n], 0.0, 1.0)
                cand_h = project_to_box_sum(cand_h, target_sum)
                cand_c5 = compute_c5(cand_h)
                if cand_c5 < best_c5:
                    best_h, best_c5 = cand_h, cand_c5
        except Exception:
            pass

    # ----- Final safety projection -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
