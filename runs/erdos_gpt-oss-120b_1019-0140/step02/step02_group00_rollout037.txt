def compute_c5_norm_binary(h):
    n = h.shape[0]
    # compute autocorrelation via FFT
    # use convolution between h and reversed h
    h_rev = h[::-1]
    length = 2 * n
    # FFT
    H = np.fft.fft(h, length)
    H_rev = np.fft.fft(h_rev, length)
    conv = np.fft.ifft(H * H_rev).real
    # round to nearest integer (due to floating error)
    conv = np.rint(conv).astype(np.int64)
    # extract relevant part length 2n-1
    conv = conv[:2*n-1]
    min_conv = conv.min()   # includes shift 0 maybe max conv; we need min across all shifts
    # compute normalized bound
    return 1.0 - (2.0 * min_conv) / n
```

    n = h.shape[0]
    H = np.fft.fft(h, 2*n)
    G = np.fft.fft((1.0 - h)[::-1], 2*n)

            term1[:n - shift] = h[shift:]
            term2 = np.zeros(n, dtype=h.dtype)

            term2 = np.zeros(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)

import time
import math
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # enlarge bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation to a larger grid."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Linear interpolation to a smaller grid."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_fft(h):
    """Linear correlation of h with (1-h) using FFT (full length)."""
    n = h.shape[0]
    H = np.fft.fft(h, 2 * n)
    G = np.fft.fft((1.0 - h)[::-1], 2 * n)
    corr = np.fft.ifft(H * G).real
    return corr[: 2 * n - 1]  # length 2n-1


def compute_c5_continuous(h):
    """Normalized C5 for a (possibly fractional) step function."""
    n = h.shape[0]
    max_corr = np.max(compute_corr_fft(h))
    return max_corr * (2.0 / n)


def compute_c5_binary(h):
    """Normalized C5 for a binary vector (0/1)."""
    n = h.shape[0]
    H = np.fft.fft(h, 2 * n)
    H_rev = np.fft.fft(h[::-1], 2 * n)
    conv = np.fft.ifft(H * H_rev).real
    conv = np.rint(conv[: 2 * n - 1]).astype(np.int64)
    min_conv = conv.min()
    return 1.0 - (2.0 * min_conv) / n


def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient w.r.t the max‑correlation term."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def slsqp_optimize_candidate(h0, max_iter=500):
    """Short SLSQP refinement of a candidate."""
    n = h0.shape[0]
    t0 = compute_c5_continuous(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        max_corr = np.max(compute_corr_fft(h))
        return t - max_corr * (2.0 / n)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5_continuous(h_opt)
    return h_opt, t_opt


def gradient_descent_momentum_fft(h0, n_iter=20000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum, using FFT for correlations."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_continuous(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_continuous(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    best_h = None
    best_score = float('inf')
    target = 0.3808

    n_small = 256
    n_large = 512

    # ----- Phase 1: Small‑scale SLSQP search -----
    base_small = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        base_small.append(project_onto_box_sum(downsample(init_full, n_small), n_small / 2.0))
    base_small.append(thue_morse(n_small))
    for _ in range(3):
        base_small.append(random_binary_balanced(n_small, rng))
    for _ in range(2):
        v = rng.random(n_small)
        base_small.append(project_onto_box_sum(v, n_small / 2.0))

    for cand in base_small:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=500)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            break

    while time_left() > 8.0 and best_score > target:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=300)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            break

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score = compute_c5_continuous(best_h)

    # ----- Phase 2: Upsample to larger grid -----
    h_large = upsample(best_h, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    sc_large = compute_c5_continuous(h_large)
    if sc_large < best_score:
        best_score, best_h = sc_large, h_large.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # ----- Phase 3: Large‑scale gradient descent -----
    remaining = time_left()
    if remaining > 5.0:
        est_iters = int(min(250000, max(5000, remaining * 500)))
        h_ref, sc_ref = gradient_descent_momentum_fft(best_h, n_iter=est_iters,
                                                      step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 4: Binary simulated annealing -----
    # Convert current best to binary (top half values)
    n = best_h.shape[0]
    idx_top = np.argsort(-best_h)[: n // 2]
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[idx_top] = 1.0

    cur_h = h_bin.copy()
    cur_score = compute_c5_binary(cur_h)
    best_h_bin = cur_h.copy()
    best_score_bin = cur_score

    # also try Thue–Morse binary start
    h_tm = thue_morse(n).astype(np.float64)
    tm_score = compute_c5_binary(h_tm)
    if tm_score < best_score_bin:
        best_score_bin = tm_score
        best_h_bin = h_tm.copy()
        cur_h = h_tm.copy()
        cur_score = tm_score

    ones = list(np.where(cur_h == 1.0)[0])
    zeros = list(np.where(cur_h == 0.0)[0])

    T0 = 0.008
    while time_left() > 0.5:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        idx_one = rng.integers(len(ones))
        idx_zero = rng.integers(len(zeros))
        i1 = ones[idx_one]
        i0 = zeros[idx_zero]

        # perform swap
        cur_h[i1] = 0.0
        cur_h[i0] = 1.0
        cand_score = compute_c5_binary(cur_h)

        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            ones[idx_one] = i0
            zeros[idx_zero] = i1
            cur_score = cand_score
            if cand_score < best_score_bin:
                best_score_bin = cand_score
                best_h_bin = cur_h.copy()
                if best_score_bin <= target:
                    break
        else:
            cur_h[i1] = 1.0
            cur_h[i0] = 0.0

    # ----- Choose best overall -----
    if best_score_bin < best_score:
        best_h = best_h_bin
        best_score = best_score_bin

    best_h = best_h.astype(np.float64)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_continuous(best_h)

    return best_h, best_score, best_h.size
