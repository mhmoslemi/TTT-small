def compute_c5_fft(h):
    n = h.shape[0]
    m = 1 << (2*n-1).bit_length()  # next power of two >= 2n
    # compute FFTs
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1-h, n=m)
    corr = np.fft.irfft(H * np.conj(G), n=m)
    # central part corresponds to full correlation length 2n-1; slice from 0 to 2n-2 inclusive
    corr = corr[:2*n-1]  # maybe due to rounding errors
    # scaling: Since using standard FFT with no scaling, not needed.
    return np.max(corr) * (2.0 / n)

  m = np.zeros_like(h)
  v = np.zeros_like(h)
  s = h.size/2.0
  for t in range(1, n_iter+1):
      # compute correlation using FFT

      # get max shift indices

      # convert to shift (negative to positive)
      shifts = idxes - (h.size - 1)
      grad = np.zeros_like(h)
      for shift in shifts:
          grad += gradient_for_shift(h, shift)
      grad /= len(shifts)
      # Adam updates
      m = beta1 * m + (1-beta1) * grad

      h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
      # projection onto box & sum constraints
      h = project_onto_box_sum(h_new, s)
  return h

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust bounds if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def correlation_raw(h):
    n = h.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * np.conj(G), n=m)

        grad[:n - s]   -= h[s:]
    return grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {x: lo<=x_i<=hi, sum x_i = s} using a simple
    bisection on the Lagrange multiplier."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def correlation_raw(h):
    """Linear (non‑circular) correlation of h with (1‑h).  Returns the
    unscaled sum for each integer shift k∈[‑n+1,…,n‑1]."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * np.conj(G), n=m)
    return corr[: 2 * n - 1]

def compute_c5(h):
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = correlation_raw(h)
    return corr.max() * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_for_shift(h, shift):
    """Sub‑gradient of the max‑correlation term for a given shift."""
    n = h.shape[0]
    grad = np.ones_like(h)
    if shift >= 0:
        grad[: n - shift] -= h[shift:]
        grad[shift:]     -= h[: n - shift]
    else:
        s = -shift
        grad[s:]         -= h[: n - s]
        grad[: n - s]    -= h[s:]
    return grad

def adam_optimize(h0, n_iter, step0, rng):
    """Adam‑style projected optimisation of the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for t in range(1, n_iter + 1):
        corr = correlation_raw(h)
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        grad = gradient_for_shift(h, shift)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = step0 / np.sqrt(t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h_new, s)

        # occasional jitter to escape flat regions
        if t % 5000 == 0:
            h = project_onto_box_sum(h + rng.normal(scale=0.02, size=n), s)

        # keep track of the best seen so far
        if t % 2000 == 0 or t == n_iter:
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # grid sizes
    n_small = 256
    n_large = 512

    best_h = None
    best_c5 = float('inf')

    # ---- Phase 0: initial candidates on the small grid ----
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        h0 = upsample(init, n_small)
        h0 = project_onto_box_sum(h0, n_small / 2.0)
        c0 = compute_c5(h0)
        best_h, best_c5 = h0, c0

    # a few random balanced binaries
    for _ in range(5):
        if time_left() < 2.0:
            break
        cand = random_binary_balanced(n_small, rng)
        c = compute_c5(cand)
        if c < best_c5:
            best_h, best_c5 = cand, c

    if best_h is None:
        best_h = random_binary_balanced(n_small, rng)
        best_c5 = compute_c5(best_h)

    # ---- Phase 1: Adam refinement on the small grid ----
    while time_left() > 30.0:
        remaining = time_left()
        iters = int(min(200000, max(2000, remaining * 3000)))
        cand, c = adam_optimize(best_h, iters, step0=0.2, rng=rng)
        if c < best_c5:
            best_h, best_c5 = cand, c
        if best_c5 <= 0.3808:
            break

    # ---- Phase 2: Upsample to a larger grid and refine ----
    if time_left() > 20.0:
        h_big = upsample(best_h, n_large)
        h_big = project_onto_box_sum(h_big, n_large / 2.0)
        c_big = compute_c5(h_big)
        if c_big < best_c5:
            best_h, best_c5 = h_big, c_big

        while time_left() > 15.0:
            remaining = time_left()
            iters = int(min(300000, max(2000, remaining * 2000)))
            cand, c = adam_optimize(best_h, iters, step0=0.2, rng=rng)
            if c < best_c5:
                best_h, best_c5 = cand, c
            if best_c5 <= 0.3808:
                break

    # ---- Phase 3: Random binary hill‑climbing on the current resolution ----
    n_cur = best_h.shape[0] if best_h is not None else n_small
    while time_left() > 5.0:
        cand = random_binary_balanced(n_cur, rng)
        c = compute_c5(cand)
        if c < best_c5:
            best_h, best_c5 = cand, c
        if best_c5 <= 0.3808:
            break

    # final safeguard
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_h.size
