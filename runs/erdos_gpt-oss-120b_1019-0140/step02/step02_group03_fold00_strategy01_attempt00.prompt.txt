Find a step function h: [0, 2] → [0, 1] that **minimizes** the overlap integral:

$$C_5 = \max_k \int h(x)(1 - h(x+k)) dx$$

**Constraints**:

- 0 ≤ h[i] ≤ 1 for all i

    """Maximum overlap integral for a given step vector h."""

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the hypercube [lo,hi]^n while fixing the sum to target."""
    # Binary search a constant λ so that sum(clip(v+λ, lo, hi)) == target
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0

class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

        h = x[:self.n]
        k = self.k

        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
