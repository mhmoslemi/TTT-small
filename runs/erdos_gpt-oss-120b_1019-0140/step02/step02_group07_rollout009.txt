import time
import math
import numpy as np

def compute_c5(arr):
    """Maximum overlap integral approximated by discrete correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Thue-Morse binary sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    """Rudin-Shapiro binary sequence (+1/-1) mapped to 0/1."""
    a = np.array([1], dtype=int)
    while a.size < n:
        a = np.concatenate([a, a * ((-1) ** np.arange(a.size))])
    a = a[:n]
    return ((a + 1) // 2).astype(int)

def generate_random_binary(n, target_sum, rng):
    """Random binary vector with exactly target_sum ones."""
    vec = np.zeros(n, dtype=int)
    vec[rng.choice(n, size=target_sum, replace=False)] = 1
    return vec

def repair_to_target(vec, target, rng):
    """Flip bits to reach the required number of ones."""
    current = int(vec.sum())
    diff = target - current
    if diff > 0:
        zeros = np.flatnonzero(vec == 0)
        if zeros.size:
            chosen = rng.choice(zeros, size=diff, replace=False)
            vec[chosen] = 1
    elif diff < 0:
        ones = np.flatnonzero(vec == 1)
        if ones.size:
            chosen = rng.choice(ones, size=-diff, replace=False)
            vec[chosen] = 0
    return vec

def tournament_select(fitness, size, rng):
    """Choose the best index among a random subset."""
    idx = rng.integers(len(fitness), size=size)
    best = idx[0]
    best_val = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_val:
            best = i
            best_val = fitness[i]
    return best

def uniform_crossover(p1, p2, rng):
    """Uniform crossover preserving bit type."""
    mask = rng.random(p1.shape) < 0.5
    child = np.where(mask, p1, p2)
    return child.astype(int)

def mutate_swap(vec, rng, swaps=1):
    """Swap bits (1↔0) a few times."""
    n = vec.size
    for _ in range(swaps):
        ones = np.flatnonzero(vec == 1)
        zeros = np.flatnonzero(vec == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        vec[i] = 0
        vec[j] = 1
    return vec

def binary_hill_climb(start_bin, time_end, rng):
    """Deterministic hill climb using swaps."""
    h = start_bin.copy()
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    while time.time() < time_end and no_improve < 2000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h.astype(float))
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h.copy()
            cur_c5 = new_c5
            no_improve = 0
        else:
            # revert
            h[i] = 1
            h[j] = 0
            no_improve += 1
    return best_h.astype(float), best_c5
