import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _adjust_sum(vec, target, rng):
    """Adjust vec so that sum(vec) == target while keeping 0≤vec≤1."""
    cur = vec.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return vec
    # try a uniform shift first
    vec = np.clip(vec + diff / len(vec), 0.0, 1.0)
    cur = vec.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return vec
    # fix remaining discrepancy by local moves
    idxs = np.arange(len(vec))
    rng.shuffle(idxs)
    for i in idxs:
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec

def _anneal_opt(h_start, best_val, start_time, deadline, rng):
    """Simulated annealing that moves mass from heavy contributors to beneficial locations."""
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    cur_val = best_val
    best_h = h.copy()
    best = best_val
    total_budget = deadline - start_time
    T0 = 0.01

    while time.time() < deadline - 0.1:
        # correlation and shift with maximal overlap
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)  # signed shift

        # overlapping indices for this shift
        if k >= 0:
            i_range = np.arange(n - k)
            prod = h[i_range] * (1.0 - h[i_range + k])
        else:
            kk = -k
            i_range = np.arange(kk, n)
            prod = h[i_range] * (1.0 - h[i_range - kk])

        # pick source i weighted by its contribution (heavier more likely)
        if prod.sum() > 0:
            prob = prod / prod.sum()
            i = rng.choice(i_range, p=prob)
        else:
            i = rng.choice(i_range)

        # choose destination j that can accept mass and preferably reduces the product
        dst_candidates = np.where(h < 1.0 - 1e-12)[0]
        if dst_candidates.size == 0:
            break
        # score destinations: smaller (1 - h_{j+k}) is better (i.e., larger h_{j+k})
        if k >= 0:
            mask = dst_candidates < n - k
            score = np.full_like(dst_candidates, 1e6, dtype=float)
            score[mask] = 1.0 - h[dst_candidates[mask] + k]
        else:
            mask = dst_candidates >= -k
            score = np.full_like(dst_candidates, 1e6, dtype=float)
            score[mask] = 1.0 - h[dst_candidates[mask] + k]  # k negative works as index shift
        min_score = score.min()
        good_js = dst_candidates[score <= min_score + 1e-12]
        j = rng.choice(good_js)

        if i == j:
            continue

        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        # bias towards smaller moves
        delta = max_delta * rng.random() ** 2
        if delta < 1e-12:
            continue

        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
                if best < 0.3808:
                    break
        else:
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            T = T0 * (remaining / total_budget)
            if T < 1e-12:
                T = 1e-12
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert move
                h[i] += delta
                h[j] -= delta

    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # problem size
    n = 80
    target_sum = n / 2.0

    # initial h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _adjust_sum(h0, target_sum, rng)
    else:
        # start from a random binary pattern (half ones)
        idx = rng.choice(n, n // 2, replace=False)
        h0 = np.zeros(n, dtype=float)
        h0[idx] = 1.0

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- SLSQP phase -----
    dx = 2.0 / n

    def sum_eq(x):
        return np.sum(x[:n]) - target_sum

    constraints = [{"type": "eq", "fun": sum_eq}]
    def make_overlap_con(k):
        def con(x, k=k):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])
            return x[-1] - overl * dx
        return con
    for k in range(-n + 1, n):
        if k == 0:
            continue
        constraints.append({"type": "ineq", "fun": make_overlap_con(k)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # multiple random starts
    while time.time() < deadline - 5.0:
        # choose random start type
        if rng.random() < 0.5:
            # binary start
            idx = rng.choice(n, n // 2, replace=False)
            h_start = np.zeros(n, dtype=float)
            h_start[idx] = 1.0
        else:
            # continuous random start
            v = rng.random(n)
            v = v - np.mean(v) + target_sum / n
            v = np.clip(v, 0.0, 1.0)
            v = _adjust_sum(v, target_sum, rng)
            h_start = v

        t0 = compute_c5(h_start)
        x0 = np.concatenate([h_start, [t0]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 1500, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            break

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # enforce exact sum if needed
        cur = h_opt.sum()
        if abs(cur - target_sum) > 1e-9:
            h_opt = _adjust_sum(h_opt, target_sum, rng)

        val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

        if time.time() > deadline - 5.0:
            break

    # ----- annealing refinement -----
    if time.time() < deadline - 0.5:
        best_h, best_val = _anneal_opt(best_h, best_val, start_time, deadline, rng)

    # final correction
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target_sum) > 1e-9:
        best_h = _adjust_sum(best_h, target_sum, rng)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
