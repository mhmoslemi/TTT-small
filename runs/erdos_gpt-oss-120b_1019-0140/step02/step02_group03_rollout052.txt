import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target via a simple bisection."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k > 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        elif k < 0:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        else:
            overlap = np.dot(h, 1.0 - h)
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=400):

    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

            contrib = cur_h[kp:] * (1.0 - cur_h[:n - kp])
            order = np.argsort(-contrib)[:5]
            for i_local in order:
                i = int(i_local) + kp
                j = i - kp
                eps_cand = min(step, cur_h[i], 1.0 - cur_h[j])
                if eps_cand > 1e-12:
                    i_star, j_target, eps = i, j, eps_cand
                    moved = True
                    break
        else:
            # k_max == 0: fallback to random pair
            moved = False

        if not moved:
            # random feasible pair
            i_star = int(rng.integers(0, n))
            j_target = int(rng.integers(0, n))
            while j_target == i_star:
                j_target = int(rng.integers(0, n))
            eps = min(step, cur_h[i_star], 1.0 - cur_h[j_target])
            if eps <= 0:
                eps = min(step * rng.random(), cur_h[i_star], 1.0 - cur_h[j_target])

        cand = cur_h.copy()
        cand[i_star] -= eps
        cand[j_target] += eps
        cand_c5 = compute_c5(cand)

        # deterministic improvement acceptance
        if cand_c5 + 1e-12 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h = cand.copy()
                best_c5 = cur_c5
        else:
            # occasional uphill acceptance to escape plateaus
            if rng.random() < 0.001:
                cur_h = cand
                cur_c5 = cand_c5

        it += 1
        if it % 5000 == 0:
            step = max(step * 0.9, 1e-4)

    # ----- final safety projection -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
