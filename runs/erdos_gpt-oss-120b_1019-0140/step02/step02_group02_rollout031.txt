def compute_gradient(h, alpha):
    n = h.shape[0]
    dx = 2.0 / n
    O = np.correlate(h, 1.0 - h, mode='full') * dx
    # compute weights
    maxO = O.max()
    exp_vals = np.exp(alpha * (O - maxO))
    w = exp_vals / exp_vals.sum()
    # compute gradient
    grad = np.zeros_like(h)
    shifts = np.arange(-n+1, n)
    for idx, k in enumerate(shifts):
        weight = w[idx]
        if k >= 0:
            # term1: 1 - h[i+k]; term2: h[i-k] where i>=k else 0
            term1 = np.empty(n)
            term1[:n-k] = 1.0 - h[k:]
            term1[n-k:] = 1.0
            term2 = np.empty(n)
            term2[:k] = 0.0
            term2[k:] = h[:n-k]
        else:
            kk = -k
            term1 = np.empty(n)
            term1[kk:] = 1.0 - h[:n-kk]
            term1[:kk] = 1.0
            term2 = np.empty(n)
            term2[:n-kk] = h[kk:]
            term2[n-kk:] = 0.0
        grad += weight * (dx * (term1 - term2))
    return grad
```

So shifts = np.arange(-n+1, n). Use enumeration accordingly.

def gradient_descent_h(h0, rng, remaining_time):
    n = h0.shape[0]
    target_sum = n/2
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    alpha = 30.0
    while time.time() - start < remaining_time - 0.05:
        O = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        maxO = O.max()
        # compute smooth objective
        maxO = O.max()  # for storing
        exp_vals = np.exp(alpha * (O - maxO))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        grad = np.zeros_like(h)
        shifts = np.arange(-n+1, n)
        dx = 2.0 / n
        for idx, k in enumerate(shifts):
            weight = w[idx]
            if k >= 0:
                term1 = np.empty(n)
                term1[:n-k] = 1.0 - h[k:]
                term1[n-k:] = 1.0
                term2 = np.empty(n)

                term1 = np.empty(n)

                term1[:kk] = 1.0
                term2 = np.empty(n)
                term2[:n-kk] = h[kk:]
                term2[n-kk:] = 0.0

        step = 1.0
        cur_sm = maxO + (1.0/alpha) * np.log(np.sum(np.exp(alpha * (O - maxO))))  # same as sm

        val = compute_c5(h)
        if val < best_val:

            if best_val < 0.3808:

def project_onto_box_sum(v, target):
    # Clip initial
    v = np.clip(v, 0.0, 1.0)
    # If sum equals target within tolerance, return
    s = v.sum()
    if abs(s - target) < 1e-9:
        return v
    # Use bisection on lambda:
    low = -1.0
    high = 1.0
    # adjust bounds to guarantee sum high <= target <= sum low
    # define function sum_clip(lam): sum_{i} clip(v_i - lam, 0,1)
    def sum_clip(lam):
        return np.clip(v - lam, 0.0, 1.0).sum()
    # ensure bounds
    while sum_clip(low) < target:
        low -= 1.0
    while sum_clip(high) > target:
        high += 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        s_mid = sum_clip(mid)
        if s_mid > target:
            low = mid
        else:
            high = mid
    proj = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # after projection the sum may be slightly off due to tolerance; adjust a little
    return proj
```

def binary_swap_refinement(h, rng, time_limit):
    n = h.shape[0]
    target = n//2
    # round to binary
    # Choose top target entries as 1, rest 0
    idx_sorted = np.argsort(-h)  # descending
    bin_h = np.zeros(n)
    bin_h[idx_sorted[:target]] = 1.0
    best_h = bin_h.copy()
    best_val = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_limit:
        # randomly select a 1 index i and 0 index j and consider swapping
        i = rng.choice(np.where(best_h == 1)[0])
        j = rng.choice(np.where(best_h == 0)[0])
        # attempt swap
        trial = best_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        val = compute_c5(trial)
        if val < best_val - 1e-12:
            best_h = trial
            best_val = val
            # early stop if good enough
            if best_val < 0.3808:
                break
        # maybe also try acceptance of equal or worse with probability (annealing) if stuck
        # but keep simple for now.
    return best_h

def binary_anneal(h_start, cur_val, start_time, deadline, rng):
    h = h_start.copy()
    cur_val = cur_val
    best_h = h.copy()
    best = cur_val
    n = h.shape[0]
    while time.time() < deadline - 0.2:
        # compute O array
        O = np.correlate(h, 1.0 - h, mode="full") * (2.0/n)
        max_idx = np.argmax(O)
        k = max_idx - (n - 1)
        # find mismatched indices for shift k
        if k >= 0:
            i_range = np.arange(n - k)
            i_candidates = i_range[(h[i_range] == 1) & (h[i_range + k] == 0)]
            j_candidates = i_range[(h[i_range] == 0) & (h[i_range + k] == 1)]
        else:
            kk = -k
            i_range = np.arange(kk, n)
            i_candidates = i_range[(h[i_range] == 1) & (h[i_range - kk] == 0)]
            j_candidates = i_range[(h[i_range] == 0) & (h[i_range - kk] == 1)]
        if len(i_candidates) > 0 and len(j_candidates) > 0:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        else:
            # fallback random swap
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            # Metropolis acceptance
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert swap
                h[i], h[j] = 1.0, 0.0
        if best < 0.3808:
            break
    return best_h, best

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_onto_box_sum(v, target):
    """Euclidean projection onto [0,1]^n ∩ {sum = target}."""
    v = np.clip(v, 0.0, 1.0)
    low, high = -1.0, 2.0
    for _ in range(50):
        mid = (low + high) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)

def random_binary_vector(n, rng):
    v = np.zeros(n, dtype=float)
    v[rng.choice(n, n // 2, replace=False)] = 1.0
    return v

def binary_anneal(h_start, cur_val, start_time, deadline, rng):
    h = h_start.copy()
    cur = cur_val
    best_h = h.copy()
    best = cur_val
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline - 0.2:
        O = np.correlate(h, 1.0 - h, mode="full") * dx
        max_idx = np.argmax(O)
        k = max_idx - (n - 1)
        if k >= 0:
            i_range = np.arange(n - k)
            i_cand = i_range[(h[i_range] == 1) & (h[i_range + k] == 0)]
            j_cand = i_range[(h[i_range] == 0) & (h[i_range + k] == 1)]
        else:
            kk = -k
            i_range = np.arange(kk, n)
            i_cand = i_range[(h[i_range] == 1) & (h[i_range - kk] == 0)]
            j_cand = i_range[(h[i_range] == 0) & (h[i_range - kk] == 1)]
        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_val = compute_c5(h)
        if new_val < cur:
            cur = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur - new_val) / T) > rng.random():
                cur = new_val
            else:
                h[i], h[j] = 1.0, 0.0
        if best < 0.3808:
            break
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():

    # ensure binary start for annealing
    if not np.all(np.isin(h0, [0.0, 1.0])):
        idx = np.argsort(-h0)
        bin_h = np.zeros(n, dtype=float)
        bin_h[idx[:int(target_sum)]] = 1.0

    best_val = compute_c5(best_h)

    # primary annealing on the (possibly provided) start
    if time.time() < deadline - 2.0:

    # additional random restarts
    while time.time() < deadline - 2.0:

        h_opt, val_opt = binary_anneal(h_start, val_start, start_time, deadline, rng)

    cur_sum = best_h.sum()
