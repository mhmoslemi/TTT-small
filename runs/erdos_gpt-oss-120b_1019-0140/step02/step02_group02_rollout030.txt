import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return the discrete overlap bound C5 for a vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_onto_box_sum(v, target, max_iter=60):
    """
    Project vector v onto the hyperplane sum(x)=target intersected with [0,1]^len(v).
    Uses a simple bisection on a global shift λ: x = clip(v+λ,0,1).
    """
    lo, hi = -5.0, 5.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def random_start_vector(n, rng, target):
    """Generate a feasible start vector in [0,1]^n with exact sum `target`."""
    v = rng.random(n)
    v = v / v.sum() * target
    return project_onto_box_sum(v, target)

def slsqp_optimize(n, target, dx, rng, deadline):
    """Run many short SLSQP runs and keep the best feasible vector."""
    best_h = None
    best_val = np.inf

    def sum_con(x):
        return np.sum(x[:n]) - target

    constraints = [{"type": "eq", "fun": sum_con}]
    def make_overlap(k):
        def con(x, k=k):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])
            return x[-1] - overl * dx
        return con
    for kk in range(-n + 1, n):
        if kk == 0:
            continue
        constraints.append({"type": "ineq", "fun": make_overlap(kk)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    while time.time() < deadline - 5.0:
        h_start = random_start_vector(n, rng, target)
        t0 = compute_c5(h_start)
        x0 = np.concatenate([h_start, [t0]])
        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 3000, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur = h_opt.sum()
        if abs(cur - target) > 1e-9:
            h_opt += (target - cur) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

def subgradient_optimize(h0, target, dx, deadline, rng):
    """Subgradient descent on the max-overlap objective with projection."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    it = 0
    while time.time() < deadline - 0.01:
        it += 1
        # correlation to find active shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        # subgradient of the active shift
        g = np.zeros_like(h)
        if k >= 0:
            i_range = np.arange(n - k)
            g[i_range] = 1.0 - h[i_range + k]
            g[i_range + k] -= h[i_range]
        else:
            kk = -k
            i_range = np.arange(kk, n)
            g[i_range] = 1.0 - h[i_range - kk]
            g[i_range - kk] -= h[i_range]

        alpha = 1.0 / np.sqrt(it)
        h_candidate = project_onto_box_sum(h - alpha * g, target)
        cand_val = compute_c5(h_candidate)

        if cand_val < cur_val:
            h = h_candidate
            cur_val = cand_val

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val < 0.3808:
                break

        # occasional random mass‑swap to escape flat regions
        if it % 500 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(0.1, h[i])
            if delta > 1e-12:
                h_tmp = h.copy()
                h_tmp[i] -= delta
                h_tmp[j] += delta
                h_tmp = project_onto_box_sum(h_tmp, target)
                tmp_val = compute_c5(h_tmp)
                if tmp_val < cur_val:
                    h = h_tmp
                    cur_val = tmp_val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        cur_sum = h0.sum()
        if abs(cur_sum - target) > 1e-12:
            h0 = project_onto_box_sum(h0, target)
    else:
        n = 80
        target = n / 2.0
        h0 = random_start_vector(n, rng, target)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # SLSQP phase (if time permits)
    if time.time() < deadline - 200.0:
        slsqp_h, slsqp_val = slsqp_optimize(n, target, dx, rng, deadline)
        if slsqp_h is not None and slsqp_val < best_val:
            best_h, best_val = slsqp_h, slsqp_val

    # ------------------------------------------------------------------
    # Subgradient refinement for the remaining budget
    if time.time() < deadline - 5.0:
        best_h, best_val = subgradient_optimize(best_h, target, dx, deadline, rng)

    # ------------------------------------------------------------------
    # Final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h = project_onto_box_sum(best_h, target)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
