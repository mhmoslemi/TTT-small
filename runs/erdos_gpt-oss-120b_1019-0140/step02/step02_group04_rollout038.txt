def refine_continuous(h, dx, target_sum, time_limit, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.05
    n = len(cur_h)
    while time.time() < time_limit:
        # pick donor >0, recipient <1
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # while not convergence and iteration limit
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # can increase
            inc_indices = np.where(vec < 1.0)[0]
            if inc_indices.size == 0:
                break
            inc = diff / inc_indices.size
            # cap inc to max allowable per index
            inc = np.minimum(inc, 1.0 - vec[inc_indices])
            total_inc = np.sum(inc)
            vec[inc_indices] += inc
            diff -= total_inc
        else:
            # diff < 0, need to decrease
            dec_indices = np.where(vec > 0.0)[0]
            if dec_indices.size == 0:
                break
            dec = (-diff) / dec_indices.size
            dec = np.minimum(dec, vec[dec_indices])
            total_dec = np.sum(dec)
            vec[dec_indices] -= dec
            diff += total_dec
    # final small correction enforce exact sum via distributing remainder
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # adjust greedily
        if diff > 0:
            idxs = np.where(vec < 1.0)[0]
            for i in idxs:
                inc = min(diff, 1.0 - vec[i])
                vec[i] += inc
                diff -= inc
                if diff <= 1e-12:
                    break
        else:
            idxs = np.where(vec > 0.0)[0]
            for i in idxs:
                dec = min(-diff, vec[i])
                vec[i] -= dec
                diff += dec
                if diff >= -1e-12:
                    break
    return vec
```

def generate_random_feasible(n, target_sum, rng):
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    # fix residual sum diff
    diff = target_sum - np.sum(x)
    if diff > 0:
        # capacity = sum of (1 - x_i) where x_i < 1
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

def binary_hill_climb_directed(h_bin, dx, rng, time_limit):
    best = h_bin.copy()
    best_val = compute_c5_vec(best.astype(float), dx)
    n = len(best)
    while time.time() < time_limit:
        overlaps = np.correlate(best.astype(float), 1.0 - best.astype(float), mode='full')
        max_idx = np.argmax(overlaps)
        k = max_idx - (n - 1)  # shift
        # compute donor and recipient indices
        if k >= 0:
            donor = np.where((best[:-k] == 1) & (best[k:] == 0))[0]
            recipient = np.where((best[:-k] == 0) & (best[k:] == 1))[0]
        else:
            kk = -k
            donor = np.where((best[kk:] == 1) & (best[:-kk] == 0))[0] + kk
            recipient = np.where((best[kk:] == 0) & (best[:-kk] == 1))[0] + kk
        if donor.size == 0 or recipient.size == 0:
            # fallback random
            ones = np.flatnonzero(best)
            zeros = np.flatnonzero(1 - best)
            if ones.size == 0 or zeros.size == 0:
                break

            j = rng.choice(zeros)

            i = rng.choice(donor)

        cand[i] = 0
        cand[j] = 1

            best_val = cand_val

    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.02
    n = len(h)
    while time.time() < time_limit:
        # compute overlaps
        overlaps = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(overlaps)
        k = max_idx - (n - 1)
        # donor indices: i where cur_h[i]>eps and (i+k) in range and cur_h[i+k] < eps
        if k >= 0:
            donor_mask = (cur_h[:n - k] > 1e-8) & (cur_h[k:] < 1e-8)
            donor_idx = np.where(donor_mask)[0]
            recipient_mask = (cur_h[:n - k] < 1 - 1e-8) & (cur_h[k:] > 1 - 1e-8)
            recipient_idx = np.where(recipient_mask)[0]

            recipients = np.where(cur_h < 1 - 1e-8)[0]

            j = rng.choice(recipients)

            i = rng.choice(donor_idx)
