def compute_swap_delta(h, i, j):
    n = len(h)
    size = 2*n - 1
    delta = np.zeros(size, dtype=float)

    # i contribution (delta_h = -1)
    # plus part (i+k)
    k_start = -i   # smallest k such that i+k >=0 => k >= -i
    k_end = n-1 - i   # largest k such that i+k <= n-1 => k <= n-1 - i
    # k range inclusive: -i ... n-1-i
    # corresponding indices in delta:
    start_idx = k_start + n - 1
    end_idx = k_end + n - 1
    # assign
    if start_idx <= end_idx:
        # slice of length end_idx - start_idx + 1
        ks = np.arange(start_idx, end_idx+1)
        # values = -(1 - h[i + (k)]) where k = ks - (n-1)
        delta[ks] -= (1 - h[i + ks - (n-1)])

    # minus part (i-k)
    # i - k in [0,n-1] => k in [i - (n-1), i]
    k_start2 = i - (n-1)
    k_end2 = i
    start_idx2 = k_start2 + n - 1
    end_idx2 = k_end2 + n - 1
    if start_idx2 <= end_idx2:
        ks2 = np.arange(start_idx2, end_idx2+1)
        # values for each k: + h[i - k] = h[i - (k)] = h[i - (ks2 - (n-1))]

    # j contribution (delta_h = +1)
    # plus part (j+k)
    k_start = -j
    k_end = n-1 - j
    start_idx = k_start + n - 1

def exhaustive_hill_climb(h):
    n = len(h)
    cur_corr = np.correlate(h, 1-h, mode='full')
    cur_max = cur_corr.max()
    improved = True
    while improved:
        improved = False
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        for i in ones:
            for j in zeros:
                delta = compute_swap_delta(h, i, j)
                new_max = (cur_corr + delta).max()
                if new_max < cur_max - 1e-12:
                    # Accept swap
                    h[i], h[j] = 0.0, 1.0
                    cur_corr += delta
                    cur_max = new_max
                    improved = True
                    # break out to restart scanning
                    break
            if improved:
                break
    return h, cur_corr.max()

import time
import math
import numpy as np

def compute_c5(arr):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Binary Thue‑Morse sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    """Binary Rudin‑Shapiro sequence (+1/-1→1/0) of length n."""
    a = np.array([1], dtype=int)
    while a.size < n:
        a = np.concatenate([a, a * ((-1) ** np.arange(a.size) % 2)])
    a = a[:n]
    return ((a + 1) // 2).astype(int)

def adjust_to_target(vec, target, rng):
    """Flip bits of binary vector to achieve exact sum = target."""
    cur = int(vec.sum())
    diff = target - cur
    if diff == 0:
        return
    idxs = np.arange(len(vec))
    rng.shuffle(idxs)
    for idx in idxs:
        if diff == 0:
            break
        if diff > 0 and vec[idx] == 0:
            vec[idx] = 1
            diff -= 1
        elif diff < 0 and vec[idx] == 1:
            vec[idx] = 0
            diff += 1

def compute_swap_delta(h, i, j):
    """Delta of correlation array (full mode) caused by swapping 1 at i with 0 at j."""
    n = len(h)
    size = 2 * n - 1
    delta = np.zeros(size, dtype=float)

    # helper to add slice contributions
    def add_slice(start_k, end_k, sign, base_idx):
        # k = start_k ... end_k inclusive
        # corresponding indexes in delta array:
        s = start_k + n - 1
        e = end_k + n - 1
        if s > e:
            return
        ks = np.arange(s, e + 1, dtype=int)
        indices = base_idx + ks - (n - 1)  # i + k or j + k etc.
        delta[ks] += sign * (1 - h[indices])  # for (1 - h[i+k]) part

    def add_slice_minus(start_k, end_k, sign, base_idx):
        # corresponds to + h[i - k] or - h[j - k]
        s = start_k + n - 1
        e = end_k + n - 1
        if s > e:
            return
        ks = np.arange(s, e + 1, dtype=int)
        indices = base_idx - (ks - (n - 1))  # i - k or j - k
        delta[ks] += sign * h[indices]

    # i: delta_h = -1
    # (1 - h[i+k]) term: sign = -1
    add_slice(-i, n - 1 - i, -1, i)
    # h[i - k] term: sign = +1
    add_slice_minus(i - (n - 1), i, +1, i)

    # j: delta_h = +1
    # (1 - h[j+k]) term: sign = +1
    add_slice(-j, n - 1 - j, +1, j)
    # h[j - k] term: sign = -1
    add_slice_minus(j - (n - 1), j, -1, j)

    # zero lag (k=0) should stay unchanged for binary swaps
    delta[n - 1] = 0.0
    return delta

def exhaustive_hill_climb(h):
    """Deterministic improvement by exhaustive pair swaps (binary)."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    improved = True
    while improved:
        improved = False
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        for i in ones:
            for j in zeros:
                delta = compute_swap_delta(h, i, j)
                new_max = (corr + delta).max()
                if new_max + 1e-12 < cur_max:
                    # apply swap
                    h[i] = 0.0
                    h[j] = 1.0
                    corr += delta
                    cur_max = new_max
                    improved = True
                    break
            if improved:
                break
    return h, cur_max

def random_binary(n, target, rng):
    vec = np.zeros(n, dtype=float)
    vec[rng.choice(n, target, replace=False)] = 1.0
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # --------- initial vector ----------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        # binarize and adjust sum
        h0 = (h0 >= 0.5).astype(float)
        target_sum = n // 2
        adjust_to_target(h0, target_sum, rng)
    else:
        n = 80
        target_sum = n // 2
        h0 = random_binary(n, target_sum, rng)

    dx = 2.0 / n
    best_h = h0.copy()
    best_corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_raw = best_corr.max()
    best_c5 = best_raw * dx

    # --------- auxiliary seeds ----------
    seeds = [best_h.copy()]
    tm = thue_morse_seq(n).astype(float)
    adjust_to_target(tm, target_sum, rng)
    seeds.append(tm.copy())
    rs = rudin_shapiro_seq(n).astype(float)
    adjust_to_target(rs, target_sum, rng)
    seeds.append(rs.copy())

    # --------- annealing parameters ----------
    init_temp = 5e-3
    final_temp = 1e-7
    restart_prob = 0.015
    target_c5 = 0.3808

    # start from a random seed
    cur_h = best_h.copy()
    cur_corr = best_corr.copy()
    cur_raw = best_raw

    while time.time() - start_time < budget_s - 0.5:
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        T = init_temp * (1 - frac) + final_temp * frac

        # occasional restart
        if rng.random() < restart_prob:
            cur_h = rng.choice(seeds).copy()
            # perturb a few swaps to explore neighbourhood
            perturb = rng.integers(1, 4)
            for _ in range(perturb):
                ones = np.flatnonzero(cur_h > 0.5)
                zeros = np.flatnonzero(cur_h < 0.5)
                if ones.size and zeros.size:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    cur_h[i] = 0.0
                    cur_h[j] = 1.0
            cur_corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            cur_raw = cur_corr.max()
            continue

        # choose a random 1 and 0 to swap
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        delta = compute_swap_delta(cur_h, i, j)
        new_corr = cur_corr + delta
        new_raw = new_corr.max()

        d_raw = new_raw - cur_raw
        if d_raw <= 0 or rng.random() < math.exp(-d_raw / (T + 1e-20)):
            # accept swap
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            cur_corr = new_corr
            cur_raw = new_raw
            # update best if improved
            new_c5 = new_raw * dx
            if new_c5 + 1e-12 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                best_raw = new_raw
                if best_c5 <= target_c5:
                    break

    # Use exhaustive hill climbing from the best binary solution found.
    if time.time() - start_time < budget_s - 0.2:
        refined_h, refined_raw = exhaustive_hill_climb(best_h.copy())
        refined_c5 = refined_raw * dx
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-10:
        # adjust by tiny amount uniformly (should not happen for binary)
        best_h += (target_sum - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
