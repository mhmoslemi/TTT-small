import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given discretized step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def enforce_sum(h, M):
    """Adjust h so that sum(h) == M while keeping 0≤h≤1."""
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) <= 1e-12:
        return h
    h = h + diff / h.size
    h = np.clip(h, 0.0, 1.0)
    cur2 = np.sum(h)
    if abs(cur2 - M) <= 1e-12:
        return h
    deficit = M - cur2
    if deficit > 0:
        order = np.argsort(h)
        for i in order:
            inc = min(deficit, 1.0 - h[i])
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    else:
        excess = -deficit
        order = np.argsort(-h)
        for i in order:
            dec = min(excess, h[i])
            h[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    return h

def generate_initial_h(n, rng):
    """Return a random feasible h with sum = n//2."""
    M = n // 2
    method = rng.integers(3)
    if method == 0:
        # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:
        # continuous random scaled to sum M
        h = rng.random(n)
        h = h / np.sum(h) * M
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, M)
    else:
        # linear decreasing shape, then repair
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, M)
    return h

def simulated_annealing_h(h0, time_budget, rng):
    """Simple SA that preserves sum and bounds, returns best h found."""
    n = h0.size
    h = h0.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    T0 = 0.02  # initial temperature

    while time.time() - start < time_budget - 1e-6:
        elapsed = time.time() - start
        t = T0 * (1.0 - elapsed / time_budget)
        if t < 1e-7:
            t = 1e-7

        i, j = rng.choice(n, size=2, replace=False)
        # random direction of transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        max_transfer = min(1.0 - h[dst], h[src])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        # apply move
        h[src] -= delta
        h[dst] += delta
        new_c5 = compute_c5(h)
        dE = new_c5 - cur_c5
        if dE <= 0.0 or rng.random() < np.exp(-dE / t):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[dst] -= delta
            h[src] += delta

        # occasional tiny random perturbation to escape plateaus
        if rng.random() < 0.001:
            idx1 = rng.integers(n)
            idx2 = rng.integers(n - 1)
            if idx2 >= idx1:
                idx2 += 1
            old1, old2 = h[idx1], h[idx2]
            new1 = rng.random()
            diff = new1 - old1
            if 0.0 <= old2 - diff <= 1.0:
                h[idx1] = new1
                h[idx2] = old2 - diff
                cur_c5 = compute_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
    return best_h, best_c5
