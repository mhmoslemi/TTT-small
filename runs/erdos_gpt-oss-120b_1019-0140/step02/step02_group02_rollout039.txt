def project_to_capped_simplex(v, target_sum):
    # Clip v to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # If sum is already close to target, return v
    cur = v.sum()
    if abs(cur - target_sum) < 1e-12:
        return v
    # Bisection to find lambda
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(60):
        lam = (low + high) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            low = lam
        else:
            high = lam
    v_proj = np.clip(v - (low+high)/2.0, 0.0, 1.0)
    # After projecting, due to rounding may not be exact; adjust small diff:
    diff = target_sum - v_proj.sum()
    if abs(diff) > 1e-8:
        # distribute diff to elements not at bounds
        mask = (v_proj > 1e-12) & (v_proj < 1.0-1e-12)
        if np.any(mask):
            # add diff equally among masked positions
            v_proj[mask] += diff / mask.sum()
            v_proj = np.clip(v_proj, 0.0, 1.0)
        else:
            # fallback: adjust first element
            v_proj[0] = np.clip(v_proj[0] + diff, 0.0, 1.0)
    return v_proj

    dx = 2.0 / n
    best_h = None
    best_val = np.inf

    # function to compute C5 from h

        conv = np.convolve(h, h, mode='full')
        overl = target - conv
        return np.max(overl) * dx

    # projection function defined above

def project_to_capped_simplex(v, target):
    # Clip v into [0,1]
    v = np.clip(v, 0.0, 1.0)
    # If sum close to target, return
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # Bisection for lambda
    low = np.min(v) - 1.0   # ensures all values after subtract >= 1, clipped to 1
    high = np.max(v)        # ensures all values after subtract <=0, clipped to 0
    for _ in range(60):
        lam = (low + high) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    v_proj = np.clip(v - lam, 0.0, 1.0)
    # Fix minor difference due to rounding:
    diff = target - v_proj.sum()
    if abs(diff) > 1e-12:
        # distribute diff to non-bound elements if possible
        free = (v_proj > 1e-12) & (v_proj < 1.0 - 1e-12)
        if np.any(free):
            v_proj[free] += diff / free.sum()
            v_proj = np.clip(v_proj, 0.0, 1.0)
        else:
            # fallback to adjusting first element
            v_proj[0] = np.clip(v_proj[0] + diff, 0.0, 1.0)
    return v_proj

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def local_optimize(h_start, deadline):
    h = h_start.copy()
    n = h.shape[0]
    target = n / 2.0
    T = 10.0
    T_min = 0.01
    decay = 0.995
    eta = 0.2
    best_h = h.copy()
    best_val = compute_c5(h)
    while time.time() < deadline - 0.01 and T > T_min:
        # correlation
        overl = np.correlate(h, 1.0 - h, mode='full')
        max_overl = overl.max()
        # weight
        # to avoid large exponent, subtract max_overl for stability
        shift = (overl - max_overl) / T
        w = np.exp(shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient components
        # conv w_rev * (1 - h)
        w_rev = w[::-1]
        conv1 = np.convolve(w_rev, 1.0 - h, mode='full')
        grad_pos = conv1[n-1:n-1+n]   # slice length n
        conv2 = np.convolve(w, h, mode='full')
        grad_neg = conv2[n-1:n-1+n]
        grad = grad_pos - grad_neg
        # gradient step
        h_new = h - eta * grad
        # projection
        h_new = project_to_capped_simplex(h_new, target)
        # evaluate
        cur_val = compute_c5(h_new)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h_new.copy()
        # accept step (always)
        h = h_new
        # decay temperature
        T *= decay
        # optionally adapt eta: maybe increase when T reduces? Not needed.
        if best_val < 0.3808:
            break
    return best_h, best_val

import time
import numpy as np

def compute_c5(h):
    """Return C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_capped_simplex(v, target):
    """Euclidean projection of v onto {x | 0<=x_i<=1, sum x_i = target}."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    lo = np.min(v) - 1.0   # ensures all clipped values become 1
    hi = np.max(v)         # ensures all clipped values become 0
    for _ in range(60):
        lam = 0.5 * (lo + hi)
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = 0.5 * (lo + hi)
    proj = np.clip(v - lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if np.any(free):
            idx = np.where(free)[0][0]
            proj[idx] = np.clip(proj[idx] + diff, 0.0, 1.0)
        else:
            proj[0] = np.clip(proj[0] + diff, 0.0, 1.0)
    return proj

def generate_start(n, target, rng):
    """Create an initial h with the required sum; upsample if a parent exists."""
    if "initial_h_values" in globals():
        old_h = np.array(initial_h_values, dtype=float)
        old_n = old_h.shape[0]
        x_old = np.linspace(0, 2, old_n, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        new_h = np.interp(x_new, x_old, old_h)
        new_h = np.clip(new_h, 0.0, 1.0)
        return project_to_capped_simplex(new_h, target)
    else:
        v = rng.random(n)
        v = np.clip(v, 0.0, 1.0)
        return project_to_capped_simplex(v, target)

def local_optimize(h_start, deadline):
    """Smooth‑max gradient descent with projection."""
    h = h_start.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)

    T = 10.0               # initial temperature for smooth max
    T_min = 0.01
    decay = 0.995
    eta = 0.5              # base step size

    while time.time() < deadline - 0.01 and T > T_min:
        # correlation for smooth max
        overl = np.correlate(h, 1.0 - h, mode="full")
        max_overl = overl.max()
        # softmax weights
        shift = (overl - max_overl) / T
        w = np.exp(shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient: conv(w_rev, 1-h) - conv(w, h)
        w_rev = w[::-1]
        conv_pos = np.convolve(w_rev, 1.0 - h, mode="full")[n - 1 : n - 1 + n]
        conv_neg = np.convolve(w, h, mode="full")[n - 1 : n - 1 + n]
        grad = conv_pos - conv_neg

        # gradient step with projection
        h_new = h - eta * grad
        h_new = project_to_capped_simplex(h_new, target)
        cur_val = compute_c5(h_new)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h_new.copy()
            if best_val < 0.3808:
                break

        h = h_new
        T *= decay
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    n = 200                     # number of discretisation points
    target = n / 2.0

    best_h = None
    best_val = np.inf

    # multiple random restarts within the time budget
    while time.time() < deadline - 2.0:
        h0 = generate_start(n, target, rng)
        # quick check of the starting value
        val0 = compute_c5(h0)
        if val0 < best_val:
            best_val = val0
            best_h = h0.copy()
            if best_val < 0.3808:
                break

        h_opt, val_opt = local_optimize(h0, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # final safety projection and evaluation
    if best_h is None:
        # fall back to a simple feasible vector
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_capped_simplex(best_h, target)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
