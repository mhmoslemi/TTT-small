def project_sum_and_bounds(x, target, tol=1e-8, max_iter=1000):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral C5 = dx * max correlation of h and (1-h)."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate returns length 2n-1 array (full convolution)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def project_sum_and_bounds(x, target, tol=1e-12, max_iter=100):
    """
    Clip entries to [0,1] and adjust so that sum(x) == target (within tol).
    Simple iterative repair that preserves feasibility.
    """
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) <= tol:
        return x
    # First uniform correction
    x += diff / len(x)
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) <= tol:
        return x
    # Iterative elementwise fix
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to increase some entries
            for i in range(len(x)):
                if diff <= 0:
                    break
                if x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
        else:
            # need to decrease some entries
            for i in range(len(x)):
                if diff >= 0:
                    break
                if x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    # Final clipping (should already be feasible)
    return np.clip(x, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Simulated annealing / stochastic search for a step function h on [0,2]
    minimizing C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    else:
        n = 80
        h = rng.random(n)
    target_sum = n / 2.0
    h = project_sum_and_bounds(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    curr_h = h.copy()
    curr_c5 = best_c5

    # Annealing schedule parameters
    T0 = 0.02                # initial temperature
    sigma0 = 0.12            # initial std for global perturbations
    p_global = 0.025         # probability of a global move
    no_improve = 0
    max_no_improve = 20000   # after which we re‑seed from best

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Linear temperature decay
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        # Generate a candidate
        if rng.random() < p_global:
            # Global Gaussian perturbation (zero‑sum)
            sigma = sigma0 * max(0.0, 1.0 - elapsed / budget_s)
            eps = rng.normal(scale=sigma, size=n)
            eps -= eps.mean()
            cand = curr_h + eps
        else:
            # Local two‑index delta move
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            # Compute allowable delta range so that both stay in [0,1]
            delta_min = max(-curr_h[i], -(1.0 - curr_h[j]))
            delta_max = min(1.0 - curr_h[i], curr_h[j])
            if delta_max <= delta_min:
                continue  # cannot move; skip this iteration
            # Scale step size with temperature (smaller as T drops)
            step_frac = 0.05 + 0.45 * (T / T0)  # between 0.05 and 0.5
            delta = delta_min + rng.random() * (delta_max - delta_min) * step_frac
            cand = curr_h.copy()
            cand[i] += delta
            cand[j] -= delta

        # Project back to feasible region
        cand = project_sum_and_bounds(cand, target_sum)

        # Evaluate
        cand_c5 = compute_c5(cand)

        # Acceptance rule (Metropolis)
        if cand_c5 < curr_c5 or rng.random() < np.exp(-(cand_c5 - curr_c5) / (T + 1e-16)):
            curr_h = cand
            curr_c5 = cand_c5
            # Update best if improved
            if cand_c5 < best_c5 - 1e-12:
                best_c5 = cand_c5
                best_h = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # If stuck for long, re‑seed around the best solution
        if no_improve >= max_no_improve:
            eps = rng.normal(scale=0.03, size=n)
            eps -= eps.mean()
            curr_h = project_sum_and_bounds(best_h + eps, target_sum)
            curr_c5 = compute_c5(curr_h)
            no_improve = 0

    # Final safety projection and evaluation
    best_h = project_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
