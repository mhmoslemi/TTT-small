def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rnd = np.random.default_rng(seed)
    # Determine dimensionality
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rnd.random(n)
    target = n/2.0
    h0 = project_to_box_sum(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # parameters
    phase1_time = budget_s * 0.2
    phase2_time = budget_s - phase1_time
    # Phase1: random restarts
    while time.time() - start < start + phase1_time:
        # generate random feasible start
        h = rnd.random(n)
        h = project_to_box_sum(h, target)
        # run subgradient for some iterations
        max_iter = 300
        for it in range(max_iter):
            # compute inner products
            inner_all = np.correlate(h, h, mode='full')
            # extract positive lags
            # lags = 1..n-1 -> indices n-1+lag
            # compute min inner
            # We'll ignore lag=0
            lags_inner = inner_all[n:]
            # find min
            k_min = np.argmin(lags_inner) + 1  # +1 shift offset (since lags start at 1)
            # gradient
            shift_plus = np.concatenate([h[k_min:], np.zeros(k_min)])
            shift_minus = np.concatenate([np.zeros(k_min), h[:n - k_min]])
            grad = shift_plus + shift_minus
            # step size schedule
            step = 0.5 / np.sqrt(it + 1)  # arbitrary
            h_new = h + step * grad
            h_new = project_to_box_sum(h_new, target)
            h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if best_c5 < 0.3805:
            break

    # Phase2: Subgradient from best_h
    h = best_h.copy()
    cur_c5 = best_c5
    iter_cnt = 0
    while time.time() - start < budget_s - 0.5:
        # compute inner all
        inner_all = np.correlate(h, h, mode='full')
        lags_inner = inner_all[n:] # n = n points, include lag up to n-1
        k_min = np.argmin(lags_inner) + 1
        shift_plus = np.concatenate([h[k_min:], np.zeros(k_min)])
        shift_minus = np.concatenate([np.zeros(k_min), h[:n - k_min]])
        grad = shift_plus + shift_minus
        # step size: adapt
        step = 0.1 / np.sqrt(iter_cnt + 1)
        # maybe occasionally add random noise
        if rnd.random() < 0.01:
            noise = rnd.normal(scale=0.1, size=n)
            grad = grad + noise
        h_new = h + step * grad
        h_new = project_to_box_sum(h_new, target)
        h = h_new
        iter_cnt += 1
        # evaluate occasionally
        if iter_cnt % 10 == 0:
            c5 = compute_c5(h)
            if c5 < cur_c5 - 1e-12:
                cur_c5 = c5
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h.copy()
            # early break if good enough
            if best_c5 < 0.3804:
                break

    # final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    high = (hi - np.min(v)) + 5.0

        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def run(seed=42, budget_s=1100, **kwargs):

    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # initialise dimension and feasible start
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target = n / 2.0
    h0 = project_to_box_sum(h0, target)

    best_c5 = compute_c5(best_h)

    # Phase 1 : many random restarts with a few subgradient steps
    # -------------------------------------------------------------
