import time
import numpy as np

# ----- Helper functions -----
def compute_c5(h, dx):
    """Maximum overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target_sum):
    """Project vec onto the hyperplane sum(vec)=target_sum while staying in [0,1]."""
    v = np.clip(vec, 0.0, 1.0)
    cur = np.sum(v)
    diff = target_sum - cur
    # if already satisfied
    if abs(diff) < 1e-12:
        return v
    # try uniform adjustment
    v += diff / len(v)
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - np.sum(v)
    # distribute remaining diff greedily
    if diff > 0:
        idx = np.where(v < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(v > 0.0)[0]
        for i in idx:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return v

def generate_random_feasible(n, target_sum, rng):
    """Random vector in [0,1]^n with exact sum target_sum."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0.0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    # fix residual sum difference
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

def gradient_of_shift(h, shift):
    """Sub‑gradient of the overlap term for a given shift."""
    n = len(h)
    grad = np.zeros_like(h)
    if shift >= 0:
        fwd_len = n - shift
        if fwd_len > 0:
            grad[:fwd_len] += 1.0 - h[shift:shift + fwd_len]
            grad[shift:] -= h[:fwd_len]
    else:
        m = -shift
        fwd_len = n - m
        if fwd_len > 0:
            grad[m:] += 1.0 - h[:fwd_len]
            grad[:fwd_len] -= h[m:]
    return grad

    rng = np.random.default_rng(seed)

    # problem size
    if "initial_h_values" in globals():

    else:
        n = 80
        h0 = None

    dx = 2.0 / n
    target_sum = n / 2.0

    # initial best solution
    if h0 is not None:
        best_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        best_h = generate_random_feasible(n, target_sum, rng)
    best_val = compute_c5(best_h, dx)

    # ----- Gradient‑descent local search -----
    def local_descent(start_h, max_iters=200, init_step=0.5):
        h = start_h.copy()
        step = init_step
        val = compute_c5(h, dx)
        for _ in range(max_iters):
            if time.time() - start_time >= budget_s:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = np.argmax(corr)
            shift = idx - (n - 1)
            grad = gradient_of_shift(h, shift)

            # backtracking line search
            improved = False
            alpha = step
            while alpha > 1e-7:
                h_new = h - alpha * grad
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)

                return x[-1] - overl * dx
            return con

        constraints = [{"type": "eq", "fun": sum_con}]

            constraints.append({"type": "ineq", "fun": make_overlap_con(k)})

    best_val = compute_c5(best_h, dx)
