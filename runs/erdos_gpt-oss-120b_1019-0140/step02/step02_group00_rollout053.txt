def upsample(arr, new_n):

    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1 = h shifted left by shift (i.e., h[i+shift] for i)
            term1 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad
```

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9, deadline=None):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while it < n_iter:
        if deadline is not None and time.time() >= deadline:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts achieving max (within eps)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it + 1)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def hill_climb_binary(h0, max_iters, deadline, rng):
    h = h0.copy()
    best_score = compute_c5(h)
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    it = 0
    while it < max_iters:
        if time.time() >= deadline:
            break
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # create swapped version
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        if new_score < best_score - 1e-12:
            # accept move
            h = h_new
            best_score = new_score
            # update index lists: swap i and j
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
        it += 1
    return h, best_score
```

def hill_climb_binary_anneal(h0, max_iters, deadline, rng, temp0=0.001):

    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        if not ones or not zeros:
            break

        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0

            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)

import time
import numpy as np
import math

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    low = np.min(v) - hi
    high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        total = proj.sum()
        if abs(total - s) < 1e-12:
            return proj
        if total > s:
            low = lam
        else:
            high = lam
    return np.clip(v - (low + high) * 0.5, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    v[idx] = 1.0
    return v

def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2 = np.zeros(n, dtype=h.dtype)
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9, deadline=None):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while it < n_iter:
        if deadline is not None and time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it + 1)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def hill_climb_binary_anneal(h0, max_iters, deadline, rng, temp0=5e-4):
    """Binary hill climbing with a simple annealing schedule."""
    h = h0.copy()
    best_score = compute_c5(h)
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        temp = max(temp0 * (1.0 - it / max_iters), 0.0)
        accept = False
        if new_score < best_score - 1e-12:
            accept = True
        elif temp > 0.0:
            prob = math.exp(-(new_score - best_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            h = h_new
            best_score = new_score
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
    return h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    n = 256  # target resolution
    starts = []

    # Use upsampled initial construction if provided
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size > 0:
            up = upsample(init_full, n)
            idx = np.argpartition(-up, n // 2)[: n // 2]
            binary = np.zeros(n, dtype=float)
            binary[idx] = 1.0
            starts.append(binary)

    # Thue–Morse binary start
    starts.append(thue_morse(n))

    # A few random balanced binary starts
    for _ in range(3):
        starts.append(random_binary_balanced(n, rng))

    best_h = None
    best_score = math.inf

    # Phase 1: annealed binary hill climbing
    for h0 in starts:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        max_iters = int(min(20000, remaining * 1500))
        h_opt, sc_opt = hill_climb_binary_anneal(h0, max_iters, deadline, rng, temp0=5e-4)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
            if best_score <= 0.3808:
                break

    # Fallback initialisation if nothing succeeded
    if best_h is None:
        best_h = random_binary_balanced(n, rng)
        best_score = compute_c5(best_h)

    # Phase 2: continuous refinement via momentum gradient descent
    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        est_iters = int(min(50000, max(5000, remaining * 500)))
        h_ref, sc_ref = gradient_descent_momentum(
            best_h.astype(float),
            n_iter=est_iters,
            step0=0.4,
            beta=0.9,
            deadline=deadline
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Final projection to enforce constraints precisely
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
