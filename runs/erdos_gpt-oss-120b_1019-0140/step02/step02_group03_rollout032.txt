    start = time.time()
    rng = np.random.default_rng(seed)

        n = h.size

    # project onto constraints
    # using project_to_box_sum defined (or implement own)
    h = project_to_box_sum(h, target_sum)
    dx = 2.0 / n

    def compute_c5_vec(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return np.max(corr) * dx

    cur_c5 = compute_c5_vec(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # SA parameters
    T0 = 0.02  # start temperature (maybe choose based on initial C5)
    # We could set T0 = cur_c5 * 0.05 or something.
    T = T0
    decay = 0.9995  # per iteration
    step_factor = 0.5  # initial max delta fraction of allowable

    # stats for dynamic adapt
    it = 0
    accepted = 0
    while time.time() - start < budget_s * 0.94:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # randomly decide direction: inc_i_dec_j (increase i)
        if rng.random() < 0.5:
            # increase i, decrease j
            delta_max = min(1.0 - h[i], h[j])
            if delta_max <= 0:
                continue
            delta = rng.random() * delta_max * step_factor
            h_cand = h.copy()
            h_cand[i] += delta
            h_cand[j] -= delta
        else:
            # increase j, decrease i
            delta_max = min(1.0 - h[j], h[i])
            if delta_max <= 0:
                continue
            delta = rng.random() * delta_max * step_factor
            h_cand = h.copy()
            h_cand[j] += delta
            h_cand[i] -= delta
        # compute c5 candidate
        cand_c5 = compute_c5_vec(h_cand)
        # acceptance
        if cand_c5 < cur_c5 or rng.random() < np.exp(-(cand_c5 - cur_c5) / max(T, 1e-12)):
            h = h_cand
            cur_c5 = cand_c5
            accepted += 1
            # keep best
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = h.copy()
        # temperature update
        T *= decay
        it += 1
        # adjust step_factor heuristically every 1000 it
        if it % 2000 == 0:
            # Decay step_factor slowly
            step_factor *= 0.9
            # Optionally reset T if too low
            if T < 1e-8:
                T = T0
        # if time low break
        if time.time() - start > budget_s * 0.94:
            break

    # after SA, local search (greedy)
    # Use random pair improvement only accept if improves
    cur_h = best_h.copy()
    cur_c5 = best_c5
    # we can use decreasing step factor schedule
    local_step_factor = 0.2
    while time.time() - start < budget_s * 0.98:
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # direction both ways
        # maybe try inc i, dec j
        # compute delta_max
        delta_max = min(1.0 - cur_h[i], cur_h[j])
        if delta_max <= 0:
            continue
        delta = rng.random() * delta_max * local_step_factor
        cand = cur_h.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_c5 = compute_c5_vec(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5

    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5_vec(best_h)

    return best_h, best_c5, n

import time, numpy as np

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    corr[n-1] = -np.inf
    return np.max(corr) * dx

def max_overlap_and_lag(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    corr[n-1] = -np.inf
    idx = np.argmax(corr)
    val = corr[idx] * dx
    lag = idx - (n - 1)
    return val, lag

def compute_gradient(h, k):
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    if k >= 0:
        i1 = np.arange(n - k)
        g[i1] += 1.0 - h[i1 + k]
        i2 = np.arange(k, n)
        g[i2] -= h[i2 - k]
    else:
        kp = -k
        i1 = np.arange(n - kp)
        g[i1] -= h[i1 + kp]
        i2 = np.arange(kp, n)
        g[i2] += 1.0 - h[i2 - kp]
    return g

import time
import numpy as np

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing sum(v)=target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)


def compute_c5(h):
    """Maximum overlap integral (excluding zero shift)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    corr[n - 1] = -np.inf  # exclude k=0
    return np.max(corr) * dx


def max_overlap_and_lag(h):
    """Return (max value, lag) for the overlap, ignoring k=0."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    corr[n - 1] = -np.inf
    idx = np.argmax(corr)
    val = corr[idx] * dx
    lag = idx - (n - 1)
    return val, lag


def compute_gradient(h, k):
    """Gradient of overlap(k) w.r.t. h (for a fixed lag k)."""
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    if k >= 0:
        i1 = np.arange(n - k)
        g[i1] += 1.0 - h[i1 + k]
        i2 = np.arange(k, n)
        g[i2] -= h[i2 - k]
    else:
        kp = -k
        i1 = np.arange(n - kp)
        g[i1] -= h[i1 + kp]
        i2 = np.arange(kp, n)
        g[i2] += 1.0 - h[i2 - kp]
    return g
