import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper Functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project vector v onto [lo,hi]^n while forcing sum(v) == target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# ---------- Constraint Classes ----------
class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    """Run a single SLSQP optimisation starting from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Choose discretisation ----
    n = 120  # more points for finer control, still cheap
    target_sum = n / 2.0

    # ---- Initialise h ----
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        m = h_init.size
        # linear interpolation from m points to n points
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, h_init)
    else:
        h0 = rng.random(n)
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---- Constraints for SLSQP ----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # ---- Phase 1: SLSQP restarts (≈65% of budget) ----
    phase1_deadline = start_time + budget_s * 0.65
    while time.time() < phase1_deadline:
        # Random start, sometimes perturb the best known one
        if rng.random() < 0.2:
            h_start = best_h + 0.1 * (rng.random(n) - 0.5)
        else:
            h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, target_sum)
        t_start = compute_c5(h_start)

        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800)
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # enforce exact sum after optimisation
        if abs(h_opt.sum() - target_sum) > 1e-9:
            h_opt = project_to_box_sum(h_opt, target_sum)

        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            if best_c5 < 0.3806:  # early exit if already very good
                break

    # ---- Phase 2: Gradient‑based hill climbing / simulated annealing ----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    step = 0.2  # initial step magnitude
    while True:
        now = time.time()
        if now - start_time > budget_s - 0.6:
            break

        # Determine shift giving the current maximum overlap
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        # Gradient of the active overlap term (subgradient for the max)
        g = np.ones(n)
        if k_star >= 0:
            k = k_star
            if k > 0:
                g[:n - k] -= cur_h[k:]
                g[k:]      -= cur_h[:n - k]
        else:
            kk = -k_star
            g[kk:]      -= cur_h[:n - kk]
            g[:n - kk] -= cur_h[kk:]

        # Take a projected step
        cand = cur_h - step * g
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)

        if cand_c5 + 1e-12 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                if best_c5 < 0.3808:
                    break
            step = min(step * 1.05, 1.0)  # accelerate when successful
        else:
            step *= 0.7
            if step < 1e-4:
                # occasional random perturbation to escape stagnation
                cand = best_h + 0.05 * (rng.random(n) - 0.5)
                cand = project_to_box_sum(cand, target_sum)
                cand_c5 = compute_c5(cand)
                if cand_c5 < cur_c5:
                    cur_h, cur_c5 = cand, cand_c5
                step = 0.2

    # ---- Optional final polishing with a short SLSQP run ----
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5.0:
        final_res = slsqp_optimize(best_h, best_c5, sum_con, overlap_cons, maxiter=200)
        if final_res.success:
            h_tmp = np.clip(final_res.x[:n], 0.0, 1.0)
            if abs(h_tmp.sum() - target_sum) > 1e-9:
                h_tmp = project_to_box_sum(h_tmp, target_sum)
            c5_tmp = compute_c5(h_tmp)
            if c5_tmp + 1e-12 < best_c5:
                best_h, best_c5 = h_tmp, c5_tmp

    # ---- Final safety projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
