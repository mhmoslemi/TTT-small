def gradient_overlap(k, h):
    n = len(h)
    g = np.zeros_like(h)
    if k >= 0:
        # term from first factor: (1 - h[i+k]) for i <= n-1-k
        # j = i
        g[: n - k] += 1 - h[k: ]  # since for j from 0 to n-1-k, h[j+k] is h[k to end]
        # term from second factor: -h[i] for i = j - k, when j >= k
        g[k: ] -= h[: n - k]
    else:
        kk = -k
        # k negative: overlap uses shift left
        # first term: (1 - h[j-kk]) for j >= kk
        g[kk: ] += 1 - h[: n - kk]
        # second term: -h[j+kk] for j + kk <= n-1 => j <= n-1-kk
        g[: n - kk] -= h[kk: ]
    return g

import time
import numpy as np


def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Project vector to the simplex with bounds [0,1] and exact sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible vector (values in [0,1] and exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def gradient_overlap(k, h):
    """Gradient of raw overlap (without dx) for shift k."""
    n = len(h)
    g = np.zeros_like(h)
    if k >= 0:
        # term (1 - h[i+k]) for i=0..n-1-k
        g[: n - k] += 1.0 - h[k:]
        # term -h[i] for i=k..n-1 (i = j - k)
        g[k:] -= h[: n - k]
    else:
        kk = -k
        # term (1 - h[i-kk]) for i=kk..n-1
        g[kk:] += 1.0 - h[: n - kk]
        # term -h[i] for i=0..n-kk-1 (i = j + kk)
        g[: n - kk] -= h[kk:]
    return g


def binary_hill_climb(h_bin, cur_val, dx, time_limit, rng):
    """Simple swap hill‑climbing on a binary vector."""
    best_h = h_bin.copy()
    best_val = cur_val
    n = len(h_bin)
    while time.time() < time_limit:
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best_h = cand
            best_val = cand_val
            if best_val < 0.3808:
                break
    return best_h.astype(float), best_val

    # Main search loop
    while time.time() - start_time < budget_s - 5.0:
        # random restart
        h = generate_random_feasible(n, target_sum, rng)

        for it in range(max_iter_per_restart):
            now = time.time()
            if now - start_time > budget_s - 5.0:
                break

            raw = np.correlate(h, 1.0 - h, mode="full")
            max_idx = int(np.argmax(raw))
            max_raw = raw[max_idx]
            cur_val = max_raw * dx

            if cur_val < best_val - 1e-12:
                best_h = h.copy()
                best_val = cur_val
                if best_val < 0.3808:
                    # Excellent result – stop early
                    break

            # shift corresponding to the worst overlap
            k = max_idx - (n - 1)

            # gradient descent step
            g = gradient_overlap(k, h)
            lr = learning_base * (1.0 - it / max_iter_per_restart) + 0.001
            h = h - lr * g
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)

        if best_val < 0.3808:
            break

    # Binary refinement (if time permits)
    time_left = budget_s - (time.time() - start_time)
    if time_left > 8.0:
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argsort(-best_h)[: int(target_sum)]
        binary_h[top_idx] = 1
        bin_val = compute_c5_vec(binary_h.astype(float), dx)
        bin_h_opt, bin_val_opt = binary_hill_climb(
            binary_h,
            bin_val,
            dx,
            start_time + budget_s - 0.5,
            rng,
        )
        if bin_val_opt < best_val - 1e-12:
            best_h = bin_h_opt
            best_val = bin_val_opt

    # Final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
