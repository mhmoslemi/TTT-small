def subgrad_max(h, max_shifts):
    n = len(h)
    g = np.zeros(n)
    for k in max_shifts:
        offset = n - 1
        shift = k - offset  # actual shift value (negative to positive)
        if shift >= 0:
            tail = slice(0, n - shift)
            head = slice(shift, n)
        else:
            s = -shift
            tail = slice(s, n)
            head = slice(0, n - s)
        # tail part: derivative from h[tail] * (1 - h[head]) -> (1 - h[head])
        g[tail] += (1.0 - h[head])
        # head part: derivative from h[tail] * (1 - h[head]) includes -h[tail] for h[head]
        g[head] -= h[tail]
    # Now g is gradient of v_max. Since we want to minimize v_max, use -g as descent direction.
    # But need to preserve sum; we can subtract mean of g so it doesn't affect sum.
    g_centered = g - np.mean(g)
    return g_centered

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral multiplied by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (repair by uniform then greedy)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # first uniform adjustment
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # greedy fix
            for i in range(len(x)):
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
                if abs(diff) < 1e-12:
                    break
    return x
