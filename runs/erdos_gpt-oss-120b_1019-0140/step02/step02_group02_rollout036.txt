    dx = 2.0 / n
    # compute overlaps O as array length 2n-1
    corr = np.correlate(h, 1.0 - h, mode='full')

    # compute softmax weights
    # to avoid overflow:
    max_O = np.max(beta * O)
    exp_vals = np.exp(beta * O - max_O)

def smooth_descent(h0, betas, iters_per_beta, deadline, rng):

    best_h = h.copy()
    best_val = compute_c5(h)
    n = len(h)

            step = 0.5  # step size base

            for _ in range(5):

    return best_h, best_val

def compute_f_and_grad(h, beta):
    n = h.shape[0]
    dx = 2.0 / n
    # compute correlation full
    corr = np.correlate(h, 1.0 - h, mode='full')
    O = corr * dx  # array of length 2n-1
    # compute softmax weights
    max_bO = np.max(beta * O)
    exp_vals = np.exp(beta * O - max_bO)
    w = exp_vals / np.sum(exp_vals)  # shape (2n-1,)
    f = np.dot(w, O)  # smooth max
    # gradient
    g = np.zeros_like(h)
    # k = 0
    weight0 = w[n-1]
    g += weight0 * (1.0 - 2.0 * h)  # derivative of O0 = dx * sum h_i (1-h_i); derivative w.r.t h_i: dx*(1-2h_i). We'll multiply dx later.
    # other shifts
    for s in range(2*n-1):
        if s == n-1:
            continue
        weight = w[s]
        k = s - (n - 1)
        if k > 0:
            i_len = n - k
            # slices
            g[:i_len] += weight * (1.0 - h[k:k+i_len])
            g[k:k+i_len] += weight * (-h[:i_len])
        else:  # k < 0
            kk = -k
            i_len = n - kk
            g[kk:kk+i_len] += weight * (1.0 - h[:i_len])
            g[:i_len] += weight * (-h[kk:kk+i_len])
    g *= dx
    return f, g

def pgd_step(h, grad, target_sum, step_factor=0.5):
    norm = np.linalg.norm(grad)
    if norm < 1e-12:
        return h
    step = step_factor / norm
    h_new = project_to_feasible(h - step * grad, target_sum)
    return h_new

def project_to_feasible(y, target):
    # Clip to [0,1] after shifting via lambda.
    # binary search for lambda (shift) such that sum(clamp(y - lam,0,1)) = target.
    lo = -1.0  # lower bound for lambda; ensures y - lo large -> many >1
    hi = 1.0   # upper bound for lambda; ensures y - hi small -> many <0
    # Expand lo/hi until condition satisfied
    # Actually we need hi low enough to make sum <= target, lo high enough to make sum >= target.
    # Start with lo = -1e5, hi = 1e5.
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - hi, 0.0, 1.0)
    # adjust small diff
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # adjust free entries
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_feasible(y, target):
    """Project y onto the set {0≤x≤1, sum(x)=target} via bisection on a shift λ."""
    lo, hi = -1e5, 1e5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - hi, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
    return x

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    y = rng.random(n)
    return project_to_feasible(y, n / 2.0)

def compute_f_and_grad(h, beta):
    """
    Smoothed max overlap (softmax with temperature 1/beta) and its gradient.
    Returns (smooth_value, gradient, max_overlap).
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    O = corr * dx                     # overlaps for each shift
    max_O = O.max()
    # softmax weights
    bO = beta * O
    max_bO = np.max(bO)
    exp_vals = np.exp(bO - max_bO)
    w = exp_vals / np.sum(exp_vals)  # shape (2n-1,)

    smooth = np.dot(w, O)

    g = np.zeros_like(h)
    # contribution from shift 0
    weight0 = w[n - 1]
    g += weight0 * (1.0 - 2.0 * h)

    # contributions from other shifts
    for s in range(2 * n - 1):
        if s == n - 1:
            continue
        weight = w[s]
        k = s - (n - 1)
        if k > 0:
            i_len = n - k
            g[:i_len] += weight * (1.0 - h[k:k + i_len])
            g[k:k + i_len] += weight * (-h[:i_len])
        else:
            kk = -k
            i_len = n - kk
            g[kk:kk + i_len] += weight * (1.0 - h[:i_len])
            g[:i_len] += weight * (-h[kk:kk + i_len])

    g *= dx
    return smooth, g, max_O

def pgd_step(h, grad, target, step_factor):
    """One projected gradient descent step with step size scaled by step_factor."""
    norm = np.linalg.norm(grad)
    if norm < 1e-12:
        return h
    step = step_factor / norm
    h_new = project_to_feasible(h - step * grad, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        if abs(h0.sum() - target) > 1e-12:
            h0 = project_to_feasible(h0, target)
    else:
        n = 120
        target = n / 2.0
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    betas = [10, 30, 80, 200, 500]
    max_iters_per_beta = 2000
    restart = 0

    while time.time() < deadline - 0.5:
        # Choose a start point
        if restart == 0 and "initial_h_values" in globals():
            h = best_h.copy()
        else:
            if rng.random() < 0.3:
                h = best_h.copy()
                h = project_to_feasible(h + rng.normal(scale=0.05, size=n), target)
            else:
                h = random_feasible_vector(n, rng)

        # Gradient‑descent phase with increasing beta
        for beta in betas:
            it = 0
            while it < max_iters_per_beta and time.time() < deadline - 0.2:
                _, grad, _ = compute_f_and_grad(h, beta)
                step_factor = 0.5 / np.sqrt(it + 1)
                h = pgd_step(h, grad, target, step_factor)

                cur_val = compute_c5(h)
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
                    if best_val < 0.3808:
                        return best_h.astype(float), best_val, n
                it += 1
        restart += 1
        if restart >= 50:
            break

    # Final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - target) > 1e-9:
        best_h = project_to_feasible(best_h, target)

    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
