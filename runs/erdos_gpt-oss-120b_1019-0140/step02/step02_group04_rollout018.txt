    step = 0.05  # max transfer amount

def compute_c5_vec(h, dx):
    """Maximum overlap integral for vector h."""

def project_to_sum(vec, target_sum):
    """Scale vec into [0,1] while forcing exact sum."""
    v = np.clip(vec, 0.0, 1.0)
    cur = np.sum(v)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return v
    # uniform correction
    v += diff / len(v)
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - np.sum(v)
    n = len(v)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and v[i] < 1.0:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        elif diff < 0 and v[i] > 0.0:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    return v


def generate_random_binary(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    half = n // 2
    arr = np.zeros(n, dtype=int)
    idx = rng.permutation(n)[:half]
    arr[idx] = 1
    return arr


# --------- Local refinement (continuous) ----------
def refine_continuous(h, dx, target_sum, deadline, rng):
    cur = h.copy()
    cur_val = compute_c5_vec(cur, dx)
    step = 0.05
    n = len(h)
    while time.time() < deadline:
        donors = np.where(cur > 1e-12)[0]
        recips = np.where(cur < 1.0 - 1e-12)[0]
        if donors.size == 0 or recips.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recips)
        max_delta = min(step, cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    cur = project_to_sum(cur, target_sum)
    cur_val = compute_c5_vec(cur, dx)
    return cur, cur_val


# --------- Simulated annealing for binary vectors ----------
def anneal_binary(start_h, dx, rng, deadline, target_score=None):
    n = len(start_h)
    cur = start_h.copy()
    cur_val = compute_c5_vec(cur.astype(float), dx)
    best = cur.copy()
    best_val = cur_val
    temp = 0.015
    decay = 0.9995
    no_improve = 0
    while time.time() < deadline:
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val - 1e-12:
                best = cur.copy()
                best_val = cur_val
                no_improve = 0
                temp = max(temp, 0.015)  # reheating after improvement
            else:
                no_improve += 1
        else:
            no_improve += 1

        temp = max(temp * decay, 1e-7)
        if no_improve > 25000:
            # restart from best or random
            if target_score is not None and best_val < target_score:
                cur = best.copy()
                cur_val = best_val
            else:
                cur = generate_random_binary(n, rng)
                cur_val = compute_c5_vec(cur.astype(float), dx)
            no_improve = 0
            temp = 0.015
    return best.astype(float), best_val


# --------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        init_vec = np.array(initial_h_values, dtype=float)
        half = n // 2
        binary_start = np.zeros(n, dtype=int)
        binary_start[np.argsort(-init_vec)[:half]] = 1
    else:
        n = 120
        if n % 2:
            n += 1
        half = n // 2
        binary_start = generate_random_binary(n, rng)

    dx = 2.0 / n
    target_sum = n / 2.0

    # Initial best (binary converted to float)
    best_h = binary_start.astype(float)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    # Simulated annealing phase (≈75 % of budget)
    deadline_sa = start_time + budget_s * 0.75
    while time.time() < deadline_sa:
        # start from current best (binary) or occasionally random
        start_vec = best_h.astype(int)
        if rng.random() < 0.15:
            start_vec = generate_random_binary(n, rng)
        cur_h, cur_val = anneal_binary(start_vec, dx, rng, deadline_sa, target_score=0.3808)
        if cur_val < best_val - 1e-12:
            best_h, best_val = cur_h, cur_val
            if best_val < 0.3808:
                break

    # Continuous refinement phase (remaining time)
    deadline_ref = start_time + budget_s - 4.0
    if time.time() < deadline_ref:
        cur_h, cur_val = refine_continuous(best_h, dx, target_sum, deadline_ref, rng)
        if cur_val < best_val - 1e-12:
            best_h, best_val = cur_h, cur_val

    # Final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
