import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a (discretized) step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def binary_from_real(x, M):
    """Convert real vector x to a binary vector with exactly M ones (by largest entries)."""
    idx = np.argpartition(-x, M - 1)[:M]
    b = np.zeros_like(x, dtype=np.float64)
    b[idx] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate resolutions (even)
    candidate_ns = [80, 96, 112, 128, 144, 160]

    # prioritize parent size if present
    has_parent = "initial_h_values" in globals()
    if has_parent:
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidate_ns:
                candidate_ns.remove(n0)
            candidate_ns.insert(0, n0)

    # Main loop over resolutions
    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining_total = budget_s - (time.time() - start_time)
        # allocate at most 300 s per resolution, else a fraction of remaining time
        alloc = min(300.0, remaining_total * 0.7)

        M = n // 2  # required sum of h

        # ----- initialise binary candidate -----
        if has_parent and len(initial_h_values) == n:
            init_real = np.array(initial_h_values, dtype=np.float64)
            h_bin = binary_from_real(init_real, M)
        else:
            h_bin = np.zeros(n, dtype=np.float64)
            h_bin[rng.choice(n, M, replace=False)] = 1.0

        c5_cur = compute_c5(h_bin)
        best_h_n = h_bin.copy()
        best_c5_n = c5_cur

        # ----- binary simulated annealing -----
        anneal_start = time.time()
        while time.time() - anneal_start < alloc * 0.7:
            # linear temperature schedule
            elapsed = time.time() - anneal_start
            progress = elapsed / (alloc * 0.7)
            T = 0.2 * (1.0 - progress) + 1e-6

            # Occasionally perform a targeted swap that directly attacks the worst shift
            if rng.random() < 0.12:
                corr = np.correlate(h_bin, 1.0 - h_bin, mode="full")
                k_idx = int(np.argmax(corr))
                k_shift = k_idx - (n - 1)  # shift value
                if k_shift != 0:
                    k_abs = abs(k_shift)
                    if k_abs < n:
                        # i where h[i]=1 and h[i+k_abs]=0
                        i_candidates = np.where(
                            (h_bin[: n - k_abs] == 1) & (h_bin[k_abs:] == 0)
                        )[0]
                        if i_candidates.size:
                            i = rng.choice(i_candidates)
                            j = i + k_abs
                            h_new = h_bin.copy()
                            h_new[i] = 0.0
                            h_new[j] = 1.0
                            c5_new = compute_c5(h_new)
                            delta = c5_new - c5_cur
                            if delta < 0 or rng.random() < np.exp(-delta / T):
                                h_bin = h_new
                                c5_cur = c5_new
                                if c5_cur + 1e-12 < best_c5_n:
                                    best_c5_n = c5_cur
                                    best_h_n = h_bin.copy()
                                    if best_c5_n < 0.3808:
                                        break
                            continue  # skip the generic swap this iteration

            # generic random swap (maintains exact sum)
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = h_bin.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            c5_new = compute_c5(h_new)
            delta = c5_new - c5_cur
            if delta < 0 or rng.random() < np.exp(-delta / T):
                h_bin = h_new
                c5_cur = c5_new
                if c5_cur + 1e-12 < best_c5_n:
                    best_c5_n = c5_cur
                    best_h_n = h_bin.copy()
                    if best_c5_n < 0.3808:
                        break
            if best_c5_n < 0.3808:
                break

        # ----- continuous fine‑tuning (small mass transfers) -----
        remaining_n = alloc - (time.time() - anneal_start)
        if remaining_n > 0.8:
            cont_start = time.time()
            while time.time() - cont_start < remaining_n * 0.9:
                i, j = rng.choice(n, 2, replace=False)
                max_transfer = min(1.0 - best_h_n[i], best_h_n[j])
                if max_transfer <= 0:

                delta = rng.random() * max_transfer
                h_new = best_h_n.copy()
                h_new[i] += delta
                h_new[j] -= delta
                c5_new = compute_c5(h_new)
                if c5_new + 1e-12 < best_c5_n:

                    best_h_n = h_new

        if best_c5_n < best_c5:
            best_c5 = best_c5_n
            best_h = best_h_n.copy()
            best_n = n
            if best_c5 < 0.3808:

    if best_h is None:

        best_h = np.full(best_n, 0.5, dtype=np.float64)
        best_c5 = compute_c5(best_h)

    # enforce exact constraints one last time
    best_h = np.clip(best_h, 0.0, 1.0)
    M = best_n // 2
    cur_sum = np.sum(best_h)
    if abs(cur_sum - M) > 1e-9:
        best_h += (M - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
