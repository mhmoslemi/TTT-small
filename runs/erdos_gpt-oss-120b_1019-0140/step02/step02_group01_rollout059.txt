import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def local_search(n, h_start, time_limit, rng, target=0.3808):
    """
    Simulated annealing / hill‑climbing on a continuous vector h of length n.
    Returns the best (h, c5) found within time_limit seconds.
    """
    M = n / 2.0               # required sum
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    # current state for the annealing loop
    h = best_h.copy()
    cur_c5 = best_c5

    start = time.time()
    it = 0
    T0 = 0.04                  # initial temperature
    decay = 0.9995

    while True:
        now = time.time()
        if now - start > time_limit - 0.5:
            break
        if cur_c5 < target:
            break
        if it >= 250000:       # safety cap
            break

        # decide move type: 10% chance for a targeted worst‑shift move
        if rng.random() < 0.1:
            # targeted move using the worst (largest) overlap shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_idx = np.argmax(corr)
            lag = k_idx - (n - 1)   # shift value
            # overlapping indices
            if lag >= 0:
                i_idx = np.arange(n - lag)
                j_idx = i_idx + lag
            else:
                i_idx = np.arange(-lag, n)
                j_idx = i_idx + lag
            # contribution of each pair to the overlap
            contrib = h[i_idx] * (1.0 - h[j_idx])
            # pick the pair with largest contribution
            pair = np.argmax(contrib)
            i = int(i_idx[pair])
            j = int(j_idx[pair])
            # transfer from i (high) to j (low)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0.0:
                it += 1
                continue
            delta = rng.random() * max_transfer
            h[i] -= delta
            h[j] += delta
        else:
            # generic random transfer between two distinct positions
            i, j = rng.choice(n, 2, replace=False)
            hi, hj = h[i], h[j]
            # allowable transfer d from i to j (positive d) or reverse (negative d)
            d_min = -min(hj, 1.0 - hi)   # move from j to i
            d_max =  min(hi, 1.0 - hj)   # move from i to j
            if d_max <= d_min:
                it += 1
                continue
            d = rng.random() * (d_max - d_min) + d_min
            # apply transfer: hi -> hi - d, hj -> hj + d
            h[i] = hi - d
            h[j] = hj + d

        new_c5 = compute_c5(h)

        # acceptance
        if new_c5 < cur_c5:
            cur_c5 = new_c5
        else:
            T = T0 * (decay ** it)
            # avoid division by zero
            if T <= 0:
                T = 1e-12
            prob = np.exp(-(new_c5 - cur_c5) / T)
            if rng.random() < prob:
                cur_c5 = new_c5
            else:
                # revert move
                if rng.random() < 0.1:
                    # revert targeted move
                    # we have stored i,j,delta in locals
                    if 'delta' in locals():
                        # reverse the same transfer
                        if lag >= 0 or True:  # both cases use same i,j
                            h[i] += delta
                            h[j] -= delta
                else:
                    # revert generic transfer using original hi,hj
                    h[i] = hi
                    h[j] = hj

        # keep global best
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            # early exit if we already beat the target
            if best_c5 < target:
                break

        it += 1

    # final correction of sum (tiny numerical drift)
    cur_sum = best_h.sum()
    if abs(cur_sum - M) > 1e-9:
        best_h += (M - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5
