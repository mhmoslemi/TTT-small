def gradient_descent(h0, time_budget, rng):
    n = len(h0)
    target = n/2.0
    dx = 2.0/n
    h = h0.copy()
    v = np.zeros_like(h)  # momentum
    mu = 0.9
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.1
    no_improve = 0
    while time.time() - start < time_budget:
        # compute correlation v
        v_corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = v_corr.max() * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # increase beta
        if beta < max_beta:
            beta = min(max_beta, beta * 1.0015)
        # compute softmax weights
        vmax = v_corr.max()
        w = np.exp(beta * (v_corr - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient
        g = np.zeros_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # enforce zero-sum direction: subtract mean
        g -= np.mean(g)
        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        # update momentum
        v = mu * v - alpha * g
        h_new = h + v
        h_new = project_sum_and_bounds(h_new, target)
        # accept if improvement or random
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        # random shake
        if no_improve > 2000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            v = np.zeros_like(h)
            no_improve = 0
    return best_h, best_c5

def simulated_annealing(h0, best_c5, target, time_budget, rng):
    n = len(h0)
    dx = 2.0 / n
    current_h = h0.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5_local = best_c5
    start = time.time()
    # initial temperature based on current_c5
    T0 = (current_c5) * 0.01
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / time_budget)
        # propose pair move
        i, j = rng.choice(n, size=2, replace=False)
        # decide direction
        if rng.random() < 0.5:
            # move from i to j
            max_move = min(current_h[i], 1.0 - current_h[j])
            if max_move <= 0:
                continue
            delta = rng.random() * max_move
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move from j to i
            max_move = min(current_h[j], 1.0 - current_h[i])
            if max_move <= 0:
                continue
            delta = rng.random() * max_move
            new_h = current_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        # no need to project: sum preserved, values in [0,1] by construction
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(T, 1e-12)):
            current_h = new_h
            current_c5 = new_c5
            # update best
            if current_c5 < best_c5_local:
                best_c5_local = current_c5
                best_h = current_h.copy()
        # occasional random restart if stuck: if no improvement for many iterations
        # maybe track counter, but not needed.
    return best_h, best_c5_local

def project_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # distribute diff uniformly
        # need to maintain bounds
        # iterative approach
        # we try simple scaling: add diff/len(x) to each, clip
        inc = diff / len(x)
        x += inc
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        diff = target - s
        # if still diff significant, adjust in loop
        if abs(diff) > 1e-12:
            # adjust greedily
            # if need to increase sum, fill from indices with <1
            # if need to decrease sum, reduce from indices >0
            if diff > 0:
                # need to increase sum
                for i in range(len(x)):
                    if diff <= 1e-12:
                        break
                    space = 1.0 - x[i]
                    if space > 1e-12:
                        add = min(space, diff)
                        x[i] += add
                        diff -= add
            else:
                # need to decrease sum
                for i in range(len(x)):
                    if diff >= -1e-12:
                        break
                    space = x[i]
                    if space > 1e-12:
                        sub = min(space, -diff)
                        x[i] -= sub
                        diff += sub
    return x

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap scaled by dx."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target, tol=1e-12):
    """Clip to [0,1] and enforce sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    diff = target - s
    if abs(diff) <= tol:
        return x
    # distribute diff uniformly, then clip again
    inc = diff / len(x)
    x += inc
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    diff = target - s
    if abs(diff) <= tol:
        return x
    # greedy correction
    if diff > 0:
        for i in range(len(x)):
            if diff <= tol:
                break
            space = 1.0 - x[i]
            if space > 0:
                add = min(space, diff)
                x[i] += add
                diff -= add
    else:
        for i in range(len(x)):
            if diff >= -tol:
                break
            space = x[i]
            if space > 0:
                sub = min(space, -diff)
                x[i] -= sub
                diff += sub
    return x

def gradient_descent(h0, time_budget, rng):
    """Smooth‐max gradient descent with momentum."""
    n = h0.shape[0]
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    vel = np.zeros_like(h)
    mu = 0.9
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.1
    no_improve = 0

    while time.time() - start < time_budget:
        # correlation and current C5
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.0015)

        # soft‑max weights
        vmax = corr.max()
        w = np.exp(beta * (corr - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth objective
        g = np.zeros_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # enforce zero‑sum direction
        g -= np.mean(g)

        # step‑size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        # momentum update
        vel = mu * vel - alpha * g
        h_new = h + vel
        h_new = project_sum_and_bounds(h_new, target)

        # accept if improves or tiny random acceptance
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        # random shake when stuck
        if no_improve > 2000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            vel = np.zeros_like(h)
            no_improve = 0

    return best_h, best_c5

def simulated_annealing(h_start, c5_start, target, time_budget, rng):
    """Pairwise mass‑exchange SA."""
    n = h_start.shape[0]
    dx = 2.0 / n
    cur_h = h_start.copy()
    cur_c5 = c5_start
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    T0 = max(0.001, cur_c5 * 0.02)

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / time_budget)

        # choose distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # direction of transfer
        if rng.random() < 0.5:
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0:
                continue
            delta = rng.random() * max_move
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            max_move = min(cur_h[j], 1.0 - cur_h[i])
            if max_move <= 0:
                continue
            delta = rng.random() * max_move
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        new_c5 = compute_c5(new_h)
        d = new_c5 - cur_c5
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-14)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    # initialise

    gd_time = remaining * 0.5

    best_h, best_c5 = simulated_annealing(best_h, best_c5, target, sa_time, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
