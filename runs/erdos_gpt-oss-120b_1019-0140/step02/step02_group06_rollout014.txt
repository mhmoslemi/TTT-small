def gradient_descent_optimize(h0, time_budget, seed):

    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n/2
    dx = 2.0/n

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start = time.time()
    beta = 2.0
    max_beta = 40.0
    alpha0 = 0.2
    no_improve = 0

    while time.time() - start < time_budget:

         v = np.correlate(h, 1.0 - h, mode='full')
         c5 = v.max()*dx
         if c5 < best_c5 - 1e-12:
              best_c5 = c5
              best_h = h.copy()
              no_improve=0
         else:
              no_improve+=1

         # update smoothing
         if beta < max_beta:
              beta = min(max_beta, beta*1.01)

         # softmax weights
         vmax = v.max()
         w = np.exp(beta*(v - vmax))
         wsum = w.sum()
         if wsum == 0:
              w = np.ones_like(v)/len(v)
         else:
              w = w / wsum

         # compute gradient
         g = np.zeros(n)
         offset = n-1
         for idx, weight in enumerate(w):
              if weight < 1e-16:
                  continue
              k = idx - offset
              if k >= 0:
                  tail = slice(0, n-k)
                  head = slice(k, n)
              else:
                  k2 = -k
                  tail = slice(k2, n)
                  head = slice(0, n-k2)
              g[tail] += weight*(1.0 - h[head])
              g[head] -= weight*h[tail]
         # zero sum direction
         g = g - g.mean()

         # step-size schedule
         elapsed = time.time() - start
         alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

         h_new = h - alpha*g
         h_new = project_sum_and_bounds(h_new, target)
         c5_new = compute_c5(h_new)

         if c5_new < c5 or rng.random() < 0.001:
               h = h_new

         if no_improve > 2000:
               # random shake
               h = project_sum_and_bounds(h + rng.normal(scale=0.02,size=n), target)
               no_improve = 0

    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap value multiplied by dx."""
    n = len(h)
    dx = 2.0 / n
    # np.correlate returns array of size 2n-1, same as full convolution.
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute equally
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # fix any residual by scanning
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def gradient_descent_optimize(h0, time_budget, seed):
    """Local refinement using smooth max approximation."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 40.0
    alpha0 = 0.2
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # increase smoothing slowly
        if beta < max_beta:
            beta = min(max_beta, beta * 1.01)
        # soft-max weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / len(v)
        else:
            w = w / w_sum
        # gradient of smooth objective
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:

                head = slice(k, n)

                tail = slice(k2, n)

def compute_c5(h):
    """Maximum overlap value multiplied by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur

                    x[i] += inc
                    diff -= inc

                    diff += dec
    return x
