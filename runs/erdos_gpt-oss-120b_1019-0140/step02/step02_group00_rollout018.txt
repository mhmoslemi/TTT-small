def compute_c5_fft(h):
    n = h.shape[0]
    # compute correlation using FFT with zero padding length 2*n
    H = np.fft.fft(h, n*2)

    # conv length 2*n, but values near edges maybe due to zero padding; max among conv is fine.
    # scaling: use dx = 2.0 / n
    return np.max(conv) * (2.0 / n)

def anneal_binary(h, rng, time_budget):
    n = h.shape[0]
    # ones and zeros lists, map dicts
    ones_list = list(np.flatnonzero(h == 1))
    zeros_list = list(np.flatnonzero(h == 0))
    pos_one = {idx: i for i, idx in enumerate(ones_list)}
    pos_zero = {idx: i for i, idx in enumerate(zeros_list)}
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    score = best_score
    start = time.time()
    total = time_budget
    T0 = 0.002  # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= total:
            break
        # compute temperature linearly decreasing
        T = T0 * (1 - elapsed / total)
        # random swap
        i_pos = rng.integers(len(ones_list))
        j_pos = rng.integers(len(zeros_list))
        i_idx = ones_list[i_pos]
        j_idx = zeros_list[j_pos]
        # swap in h
        h[i_idx] = 0
        h[j_idx] = 1
        # compute new score quickly
        new_score = compute_c5_fft(h)
        delta = new_score - score
        # Accept?
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # update score
            score = new_score
            # update best if improvement
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # update lists: i becomes zero, j becomes one
            # swap entries in lists
            ones_list[i_pos] = j_idx
            zeros_list[j_pos] = i_idx
            # update mapping
            pos_one[j_idx] = i_pos
            pos_zero[i_idx] = j_pos
            # remove old mapping
            del pos_one[i_idx]
            del pos_zero[j_idx]
        else:
            # revert swap
            h[i_idx] = 1
            h[j_idx] = 0
        # continue
    return best_h, best_score

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0., hi=1., max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_c5_fft(h):
    n = h.shape[0]
    H = np.fft.fft(h, n*2)
    G = np.fft.fft(1.0 - h, n*2)
    conv = np.fft.ifft(H * np.conjugate(G)).real
    return np.max(conv) * (2.0 / n)

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term1[n-shift:] = 0
            term2[shift:] = h[:n-shift]
            term2[:shift] = 0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term1[:s] = 0
            term2[:n-s] = h[s:]
            term2[n-s:] = 0
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def anneal_binary(h_init, rng, max_time):
    h = h_init.copy()
    n = h.shape[0]
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    pos_one = {idx: i for i, idx in enumerate(ones)}
    pos_zero = {idx: i for i, idx in enumerate(zeros)}
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score
    start = time.time()
    T0 = 0.02
    while True:
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        T = T0 * max(0.0, 1.0 - elapsed / max_time)
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i_idx = ones[i_pos]
        j_idx = zeros[j_pos]
        h[i_idx] = 0
        h[j_idx] = 1
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # update lists
            ones[i_pos] = j_idx
            zeros[j_pos] = i_idx
            pos_one[j_idx] = i_pos
            pos_zero[i_idx] = j_pos
            del pos_one[i_idx]
            del pos_zero[j_idx]
        else:
            h[i_idx] = 1
            h[j_idx] = 0
    return best_h, best_score

import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_c5_fft(h):
    n = h.shape[0]
    H = np.fft.fft(h, n * 2)
    G = np.fft.fft(1.0 - h, n * 2)
    conv = np.fft.ifft(H * np.conjugate(G)).real
    return np.max(conv) * (2.0 / n)

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h, dtype=float)
    for shift in shifts:
        term1 = np.zeros(n, dtype=float)
        term2 = np.zeros(n, dtype=float)
        if shift >= 0:
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def anneal_binary(h_init, rng, max_time):
    h = h_init.copy()
    n = h.shape[0]
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    pos_one = {idx: i for i, idx in enumerate(ones)}
    pos_zero = {idx: i for i, idx in enumerate(zeros)}
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    cur_score = best_score
    start = time.time()
    T0 = 0.02
    while True:
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        T = T0 * max(0.0, 1.0 - elapsed / max_time)
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i_idx = ones[i_pos]
        j_idx = zeros[j_pos]
        h[i_idx] = 0
        h[j_idx] = 1
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            ones[i_pos] = j_idx
            zeros[j_pos] = i_idx
            pos_one[j_idx] = i_pos
            pos_zero[i_idx] = j_pos
            del pos_one[i_idx]
            del pos_zero[j_idx]
        else:
            h[i_idx] = 1
            h[j_idx] = 0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 256
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        m = init_full.shape[0]
        x_old = np.linspace(0.0, 1.0, m, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        init_interp = np.interp(x_new, x_old, init_full)
    else:
        init_interp = thue_morse_seq(n)

    init_bin = (init_interp >= 0.5).astype(int)
    target = n // 2
    cur = init_bin.sum()
    diff = cur - target
    if diff > 0:
        ones = np.flatnonzero(init_bin == 1)
        to_zero = rng.choice(ones, size=diff, replace=False)
        init_bin[to_zero] = 0
    elif diff < 0:
        zeros = np.flatnonzero(init_bin == 0)
        to_one = rng.choice(zeros, size=-diff, replace=False)
        init_bin[to_one] = 1

    best_h = init_bin.copy()
    best_score = compute_c5_fft(best_h)

    # Phase 1: Simulated annealing on binary grid
    anneal_time = max(0.0, time_left() - 60.0)
    if anneal_time > 5.0:
        h_ann, s_ann = anneal_binary(best_h.copy(), rng, anneal_time)
        if s_ann < best_score:
            best_h, best_score = h_ann, s_ann

    # Phase 2: Local continuous refinement via SLSQP
    cont_time = max(0.0, time_left() - 20.0)
    if cont_time >= 5.0:
        attempts = max(1, int(cont_time // 10))
        for _ in range(attempts):
            cand = project_onto_box_sum(best_h.astype(float) + 0.02 * (rng.random(n) - 0.5), n / 2.0)
            h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=800)
            if sc_opt < best_score:
                best_h, best_score = h_opt, sc_opt

    # Phase 3: Momentum gradient descent refinement
    remaining = time_left()
    if remaining > 2.0:
        est_iters = int(min(20000, max(5000, remaining * 800)))
        h_ref, sc_ref = gradient_descent_momentum(best_h.astype(float),
                                                  n_iter=est_iters,
                                                  step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
