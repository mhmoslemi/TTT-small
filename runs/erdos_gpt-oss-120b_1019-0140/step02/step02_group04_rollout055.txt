import time, numpy as np

def compute_c5_vec(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum(vec, target):
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target - cur
    # if diff small, done
    # Distribute diff uniformly then clip again iteratively
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    # adjust remaining diff
    diff = target - np.sum(vec)
    if abs(diff) > 1e-12:
        # fix by adding/subtracting from entries with slack
        if diff > 0:
            slack = 1.0 - vec
            # allocate as much as possible, iterating
            while diff > 1e-12 and np.any(slack > 1e-12):
                idx = np.argmax(slack) # get max slack
                inc = min(diff, slack[idx])
                vec[idx] += inc
                diff -= inc
                slack[idx] = 1.0 - vec[idx]
        else:
            slack = vec
            while diff < -1e-12 and np.any(slack > 1e-12):
                idx = np.argmax(slack)
                dec = min(-diff, slack[idx])
                vec[idx] -= dec
                diff += dec
                slack[idx] = vec[idx]
    return vec

def generate_random_feasible(n, target, rng):
    # generate random vector and project
    x = rng.random(n)
    x = x / np.sum(x) * target
    # enforce [0,1]
    x = np.clip(x, 0.0, 1.0)
    # adjust sum
    return project_to_sum(x, target)

def worst_shift(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # np.argmax on array of length 2*n-1
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    val = corr[max_idx] * dx
    return k, val, corr

def binary_hill_climb(h_bin, cur_val, dx, budget_s, start_time, rng):
    # similar to previous but with simulated annealing acceptance
    best_h = h_bin.copy()
    best_val = cur_val
    n = len(h_bin)
    # initial temperature similar to continuous
    T = 0.01
    cooling = 0.999
    while time.time() - start_time < budget_s:
        # pick a 1 and a 0 to swap
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val or rng.random() < np.exp(-(cand_val - best_val)/max(T, 1e-12)):
            best_h = cand
            best_val = cand_val
        T *= cooling
    return best_h.astype(float), best_val

    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target):
    """Clip to [0,1] and adjust to have exact sum = target."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target - cur
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target - np.sum(vec)
    # distribute remaining diff greedily
    if diff > 0:
        slack = 1.0 - vec
        while diff > 1e-12 and np.any(slack > 1e-12):
            i = np.argmax(slack)
            inc = min(diff, slack[i])
            vec[i] += inc
            diff -= inc
            slack[i] = 1.0 - vec[i]
    elif diff < 0:
        slack = vec
        while diff < -1e-12 and np.any(slack > 1e-12):
            i = np.argmax(slack)
            dec = min(-diff, slack[i])
            vec[i] -= dec
            diff += dec
            slack[i] = vec[i]
    return vec


def generate_random_feasible(n, target, rng):
    """Random vector in [0,1] with exact sum = target."""
    x = rng.random(n)
    if np.sum(x) == 0:
        x = np.full(n, target / n)
    else:
        x = x * (target / np.sum(x))
    x = np.clip(x, 0.0, 1.0)
    return project_to_sum(x, target)


def worst_shift(h):
    """Return shift k attaining maximal overlap (ignoring dx factor)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    return k, corr[max_idx]


def anneal_optimize(h0, target, dx, budget_s, start_time, rng):
    """Simulated‑annealing style mass‑transfer search."""
    cur_h = h0.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val

    T = 0.02          # initial temperature
    cooling = 0.9995
    step = 0.05
    it = 0

    while time.time() - start_time < budget_s:
        k, _ = worst_shift(cur_h)
        n = len(cur_h)

        if k == 0:
            # fallback random move
            donors = np.where(cur_h > 1e-12)[0]
            recips = np.where(cur_h < 1.0 - 1e-12)[0]
            if donors.size == 0 or recips.size == 0:
                break
            i = rng.choice(donors)
            j = rng.choice(recips)
        else:
            if k > 0:
                valid = np.arange(n - k)
                probs = cur_h[valid]
                if probs.sum() > 0:
                    i = rng.choice(valid, p=probs / probs.sum())
                else:
                    i = rng.choice(valid)
                j = i + k
            else:
                kp = -k
                valid = np.arange(kp, n)
                probs = cur_h[valid]
                if probs.sum() > 0:
                    i = rng.choice(valid, p=probs / probs.sum())
                else:
                    i = rng.choice(valid)
                j = i - kp

        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            # random small move
            donors = np.where(cur_h > 1e-12)[0]
            recips = np.where(cur_h < 1.0 - 1e-12)[0]
            if donors.size == 0 or recips.size == 0:
                break
            i = rng.choice(donors)
            j = rng.choice(recips)
            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        delta = rng.uniform(0.0, max_delta)

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = compute_c5_vec(new_h, dx)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
                if best_val < 0.3808:
                    break

        T *= cooling
        it += 1
        if it % 1000 == 0:
            step = max(step * 0.9, 0.001)

        # occasional random jitter to escape plateaus
        if it % 5000 == 0:
            for _ in range(3):
                donors = np.where(cur_h > 1e-12)[0]
                recips = np.where(cur_h < 1.0 - 1e-12)[0]
                if donors.size == 0 or recips.size == 0:
                    break
                i = rng.choice(donors)
                j = rng.choice(recips)
                dmax = min(step, cur_h[i], 1.0 - cur_h[j])
                if dmax <= 0:
                    continue
                d = rng.uniform(0.0, dmax)
                cur_h[i] -= d
                cur_h[j] += d
            cur_val = compute_c5_vec(cur_h, dx)

    return best_h, best_val


def binary_hill_climb(bin_h, cur_val, dx, budget_s, start_time, rng):
    """Simulated‑annealing hill‑climb on binary vectors (swap 1↔0)."""
    best_h = bin_h.copy()
    best_val = cur_val
    T = 0.01
    cooling = 0.999
    while time.time() - start_time < budget_s:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val or rng.random() < np.exp(-(cand_val - best_val) / max(T, 1e-12)):
            best_h, best_val = cand, cand_val
        T *= cooling
    return best_h.astype(float), best_val


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 100
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # Initial best solution
    if h0 is not None:
        cur_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        cur_h = generate_random_feasible(n, target_sum, rng)
    cur_val = compute_c5_vec(cur_h, dx)
    best_h, best_val = cur_h.copy(), cur_val

    # Main multi‑restart annealing loop
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 5.0:
            break
        seg_start = time.time()
        seg_time = min(30.0, time_left)

        # start point: either perturb current best or new random
        if rng.random() < 0.2 and best_val < 0.5:
            init_h = best_h.copy()
            for _ in range(5):
                donors = np.where(init_h > 1e-12)[0]
                recips = np.where(init_h < 1.0 - 1e-12)[0]
                if donors.size == 0 or recips.size == 0:
                    break
                i = rng.choice(donors)
                j = rng.choice(recips)
                delta = rng.uniform(0.0, min(0.05, init_h[i], 1.0 - init_h[j]))
                init_h[i] -= delta
                init_h[j] += delta
            init_h = project_to_sum(init_h, target_sum)
        else:
            init_h = generate_random_feasible(n, target_sum, rng)

        h_candidate, val_candidate = anneal_optimize(
            init_h, target_sum, dx, seg_time, seg_start, rng
        )
        if val_candidate < best_val:
            best_h, best_val = h_candidate, val_candidate
            if best_val < 0.3808:
                break

    # Binary conversion + hill climbing if time permits
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argsort(-best_h)[: int(round(target_sum))]
        binary_h[top_idx] = 1
        bin_val = compute_c5_vec(binary_h.astype(float), dx)
        binary_h_opt, bin_val_opt = binary_hill_climb(
            binary_h, bin_val, dx, remaining, start_time, rng
        )
        if bin_val_opt < best_val:
            best_h, best_val = binary_h_opt, bin_val_opt

    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h.tolist(), float(best_val), n
