def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine base n0 (if initial values exist)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 80
        h0 = None

    # Choose upsampling factor m
    m = 4  # could be 4 => n = 320 if n0=80
    n = n0 * m
    target_sum = n / 2.0
    dx = 2.0 / n

    # Prepare initial continuous vector
    if h0 is not None:
        # repeat to upsample
        h_cont = np.repeat(h0, m)
        # ensure within bounds
        h_cont = np.clip(h_cont, 0.0, 1.0)
        # project to exact sum
        h_cont = project_to_sum(h_cont, target_sum)
    else:
        h_cont = generate_random_feasible(n, target_sum, rng)

    # Compute initial best
    best_h = h_cont.copy()
    best_val = compute_c5_vec(best_h, dx)

    # time allocation
    total_budget = budget_s
    # allocate 55% for continuous SA, 45% for binary SA
    cont_budget = total_budget * 0.55
    bin_budget = total_budget - cont_budget - 2.0  # 2 sec safety

    # Continuous SA run
    best_h, best_val = simulated_annealing_continuous(best_h, target_sum, dx, start_time + cont_budget, start_time, rng, best_val)

    # After continuous, generate binary rounding from best
    binary_h = np.zeros(n, dtype=float)
    # choose top target_sum entries (target_sum is integer? n/2 may be int if n even)
    num_ones = int(round(target_sum))
    # In case of tie due to equal values, break ties by random permutation
    idx = np.argsort(-best_h)[:num_ones]
    binary_h[idx] = 1.0
    # compute binary C5
    binary_val = compute_c5_vec(binary_h, dx)
    if binary_val < best_val:
        best_h = binary_h.copy()
        best_val = binary_val

    # Binary SA run
    best_h, best_val = simulated_annealing_binary(best_h, target_sum, dx, start_time + total_budget - 2.0, start_time, rng, best_val)

    # final safety: clip and project
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)

def simulated_annealing_continuous(cur_h, target_sum, dx, time_limit, start_time, rng, cur_val):
    best_h, best_val = cur_h.copy(), cur_val
    n = len(cur_h)
    step = 0.1  # initial max delta
    # schedule parameters
    T0 = 0.01
    Tf = 1e-7
    total_time = time_limit - start_time
    while True:
        now = time.time()
        if now >= time_limit - 0.01:
            break
        elapsed = now - start_time
        frac = elapsed / total_time
        T = T0 * (Tf / T0) ** frac
        # choose donor and receiver
        donors = np.where(cur_h > 1e-12)[0]
        receivers = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(receivers)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        diff = cand_val - cur_val
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            cur_h = cand
            cur_val = cand_val
            if cur_val < best_val - 1e-12:
                best_h = cur_h.copy()
                best_val = cur_val
                # adapt step up
                step = min(step * 1.2, 0.5)
        else:
            # adapt step down can be occasionally
            if rng.random() < 0.3:
                step *= 0.9

import time
import numpy as np
import math

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given vector h."""
    # full correlation gives lags from -(n-1) to (n-1)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Project vec into [0,1] while fixing its sum to target_sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # add uniform correction first
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    # greedy fix the remaining difference
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Random continuous feasible vector (values in [0,1], exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    # fix residual sum if any
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def simulated_annealing_continuous(start_h, target_sum, dx, deadline, start_time, rng, start_val):
    """SA on continuous vectors (mass transfer)."""
    cur_h = start_h.copy()
    cur_val = start_val
    best_h, best_val = cur_h.copy(), cur_val
    n = len(cur_h)
    step = 0.12                      # initial max transfer
    T0, Tf = 0.012, 1e-7             # temperature schedule
    total = max(deadline - start_time, 1e-6)

    while True:
        now = time.time()
        if now >= deadline - 0.025:
            break
        elapsed = now - start_time
        frac = elapsed / total
        T = T0 * (Tf / T0) ** frac

        donors = np.where(cur_h > 1e-12)[0]
        recvs = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recvs.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recvs)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        diff = cand_val - cur_val
        accept = diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12))
        if accept:
            cur_h, cur_val = cand, cand_val
            if cur_val < best_val - 1e-12:
                best_h, best_val = cur_h.copy(), cur_val
                step = min(step * 1.2, 0.45)
                if best_val <= 0.3808:
                    return best_h, best_val
        else:
            if rng.random() < 0.3:
                step *= 0.9
    return best_h, best_val


def simulated_annealing_binary(start_h, target_sum, dx, deadline, start_time, rng, start_val):
    """SA on binary vectors (swap a 1 with a 0)."""
    cur_h = np.rint(start_h).astype(int)
    cur_val = compute_c5_vec(cur_h.astype(float), dx)
    best_h, best_val = cur_h.copy(), cur_val
    n = len(cur_h)

    total = max(deadline - start_time, 1e-6)

    while True:
        now = time.time()
        if now >= deadline - 0.025:
            break
        elapsed = now - start_time
        frac = elapsed / total
        T = T0 * (Tf / T0) ** frac

        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # base size (use initial if present)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 80
        h0 = None

    # Upsample factor (keep total points < 1000)
    up_factor = 4
    n = n0 * up_factor
    target_sum = n / 2.0
    dx = 2.0 / n

    # Initial continuous vector
    if h0 is not None:
        h_cont = np.repeat(h0, up_factor)
        h_cont = np.clip(h_cont, 0.0, 1.0)
        h_cont = project_to_sum(h_cont, target_sum)
    else:
        h_cont = generate_random_feasible(n, target_sum, rng)

    best_h = h_cont.copy()
    best_val = compute_c5_vec(best_h, dx)

    # time allocation
    total_budget = budget_s
    cont_time = total_budget * 0.55
    bin_time = total_budget - cont_time - 2.0   # spare margin

    # Continuous simulated annealing
    cont_deadline = start_time + cont_time
    best_h, best_val = simulated_annealing_continuous(
        best_h, target_sum, dx, cont_deadline, start_time, rng, best_val
    )
    if best_val <= 0.3808:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_to_sum(best_h, target_sum)
        return best_h, compute_c5_vec(best_h, dx), n

    # Convert to binary by taking the largest entries
    num_ones = int(round(target_sum))
    idx = np.argsort(-best_h)[:num_ones]
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx] = 1.0
    bin_val = compute_c5_vec(bin_h, dx)
    if bin_val < best_val:
        best_h, best_val = bin_h, bin_val

    # Binary simulated annealing
    bin_deadline = start_time + total_budget - 2.0
    best_h, best_val = simulated_annealing_binary(
        best_h, target_sum, dx, bin_deadline, start_time, rng, best_val
    )
    # Final safety checks
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
