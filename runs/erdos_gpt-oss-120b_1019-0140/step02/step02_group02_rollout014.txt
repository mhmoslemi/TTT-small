def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def random_binary_vector(n, rng):
    h = np.zeros(n, dtype=np.float64)
    ones_idx = rng.choice(n, n//2, replace=False)
    h[ones_idx] = 1.0
    return h

def hill_climb(h, rng, budget_deadline):
    best_h = h.copy()
    best_val = compute_c5(best_h)
    # maintain list of indices of 1s and 0s
    while time.time() < budget_deadline:
        # choose random 1 and 0 to swap
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        best_h[i], best_h[j] = 0.0, 1.0
        val = compute_c5(best_h)
        if val < best_val:
            best_val = val
            # keep changes
        else:
            # revert with probability exp(-(val - best_val)/T)
            # temperature schedule: linear cooling based on remaining time
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = remaining / (deadline - start_time) * initial_T  # start T = something
            # compute acceptance
            if rng.random() < math.exp(-(val - best_val) / (T + 1e-12)):
                best_val = val  # accept uphill
            else:
                # revert swap
                best_h[i], best_h[j] = 1.0, 0.0
        # maybe track best overall solution across restarts.
    return best_h, best_val

def targeted_anneal(h, cur_val, start_time, deadline, rng):
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    while time.time() < deadline - 0.01:
        # compute correlation and max shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute region indices
        if k >= 0:
            i_range = np.arange(0, n - k)
            j_range = i_range + k
        else:
            kk = -k
            i_range = np.arange(kk, n)
            j_range = i_range - kk
        prod = h[i_range] * (1.0 - h[j_range])
        if prod.sum() > 0:
            probs = prod / prod.sum()
            i_local = rng.choice(i_range, p=probs)
        else:
            i_local = rng.choice(i_range)
        # candidate destination
        zero_idxs = np.where(h < 1.0 - 1e-12)[0]
        if zero_idxs.size == 0:
            break
        # compute w for each zero idx
        # compute counterpart presence
        if k >= 0:
            counterpart = zero_idxs + k
            mask = counterpart < n
        else:
            kk = -k
            counterpart = zero_idxs - kk
            mask = counterpart >= 0
        w = np.empty_like(zero_idxs, dtype=float)
        w.fill(0.0)
        if mask.any():
            w[mask] = 1.0 - h[counterpart[mask]]
        # choose j with minimal w
        min_w = w.min()
        min_candidates = zero_idxs[w == min_w]
        j_local = rng.choice(min_candidates)
        # compute max transfer
        max_transfer = min(h[i_local], 1.0 - h[j_local])
        if max_transfer <= 0:
            continue
        # delta. Use random fraction to avoid too large jumps
        delta = max_transfer * rng.random()
        # apply
        h[i_local] -= delta
        h[j_local] += delta
        new_val = compute_c5(h)
        if new_val < best_val:
            best_val = new_val
            best_h = h.copy()
        # acceptance: always accept if new_val <= cur_val else accept with prob
        if new_val <= cur_val:
            cur_val = new_val
        else:
            # temperature
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            # degrade temperature linearly: start_T maybe 0.01, end_T maybe 1e-6
            # compute proportion of remaining time
            prop = max(remaining / (deadline - start_time), 0.0)
            T = 0.01 * prop + 1e-6
            if rng.random() < np.exp(-(new_val - cur_val) / T):
                cur_val = new_val
            else:
                # revert
                h[i_local] += delta
                h[j_local] -= delta
        # early exit if best_val lower than target threshold
        if best_val < 0.3808:
            break
    return best_h, best_val

import time
import math
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector_continuous(n, rng):
    """Random feasible vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    inc = min(diff, 1.0 - v[i])
                    v[i] += inc
                    diff -= inc
                elif diff < 0 and v[i] > 0.0:
                    dec = min(-diff, v[i])
                    v[i] -= dec
                    diff += dec
    return v

def targeted_anneal(h_start, cur_val_start, overall_start, deadline, rng):
    """Focused simulated annealing: move mass from heavy contributors to low‑impact sites."""
    h = h_start.copy()
    cur_val = cur_val_start
    best_h = h.copy()
    best_val = cur_val_start
    n = h.shape[0]

    while time.time() < deadline - 0.01:
        # find shift with maximum overlap
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)  # shift index

        # region indices for this shift
        if k >= 0:
            i_range = np.arange(0, n - k)
            j_range = i_range + k
        else:
            kk = -k
            i_range = np.arange(kk, n)
            j_range = i_range - kk

        # weighted choice of source index i (heavy contributor)
        prod = h[i_range] * (1.0 - h[j_range])
        if prod.sum() > 0:
            probs = prod / prod.sum()
            i_local = rng.choice(i_range, p=probs)
        else:
            i_local = rng.choice(i_range)

        # choose destination index j (low impact)
        zero_idxs = np.where(h < 1.0 - 1e-12)[0]
        if zero_idxs.size == 0:
            break
        # compute impact weight w = 1 - h[partner] when partner exists, else 0
        if k >= 0:
            partner = zero_idxs + k
            mask = partner < n
        else:
            kk = -k
            partner = zero_idxs - kk
            mask = partner >= 0
        w = np.empty_like(zero_idxs, dtype=float)
        w.fill(0.0)
        if mask.any():
            w[mask] = 1.0 - h[partner[mask]]
        min_w = w.min()
        cand_js = zero_idxs[w == min_w]
        j_local = rng.choice(cand_js)

        # perform a transfer of mass
        max_transfer = min(h[i_local], 1.0 - h[j_local])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random()
        h[i_local] -= delta
        h[j_local] += delta

        new_val = compute_c5(h)

        # update best found
        if new_val < best_val:
            best_val = new_val
            best_h = h.copy()

        # Metropolis acceptance for uphill moves
        if new_val <= cur_val:
            cur_val = new_val
        else:
            elapsed = time.time() - overall_start
            remaining = max(deadline - time.time(), 0.0)
            # linear cooling: start temperature ~0.02, end ~1e-6
            total_time = deadline - overall_start
            prop = remaining / total_time if total_time > 0 else 0.0
            T = 0.02 * prop + 1e-6
            if math.exp(-(new_val - cur_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert move
                h[i_local] += delta
                h[j_local] -= delta

        if best_val < 0.3808:
            break

    return best_h, best_val
