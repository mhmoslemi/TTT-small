def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # distribute diff uniformly as much as possible
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    # residual
    diff = target_sum - np.sum(vec)
    # fix greedily
    indices = np.arange(len(vec))
    if diff > 0:
        # need increase some entries
        for i in indices:
            if diff <= 0:
                break
            if vec[i] < 1.0:
                inc = min(diff, 1.0 - vec[i])
                vec[i] += inc
                diff -= inc
    elif diff < 0:
        for i in indices:
            if diff >= 0:
                break
            if vec[i] > 0.0:
                dec = min(-diff, vec[i])
                vec[i] -= dec
                diff += dec
    return vec
```

    cur_h = h.copy()
    cur_val = compute_c5(cur_h, dx)
    step = 0.05
    n = len(h)
    while time.time() < time_deadline:
        # pick donor and recipient
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]

def compute_c5(h, dx):
    """Maximum overlap integral for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target_sum):
    """Clip to [0,1] and adjust to exact sum target_sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if diff > 0:
        for i in range(len(vec)):
            if diff <= 0:
                break
            if vec[i] < 1.0:
                inc = min(diff, 1.0 - vec[i])
                vec[i] += inc
                diff -= inc
    elif diff < 0:
        for i in range(len(vec)):
            if diff >= 0:
                break
            if vec[i] > 0.0:
                dec = min(-diff, vec[i])
                vec[i] -= dec
                diff += dec
    return vec

def random_binary_vector(n, weight, rng):
    """Generate a binary vector of length n with exactly weight entries equal to 1."""
    vec = np.zeros(n, dtype=float)
    vec[rng.choice(n, size=weight, replace=False)] = 1.0
    return vec

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        new_val = compute_c5(h, dx)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if cur_val <= 0.3808:
                break
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, cur_val

def refine_continuous(h, dx, target_sum, deadline, rng):
    """Simple stochastic local search allowing fractional values."""
    cur_h = h.copy()
    cur_val = compute_c5(cur_h, dx)
    step = 0.05
    n = len(h)
    while time.time() < deadline:
        donors = np.where(cur_h > 1e-12)[0]
        recps = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recps.size == 0:
            break
        i = int(rng.choice(donors))
        j = int(rng.choice(recps))
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.random() * max_delta
        cur_h[i] -= delta
        cur_h[j] += delta
        new_val = compute_c5(cur_h, dx)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            step = min(step * 1.1, 0.5)
            if cur_val <= 0.3808:
                break
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5(cur_h, dx)
    return cur_h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    dx = 2.0 / n
    target_sum = n / 2.0
    weight = int(target_sum)

    # initialise best solution
    if h0 is not None:
        best_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        best_h = np.zeros(n, dtype=float)
        best_h[:weight] = 1.0
        best_h = best_h[rng.permutation(n)]

    best_val = compute_c5(best_h, dx)

    end_time = start_time + budget_s
    binary_phase_end = start_time + budget_s * 0.70

    # binary optimisation phase
    while time.time() < binary_phase_end:
        # occasional warm‑start from current best if it's already binary
        if rng.random() < 0.1 and np.all((best_h == 0) | (best_h == 1)):
            start_vec = best_h.copy()
        else:
            start_vec = random_binary_vector(n, weight, rng)

        cur_val = compute_c5(start_vec, dx)
        local_deadline = min(binary_phase_end, time.time() + 0.4)
        h_opt, val_opt = hill_climb_binary(start_vec, cur_val, dx, local_deadline, rng)

        if val_opt < best_val - 1e-12:
            best_h = h_opt.copy()
            best_val = val_opt
            if best_val <= 0.3808:
                break

    # continuous refinement on the remaining budget
    cont_deadline = start_time + budget_s * 0.95
    if time.time() < cont_deadline:
        best_h, best_val = refine_continuous(best_h, dx, target_sum, cont_deadline, rng)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5(best_h, dx)

    return best_h, float(best_val), n
