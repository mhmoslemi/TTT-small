    m = 2*n - 1   # length for full correlation
    # pad next power of two for speed? major not needed. Use np.fft.fft.

    H = np.fft.rfft(h, n=N)

def sa_optimize(n, time_budget, rng, init_h=None):
    M = n // 2
    dx = 2.0 / n
    # Initialize h
    if init_h is not None and len(init_h)==n:
        h = np.array(init_h, dtype=float)
        # ensure constraints
        cur = h.sum()
        if abs(cur - M) > 1e-9:
            h += (M - cur) / n
        np.clip(h, 0, 1, out=h)
    else:
        h = generate_initial_h(n, M, rng)

    # compute current C5
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    # heating schedule
    start = time.time()
    # Use exponential cooling schedule based on time fraction t = (now - start)/time_budget
    # Let's set initial temperature as maybe 0.005 * c5 (just guess)
    T0 = 1e-3  # small but allow some acceptance initially
    # We'll compute total steps as maybe int(1e6) or based on time.
    # We'll usage while time left.
    while time.time() - start < time_budget:
        # compute temperature using linear schedule decreasing to near zero
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1 - frac)  # linear cooling
        # pick random pair
        i, j = rng.choice(n, size=2, replace=False)
        # compute max transfer
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        # delta magnitude: maybe scaled by temperature
        delta = rng.random() * max_transfer * (0.5 + 0.5* (1-frac))  # if frac small (early), large delta; later small
        # modify copy
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # compute new C5
        c5_new = compute_c5(h_new)
        # accept criteria
        if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / (T + 1e-12)):
            h = h_new
            c5 = c5_new

def greedy_refine(h, overl, c5, time_budget, rng):
    n = len(h)
    start = time.time()
    best_h = h.copy()
    best_c5 = c5
    best_overl = overl.copy()
    while time.time() - start < time_budget:

        max_transfer = min(1.0 - h[i], h[j])

        c5_new = np.max(overl_new) * (2.0 / n)

            h[j] -= delta
            overl = overl_new
            c5 = c5_new
            best_c5 = c5_new
            best_h = h.copy()
            best_overl = overl.copy()
    return best_h, best_c5

import time
import numpy as np

def generate_initial_h(n, M, rng):
    """Linear decreasing shape with correct mass."""
    lin = np.linspace(1.0, 0.0, n)
    h = lin * (M / lin.sum())
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - M) > 1e-12:
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    """Maximum overlap integral for a given discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def sa_optimize_for_n(n, time_budget, rng, init_h=None):
    """Simulated‑annealing + greedy refinement for a fixed resolution."""
    M = n // 2
    dx = 2.0 / n

    if init_h is not None:
        h = np.array(init_h, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)
    else:
        h = generate_initial_h(n, M, rng)
        # tiny noise to break symmetry
        h += (rng.random(n) - 0.5) * 0.02
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)

    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    start = time.time()
    T0 = 5e-3  # initial temperature

    # ---------- simulated annealing ----------
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max(1e-9, T0 * (1.0 - frac))

        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0.0:
            continue

        scale = 0.2 + 0.8 * (1.0 - frac)  # larger steps early
        delta = rng.random() * max_transfer * scale
        if delta <= 0.0:
            continue

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)

        # Metropolis acceptance
        if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / (T + 1e-15)):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:

            c5 = c5_new
            best_c5 = c5_new
            best_h = h.copy()
            if best_c5 < 0.3808:
                break

        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5
