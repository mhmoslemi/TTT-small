def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # use np.correlate, mode 'full'
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def binary_anneal(h, cur_val, rng, time_deadline):
    # local SA using swaps
    best_h = h.copy()
    best_val = cur_val
    cur_h = h.copy()
    cur_val = cur_val
    n = h.shape[0]
    # Precompute arrays of indices where h_i = 1 and =0 we need to update after each swap
    # We'll recompute each step for simplicity.
    # Determine number of iterations to perform: loop until time runs out.
    # Temperature schedule:
    start_temp = 0.01
    # maybe compute schedule based on time remaining?
    while time.time() < time_deadline - 0.01:
        # get indices
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

def continuous_anneal(h, cur_val, rng, deadline):
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        # pick random i with h_i > 0 and random j with h_j < 1 (j maybe same as i? no)
        i = rng.choice(np.where(h > 1e-12)[0])
        j = rng.choice(np.where(h < 1 - 1e-12)[0])
        if i == j:
            continue
        delta_max = min(h[i], 1 - h[j])
        # random delta as fraction of delta_max
        delta = rng.random() * delta_max
        if delta < 1e-12:
            continue
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            # temperature schedule based on remaining time
            remaining = deadline - time.time()
            total = deadline - start  # start defined outer
            T = max(remaining / total, 1e-6) * 0.01  # scaling
            if np.exp(-(new_val - cur_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert
                h[i] += delta
                h[j] -= delta
        # maybe occasionally rebalance any stray rounding errors
    return best_h, best_val

import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a step function h given as a NumPy array."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation between h and (1-h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    # simple shift to get close to target, then clip
    shift = (target - v.sum()) / n
    v = v + shift
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # distribute any remaining tiny difference
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(v < 1.0)[0]
            if idx.size == 0:
                break
            inc = min(diff, 1.0 - v[idx[0]])
            v[idx[0]] += inc
        else:
            idx = np.where(v > 0.0)[0]
            if idx.size == 0:
                break
            dec = min(-diff, v[idx[0]])
            v[idx[0]] -= dec
        diff = target - v.sum()
    return v

    cur_val = compute_c5(h)

    best_val = cur_val
    T_start = 0.02
    T_end = 1e-6
