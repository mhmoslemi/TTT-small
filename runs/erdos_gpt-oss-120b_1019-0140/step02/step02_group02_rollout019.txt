def continuous_anneal(h_start, best_val, start_time, deadline, rng):
    h = h_start.copy()
    cur_val = best_val
    best_h = h.copy()
    best = best_val
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        # compute correlation for current h
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # produce contributions:
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        # find index with largest product
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # find candidate j where h[j] < 1
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            # simple temp acceptance not needed
            # revert
            h[i] += delta
            h[j] -= delta
        if best < 0.3808:
            break
    return best_h, best

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _project_to_bounds_and_sum(x, target_sum):
    """Euclidean projection of x onto {0≤y≤1, sum(y)=target_sum}."""
    low, high = -2.0, 2.0
    for _ in range(40):
        lam = (low + high) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def random_feasible_vector(n, target_sum, rng):
    """Uniform random vector projected onto the feasible set."""
    x = rng.random(n)
    return _project_to_bounds_and_sum(x, target_sum)

def gradient_descent_local(h_start, target_sum, dx, deadline, rng):
    """Projected gradient descent on a smooth max-overlap surrogate."""
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(h)
    alpha = 10.0           # softmax sharpness
    step0 = 0.5            # initial step size
    max_iter = 1500
    it = 0
    while time.time() < deadline - 0.1 and it < max_iter:
        it += 1
        # correlation and overlap values
        corr = np.correlate(h, 1.0 - h, mode="full")
        O = corr * dx
        center = n - 1
        # exclude zero shift from the surrogate
        O_tmp = O.copy()
        O_tmp[center] = -np.inf
        max_O = O_tmp.max()
        # softmax weights (zero for the ignored shift)
        w = np.exp(alpha * (O_tmp - max_O))
        sum_w = w.sum()
        if sum_w == 0:
            break
        p = w / sum_w

        # gradient of the surrogate
        grad = np.zeros_like(h)
        for k in range(-(n - 1), n):
            if k == 0:
                continue
            weight = p[k + (n - 1)]
            if weight == 0:
                continue
            if k >= 0:
                i_slice = slice(0, n - k)
                j_slice = slice(k, n)
                grad[i_slice] += weight * (1.0 - h[j_slice])
                grad[j_slice] += weight * (-h[i_slice])
            else:
                kk = -k
                i_slice = slice(kk, n)
                j_slice = slice(0, n - kk)
                grad[i_slice] += weight * (1.0 - h[j_slice])
                grad[j_slice] += weight * (-h[i_slice])
        grad *= dx

        # step with linear decay
        step = step0 * (1.0 - it / max_iter)
        h_new = h - step * grad
        h = _project_to_bounds_and_sum(h_new, target_sum)

        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

def continuous_anneal(h_start, best_val, start_time, deadline, rng):
    """Simple mass‑transfer annealing on the original max‑overlap."""
    h = h_start.copy()
    cur_val = best_val
    best_h = h.copy()
    best = best_val
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i = i_range[np.argmax(prod)]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
        if best < 0.3808:
            break
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = _project_to_bounds_and_sum(h0, target)
    else:
        n = 80
        target = n / 2.0
        h0 = random_feasible_vector(n, target, rng)

    dx = 2.0 / n
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- multistart gradient descent -----
    while time.time() < deadline - 6.0:
        # choose start point
        if rng.random() < 0.6:
            h_start = random_feasible_vector(n, target, rng)
        else:
            pert = rng.normal(scale=0.05, size=n)
            h_start = best_h + pert
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = _project_to_bounds_and_sum(h_start, target)

        h_local, val_local = gradient_descent_local(
            h_start, target, dx, deadline, rng
        )
        if val_local < best_val:
            best_val = val_local
            best_h = h_local.copy()
            if best_val < 0.3808:
                break
        if time.time() > deadline - 6.0:
            break

    # ----- final annealing refinement -----
    if time.time() < deadline - 0.5:
        current_val = compute_c5(best_h)
        best_h, best_val = continuous_anneal(best_h, current_val, start, deadline, rng)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_bounds_and_sum(best_h, target)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
