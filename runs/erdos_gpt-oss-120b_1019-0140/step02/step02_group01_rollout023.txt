import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def binary_hill_climb(h_init, n, M, time_budget, rng):
    """Heuristic search on binary vectors with fixed sum M."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    start = time.time()
    while time.time() - start < time_budget - 0.01:
        # correlation of h with itself
        corr_hh = np.correlate(h, h, mode='full')
        overlap = M - corr_hh
        # ignore zero shift
        overlap[n - 1] = -np.inf
        worst_off = np.argmax(overlap)
        k = worst_off - (n - 1)

        improved = False
        # try a targeted swap that improves the worst shift
        if k >= 0:
            L = n - k
            if L > 0:
                h_i = h[:L]
                h_i_k = h[k:]
                bad_one = np.where((h_i == 1) & (h_i_k == 0))[0]
                good_zero = np.where((h_i == 0) & (h_i_k == 1))[0]
                if bad_one.size and good_zero.size:
                    for _ in range(3):
                        i_rel = rng.choice(bad_one)
                        j_rel = rng.choice(good_zero)
                        i, j = int(i_rel), int(j_rel)
                        h_new = h.copy()
                        h_new[i], h_new[j] = h_new[j], h_new[i]
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5 - 1e-12:
                            h = h_new
                            best_h = h_new.copy()
                            best_c5 = c5_new
                            no_improve = 0
                            improved = True
                            break
        else:
            k_abs = -k
            L = n - k_abs
            if L > 0:
                h_i = h[k_abs:]
                h_i_k = h[:L]
                bad_one = np.where((h_i == 1) & (h_i_k == 0))[0]
                good_zero = np.where((h_i == 0) & (h_i_k == 1))[0]
                if bad_one.size and good_zero.size:
                    for _ in range(3):
                        i_rel = rng.choice(bad_one)
                        j_rel = rng.choice(good_zero)
                        i = int(k_abs + i_rel)
                        j = int(k_abs + j_rel)
                        h_new = h.copy()
                        h_new[i], h_new[j] = h_new[j], h_new[i]
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5 - 1e-12:
                            h = h_new
                            best_h = h_new.copy()
                            best_c5 = c5_new
                            no_improve = 0
                            improved = True
                            break
        if not improved:
            # random swaps, keep the best improvement among a few trials
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            local_best = best_c5
            local_swap = None
            for _ in range(5):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h_new = h.copy()
                h_new[i], h_new[j] = h_new[j], h_new[i]
                c5_new = compute_c5(h_new)

                    local_best = c5_new
                    local_swap = h_new
            if local_swap is not None:
                h = local_swap
                best_h = local_swap.copy()
                best_c5 = local_best

            if c5_restart < best_c5 - 1e-12:
                best_h = h.copy()
                best_c5 = c5_restart
            no_improve = 0
    return best_h, best_c5
