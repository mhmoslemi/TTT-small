def project_to_feasible(x):
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    target = n/2
    # enforce sum by simple water-filling
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # find indices not at bound
        if diff > 0:
            free = np.where(x < 1 - 1e-12)[0]
        else:
            free = np.where(x > 1e-12)[0]
        if free.size == 0:
            break  # cannot adjust
        # distribute equally among free
        delta = diff / free.size
        # limit delta to not exceed bounds
        delta = np.clip(delta, -x[free], 1 - x[free])
        x[free] += delta
    return x

def hill_climb_binary(n, rng, time_end, max_iters_per_restart=20000):
    best_h = None
    best_c5 = np.inf
    # precompute positions of ones and zeros for quickly picking
    while time.time() < time_end:
        # random start
        h = np.zeros(n, dtype=int)
        ones_idx = rng.choice(n, n//2, replace=False)
        h[ones_idx] = 1
        cur_c5 = compute_c5(h.astype(float))
        # track local best
        loc_best_h = h.copy()
        loc_best_c5 = cur_c5
        no_improve = 0
        # loop
        it = 0
        while time.time() < time_end and it < max_iters_per_restart:
            # for speed, we can sample several attempts per iteration? but keep simple.
            # Choose indices
            ones_pos = np.flatnonzero(h)
            zeros_pos = np.flatnonzero(1 - h)
            if not ones_pos.size or not zeros_pos.size:
                break
            i = rng.choice(ones_pos)
            j = rng.choice(zeros_pos)
            # swap
            h[i], h[j] = 0, 1
            new_c5 = compute_c5(h.astype(float))
            # accept if improvement
            if new_c5 + 1e-12 < cur_c5:
                cur_c5 = new_c5
                if new_c5 < loc_best_c5:
                    loc_best_c5 = new_c5
                    loc_best_h = h.copy()
                no_improve = 0
            else:
                # revert swap
                h[i], h[j] = 1, 0
                no_improve += 1
            # early break if no_improve > some threshold
            if no_improve >= 1000:
                break
            it += 1
        # update global best
        if loc_best_c5 < best_c5:
            best_c5 = loc_best_c5
            best_h = loc_best_h.copy()
            # early exit if good enough
            if best_c5 <= 0.3808:
                return best_h.astype(float), best_c5, n
    # final conversion to float
    if best_h is None:
        # fallback: produce uniform
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
    return best_h.astype(float), best_c5, n

def compute_c5(arr):
    # arr is length n float
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def binary_anneal(start_h, time_end, rng, n):
    h = (start_h >= 0.5).astype(int)  # binary representation
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    local_start = time.time()
    init_temp = 5e-2
    final_temp = 1e-6
    while time.time() < time_end:
        # compute temperature
        progress = (time.time() - local_start) / (time_end - local_start)
        T = init_temp * (1 - progress) + final_temp * progress
        # select random swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0, 1
        new_c5 = compute_c5(h.astype(float))
        dE = new_c5 - cur_c5
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1, 0
        # loop
    return best_h.astype(float), best_c5

        a = b

def project_sum_bounds(x, target):
    # clip to [0,1] then enforce sum = target by water-filling iterative approach.
    y = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        s = y.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine free indices: those not at bound or those we can adjust in direction of diff.
        if diff > 0:
            free = np.where(y < 1 - 1e-12)[0]
        else:
            free = np.where(y > 1e-12)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        # limit delta to bound capacities
        delta = np.clip(delta, -y[free], 1 - y[free])
        y[free] += delta
    # final clipping
    y = np.clip(y, 0.0, 1.0)
    return y

import time
import math
import numpy as np

def compute_c5(arr):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    n = arr.shape[0]
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Binary Thue‑Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)

def rudin_shapiro_seq(n):
    """Binary Rudin‑Shapiro sequence (+1/‑1 → 1/0) of length n."""
    a = np.array([1], dtype=int)
    while a.shape[0] < n:
        k = np.arange(a.shape[0])
        sign = np.where(k % 2 == 0, 1, -1)
        b = np.empty(2 * a.shape[0], dtype=int)
        b[0::2] = a
        b[1::2] = a * sign
        a = b
    a = a[:n]
    return ((a + 1) // 2).astype(int)

def adjust_binary_to_sum(vec, target, rng):
    """Flip bits of binary vec to achieve exact sum=target."""
    diff = target - vec.sum()
    if diff == 0:
        return vec
    idxs = np.arange(vec.shape[0])
    rng.shuffle(idxs)
    for idx in idxs:
        if diff == 0:
            break
        if diff > 0 and vec[idx] == 0:
            vec[idx] = 1
            diff -= 1
        elif diff < 0 and vec[idx] == 1:
            vec[idx] = 0
            diff += 1
    return vec

def project_sum_bounds(x, target):
    """Clip to [0,1] and adjust to meet sum=target (water‑filling)."""
    y = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        s = y.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(y < 1 - 1e-12)[0]
        else:
            free = np.where(y > 1e-12)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        delta = np.clip(delta, -y[free], 1 - y[free])
        y[free] += delta
    return np.clip(y, 0.0, 1.0)

def binary_anneal(start_h, time_end, rng, n):
    """Simulated annealing on binary vectors (swap moves)."""
    h = (start_h >= 0.5).astype(int)
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    local_start = time.time()
    init_temp = 5e-2
    final_temp = 1e-6
    while time.time() < time_end:
        progress = (time.time() - local_start) / (time_end - local_start)
        T = init_temp * (1 - progress) + final_temp * progress
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        new_c5 = compute_c5(h.astype(float))
        dE = new_c5 - cur_c5
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                if best_c5 <= 0.3808:
                    break
        else:
            h[i], h[j] = 1, 0
        if time.time() >= time_end:
            break
    return best_h.astype(float), best_c5

        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        new_c5 = compute_c5(h.astype(float))
        if new_c5 + 1e-12 < cur_c5:
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                if best_c5 <= 0.3808:
                    break
        else:
            h[i], h[j] = 1, 0
            no_improve += 1
    return best_h.astype(float), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    n = 200                     # number of discretisation points (≤1000)
    target_sum = n / 2.0

    # initialise from parent if available, otherwise random binary
    if 'initial_h_values' in globals():
        small = np.array(initial_h_values, dtype=float)
        x_small = np.linspace(0, 2, small.shape[0], endpoint=False)
        x_big = np.linspace(0, 2, n, endpoint=False)
        h0 = np.interp(x_big, x_small, small)
    else:
        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_sum_bounds(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # deterministic binary seeds
    seeds = []
    rand_bin = np.zeros(n, dtype=int)
    rand_bin[rng.choice(n, n // 2, replace=False)] = 1
    seeds.append(rand_bin.astype(float))

    tm = thue_morse_seq(n)
    tm = adjust_binary_to_sum(tm.astype(int), n // 2, rng)
    seeds.append(tm.astype(float))

    rs = rudin_shapiro_seq(n)
    rs = adjust_binary_to_sum(rs.astype(int), n // 2, rng)
    seeds.append(rs.astype(float))

    # Main search loop
    while time.time() - start_time < budget_s - 0.5:
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 0:
            break
        chunk = min(7.0, remaining * 0.4)
        end_time = time.time() + chunk

        if rng.random() < 0.2:
            start_vec = seeds[rng.integers(len(seeds))]
        else:
            start_vec = best_h.copy()

        r = rng.random()
        if r < 0.4:
            cand_h, cand_c5 = binary_anneal(start_vec, end_time, rng, n)
        elif r < 0.7:
            cand_h, cand_c5 = continuous_anneal(start_vec, end_time, rng, n)
        else:
            cand_h, cand_c5 = random_binary_hill_climb(n, end_time, rng)

        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h, cand_c5
            if best_c5 <= 0.3808:
                break

    # Final refinement with remaining time
    if time.time() - start_time < budget_s - 0.3:
        final_end = start_time + budget_s - 0.1
        final_h, final_c5 = binary_anneal(best_h, final_end, rng, n)
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5

    # enforce feasibility exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    s = best_h.sum()
    if abs(s - target_sum) > 1e-12:
        best_h = project_sum_bounds(best_h, target_sum)
        diff = target_sum - best_h.sum()
        if abs(diff) > 1e-12:
            best_h += diff / n
            best_h = np.clip(best_h, 0.0, 1.0)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
