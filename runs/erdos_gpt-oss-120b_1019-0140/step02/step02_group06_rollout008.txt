def cem_optimize(h0, target, time_budget, rng):
    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    # initial distribution
    mu = h0.copy()
    sigma = np.full(n, 0.3)  # initial std
    # hyperparams
    N_samples = 200
    elite_frac = 0.2
    elite_size = max(1, int(N_samples * elite_frac))
    alpha = 0.7   # update factor for mean and sigma
    min_sigma = 0.01
    start = time.time()
    while time.time() - start < time_budget:
        # generate samples
        samples = rng.normal(loc=mu, scale=sigma, size=(N_samples, n))
        # clip and project
        # allocate list for candidate scores
        scores = np.empty(N_samples)
        proj_samples = np.empty_like(samples)
        for i in range(N_samples):
            s = samples[i]
            s = np.clip(s, 0.0, 1.0)
            s = project_sum_and_bounds(s, target)
            proj_samples[i] = s
            scores[i] = compute_c5(s)
        # sort elite indices
        elite_idx = np.argpartition(scores, elite_size)[:elite_size]
        elite = proj_samples[elite_idx]
        # update mu and sigma
        new_mu = elite.mean(axis=0)
        new_sigma = elite.std(axis=0)
        mu = (1 - alpha) * mu + alpha * new_mu
        sigma = (1 - alpha) * sigma + alpha * new_sigma
        sigma = np.maximum(sigma, min_sigma)
        # update best
        min_idx = scores.argmin()
        if scores[min_idx] < best_c5:
            best_c5 = scores[min_idx]
            best_h = proj_samples[min_idx].copy()
        # early exit if near target
        if best_c5 < 0.38:
            break
    return best_h, best_c5

def local_optimize(h0, target, time_budget, rng):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.3
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # update beta
        beta = min(max_beta, beta * 1.002)
        # compute softmax weights
        vmax = v.max()
        # stable softmax
        exp = np.exp(beta * (v - vmax))
        w = exp / exp.sum()
        # gradient
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # enforce zero sum direction to preserve mass
        g -= g.mean()
        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        # take step
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        # acceptance: always accept improvement, else maybe random
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        # optional random shake if stuck: we can add a check for no improvement jumps.
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        n = len(x)
        # first attempt: distribute uniformly
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # fix remaining diff by scanning
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def local_optimize(h0, target, time_budget, rng):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.3
    no_improve = 0
    while time.time() - start < time_budget:
        # correlation
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # increase smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.002)
        # softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # zero‑sum direction
        g -= g.mean()
        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)

            no_improve = 0
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral (with dx factor)."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        n = len(x)
        # first uniform adjustment
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # fix remaining discrepancy elementwise
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def local_optimize(h0, target, time_budget, rng):
    """Smooth‑max gradient descent."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.3
    no_improve = 0
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if beta < max_beta:
            beta = min(max_beta, beta * 1.002)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        g -= g.mean()          # preserve total mass
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        if no_improve > 2000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
    return best_h, best_c5

    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
