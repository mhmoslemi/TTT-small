def compute_c5_fft(h):
    n = h.shape[0]
    L = 2 * n
    # FFT of h and 1-h padded to L
    fft_h = np.fft.rfft(h, n=L)
    fft_1mh = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fft_h * np.conj(fft_1mh), n=L)
    # corr length L, linear correlation of length 2n-1
    # np.correlate returns length 2n-1. We can ignore last value.
    # Compute max over all k.
    # Use float64
    max_val = np.max(corr[:2*n-1])
    return max_val * (2.0 / n)
```

def hill_climb(h, max_passes=5):
    # repeat pass:
    cur_score = compute_c5_fft(h)
    n = len(h)
    for _ in range(max_passes):
        improved = False
        # iterate over random swap pairs maybe
        for _ in range(5*n):

            i = rng.choice(np.where(h==1)[0])
            j = rng.choice(np.where(h==0)[0])
            # swap
            h[i], h[j] = 0, 1
            new_score = compute_c5_fft(h)
            if new_score < cur_score:
                cur_score = new_score
                improved = True

import time
import numpy as np

# ---------- Utility functions ----------

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def random_binary_balanced(n, rng):
    """Return a binary (0/1) array of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def adjust_to_balance(arr, rng):
    """Force exact n/2 ones by random flips."""
    n = arr.size
    target = n // 2
    cur = arr.copy()
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    diff = ones.size - target
    if diff > 0:
        idx = rng.choice(ones, size=diff, replace=False)
        cur[idx] = 0.0
    elif diff < 0:
        idx = rng.choice(zeros, size=-diff, replace=False)
        cur[idx] = 1.0
    return cur


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx on a uniform grid, via FFT."""
    n = h.shape[0]
    L = 2 * n
    fft_h = np.fft.rfft(h, n=L)
    fft_1mh = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fft_h * np.conj(fft_1mh), n=L)
    max_corr = np.max(corr[:2 * n - 1])
    return max_corr * (2.0 / n)


compute_c5 = compute_c5_fft  # alias used by other helpers


def compute_gradients_for_shifts(h, shifts):
    """Gradient approximation for the max‑correlation term."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)
