def project_sum_and_bounds(x, target):

    x = np.clip(x, 0, 1)

    cur = x.sum()

    diff = target - cur

    # distribute diff equally

    if abs(diff) > 1e-12:

        x += diff / len(x)

        x = np.clip(x, 0, 1)

        cur = x.sum()

        diff = target - cur

        if abs(diff) > 1e-12:

            # adjust one by one

            # if diff > 0, fill from indices with <1

            # else fill from >0.

            for i in range(len(x)):

                if diff <= 0:

                    break

                if diff > 0 and x[i] < 1.0:

                    inc = min(diff, 1.0 - x[i])

                    x[i] += inc

                    diff -= inc

                elif diff < 0 and x[i] > 0.0:

                    dec = min(-diff, x[i])

                    x[i] -= dec

                    diff += dec

    return x

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

    rng = np.random.default_rng(seed)

    n = 200

    target = n/2.0

    # build initial candidate

    h_cur = prepare_initial(n, rng)

    c5_cur = compute_c5(h_cur)

    best_h = h_cur.copy()

    best_c5 = c5_cur

    # Simulated annealing parameters

    T0 = 0.01

    sigma0 = 0.1

    min_sigma = 0.001

    # iteration counters

    it = 0

    no_improve = 0

    # while time left

    while time.time() - start < budget_s - 0.05:  # leave small margin

        elapsed = time.time() - start

        frac = elapsed / budget_s

        T = max(T0 * (1 - frac), 1e-8)   # decays to near zero

        sigma = max(sigma0 * (1 - frac), min_sigma)

        # propose candidate

        h_new = h_cur + rng.normal(scale=sigma, size=n)

        # clip and project

        h_new = np.clip(h_new, 0.0, 1.0)

        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)

        # update best

        if c5_new < best_c5 - 1e-12:

            best_c5 = c5_new

            best_h = h_new.copy()

            no_improve = 0

        else:

            no_improve += 1

        # accept based on SA

        if c5_new < c5_cur:

            h_cur = h_new

            c5_cur = c5_new

        else:

            # probability

            # compute delta = c5_new - c5_cur

            delta = c5_new - c5_cur

            accept_prob = np.exp(-delta / T) if T > 0 else 0.0

            if rng.random() < accept_prob:

                h_cur = h_new

                c5_cur = c5_new

        # occasional restart if stuck

        if no_improve > 2000:

            # random restart

            h_cur = rng.random(n)

            h_cur = project_sum_and_bounds(h_cur, target)

            c5_cur = compute_c5(h_cur)

            no_improve = 0

        it += 1

    # final projection

    best_h = np.clip(best_h, 0.0, 1.0)

    best_h = project_sum_and_bounds(best_h, target)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap value multiplied by dx."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_sum_and_bounds(x, target, tol=1e-12):
    """Clip to [0,1] and enforce sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        # First equal redistribution
        x += diff / x.size
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # Greedy fill/empty if still off
        if abs(diff) > tol:
            for i in range(x.size):
                if diff <= 0:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def upsample_h(old_h, new_n):
    """Linearly interpolate old_h to length new_n."""
    old_n = old_h.size
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h, left=old_h[0], right=old_h[-1])
    return new_h

    # simulated‑annealing‑style hill climbing parameters
    T0 = 0.01          # initial temperature for uphill moves
