def subgradient_max_overlap(h):
    n = len(h)
    v = np.correlate(h, 1.0 - h, mode='full')
    maxv = v.max()
    # find indices of max ops
    max_idxs = np.where(np.abs(v - maxv) < 1e-12)[0]
    # compute gradient as average of each argmax gradient
    g = np.zeros_like(h)
    offset = n - 1
    for idx in max_idxs:
        k = idx - offset
        # get overlapping slices
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
        # grad contributions: derivative of v_k w.r.t h[tail] is (1 - h[head])
        # derivative w.r.t h[head] is -h[tail]
        g[tail] += (1.0 - h[head])
        g[head] -= h[tail]
    # average over count
    g /= len(max_idxs)
    # zero-mean to preserve sum
    g -= np.mean(g)
    return g
```

def gradient_descent_optimize(h0, time_budget, rng):
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    # Parameters
    alpha0 = 0.2
    # no smoothing needed
    while time.time() - start < time_budget:
        cur_c5 = compute_c5(h)
        # subgradient
        g = subgradient_max_overlap(h)
        elapsed = time.time() - start
        remain = time_budget - elapsed
        # step size schedule
        alpha = max(0.01, alpha0 * (remain / time_budget))
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Accept condition
        if c5_new < cur_c5 or rng.random() < 0.01:
            h = h_new
        else:
            # occasionally random perturb to escape
            if rng.random() < 0.02:
                h = project_sum_and_bounds(h + rng.normal(scale=0.01, size=n), target)
    return best_h, best_c5

def subgradient_max_overlap(h):
    n = len(h)
    v = np.correlate(h, 1.0 - h, mode='full')
    maxv = v.max()
    # tolerance for ties
    idxs = np.where(v >= maxv - 1e-12)[0]
    g = np.zeros_like(h)
    offset = n - 1
    for idx in idxs:
        k = idx - offset
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
        g[tail] += (1.0 - h[head])
        g[head] -= h[tail]
    # average subgradient
    g = g / len(idxs)
    # zero sum direction
    g -= np.mean(g)
    return g
```

    n = h.shape[0]
    v = np.correlate(h, 1.0 - h, mode='full')
    maxv = v.max()
    # indices achieving the maximum (allow tiny tolerance)
    idxs = np.where(v >= maxv - 1e-12)[0]
    g = np.zeros_like(h)
    offset = n - 1
    for idx in idxs:
        k = idx - offset
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
        g[tail] += (1.0 - h[head])
        g[head] -= h[tail]
    if len(idxs) > 1:
        g /= len(idxs)
    # remove any component that changes the total sum
    g -= np.mean(g)
    return g

def gradient_descent_optimize(h0, time_budget, rng):
    """Projected subgradient descent with simple step‑size decay."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    alpha0 = 0.3
    while time.time() - start < time_budget:
        cur_c5 = compute_c5(h)
        g = subgradient_max_overlap(h)
        # linear decay of step size
        elapsed = time.time() - start
        remain = max(0.0, time_budget - elapsed)
        alpha = max(0.01, alpha0 * (remain / time_budget))
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        # acceptance rule
        if new_c5 < cur_c5 or rng.random() < 0.01:
            h = h_new
        else:
            # occasional random shake
            if rng.random() < 0.02:
                h = project_sum_and_bounds(h + rng.normal(scale=0.01, size=n), target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = rng.random(n)
    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 1: Cross‑Entropy Method (global exploration)
    cems_time = budget_s * 0.55          # reserve ~45 % for refinement
    N_per_iter = 150                     # samples each iteration
    elite_frac = 0.2
    sigma_min = 0.02
    mu = h0.copy()
    sigma = np.full(n, 0.5)              # start with wide spread

    while True:
        elapsed = time.time() - start
        if elapsed >= cems_time:
            break
        remaining = budget_s - elapsed
        if remaining < 0.2:              # keep enough for final refinement
            break

        # generate candidates
        samples = rng.normal(loc=mu, scale=sigma, size=(N_per_iter, n))
        candidates = []
        for s in samples:
            x = np.clip(s, 0.0, 1.0)
            x = project_sum_and_bounds(x, target)
            c5 = compute_c5(x)
            candidates.append((c5, x))
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.copy()
                # early stop if we already beat the target benchmark
                if best_c5 < 0.3808:
                    break
        # elite selection
        elite_k = max(1, int(N_per_iter * elite_frac))
        candidates.sort(key=lambda t: t[0])
        elite_vecs = np.stack([c[1] for c in candidates[:elite_k]], axis=0)

        # update distribution (exponential smoothing)
        new_mu = elite_vecs.mean(axis=0)
        new_sigma = np.maximum(elite_vecs.std(axis=0), sigma_min)
        mu = 0.9 * mu + 0.1 * new_mu
        sigma = 0.9 * sigma + 0.1 * new_sigma

        # if we already have a very good bound, exit early
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Phase 2: Local refinement via projected subgradient descent
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 0.1:
        refined_h, refined_c5 = gradient_descent_optimize(best_h, remaining, rng)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    # final safety projection
    best_h = project_sum_and_bounds(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, int(n)
