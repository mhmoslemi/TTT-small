def smooth_max_gradient(h, alpha=20.0):
    n = h.size
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    max_corr = corr.max()
    # shift to avoid large exp
    w = np.exp(alpha * (corr - max_corr))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    # gradient: start with ones
    grad = np.ones_like(h)
    # compute contributions for each offset
    # offset mapping: offset = idx - (n-1)
    for idx, offset in enumerate(range(-(n-1), n)):
        weight = w[idx]
        if weight == 0:
            continue
        if offset == 0:
            grad -= 2.0 * weight * h
        elif offset > 0:
            # subtract h_{j+offset} from grad[j] where valid
            grad[:n - offset] -= weight * h[offset:]
            # subtract h_{j-offset} from grad[j]
            grad[offset:] -= weight * h[:n - offset]
        else:  # offset < 0
            off = -offset
            # offset negative => offset = -off
            # subtract h_{j+offset} = h_{j-off} from grad[j] for j>=off
            grad[off:] -= weight * h[:n - off]
            # subtract h_{j-offset} = h_{j+off} from grad[j] for j < n-off
            grad[:n - off] -= weight * h[off:]
    return grad
```

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Max overlap integral approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from length m to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def make_balanced_binary(v):
    """Return a balanced binary vector by thresholding the largest entries."""
    n = v.size
    k = n // 2
    idx = np.argpartition(-v, k - 1)[:k]  # indices of k largest values
    h = np.zeros_like(v)
    h[idx] = 1.0
    return h

def smooth_max_gradient(h, alpha=20.0):
    """Gradient of the smoothed max_{k} correlation, using log‑sum‑exp."""
    n = h.size
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    max_corr = corr.max()
    w = np.exp(alpha * (corr - max_corr))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    grad = np.ones_like(h)                 # constant term from derivative of sum_i h_i
    # offsets correspond to indices - (n-1)
    for idx, offset in enumerate(range(-(n - 1), n)):
        weight = w[idx]
        if weight == 0.0:
            continue
        if offset == 0:
            grad -= 2.0 * weight * h
        elif offset > 0:
            # subtract h_{j+offset}
            grad[: n - offset] -= weight * h[offset:]
            # subtract h_{j-offset}
            grad[offset:] -= weight * h[: n - offset]
        else:  # offset < 0
            off = -offset
            grad[off:] -= weight * h[: n - off]
            grad[: n - off] -= weight * h[off:]
    return grad

def gradient_descent_smooth(h0, max_iters, step0=0.5, beta=0.9,
                            alpha=20.0, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on smoothed max objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        grad = smooth_max_gradient(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climb_binary(h0, max_iters, rng, time_left=lambda: 1e9):
    """Swap‑based hill climbing on balanced binary vectors."""
    h = h0.copy()
    n = h.size
    best_score = compute_c5(h)
    # maintain index lists for 1s and 0s
    ones = np.flatnonzero(h == 1.0)
    zeros = np.flatnonzero(h == 0.0)
    it = 0
    while it < max_iters and time_left() > 0.01:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        candidate = h.copy()
        candidate[i] = 0.0
        candidate[j] = 1.0
        sc = compute_c5(candidate)
        if sc < best_score - 1e-12:
            h = candidate
            best_score = sc
            # update the index sets efficiently
            ones = np.flatnonzero(h == 1.0)
            zeros = np.flatnonzero(h == 0.0)
        it += 1
    return h, best_score

def slsqp_optimize_candidate(h0, max_iter=300):
    """Short SLSQP refinement of a candidate."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [
        {'type': 'eq', 'fun': eq_sum},
        {'type': 'ineq', 'fun': ineq_fun}
    ]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
