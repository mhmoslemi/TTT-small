import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral using FFT-based convolution."""
    n = h.shape[0]
    dx = 2.0 / n
    zh = 1.0 - h
    L = 2 * n - 1
    H = np.fft.rfft(h, L)
    Z = np.fft.rfft(zh, L)
    conv = np.fft.irfft(H * Z, L)
    return np.max(conv) * dx


def _random_initial_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a feasible h with sum = n/2, values in [0,1]."""
    M = n / 2.0
    h = rng.random(n)
    h = h / np.sum(h) * M
    # clip and fix possible overflow/underflow
    overflow = h - 1.0
    if np.any(overflow > 0):
        excess = np.sum(overflow[overflow > 0])
        h[overflow > 0] = 1.0
        under = h < 1.0
        if np.any(under):
            h[under] += excess / np.count_nonzero(under)
    underflow = -h
    if np.any(underflow > 0):
        deficit = np.sum(underflow[underflow > 0])
        h[underflow > 0] = 0.0
        over = h > 0.0
        if np.any(over):
            h[over] -= deficit / np.count_nonzero(over)
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) > 1e-12:
        diff = M - cur
        for i in range(n):
            if diff > 0 and h[i] + diff <= 1.0:
                h[i] += diff
                break
            if diff < 0 and h[i] + diff >= 0.0:
                h[i] += diff
                break
    return h


def _optimise_sa(n: int, init_h: np.ndarray, budget: float, rng: np.random.Generator):
    """Simulated‑annealing search for a low C5 value."""
    h = init_h.copy()
    cur_c5 = _compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    start = time.time()
    T0, T_end = 0.01, 1e-6

    while time.time() - start < budget:
        elapsed = time.time() - start
        frac = elapsed / budget
        T = T0 * (1 - frac) + T_end * frac

        i, j = rng.choice(n, 2, replace=False)
        # random direction of transfer
        if rng.random() < 0.5:
            donor, rec = i, j
        else:
            donor, rec = j, i
        max_tr = min(h[donor], 1.0 - h[rec])
        if max_tr <= 1e-12:
            continue
        delta = rng.random() * max_tr
        h_new = h.copy()
        h_new[donor] -= delta
        h_new[rec] += delta
        c5_new = _compute_c5(h_new)

        accept = False
        if c5_new < cur_c5:
            accept = True
        else:
            if rng.random() < np.exp(-(c5_new - cur_c5) / T):
                accept = True

        if accept:
            h, cur_c5 = h_new, c5_new
            if cur_c5 < best_c5 - 1e-12:
                best_c5, best_h = cur_c5, h.copy()
        # occasional sum re‑projection to avoid drift
        if (elapsed % 1.0) < 0.001:
            s = np.sum(h)
            M = n / 2.0
            if abs(s - M) > 1e-12:
                diff = M - s
                for _ in range(5):
                    idx = rng.integers(n)
                    if diff > 0 and h[idx] + diff <= 1.0:
                        h[idx] += diff
                        break
                    if diff < 0 and h[idx] + diff >= 0.0:
                        h[idx] += diff
                        break
        if best_c5 < 0.3808:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.time()
    rng = np.random.default_rng(seed)

    best_h, best_c5, best_n = None, np.inf, None

    # candidate discretisations (even)
    candidates = [80, 96, 112, 128, 144, 160, 200]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidates:
                candidates.insert(0, n0)
            else:
                candidates = [n0] + [n for n in candidates if n != n0]

    for n in candidates:
        if time.time() - start_total > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_total)
        # allocate at most 180 s per n, otherwise half of what remains
        alloc = min(180.0, remaining * 0.5)

        # initialise from parent if size matches, else from a binary seed
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
        else:
            M = n // 2
            idx = rng.choice(n, M, replace=False)
            h0 = np.zeros(n, dtype=float)
            h0[idx] = 1.0

        # primary SA run
        h_opt, c5_opt = _optimise_sa(n, h0, alloc, rng)

        # secondary run with a continuous random start if time permits
        if time.time() - start_total < budget_s - 5:
            extra = min(alloc * 0.3, 60.0)
            h_cont = _random_initial_h(n, rng)
            h_opt2, c5_opt2 = _optimise_sa(n, h_cont, extra, rng)
            if c5_opt2 < c5_opt:
                h_opt, c5_opt = h_opt2, c5_opt2

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
            if best_c5 < 0.3808:
                break

    # final safety fallback
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = _compute_c5(best_h)
        best_n = n
    else:
        # enforce exact constraints one last time
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = np.sum(best_h)
        M = best_n / 2.0
        if abs(cur - M) > 1e-9:
            diff = M - cur
            for i in range(best_n):
                if diff > 0 and best_h[i] + diff <= 1.0:
                    best_h[i] += diff
                    break
                if diff < 0 and best_h[i] + diff >= 0.0:
                    best_h[i] += diff
                    break
        best_c5 = _compute_c5(best_h)

    return best_h, best_c5, best_n
