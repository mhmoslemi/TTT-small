        # compute final c5

        w[k:] += h[:n - k]

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap C5 for a given step function h (discrete)."""
    n = h.shape[0]
    M = n // 2
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full')
    min_corr = corr.min()
    return (M - min_corr) * dx

def weight_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient weight for shift k: w_i = h_{i+k}+h_{i-k} (zero outside)."""
    n = h.shape[0]
    w = np.zeros(n, dtype=h.dtype)
    if k >= 0:
        w[: n - k] += h[k:]
        w[k:] += h[: n - k]
    else:
        kk = -k
        w[kk:] += h[: n - kk]
        w[: n - kk] += h[kk:]
    return w

def init_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a feasible h with sum = n/2. Use parent if size matches."""
    M = n // 2
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - M) > 1e-9:
            diff = M - cur
            idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
            if idx.size:
                h[idx[0]] += diff
                h = np.clip(h, 0.0, 1.0)
        return h
    # start from uniform 0.5 and add a few random transfers
    h = np.full(n, 0.5, dtype=float)
    steps = rng.integers(5, 15)
    for _ in range(steps):
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] += delta
        h[j] -= delta
    cur = h.sum()
    diff = M - cur
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def greedy_optimize(h: np.ndarray, n: int, rng: np.random.Generator,
                    time_limit: float) -> tuple[np.ndarray, float]:
    """Iteratively improve h by moving mass to increase the worst autocorrelation."""
    start = time.time()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    no_improve = 0
    max_no_improve = 5000

    while time.time() - start < time_limit - 0.01:
        corr = np.correlate(h, h, mode='full')
        min_idx = int(corr.argmin())
        k = min_idx - (n - 1)
        w = weight_for_shift(h, k)

        inc_mask = h < 1.0 - 1e-12
        if not np.any(inc_mask):
            break
        w_inc = w.copy()
        w_inc[~inc_mask] = -np.inf
        i = int(np.argmax(w_inc))

        dec_mask = h > 1e-12
        if not np.any(dec_mask):
            break
        w_dec = w.copy()
        w_dec[~dec_mask] = np.inf
        j = int(np.argmin(w_dec))

        if i == j:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        delta_max = min(1.0 - h[i], h[j])
        if delta_max <= 1e-12:
            # try opposite direction
            i, j = j, i
            delta_max = min(1.0 - h[i], h[j])
            if delta_max <= 1e-12:
                continue
        delta = rng.random() * delta_max
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_c = compute_c5(new_h)
        if new_c + 1e-15 < cur_c:
            h = new_h
            cur_c = new_c
            no_improve = 0
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
                if best_c <= 0.3808:
                    break
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    return best_h, best_c

def search_n(n: int, time_budget: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Repeatedly try random starts and greedy improve until time runs out."""
    best_h = None
    best_c = np.inf
    start = time.time()
    while time.time() - start < time_budget - 0.1:
        h0 = init_h(n, rng)
        remaining = time_budget - (time.time() - start)
        h_opt, c_opt = greedy_optimize(h0, n, rng, remaining)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            if best_c <= 0.3808:
                break
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    target = 0.3808

    best_h_global = None
    best_c_global = np.inf
    best_n_global = None

    candidate_ns = [80, 96, 112, 128, 144, 160]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        alloc = min(200.0, remaining * 0.5)
        h_opt, c_opt = search_n(n, alloc, rng)
        if h_opt is not None and c_opt < best_c_global:
            best_c_global = c_opt
            best_h_global = h_opt.copy()
            best_n_global = n
            if best_c_global <= target:
                break

    if best_h_global is None:
        n = candidate_ns[0]
        best_h_global = np.full(n, 0.5, dtype=float)
        best_c_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        M = n // 2
        cur_sum = best_h_global.sum()
        if abs(cur_sum - M) > 1e-9:
            diff = M - cur_sum
            if diff > 0:
                idx = np.where(best_h_global < 1.0 - 1e-12)[0]
            else:
                idx = np.where(best_h_global > 1e-12)[0]
            if idx.size:
                best_h_global[idx[0]] += diff
                best_h_global = np.clip(best_h_global, 0.0, 1.0)
            best_c_global = compute_c5(best_h_global)

    return best_h_global, best_c_global, best_n_global
