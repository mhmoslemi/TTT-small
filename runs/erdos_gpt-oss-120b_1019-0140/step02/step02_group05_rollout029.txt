import time, numpy as np, cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def local_refine(h0, target_sum, time_limit, rng):
    # local random moves
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    n = len(h0)
    start = time.time()
    while time.time() - start < time_limit:
        # random transfer move
        i, j = rng.choice(n, size=2, replace=False)
        # compute max transfer respecting bounds
        max_tr = min(cur_h[i], 1.0 - cur_h[j])
        if max_tr <= 0:
            continue
        eps = rng.random() * max_tr
        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        # ensure sum unchanged
        new_val = compute_c5(cand)
        if new_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break
        else:
            # occasional acceptance
            # simple probability decreasing with time
            # compute temperature
            elapsed = time.time() - start
            T = max(0.0, 0.01 * (1 - elapsed / time_limit))
            if T > 0 and rng.random() < np.exp(-(new_val - cur_val) / T):
                cur_h = cand
                cur_val = new_val
    # smoothing passes
    for _ in range(5):
        # smoothing kernel
        kernel = np.array([0.25, 0.5, 0.25])
        sm = np.convolve(best_h, kernel, mode='same')
        sm = np.clip(sm, 0.0, 1.0)
        # adjust sum
        cur = np.sum(sm)
        diff = target_sum - cur
        if abs(diff) > 1e-12:
            free = np.where((sm > 0) & (sm < 1))[0]
            if free.size:
                inc = diff / free.size
                sm[free] += inc
                sm = np.clip(sm, 0.0, 1.0)
            # residual fix
            diff = target_sum - np.sum(sm)
            for ii in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and sm[ii] < 1.0:
                    add = min(diff, 1.0 - sm[ii])
                    sm[ii] += add
                    diff -= add
                elif diff < 0 and sm[ii] > 0.0:
                    sub = min(-diff, sm[ii])
                    sm[ii] -= sub
                    diff += sub
            sm = np.clip(sm, 0.0, 1.0)
        sm_val = compute_c5(sm)
        if sm_val + 1e-12 < best_val:
            best_val = sm_val
            best_h = sm
        else:
            break
    # final correction of sum
    cur = np.sum(best_h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        # residual fix
        diff = target_sum - np.sum(best_h)
        for ii in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and best_h[ii] < 1.0:
                add = min(diff, 1.0 - best_h[ii])
                best_h[ii] += add
                diff -= add
            elif diff < 0 and best_h[ii] > 0.0:
                sub = min(-diff, best_h[ii])
                best_h[ii] -= sub
                diff += sub
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_val

    rng = np.random.default_rng(seed)
    # if initial_h_values given explicit

    n = 160
    # LP part, try
    try:
        h_var = cp.Variable(n)

        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]

        for k in range(1, n):

        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-9, reltol=1e-9)

        diff = target_sum - cur_sum

                    h0[i] -= sub
                    diff += sub

        # Straight compute c5
        best_h = h0
        best_val = compute_c5(best_h)
    else:
        # fallback start from initial_h_values if present else random binary with 0.5 density
        if "initial_h_values" in globals():
            best_h = np.array(initial_h_values, dtype=float)
            n = len(best_h)
        else:
            n = 80
            best_h = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            best_h[idx] = 1.0
        # enforce sum exactly
        target_sum = n / 2.0
        cur_sum = np.sum(best_h)
        if abs(cur_sum - target_sum) > 1e-12:
            diff = target_sum - cur_sum
            best_h += diff / n
            best_h = np.clip(best_h, 0.0, 1.0)
        best_val = compute_c5(best_h)

    # local refinement: allocate remaining time
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        # use remaining for local refinement
        target_sum = n / 2.0
        # run local refine for up to remaining - small margin
        refine_time = remaining - 2.0
        best_h, best_val = local_refine(best_h, target_sum, refine_time, rng)

    # final safety clipping and sum correction
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = np.sum(best_h)
    target_sum = n / 2.0
    if abs(cur - target_sum) > 1e-9:
        best_h += (target_sum - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_val, n

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def adjust_sum_exact(h, target):
    """Project h onto the hyperplane sum(h)=target while respecting [0,1]."""
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # distribute uniformly among free entries
    free = np.where((h > 0) & (h < 1))[0]
    if free.size:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    # residual correction
    diff = target - np.sum(h)
    for i in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0.0, 1.0)

def local_refine(start_h, target_sum, time_budget, rng):
    """Simple stochastic hill‑climbing with a few smoothing passes."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    start = time.time()
    while time.time() - start < time_budget:
        i, j = rng.choice(n, size=2, replace=False)
        max_tr = min(cur_h[i], 1.0 - cur_h[j])
        if max_tr <= 0:
            continue
        eps = rng.random() * max_tr
        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        new_val = compute_c5(cand)
        # accept improvements; occasional occasional uphill moves
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = cand, new_val
            if new_val < best_val:
                best_h, best_val = cand.copy(), new_val
                if best_val < 0.3808:
                    break
        else:
            # simple simulated‑annealing acceptance
            elapsed = time.time() - start
            T = max(0.0, 0.01 * (1.0 - elapsed / time_budget))
            if T > 0 and rng.random() < np.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = cand, new_val

    # a few Gaussian smoothing passes
    kernel = np.array([0.25, 0.5, 0.25])
    for _ in range(5):
        sm = np.convolve(best_h, kernel, mode="same")
        sm = np.clip(sm, 0.0, 1.0)
        sm = adjust_sum_exact(sm, target_sum)
        sm_val = compute_c5(sm)
        if sm_val + 1e-12 < best_val:
            best_h, best_val = sm, sm_val
        else:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- grid size -------------------------------------------------
    n = 160                     # a moderate resolution; works within budget
    target_sum = n / 2.0

    # ----- try a convex LP relaxation --------------------------------
    h_opt = None
    try:
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]

        # for each shift k = 1 .. n-1 create McCormick variables
        for k in range(1, n):
            y = cp.Variable(n - k)
            constraints += [
                y <= h_var[: n - k],
                y <= h_var[k:],
                y >= h_var[: n - k] + h_var[k:] - 1,
                y >= 0,
                cp.sum(y) >= t_var,
            ]

        prob = cp.Problem(cp.Maximize(t_var), constraints)
        prob.solve(solver=cp.ECOS, max_iters=20000, abstol=1e-9, reltol=1e-9)
        if prob.status in ("optimal", "optimal_inaccurate"):
            h_opt = np.array(h_var.value).flatten()
            h_opt = np.clip(h_opt, 0.0, 1.0)
            h_opt = adjust_sum_exact(h_opt, target_sum)
    except Exception:
        h_opt = None
