def compute_c5_fft(h):

    # compute full correlation of h and 1-h

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9, eps=1e-12):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        max_corr, shifts, _ = compute_c5_and_shifts(h, eps=eps)
        # compute gradient average for active shifts
        grad = np.zeros_like(h)
        for k in shifts:
            grad += gradient_for_shift(h, k)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1) as float array."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones) as float array."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def correlation_full_fft(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    a = h.astype(np.float64)
    b = 1.0 - a
    size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    corr = np.fft.irfft(A * np.conj(B), n=size)[: 2 * n - 1]
    return corr


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = correlation_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_c5_and_shifts(h, eps=1e-9):
    """Return max correlation (unscaled), list of shifts attaining it, and the correlation array."""
    n = h.shape[0]
    corr = correlation_full_fft(h)
    max_corr = np.max(corr)
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    return max_corr, shifts, corr


def gradient_for_shift(h, shift):
    """Gradient of correlation term for a single shift."""
    n = h.shape[0]
    grad = np.ones_like(h)
    if shift >= 0:
        k = shift
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        k = -shift
        grad[k:] -= h[: n - k]
        grad[: n - k] -= h[k:]
    return grad


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9, eps=1e-9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        max_corr, shifts, _ = compute_c5_and_shifts(h, eps=eps)
        # average gradient over active shifts
        grad = np.zeros_like(h)
        for sh in shifts:
            grad += gradient_for_shift(h, sh)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def round_to_binary(h):
    """Return a binary (0/1) array with exactly half ones, keeping the largest entries."""
    n = h.shape[0]
    idx = np.argpartition(-h, n // 2)[: n // 2]
    out = np.zeros_like(h, dtype=float)
    out[idx] = 1.0
    return out


def hill_climb_binary(h, max_swaps, rng, target=None):
    """Greedy swap‑based hill climbing on a binary vector."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    # pre‑compute positions
    ones = np.flatnonzero(best_h == 1)
    zeros = np.flatnonzero(best_h == 0)
    for _ in range(max_swaps):
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        sc = compute_c5(cand)
        if sc < best_score:
            best_h = cand
            best_score = sc
            ones = np.flatnonzero(best_h == 1)
            zeros = np.flatnonzero(best_h == 0)
            if target is not None and best_score <= target:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    target = 0.3808
    n = 512  # target resolution

    best_h = None
    best_score = float('inf')

    # ---- generate initial candidates ---------------------------------
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        candidates.append(project_onto_box_sum(upsample(init_full, n), n / 2.0))

    candidates.append(project_onto_box_sum(thue_morse(n).astype(float), n / 2.0))

    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))

    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng).astype(float))

    # ---- process each candidate --------------------------------------
    for cand in candidates:
        if time_left() < 5.0:
            break
        # allocate iterations based on remaining time
        max_iters = int(min(20000, max(1000, time_left() * 600)))
        h_opt, sc_opt = gradient_descent_momentum(cand, n_iter=max_iters,
                                                  step0=0.8, beta=0.9)
        if sc_opt < best_score:
            best_h, best_score = h_opt.copy(), sc_opt

        # binary rounding + hill climbing
        if time_left() < 2.0:
            continue
        h_bin = round_to_binary(best_h)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_h, best_score = h_bin.copy(), sc_bin

        max_swaps = int(min(8000, time_left() * 20))
        h_hc, sc_hc = hill_climb_binary(best_h.copy(), max_swaps, rng,
                                        target=target)
        if sc_hc < best_score:
            best_h, best_score = h_hc.copy(), sc_hc

        if best_score <= target:
            break

    # ---- random binary search with hill climbing ----------------------
    while best_score > target and time_left() > 5.0:
        h_rand = random_binary_balanced(n, rng).astype(float)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_h, best_score = h_rand.copy(), sc_rand
        max_swaps = int(min(8000, time_left() * 20))
        h_hc, sc_hc = hill_climb_binary(best_h.copy(), max_swaps, rng,
                                        target=target)
        if sc_hc < best_score:
            best_h, best_score = h_hc.copy(), sc_hc
        if best_score <= target:
            break

    # final guard against drift
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
