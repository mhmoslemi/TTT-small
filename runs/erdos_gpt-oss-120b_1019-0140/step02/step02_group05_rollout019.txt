def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx on the discrete grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_and_fix(h, target_sum):
    """Gaussian‑type smoothing step with exact‑sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    diff = target_sum - np.sum(sm)

                sm[i] -= sub
                diff += sub

                if cur_h[donor] > eps:
                    break
                donor = rng.integers(n)
            if cur_h[donor] <= eps:
                continue
            receiver = rng.integers(n)
            for _ in range(10):
                if cur_h[receiver] < 1.0 - eps:
                    break
                receiver = rng.integers(n)
            if cur_h[receiver] >= 1.0 - eps:
                continue
            max_tr = min(cur_h[donor], 1.0 - cur_h[receiver])
            if max_tr <= 0:
                continue
            delta = max_tr * rng.random()
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[receiver] += delta
        else:
            # ----------- gradient move --------------------------------
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = int(np.argmax(corr))
            k = idx_max - (n - 1)
            neigh = np.zeros(n)
            if k >= 0:
                neigh[: n - k] += cur_h[k:]
                neigh[k:] += cur_h[: n - k]
            else:
                kk = -k
                neigh[kk:] += cur_h[: n - kk]
                neigh[: n - kk] += cur_h[kk:]
            grad = dx * (1.0 - neigh)
            grad_centered = grad - np.mean(grad)
            delta_vec = -grad_centered
            mabs = np.max(np.abs(delta_vec))
            if mabs > 0.2:
                delta_vec = delta_vec * (0.2 / mabs)
            cand = cur_h + delta_vec
            cand = np.clip(cand, 0.0, 1.0)
            cand = enforce_sum(cand, target_sum)

        new_val = compute_c5(cand)
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0:
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            cur_h = cand
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                if best_val < 0.3808:
                    break
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # perturb around the current best to escape stagnation
            cand = best_h.copy()
            for _ in range(5):
                donor = rng.integers(n)
                while cand[donor] <= eps:
                    donor = rng.integers(n)
                receiver = rng.integers(n)
                while cand[receiver] >= 1.0 - eps:
                    receiver = rng.integers(n)
                max_tr = min(cand[donor], 1.0 - cand[receiver])
                if max_tr > 0:
                    delta = max_tr * rng.random()
                    cand[donor] -= delta
                    cand[receiver] += delta
            cand = enforce_sum(cand, target_sum)
            cur_h = cand
            cur_val = compute_c5(cur_h)
            no_improve = 0

    # ----- optional smoothing passes ---------------------------------
    if time.time() - start < budget_s - 5.0:
        for _ in range(10):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
            else:
                break

    # ----- final safety correction -----------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(np.sum(best_h) - target_sum) > 1e-9:
        best_h = enforce_sum(best_h, target_sum)

    return best_h, best_val, n
