def anneal_optimize(n, time_limit, rng):
    M = n // 2
    # initial h
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        # adjust sum exactly:
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)
    else:
        h = random_feasible_h(n, M, rng)
    # ensure within bounds and sum
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    # SA parameters
    start = time.time()
    # set initial temperature relative to c5 scale
    T0 = 0.05 * c5  # maybe 5% of c5
    Tmin = 1e-6
    # schedule linear cooling over time budget
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = max(Tmin, T0 * (1 - frac))
        # propose move
        i, j = rng.choice(n, 2, replace=False)
        # direction: from j to i (increase i, decrease j)
        # max transfer limited by h[j] and 1-h[i]
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0:
            continue
        # choose delta scaled by temperature: proportion of max_transfer
        # For high T, allow larger deltas; for low T, smaller.
        # Let delta = rng.random() * max_transfer * (T / T0)
        # Ensure at least minimal delta
        scale = T / T0 if T0 > 0 else 1.0
        delta = rng.random() * max_transfer * max(scale, 0.01)  # maybe cap at 0.1?
        # apply temporarily
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        delta_c5 = c5_new - c5
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / (T + 1e-12)):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                # early stop if target reached
                if best_c5 < 0.3808:
                    break
    return best_h, best_c5

def anneal_optimize(n, time_limit, rng):
    M = n // 2
    # initial h: check parent
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)
    else:
        # generate random feasible h
        # generate uniform random and then adjust
        # approach: start with random values, project onto [0,1] and adjust sum
        h = rng.random(n)
        # scale to sum M via linear scaling and clipping
        # using simple algorithm like in previous code:
        h = h / h.sum() * M
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - M) > 1e-12:
            # adjust via water-filling
            diff = M - cur
            # add diff/n equally then reclip
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            # correction again
            cur = h.sum()
            if abs(cur - M) > 1e-12:
                # fallback to simple iterative approach to meet sum
                # Use smaller adjustments
                for _ in range(10):
                    cur = h.sum()
                    diff = M - cur
                    if abs(diff) < 1e-12:
                        break
                    h += diff / n
                    h = np.clip(h, 0.0, 1.0)
    # ensure within bounds
    h = np.clip(h, 0.0, 1.0)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    start = time.time()
    # parameters
    T0 = 0.02  # initial temperature
    Tmin = 1e-6
    # compute approximate number of iterations? We'll loop until time
    while True:
        elapsed = time.time() - start
        if elapsed > time_limit - 0.01:
            break
        # temperature schedule
        frac = elapsed / time_limit
        T = max(Tmin, T0 * (1 - frac))
        # propose swap
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # direction: move from j to i
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0:

        delta = rng.random() * max_transfer
        # if T not zero, we could shape delta with T but not needed

        h[j] -= delta
        # compute new c5
        c5_new = compute_c5(h)

        accept = False

                if best_c5 < 0.3808:
                    # early exit
                    break
        else:
            # revert move
            h[i] -= delta
            h[j] += delta
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Constraints for SLSQP ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        # x includes h and auxiliary t at the end
        return np.sum(x[: self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        # constraint: t - overl*dx >= 0
        return x[-1] - overl * self.dx


# ---------- Utility ----------
def compute_c5(h):
    """Maximum overlap integral for a (continuous) step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def generate_random_h(n, M, rng):
    """Random feasible h with sum M and 0≤h≤1."""
    # start from uniform random vector
    h = rng.random(n)
    h = h / np.sum(h) * M
    h = np.clip(h, 0.0, 1.0)
    # iterative water‑filling to hit exact sum
    for _ in range(5):
        cur = h.sum()
        diff = M - cur
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


# ---------- Simulated annealing ----------
def simulated_annealing(h0, M, time_limit, rng):
    """SA moves mass between two coordinates while keeping sum=M."""
    h = h0.copy()
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    start = time.time()
    T0 = max(0.01, c5 * 0.1)      # initial temperature
    Tmin = 1e-6

    while True:
        elapsed = time.time() - start
        if elapsed > time_limit - 0.01:
            break
        frac = elapsed / time_limit
        T = max(Tmin, T0 * (1.0 - frac))
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        # apply move
        h[i] += delta
        h[j] -= delta
        c5_new = compute_c5(h)
        delta_c5 = c5_new - c5
        accept = False
        if delta_c5 < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta_c5 / (T + 1e-12)):
                accept = True

        if accept:
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                if best_c5 < 0.3808:
                    break
        else:
            # revert
            h[i] -= delta
            h[j] += delta
    return best_h, best_c5


# ---------- SLSQP refinement ----------
def slsqp_refine(h_init, M, time_limit, rng):
    """Single SLSQP run with auxiliary variable t."""
    n = len(h_init)
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{"type": "eq", "fun": sum_con}]
    constraints += [{"type": "ineq", "fun": oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    t0 = compute_c5(h_init)
    x0 = np.concatenate([h_init, [t0]])

    try:
        res = minimize(lambda x: x[-1],
                       x0,
                       method="SLSQP",
                       bounds=bounds,
                       constraints=constraints,
                       options={"maxiter": 400, "ftol": 1e-10, "disp": False})
    except Exception:
        return h_init, compute_c5(h_init)

    if not res.success:
        return h_init, compute_c5(h_init)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = h_opt.sum()
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)


# ---------- Local pairwise refinement ----------
def local_refine(h_start, M, time_limit, rng):
    """Greedy pairwise mass transfer improving C5."""
    n = len(h_start)
    cur_h = h_start.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()

    while True:
        if time.time() - start > time_limit - 0.01:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(1.0 - cur_h[i], cur_h[j])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        cur_h[i] += delta
        cur_h[j] -= delta
        new_c5 = compute_c5(cur_h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                if best_c5 < 0.3808:
                    break
        else:
            cur_h[i] -= delta
            cur_h[j] += delta
    return best_h, best_c5


# ---------- Optimisation for a given discretisation ----------
def optimise_n(n, budget, rng):
    """Run SA → SLSQP → local refinement within allocated budget."""
    M = n // 2
    # initial vector
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        h0 = np.array(initial_h_values, dtype=float)
        cur = h0.sum()
        if abs(cur - M) > 1e-12:
            h0 += (M - cur) / n
            h0 = np.clip(h0, 0.0, 1.0)
    else:
        h0 = generate_random_h(n, M, rng)

    # time split
    start = time.time()
    t_sa = budget * 0.5
    t_slsqp = budget * 0.3
    t_local = max(0.0, budget - t_sa - t_slsqp)

    # 1) Simulated annealing
    h_best, c5_best = simulated_annealing(h0, M, t_sa, rng)

    # 2) SLSQP refinement
    if t_slsqp > 0.5:
        h_ref, c5_ref = slsqp_refine(h_best, M, t_slsqp, rng)
        if c5_ref < c5_best:
            h_best, c5_best = h_ref, c5_ref

    # 3) Local pairwise refinement
    if t_local > 0.5:
        h_loc, c5_loc = local_refine(h_best, M, t_local, rng)
        if c5_loc < c5_best:
            h_best, c5_best = h_loc, c5_loc

    return h_best, c5_best


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidate_ns = [80, 96, 112, 128, 144]

    # prioritize parent size if present
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        remaining = budget_s - elapsed
        # allocate at most 250 s per resolution, otherwise use half of remaining
        alloc = min(250.0, remaining * 0.5)
        h_opt, c5_opt = optimise_n(n, alloc, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n
            if best_c5 < 0.3808:
                break

    # fallback if everything failed
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # final safety fixes
    best_h = np.clip(best_h, 0.0, 1.0)
    if best_n is not None:
        M = best_n // 2
        cur = best_h.sum()
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / best_n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
