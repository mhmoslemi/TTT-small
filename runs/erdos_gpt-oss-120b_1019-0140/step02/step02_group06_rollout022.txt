            w /= w_sum

        # gradient of the smooth objective
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # enforce zero‑sum direction (preserve total mass)
        g -= np.mean(g)

        # step‑size schedule (linear decay)
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        # occasional random shake when stuck
        if no_improve > 5000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.015, size=n), target)
            no_improve = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function (sampled values), its C5 bound, and grid size."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # Choose a resolution that balances quality and speed.
    n_target = 160                     # 2 × the parent resolution
    target_sum = n_target / 2.0

    # Build initial feasible vector.
    if "initial_h_values" in globals():
        h0 = upsample_initial(initial_h_values, n_target)
    else:
        h0 = rng.random(n_target)
        h0 = project_sum_and_bounds(h0, target_sum)

    # Split the remaining budget: majority for global search, rest for local refinement.
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    time_cem = remaining * 0.6
    h_cem, _ = cem_optimize(h0, target_sum, time_cem, seed + 1)

    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    h_ref, _ = gradient_refine(h_cem, target_sum, remaining, seed + 2)

    # Final safeguard projection and evaluation.
    h_final = np.clip(h_ref, 0.0, 1.0)
    h_final = project_sum_and_bounds(h_final, target_sum)
    final_c5 = compute_c5(h_final)

    return h_final, final_c5, n_target
