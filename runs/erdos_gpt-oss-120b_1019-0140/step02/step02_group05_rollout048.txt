def initialize(rng):
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        factor = 2
        n = n0 * factor
        # interpolation
        old_x = np.linspace(0, 2, n0, endpoint=False)
        new_x = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(new_x, old_x, h0)
    else:
        n = 120   # default
        h = np.zeros(n)
        # random half fill
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
    # clip
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    target = n/2.0
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        h += (target - cur) / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        # fix residual
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
        h = np.clip(h, 0.0, 1.0)
    return h

                conv_minus[:n - s] += wk * h[s:]

        grad = dx * (sum_w - (conv_plus + conv_minus))

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_sum(h, target):
    """Clip to [0,1] and enforce exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        for i in range(h.shape[0]):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_fix(h, target):
    """One Gaussian‑type smoothing pass with sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return _project_sum(sm, target)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        factor = 2
        n = n0 * factor
        old_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(new_x, old_x, h0)
    else:
        n = 120
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    target_sum = n / 2.0
    h = _project_sum(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    dx = 2.0 / n
    # ----- continuation gradient descent -------------------------------
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0]
    step0 = dx * 5.0

    for beta in betas:
        for it in range(500):
            if time.time() - start >= budget_s * 0.85:
                break

            corr = np.correlate(h, 1.0 - h, mode="full")
            O = corr * dx
            cur_val = O.max()
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val < 0.3808:
                    # keep searching a little longer for refinement
                    pass

            maxO = O.max()
            w = np.exp(beta * (O - maxO))
            w /= w.sum()

            conv_plus = np.zeros(n)
            conv_minus = np.zeros(n)
            # compute weighted shifted sums
            for shift in range(-(n - 1), n):
                wk = w[shift + n - 1]
                if wk == 0.0:
                    continue
                if shift >= 0:
                    conv_plus[: n - shift] += wk * h[shift:]
                    conv_minus[shift:] += wk * h[: n - shift]
                else:
                    s = -shift
                    conv_plus[s:] += wk * h[: n - s]
                    conv_minus[: n - s] += wk * h[s:]

            grad = dx * (1.0 - (conv_plus + conv_minus))
            step = step0 / np.sqrt(it + 1.0)
            h = h - step * grad
            h = _project_sum(h, target_sum)

    # ----- optional smoothing passes ---------------------------------
    if time.time() - start < budget_s - 5.0:
        for _ in range(5):
            sm = _smooth_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm.copy()
            else:
                break

    # ----- final SLSQP refinement ------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 10.0:
        n = best_h.shape[0]
        dx = 2.0 / n
        target_sum = n / 2.0

        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

        constraints = [{"type": "eq", "fun": sum_eq}]
        for kk in range(-(n - 1), n):
            if kk == 0:
                continue

            def make_fun(k):
                def fun(x):
                    h = x[:n]
                    if k >= 0:
                        overl = np.dot(h[: n - k], 1.0 - h[k:])
                    else:
                        s = -k
                        overl = np.dot(h[s:], 1.0 - h[: n - s])
                    return x[-1] - overl * dx
                return fun

            constraints.append({"type": "ineq", "fun": make_fun(kk)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 600, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = _project_sum(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt.copy()
        except Exception:
            pass

    # ----- final safety correction ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum(best_h, target_sum)

    return best_h, best_val, n
