def adam_optimize(h0, n_iter, lr, beta1, beta2, eps, time_check):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter+1):
        if not time_check():
            break
        # compute gradient for worst shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        # bias correction
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_swap_hill_climb(h_bin, max_iters, rng, time_check):
    # h_bin is binary 0-1 array sum = n/2
    n = len(h_bin)
    best_h = h_bin.copy()
    best_score = compute_c5(h_bin)
    for it in range(max_iters):
        if not time_check():
            break
        # compute corr and max shift
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        # Find index sets.
        if shift >= 0:
            bad = np.where((best_h[:n-shift] == 1) & (best_h[shift:] == 0))[0]
            good = np.where((best_h[:n-shift] == 0) & (best_h[shift:] == 1))[0]
            # shift index mapping: bad positions refer to original i where h[i]==1 and h[i+shift]==0.
            # good positions refer to i where h[i]==0 and h[i+shift]==1.
        else:
            s = -shift
            bad = np.where((best_h[s:] == 1) & (best_h[:n-s] == 0))[0] + s
            good = np.where((best_h[s:] == 0) & (best_h[:n-s] == 1))[0] + s
        if bad.size == 0 or good.size == 0:
            # cannot improve via swap for this shift; maybe try random
            # Randomly pick any 1 and any 0 and swap; but may not improve
            # We'll break if stuck
            continue
        # Randomly select a pair
        i_from = rng.choice(bad)
        i_to = rng.choice(good)
        # Perform swap
        new_h = best_h.copy()
        new_h[i_from] = 0
        new_h[i_to] = 1
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
            # Continue - maybe multiple improvements
        # else maybe accept rarely? We'll ignore.
        # Check early exit if bound reached
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    start = time.time()
    rng = np.random.default_rng(seed)

    def time_left():
        return budget_s - (time.time() - start)

    def time_check():
        return time_left() > 0.01  # minimal

    n = 256  # target resolution
    dx = 2.0 / n

    best_h = None
    best_score = float('inf')

    # Candidate generation
    candidates = []

    # initial_h_values if present
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        # upsample to n
        # define upsample interpolation
        def upsample(arr, new_n):
            m = arr.shape[0]
            if m == new_n:
                return arr.copy()
            x_old = np.linspace(0, 1, m, endpoint=False)
            x_new = np.linspace(0, 1, new_n, endpoint=False)
            return np.interp(x_new, x_old, arr)
        cand = upsample(init_full, n)
        # ensure bounds [0,1] (interp between 0 and1, should be within)
        cand = project_onto_box_sum(cand, n/2.0)
        candidates.append(cand)

    # Thue-Morse binary
    candidates.append(thue_morse(n).astype(float))

    # random binary balanced 2-3 times
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))

    # random continuous vectors
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n/2.0))

    # maybe start from smooth sinusoid
    x = np.linspace(0, 2, n, endpoint=False)
    sinusoid = 0.5 + 0.5 * np.sin(2*np.pi*x/2)  # period 2
    candidates.append(project_onto_box_sum(sinusoid, n/2.0))

    # Process each candidate within time
    for cand in candidates:
        if not time_check():
            break
        # limit iterations based on remaining time and n
        remaining = time_left()
        # guess ~0.001 per iteration, allocate up to 20000 or leftover/0.001
        max_iters = min(20000, int(remaining * 800))  # ~800 iterations per second
        h_opt, sc_opt = adam_optimize(cand, n_iter=max_iters, lr=0.05,
                                      beta1=0.9, beta2=0.999, eps=1e-8,
                                      time_check=time_check)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        # If already good enough, break
        if best_score <= 0.3808:
            break

    # After continuous optimization, try binary rounding and swap hill climb
    if best_h is not None:
        # Round to binary by threshold .5
        h_bin = (best_h >= 0.5).astype(float)
        # Ensure sum correct: adjust by swapping nearest to .5 if needed.
        current_sum = h_bin.sum()
        target_sum = n/2
        if current_sum > target_sum:
            # need to turn some 1's to 0; choose those with smallest value (closest to .5)
            ones_idx = np.where(h_bin == 1)[0]
            # values in original best_h for those indices
            vals = best_h[ones_idx]
            # sort by ascending (smallest first)
            sort_idx = np.argsort(vals)  # smallest first
            to_flip = int(current_sum - target_sum)
            flip_idx = ones_idx[sort_idx[:to_flip]]
            h_bin[flip_idx] = 0
        elif current_sum < target_sum:
            zeros_idx = np.where(h_bin == 0)[0]
            vals = best_h[zeros_idx]
            sort_idx = np.argsort(-vals)  # largest first
            to_flip = int(target_sum - current_sum)
            flip_idx = zeros_idx[sort_idx[:to_flip]]
            h_bin[flip_idx] = 1
        # Final projection not needed because binary sum exact.

        # Run hill climbing
        remaining = time_left()
        max_hill_iters = int(remaining * 400)  # about 400 per second
        h_bin_opt, sc_bin_opt = binary_swap_hill_climb(h_bin, max_hill_iters,
                                                       rng, time_check)
        if sc_bin_opt < best_score:
            best_score, best_h = sc_bin_opt, h_bin_opt.copy()

    # Final projection to ensure constraints (floating may deviate a bit)
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, n/2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, n

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Average sub‑gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def thue_morse(n):
    """Binary Thue–Morse sequence of length n (values 0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = bin(i).count('1') & 1
    return seq


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample(arr, new_n):
    """Linear interpolation of arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def adam_optimize(h0, n_iter, lr, beta1, beta2, eps, start_time, budget_s):
    """Adam‑style projected gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for t in range(1, n_iter + 1):
        if time.time() - start_time >= budget_s:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score


def binary_swap_hill_climb(h0, max_iters, rng, start_time, budget_s):
    """Swap‑based hill climbing on a binary vector."""
    n = h0.shape[0]
    best_h = h0.copy()
    best_score = compute_c5(best_h)

    for _ in range(max_iters):
        if time.time() - start_time >= budget_s:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift == 0:
            continue

        if shift > 0:
            bad = np.where((best_h[:n - shift] == 1) & (best_h[shift:] == 0))[0]
            good = np.where((best_h[:n - shift] == 0) & (best_h[shift:] == 1))[0]
        else:
            s = -shift
            bad = np.where((best_h[s:] == 1) & (best_h[:n - s] == 0))[0] + s
            good = np.where((best_h[s:] == 0) & (best_h[:n - s] == 1))[0] + s

        if bad.size == 0 or good.size == 0:
            continue

        i_from = rng.choice(bad)
        i_to = rng.choice(good)

        new_h = best_h.copy()
        new_h[i_from] = 0.0
        new_h[i_to] = 1.0

        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
            if best_score <= 0.3808:
                break

    return best_h, best_score

            start_time=start,
            budget_s=budget_s,
        )
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # ----------- Binary rounding + swap hill climbing ----------
    if best_h is not None and time.time() - start < budget_s:
        # Round while preserving the exact half‑mass constraint
        h_bin = (best_h >= 0.5).astype(float)
        cur = int(h_bin.sum())
        target = h_bin.size // 2
        if cur != target:
            if cur > target:
                ones_idx = np.where(h_bin == 1)[0]
                vals = best_h[ones_idx]
                order = np.argsort(vals)  # smallest first
                to_flip = cur - target
                h_bin[ones_idx[order[:to_flip]]] = 0.0
            else:
                zeros_idx = np.where(h_bin == 0)[0]
                vals = best_h[zeros_idx]
                order = np.argsort(-vals)  # largest first
                to_flip = target - cur
                h_bin[zeros_idx[order[:to_flip]]] = 1.0

        # Swap‑based refinement
        remaining = budget_s - (time.time() - start)
        max_swaps = int(remaining * 300)
        if max_swaps > 0:
            h_bin_opt, sc_bin_opt = binary_swap_hill_climb(
                h_bin, max_swaps, rng, start, budget_s
            )
            if sc_bin_opt < best_score:
                best_score, best_h = sc_bin_opt, h_bin_opt.copy()

    # ----------- Optional up‑sampling to 512 points ----------
    if best_h is not None and budget_s - (time.time() - start) > 5.0:
        n2 = 512
        h2 = upsample(best_h, n2)
        h2 = project_onto_box_sum(h2, n2 // 2)
        remaining = budget_s - (time.time() - start)
        est_iters2 = int(min(5000, remaining * 300))
        if est_iters2 > 0:
            h2_opt, sc2_opt = adam_optimize(
                h2,
                n_iter=est_iters2,
                lr=0.03,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                start_time=start,
                budget_s=budget_s,
            )
            if sc2_opt < best_score:
                best_score, best_h = sc2_opt, h2_opt.copy()

    # ----------- Final projection and evaluation -------------
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size // 2)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
