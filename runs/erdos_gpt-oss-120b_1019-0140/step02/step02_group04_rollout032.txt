import time
import numpy as np

# ---------- Helper Functions ----------
def compute_c5_vec(h, dx):
    """Return max overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Project vec onto the feasible set: 0<=vec<=1, sum(vec)=target_sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    # distribute remaining difference greedily
    for i in range(len(vec)):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible continuous vector (values in [0,1] and exact sum)."""
    x = rng.random(n)
    x = x / np.sum(x) * target_sum
    x = np.clip(x, 0.0, 1.0)
    return project_to_sum(x, target_sum)


def refine_continuous(h, dx, target_sum, time_limit, start_time, rng):
    """Local coordinate descent transferring mass between entries."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.05
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val
