def project_to_feasible(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # identify free: where 0 < h < 1
    free = np.where((h > 0) & (h < 1))[0]
    if free.size > 0:
        # distribute diff evenly
        inc = diff / free.size
        h[free] += inc
        # clip again
        h = np.clip(h, 0.0, 1.0)
    # recompute diff and fix residual greedily
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # adjust sequentially
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h
```

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_feasible(h, target):
    """Clip to [0,1] and enforce exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)

def smooth_and_fix(h, target):
    """Mild Gaussian smoothing followed by exact-sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    sm = _project_to_feasible(sm, target)
    return sm

def _neighbor_sum(h, k):
    n = len(h)
    out = np.zeros_like(h)
    if k >= 0:
        out[: n - k] += h[k:]
        out[k:] += h[: n - k]
    else:
        kk = -k
        out[kk:] += h[: n - kk]
        out[: n - kk] += h[kk:]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---------- choose discretisation ----------
    n = 200                     # number of samples (≤1000)
    dx = 2.0 / n
    target_sum = n / 2.0        # integral constraint

        if base.size != n:

            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            base = np.interp(x_new, x_old, base)
        base = _project_to_feasible(base, target_sum)
        inits.append(base)

    # a few random starts (including a binary start)
    for _ in range(5):
        h = rng.random(n)
        h = _project_to_feasible(h, target_sum)
        inits.append(h)

    best_val = np.inf

    # ---------- subgradient descent from each start ----------
    for h0 in inits:

        step0 = 5.0
        for it in range(4000):
            # time guard
            if time.time() - start > budget_s * 0.85:
                break

            # compute overlap values O_k
            corr = np.correlate(h, 1.0 - h, mode="full")
            O = corr * dx

            val = O[idx_max]

        best_h = _project_to_feasible(best_h, target_sum)
