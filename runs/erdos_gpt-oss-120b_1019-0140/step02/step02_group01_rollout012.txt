    H = np.fft.rfft(h, n*2)

def anneal_optimize(h, time_budget, rng):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val
    start = time.time()
    # parameters
    T0 = 0.02 # may adjust
    # schedule: T = T0 * (1 - (elapsed / time_budget))**2 maybe
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(1e-4, T0 * (1 - elapsed / time_budget))
        # propose move
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - cur_h[i], cur_h[j])
        if max_transfer <= 0:
            continue
        # Use random fraction
        delta = rng.random() * max_transfer
        # apply
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = compute_c5(new_h)
        d = new_val - cur_val
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = new_val
    return best_h, best_val

def repair_sum(h):
    n = len(h)
    M = n // 2
    cur = np.sum(h)
    # adjust by adding uniform epsilon:
    diff = M - cur
    if abs(diff) < 1e-12:
        return np.clip(h,0,1)
    # distribute diff evenly while respecting bounds.
    h = h + diff / n
    # clip
    np.clip(h,0,1,inplace=True)
    # after clipping may have leftover diff; fix iteratively
    # compute new diff
    cur = np.sum(h)
    diff = M - cur
    # if diff >0, need to add more to entries not at 1
    if diff > 1e-12:
        # indices sorted by h (ascending) where we can add
        idx = np.argsort(h)
        for i in idx:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < -1e-12:
        idx = np.argsort(-h)  # descending
        for i in idx:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return h
```

def optimize_n_anneal(n, start_h, time_budget, rng):
    best_h = None
    best_val = np.inf
    start_time = time.time()
    # We'll allocate multiple restarts by splitting time_budget equally.
    # Let's allocate restarts of 6, each with time_budget/6.
    num_restarts = max(1, int(time_budget // 30))  # each at least 30 secs
    per_restart_time = time_budget / num_restarts
    for r in range(num_restarts):
        if time.time() - start_time > time_budget - 0.5:

        if r == 0:
            h0 = start_h.copy()

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # compute cross-correlation using FFT
    H = np.fft.rfft(h, 2 * n)
    G = np.fft.rfft(1.0 - h, 2 * n)
    conv = np.fft.irfft(H * np.conj(G), 2 * n)
    # get linear convolution length 2n-1
    conv = conv[:2 * n - 1]
    max_val = np.max(conv)
    return max_val * dx

def repair_sum(h):
    n = h.shape[0]
    M = n // 2
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) < 1e-12:
        # just clip to [0,1]
        np.clip(h, 0.0, 1.0, out=h)
        return h
    # distribute diff uniformly
    h += diff / n
    np.clip(h, 0.0, 1.0, out=h)
    cur = np.sum(h)
    diff = M - cur
    if diff > 1e-12:
        # need to add mass
        idx = np.argsort(h)
        for i in idx:
            inc = min(diff, 1.0 - h[i])
            if inc <= 0:
                continue
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < -1e-12:
        idx = np.argsort(-h)
        for i in idx:
            dec = min(-diff, h[i])
            if dec <= 0:
                continue
            h[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    np.clip(h, 0.0, 1.0, out=h)
    return h
```

def anneal_optimize(h0, time_budget, rng):
    n = h0.shape[0]
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    T0 = 0.02  # initial temperature
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(1e-5, T0 * (1.0 - elapsed / time_budget))
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - cur_h[i], cur_h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = compute_c5(new_h)
        d = new_val - cur_val
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = new_val
                if best_val < 0.3808:
                    break
    return best_h, best_val

def optimize_n_anneal(n, start_h, time_budget, rng):

    best_val = np.inf

    min_restart_time = 30.0

    if max_restarts < 1:
        max_restarts = 1

    for r in range(max_restarts):
        if time.time() - start > time_budget - 0.5:
            break
        if r == 0:
            h0 = start_h.copy()
        else:

    return np.max(conv) * dx

def repair_sum(h):
    n = h.shape[0]
    M = n // 2
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) < 1e-12:
        np.clip(h, 0.0, 1.0, out=h)
        return h
    h += diff / n
    np.clip(h, 0.0, 1.0, out=h)
    cur = np.sum(h)
    diff = M - cur
    if diff > 1e-12:
        idx = np.argsort(h)
        for i in idx:
            inc = min(diff, 1.0 - h[i])
            if inc <= 0:
                continue
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < -1e-12:
        idx = np.argsort(-h)
        for i in idx:
            dec = min(-diff, h[i])
            if dec <= 0:
                continue
            h[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    np.clip(h, 0.0, 1.0, out=h)
    return h

def upscale(h, n_new):
    n_old = h.shape[0]
    t_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    t_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(t_new, t_old, h)
    return repair_sum(h_new)

def anneal_optimize(h0, time_budget, rng):
    n = h0.shape[0]
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    T0 = 0.05
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(1e-5, T0 * (1.0 - elapsed / time_budget))
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - cur_h[i], cur_h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = compute_c5(new_h)
        d = new_val - cur_val
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = new_val
                if best_val < 0.3808:
                    break
    return best_h, best_val

def optimize_n_anneal(n, start_h, time_budget, rng):
    best_h = None
    best_val = np.inf
    start = time.time()
    min_restart = 30.0
    max_restarts = max(1, int(time_budget // min_restart))
    per_restart = time_budget / max_restarts
    for r in range(max_restarts):
        if time.time() - start > time_budget - 0.5:
            break
        if r == 0:
            h0 = start_h.copy()
        else:
            M = n // 2
            h0 = np.zeros(n)
            idx = rng.choice(n, M, replace=False)
            h0[idx] = 1.0
            noise = rng.random(n) * 0.1
            h0 = np.clip(h0 + noise, 0.0, 1.0)
            h0 = repair_sum(h0)
        h_opt, val_opt = anneal_optimize(h0, per_restart, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt
        if best_val < 0.3808:
            break
    return best_h, best_val

def refine_local(h0, time_budget, rng):
    n = h0.shape[0]
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_budget:
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - best_h[i], best_h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        val_new = compute_c5(new_h)
        if val_new + 1e-12 < best_val:
            best_h = new_h
            best_val = val_new
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h_global = None
    best_val_global = np.inf
    best_n_global = None
    # candidate resolutions
    candidate_ns = [80, 120, 200, 280, 360]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns = [n0] + candidate_ns
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]
    prev_h = None
    prev_n = None
    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:
            break
        alloc = min(200.0, remaining * 0.5)
        if prev_h is not None and prev_n is not None and prev_n < n:
            start_h = upscale(prev_h, n)
        elif 'initial_h_values' in globals() and len(initial_h_values) == n:
            start_h = np.array(initial_h_values, dtype=float)
            start_h = repair_sum(start_h)
        else:
            M = n // 2
            start_h = np.zeros(n)
            idx = rng.choice(n, M, replace=False)
            start_h[idx] = 1.0
            start_h = repair_sum(start_h)
        h_opt, val_opt = optimize_n_anneal(n, start_h, alloc, rng)
        if h_opt is not None and val_opt < best_val_global:
            best_h_global = h_opt
            best_val_global = val_opt
            best_n_global = n
        prev_h = h_opt if h_opt is not None else prev_h
        prev_n = n
        if best_val_global < 0.3808:
            break
    if best_h_global is not None:
        remaining = budget_s - (time.time() - start_time)
        if remaining > 5:
            h_ref, val_ref = refine_local(best_h_global, remaining, rng)
            if val_ref < best_val_global:
                best_h_global = h_ref
                best_val_global = val_ref
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        best_h_global = repair_sum(best_h_global)
        best_val_global = compute_c5(best_h_global)
        best_n_global = len(best_h_global)
    else:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_n_global = n
        best_val_global = compute_c5(best_h_global)
    return best_h_global, best_val_global, best_n_global
