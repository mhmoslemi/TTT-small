def binary_hill_climb_targeted(h_bin, dx, time_limit, rng):
    h = h_bin.astype(float).copy()
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5_vec(h, dx)
    zero_idx = n - 1
    while time.time() < time_limit:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr[zero_idx] = -np.inf
        max_idx = int(np.argmax(corr))
        max_val = corr[max_idx]
        # If max_val is already <= best_val (should be same), but we can stop if no improvement possible:
        # Try move
        shift = max_idx - zero_idx
        donors = np.array([], dtype=int)
        receivers = np.array([], dtype=int)
        if shift > 0:
            s = shift
            mask_i = slice(0, n - s)
            mask_j = slice(s, n)
            donors_mask = (h[mask_i] > 0.5) & (h[mask_j] < 0.5)
            receivers_mask = (h[mask_i] < 0.5) & (h[mask_j] > 0.5)
            donors = np.where(donors_mask)[0]
            receivers = np.where(receivers_mask)[0]
        elif shift < 0:
            s = -shift
            mask_i = slice(s, n)
            mask_j = slice(0, n - s)
            donors_mask = (h[mask_i] > 0.5) & (h[mask_j] < 0.5)
            receivers_mask = (h[mask_i] < 0.5) & (h[mask_j] > 0.5)
            donors = np.where(donors_mask)[0] + s
            receivers = np.where(receivers_mask)[0] + s
        # else shift == 0 not possible as we zeroed it
        moved = False
        if donors.size and receivers.size:
            i = rng.choice(donors)
            j = rng.choice(receivers)
            # swap
            h[i], h[j] = h[j], h[i]
            new_val = compute_c5_vec(h, dx)
            if new_val < best_val - 1e-12:

                best_h = h.copy()
                moved = True

            else:
                # revert
                h[i], h[j] = h[j], h[i]

            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]

                    break

def continuous_local_search(h, dx, target_sum, time_limit, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.1
    while time.time() < time_limit:
        # try random donor/receiver swap similar to earlier but with delta
        donors = np.where(cur_h > 1e-8)[0]
        receivers = np.where(cur_h < 1 - 1e-8)[0]
        if donors.size ==0 or receivers.size ==0:
            break
        i = rng.choice(donors)
        j = rng.choice(receivers)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5

        delta = rng.uniform(0.0, max_delta)

            cur_val = cand_val

import time
import numpy as np

# ---------- Helper functions ----------
def compute_c5_vec(h, dx):
    """Maximum overlap integral C5 for a given discretized h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Clip to [0,1] and adjust to achieve exact sum `target_sum`."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # simple iterative fill
    n = len(vec)
    # distribute uniformly as a first correction
    vec += diff / n
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - vec.sum()
    # fix any remaining deficit/excess greedily
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Random feasible vector in [0,1] with exact sum `target_sum`."""
    x = rng.random(n)
    s = x.sum()
    if s > 0:
        x = x * (target_sum / s)
    else:
        x = np.full(n, target_sum / n)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


# ---------- Binary targeted hill climbing ----------
def binary_hill_climb_targeted(seed_bin, dx, time_limit, rng):
    """Hill‑climb on a binary vector focusing on the worst shift."""
    h = seed_bin.astype(float).copy()
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5_vec(h, dx)
    zero_idx = n - 1
    while time.time() < time_limit:
        # full correlation, ignore zero shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        corr[zero_idx] = -np.inf
        max_idx = int(np.argmax(corr))
        max_val = corr[max_idx]
        if max_val <= 0:
            break
        shift = max_idx - zero_idx
        donors = np.array([], dtype=int)
        receivers = np.array([], dtype=int)
        if shift > 0:
            s = shift
            left = slice(0, n - s)
            right = slice(s, n)
            donors_mask = (h[left] > 0.5) & (h[right] < 0.5)
            receivers_mask = (h[left] < 0.5) & (h[right] > 0.5)
            donors = np.where(donors_mask)[0]
            receivers = np.where(receivers_mask)[0]
        elif shift < 0:
            s = -shift
            left = slice(s, n)
            right = slice(0, n - s)
            donors_mask = (h[left] > 0.5) & (h[right] < 0.5)
            receivers_mask = (h[left] < 0.5) & (h[right] > 0.5)
            donors = np.where(donors_mask)[0] + s
            receivers = np.where(receivers_mask)[0] + s
        moved = False
        if donors.size and receivers.size:
            i = rng.choice(donors)
            j = rng.choice(receivers)
            h[i], h[j] = h[j], h[i]   # swap
            new_val = compute_c5_vec(h, dx)
            if new_val < best_val - 1e-12:
                best_val, best_h = new_val, h.copy()
                moved = True
                if best_val < 0.3808:
                    break
            else:
                h[i], h[j] = h[j], h[i]   # revert
        else:
            # fallback random swap of any 1 and 0
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = h[j], h[i]
            new_val = compute_c5_vec(h, dx)
            if new_val < best_val - 1e-12:
                best_val, best_h = new_val, h.copy()
                moved = True
                if best_val < 0.3808:
                    break
            else:
                h[i], h[j] = h[j], h[i]
        # optional early stop when no improvement for many iterations
        if not moved and time.time() - time_limit > -0.1:
            # little timeout safeguard
            continue
    return best_h, best_val


# ---------- Continuous local refinement ----------
def continuous_refine(h, dx, target_sum, time_limit, rng):
    """Simple coordinate‑descent style refinement keeping the sum fixed."""
    cur = h.copy()
    cur_val = compute_c5_vec(cur, dx)
    step = 0.1
    while time.time() < time_limit:
        donors = np.where(cur > 1e-8)[0]
        receivers = np.where(cur < 1.0 - 1e-8)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(receivers)
        max_delta = min(step, cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur, cur_val = cand, cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    cur = project_to_sum(cur, target_sum)
    cur_val = compute_c5_vec(cur, dx)
    return cur, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        init_cont = np.array(initial_h_values, dtype=float)
        init_cont = np.clip(init_cont, 0.0, 1.0)
        target_sum = n / 2.0
        init_cont = project_to_sum(init_cont, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        init_cont = generate_random_feasible(n, target_sum, rng)

    dx = 2.0 / n

    # best solution so far (continuous)
    best_h = init_cont.copy()
    best_val = compute_c5_vec(best_h, dx)

    # binary seed from continuous
    bin_h = np.zeros(n, dtype=float)
    idx = np.argsort(-init_cont)[:int(target_sum)]
    bin_h[idx] = 1.0
    bin_val = compute_c5_vec(bin_h, dx)
    if bin_val < best_val:
        best_h, best_val = bin_h, bin_val

    # ----------- binary hill‑climb stage -----------
    # allocate most of the budget to many short restarts
    time_budget = budget_s * 0.9
    per_restart = 0.8  # seconds per binary climb
    while time.time() - start < time_budget:
        # random binary start
        seed_bin = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=int(target_sum), replace=False)
        seed_bin[idx] = 1.0
        # run targeted hill climb
        limit = time.time() + per_restart
        h_cand, v_cand = binary_hill_climb_targeted(seed_bin, dx, limit, rng)
        if v_cand < best_val - 1e-12:
            best_h, best_val = h_cand, v_cand
            if best_val < 0.3808:
                break
        # shrink per_restart as time runs out
        elapsed = time.time() - start
        if elapsed > budget_s * 0.7:
            per_restart = max(0.2, per_restart * 0.9)

    # ----------- continuous refinement stage -----------
    remaining = budget_s - (time.time() - start)
    if remaining > 2.0:
        limit = time.time() + remaining
        h_ref, v_ref = continuous_refine(best_h, dx, target_sum, limit, rng)
        if v_ref < best_val - 1e-12:
            best_h, best_val = h_ref, v_ref

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
