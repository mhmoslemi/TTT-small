def project_to_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    diff = target - cur
    if abs(diff) < 1e-12:
        return v
    # distribute diff across non-boundary entries
    # Determine which indices are not at bound
    free = np.where((v > 0) & (v < 1))[0]
    if free.size > 0:
        incr = diff / free.size
        v[free] += incr
        v = np.clip(v, 0.0, 1.0)
    # after adjustment recalc diff
    cur = np.sum(v)
    diff = target - cur
    # fix residual by adding to arbitrary positions
    i = 0
    while abs(diff) > 1e-12 and i < len(v):
        if diff > 0 and v[i] < 1:
            inc = min(diff, 1 - v[i])
            v[i] += inc
            diff -= inc
        elif diff < 0 and v[i] > 0:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        i += 1
    return v

import time
import math
import numpy as np

# ---------- Helper functions ----------
def compute_c5(h, dx):
    """Compute the C5 value for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def compute_c5_and_worst_k(h, dx):
    """Return (C5, worst_shift) for h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    best_val = corr[max_idx] * dx
    k = max_idx - (len(h) - 1)
    return best_val, k

def project_to_sum(v, target):
    """Project vector v onto the feasible set sum(v)=target, 0<=v<=1."""
    v = np.clip(v, 0.0, 1.0)
    cur = float(np.sum(v))
    diff = target - cur
    if abs(diff) < 1e-12:
        return v
    # distribute diff uniformly over non‑bound entries
    free = np.where((v > 0) & (v < 1))[0]
    if free.size:
        inc = diff / free.size
        v[free] += inc
        v = np.clip(v, 0.0, 1.0)
    # fix any residual
    diff = target - float(np.sum(v))
    i = 0
    n = len(v)
    while abs(diff) > 1e-12 and i < n:
        if diff > 0 and v[i] < 1.0:
            add = min(diff, 1.0 - v[i])
            v[i] += add
            diff -= add
        elif diff < 0 and v[i] > 0.0:
            sub = min(-diff, v[i])
            v[i] -= sub
            diff += sub
        i += 1
    return v

def generate_random_feasible(n, target, rng):
    """Create a random feasible vector (values in [0,1] and exact sum)."""
    v = rng.random(n)
    s = float(v.sum())
    if s == 0.0:
        v = np.full(n, target / n)
    else:
        v = v * (target / s)
    v = np.clip(v, 0.0, 1.0)
    v = project_to_sum(v, target)
    return v

def gradient_worst_k(h, k):
    """Gradient of the worst overlap term with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[: n - k] += 1.0 - h[k:]
            grad[k:] -= h[: n - k]
    else:
        kk = -k
        if kk < n:
            grad[kk:] += 1.0 - h[: n - kk]
            grad[: n - kk] -= h[kk:]
    return grad

def binary_hill_climb(h_bin, cur_val, dx, time_limit, rng):
    """Simple hill climbing on a binary vector."""
    best_h = h_bin.copy()
    best_val = cur_val
    n = len(h_bin)
    while time.time() < time_limit:
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best_h = cand
            best_val = cand_val
    return best_h.astype(float), best_val

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and starting vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    dx = 2.0 / n
    target_sum = n / 2.0

    # Initialise best solution
    if h0 is not None:
        cur_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        cur_h = generate_random_feasible(n, target_sum, rng)

    cur_val = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val

    # Parameters for stochastic search
    T0 = 0.001                # initial temperature
    guided_prob = 0.6         # probability to use gradient‑guided move
    restart_limit = 20000     # steps without improvement before a restart
    no_improve = 0

    end_time = start_time + budget_s - 2.0  # safety margin

    while time.time() < end_time:
        # Refresh current value and worst shift
        cur_val, worst_k = compute_c5_and_worst_k(cur_h, dx)

        # --------- Propose a move ----------
        if rng.random() < guided_prob:
            # Gradient‑guided transfer
            grad = gradient_worst_k(cur_h, worst_k)
            pos = np.where((grad > 1e-12) & (cur_h > 1e-12))[0]
            neg = np.where((grad < -1e-12) & (cur_h < 1.0 - 1e-12))[0]
            if pos.size and neg.size:
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_delta = min(cur_h[i], 1.0 - cur_h[j])
                if max_delta > 0:
                    delta = rng.random() * max_delta
                    cand = cur_h.copy()
                    cand[i] -= delta
                    cand[j] += delta
                else:
                    continue
            else:
                # Fallback to random transfer
                donors = np.where(cur_h > 1e-12)[0]
                recips = np.where(cur_h < 1.0 - 1e-12)[0]
                if donors.size == 0 or recips.size == 0:
                    continue
                i = rng.choice(donors)
                j = rng.choice(recips)
                max_delta = min(cur_h[i], 1.0 - cur_h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                cand = cur_h.copy()
                cand[i] -= delta
                cand[j] += delta
        else:
            # Pure random mass transfer
            donors = np.where(cur_h > 1e-12)[0]
            recips = np.where(cur_h < 1.0 - 1e-12)[0]
            if donors.size == 0 or recips.size == 0:
                continue
            i = rng.choice(donors)
            j = rng.choice(recips)
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta

        # Ensure feasibility (tiny drift correction)
        if abs(np.sum(cand) - target_sum) > 1e-10:
            cand = project_to_sum(cand, target_sum)

        cand_val = compute_c5(cand, dx)

        # --------- Acceptance ----------
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            no_improve = 0
            if cand_val < best_val - 1e-12:
                best_h = cand.copy()
                best_val = cand_val
        else:
            # Simulated‑annealing acceptance
            elapsed = time.time() - start_time
            fraction = max(0.0, (end_time - elapsed) / (end_time - start_time))
            T = T0 * fraction
            if T > 0.0 and rng.random() < math.exp(-(cand_val - cur_val) / T):
                cur_h = cand
                cur_val = cand_val
            else:
                no_improve += 1

        # --------- Restart if stagnated ----------
        if no_improve >= restart_limit:
            cur_h = generate_random_feasible(n, target_sum, rng)
            cur_val = compute_c5(cur_h, dx)
            no_improve = 0

    # ----- Binary refinement phase (if time permits) -----
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 2.0:
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argpartition(best_h, -int(target_sum))[-int(target_sum) :]
        binary_h[top_idx] = 1
        bin_val = compute_c5(binary_h.astype(float), dx)
        limit = time.time() + time_remaining - 0.5
        binary_h, bin_val = binary_hill_climb(binary_h, bin_val, dx, limit, rng)
        if bin_val < best_val - 1e-12:
            best_h = binary_h.astype(float)
            best_val = bin_val

    # Final safety projection
    best_h = project_to_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_val = compute_c5(best_h, dx)

    return best_h, best_val, n
