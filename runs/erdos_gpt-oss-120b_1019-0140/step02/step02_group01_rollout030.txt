import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given discretised h."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def hill_climb_binary(n, rng, deadline):
    """Simulated‑annealing style binary hill‑climbing (swap 0/1) respecting sum=M."""
    M = n // 2
    # start from a random binary vector with exactly M ones
    h = np.zeros(n, dtype=float)
    h[:M] = 1.0
    rng.shuffle(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score

    start = time.time()
    max_iters = 500000
    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break
        # linear temperature schedule between 0.01 and 1e-5
        frac = (now - start) / (deadline - start)
        T = 0.01 * (1.0 - frac) + 1e-5 * frac
        # propose a swap of opposite bits
        i, j = rng.choice(n, 2, replace=False)
        if cur_h[i] == cur_h[j]:
            continue
        new_h = cur_h.copy()
        new_h[i], new_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(new_h)
        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / T):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_h = new_h
                best_score = new_score
                if best_score < 0.3808:
                    break
    return best_h, best_score

def continuous_refine(h, rng, deadline):
    """Small mass‑transfer moves (fractional) to improve the objective."""
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while time.time() < deadline:
        i, j = rng.choice(n, 2, replace=False)
        # random direction of transfer
        if rng.random() < 0.5:
            # from i to j
            max_transfer = min(best_h[i], 1.0 - best_h[j])
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            new_h = best_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # from j to i
            max_transfer = min(best_h[j], 1.0 - best_h[i])
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            new_h = best_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        new_score = compute_c5(new_h)
        if new_score + 1e-12 < best_score:
            best_h = new_h
            best_score = new_score
            if best_score < 0.3808:
                break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    # a small safety margin
    final_deadline = start_time + budget_s - 0.5

    rng = np.random.default_rng(seed)

    best_h_global = None
    best_score_global = np.inf
    best_n_global = None

    # Candidate discretisations (even numbers)
    candidate_ns = [80, 96, 112, 128, 144, 160]

    # If a parent solution is supplied, try its size first
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() >= final_deadline:
            break

        # allocate a chunk of time for this n
        remaining_total = final_deadline - time.time()
        if remaining_total < 30.0:
            break
        sub_deadline = time.time() + min(250.0, remaining_total * 0.6)

        M = n // 2
        # start with parent if it matches size
        best_h_n = None
        best_score_n = np.inf
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h_parent = np.array(initial_h_values, dtype=float)
            cur = np.sum(h_parent)
            if abs(cur - M) > 1e-12:
                # simple uniform correction
                h_parent = np.clip(h_parent + (M - cur) / n, 0.0, 1.0)
            best_h_n = h_parent
            best_score_n = compute_c5(best_h_n)

        # run several binary hill‑climbing restarts while time permits
        while True:
            time_left = sub_deadline - time.time()
            if time_left < 20.0:
                break
            inner_deadline = time.time() + min(30.0, time_left)
            h_local, score_local = hill_climb_binary(n, rng, inner_deadline)
            if score_local < best_score_n - 1e-12:
                best_h_n = h_local
                best_score_n = score_local
                if best_score_n < 0.3808:
                    break
        # final local fractional refinement
        if best_h_n is not None and time.time() < sub_deadline:
            best_h_n, best_score_n = continuous_refine(best_h_n, rng, sub_deadline)

        # update global best
        if best_h_n is not None and best_score_n < best_score_global - 1e-12:
            best_h_global = best_h_n
            best_score_global = best_score_n
            best_n_global = n
            if best_score_global < 0.3808:
                break

    # Fallback if nothing was found
    if best_h_global is None:
        n = 80
        best_n_global = n
        best_h_global = np.full(n, 0.5, dtype=float)
        best_score_global = compute_c5(best_h_global)

    # Final feasibility enforcement
    n = best_n_global
    h_final = np.clip(best_h_global, 0.0, 1.0)
    M = n // 2
    cur = np.sum(h_final)
    if abs(cur - M) > 1e-9:
        # distribute the tiny discrepancy uniformly, then clip again
        h_final = np.clip(h_final + (M - cur) / n, 0.0, 1.0)
        # a final tiny fix if needed
        cur = np.sum(h_final)
        if abs(cur - M) > 1e-9:
            diff = M - cur
            h_final[0] = np.clip(h_final[0] + diff, 0.0, 1.0)

    return h_final, best_score_global, n
