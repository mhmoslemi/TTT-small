def project_to_bounds_and_sum(v, target, eps=1e-9, max_iter=100):
    lo = -1.0
    hi = 1.0
    # Actually lambda can be any real; compute lower bound: if lambda very negative, v - lambda huge positive, clipped to 1 => sum = n (max). If lambda very large positive, v - lambda very negative, clipped to 0 => sum =0. So lambda monotonic decreasing sum.
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < eps:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

```
def solve_milp(n, time_limit):
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == n/2]
    # Add t >= 0 optionally
    constraints.append(t >= 0)
    for k in range(-(n-1), n):
        if k == 0:
            continue
        if k > 0:
            idx_i = np.arange(n - k)
            idx_j = idx_i + k
        else:
            kk = -k
            idx_i = np.arange(kk, n)
            idx_j = idx_i - kk
        # create y_k variable
        m = len(idx_i)
        y = cp.Variable(m, boolean=True)
        constraints.append(y <= x[idx_i])
        constraints.append(y <= x[idx_j])
        constraints.append(y >= x[idx_i] + x[idx_j] - 1)
        overlap_expr = cp.sum(x[idx_i]) - cp.sum(y)
        constraints.append(overlap_expr <= t)
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, timeLimit=time_limit)
    # After solving, check status
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h = np.array(x.value, dtype=float)
        t_val = t.value
        return h, t_val
    else:
        return None, None
```

    lo = -2.0
    hi = 2.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0

        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

            idx_j = idx_i + k

        return None, None

    t_val = float(t.value)
    return h, t_val

            idx_j = idx_i - kk
        m = len(idx_i)
        y = cp.Variable(m, boolean=True)
        constraints += [
            y <= x[idx_i],
            y <= x[idx_j],
            y >= x[idx_i] + x[idx_j] - 1,
            cp.sum(x[idx_i]) - cp.sum(y) <= t,
        ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=max_seconds, warm_start=True, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, warm_start=True, verbose=False)
        except Exception:
            return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None
    h = np.array(x.value, dtype=float)
    t_val = float(t.value)
    return h, t_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ----- initialise h ----------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        cur = h0.sum()
        if abs(cur - target) > 1e-12:
            h0 = project_bounds_sum(h0, target)
    else:
        n = 80
        h0 = random_binary_vector(n, rng)
        target = n / 2.0

    h0 = np.clip(h0, 0.0, 1.0)
    cur = h0.sum()
    if abs(cur - target) > 1e-9:
        h0 = project_bounds_sum(h0, target)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- greedy refinement from the supplied construction ---------------
    if time.time() < deadline - 5.0:
        best_h, best_val = refine_greedy(best_h, deadline, rng)

    # ----- several random restarts ----------------------------------------
    while time.time() < deadline - 30.0:
        h_start = random_binary_vector(n, rng)
        h_ref, val = refine_greedy(h_start, deadline, rng)
        if val < best_val:
            best_val = val
            best_h = h_ref.copy()
            if best_val <= 0.3808:
                break

    # ----- attempt exact MILP if time permits -----------------------------
    remaining = deadline - time.time()
    if remaining > 10.0 and cp is not None:
        milp_time = min(300.0, remaining)
        h_milp, t_val = milp_optimize(n, max_seconds=milp_time)
        if h_milp is not None:
            val = t_val * (2.0 / n)
            if val < best_val:
                best_val = val
                best_h = h_milp.copy()

    # ----- final safety projection ----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h = project_bounds_sum(best_h, target)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
