import time, numpy as np, cvxpy as cp

def compute_c5_vec(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def solve_milp(n, budget_s, start_time):
    dx = 2.0 / n
    target = n // 2  # integer
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == target, t >= 0]
    # For each shift k != 0
    for k in range(-n+1, n):
        if k == 0:
            continue
        length = n - abs(k)
        # get i indices
        if k > 0:
            i_idx = np.arange(0, n - k)
            j_idx = i_idx + k
        else:
            i_idx = np.arange(-k, n)
            j_idx = i_idx + k  # will be i_idx - abs(k)
        # define w variable for this shift
        w = cp.Variable(length, boolean=True)
        # constraints linking w to h[i], h[j]
        constraints += [w <= h[i_idx]]
        constraints += [w <= 1 - h[j_idx]]
        # w >= h_i - h_j
        constraints += [w >= h[i_idx] - h[j_idx]]
        # bound on sum
        constraints += [cp.sum(w) * dx <= t]
    # define objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    # Solve with time limit (maybe)
    # Provide solver arguments
    time_left = budget_s - (time.time() - start_time)
    # If time_left > 0 else skip.
    try:
        prob.solve(solver=cp.CBC, maximumSeconds=time_left)
    except Exception:
        # fallback
        return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        # fallback
        return None, None
    h_val = np.array(h.value).astype(float)
    # Ensure rounding to 0/1 via clipping
    h_val = np.clip(np.round(h_val), 0, 1)
    t_val = t.value
    return h_val, t_val
```

def hill_climb_binary(h, dx, target_sum, time_limit, rng):
    best = h.copy()
    best_val = compute_c5_vec(best, dx)
    while time.time() < time_limit:
        # randomly select 1 and 0 indices
        ones = np.flatnonzero(best == 1)
        zeros = np.flatnonzero(best == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < best_val - 1e-12:
            best = cand
            best_val = cand_val
    return best, best_val

import time
import numpy as np
import cvxpy as cp

# -------------------------------------------------
def compute_c5_vec(h, dx):
    """Maximum overlap integral for a discrete step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def generate_random_binary(n, on_cnt, rng):
    """Return a random binary array of length n with exactly on_cnt ones."""
    idx = rng.choice(n, size=on_cnt, replace=False)
    arr = np.zeros(n, dtype=int)
    arr[idx] = 1
    return arr

def hill_climb_binary(start_h, dx, on_cnt, time_limit, rng):
    """Simple deterministic hill‑climb on binary vectors."""
    best = start_h.copy()
    best_val = compute_c5_vec(best.astype(float), dx)
    while time.time() < time_limit:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best = cand
            best_val = cand_val
    return best, best_val

def solve_milp(n, time_limit, start_time):
    """Mixed‑integer linear programming formulation for the exact problem."""
    dx = 2.0 / n
    target = n // 2  # number of ones

    h = cp.Variable(n, boolean=True)
    t = cp.Variable(nonneg=True)

    constraints = [cp.sum(h) == target, t >= 0]

    # Build w variables and constraints for each shift k != 0
    for k in range(-n + 1, n):
        if k == 0:
            continue
        length = n - abs(k)
        if k > 0:
            i_idx = np.arange(0, n - k)
            j_idx = i_idx + k
        else:
            i_idx = np.arange(-k, n)
            j_idx = i_idx + k
        w = cp.Variable(length, boolean=True)
        # w <= h[i]
        constraints.append(w <= h[i_idx])
        # w <= 1 - h[j]
        constraints.append(w <= 1 - h[j_idx])
        # w >= h[i] - h[j]
        constraints.append(w >= h[i_idx] - h[j_idx])
        # overlap bound for this shift
        constraints.append(cp.sum(w) * dx <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # remaining time for the solver
    remaining = time_limit - (time.time() - start_time)
    if remaining <= 0:
        return None, None

    try:
        prob.solve(solver=cp.CBC, seconds=remaining, verbose=False)
    except Exception:
        return None, None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    h_val = np.rint(np.array(h.value)).astype(int)
    t_val = float(t.value) if t.value is not None else None
    return h_val, t_val
