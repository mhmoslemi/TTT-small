def fix_sum(h, target):
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # find free entries (0 < h < 1)
    free = np.where((h > 0) & (h < 1))[0]
    if free.size > 0:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0., 1.)
        # recompute diff...
    # residual fix:
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        # adjust sequentially
        for i in range(len(h)):
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(h, 0., 1.)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def fix_sum(h, target):
    """Project h onto the hyperplane sum(h)=target while keeping 0<=h<=1."""
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Distribute equally among interior (non‑bound) entries
    free = np.where((h > 0) & (h < 1))[0]
    if free.size:
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = target - cur
    # Residual correction by scanning
    if abs(diff) > 1e-12:
        for i in range(len(h)):
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def gradient_for_max_shift(h, k, dx):
    """Gradient of the overlap term f_k = ∑ h_i (1‑h_{i+k})·dx."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[: n - k] += (1.0 - h[k:]) * dx          # derivative of first factor
        grad[k:] -= h[: n - k] * dx                  # derivative of second factor
    else:
        kk = -k
        grad[kk:] += (1.0 - h[: n - kk]) * dx
        grad[: n - kk] -= h[kk:] * dx
    return grad

def overlap_shift(h, k, dx):
    """Return ∫ h(x)(1‑h(x+k))dx for a given integer shift k on the grid."""
    n = len(h)

        return np.dot(h[kk:], (1.0 - h[: n - kk])) * dx
