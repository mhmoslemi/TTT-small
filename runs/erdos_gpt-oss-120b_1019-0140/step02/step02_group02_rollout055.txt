def get_product_weights(h, k):
    n = len(h)
    w = np.zeros(n)
    if k >= 0:
        # For i in [0, n-k-1], h_i contributes to product sum via h_i*h_{i+k}
        w[: n - k] += h[k:]  # note: for i, partner at i+k
        # For i in [k, n-1], h_i also appears as partner when j = i-k, product term h_{i-k}*h_i
        w[k:] += h[: n - k]
    else:
        kk = -k
        w[kk:] += h[: n - kk]
        w[: n - kk] += h[kk:]
    return w

def hill_climb(h_start, start_time, deadline, rng):
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    # compute initial best
    best_h = h.copy()
    best_val = compute_c5(h)
    # parameters
    step_base = 0.01  # typical delta
    # We may adapt step size decaying
    iter_count = 0
    # Precompute index sets for each shift? Not needed.

    while time.time() < deadline - 0.05:
        # compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        # find worst shift (max correlation)
        max_idx = np.argmax(corr)
        # compute value
        cur_val = corr[max_idx] * dx
        # if current best improved
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            # early exit if very good
            if best_val < 0.3808:
                break

        # Determine shift k
        k = max_idx - (n - 1)
        # compute gradient weights for product sum
        w = get_product_weights(h, k)

        # Identify source and destination indices
        # Sources: h_i > eps (to subtract)
        # Destinations: h_i < 1 - eps (to add)
        eps = 1e-12
        src_candidates = np.where(h > eps)[0]
        dst_candidates = np.where(h < 1.0 - eps)[0]

        if src_candidates.size == 0 or dst_candidates.size == 0:
            # No move possible
            break

        # Choose src with minimum w, dst with maximum w, but ensure src != dst
        # Compute w values for candidates
        src_w = w[src_candidates]
        dst_w = w[dst_candidates]

        # Choose src idx with minimal w
        i_src = src_candidates[np.argmin(src_w)]
        i_dst = dst_candidates[np.argmax(dst_w)]

        if i_src == i_dst:
            # pick alternative random
            i_src = rng.choice(src_candidates)
            i_dst = rng.choice(dst_candidates)

        # Transfer amount
        max_transfer = min(h[i_src], 1.0 - h[i_dst])
        if max_transfer < 1e-12:
            # cannot transfer
            break

        # step size maybe scaled by step_base relative to max_transfer
        delta = min(step_base, max_transfer)

        # Optionally use some random factor
        # delta = rng.random() * max_transfer

        # Apply transfer
        h[i_src] -= delta
        h[i_dst] += delta

        iter_count += 1
        # Optionally adapt step size: reduce after many iterations
        if iter_count % 200 == 0:
            step_base *= 0.9  # gradually shrink

    return best_h, best_val

def random_start_vector(n, rng):
    # generate random vector with entries in [0,1], sum = n/2
    # We'll sample uniform [0,1], then scale to sum, then clip and fix residual.
    v = rng.random(n)
    target = n / 2.0
    # Scale to achieve sum, but scaling might push some values >1; we will clip then adjust
    scale = target / v.sum()
    v = v * scale
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Compute current sum and adjust
    cur = v.sum()
    diff = target - cur
    # While diff not close to zero, adjust random entries
    max_iter = 1000
    itr = 0
    while abs(diff) > 1e-12 and itr < max_iter:
        # pick random index
        i = rng.integers(0, n)
        if diff > 0:
            # need to increase sum
            inc = min(diff, 1.0 - v[i])
            if inc > 0:
                v[i] += inc
                diff -= inc
        else:
            # need to decrease sum
            dec = min(-diff, v[i])
            if dec > 0:
                v[i] -= dec
                diff += dec
        itr += 1
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_product_weights(h, k):
    n = len(h)
    w = np.zeros(n, dtype=float)
    if k >= 0:
        nk = n - k
        w[:nk] += h[k:]
        w[k:] += h[:nk]
    else:
        kk = -k
        nk = n - kk
        w[kk:] += h[:nk]
        w[:nk] += h[kk:]
    return w

def random_start_vector(n, rng):
    target = n / 2.0
    v = rng.random(n)
    # scale to target sum
    scale = target / v.sum()
    v = v * scale
    # clip and fix sum
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    # iterative fix
    max_iter = 10 * n
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        # choose random indices to adjust
        i = rng.integers(0, n)
        if diff > 0:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        it += 1
    # Final clip
    v = np.clip(v, 0.0, 1.0)
    return v

def hill_climb(h_start, deadline, rng):
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    # precompute correlation and value
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_val = np.max(corr) * dx
    best_h = h.copy()
    best_val = cur_val
    step = 0.01  # base transfer amount
    iter_count = 0
    eps = 1e-12
    max_iterations = 50000  # safety
    while time.time() < deadline - 1e-3 and iter_count < max_iterations:
        # find worst shift
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute gradient weights for product sum
        w = get_product_weights(h, k)
        # candidates
        src_cands = np.where(h > eps)[0]
        dst_cands = np.where(h < 1.0 - eps)[0]
        if src_cands.size == 0 or dst_cands.size == 0:
            break
        # pick src with minimal w, dst with maximal w
        src_w = w[src_cands]
        dst_w = w[dst_cands]
        i_src = src_cands[np.argmin(src_w)]
        i_dst = dst_cands[np.argmax(dst_w)]
        if i_src == i_dst:
            # fallback random
            i_src = rng.choice(src_cands)
            i_dst = rng.choice(dst_cands)
        max_transfer = min(h[i_src], 1.0 - h[i_dst])
        if max_transfer < eps:
            # try random tiny move
            i_src = rng.choice(src_cands)
            i_dst = rng.choice(dst_cands)
            max_transfer = min(h[i_src], 1.0 - h[i_dst])
            if max_transfer < eps:
                break
        delta = min(step, max_transfer)
        # apply
        h[i_src] -= delta
        h[i_dst] += delta
        # evaluate new
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_val = np.max(new_corr) * dx
        if new_val <= cur_val + 1e-14:
            # accept
            corr = new_corr
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            # revert
            h[i_src] += delta
            h[i_dst] -= delta
            # optionally random perturbation
            # if no improvement after many steps, reduce step
        iter_count += 1
        if iter_count % 500 == 0:
            step *= 0.9
            if step < 1e-5:
                step = 0.01  # reset maybe?
    return best_h, best_val

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def get_product_weights(h, k):
    """Gradient of Σ h_i*h_{i+k} w.r.t. h (used for k‑shift)."""
    n = len(h)
    w = np.zeros(n, dtype=float)
    if k >= 0:
        nk = n - k
        w[:nk] += h[k:]          # h_i paired with h_{i+k}
        w[k:] += h[:nk]          # h_i paired with h_{i-k}
    else:
        kk = -k
        nk = n - kk
        w[kk:] += h[:nk]         # h_i paired with h_{i-kk}
        w[:nk] += h[kk:]         # h_i paired with h_{i+kk}
    return w

def random_start_vector(n, rng):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    scale = target / v.sum()
    v = v * scale
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    max_iter = 10 * n
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        it += 1
    return np.clip(v, 0.0, 1.0)

def hill_climb(h_start, deadline, rng):
    """Greedy mass‑transfer hill climbing guided by the worst shift."""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_val = np.max(corr) * dx
    best_h = h.copy()
    best_val = cur_val

    step = 0.01          # base transfer amount
    eps = 1e-12
    non_improve = 0
    iter_cnt = 0
    max_iter = 50000

    while time.time() < deadline - 1e-3 and iter_cnt < max_iter:
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        w = get_product_weights(h, k)

        src_cands = np.where(h > eps)[0]
        dst_cands = np.where(h < 1.0 - eps)[0]
        if src_cands.size == 0 or dst_cands.size == 0:
            break

        src_w = w[src_cands]
        dst_w = w[dst_cands]

        i_src = src_cands[np.argmin(src_w)]
        i_dst = dst_cands[np.argmax(dst_w)]

        if i_src == i_dst:
            i_src = rng.choice(src_cands)
            i_dst = rng.choice(dst_cands)

        max_transfer = min(h[i_src], 1.0 - h[i_dst])
        if max_transfer < eps:
            # fall back to a random feasible pair
            i_src = rng.choice(src_cands)
            i_dst = rng.choice(dst_cands)
            max_transfer = min(h[i_src], 1.0 - h[i_dst])
            if max_transfer < eps:
                break

        delta = min(step, max_transfer)
        h[i_src] -= delta
        h[i_dst] += delta

        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_val = np.max(new_corr) * dx

        if new_val <= cur_val + 1e-14:
            # accept
            corr = new_corr
            cur_val = new_val
            non_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            # reject
            h[i_src] += delta
            h[i_dst] -= delta
            non_improve += 1

        iter_cnt += 1
        if iter_cnt % 500 == 0:
            step = max(step * 0.9, 1e-5)

        # occasional random perturbation to escape plateaus
        if non_improve >= 500:
            src_cands = np.where(h > eps)[0]
            dst_cands = np.where(h < 1.0 - eps)[0]
            if src_cands.size and dst_cands.size:
                i_src = rng.choice(src_cands)
                i_dst = rng.choice(dst_cands)
                max_transfer = min(h[i_src], 1.0 - h[i_dst])
                if max_transfer > eps:
                    delta = rng.random() * max_transfer
                    h[i_src] -= delta
                    h[i_dst] += delta
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    cur_val = np.max(corr) * dx
            non_improve = 0

    return best_h, best_val
