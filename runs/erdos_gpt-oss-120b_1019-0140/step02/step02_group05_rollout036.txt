def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur_sum = np.sum(h)
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        # residual
        cur_sum = np.sum(h)
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # adjust iteratively
            for i in range(len(h)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
            h = np.clip(h, 0.0, 1.0)
    return h
```

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated on the discretization."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Clip to [0,1] and adjust so that sum equals target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # distribute uniformly among interior points
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            incr = diff / free.size
            h[free] += incr
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
        # residual fix (rare)
        if abs(diff) > 1e-12:
            for i in range(h.size):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_and_fix(h, target_sum):
    """One pass of a simple 3‑point smoothing followed by feasibility projection."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return project_to_feasible(sm, target_sum)

def subgradient_optimize(
    h0,
    target_sum,
    dx,
    max_iter,
    step0,
    budget_s,
    start_time,
    best_val_overall,
    target_c5,
):
    """
    Projected sub‑gradient descent that repeatedly reduces the worst overlap.
    Returns the best (h, value) found for this start.
    """
    n = h0.shape[0]
    h = h0.copy()
    best_h_local = h.copy()
    best_val_local = compute_c5(h)

    for it in range(max_iter):
        # time bound
        if time.time() - start_time >= budget_s * 0.99:
            break

        # current overlaps
        overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
        idx_max = int(np.argmax(overlaps))
        k = idx_max - (n - 1)  # shift corresponding to maximal overlap

        # direction that decreases this overlap (i.e. increases autocorrelation)
        d = np.zeros_like(h)
        if k == 0:
            d = dx * (1.0 - 2.0 * h)          # -grad of O(0)
        else:
            if k > 0:
                d[: n - k] += h[k:]
                d[k:] += h[: n - k]
            else:
                kk = -k
                d[kk:] += h[: n - kk]
                d[: n - kk] += h[kk:]
            d = d * dx                       # include the dx factor

        step = step0 * (1.0 - it / max_iter)
        if step <= 0:
            break
        h = project_to_feasible(h + step * d, target_sum)

        # periodic evaluation
        if (it & 31) == 0:
            cur_val = compute_c5(h)
            if cur_val < best_val_local:
                best_val_local = cur_val
                best_h_local = h.copy()
                if best_val_local < target_c5:
                    return best_h_local, best_val_local
            # keep track against overall best to possibly exit early
            if best_val_overall is not None and best_val_local < best_val_overall:
                best_val_overall = best_val_local

    return best_h_local, best_val_local
