def perturb_swap(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # compute delta bounds
    low = max(-h[i], -(1 - h[j]))

        return h  # no feasible perturbation
    delta = rng.uniform(low, high)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta

def gradient_descent_local(h, max_iters, rng, target_sum, time_limit):
    best_local_h = h
    best_local_c5 = compute_c5(h)
    dx = 2.0/len(h)
    beta = 2.0
    max_beta = 50.0
    alpha0 = 0.2
    start = time.time()
    no_improve = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_local_c5 - 1e-12:
            best_local_c5 = c5
            best_local_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # Update beta slowly
        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)
        # Compute softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(v) / len(v)
        else:
            w /= w_sum
        # gradient
        n = len(h)
        offset = n - 1
        g = np.zeros(n)
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            k = idx - offset
            if k >= 0:
                # shift right by k: overlap between h[0:n-k] and h[k:n]
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                # k negative: shift left
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # zero-mean step for sum constraint
        g -= np.mean(g)
        # step size decays linearly over the total budget maybe
        elapsed = time.time() - start
        # we can compute fraction based on max_iters? simple: alpha = alpha0 * max(0.01, 1 - it/max_iters)
        alpha = alpha0 * max(0.01, 1 - it / max_iters)
        h_new = h - alpha * g
        # project feasibility
        h_new = project_to_capped_simplex(h_new, target_sum)
        # compute objective
        c5_new = compute_c5(h_new)
        # accept if improves or random small prob to escape
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        # random shake if stagnant
        if no_improve > 500:
            # add random zero-mean noise then project
            h = project_to_capped_simplex(h + rng.normal(scale=0.02, size=n), target_sum)
            no_improve = 0
    return best_local_h, best_local_c5

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap value multiplied by dx."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n with sum(v)=target (capped simplex)."""
    v = np.asarray(v, dtype=float)
    lo = -1.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def perturb_swap(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Randomly move mass between two indices while preserving sum."""
    n = h.size
    for _ in range(5):
        i, j = rng.choice(n, size=2, replace=False)
        low = max(-h[i], -(1.0 - h[j]))
        high = min(1.0 - h[i], h[j])
        if low < high:
            delta = rng.uniform(low, high)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            return h_new
    return h.copy()  # fallback (unlikely)

def gradient_descent_local(h0: np.ndarray, max_iters: int,
                          rng: np.random.Generator,
                          target_sum: float,
                          time_limit: float):
    """Projected gradient descent on a smoothed max-overlap objective."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 50.0
    alpha0 = 0.2
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        v = np.correlate(h, 1.0 - h, mode="full")
        c5 = v.max() * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if beta < max_beta:
            beta = min(max_beta, beta * 1.004)

        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        ws = w.sum()
        if ws == 0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        g = np.zeros_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        g -= np.mean(g)  # keep zero-sum direction
        alpha = alpha0 * max(0.01, 1.0 - it / max_iters)
        h_new = h - alpha * g
        h_new = project_to_capped_simplex(h_new, target_sum)

        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 600:
            h = project_to_capped_simplex(h + rng.normal(scale=0.025, size=n), target_sum)
            no_improve = 0

    return best_h, best_c5

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.size
    else:
        n = 100
        h0 = rng.random(n)

    target_sum = n / 2.0
    h0 = project_to_capped_simplex(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Main optimization loop
    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 0.5:
            break
        # Random start: either jitter best or fresh random
        if rng.random() < 0.4:
            h_start = best_h + rng.normal(scale=0.03, size=n)
        else:
            h_start = rng.random(n)
        h_start = project_to_capped_simplex(h_start, target_sum)

        # Allocate iterations proportional to remaining time
        max_iters = int(min(2000, remaining * 500))
        if max_iters < 10:
            max_iters = 10
        local_time = remaining * 0.6  # give most of remaining budget to GD
        h_loc, c5_loc = gradient_descent_local(
            h_start, max_iters, rng, target_sum, local_time
        )
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc.copy()
            if best_c5 < 0.3808:
                break

        # Short hill‑climbing with pair swaps
        cur_h = best_h.copy()
        cur_c5 = best_c5
        moves = 300
        for _ in range(moves):
            if time.time() - start > budget_s - 0.1:
                break
            cand = perturb_swap(cur_h, rng)
            cand_c5 = compute_c5(cand)
            if cand_c5 < cur_c5:
                cur_h, cur_c5 = cand, cand_c5
                if cand_c5 < best_c5:
                    best_c5, best_h = cand_c5, cand.copy()
                    if best_c5 < 0.3808:
                        break
            elif rng.random() < 0.0005:
                cur_h, cur_c5 = cand, cand_c5

    # Final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_capped_simplex(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
