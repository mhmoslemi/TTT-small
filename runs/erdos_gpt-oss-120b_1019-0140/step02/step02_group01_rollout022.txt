def random_feasible_h(n, M, rng):
    h = np.zeros(n)
    remaining_mass = M
    while remaining_mass > 1e-12:
        # choose random index that is not full
        candidates = np.where(h < 1 - 1e-12)[0]
        if len(candidates)==0:
            break
        i = rng.choice(candidates)
        # assign random amount up to min(1-h[i], remaining_mass)
        max_add = min(1 - h[i], remaining_mass)
        add = rng.random() * max_add
        # ensure non-zero addition
        if add < 1e-12:
            add = max_add
        h[i] += add
        remaining_mass -= add
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ----- basic utilities -----
def compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def generate_initial_h(n, M, rng):
    """Random feasible h with sum M and 0<=h<=1."""
    method = rng.integers(3)
    if method == 0:  # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:  # random Dirichlet‑like then project
        h = rng.random(n)
        h = h / h.sum() * M
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if cur < M:
            deficit = M - cur
            order = np.argsort(h)
            for i in order:
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            order = np.argsort(-h)
            for i in order:
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    else:  # linear decreasing shape
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if cur < M:
            deficit = M - cur
            order = np.argsort(h)
            for i in order:
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            order = np.argsort(-h)
            for i in order:
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    return h

# ----- constraints for SLSQP -----
class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def objective_t(x):
    return x[-1]

def optimise_for_n(n, time_budget, rng):
    """SLSQP based optimisation for a fixed resolution."""
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h, best_c5 = None, np.inf
    start = time.time()

    # parent solution if available
    use_parent = False
    if 'initial_h_values' in globals():
        ih = globals()['initial_h_values']
        if isinstance(ih, (list, tuple, np.ndarray)) and len(ih) == n:
            use_parent = True
            parent_h = np.array(ih, dtype=float)
            cur = parent_h.sum()
            if abs(cur - M) > 1e-12:
                parent_h += (M - cur) / n
                parent_h = np.clip(parent_h, 0.0, 1.0)
            best_h, best_c5 = parent_h, compute_c5(parent_h)

    while time.time() - start < time_budget - 0.1:
        if use_parent:
            h0 = parent_h.copy()
            use_parent = False
        else:
            h0 = generate_initial_h(n, M, rng)
        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])

        try:
            res = minimize(
                fun=objective_t,
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 800, 'ftol': 1e-10, 'disp': False}
            )
        except Exception:
            continue
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur = h_opt.sum()
        if abs(cur - M) > 1e-9:
            h_opt += (M - cur) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_c5, best_h = c5_opt, h_opt
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

# ----- directed local refinement -----
def local_search(h_start, time_budget, rng):
    """Greedy mass‑transfer hill climbing with a small simulated‑annealing component."""
    n = h_start.size
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    eps = 1e-12
    # initial temperature proportional to current objective
    T0 = max(1e-4, best_c5 * 0.01)

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(1e-12, T0 * (1.0 - elapsed / time_budget))

        # correlation and shift with maximal overlap
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (n - 1)

        # contributions for this shift
        if shift >= 0:
            L = n - shift
            hi = best_h[:L]
            hj = best_h[shift:]
            i_off, j_off = 0, shift
        else:
            s = -shift
            L = n - s
            hi = best_h[s:]
            hj = best_h[:L]
            i_off, j_off = s, 0

        contrib = hi * (1.0 - hj)

        # weighted sampling (avoid all‑zero case)
        sum_c = contrib.sum()
        if sum_c > eps:
            probs_i = contrib / sum_c
            i_loc = rng.choice(L, p=probs_i)
        else:
            i_loc = rng.integers(L)

        inv = (1.0 - contrib) + eps
        probs_j = inv / inv.sum()
        j_loc = rng.choice(L, p=probs_j)

        i = i_loc + i_off
        j = j_loc + j_off
        if i == j:
            i, j = rng.choice(n, 2, replace=False)

        # two possible directions of transfer
        max_ij = min(best_h[i], 1.0 - best_h[j])
        max_ji = min(best_h[j], 1.0 - best_h[i])
        if max_ij <= 0 and max_ji <= 0:
            continue

        # decide direction
        if max_ij > 0 and (max_ji == 0 or rng.random() < 0.5):
            delta = max_ij * rng.random()
            if delta < eps:
                delta = max_ij * 0.1
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            delta = max_ji * rng.random()
            if delta < eps:
                delta = max_ji * 0.1
            h_new = best_h.copy()
            h_new[i] += delta
            h_new[j] -= delta

        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            best_h, best_c5 = h_new, c5_new
        else:
            # accept worse move with Metropolis probability
            if np.exp((best_c5 - c5_new) / T) > rng.random():
                best_h, best_c5 = h_new, c5_new
    return best_h, best_c5

# ----- main driver -----
def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    candidate_ns = [80, 96, 112, 128, 144, 160]
    # prioritize parent size if present
    if 'initial_h_values' in globals():
        ih = globals()['initial_h_values']
        if isinstance(ih, (list, tuple, np.ndarray)):
            n0 = len(ih)
            if n0 % 2 == 0:
                if n0 not in candidate_ns:
                    candidate_ns = [n0] + candidate_ns
                else:
                    candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_total > budget_s - 2.0:
            break
        remaining = budget_s - (time.time() - start_total)
        alloc_n = min(remaining * 0.5, 300.0)   # at most 5 min per resolution

        # ---- SLSQP phase ----
        h_opt, c5_opt = optimise_for_n(n, alloc_n * 0.4, rng)
        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global, best_c5_global, best_n_global = h_opt, c5_opt, n
            if best_c5_global < 0.3808:
                break

        # ---- directed local refinement ----
        if h_opt is not None:
            h_ref, c5_ref = local_search(h_opt, alloc_n * 0.5, rng)
            if c5_ref < best_c5_global:
                best_h_global, best_c5_global, best_n_global = h_ref, c5_ref, n
                if best_c5_global < 0.3808:
                    break

    # fallback to a simple feasible constant if nothing succeeded
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        cur = best_h_global.sum()
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h_global += (M - cur) / n
            best_h_global = np.clip(best_h_global, 0.0, 1.0)

    return best_h_global, best_c5_global, n
