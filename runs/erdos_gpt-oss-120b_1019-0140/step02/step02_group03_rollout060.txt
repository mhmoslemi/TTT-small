class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n/2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        n = self.n
        if k >= 0:
            overlap = np.dot(h[:n - k], 1.0 - h[k:]) * self.dx
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n - kk]) * self.dx
        return x[-1] - overlap

def slsqp_optimize(h_start, t_start, n, maxiter=400):
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    bounds = [(0.0, 1.0)]*n + [(0.0, None)]
    constraints = [{'type':'eq','fun':sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol':1e-9, 'disp':False})
    if not res.success:
        return None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # ensure sum exact
    target = n/2.0
    cur_sum = h_opt.sum()
    if abs(cur_sum - target) > 1e-9:
        h_opt = project_to_box_sum(h_opt, target)
    return h_opt

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = hi - np.min(v) + 5.0
    for _ in range(50):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def overlap_gradient(h, k):
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        L = n - k
        if L > 0:
            g[:L] += (1.0 - h[k:k+L])
            g[k:k+L] += -h[:L]
    else:
        k = -k
        L = n - k
        if L > 0:
            g[k:k+L] += (1.0 - h[:L])
            g[:L] += -h[k:k+L]
    return g

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        n = self.n
        if k >= 0:
            overlap = np.dot(h[:n - k], 1.0 - h[k:]) * self.dx
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n - kk]) * self.dx
        return x[-1] - overlap

def slsqp_optimize(h_start, t_start, n, maxiter=400):
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if not res.success:
        return None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # final projection for exact sum
    target = n/2.0
    if abs(h_opt.sum() - target) > 1e-9:
        h_opt = project_to_box_sum(h_opt, target)
    return h_opt
