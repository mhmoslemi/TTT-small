def compute_gradient(h, k, dx):

    if k >= 0:
        g[: n - k] += h[k:]
        g[k:] += h[: n - k]
    else:

        g[kk:] += h[: n - kk]
        g[: n - kk] += h[kk:]

import time
import numpy as np
from scipy.optimize import minimize

def _project_capped_simplex(v, target, lower=0.0, upper=1.0):
    """Project vector v onto the set {x: sum(x)=target, lower<=x<=upper}."""
    # binary search on lambda such that sum(clip(v - lam, lower, upper)) = target
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, lower, upper)


def _random_feasible_h(n, rng, target):
    """Return a random vector in [0,1]^n with exact sum = target."""
    v = rng.random(n)
    return _project_capped_simplex(v, target, 0.0, 1.0)


def _gradient_min_corr(h, shift, dx):
    """Gradient of the correlation for a given shift (the worst shift)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        g[: n - k] += h[k:]
        g[k:] += h[: n - k]
    else:
        k = -shift
        g[k:] += h[: n - k]
        g[: n - k] += h[k:]
    return g * dx


def _min_corr_and_shift(h, dx):
    """Return (minimum correlation (including dx), shift achieving it)."""
    n = h.shape[0]
    corr = np.correlate(h, h, mode="full") * dx
    centre = n - 1
    corr[centre] = np.inf  # ignore k=0
    idx = np.argmin(corr)
    return corr[idx], idx - centre


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem dimensions
    n = 200                     # number of sample points
    dx = 2.0 / n
    target_sum = n / 2.0        # sum of h values (before multiplying by dx)

    # ------------------------------------------------------------------
    # initial h
    if "initial_h_values" in globals():
        hv = np.array(initial_h_values, dtype=float)
        if hv.shape[0] != n:
            x_old = np.linspace(0.0, 2.0, hv.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, hv)
        else:
            h = hv.copy()
        h = np.clip(h, 0.0, 1.0)
        h = _project_capped_simplex(h, target_sum, 0.0, 1.0)
    else:
        h = _random_feasible_h(n, rng, target_sum)

    best_h = h.copy()
    min_corr, _ = _min_corr_and_shift(h, dx)
    best_val = 1.0 - min_corr  # current C5 bound

    # ------------------------------------------------------------------
    # main gradient-ascent loop
    iter_no = 0
    no_improve = 0
    BASE_STEP = 0.5
    RESTART_AFTER = 5000

    centre = n - 1

    while time.time() < deadline - 0.5:
        # current worst shift
        min_corr, shift = _min_corr_and_shift(h, dx)

        # gradient step toward increasing that correlation
        grad = _gradient_min_corr(h, shift, dx)
        step = BASE_STEP / np.sqrt(iter_no + 1.0)
        h_candidate = h + step * grad
        h = _project_capped_simplex(h_candidate, target_sum, 0.0, 1.0)

        # evaluate new C5
        cur_min_corr, _ = _min_corr_and_shift(h, dx)
        cur_val = 1.0 - cur_min_corr

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
            if best_val < 0.3808:
                break
        else:
            no_improve += 1

        if no_improve >= RESTART_AFTER:
            h = _random_feasible_h(n, rng, target_sum)
            no_improve = 0

        iter_no += 1

    # ------------------------------------------------------------------
    # optional short SLSQP refinement (only if time permits)
    remaining = deadline - time.time()
    if remaining > 2.0:
        def obj(x):
            # objective = C5 = 1 - min_corr
            corr = np.correlate(x, x, mode="full") * dx
            corr[centre] = np.inf
            return 1.0 - corr.min()

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                obj,
                best_h,
                method="SLSQP",
                bounds=bnds,
                constraints=cons,
                options={"maxiter": 800, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x, 0.0, 1.0)
                # re‑project tiny sum errors
                cur = h_opt.sum()
                if abs(cur - target_sum) > 1e-9:
                    h_opt = _project_capped_simplex(h_opt, target_sum, 0.0, 1.0)
                opt_min_corr, _ = _min_corr_and_shift(h_opt, dx)
                opt_val = 1.0 - opt_min_corr
                if opt_val < best_val:
                    best_val = opt_val
                    best_h = h_opt.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    if abs(cur - target_sum) > 1e-9:
        best_h = _project_capped_simplex(best_h, target_sum, 0.0, 1.0)

    final_min_corr, _ = _min_corr_and_shift(best_h, dx)
    final_val = 1.0 - final_min_corr

    return best_h.astype(float), float(final_val), n
