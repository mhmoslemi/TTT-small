def anneal_binary(h_bin, max_iter, temp_start, temp_end, rng):
    n = h_bin.size
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    cur_h = h_bin.copy()
    cur_score = best_score
    # exponential cooling schedule
    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        # propose swap: pick i where cur_h[i]==1, j where cur_h[j]==0
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Approximate C5 = max_k ∫ h(x)(1-h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term averaged over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def slsqp_optimize_candidate(h0, max_iter=400):
    """SLSQP to minimize max correlation on a small scale."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simple simulated annealing on binary balanced vectors."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # pre‑compute index sets
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        # linear cooling in log‑temperature space
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))

        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            # update index lists
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    best_h = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1: small‑scale exploration (n = 64)
    # ------------------------------------------------------------------
    n_small = 64
    small_cands = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n_small:
            small = downsample(init_full, n_small)
        else:
            small = init_full.copy()
        small = project_onto_box_sum(small, n_small / 2.0)
        small_cands.append(small)

    small_cands.append(thue_morse(n_small))
    for _ in range(2):
        small_cands.append(random_binary_balanced(n_small, rng))
    for _ in range(2):
        v = rng.random(n_small)
        small_cands.append(project_onto_box_sum(v, n_small / 2.0))

    # SLSQP refinements
    for cand in small_cands:
        if time_left() < 2.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=400)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Additional random SLSQP starts
    while time_left() > 5.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=300)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Gradient refinement on small scale
    if best_h is not None and time_left() > 5.0:
        h_gd, sc_gd = gradient_descent_momentum(best_h, n_iter=5000,
                                                step0=0.5, beta=0.9)
        if sc_gd < best_score:
            best_score, best_h = sc_gd, h_gd.copy()

    # ------------------------------------------------------------------
    # Phase 2: Upsample to larger resolution (n = 512)
    # ------------------------------------------------------------------
    n_large = 512
    if best_h is None:
        best_h_large = random_binary_balanced(n_large, rng)
    else:
        best_h_large = upsample(best_h, n_large)
    best_h_large = project_onto_box_sum(best_h_large, n_large / 2.0)
    best_score_large = compute_c5(best_h_large)
    if best_score_large < best_score:
        best_score, best_h = best_score_large, best_h_large.copy()

    # ------------------------------------------------------------------
    # Phase 3: Large‑scale momentum gradient descent
    # ------------------------------------------------------------------
    remaining = time_left()
    if remaining > 6.0:
        # Rough estimate: ~1500 iterations per second for n=512
        max_iters = int(min(200000, remaining * 1500))
        if max_iters > 0:
            h_gd, sc_gd = gradient_descent_momentum(best_h, n_iter=max_iters,
                                                    step0=0.5, beta=0.9)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()

    # ------------------------------------------------------------------
    # Phase 4: Simulated annealing on a binary version
    # ------------------------------------------------------------------
    remaining = time_left()
    if remaining > 5.0:
        # Build a balanced binary start from the continuous best
        idx = np.argsort(best_h)[-n_large // 2:]
        h_bin = np.zeros(n_large, dtype=float)
        h_bin[idx] = 1.0
        # annealing schedule parameters
        max_iter_ann = int(remaining * 500)  # roughly 500 proposals per sec
        temp_start = 0.01
        temp_end = 1e-5
        h_ann, sc_ann = anneal_binary(h_bin, max_iter_ann,
                                      temp_start, temp_end, rng)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()

    # Final projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
