import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum_bounds(h, target):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # iterative waterfill
    for _ in range(10):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            h[free] += inc
        else:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            h[free] += dec
        h = np.clip(h, 0.0, 1.0)
    # final small fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, alpha, dx):
    # overl = np.correlate(h, 1 - h, mode='full') * dx
    overl = np.correlate(h, 1.0 - h, mode='full')
    overl_f = overl * dx
    max_val = np.max(overl_f)
    # compute weights
    exp_vals = np.exp(alpha * (overl_f - max_val))
    p = exp_vals / np.sum(exp_vals)
    n = len(h)
    offset = len(p) - 1  # = 2n-2
    # forward: conv of reversed p with h
    conv_fwd = np.convolve(p[::-1], h, mode='full')[offset:offset + n]
    conv_bwd = np.convolve(p, h, mode='full')[offset:offset + n]
    grad = dx * (1.0 - conv_fwd - conv_bwd)
    return grad

def compute_c5(h):
    """Exact max overlap C5 for a step function h on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_bounds(h, target, tol=1e-12, max_iter=10):
    """Project vector onto [0,1]^n and enforce sum==target via water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):

        diff = target - s
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            h[free] += inc
        else:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            h[free] += dec
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    diff = target - np.sum(h)
    if abs(diff) > tol:
        for i in range(len(h)):
            if abs(diff) < tol:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, alpha, dx):
    """Gradient of smooth max‑overlap (softmax with weight alpha)."""
    # raw overlaps (no dx)
    raw = np.correlate(h, 1.0 - h, mode="full")
    overlaps = raw * dx

    # backward average: conv of p with h
    bwd = np.convolve(p, h, mode="full")[offset:offset + n]
    grad = dx * (1.0 - fwd - bwd)
    return grad
