def hill_climb(h, max_steps, deadline, rng):
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    n = cur_h.shape[0]
    dx = 2.0 / n
    step = 0
    while step < max_steps and time.time() < deadline - 0.01:
        # compute overlap
        O = np.correlate(cur_h, 1.0 - cur_h, mode='full') * dx
        max_idx = np.argmax(O)
        k = max_idx - (n - 1)

        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        # valid donors/recipients for shift k
        if k >= 0:
            max_idx_shift = n - 1 - k
            donors_valid = donors[donors <= max_idx_shift]
            recipients_valid = recipients[recipients <= max_idx_shift]
            if donors_valid.size == 0 or recipients_valid.size == 0:
                break
            donor_shift_vals = cur_h[donors_valid + k]
            recipient_shift_vals = cur_h[recipients_valid + k]
        else:
            l = -k
            donors_valid = donors[donors >= l]
            recipients_valid = recipients[recipients >= l]
            if donors_valid.size == 0 or recipients_valid.size == 0:
                break
            donor_shift_vals = cur_h[donors_valid - l]
            recipient_shift_vals = cur_h[recipients_valid - l]

        # find best diff
        diff = donor_shift_vals.max() - recipient_shift_vals.min()
        if diff <= 1e-12:
            break
        donor_idx = donors_valid[np.argmax(donor_shift_vals)]
        recipient_idx = recipients_valid[np.argmin(recipient_shift_vals)]
        # full transfer
        max_transfer = min(cur_h[donor_idx], 1.0 - cur_h[recipient_idx])
        if max_transfer <= 0:
            break
        delta = max_transfer
        # apply
        cur_h[donor_idx] -= delta
        cur_h[recipient_idx] += delta
        new_val = compute_c5(cur_h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
        else:
            # revert
            cur_h[donor_idx] += delta
            cur_h[recipient_idx] -= delta
            break
        step += 1
    return cur_h, cur_val

import time
import numpy as np

def compute_c5(h):
    """Return the discrete C5 value for vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        inc = diff / n
        v = np.clip(v + inc, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    delta = min(diff, 1.0 - v[i])
                    v[i] += delta
                    diff -= delta
                elif diff < 0 and v[i] > 0.0:
                    delta = min(-diff, v[i])
                    v[i] -= delta
                    diff += delta
    return v

def random_binary(n, rng):
    """Binary vector (0/1) with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    arr[idx] = 1.0
    return arr

def hill_climb(h_start, max_steps, deadline, rng):
    """Greedy hill‑climbing: move full mass from a donor to a recipient if it lowers C5."""
    h = h_start.copy()
    cur_val = compute_c5(h)
    n = h.shape[0]
    dx = 2.0 / n
    step = 0
    while step < max_steps and time.time() < deadline - 0.01:
        # compute overlaps
        O = np.correlate(h, 1.0 - h, mode="full") * dx
        max_idx = int(np.argmax(O))
        k = max_idx - (n - 1)  # shift offset

        donors = np.where(h > 1e-12)[0]
        recipients = np.where(h < 1.0 - 1e-12)[0]

        if k >= 0:
            bound = n - 1 - k
            donors_valid = donors[donors <= bound]
            rec_valid = recipients[recipients <= bound]
            if donors_valid.size == 0 or rec_valid.size == 0:
                break
            donor_shift = h[donors_valid + k]
            rec_shift = h[rec_valid + k]
        else:
            l = -k
            donors_valid = donors[donors >= l]
            rec_valid = recipients[recipients >= l]
            if donors_valid.size == 0 or rec_valid.size == 0:
                break
            donor_shift = h[donors_valid - l]
            rec_shift = h[rec_valid - l]

        # best improvement candidate
        diff = donor_shift.max() - rec_shift.min()
        if diff <= 1e-12:
            break
        donor_idx = donors_valid[np.argmax(donor_shift)]
        rec_idx = rec_valid[np.argmin(rec_shift)]

        max_transfer = min(h[donor_idx], 1.0 - h[rec_idx])
        if max_transfer <= 0:
            break
        delta = max_transfer  # move whole possible mass

        # apply and test
        h[donor_idx] -= delta
        h[rec_idx] += delta
        new_val = compute_c5(h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
        else:
            # revert
            h[donor_idx] += delta
            h[rec_idx] -= delta
            break
        step += 1
    return h, cur_val
