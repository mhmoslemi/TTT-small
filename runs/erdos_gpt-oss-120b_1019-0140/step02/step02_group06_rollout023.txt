def project_sum_and_bounds(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Compute current sum
    cur = x.sum()
    # Need to adjust to target: diff = target - cur
    # Add diff equally to all entries
    x += diff / len(x)
    # Clip again
    x = np.clip(x, 0.0, 1.0)
    # Iterate maybe a few times to adjust remaining diff
    for _ in range(5):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        # Distribute diff among non-bound entries
        # Find indices where x not at 0 or 1 (i.e., 0 < x < 1)
        mask = (x > 0.0) & (x < 1.0)
        nfree = np.count_nonzero(mask)
        if nfree == 0:
            # all at bounds; cannot adjust further
            break
        add = diff / nfree
        x[mask] += add
        # Clip again
        x = np.clip(x, 0.0, 1.0)
    return x

import time, numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target, tolerance=1e-12, max_iter=10):
    # Clip
    x = np.clip(x, 0.0, 1.0)
    # Compute sum
    cur = x.sum()
    diff = target - cur
    # Add diff evenly
    if abs(diff) > tolerance:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < tolerance:
            break
        mask = (x > 0.0) & (x < 1.0)
        free = np.count_nonzero(mask)
        if free == 0:
            break
        add = diff / free
        x[mask] += add
        x = np.clip(x, 0.0, 1.0)
    return x

def tournament_select(pop, fit, rng, tournament_size):
    idxs = rng.choice(len(pop), size=tournament_size, replace=False)
    best_local = idxs[np.argmin(fit[idxs])]
    return pop[best_local]

def crossover(parent1, parent2, rng, crossover_rate):
    n = len(parent1)
    if rng.random() < crossover_rate:
        cp = rng.integers(1, n)  # 1..n-1
        child1 = np.concatenate([parent1[:cp], parent2[cp:]])
        child2 = np.concatenate([parent2[:cp], parent1[cp:]])
    else:
        child1 = parent1.copy()
        child2 = parent2.copy()
    return child1, child2

def mutate_ind(ind, target_sum, rng, mutation_rate, n):
    # In-place copy?
    ind = ind.copy()
    if rng.random() < mutation_rate:
        # redistributive moves
        num_swaps = rng.integers(1, 4)
        for _ in range(num_swaps):
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(ind[i], 1.0 - ind[j])
            min_transfer = -min(ind[j], 1.0 - ind[i])
            if max_transfer > 0 or min_transfer < 0:
                delta = rng.uniform(min_transfer, max_transfer)
                ind[i] -= delta
                ind[j] += delta
        # optional Gaussian noise on some indices
        if rng.random() < 0.3:
            num_idxs = rng.integers(1, max(2, n // 4 + 1))
            idxs = rng.choice(n, size=num_idxs, replace=False)
            ind[idxs] += rng.normal(scale=0.02, size=idxs.shape)
    # final projection
    ind = project_sum_and_bounds(ind, target_sum)
    return ind

def initialize_population(n, pop_size, target_sum, rng):
    pop = []
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float).copy()
        base = project_sum_and_bounds(base, target_sum)

def hill_climb_refine(h, target_sum, rng, time_budget):
    start = time.time()
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    n = len(h)
    while time.time() - start < time_budget:
        cand = cur.copy()
        # random redistributive moves
        for _ in range(rng.integers(1, 4)):
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(cand[i], 1 - cand[j])
            min_transfer = -min(cand[j], 1 - cand[i])
            if max_transfer > 0 or min_transfer < 0:
                delta = rng.uniform(min_transfer, max_transfer)
                cand[i] -= delta
                cand[j] += delta
        # optional noise
        if rng.random() < 0.2:
            num_idxs = rng.integers(1, max(2, n // 4 + 1))
            idxs = rng.choice(n, size=num_idxs, replace=False)
            cand[idxs] += rng.normal(scale=0.01, size=idxs.shape)
        cand = project_sum_and_bounds(cand, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
        # occasional random restart
        if rng.random() < 0.001:
            cur = rng.random(n)
            cur = project_sum_and_bounds(cur, target_sum)
            cur_c5 = compute_c5(cur)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    # Determine n_points
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80
    target_sum = n / 2.0
    # GA parameters
    pop_size = 60
    tournament_size = 3
    crossover_rate = 0.9
    mutation_rate = 0.5
    max_generations = 1000000  # effectively unbounded
    # Initialize population
    population = initialize_population(n, pop_size, target_sum, rng)
    # Compute fitness
    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    no_improve = 0
    generation = 0
    # main GA loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        if generation >= max_generations:
            break
        # elite preservation
        elite_frac = 0.1
        elite_count = max(1, int(pop_size * elite_frac))
        elite_idxs = np.argsort(fitness)[:elite_count]
        new_pop = [population[i].copy() for i in elite_idxs]
        # fill rest of population
        while len(new_pop) < pop_size:
            parent1 = tournament_select(population, fitness, rng, tournament_size)
            parent2 = tournament_select(population, fitness, rng, tournament_size)
            child1, child2 = crossover(parent1, parent2, rng, crossover_rate)
            child1 = mutate_ind(child1, target_sum, rng, mutation_rate, n)
            child2 = mutate_ind(child2, target_sum, rng, mutation_rate, n)
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
        population = new_pop
        fitness = np.array([compute_c5(ind) for ind in population])
        generation += 1
        cur_best_idx = np.argmin(fitness)
        cur_best_c5 = fitness[cur_best_idx]
        if cur_best_c5 + 1e-12 < best_c5:
            best_c5 = cur_best_c5
            best_h = population[cur_best_idx].copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 2000:
            break
    # local refinement
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        refine_budget = remaining * 0.8
        best_h, best_c5 = hill_climb_refine(best_h, target_sum, rng, refine_budget)
    # final projection and compute c5
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given h."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target, tol=1e-12, max_iter=10):
    """Clip to [0,1] and adjust sum to target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > tol:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < tol:
            break
        mask = (x > 0.0) & (x < 1.0)
        free = np.count_nonzero(mask)
        if free == 0:
            break
        add = diff / free
        x[mask] += add
        x = np.clip(x, 0.0, 1.0)
    return x

def tournament_select(pop, fit, rng, tsize):
    """Return one individual from a tournament."""
    idx = rng.choice(len(pop), size=tsize, replace=False)
    best = idx[np.argmin(fit[idx])]
    return pop[best]

def crossover(p1, p2, rng, rate):
    """Single‑point crossover."""
    n = len(p1)
    if rng.random() < rate:
        cp = rng.integers(1, n)
        c1 = np.concatenate([p1[:cp], p2[cp:]])
        c2 = np.concatenate([p2[:cp], p1[cp:]])
    else:
        c1 = p1.copy()
        c2 = p2.copy()
    return c1, c2

def mutate_ind(ind, target, rng, mrate, n):
    """Redistributive mutation respecting bounds."""
    ind = ind.copy()
    if rng.random() < mrate:
        for _ in range(rng.integers(1, 4)):
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(ind[i], 1.0 - ind[j])
            min_transfer = -min(ind[j], 1.0 - ind[i])
            if max_transfer > 0 or min_transfer < 0:
                delta = rng.uniform(min_transfer, max_transfer)
                ind[i] -= delta
                ind[j] += delta
        if rng.random() < 0.3:
            cnt = rng.integers(1, max(2, n // 4 + 1))
            idx = rng.choice(n, size=cnt, replace=False)
            ind[idx] += rng.normal(scale=0.02, size=idx.shape)
    ind = project_sum_and_bounds(ind, target)
    return ind

def initialize_population_generic(n, pop_size, target, rng, base=None):
    """Create an initial feasible population, optionally seeded by `base`."""
    pop = []
    if base is not None:
        base = np.array(base, dtype=float).copy()
        if len(base) != n:
            # linear interpolation to required size
            x_old = np.linspace(0, 2, len(base), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            base = np.interp(x_new, x_old, base)
        base = project_sum_and_bounds(base, target)
        pop.append(base)
        while len(pop) < pop_size:
            ind = base.copy()
            for _ in range(rng.integers(1, 5)):
                i, j = rng.choice(n, size=2, replace=False)
                max_transfer = min(ind[i], 1.0 - ind[j])
                min_transfer = -min(ind[j], 1.0 - ind[i])
                if max_transfer > 0 or min_transfer < 0:
                    delta = rng.uniform(min_transfer, max_transfer)
                    ind[i] -= delta
                    ind[j] += delta
            if rng.random() < 0.5:
                cnt = rng.integers(1, max(2, n // 4 + 1))
                idx = rng.choice(n, size=cnt, replace=False)
                ind[idx] += rng.normal(scale=0.02, size=idx.shape)
            ind = project_sum_and_bounds(ind, target)
            pop.append(ind)
    else:
        for _ in range(pop_size):
            ind = rng.random(n)
            ind = project_sum_and_bounds(ind, target)
            pop.append(ind)
    return pop

def upscale_h(h_small, n_large):
    """Upsample a step vector to a larger resolution."""
    n_small = len(h_small)
    if n_large % n_small == 0:
        factor = n_large // n_small
        return np.repeat(h_small, factor)
    x_small = np.linspace(0, 2, n_small, endpoint=False)
    x_large = np.linspace(0, 2, n_large, endpoint=False)
    return np.interp(x_large, x_small, h_small)

def hill_climb_refine(h, target, rng, time_budget):
    """Greedy local improvement limited by a time budget."""
    start = time.time()
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    n = len(h)
    while time.time() - start < time_budget:
        cand = cur.copy()
        for _ in range(rng.integers(1, 4)):
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(cand[i], 1.0 - cand[j])
            min_transfer = -min(cand[j], 1.0 - cand[i])
            if max_transfer > 0 or min_transfer < 0:
                delta = rng.uniform(min_transfer, max_transfer)
                cand[i] -= delta
                cand[j] += delta
        if rng.random() < 0.2:
            cnt = rng.integers(1, max(2, n // 4 + 1))
            idx = rng.choice(n, size=cnt, replace=False)
            cand[idx] += rng.normal(scale=0.01, size=idx.shape)
        cand = project_sum_and_bounds(cand, target)
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
        if rng.random() < 0.001:
            cur = rng.random(n)
            cur = project_sum_and_bounds(cur, target)
            cur_c5 = compute_c5(cur)
    return best_h, best_c5

def ga_optimize(n, target, rng, time_budget, base=None):
    """Simple GA with elitism and a final hill‑climb."""
    pop_size = 50
    tournament_size = 3
    crossover_rate = 0.9
    mutation_rate = 0.5
    elite_frac = 0.1

    population = initialize_population_generic(n, pop_size, target, rng, base)
    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    no_improve = 0
    start = time.time()

    while time.time() - start < time_budget:
        elite_count = max(1, int(pop_size * elite_frac))
        elite_idxs = np.argsort(fitness)[:elite_count]
        new_pop = [population[i].copy() for i in elite_idxs]
        while len(new_pop) < pop_size:
            p1 = tournament_select(population, fitness, rng, tournament_size)
            p2 = tournament_select(population, fitness, rng, tournament_size)
            c1, c2 = crossover(p1, p2, rng, crossover_rate)
            c1 = mutate_ind(c1, target, rng, mutation_rate, n)
            c2 = mutate_ind(c2, target, rng, mutation_rate, n)
            new_pop.append(c1)
            if len(new_pop) < pop_size:
                new_pop.append(c2)
        population = new_pop
        fitness = np.array([compute_c5(ind) for ind in population])
        cur_best = np.argmin(fitness)
        cur_c5 = fitness[cur_best]
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = population[cur_best].copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 2000:
            break

    remaining = time_budget - (time.time() - start)
    if remaining > 0.5:
        best_h, best_c5 = hill_climb_refine(best_h, target, rng, remaining * 0.5)

    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5
