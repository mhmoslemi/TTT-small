```
def adjust_sum(h, target):
    # Clip to [0,1], then adjust sum using water-filling on free indices
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    # iterative simple fix: allocate diff uniformly to free entries
    # free: those not at bound
    for _ in range(2):
        free = (h > 0) & (h < 1)
        if not np.any(free):
            break
        inc = diff / np.sum(free)
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = target - cur
    # final residual fix loop
    if abs(diff) > 1e-12:
        # go through entries to fix residual
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_improve(h, target_sum, budget, start_time):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(best_h)
    step = 0.2
    step_decay = 0.99
    max_iter = 100000
    for it in range(max_iter):
        if time.time() - start_time >= budget * 0.95:
            break
        # find worst shift (max C5)
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = np.argmax(corr)
        k = idx_max - (n - 1)
        # compute gradient of C5 = w - overlap, i.e., derivative is 1 - (h_{i+k}+h_{i-k})
        g = np.zeros_like(h)
        if k >= 0:
            # term for i in [0, n-k-1]: -h_{i+k}
            g[: n - k] -= h[k:]
            # term for i in [k, n-1]: -h_{i - k}
            g[k:] -= h[: n - k]
            # plus constant derivative +1 for all i (since d Σ h_i = 1)
            g += 1.0
        else:
            kk = -k
            g[kk:] -= h[: n - kk]
            g[: n - kk] -= h[kk:]
            g += 1.0
        # This gradient gives direction of increase of C5; we want to decrease C5 => move opposite
        grad_desc = -g
        # remove mean to keep sum direction zero (or rely on projection)
        grad_desc_centered = grad_desc - np.mean(grad_desc)
        max_abs = np.max(np.abs(grad_desc_centered))
        if max_abs < 1e-12:
            break
        alpha = step / max_abs
        cand = h + alpha * grad_desc_centered
        cand = np.clip(cand, 0.0, 1.0)
        cand = adjust_sum(cand, target_sum)
        cand_val = compute_c5(cand)
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_h = cand.copy()
            h = cand.copy()
            step = min(step * 1.05, 0.5)
        else:
            step *= step_decay
            # occasional random jitter
            if step < 1e-4:
                jitter = rng.normal(scale=0.01, size=n)
                cand2 = h + jitter
                cand2 = np.clip(cand2, 0, 1)
                cand2 = adjust_sum(cand2, target_sum)
                cand2_val = compute_c5(cand2)
                if cand2_val < best_val:
                    best_val, best_h, h = cand2_val, cand2.copy(), cand2.copy()
                step = 0.2

def slsqp_refine(h, target_sum, best_val, budget, start_time):
    n = len(h)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    # define constraint functions
    def sum_eq(x):
        return np.sum(x) - target_sum
    cons = [{"type":"eq","fun": sum_eq}]
    for kk in range(-n+1, n):
        if kk == 0:
            continue
        def make_ineq(k):
            def fun(x):
                hvec = x
                if k >= 0:
                    overl = np.dot(hvec[: n - k], 1.0 - hvec[k:])
                else:
                    kk2 = -k
                    overl = np.dot(hvec[kk2:], 1.0 - hvec[: n - kk2])
                return best_val - overl * dx  # we require best_val >= overl*dx, i.e., overl*dx <= best_val
            return fun
        cons.append({"type":"ineq","fun": make_ineq(kk)})
    # objective to minimize best_val (t)
    # Instead of adding extra variable for bound, we can directly minimize t which equals best_val, but we need variable for bound. We'll use t variable.
    # Include t as last variable
    def obj(x):
        return x[-1]
    # Build x0 = [h, best_val]
    x0 = np.concatenate([h, [best_val]])
    # Expand bounds
    bounds_full = bounds + [(0.0, None)]  # t >= 0
    # define constraints as before but using x[-1] in each
    cons_full = [{"type":"eq","fun": lambda x: np.sum(x[:n]) - target_sum}]
    for kk in range(-n+1, n):
        if kk == 0:
            continue
        def make_ineq(k):
            def fun(x):
                hvec = x[:n]
                t = x[-1]
                if k >= 0:
                    overl = np.dot(hvec[: n - k], 1.0 - hvec[k:])
                else:
                    kk2 = -k
                    overl = np.dot(hvec[kk2:], 1.0 - hvec[: n - kk2])
                return t - overl * dx
            return fun
        cons_full.append({"type":"ineq","fun": make_ineq(kk)})
    # call minimize
    try:
        res = minimize(fun=obj, x0=x0, method="SLSQP", bounds=bounds_full, constraints=cons_full,
                       options={"maxiter": 2000, "ftol": 1e-9, "disp": False})
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = adjust_sum(h_opt, target_sum)
            val_opt = compute_c5(h_opt)
            return h_opt, val_opt
    except Exception:
        pass
    return h, compute_c5(h)

import time
import numpy as np
from scipy.optimize import minimize

def _adjust_sum(vec, target):
    """Clip to [0,1] and adjust sum to target (exactly) by water-filling."""
    v = np.clip(vec, 0.0, 1.0)
    cur = np.sum(v)
    diff = target - cur
    # distribute diff over free entries
    for _ in range(2):
        free = (v > 0.0) & (v < 1.0)
        cnt = np.count_nonzero(free)
        if cnt == 0:
            break
        inc = diff / cnt
        v[free] += inc
        v = np.clip(v, 0.0, 1.0)
        cur = np.sum(v)
        diff = target - cur
    # residual fix (few steps)
    if abs(diff) > 1e-12:
        for i in range(len(v)):
            if abs(diff) < 1e-12:
                break
            if diff > 0.0 and v[i] < 1.0:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
            elif diff < 0.0 and v[i] > 0.0:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _gradient_step(h, target, rng, step):
    """One descent step on the worst shift (max C5)."""
    n = len(h)
    dx = 2.0 / n
    # worst shift (max C5)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)          # shift value (could be negative)
    # gradient of C5 for this shift: g = 1 - (h_{i+k}+h_{i-k})
    g = np.zeros_like(h)
    if k >= 0:
        g[: n - k] -= h[k:]
        g[k:]     -= h[: n - k]
    else:
        kk = -k
        g[kk:]    -= h[: n - kk]
        g[: n - kk] -= h[kk:]
    g += 1.0                     # add the constant part from Σ h_i
    # descent direction (we want to *decrease* C5)
    d = -g
    # zero‑mean direction to keep sum after projection
    d -= np.mean(d)
    max_abs = np.max(np.abs(d))
    if max_abs < 1e-12:
        return h, step
    alpha = step / max_abs        # scale so max change equals step
    cand = h + alpha * d
    cand = np.clip(cand, 0.0, 1.0)
    cand = _adjust_sum(cand, target)
    return cand, step

def _local_optimize(start_h, target, rng, start_time, budget):
    """Iterative gradient descent with adaptive step size."""
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_value(best_h)
    step = 0.2
    decay = 0.99
    while True:
        now = time.time()
        if now - start_time >= budget * 0.95:
            break
        cand, step = _gradient_step(h, target, rng, step)
        cand_val = _c5_value(cand)
        if cand_val + 1e-12 < best_val:
            best_val = cand_val
            best_h = cand.copy()
            h = cand.copy()
            step = min(step * 1.05, 0.5)   # modest increase after improvement
        else:
            step *= decay
            # occasional random jitter when step becomes tiny
            if step < 1e-4:
                jitter = rng.normal(scale=0.01, size=len(h))
                cand2 = h + jitter
                cand2 = np.clip(cand2, 0.0, 1.0)
                cand2 = _adjust_sum(cand2, target)
                cand2_val = _c5_value(cand2)
                if cand2_val < best_val:
                    best_val = cand2_val
                    best_h = cand2.copy()
                    h = cand2.copy()
                step = 0.2
    return best_h, best_val

def _slsqp_refine(h, target, init_val, start_time, budget):
    """Optional SLSQP polishing (same constraints as before)."""
    n = len(h)
    dx = 2.0 / n
    # bounds for h and for the auxiliary t variable
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    # equality sum constraint
    def sum_eq(x):
        return np.sum(x[:n]) - target
    constraints = [{"type": "eq", "fun": sum_eq}]
    # inequality for each non‑zero shift: t >= overlap_k*dx
    def make_ineq(k):
        if k >= 0:
            def fun(x):
                hv = x[:n]
                ov = np.dot(hv[: n - k], 1.0 - hv[k:])
                return x[-1] - ov * dx
        else:
            kk = -k
            def fun(x):
                hv = x[:n]
                ov = np.dot(hv[kk:], 1.0 - hv[: n - kk])
                return x[-1] - ov * dx
        return fun
    for kk in range(-n + 1, n):
        if kk == 0:
            continue
        constraints.append({"type": "ineq", "fun": make_ineq(kk)})
    x0 = np.concatenate([h, [init_val]])
    remaining = budget - (time.time() - start_time)
    if remaining < 5.0:
        return h, init_val
    try:
        res = minimize(
            fun=lambda x: x[-1],
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 2000, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            h_opt = _adjust_sum(np.clip(res.x[:n], 0.0, 1.0), target)
            val_opt = _c5_value(h_opt)
            if val_opt + 1e-12 < init_val:
                return h_opt, val_opt
    except Exception:
        pass
    return h, init_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        h0 = None
    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- build a pool of starting vectors ----------------------------
    candidates = []

    # use provided initial vector if present
    if h0 is not None:
        candidates.append(_adjust_sum(h0.copy(), target_sum))

    # triangular (tent) shape, peaks at centre, integral = 1 already
    xs = np.linspace(0.0, 2.0 - dx, n) + dx / 2.0
    tri = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    candidates.append(_adjust_sum(tri, target_sum))

    # sine‑squared shape (same integral)
    sin_sq = np.sin(np.pi * xs / 2.0) ** 2
    candidates.append(_adjust_sum(sin_sq, target_sum))

    # binary masks (random)
    for _ in range(2):
        mask = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        mask[idx] = 1.0
        candidates.append(mask)

    # random fractional vectors
    for _ in range(2):
        rnd = rng.random(n)
        candidates.append(_adjust_sum(rnd, target_sum))

    # two‑level block families: a block of 1’s centred, rest a constant level
    for m in range(10, n//2 + 1, 5):
        if m >= n:
            continue
        l = (target_sum - m) / (n - m)
        if not (0.0 <= l <= 1.0):
            continue
        block = np.full(n, l, dtype=float)
        start_idx = (n - m) // 2
        block[start_idx:start_idx + m] = 1.0
        candidates.append(_adjust_sum(block, target_sum))

    best_h = None
    best_val = float("inf")

    # ----- main optimisation loop (gradient descent per candidate) -----
    for cand in candidates:
        if time.time() - start >= budget_s * 0.90:
            break
        h_opt, val_opt = _local_optimize(cand, target_sum, rng, start, budget_s)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break  # early success

    # ----- optional smoothing passes (simple 3‑point average) ---------
    if best_h is not None and time.time() - start < budget_s - 5.0:
        kernel = np.array([0.25, 0.5, 0.25])
        for _ in range(6):
            sm = np.convolve(best_h, kernel, mode="same")
            sm = np.clip(sm, 0.0, 1.0)
            sm = _adjust_sum(sm, target_sum)
            sm_val = _c5_value(sm)
            if sm_val + 1e-12 < best_val:
                best_val = sm_val
                best_h = sm
            else:
                break

    # ----- final SLSQP refinement if time permits --------------------
    if best_h is not None and (time.time() - start) < budget_s - 3.0:
        best_h, best_val = _slsqp_refine(best_h, target_sum, best_val, start, budget_s)

    # ----- final safety corrections ------------------------------------
    if best_h is None:
        # fallback: uniform 0.5 (should never happen)
        best_h = np.full(n, 0.5, dtype=float)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _adjust_sum(best_h, target_sum)
    best_val = _c5_value(best_h)

    return best_h, best_val, n
