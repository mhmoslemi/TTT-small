def gradient_worst(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full')
    f = corr * dx
    worst_idx = np.argmax(f)
    k = worst_idx - (n - 1)
    # compute gradient
    grad = np.zeros_like(h)
    if k >= 0:
        # h_i h_{i+k} term exists for i in 0..n-1-k
        # derivative -dx * h_{i+k} for i in 0..n-1-k
        grad[:n-k] -= dx * h[k:]   # for i in 0..n-1-k: -dx * h[i+k]

        grad[k:] -= dx * h[:n-k]   # for i in k..n-1: -dx * h[i-k]

def get_worst_shift_and_grad(h):

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # v is a 1D numpy array
    # Find λ such that sum(clip(v + λ, lo, hi)) = target
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(50):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while enforcing sum(v)=target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def worst_shift_and_grad(h):
    """Return subgradient for the shift attaining the maximal overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)                # index of worst shift (no dx needed for argmax)
    k = idx - (n - 1)                    # actual shift
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] -= dx * h[k:]       # -dx * h_{i+k}
        grad[k:]     -= dx * h[:n - k]   # -dx * h_{i-k}
    else:
        kk = -k
        grad[kk:]    -= dx * h[:n - kk]
        grad[:n - kk] -= dx * h[kk:]
    return grad, idx, k, corr[idx] * dx

# ---------- Constraint classes ----------
class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k > 0:
            ov = np.dot(h[:self.n - k], 1.0 - h[k:])
        elif k < 0:
            kk = -k
            ov = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        else:
            ov = 0.0
        return x[-1] - ov * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{'type': 'eq', 'fun': sum_con}]
    cons += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Initialize dimensionality and starting vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        factor = 2                           # double the resolution (80 -> 160)
        h0 = np.repeat(base, factor)
        n = h0.size
    else:
        n = 160
        h0 = rng.random(n)

    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---------- Phase 1: Projected Gradient Descent (multiple starts) ----------
    gd_deadline = start_time + budget_s * 0.55
    num_starts = 8
    max_iter_gd = 3000
    for s in range(num_starts):
        if time.time() > gd_deadline:
            break
        if s == 0:
            cur_h = best_h.copy()
        else:
            cur_h = rng.random(n)
            cur_h = project_to_box_sum(cur_h, target_sum)

        cur_best_h = cur_h.copy()
        cur_best_c5 = compute_c5(cur_h)

        for it in range(1, max_iter_gd + 1):
            if time.time() > gd_deadline:
                break
            grad, _, _, _ = worst_shift_and_grad(cur_h)
            step = 5.0 / np.sqrt(it)
            cur_h = cur_h - step * grad
            cur_h = project_to_box_sum(cur_h, target_sum)
            cur_c5 = compute_c5(cur_h)
            if cur_c5 < cur_best_c5:
                cur_best_c5 = cur_c5
                cur_best_h = cur_h.copy()
                if cur_best_c5 < best_c5:
                    best_c5 = cur_best_c5
                    best_h = cur_best_h.copy()
                if best_c5 <= 0.3808:
                    break
        if cur_best_c5 < best_c5:
            best_c5 = cur_best_c5
            best_h = cur_best_h.copy()
        if best_c5 <= 0.3808:
            break

    # ---------- Phase 2: Simulated Annealing ----------
    sa_deadline = start_time + budget_s * 0.94  # reserve a little for final polish
    cur_h = best_h.copy()
    cur_c5 = best_c5
    T0 = 0.02
    Tf = 1e-5
    while time.time() < sa_deadline:
        elapsed = time.time() - start_time
        total = sa_deadline - start_time
        frac = (elapsed - (budget_s * 0.55)) / (total - (budget_s * 0.55))
        frac = max(0.0, min(1.0, frac))
        T = T0 * (Tf / T0) ** frac
        sigma = 0.5 * T
        cand = cur_h + sigma * rng.normal(size=n)
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 < cur_c5 or rng.random() < np.exp(-(cand_c5 - cur_c5) / max(T, 1e-12)):
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                if best_c5 <= 0.3808:
                    break

    # ---------- Phase 3: SLSQP refinement ----------
    final_deadline = start_time + budget_s - 0.5
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    for restart in range(3):
        if time.time() > final_deadline:
            break
        t_start = compute_c5(best_h)
        res = slsqp_optimize(best_h, t_start, sum_con, overlap_cons, maxiter=400)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                if best_c5 <= 0.3808:
                    break
        # small random perturbation for next restart
        best_h = project_to_box_sum(best_h + 0.02 * rng.normal(size=n), target_sum)

    # ---------- Final safety projection ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
