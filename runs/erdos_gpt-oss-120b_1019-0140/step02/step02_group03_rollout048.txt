def project_to_box_sum(v, target, lo=0.0, hi=1.0, max_iter=60, tol=1e-9):
    lo_l = lo - v
    hi_l = hi - v
    # The lambda must be in interval such that sum clip(v+λ) = target.
    # Since clip is monotonic in λ, we can binary search.
    left = -np.max(hi_l) - 1.0  # small more extreme
    right = np.max(lo_l) + 1.0 
    for _ in range(max_iter):
        mid = (left + right) / 2.0
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            left = mid
        else:
            right = mid
    return np.clip(v + (left + right) / 2.0, lo, hi)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = None

    target = n / 2.0

    if h0 is not None:
        # ensure feasible
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target)
        best_h = h0
        best_score = compute_c5(best_h)
    else:
        # random feasible start
        best_h = project_to_box_sum(rng.random(n), target)
        best_score = compute_c5(best_h)

    # Build initial population
    pop_size = 30
    population = []
    # include best_h as part
    population.append((best_score, best_h.copy()))
    # fill rest
    while len(population) < pop_size:
        vec = project_to_box_sum(rng.random(n), target)
        score = compute_c5(vec)
        population.append((score, vec))
    # sort
    population.sort(key=lambda x: x[0])

    # GA loop
    # dynamic parameters
    base_mut_sigma = 0.15
    mutation_rate = 0.2
    # time budgeting
    while time.time() < start + budget_s * 0.7: # use 70% for GA
        # tournament selection
        def select():
            idxs = rng.choice(pop_size, size=3, replace=False)
            # pick the one with smallest score
            best_idx = min(idxs, key=lambda i: population[i][0])
            return population[best_idx][1]
        parent_a = select()
        parent_b = select()
        # crossover
        mask = rng.random(n) < 0.5
        child = np.where(mask, parent_a, parent_b)
        # mutation
        mut_mask = rng.random(n) < mutation_rate
        child[mut_mask] += rng.normal(scale=base_mut_sigma, size=mut_mask.sum())
        # repair
        child = project_to_box_sum(np.clip(child, 0.0, 1.0), target)
        child_score = compute_c5(child)
        # insertion if better than worst
        if child_score < population[-1][0]:
            # replace worst
            population.pop()
            # insert sorted using binary insertion (insort)
            # but we can just append and sort, simpler given small size
            population.append((child_score, child))
            population.sort(key=lambda x: x[0])
            # update best if necessary
            if child_score < best_score:
                best_h = child.copy()
                best_score = child_score
                # early exit if good enough
                if best_score <= 0.3808:
                    break
        # adapt mutation sigma
        # maybe gradually reduce sigma
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.7)
        # linearly decay sigma
        base_mut_sigma = max(0.01, base_mut_sigma * (1 - frac))

    # Hill climbing phase for remaining budget
    cur_h = best_h.copy()
    cur_score = best_score
    base_step = 0.5
    # stage until total budget
    while time.time() < start + budget_s - 0.5:
        # random direction
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2)
        cand = cur_h + step * d
        cand = project_to_box_sum(np.clip(cand, 0.0, 1.0), target)
        cand_score = compute_c5(cand)
        if cand_score + 1e-12 < cur_score:
            cur_h = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score
                if best_score <= 0.3808:
                    break
        # occasional shrink step
        if rng.random() < 0.001:
            base_step *= 0.9

    # final projection
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0, max_iter=60):
    # binary search lambda
    # ensure v is numpy array
    v = np.asarray(v, dtype=np.float64)
    low = -np.max(v) - 1.0
    high = (hi - np.min(v)) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) / 2.0, lo, hi)
