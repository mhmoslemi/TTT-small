import time
import numpy as np

# ----- Helper: exact C5 evaluation -----
def compute_c5(h):
    """Maximum overlap integral for discretised h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# ----- Helper: projection onto [0,1]∩{sum=M} -----
def proj_sum_bounds(x, M, max_iter=60, tol=1e-12):
    """
    Project vector x onto the hypercube [0,1] while enforcing sum(x)=M.
    Finds λ such that sum(clip(x-λ,0,1)) = M via bisection.
    """
    lo, hi = -1.0, 2.0  # guarantees sum at lo >= M, at hi <= M
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s < M:
            hi = lam
        else:
            lo = lam
        if hi - lo < tol:
            break
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

# ----- Helper: generate a random feasible h -----
def random_feasible_h(n, M, rng):
    """Uniform random start then projection onto constraints."""
    x = rng.random(n)
    return proj_sum_bounds(x, M)

# ----- Gradient descent with soft‑max smoothing -----
def smooth_grad_step(h, M, alpha=5.0, eta=0.2):
    """
    One gradient descent step on the smoothed max-overlap objective.
    Returns the projected next iterate.
    """
    n = h.shape[0]
    # correlation (without scaling)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = corr.max()
    # soft‑max weights
    w = np.exp(alpha * (corr - max_corr))
    w /= w.sum()
    # gradient accumulation
    grad = np.zeros_like(h)
    # shifts from -(n-1) .. (n-1)
    for idx, k in enumerate(range(-n + 1, n)):
        wk = w[idx]
        if wk == 0.0:
            continue
        if k >= 0:
            grad[: n - k] += wk * (1.0 - h[k:])
            grad[k:]       -= wk * h[: n - k]
        else:
            kk = -k
            grad[kk:]      += wk * (1.0 - h[: n - kk])
            grad[: n - kk] -= wk * h[kk:]
    h_next = h - eta * grad
    return proj_sum_bounds(h_next, M)

# ----- Local annealing (mass‑transfer moves) -----
def anneal_h(h, M, time_limit, rng, best_global, best_h_global, n):
    """
    Simulated‑annealing-like refinement using pairwise mass transfers.
    Returns possibly improved (h, c5) and updates global best via references.
    """
    start = time.time()
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    T0 = 0.001
    while time.time() - start < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # decide direction of transfer
        if cur_h[j] > 0 and cur_h[i] < 1:
            max_tr = min(cur_h[j], 1.0 - cur_h[i])
            a, b = i, j
        elif cur_h[i] > 0 and cur_h[j] < 1:
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            a, b = j, i
        else:
            continue
        if max_tr <= 0:
            continue
        delta = max_tr * rng.random()
        new_h = cur_h.copy()
        new_h[a] += delta
        new_h[b] -= delta
        new_c5 = compute_c5(new_h)
        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / time_limit)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        elif T > 0 and rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            accept = True
        if accept:

            if new_c5 < best_global:
                best_global = new_c5
                best_h_global[:] = new_h
        if best_global < 0.3808:
            break
    return best_global, best_h_global

            # ensure array is mutable
            best_vec_n = best_vec_n.copy()
            best_c5, best_vec_n = anneal_h(
                best_vec_n, M, remaining_n, rng, best_c5, best_vec_n, n
            )
            if best_c5 < best_c5:
                best_c5, best_h, best_n = best_c5, best_vec_n.copy(), n

        if best_c5 < 0.3808:
            break

    # ----- fallback if nothing found -----
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    # final safety: enforce constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    M = best_n // 2
    cur_sum = best_h.sum()
    if abs(cur_sum - M) > 1e-9:
        best_h = proj_sum_bounds(best_h, M)

    # use the official evaluator for the final returned bound (if available)
    try:
        best_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        best_c5 = float(compute_c5(best_h))

    return best_h, best_c5, best_n
