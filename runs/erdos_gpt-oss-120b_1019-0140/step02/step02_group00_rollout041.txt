def legendre_sequence(p):
    # p must be prime, odd
    seq = np.zeros(p-1, dtype=float)
    for i in range(1, p):
        # Legendre symbol (i|p) is 1 if i is qr, -1 if non-residue, 0 if i mod p = 0 (i=0)
        # Use pow(i, (p-1)//2, p) returns 1 for residues, p-1 for non-residues (i.e., -1)
        if pow(i, (p-1)//2, p) == 1:
            seq[i-1] = 1.0
        else:
            seq[i-1] = 0.0 # treat non-residues as 0
    return seq

def thue_morse_binary(n):
    # produce binary sequence of length n with values 0/1 (Thue-Morse).
    # Use parity of number of 1 bits: 0 if even #bits else 1.
    seq = np.arange(n, dtype=int)
    # compute bit count parity
    cnt = np.vectorize(lambda x: bin(x).count('1') & 1, otypes=[int])
    values = cnt(seq)
    # values is array of 0/1
    return values.astype(float)

        pos_j = rng.integers(len(zeros_idx))

                best_h = h.copy()

            # revert
            h[i] = 1
            h[j] = 0
        # optional acceptance with probability (annealing) skipping
    return best_h, best_score

    x0 = np.concatenate([h0, [0.0]])  # t guess possibly high
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.-h, mode='full') * (2.0/n)
        # t must be >= max corr
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,

    h_opt = res.x[:n]
    # project to box sum
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=100000):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

```python
import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to change length (up‑ or down‑sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    """Set exactly half of entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    # argpartition for speed
    idx = np.argpartition(-arr, half-1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        # Legendre symbol (i|p) is 1 if i is a quadratic residue, else -1
        if pow(i, exp, p) == 1:
            seq[i - 1] = 1.0
        else:
            seq[i - 1] = 0.0
    return seq

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    """Projected GD with momentum on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=400):
    """Short SLSQP to improve a small‑scale candidate."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to change length (up‑ or down‑sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):

def random_binary_balanced(n, rng):

def binarize_from_real(arr):
    """Set exactly half of entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        if pow(i, exp, p) == 1:
            seq[i - 1] = 1.0
        else:
            seq[i - 1] = 0.0
    return seq

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    """Projected GD with momentum on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
