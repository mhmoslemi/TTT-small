    n = len(initial_h)
    cur_h = initial_h.copy()
    cur_val = compute_c5_vec(cur_h.astype(float), dx)
    best_h = cur_h.copy()
    best_val = cur_val

    # Setting initial temperature as 0.001 or something based on cur_val.
    T0 = 0.001
    Tmin = 1e-6
    start = time.time()
    iteration = 0
    while True:
        elapsed = time.time() - start
        if time.time() > time_limit:
            break
        # schedule temperature decreasing linearly
        frac = elapsed / (time_limit - start)
        T = T0 * (1 - frac) + Tmin * frac  # linear cooling

import time
import numpy as np

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    cur = np.sum(x)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return x
    # If diff > 0, distribute to entries with capacity (1-x)
    # fill sorted by capacity descending
    # For simplicity, loop over entries
    # We'll add uniform distribution then clip, similar to before
    if diff > 0:
        # Try to add uniformly
        increment = diff / len(x)
        x += increment
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - np.sum(x)
        if diff > 1e-12:
            # add extra to entries not yet at 1
            for idx in np.where(x < 1.0)[0]:
                add = min(diff, 1.0 - x[idx])
                x[idx] += add
                diff -= add
                if diff <= 1e-12:
                    break
    else:
        # diff < 0: reduce uniformly
        decrement = diff / len(x)  # negative value
        x += decrement
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - np.sum(x)
        if diff < -1e-12:
            for idx in np.where(x > 0.0)[0]:
                sub = min(-diff, x[idx])
                x[idx] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    # final adjust in case off by small epsilon
    # distribute remaining diff
    if abs(target_sum - np.sum(x)) > 1e-10:
        # adjust by small epsilon on arbitrary indices
        for idx in range(len(x)):
            if target_sum - np.sum(x) > 1e-12 and x[idx] < 1.0:
                inc = min(1.0 - x[idx], target_sum - np.sum(x))
                x[idx] += inc
            elif target_sum - np.sum(x) < -1e-12 and x[idx] > 0.0:
                dec = min(x[idx], np.sum(x) - target_sum)
                x[idx] -= dec
            if abs(target_sum - np.sum(x)) < 1e-12:
                break
    return x

import time

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    cur = np.sum(x)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return x
    # Try uniform adjustment then clip
    if diff > 0:
        inc = diff / len(x)
        x += inc
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - np.sum(x)
        if diff > 1e-12:
            for idx in np.where(x < 1.0)[0]:
                add = min(diff, 1.0 - x[idx])
                x[idx] += add
                diff -= add
                if diff <= 1e-12:
                    break
    else:
        dec = diff / len(x)
        x += dec
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - np.sum(x)
        if diff < -1e-12:
            for idx in np.where(x > 0.0)[0]:
                sub = min(-diff, x[idx])
                x[idx] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    # Final tiny correction if needed
    if abs(target_sum - np.sum(x)) > 1e-10:
        for idx in range(len(x)):
            cur_sum = np.sum(x)
            if cur_sum < target_sum - 1e-12 and x[idx] < 1.0:
                inc = min(target_sum - cur_sum, 1.0 - x[idx])
                x[idx] += inc
            elif cur_sum > target_sum + 1e-12 and x[idx] > 0.0:
                dec = min(cur_sum - target_sum, x[idx])
                x[idx] -= dec
            if abs(np.sum(x) - target_sum) < 1e-12:
                break
    return x

        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5(cur_h, dx)
    return cur_h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    if h0 is not None:
        best_h = np.clip(h0, 0.0, 1.0)
        best_h = project_to_sum(best_h, target_sum)
        best_val = compute_c5(best_h, dx)
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argsort(-best_h)[:int(target_sum)]
        binary_h[top_idx] = 1
    else:
        binary_h = generate_random_binary(n, target_sum, rng)
        best_h = binary_h.astype(float)
        best_val = compute_c5(best_h, dx)

    binary_h = np.rint(best_h).astype(int)

    total_budget = budget_s
    time_for_sa = total_budget * 0.85
    time_for_refine = total_budget - time_for_sa
    if time_for_refine < 5.0:
        time_for_refine = 5.0
        time_for_sa = max(0.0, total_budget - time_for_refine)

    min_run = 12.0
    while True:
        elapsed = time.time() - start_time
        remaining_sa = time_for_sa - elapsed
        if remaining_sa < min_run:
            break
        run_time = min(remaining_sa, max(min_run, remaining_sa * 0.5))
        cand_h, cand_val = binary_sa(binary_h, dx, target_sum,
                                    time.time() + run_time, rng)
        if cand_val < best_val - 1e-12:
            best_h = cand_h
            best_val = cand_val
            binary_h = np.rint(cand_h).astype(int)
        if best_val < 0.3808:
            break

    time_left = budget_s - (time.time() - start_time)
    if time_left > 2.0:
        refined_h, refined_val = refine_continuous(best_h, dx, target_sum,
                                                   time.time() + time_left,
                                                   start_time, rng)
        if refined_val < best_val - 1e-12:
            best_h = refined_h
            best_val = refined_val

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5(best_h, dx)

    return best_h, float(best_val), n
