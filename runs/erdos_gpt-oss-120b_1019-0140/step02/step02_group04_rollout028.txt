def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # add uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc

import time
import numpy as np

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def compute_c5_raw(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    if diff > 0:
        idx = np.where(vec < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(vec > 0.0)[0]
        for i in idx:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return vec

def random_feasible(n, target_sum, rng):
    vec = rng.random(n)
    return project_to_sum(vec, target_sum)

def gradient_for_shift(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # base sum derivative for i in [0, n-k)
        grad[: n - k] += 1.0
        # subtract product contributions
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        d = -k
        grad[d:] += 1.0
        grad[d:] -= h[: n - d]
        grad[: n - d] -= h[d:]
    return grad

def binary_hill_climb(h_bin, cur_val, dx, time_limit, rng):
    best_h = h_bin.copy()
    best_val = cur_val
    n = len(h_bin)
    while time.time() < time_limit:
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        if len(ones) == 0 or len(zeros) == 0:
            break

        j = rng.choice(zeros)
