def adjust_sum_and_bounds(h, M):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur == M:
        return h
    # simple scaling (if scaling stays within bounds)
    if cur > 0:
        scale = M / cur
        h = np.clip(h * scale, 0.0, 1.0)
        cur = np.sum(h)
    # after scaling may still be off due to clipping; fix by adding/subtracting small amounts
    diff = M - cur
    if abs(diff) > 1e-12:
        # distribute diff across indices that are not at bounds
        # iterate
        indices = np.arange(len(h))
        # order doesn't matter; just loop
        for idx in indices:
            if diff > 0:
                inc = min(diff, 1.0 - h[idx])
                if inc > 0:
                    h[idx] += inc
                    diff -= inc
            else:
                dec = min(-diff, h[idx])
                if dec > 0:
                    h[idx] -= dec
                    diff += dec
            if abs(diff) < 1e-12:
                break
    return h
```

def anneal_h(initial_h, M, budget, rng):
    h = initial_h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    start = time.time()
    # temperature schedule
    T0 = 0.005
    while True:
        elapsed = time.time() - start
        if elapsed > budget:
            break
        # progress fraction
        progress = elapsed / budget
        T = T0 * (1 - progress)
        # random pair
        i, j = rng.choice(len(h), 2, replace=False)
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        # delta proportion: maybe exponential for early big moves
        delta = rng.random() * max_transfer
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # compute c5
        c5_new = compute_c5(h_new)
        # accept
        if c5_new < c5 or rng.random() < math.exp(-(c5_new - c5) / max(T, 1e-9)):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                # early exit if below target
                if best_c5 < 0.3808:
                    break
        # else reject
    return best_h, best_c5

import time
import math
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discrete h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def adjust_sum_and_bounds(h, M):
    """Clip to [0,1] and enforce exact sum M (M = n//2)."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur != M:
        # scale if possible
        if cur > 0:
            scale = M / cur
            h = np.clip(h * scale, 0.0, 1.0)
            cur = np.sum(h)
        diff = M - cur
        # distribute remaining difference
        for idx in range(len(h)):
            if diff > 0:
                inc = min(diff, 1.0 - h[idx])
                if inc > 0:
                    h[idx] += inc
                    diff -= inc
            else:
                dec = min(-diff, h[idx])
                if dec > 0:
                    h[idx] -= dec
                    diff += dec
            if abs(diff) < 1e-12:
                break
    # final tiny correction if needed
    diff = M - np.sum(h)
    if abs(diff) > 1e-12:
        for idx in range(len(h)):
            if diff > 0:
                inc = min(diff, 1.0 - h[idx])
                h[idx] += inc
                diff -= inc
            else:
                dec = min(-diff, h[idx])
                h[idx] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return h

def anneal_h(h0, alloc, rng):
    """Simulated annealing on a feasible h."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    start = time.time()
    T0 = 0.005
    n = len(h)
    while True:
        elapsed = time.time() - start
        if elapsed > alloc:
            break
        progress = elapsed / alloc
        T = T0 * (1.0 - progress)
        i, j = rng.choice(n, 2, replace=False)
        max_transfer = min(1.0 - h[i], h[j])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            accept = True
        else:
            prob = math.exp(-(c5_new - c5_cur) / max(T, 1e-12))
            accept = rng.random() < prob
        if accept:
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h.copy()
                if best_c5 < 0.3808:
                    break
    return best_h, best_c5

def optimise_for_n(n, budget, rng):
    """Run several annealing restarts for a given n within budget seconds."""
    M = n // 2
    # base initial vector
    parent = globals().get('initial_h_values')
    if parent is not None and len(parent) == n:
        init = adjust_sum_and_bounds(np.asarray(parent, dtype=float), M)
    else:
        init = np.full(n, 0.5)
    best_h = None
    best_c5 = np.inf
    # number of restarts adapts to budget
    restarts = max(1, int(budget // 30))
    per_restart = budget / restarts
    for r in range(restarts):
        if r == 0:
            h0 = init.copy()
        else:
            # random feasible start
            x = rng.random(n)
            x = x / np.sum(x) * M
            h0 = adjust_sum_and_bounds(x, M)
        h_opt, c5_opt = anneal_h(h0, per_restart, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            if best_c5 < 0.3808:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate discretisations (even)
    candidates = [80, 96, 112, 128, 144]
    parent = globals().get('initial_h_values')
    if parent is not None:
        n0 = len(parent)
        if n0 % 2 == 0:
            if n0 not in candidates:
                candidates = [n0] + candidates
            else:
                candidates = [n0] + [n for n in candidates if n != n0]

    for n in candidates:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        # allocate at most 250 s per size, but keep some for later steps
        alloc = min(250.0, remaining * 0.4)
        h_opt, c5_opt = optimise_for_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n
            if best_c5 < 0.3808:
                break

    # final local refinement (pairwise transfers) with spare time
    if best_h is not None:
        n = best_n
        M = n // 2
        h_cur = best_h.copy()
        c5_cur = best_c5
        while time.time() - start_time < budget_s - 0.5:
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(1.0 - h_cur[i], h_cur[j])
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < c5_cur:
                h_cur = h_new
                c5_cur = c5_new
                best_h = h_cur
                best_c5 = c5_cur
                if best_c5 < 0.3808:
                    break
    else:
        # fallback constant solution
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # final constraint enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_and_bounds(best_h, best_n // 2)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
