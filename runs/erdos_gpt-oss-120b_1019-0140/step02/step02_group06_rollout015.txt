def hill_climb_binary(h_bin, target_sum, time_limit, rng):
    # h_bin is binary (0/1) with sum = target_sum.
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

            best_c5 = c5_new
    return best_h, best_c5

def optimize_h(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0/n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 5.0
    max_beta = 200.0
    alpha0 = 0.1
    momentum = np.zeros_like(h)
    mu = 0.9
    no_improve = 0
    while time.time() - start < time_budget:
        # compute correlation
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # update beta
        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)  # faster increase
        # compute softmax weights
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w /= w.sum()
        # compute gradient
        g = np.zeros(n)
        offset = n - 1
        # iterate over non-negligible weights
        # pre-select indices where w > threshold
        thresh = 1e-12
        # Use loop of all w but skip small ones
        for idx in range(len(w)):
            weight = w[idx]
            if weight <= thresh:
                continue
            k = idx - offset
            if k >= 0:
                # tail: i from 0 to n - k -1, head: i+k from k to n-1
                # length = n - k
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            # g[tail] += weight * (1 - h[head])
            # g[head] -= weight * h[tail]
            # Use vectorized operations
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # zero sum gradient to preserve total mass
        g -= np.mean(g)
        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        # momentum update
        momentum = mu * momentum - alpha * g
        h_new = h + momentum
        # project to feasible
        h_new = project_sum_and_bounds(h_new, target)
        # compute new c5
        c5_new = compute_c5(h_new)
        # acceptance
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        # random shake
        if no_improve > 5000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            momentum = np.zeros_like(h)
            no_improve = 0
    return best_h, best_c5

def project_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # simple uniform adjustment
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # iterative fix
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap value multiplied by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip x to [0,1] and adjust to make sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def optimize_h(h0, time_budget, seed):
    """Gradient descent with soft‑max smoothing and momentum."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 5.0                # initial smoothing
    max_beta = 200.0
    alpha0 = 0.1              # base step size
    mu = 0.9                  # momentum factor
    momentum = np.zeros_like(h)
    no_improve = 0

    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)

        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w /= w.sum()

        g = np.zeros(n)
        offset = n - 1
        thresh = 1e-16
        for idx, weight in enumerate(w):
            if weight <= thresh:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        g -= np.mean(g)  # preserve total mass in the update

        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        momentum = mu * momentum - alpha * g
        h_new = h + momentum
        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        if no_improve > 5000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            momentum = np.zeros_like(h)
            no_improve = 0

    return best_h, best_c5

def hill_climb_binary(h_bin, time_budget, rng):
    """Swap‑based hill climbing on a binary vector."""
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_budget:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_h = new_h
            best_c5 = c5_new
    return best_h, best_c5
