def gradient_refine(h, current_best, time_limit, rng):
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    # we may optionally use current_best to early stop
    while time.time() - start < time_limit - 0.1:
        # compute full correlation to get max overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        overlaps = corr * dx
        max_idx = np.argmax(overlaps)
        k = max_idx - (n - 1)
        # if k==0, pick random non-zero shift
        if k == 0:
            # pick random shift not zero
            k = rng.choice(np.delete(np.arange(-n+1, n), n-1))
        # compute benefit b_i
        benefit = np.zeros(n)
        if k > 0:
            benefit[:n - k] += best_h[k:]
            benefit[k:] += best_h[:n - k]
        elif k < 0:
            kk = -k
            benefit[kk:] += best_h[:n - kk]
            benefit[:n - kk] += best_h[kk:]
        # find candidates
        # To make robust, we can sort indices by benefit and pick extremes.
        # But we will just pick random among low/big benefit sets for variability.
        # Determine dec and inc sets:
        dec_mask = best_h > 1e-12
        inc_mask = best_h < 1 - 1e-12
        if not np.any(dec_mask) or not np.any(inc_mask):

        dec_idx = dec_idxs[np.argmin(benefit[dec_idxs])]
        # pick inc index with largest benefit
        inc_idx = inc_idxs[np.argmax(benefit[inc_idxs])]
        # compute max transfer

def optimise_for_n(n, time_budget, rng):
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf
    start = time.time()
    use_parent = False
    if 'initial_h_values' in globals():
        if len(initial_h_values) == n:
            use_parent = True
            parent_h = np.array(initial_h_values, dtype=float)
            cur = np.sum(parent_h)
            if abs(cur - M) > 1e-12:
                parent_h += (M - cur) / n
                parent_h = np.clip(parent_h, 0.0, 1.0)
            parent_c5 = compute_c5(parent_h)
            best_h, best_c5 = parent_h, parent_c5

    while True:
        elapsed = time.time() - start
        if elapsed > time_budget - 0.2:
            break
        if use_parent:
            h0 = parent_h.copy()
            use_parent = False
        else:
            h0 = generate_initial_h(n, M, rng)
        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])
        try:
            res = minimize(fun=objective_t,
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 600, 'ftol': 1e-10, 'disp': False})
        except Exception:
            continue
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # enforce sum exactly
        cur = np.sum(h_opt)
        if abs(cur - M) > 1e-9:
            h_opt += (M - cur) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            # early stop if good
            if best_c5 < 0.3808:
                break

    # after SLSQP loop, attempt gradient refinement within remaining time
    remaining = time_budget - (time.time() - start)
    if remaining > 1.0 and best_h is not None:
        # allocate at most half of remaining for refinement
        refine_time = min(remaining * 0.5, remaining - 0.5)
        h_ref, c5_ref = gradient_refine(best_h, best_c5, refine_time, rng)
        if c5_ref + 1e-12 < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper Classes ----------
class SumConstraint:
    """Enforce sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Enforce t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ---------- Core Evaluation ----------
def compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_t(x):
    """Objective for the auxiliary variable t."""
    return x[-1]

# ---------- Initial Vector Generator ----------
def generate_initial_h(n, M, rng):
    """Generate a feasible h (0<=h<=1, sum = M)."""
    method = rng.integers(3)
    if method == 0:  # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:  # uniform then project
        h = rng.random(n)
        h = h / np.sum(h) * M
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    else:  # linear decreasing shape
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    return h

# ---------- Gradient‑guided Local Refinement ----------
def gradient_refine(h, time_limit, rng):
    """Move mass from low‑benefit to high‑benefit sites while keeping sum fixed."""
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    eps = 1e-12

    while time.time() - start < time_limit - 0.1:
        # full correlation to locate the worst shift
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        overlaps = corr * dx
        max_idx = np.argmax(overlaps)
        k = max_idx - (n - 1)

        # avoid zero shift (rarely maximal when binary)
        if k == 0:
            k = rng.choice(np.delete(np.arange(-n + 1, n), n - 1))

        # benefit = sum of neighboring values that increase the product when we raise h[i]
        benefit = np.zeros(n)
        if k > 0:
            benefit[:n - k] += best_h[k:]
            benefit[k:] += best_h[:n - k]
        elif k < 0:
            kk = -k
            benefit[kk:] += best_h[:n - kk]
            benefit[:n - kk] += best_h[kk:]

        # pick a donor (low benefit, can decrease) and a receiver (high benefit, can increase)
        dec_mask = best_h > eps
        inc_mask = best_h < 1.0 - eps
        if not np.any(dec_mask) or not np.any(inc_mask):
            break

        dec_idxs = np.where(dec_mask)[0]
        inc_idxs = np.where(inc_mask)[0]
        dec_idx = dec_idxs[np.argmin(benefit[dec_idxs])]
        inc_idx = inc_idxs[np.argmax(benefit[inc_idxs])]

        max_transfer = min(best_h[dec_idx], 1.0 - best_h[inc_idx])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer * 0.5
        if delta < eps:
            continue

        h_new = best_h.copy()
        h_new[dec_idx] -= delta
        h_new[inc_idx] += delta

        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            best_h = h_new
            best_c5 = c5_new
            if best_c5 < 0.3808:
                break
    return best_h, best_c5

# ---------- Per‑Resolution Optimisation ----------
def optimise_for_n(n, time_budget, rng):
    """Run a few SLSQP restarts and finish with gradient refinement."""
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf
    start = time.time()

    # use the provided parent if it matches this size
    use_parent = False
    if 'initial_h_values' in globals():
        if len(initial_h_values) == n:
            use_parent = True
            parent_h = np.array(initial_h_values, dtype=float)
            cur = np.sum(parent_h)
            if abs(cur - M) > 1e-12:
                parent_h += (M - cur) / n
                parent_h = np.clip(parent_h, 0.0, 1.0)
            parent_c5 = compute_c5(parent_h)
            best_h, best_c5 = parent_h, parent_c5

    while True:
        elapsed = time.time() - start
        if elapsed > time_budget - 0.2:
            break

        if use_parent:
            h0 = parent_h.copy()
            use_parent = False
        else:
            h0 = generate_initial_h(n, M, rng)

        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])

        try:
            res = minimize(fun=objective_t,
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 600, 'ftol': 1e-10, 'disp': False})
        except Exception:
            continue

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur = np.sum(h_opt)
        if abs(cur - M) > 1e-9:
            h_opt += (M - cur) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)

        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break

    # local gradient refinement within any remaining time for this resolution
    remaining = time_budget - (time.time() - start)
    if remaining > 1.0 and best_h is not None:
        refine_time = min(remaining * 0.5, remaining - 0.5)
        h_ref, c5_ref = gradient_refine(best_h, refine_time, rng)
        if c5_ref + 1e-12 < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    return best_h, best_c5

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h_global = None
    best_c5_global = np.inf
    best_n_global = None

    candidate_ns = [80, 120, 160, 200]

    # prioritize parent size if available
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidate_ns:
                candidate_ns.insert(0, n0)
            else:
                candidate_ns = [n0] + [n for n in candidate_ns if n != n0]

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_time)
        alloc = min(250.0, remaining * 0.5)  # generous per‑resolution budget
        h_opt, c5_opt = optimise_for_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5_global:
            best_h_global, best_c5_global, best_n_global = h_opt, c5_opt, n
            if best_c5_global < 0.3808:
                break

    # ----- Optional up‑sampling refinement -----
    if best_h_global is not None:
        remaining = budget_s - (time.time() - start_time)
        if remaining > 5.0:
            # try double resolution if still within limits
            n2 = best_n_global * 2
            if n2 <= 1000:
                h2 = np.repeat(best_h_global, 2)
                # sum stays correct (M doubles)
                # refine at the finer grid
                refine_time = min(remaining * 0.5, remaining - 0.5)
                h2_ref, c5_2 = gradient_refine(h2, refine_time, rng)
                if c5_2 + 1e-12 < best_c5_global:
                    best_h_global, best_c5_global, best_n_global = h2_ref, c5_2, n2

    # ----- Final safety & constraint enforcement -----
    if best_h_global is None:
        n = 80
        best_h_global = np.full(n, 0.5)
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n
    else:
        n = best_n_global
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        cur = np.sum(best_h_global)
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h_global += (M - cur) / n
            best_h_global = np.clip(best_h_global, 0.0, 1.0)

    return best_h_global, best_c5_global, n
