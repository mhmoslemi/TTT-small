import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_start_vector(n, rng):
    target = n / 2.0
    v = rng.random(n)
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        inc = diff / n
        v = np.clip(v + inc, 0.0, 1.0)
        cur = v.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            idxs = np.arange(n)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and v[i] < 1.0:
                    delta = min(diff, 1.0 - v[i])
                    v[i] += delta
                    diff -= delta
                elif diff < 0 and v[i] > 0.0:
                    delta = min(-diff, v[i])
                    v[i] -= delta
                    diff += delta
    return v

def continuous_anneal(h_start, best_val, start_time, deadline, rng):
    h = h_start.copy()
    best_h = h.copy()
    best = best_val
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_val = np.max(corr) * dx
    while time.time() < deadline - 0.1:
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        old_corr = corr
        h[i] -= delta
        h[j] += delta
        corr = np.correlate(h, 1.0 - h, mode="full")
        new_val = np.max(corr) * dx
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
                # keep updated corr
            else:
                h[i] += delta
                h[j] -= delta
                corr = old_corr
        if best < 0.3808:
            break
    return best_h, best

def upsample_h(h_small, n_new, rng):
    m = h_small.shape[0]
    x_small = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_small, h_small)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n_new / 2.0
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new += diff / n_new
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            idxs = np.arange(n_new)
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h_new[i] < 1.0:
                    inc = min(diff, 1.0 - h_new[i])
                    h_new[i] += inc
                    diff -= inc
                elif diff < 0 and h_new[i] > 0.0:
                    dec = min(-diff, h_new[i])
                    h_new[i] -= dec
                    diff += dec
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # initialise base h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        target0 = n0 / 2.0
        cur = h0.sum()
        if abs(cur - target0) > 1e-12:
            h0 = np.clip(h0 + (target0 - cur) / n0, 0.0, 1.0)
            diff = target0 - h0.sum()
            if abs(diff) > 1e-12:
                idxs = np.arange(n0)
                rng.shuffle(idxs)
                for i in idxs:
                    if abs(diff) < 1e-12:
                        break
                    if diff > 0 and h0[i] < 1.0:
                        inc = min(diff, 1.0 - h0[i])
                        h0[i] += inc
                        diff -= inc
                    elif diff < 0 and h0[i] > 0.0:
                        dec = min(-diff, h0[i])
                        h0[i] -= dec
                        diff += dec
    else:
        n0 = 80
        h0 = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, n0 // 2, replace=False)
        h0[idx] = 1.0

    h0 = np.clip(h0, 0.0, 1.0)
    target0 = n0 / 2.0
    diff = target0 - h0.sum()
    if abs(diff) > 1e-12:
        h0 += diff / n0
        h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # SLSQP local optimisation phase
    while time.time() < deadline - 30.0:
        h_start = random_start_vector(n0, rng)
        t_start = compute_c5(h_start)
        x0 = np.concatenate([h_start, [t_start]])
        dx0 = 2.0 / n0
        target_sum0 = n0 / 2.0
        sum_con = lambda x, target=target_sum0: np.sum(x[:n0]) - target
        constraints = [{"type": "eq", "fun": sum_con}]
        def make_overlap(k):
            def con(x, k=k):
                h = x[:n0]
                if k >= 0:
                    overl = np.dot(h[: n0 - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[: n0 - kk])
                return x[-1] - overl * dx0
            return con
        for kk in range(-n0 + 1, n0):
            if kk == 0:
                continue
            constraints.append({"type": "ineq", "fun": make_overlap(kk)})
        bounds = [(0.0, 1.0)] * n0 + [(0.0, None)]
        try:
            res = opt.minimize(fun=lambda x: x[-1],
                               x0=x0,
                               method="SLSQP",
                               bounds=bounds,
                               constraints=constraints,
                               options={"maxiter": 2000, "ftol": 1e-9, "disp": False})
        except Exception:
            break
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n0], 0.0, 1.0)
        cur_sum = h_opt.sum()
        if abs(cur_sum - target_sum0) > 1e-9:
            h_opt += (target_sum0 - cur_sum) / n0
            h_opt = np.clip(h_opt, 0.0, 1.0)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break
        # short annealing refinement on the current best
        if time.time() < deadline - 10.0:
            sub_deadline = time.time() + 5.0
            best_h, best_val = continuous_anneal(best_h, best_val, start, sub_deadline, rng)

    # long annealing on the best candidate
    if time.time() < deadline - 5.0:
        best_h, best_val = continuous_anneal(best_h, best_val, start, deadline, rng)

    # upsample and final annealing if time permits
    n_high = 160
    if time.time() < deadline - 30.0:
        h_high = upsample_h(best_h, n_high, rng)
        if time.time() < deadline - 5.0:
            best_h_high, best_val_high = continuous_anneal(h_high, compute_c5(h_high), start, deadline, rng)
            if best_val_high < best_val:
                best_h = best_h_high
                best_val = best_val_high
                n = n_high
            else:
                n = best_h.shape[0]
        else:
            n = best_h.shape[0]
    else:
        n = best_h.shape[0]

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h += (target - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
