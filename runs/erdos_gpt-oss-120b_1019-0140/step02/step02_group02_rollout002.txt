def improve_by_gradient(h, best_val, start_time, deadline, rng):
    while time.time() < deadline - 0.1:
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute gradient array
        grad = np.ones(n)  # base 1
        if k >= 0:
            grad[:n - k] -= h[k:]  # subtract h[i+k]
            grad[k:] -= h[:n - k]   # subtract h[i-k], but careful to align with index
            # Actually we need to subtract h_{i-k} where i >= k.
            # For i >= k, index i-k is valid.
            # Equivalent: grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[kk:] -= h[:n - kk]
            grad[:n - kk] -= h[kk:]
        # Now grad_i = 1 - (neighbor contributions)
        # Determine pos (grad>0) and neg (grad<0)
        pos = np.where(grad > 1e-8)[0]
        neg = np.where(grad < -1e-8)[0]
        if pos.size == 0 or neg.size == 0:
            # Could be at optimum? break.
            break
        # Choose i from pos (where decreasing h reduces C) and j from neg (increase h reduces C)
        i = rng.choice(pos)
        j = rng.choice(neg)
        # Determine max possible delta
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 1e-12:
            continue
        # Choose delta: maybe a small fraction of max_delta or relative to step size.
        delta = max_delta * rng.random() * 0.5  # move up to 50% of max
        # Apply transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)

    return h, best_val

def random_hill_climb(h, best_val, start_time, deadline, rng):
    while time.time() < deadline - 0.1:
        # pick two indices i (source) and j (dest) randomly such that h[i] > 0, h[j] < 1

def enforce_sum_and_bounds(h, target):
    # Clip to [0,1]
    h = np.clip(h, 0., 1.)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute diff equally to all entries, but keep within bounds
        # We'll iterate adjusting gradually
        # Approach: add diff/n to each, clip, then recompute diff
        h = np.clip(h + diff / len(h), 0., 1.)
        cur = h.sum()
        diff = target - cur
        # If still diff, adjust random indices
        if abs(diff) > 1e-12:
            idxs = np.arange(len(h))
            rng = np.random.default_rng()
            rng.shuffle(idxs)
            for i in idxs:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    inc = min(diff, 1 - h[i])
                    h[i] += inc
                    diff -= inc
                else:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
    return h

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1−h(x+k))dx via discrete correlation."""

    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
