def subgradient_max_overlap(h):
    n = len(h)
    v = np.correlate(h, 1.0 - h, mode='full')
    max_val = v.max()
    # indices where v equals max (within tolerance)
    idxs = np.where(np.abs(v - max_val) <= 1e-12)[0]
    g = np.zeros(n)
    offset = n - 1
    for idx in idxs:
        k = idx - offset
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
            g[tail] += 1.0 - h[head]
            g[head] -= h[tail]
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
            g[tail] += 1.0 - h[head]
            g[head] -= h[tail]
    # average if multiple max
    if len(idxs) > 1:
        g /= len(idxs)
    # subtract mean to reduce bias? Not needed; projection will handle.
    return g
```

def optimize_h_subgradient(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    no_improve = 0
    # initial step size
    alpha0 = 0.5
    while time.time() - start < time_budget:
        # compute gradient
        g = subgradient_max_overlap(h)
        # step size schedule linear decay
        elapsed = time.time() - start
        alpha = alpha0 * max(0.01, 1.0 - elapsed / time_budget)
        # noisy gradient: add small random jitter to escape plateau
        if no_improve > 500:
            # random perturb
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0
            continue
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        # decide whether to accept new h
        if c5_new < compute_c5(h) or rng.random() < 0.001:
            h = h_new
        # occasional random re-shake
        if rng.random() < 0.001:
            h = project_sum_and_bounds(h + rng.normal(scale=0.01, size=n), target)
    return best_h, best_c5

def project_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    # adjust uniformly
    x += diff / len(x)
    x = np.clip(x, 0.0, 1.0)
    # iterate if still not met (rare)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # adjust greedily
        for i in range(len(x)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
    return x
```

    n = len(h)
    dx = 2.0 / n

    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)

    if abs(diff) > 1e-12:
        x += diff / len(x)
