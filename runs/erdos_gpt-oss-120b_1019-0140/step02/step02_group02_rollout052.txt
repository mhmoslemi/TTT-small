def subgradient_optimize(h_start, time_limit, start_time):
   h = h_start.copy()
   best_h = h.copy()
   best_val = compute_c5(h)
   iter = 0
   gamma = 0.5  # initial step scaling
   while time.time() < time_limit - 0.05:
       overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
       max_val = overlaps.max()
       # early exit: if max_val < best_val: update best
       if max_val < best_val:
           best_val = max_val
           best_h = h.copy()
           if best_val < 0.3808:
               break
       # find active shifts: within tolerance relative to max, maybe > max_val - 1e-6
       eps = 1e-9  # strict tolerance
       active_idxs = np.where(np.abs(overlaps - max_val) <= eps)[0]
       # convert to shift k = idx - (n-1)
       active_shifts = active_idxs - (len(h) - 1)
       # compute gradient (average)
       grad = np.zeros_like(h)
       for k in active_shifts:
           grad += gradient_of_shift(h, k, dx)
       grad /= len(active_shifts)
       # step size
       iter += 1
       step = gamma / np.sqrt(iter)
       # subgradient step
       h_new = h - step * grad
       # project onto feasible
       h = project_onto_box_sum(h_new, 0.0, 1.0, target_sum)
   return best_h, best_val

def random_feasible_vector(n, rng):
    # start with uniform random in [0,1]
    x = rng.random(n)
    # we need sum = target = n/2
    target = n/2
    # compute current sum
    cur = x.sum()
    # scale uniformly: x = x * (target / cur)
    if cur > 0:
        x = x * (target / cur)
    # clip
    x = np.clip(x, 0, 1)
    # adjust leftover
    cur = x.sum()
    diff = target - cur
    # distribute diff across eligible indices
    # while diff not small:
    while abs(diff) > 1e-12:
        if diff > 0:
            # indices where x < 1
            idx = np.where(x < 1 - 1e-12)[0]
        else:
            idx = np.where(x > 1e-12)[0]
        if len(idx) == 0:
            break
        # add a share
        share = diff / len(idx)
        # to avoid overshoot, clip per update
        for i in idx:
            if diff > 0:
                inc = min(share, 1 - x[i])
                x[i] += inc
                diff -= inc
            else:
                dec = min(-share, x[i])
                x[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
        if len(idx) == 0:
            break
    return x

        max_idx = np.argmax(overlaps)

def compute_c5(h):
    """Return the C5 value for a given h (discrete approximation)."""
    n = h.shape[0]

    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    """Gradient of the overlap for a specific shift k (including dx factor)."""
    n = h.shape[0]

    if k >= 0:
        g[: n - k] += (1.0 - h[k:]) * dx
        g[k:]     -= h[: n - k] * dx
    else:

        g[kk:]    += (1.0 - h[: n - kk]) * dx
        g[: n - kk] -= h[kk:] * dx

    return v

def subgradient_optimize(h_start, deadline, start_time):
    """Subgradient descent on max(overlap) with projection."""
    h = h_start.copy()
    n = h.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(best_h)
    it = 0
    gamma = 0.8
    while time.time() < deadline - 0.05:
        it += 1
        overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
        max_idx = int(np.argmax(overlaps))
        cur_val = overlaps[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val < 0.3808:
                break
        k = max_idx - (n - 1)
        grad = gradient_of_shift(h, k, dx)
        step = gamma / np.sqrt(it)
        h = project_onto_box_sum(h - step * grad, 0.0, 1.0, target)
    return best_h, best_val

def local_refine(h, cur_val, deadline, start_time, rng):
    """Simple hill‑climbing/annealing refinement."""
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_val = cur_val
    h_cur = h.copy()
    while time.time() < deadline - 0.05:
        src = rng.choice(np.where(h_cur > 1e-8)[0])
        dst = rng.choice(np.where(h_cur < 1.0 - 1e-8)[0])
        max_delta = min(h_cur[src], 1.0 - h_cur[dst])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = h_cur.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_val = compute_c5(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
            h_cur = h_new
            if best_val < 0.3808:
                break
        else:
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = max(remaining * 0.001, 1e-12)
            if np.exp((best_val - new_val) / T) > rng.random():
                h_cur = h_new
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # ---- initialise h ----
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        target = n / 2.0
        h0 = project_onto_box_sum(h0, 0.0, 1.0, target)
    else:
        n = 120                     # default resolution when no parent solution
        target = n / 2.0
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ---- main optimisation loop ----
    while time.time() < deadline - 5.0:
        # pick a start point
        if rng.random() < 0.5:
            start_vec = best_h.copy()
        else:
            start_vec = random_feasible_vector(n, rng)

        slice_deadline = min(deadline, time.time() + 25.0)

        cand_h, cand_val = subgradient_optimize(start_vec, slice_deadline, start)

        # optional local refinement with remaining time
        if time.time() < deadline - 1.0:
            cand_h, cand_val = local_refine(cand_h, cand_val, deadline, start, rng)

        if cand_val < best_val:
            best_val = cand_val
            best_h = cand_h.copy()
            if best_val < 0.3808:
                break

    # final safety projection and evaluation
    best_h = project_onto_box_sum(best_h, 0.0, 1.0, n / 2.0)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
