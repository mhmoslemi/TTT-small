def shift_array(arr, shift):
    """Shift arr by shift positions, inserting zeros, no wrap. Positive shift moves values to lower indices: out[i] = arr[i+shift] if i+shift < n else 0."""
    n = len(arr)
    out = np.zeros_like(arr)
    if shift >= 0:
        out[: n - shift] = arr[shift:]
    else:
        out[-shift : ] = arr[: n + shift]
    return out

import time
import numpy as np

def compute_c5_vec(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec

def generate_random_feasible(n, target_sum, rng):
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x

def shift_array(arr, shift):
    n = len(arr)
    out = np.zeros_like(arr)
    if shift >= 0:
        if shift < n:
            out[: n - shift] = arr[shift:]
    else:
        s = -shift
        if s < n:
            out[s:] = arr[: n - s]
    return out

def refine_continuous(h, dx, target_sum, time_limit, rng):
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.05
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        rec = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or rec.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(rec)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_sum(cand, target_sum)
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val

def subgradient_optimize(h0, dx, target_sum, time_limit, rng):
    h = h0.copy()
    cur_val = compute_c5_vec(h, dx)
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    it = 0
    step0 = 0.5
    while time.time() < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)
        grad = np.ones(n) - shift_array(h, k) - shift_array(h, -k)
        step = step0 / np.sqrt(it + 1)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        val = compute_c5_vec(h_new, dx)
        if val < cur_val - 1e-12:
            h = h_new
            cur_val = val
            if cur_val < best_val - 1e-12:
                best_h = h.copy()
                best_val = cur_val
            # allow larger steps after improvement
        it += 1
        if best_val < 0.3808:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        target_sum = n / 2.0
        dx = 2.0 / n
        best_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
        best_val = compute_c5_vec(best_h, dx)
    else:
        n = 200
        target_sum = n / 2.0
        dx = 2.0 / n
        best_h = generate_random_feasible(n, target_sum, rng)
        best_val = compute_c5_vec(best_h, dx)

    end_time = start_time + budget_s
    while time.time() < end_time - 5.0:
        remaining = end_time - time.time()
        if remaining < 8.0:
            break
        # random start, occasionally start from best
        if rng.random() < 0.2:
            h_start = best_h.copy()
        else:
            h_start = generate_random_feasible(n, target_sum, rng)

        sub_limit = time.time() + remaining * 0.5
        h_sub, val_sub = subgradient_optimize(h_start, dx, target_sum, sub_limit, rng)

        # refine with remaining time
        refine_limit = end_time - 0.5  # leave tiny margin
        h_ref, val_ref = refine_continuous(h_sub, dx, target_sum, refine_limit, rng)

        if val_ref < best_val - 1e-12:
            best_h, best_val = h_ref, val_ref
            if best_val < 0.3808:
                break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
