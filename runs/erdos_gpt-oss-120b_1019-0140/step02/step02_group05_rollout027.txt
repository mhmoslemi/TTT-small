import time
import numpy as np

def compute_c5(h):
    """Exact C5 for a discretized h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_and_fix(h, target_sum):
    """
    One Gaussian-type smoothing pass with exact sum correction.
    Keeps values in [0,1] and enforces Σh = target_sum.
    """
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)

    cur = np.sum(sm)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            inc = diff / free.size
            sm[free] += inc
            sm = np.clip(sm, 0.0, 1.0)

        diff = target_sum - np.sum(sm)
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
        sm = np.clip(sm, 0.0, 1.0)
    return sm

def project_to_sum_box(v, target):
    """
    Project vector v onto the intersection of the box [0,1]^n
    and the affine hyperplane Σx = target.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_smooth_gradient(h, alpha):
    """
    Gradient of the smooth max (log‑sum‑exp) of the correlation array.
    Returns ∇_h ( 1/α log Σ exp(α·corr) ).
    """
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(corr)
    exps = np.exp(alpha * (corr - maxc))
    w = exps / np.sum(exps)          # weights sum to 1

    grad = np.zeros_like(h)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        k = idx - (n - 1)
        neigh = np.zeros(n, dtype=float)
        if k >= 0:
            neigh[: n - k] += h[k:]
            neigh[k:] += h[: n - k]
        else:
            kk = -k
            neigh[kk:] += h[: n - kk]
            neigh[: n - kk] += h[kk:]
        grad += weight * dx * (1.0 - neigh)
    return grad
