import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_constraints(h, target):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum to target
    cur = np.sum(h)
    diff = target - cur
    # First uniform adjustment
    if np.abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = target - cur
        # Distribute remaining diff over free variables
        for _ in range(5):
            if np.abs(diff) < 1e-12:
                break
            free = np.where((h > 0) & (h < 1))[0]
            if free.size:
                inc = diff / free.size
                h[free] += inc
                h = np.clip(h, 0.0, 1.0)
                cur = np.sum(h)
                diff = target - cur
            else:
                break
        # final residual fix sequentially
        if np.abs(diff) >= 1e-12:
            for i in range(len(h)):
                if np.abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
    return h

def smooth_and_fix(h, target, kernel=np.array([0.25,0.5,0.25])):
    sm = np.convolve(h, kernel, mode='same')
    sm = np.clip(sm, 0.0, 1.0)
    sm = project_constraints(sm, target)
    return sm

def gradient_move(h, target):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = dx * (1.0 - neigh)
    grad_centered = grad - np.mean(grad)
    step = -grad_centered
    max_step = 0.2
    mabs = np.max(np.abs(step))
    if mabs > max_step:
        step = step * (max_step / mabs)
    cand = h + step
    cand = np.clip(cand, 0.0, 1.0)
    cand = project_constraints(cand, target)
    return cand

def anneal(h0, target, time_budget, rng):
    n = len(h0)
    cur_h = h0.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    # parameters
    T0 = 0.02
    T_end = 1e-6
    move_probs = [0.4, 0.4, 0.2]  # swap, transfer, gradient
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # exponential temperature
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac

        r = rng.random()
        if r < move_probs[0]:
            # swap
            i, j = rng.choice(n, size=2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < move_probs[0] + move_probs[1]:
            # transfer
            i, j = rng.choice(n, size=2, replace=False)
            # transfer from i to j
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr <= 0:
                max_tr = min(cur_h[j], 1.0 - cur_h[i])
                if max_tr <= 0:
                    continue
                donor, recv = j, i
            else:
                donor, recv = i, j
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        else:
            # gradient move
            cand = gradient_move(cur_h, target)

        new_val = compute_c5(cand)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
                # early exit if we beat target 
                if best_val < 0.3808:
                    break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    total_budget = budget_s
    rng_global = np.random.default_rng(seed)
    # Choose grid size n
    n = 120  # can be tuned
    target_sum = n / 2.0

    # Prepare initial solutions list
    init_hs = []
    if 'initial_h_values' in globals():
        # upsample or interpolate to length n
        init = np.array(initial_h_values, dtype=float)
        m = len(init)
        if m != n:
            # linear interpolation onto new grid
            x_old = np.linspace(0, 2, m, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        init = project_constraints(init, target_sum)
        init_hs.append(init)

    # Add a few random starts
    num_random_starts = 4
    for _ in range(num_random_starts):
        h_rand = rng_global.random(n)
        h_rand = project_constraints(h_rand, target_sum)
        init_hs.append(h_rand)

    best_h_overall = None
    best_val_overall = np.inf

    # allocate time per start equally, leaving small buffer for final refinement
    remaining = total_budget - (time.time() - start)
    # ensure at least some time for final refinement
    final_refine_time = 30.0
    per_start_budget = max(5.0, (remaining - final_refine_time) / len(init_hs))

    for idx, h0 in enumerate(init_hs):
        # allocate per-start RNG
        rng = np.random.default_rng(seed + idx + 1)
        best_h, best_val = anneal(h0, target_sum, per_start_budget, rng)
        if best_val < best_val_overall:
            best_val_overall = best_val
            best_h_overall = best_h.copy()
            if best_val_overall < 0.3808:
                break

    # optional smoothing passes
    time_left = total_budget - (time.time() - start)
    if time_left > 5.0 and best_h_overall is not None:
        # attempt a few smoothing iterations
        for _ in range(5):
            sm = smooth_and_fix(best_h_overall, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val_overall:
                best_val_overall = sm_val
                best_h_overall = sm
            else:
                break

    # final local refinement: a few gradient moves with decreasing step
    time_left = total_budget - (time.time() - start)
    if time_left > 5.0 and best_h_overall is not None:
        cur_h = best_h_overall.copy()
        cur_val = best_val_overall
        # perform deterministic gradient descent until no improvement or time out
        while time.time() - start < total_budget - 5.0:
            cand = gradient_move(cur_h, target_sum)
            new_val = compute_c5(cand)
            if new_val < cur_val - 1e-7:
                cur_h, cur_val = cand, new_val
                if new_val < best_val_overall:
                    best_val_overall = new_val
                    best_h_overall = cur_h.copy()
                if best_val_overall < 0.3808:
                    break
            else:
                break

    # final safety projection
    if best_h_overall is not None:
        best_h_overall = np.clip(best_h_overall, 0.0, 1.0)
        cur = np.sum(best_h_overall)
        if abs(cur - target_sum) > 1e-9:
            best_h_overall += (target_sum - cur) / n
            best_h_overall = np.clip(best_h_overall, 0.0, 1.0)

    return best_h_overall, float(best_val_overall), n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral C5 for a discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_constraints(h, target):
    """Clip to [0,1] and enforce exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # uniform adjustment
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target - cur
        # distribute remaining diff over free coordinates
        for _ in range(5):
            if abs(diff) < 1e-12:
                break
            free = np.where((h > 0) & (h < 1))[0]
            if free.size == 0:
                break
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
            cur = h.sum()
            diff = target - cur
        # final sequential fix
        if abs(diff) >= 1e-12:
            for i in range(h.size):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
    return h


def smooth_and_fix(h, target, kernel=np.array([0.25, 0.5, 0.25])):
    """Gaussian‑type smoothing with re‑projection."""
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    return project_constraints(sm, target)


def gradient_move(h, target):
    """Move in the opposite direction of the gradient of the worst shift."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)          # shift that attains the maximum
    neigh = np.zeros(n)
    if k >= 0:
        neigh[: n - k] += h[k:]
        neigh[k:] += h[: n - k]
    else:
        kk = -k
        neigh[kk:] += h[: n - kk]
        neigh[: n - kk] += h[kk:]
    grad = dx * (1.0 - neigh)      # ∂C5/∂h_i for the worst shift
    grad_centered = grad - np.mean(grad)   # keep zero‑sum direction
    step = -grad_centered
    max_step = 0.2
    m = np.max(np.abs(step))
    if m > max_step:
        step = step * (max_step / m)
    cand = h + step
    cand = np.clip(cand, 0.0, 1.0)
    return project_constraints(cand, target)


def anneal(h0, target, budget, rng):
    """Simulated annealing with three move types."""
    n = h0.size
    cur_h = h0.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    T0 = 0.02
    T_end = 1e-6
    probs = [0.4, 0.4, 0.2]                 # swap, transfer, gradient

    while True:
        elapsed = time.time() - start
        if elapsed >= budget:
            break
        # exponential cooling
        frac = elapsed / budget
        T = T0 * (T_end / T0) ** frac

        r = rng.random()
        if r < probs[0]:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < probs[0] + probs[1]:
            # transfer a chunk from one coordinate to another
            i, j = rng.choice(n, size=2, replace=False)
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr <= 0:
                max_tr = min(cur_h[j], 1.0 - cur_h[i])
                if max_tr <= 0:
                    continue
                donor, recv = j, i
            else:
                donor, recv = i, j
            delta = rng.random() * max_tr
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[recv] += delta
        else:
            # gradient move
            cand = gradient_move(cur_h, target)

        new_val = compute_c5(cand)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_h, cur_val = cand, new_val
            if new_val < best_val:
                best_val, best_h = new_val, cand.copy()
                if best_val < 0.3808:
                    break
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    total_budget = budget_s
    rng_global = np.random.default_rng(seed)

    # grid size – a balance between flexibility and speed
    n = 120
    target_sum = n / 2.0

    # collect several diversified starting points
    starts = []

    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            # linear interpolation onto the new grid
            x_old = np.linspace(0, 2, init.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        init = project_constraints(init, target_sum)
        starts.append(init)

    # a few random starts
    for _ in range(4):
        rand = rng_global.random(n)
        rand = project_constraints(rand, target_sum)
        starts.append(rand)

    best_h = None
    best_val = np.inf

    # allocate time for the annealing phases, leaving a small reserve
    time_spent = time.time() - start
    remaining = total_budget - time_spent - 30.0   # reserve for final polishing
    per_start = max(5.0, remaining / len(starts))

    for idx, h0 in enumerate(starts):
        rng = np.random.default_rng(seed + idx + 1)
        h_opt, val_opt = anneal(h0, target_sum, per_start, rng)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()
            if best_val < 0.3808:
                break

    # optional smoothing passes
    time_left = total_budget - (time.time() - start)
    if time_left > 5.0 and best_h is not None:
        for _ in range(5):
            sm = smooth_and_fix(best_h, target_sum)
            sm_val = compute_c5(sm)
            if sm_val + 1e-12 < best_val:
                best_val, best_h = sm_val, sm
            else:
                break

    # final deterministic gradient descent (few steps)
    time_left = total_budget - (time.time() - start)
    if time_left > 5.0 and best_h is not None:
        cur_h, cur_val = best_h.copy(), best_val
        while time.time() - start < total_budget - 5.0:
            cand = gradient_move(cur_h, target_sum)
            new_val = compute_c5(cand)
            if new_val + 1e-9 < cur_val:
                cur_h, cur_val = cand, new_val
                if new_val < best_val:
                    best_val, best_h = new_val, cand.copy()
                if best_val < 0.3808:
                    break
            else:
                break

    # final safety projection
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = best_h.sum()
        if abs(cur - target_sum) > 1e-9:
            best_h += (target_sum - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, float(best_val), n
