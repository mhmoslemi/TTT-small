import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k_shift):
    """Gradient of overlap_k with respect to h (scaled by dx omitted)."""
    n = h.shape[0]
    g = np.ones_like(h)
    if k_shift >= 0:
        # subtract h[i+k] term
        g[:n - k_shift] -= h[k_shift:]
        # subtract h[i-k] term
        g[k_shift:] -= h[:n - k_shift]
    else:
        kk = -k_shift
        g[kk:] -= h[:n - kk]
        g[:n - kk] -= h[kk:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Initialise problem size and feasible start vector
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_box_sum(h0, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    if best_c5 < 0.3808:
        return best_h, best_c5, n

    # ---------- Phase 1: Gradient descent on the active shift ----------
    phase1_deadline = start_time + budget_s * 0.4
    cur_h = best_h.copy()
    cur_c5 = best_c5
    step_size = 0.5
    min_step = 1e-6

    while time.time() < phase1_deadline:
        # Determine shift attaining the current maximum overlap
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        dx = 2.0 / n
        corr_scaled = corr * dx
        max_idx = int(np.argmax(corr_scaled))
        k_shift = max_idx - (n - 1)

        # Gradient for that shift
        g = gradient_for_shift(cur_h, k_shift)

        # Backtracking line search
        improved = False
        local_step = step_size
        while local_step > min_step:
            cand = cur_h - local_step * g
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_to_box_sum(cand, target_sum)
            cand_c5 = compute_c5(cand)
            if cand_c5 < cur_c5 - 1e-12:
                cur_h, cur_c5 = cand, cand_c5
                improved = True
                local_step *= 1.1
                break
            local_step *= 0.5

        if not improved:
            # Reduce global step size and occasionally perturb
            step_size *= 0.7
            if rng.random() < 0.05:
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(cur_h[i], 1.0 - cur_h[j])
                if max_move > 0:
                    delta = max_move * rng.random() * 0.2
                    cur_h[i] -= delta
                    cur_h[j] += delta
                    cur_h = project_to_box_sum(cur_h, target_sum)
                    cur_c5 = compute_c5(cur_h)
            if step_size < min_step:
                break

        # Update global best
        if cur_c5 < best_c5 - 1e-12:
            best_h, best_c5 = cur_h.copy(), cur_c5
            if best_c5 < 0.3808:
                return best_h, best_c5, n

    # ---------- Phase 2: Random coordinate exchanges ----------
    phase2_deadline = start_time + budget_s * 0.8
    while time.time() < phase2_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(best_h[i], 1.0 - best_h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random() * 0.3
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5 - 1e-12:
            best_h, best_c5 = cand, cand_c5
            if best_c5 < 0.3808:
                return best_h, best_c5, n

    # ---------- Phase 3: SLSQP refinement ----------
    phase3_deadline = start_time + budget_s - 1.0
    if time.time() < phase3_deadline:
        def objective(x):
            return compute_c5(x)

        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
            )
            if res.success:
                h_opt = project_to_box_sum(np.clip(res.x, 0.0, 1.0), target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5 - 1e-12:
                    best_h, best_c5 = h_opt, c5_opt
        except Exception:
            pass

    # Final safeguard projection
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
