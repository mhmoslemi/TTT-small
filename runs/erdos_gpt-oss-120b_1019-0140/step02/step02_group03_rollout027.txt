    def project_to_box_sum(v, target, lo=0.0, hi=1.0):

                low = mid
            else:
                high = mid

import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------- Determine problem size and initial vector -------
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        n = h_init.size
    else:
        n = 80
        h_init = rng.random(n)

    # ------- Helper: projection onto box with fixed sum -------
    def project_to_box_sum(v, target, lo=0.0, hi=1.0):
        """Project v onto [lo,hi]^n while fixing sum to target."""
        low = -np.max(v) - 5.0
        high = (hi - np.min(v)) + 5.0
        for _ in range(60):
            mid = (low + high) * 0.5
            x = np.clip(v + mid, lo, hi)
            s = x.sum()
            if s < target:
                low = mid
            else:
                high = mid
        return np.clip(v + (low + high) * 0.5, lo, hi)

    # ------- Compute C5 for a candidate -------
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / h.shape[0])

    target_sum = n / 2.0
    use_binary = abs(target_sum - round(target_sum)) < 1e-9

    # ------- Initialise feasible vector -------
    if use_binary:
        half = int(round(target_sum))
        idx_desc = np.argsort(-h_init)
        h = np.zeros(n, dtype=float)
        h[idx_desc[:half]] = 1.0
    else:
        h = project_to_box_sum(h_init, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ------- Binary simulated annealing (if applicable) -------
    if use_binary:
        # Pre‑compute index lists
        ones = np.flatnonzero(best_h > 0.5)
        zeros = np.flatnonzero(best_h < 0.5)

        # Annealing schedule parameters
        start_T = 0.02
        end_T = 1e-5
        anneal_time = budget_s * 0.90
        it = 0
        cur_h = best_h.copy()
        cur_c5 = best_c5
        cur_ones = ones.copy()
        cur_zeros = zeros.copy()
        restart_interval = 20000

        while time.time() - start_time < anneal_time:
            elapsed = time.time() - start_time
            frac = 1.0 - (elapsed / anneal_time)
            T = end_T + (start_T - end_T) * max(frac, 0.0)
            it += 1

            # Periodic full random restart to avoid stagnation
            if it % restart_interval == 0:
                r = rng.random(n)
                cur_h = np.zeros(n, dtype=float)
                cur_h[np.argsort(-r)[:half]] = 1.0
                cur_c5 = compute_c5(cur_h)
                cur_ones = np.flatnonzero(cur_h > 0.5)
                cur_zeros = np.flatnonzero(cur_h < 0.5)
                continue

            # Propose a swap of a 1 and a 0
            i_one = rng.choice(cur_ones)
            i_zero = rng.choice(cur_zeros)
            cand = cur_h.copy()
            cand[i_one] = 0.0
            cand[i_zero] = 1.0
            cand_c5 = compute_c5(cand)
            delta = cand_c5 - cur_c5

            if delta < 0.0 or rng.random() < np.exp(-delta / T):
                cur_h = cand
                cur_c5 = cand_c5
                # update index sets (recalc – cheap for n≈80)
                cur_ones = np.flatnonzero(cur_h > 0.5)
                cur_zeros = np.flatnonzero(cur_h < 0.5)

                if cur_c5 + 1e-12 < best_c5:
                    best_h = cur_h.copy()
                    best_c5 = cur_c5

    # ------- Local continuous refinement with SLSQP (time‑permitting) -------
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5.0:
        # Simple equality constraint for the total sum
        eq_sum = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}
        bounds = [(0.0, 1.0)] * n
        maxiter = 400

        # Give a few restarts
        for _ in range(5):
            if time.time() - start_time > budget_s - 2.0:
                break
            start_vec = best_h.copy()
            start_vec += rng.normal(scale=0.01, size=n)
            start_vec = project_to_box_sum(start_vec, target_sum)
            res = minimize(compute_c5, start_vec,

            if not res.success:

            if cand_c5 + 1e-12 < best_c5:

    # ------- Final safety projection and evaluation -------

    best_c5 = compute_c5(best_h)
