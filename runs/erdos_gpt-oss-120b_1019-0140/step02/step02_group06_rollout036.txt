import time
import numpy as np
import math

def compute_c5(h):
    """Return max overlap integral C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip x to [0,1] and adjust so that sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute the remaining mass uniformly, then clip again
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        # if still off, fix elementwise
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def gradient_descent(h0, time_budget, rng, target):
    """Perform soft‑max gradient descent on C5 for the given time budget."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    beta = 2.0          # start smoothing
    max_beta = 80.0
    alpha0 = 0.5        # base step size
    no_improve = 0
    momentum = np.zeros_like(h)  # optional momentum (kept zero)

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        # correlation vector (no dx)
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx

        # record best
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        # soft‑max weights (stable)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth proxy
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # enforce zero‑sum direction (preserve total mass)
        g -= np.mean(g)

        # step‑size schedule (linear decay)
        prog = elapsed / time_budget
        alpha = alpha0 * max(0.1, 1.0 - prog)

        # optional momentum (here simple exponential smoothing)
        momentum = 0.9 * momentum - alpha * g
        h_new = h + momentum
        h_new = project_sum_and_bounds(h_new, target)

        # acceptance (allow occasional uphill moves)
        c5_new = compute_c5(h_new)
        temp = 0.01 * (1.0 - prog) + 0.0001  # linear cooling
        accept = c5_new < c5 or rng.random() < math.exp(-(c5_new - c5) / (temp + 1e-12))
        if accept:
            h = h_new
            c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improve = 0

        # random shake if stuck
        if no_improve > 5000:
            h = project_sum_and_bounds(best_h + rng.normal(scale=0.02, size=n), target)
            momentum.fill(0.0)
            no_improve = 0

        # early exit if we already beat target bound
        if best_c5 <= 0.3808:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modestly larger discretization for more flexibility
    n = 200
    target = n / 2.0
    dx = 2.0 / n

    # Initialise h
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float).copy()
        old_n = init.shape[0]
        x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        h0 = rng.random(n)

    h0 = project_sum_and_bounds(h0, target)

    # Allocate most of the budget to gradient descent, keep a tiny reserve for final check
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed - 2.0)

    best_h, best_c5 = gradient_descent(h0, remaining, rng, target)

    # Final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
