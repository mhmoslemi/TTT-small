import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral multiplied by dx."""
    n = h.shape[0]
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode="full")
    return np.max(v) * dx

def project_sum_and_bounds(x, target):
    """
    Clip to [0,1] and adjust to satisfy sum(x) == target.
    Simple repair: first shift the mean, then sweep adjustments.
    """
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute uniformly, then re‑clip
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # greedy sweep
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def optimize_h(h0, time_budget, seed):

    Gradient descent on a smooth approximation of the max‑overlap.
    Uses soft‑max weighting, momentum, and occasional random shakes.
    """

    target = n / 2.0
    dx = 2.0 / n

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start = time.time()
    beta = 2.0                 # smoothing parameter
    max_beta = 80.0
    alpha0 = 0.3               # initial step size
    momentum_coeff = 0.9
    momentum = np.zeros_like(h)

    no_improve = 0
    offset = n - 1

    while time.time() - start < time_budget:
        # correlation vector (raw, without dx)
        v = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(v) * dx

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # graceful exit if we have already crossed a strong target
        if best_c5 < 0.38:
            break

        # increase smoothing very slowly
        if beta < max_beta:
            beta = min(max_beta, beta * 1.0015)

        # soft‑max weights (stable)
        vmax = np.max(v)
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth objective
        g = np.zeros(n)
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # zero‑sum direction (preserve total mass)
        g -= np.mean(g)

        # momentum update
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * g

        # step‑size schedule (linear decay)
        elapsed = time.time() - start
        alpha = alpha0 * max(0.05, 1.0 - elapsed / time_budget)

        h_new = h - alpha * momentum
        h_new = project_sum_and_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        # accept if improvement or tiny random acceptance
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        # random shake when stuck
        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    return best_h, best_c5

def local_refine(h, time_budget, rng):
    """
    Simple pairwise mass‑swap hill‑climbing to polish a solution.
    Swaps a small ε from one coordinate to another while keeping sum fixed.
    """
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    start = time.time()
    while time.time() - start < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # max ε that respects bounds
        max_eps = min(1.0 - best[i], best[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random() * 0.5  # move at most half of the feasible amount
        if eps < 1e-8:
            continue
        cand = best.copy()
        cand[i] += eps
        cand[j] -= eps
        c5 = compute_c5(cand)
        if c5 < best_c5:
            best = cand
            best_c5 = c5
    # final projection to be safe
    best = project_sum_and_bounds(best, target)
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
    h_values is a length‑n NumPy array satisfying the constraints,
    c5_bound is the achieved maximum overlap, and n_points is the
    number of samples used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modestly fine discretisation (still < 1k)
    n = 200
    target_sum = n / 2.0

    # Build a warm‑start from the parent solution if available
    base_h = None
    if 'initial_h_values' in globals():
        init_parent = np.array(initial_h_values, dtype=float)
        if init_parent.size != n:
            # linear up‑sampling
            old_n = init_parent.size
            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            base_h = np.interp(new_x, old_x, init_parent)
        else:
            base_h = init_parent.copy()

    best_h = None
    best_c5 = np.inf

    # Main optimisation loop – allocate most of the remaining budget per run
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # Allocate ~80 % of the remaining budget to this run
        alloc = remaining * 0.8

        # Initialise h for this run
        if best_h is None and base_h is not None:
            h0 = base_h.copy()
        else:
            h0 = rng.random(n)
        h0 = project_sum_and_bounds(h0, target_sum)

        # Run the gradient‑descent optimiser
        h_opt, c5_opt = optimize_h(h0, alloc, rng.integers(0, 1 << 30))

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        # Early exit if we already beat the target record
        if best_c5 < 0.38:
            break

    # Final polishing with any leftover time
    final_elapsed = time.time() - start_time
    final_remaining = budget_s - final_elapsed
    if best_h is not None and final_remaining > 1.0:
        best_h, best_c5 = local_refine(best_h, final_remaining, rng)

    # Ensure feasibility before returning
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
