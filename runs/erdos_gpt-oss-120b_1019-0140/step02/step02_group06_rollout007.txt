def refine_h(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n/2.0
    dx = 2.0/n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.3
    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(v) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # update smoothing
        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)
        # softmax weights
        vmax = np.max(v)
        w = np.exp(beta * (v - vmax))
        w_sum = np.sum(w)
        if w_sum > 0:
            w /= w_sum
        # compute gradient
        g = np.zeros_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]
        # zero-sum direction
        g -= np.mean(g)
        # step size schedule
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)
        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        # accept
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        else:
            # sometimes random perturbation
            if rng.random() < 0.01:
                h = project_sum_and_bounds(h + rng.normal(scale=0.01, size=n), target)
    return best_h, best_c5

def project_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute evenly then clip again and iterate
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # fix iteratively
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def tournament_select(pop, fitness, k, rng):
    """Select the best individual among k random participants."""
    idx = rng.choice(len(pop), size=k, replace=False)
    best_idx = idx[np.argmin(fitness[idx])]
    return pop[best_idx]

def crossover(p1, p2, n, rng, rate):
    """Single‑point crossover with probability rate."""
    if rng.random() < rate:
        point = rng.integers(1, n)
        return np.concatenate([p1[:point], p2[point:]])
    return p1.copy()

def mutate(child, n, rng, rate, sigma):
    """Add Gaussian noise with probability rate."""
    if rng.random() < rate:
        child = child + rng.normal(scale=sigma, size=n)
        child = np.clip(child, 0.0, 1.0)
    return child

def refine_h(h0, time_budget, seed):
    """Local gradient‑descent refinement using a smoothed max objective."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0
    max_beta = 30.0
    alpha0 = 0.3

    while time.time() - start < time_budget:
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        if beta < max_beta:
            beta = min(max_beta, beta * 1.005)

        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum

        g = np.zeros_like(h)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        g -= np.mean(g)  # zero‑sum direction

        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)
        c5_new = compute_c5(h_new)

        if c5_new < c5 or rng.random() < 0.001:
            h = h_new
        else:
            if rng.random() < 0.01:
                h = project_sum_and_bounds(h + rng.normal(scale=0.01, size=n), target)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine dimension and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = rng.random(n)

    target = n / 2.0
    h0 = project_sum_and_bounds(h0, target)

    # GA parameters
    POP_SIZE = 30
    ELITE_FRAC = 0.2
    TOURNAMENT_SIZE = 3
    CROSSOVER_RATE = 0.9
    MUTATION_RATE = 0.2

    # Initialise population
    pop = [h0]
    for _ in range(POP_SIZE - 1):
        cand = h0 + rng.normal(scale=0.1, size=n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_sum_and_bounds(cand, target)
        pop.append(cand)

    best_h = None
    best_c5 = float('inf')

    # Main GA loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 5.0:  # reserve time for refinement
            break

        fitness = np.array([compute_c5(ind) for ind in pop])
        idx_best = np.argmin(fitness)
        if fitness[idx_best] < best_c5:
            best_c5 = fitness[idx_best]
            best_h = pop[idx_best].copy()

        elite_cnt = max(1, int(POP_SIZE * ELITE_FRAC))
        elite_idxs = np.argsort(fitness)[:elite_cnt]
        new_pop = [pop[i].copy() for i in elite_idxs]

        # Fill rest of the population
        while len(new_pop) < POP_SIZE:
            parent1 = tournament_select(pop, fitness, TOURNAMENT_SIZE, rng)
            parent2 = tournament_select(pop, fitness, TOURNAMENT_SIZE, rng)
            child = crossover(parent1, parent2, n, rng, CROSSOVER_RATE)
            sigma = 0.1 * (1.0 - elapsed / budget_s)  # linear decay
            child = mutate(child, n, rng, MUTATION_RATE, sigma)
            child = project_sum_and_bounds(child, target)
            new_pop.append(child)

        pop = new_pop

    # Ensure we have a best candidate before refinement
    if best_h is None:
        best_h = h0.copy()
        best_c5 = compute_c5(best_h)

    # Local refinement on the best discovered solution
    remaining = max(0.0, budget_s - (time.time() - start))
    refined_h, refined_c5 = refine_h(best_h, remaining, seed + 12345)

    refined_h = project_sum_and_bounds(refined_h, target)
    final_c5 = compute_c5(refined_h)

    return refined_h, final_c5, n
