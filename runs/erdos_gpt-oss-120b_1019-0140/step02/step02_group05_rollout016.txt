def project_feasible(h):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # find interior indices
        interior = np.where((h > 1e-12) & (h < 1-1e-12))[0]
        if interior.size > 0:
            inc = diff / interior.size
            h[interior] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
        # if still diff, adjust by scanning
        if abs(diff) > 1e-12:
            # go through all indices
            for i in range(len(h)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj_and_grad(h, T):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    overlaps = corr * dx
    # Softmax stable
    max_ov = np.max(overlaps)
    exp_vals = np.exp((overlaps - max_ov) / T)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = max_ov + T * np.log(sum_exp)

    grad = np.zeros_like(h)
    # vectorized contributions per shift
    # Use for loop over shift index s; could be 159 loops; fine.
    for s, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = s - (n - 1)
        if k >= 0:
            # part 1
            grad[:n-k] += wk * dx * (1 - h[k:])
            # part 2
            grad[k:] -= wk * dx * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wk * dx * (1 - h[:n-kk])
            grad[:n-kk] -= wk * dx * h[kk:]
    return obj, grad

def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Clip to [0,1] and enforce sum==target_sum by redistributing on interior points."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        interior = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if interior.size:
            inc = diff / interior.size
            h[interior] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
        if abs(diff) > 1e-12:
            for i in range(len(h)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj_and_grad(h, T):
    """Smooth approximation of max overlap (log‑sum‑exp) and its gradient."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * dx                     # length 2n‑1
    max_ov = np.max(overlaps)
    exp_vals = np.exp((overlaps - max_ov) / T)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = max_ov + T * np.log(sum_exp)

    grad = np.zeros_like(h)
    for s, wk in enumerate(w):
        if wk == 0.0:
            continue

        if k >= 0:
            grad[:n - k] += wk * dx * (1.0 - h[k:])

        else:
            kk = -k
            grad[kk:]   += wk * dx * (1.0 - h[:n - kk])
            grad[:n-kk] -= wk * dx * h[kk:]
    return obj, grad

    remaining = budget_s - (time.time() - start)
    if remaining > 8.0:
        dx = 2.0 / n

        def sum_eq(x):
            return np.sum(x[:n]) - target_sum

        def make_constr(k):
            def fun(x):
                hvec = x[:n]
                if k >= 0:
                    val = np.dot(hvec[:n - k], 1.0 - hvec[k:])
                else:
                    kk = -k
                    val = np.dot(hvec[kk:], 1.0 - hvec[:n - kk])
                return x[-1] - val * dx
            return fun

        constraints = [{"type": "eq", "fun": sum_eq}]
        for kk in range(-n + 1, n):
            if kk == 0:
                continue
            constraints.append({"type": "ineq", "fun": make_constr(kk)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_feasible(h_opt, target_sum)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # final safety correction
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, target_sum)

    return best_h, best_val, n
