                    h_sol[chosen] = 1.0
                elif diff < 0:

                    chosen = rng.choice(ones, -diff, replace=False)
                    h_sol[chosen] = 0.0
            best_h = h_sol
            best_c5 = compute_c5(best_h)

    # We'll also consider initial_h_values as candidate too

        # use initial or random binary
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values, dtype=float)
        else:
            best_h = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            best_h[idx] = 1.0
        best_c5 = compute_c5(best_h)

    # We'll limit time left
    remaining = budget_s - (time.time() - start)
    # Simple hill climbing: repeatedly try swapping a 1 and 0 that reduces C5
    # Stop after time or after many unsuccessful attempts
    # We'll treat best_h as binary (0/1)
    bin_h = (best_h >= 0.5).astype(int)

    no_improve = 0
    max_no_improve = 2000
    while (time.time() - start) < budget_s - 0.5 and no_improve < max_no_improve:
        ones = np.flatnonzero(bin_h)
        zeros = np.flatnonzero(1 - bin_h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = bin_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < cur_c5:
            bin_h = cand
            cur_c5 = cand_c5
            no_improve = 0
        else:
            no_improve += 1
    best_h = bin_h.astype(float)
    best_c5 = cur_c5
    # Final safety clip and sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - best_h.sum()
        # adjust a few entries
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - best_h.sum()
    return best_h, best_c5, n

import time
import math
import numpy as np
import scipy.optimize as opt

def compute_c5(arr):
    """Approximated C5 via linear correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode="full")
    return np.max(corr) * (2.0 / n)


def continuous_anneal(start_h, time_end, rng):
    """Simple pairwise annealing preserving the total sum."""
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 500000
    it = 0

    while time.time() < time_end and it < max_iter:
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1

    # final projection to exact sum
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)
    return best_h, best_c5


def binary_hill_climb(start_bin, time_end, rng):
    """Swap‑based hill climbing on a binary vector."""
    n = len(start_bin)
    h = start_bin.copy()
    best_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    no_improve = 0
    while time.time() < time_end and no_improve < 3000:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
            no_improve = 0
        else:
            no_improve += 1
    return best_h.astype(float), best_c5


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = np.sum(h0)
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        h0 += diff / n
        h0 = np.clip(h0, 0.0, 1.0)
        residual = target_sum - np.sum(h0)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -h0[i], 1 - h0[i])
            h0[i] += adj
            residual = target_sum - np.sum(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----- SLSQP local optimisation ------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30.0:
        # helper for a single shift
        def c_shift(k, h):
            if k >= 0:
                overlap = np.dot(h[: n - k], h[k:])
            else:
                kk = -k
                overlap = np.dot(h[kk:], h[: n - kk])
            return target_sum - overlap

        # objective: minimise auxiliary variable t
        def obj(x):
            return x[-1]

        # constraints
        cons = [{'type': 'eq',
                 'fun': lambda x: np.sum(x[:n]) - target_sum}]
        for k in range(-n + 1, n):

                continue

                         'fun': lambda x, k=k: x[-1] - c_shift(k, x[:n])})

        # initial guess for t
        init_t = max(c_shift(k, best_h) for k in range(-n + 1, n))
        x0 = np.concatenate([best_h, [init_t]])

        maxiter = int(min(2000, remaining * 10))
        try:

            diff = target_sum - np.sum(cand_h)

                    residual = target_sum - np.sum(cand_h)
            cand_c5 = compute_c5(cand_h)
            if cand_c5 < best_c5:
                best_h = cand_h
                best_c5 = cand_c5

    # ----- continuous annealing ----------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 4.0:
        end = time.time() + remaining - 1.0
        cand_h, cand_c5 = continuous_anneal(best_h, end, rng)
        if cand_c5 < best_c5:
            best_h = cand_h
            best_c5 = cand_c5

    # ----- final binary hill climb (optional) ---------------------------

    if remaining > 2.0:
        bin_start = (best_h >= 0.5).astype(int)
        final_bin, final_c5 = binary_hill_climb(
            bin_start, time.time() + remaining - 0.5, rng)
        if final_c5 < best_c5:
            best_h = final_bin.astype(float)
            best_c5 = final_c5

    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:

                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    return best_h, best_c5, n
