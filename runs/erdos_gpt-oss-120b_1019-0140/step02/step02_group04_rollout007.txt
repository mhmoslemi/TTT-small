def binary_from_initial(arr):
    # arr is np.ndarray shape (n)
    # compute sorted indices.
    # set top S entries to 1, others to 0.
    idx = np.argsort(-arr)[:S]
    out = np.zeros_like(arr, dtype=float)
    out[idx] = 1.0
    return out

import time
import numpy as np

def _c5_from_binary(h_bin, dx):
    """Compute C5 for a binary (0/1) vector."""
    n = len(h_bin)
    # full autocorrelation
    corr = np.correlate(h_bin, h_bin, mode="full")
    # drop the zero shift (center)
    corr_no_center = np.concatenate((corr[: n - 1], corr[n:]))
    min_c = np.min(corr_no_center)
    # number of ones
    s = n / 2.0
    max_mismatch = s - min_c  # integer value
    return max_mismatch * dx

def _binary_from_values(v, target_sum):
    """Return a binary vector with exactly target_sum ones: largest entries become 1."""
    idx = np.argsort(-v)[:target_sum]
    b = np.zeros_like(v, dtype=np.int8)
    b[idx] = 1
    return b

def _random_binary(n, target_sum, rng):
    """Random binary vector with given weight."""
    arr = np.concatenate([np.ones(target_sum, dtype=np.int8),
                          np.zeros(n - target_sum, dtype=np.int8)])
    rng.shuffle(arr)
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # problem size
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n = len(init)
    else:
        n = 80
        init = None

    dx = 2.0 / n
    target_c5 = 0.3808
    target_sum = n // 2  # exactly n/2 ones

    # starting vector (binary)
    if init is not None:
        cur_h = _binary_from_values(init, target_sum)
    else:
        cur_h = _random_binary(n, target_sum, rng)

    best_h = cur_h.copy()
    best_val = _c5_from_binary(best_h, dx)

    # main search loop
    # we reserve a small margin to avoid being cut off right at the deadline
    while time.time() - start < budget_s * 0.95:
        cur_val = _c5_from_binary(cur_h, dx)
        no_improve = 0
        max_iters = 5000  # per restart; adaptable to remaining time
        for _ in range(max_iters):
            if time.time() - start >= budget_s * 0.95:
                break

            # evaluate autocorrelation to locate the worst shift
            corr = np.correlate(cur_h, cur_h, mode="full")
            # indices of minimal autocorrelation (excluding zero shift)
            center = n - 1
            # mask away the centre value
            mask = np.ones_like(corr, dtype=bool)
            mask[center] = False
            min_c = np.min(corr[mask])
            worst_idxs = np.where((corr == min_c) & mask)[0]
            # pick one worst shift
            k_idx = int(rng.choice(worst_idxs))
            k = k_idx - center   # shift (can be negative)
            kk = abs(k)

            made_move = False
            if kk > 0:
                # positions where a 1 lines up with a 0 for this shift
                h_front = cur_h[: n - kk]
                h_back = cur_h[kk:]
                A = np.where((h_front == 1) & (h_back == 0))[0]
                if A.size:
                    i = int(rng.choice(A))
                    zeros = np.flatnonzero(cur_h == 0)

                    j = int(rng.choice(zeros))
