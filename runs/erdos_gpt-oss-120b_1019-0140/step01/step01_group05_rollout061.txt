def repair_vector(v):
    # Clip and adjust sum; use iterative scaling + clipping
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def simulated_annealing(h_start, time_limit, rng):
    h_cur = repair_vector(h_start.copy())
    cur_score = c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start_time = time.time()
    iter_count = 0
    while time.time() - start_time < time_limit:
        T = T0 * (0.995 ** iter_count)  # exponential cooling
        # propose move
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        hi = h_cur[i]; hj = h_cur[j]
        # compute bounds
        up = min(1.0 - hi, hj) # positive eps possible
        down = min(hi, 1.0 - hj) # negative eps possible
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        # bound eps by max_eps
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # check bounds - though we already limited.
        new_score = c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        iter_count += 1
    return best_h, best_score

    N = h.size
    max_trials = 50
    while time.time() - start < time_limit:

        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        idx = np.argmax(corr)

        # We'll use worst shift only for heuristic in selecting candidate pairs.
        # For simplicity, random pairs with evaluation.
        improved = False
        for _ in range(max_trials):
            i = rng.integers(N)
            j = rng.integers(N-1)
            if j >= i:

            if best_h[i] == 0 and best_h[j] == 1:
                continue  # no move possible
            up = min(1.0 - best_h[i], best_h[j])

def simulated_annealing(h_start, time_limit, rng, max_eps=0.5, T0=0.2):
    N = len(h_start)
    dx = 2.0/N
    h_cur = repair_vector(h_start.copy())
    cur_score = c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    start = time.time()
    iteration = 0
    while time.time() - start < time_limit:
        # compute temperature
        # Exponential decay: T = T0 * (0.995 ** iteration)
        T = T0 * (0.998 ** iteration)
        # choose random pair
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        hi = h_cur[i]; hj = h_cur[j]
        # determine allowable epsilon
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0 and down == 0:
            continue
        eps = rng.random() * (up + down) - down
        # limit epsilon
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        # apply
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # ensure bounds:
        # Already ensured.
        new_score = c5_value(h_new)
        d = new_score - cur_score
        if d < 0 or rng.random() < math.exp(-d / (T + 1e-12)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        iteration += 1
    return best_h, best_score

    if h_initial.size != base_n:

        x_old = np.linspace(0.0, 2.0, h_initial.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, base_n, endpoint=False)
        h_initial = np.interp(x_new, x_old, h_initial)
    h_initial = _repair_vector(h_initial)

import numpy as np, time, math

def _repair_vector(v):
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

import numpy as np

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)

        v *= factor2

    return v

    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
