def sa_optimize(h_start, time_limit, rng, T0=0.5, max_eps=0.5):

   h_cur = h_start.copy()
   cur_score = c5_value(h_cur)
   best_h = h_cur.copy()
   best_score = cur_score
   N = h_cur.size
   start = time.time()
   while True:
       if time.time() - start > time_limit:
           break
       elapsed = time.time() - start
       frac = 1 - elapsed / time_limit
       T = T0 * max(frac, 0.01) # keep small positive. Or maybe linear decreasing T.

       # pick i,j
       i, j = rng.integers(N, size=2)
       if i == j:
           continue

       hi, hj = h_cur[i], h_cur[j]
       up = min(1 - hi, hj)
       down = min(hi, 1 - hj)
       if up == 0 and down == 0:
           continue
       # Choose eps uniform within [-down, up]; can also random scale by temperature: e.g., choose eps from uniform [-max_eps*frac, max_eps*frac] clipped to allowed range? Let's just uniform in allowed full.
       eps = rng.random() * (up + down) - down

       if eps == 0:
           continue

       h_new = h_cur.copy()
       h_new[i] = hi + eps
       h_new[j] = hj - eps

       # Check bounds (should be within 0-1)
       # Already enforced by eps calculation but just for safety
       if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
           continue

       new_score = c5_value(h_new)
       delta = new_score - cur_score
       if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
           # Accept
           h_cur = h_new
           cur_score = new_score
           if new_score < best_score:
               best_score = new_score
               best_h = h_new.copy()
   return best_h, best_score

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0:
        v[:] = target / v.size
        return np.clip(v, 0.0, 1.0)
    factor = target / s
    v = v * factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v = v * factor2
        v = np.clip(v, 0.0, 1.0)
    return v

import numpy as np
import time
import math

def _repair_vector(v):
    """Clip to [0,1] and enforce sum(v) = len(v)/2."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        return np.clip(v, 0.0, 1.0)
    factor = target / s
    v = v * factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v = v * factor2
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start, time_limit, rng):
    """SA that respects box constraints and the integral constraint."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.5            # initial temperature
    max_eps0 = 0.5      # initial move size
    eps_min = 1e-6
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)      # linear cooling
        T = T0 * frac + 1e-12
        max_eps = max_eps0 * frac + eps_min
        i, j = rng.integers(0, N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)       # max eps to increase hi
        down = min(hi, 1.0 - hj)     # max eps to decrease hi
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        if not (0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0):
            continue
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score
