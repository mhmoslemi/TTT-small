def compute_gradient(h, shift):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        # term1 for i <= n-1-shift: derivative w.r.t h[i] = 1 - h[i+shift]
        grad[:n-shift] += 1.0 - h[shift:]
        # term2 for i >= shift: derivative w.r.t h[i] adds -h[i-shift]
        grad[shift:] += -h[:n-shift]
    else:
        s = -shift
        # shift negative means correlation with w shifted left: equivalent to shift positive on reversed? Compute similarly:
        # For shift negative, correlation formula: C = sum_{i=s}^{n-1} h[i] * (1 - h[i-s]) (since shift = -s)
        # So for p >= s: grad includes (1 - h[p-s]) from term1
        # For p <= n-1 - s: grad includes -h[p+s]
        grad[s:] += 1.0 - h[:n-s] # p from s to n-1
        grad[:n-s] += -h[s:]       # p from 0 to n-1-s
    return grad

import time, math
import numpy as np

# Global RNG (set in run)
rng = None

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand brackets if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Discrete C5 value for a vector h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of C5 w.r.t. h for a fixed shift (worst shift)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[:n - shift] += 1.0 - h[shift:]
        grad[shift:] += -h[:n - shift]
    else:
        s = -shift
        grad[s:] += 1.0 - h[:n - s]
        grad[:n - s] += -h[s:]
    return grad

def projected_momentum_descent(h0, n_iter=5000, step0=0.5, momentum=0.9,
                               start_time=None, time_limit=None):
    """Projected gradient descent with momentum on the max-correlation objective."""
    h = h0.copy()
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, n_iter + 1):
        # Optional early stop
        if time_limit is not None and time.time() - start_time >= time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (h.size - 1)
        grad = compute_gradient(h, shift)

        v = momentum * v + (1.0 - momentum) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * v
        h_new = project_onto_box_sum(h_new, s)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score

def simulated_annealing_refine(h0, end_time):
    """Simple simulated annealing that preserves the sum constraint."""
    global rng
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    n = h.size

    T = 0.05
    T_min = 1e-7
    decay_T = 0.9995
    step_max = 0.2
    decay_step = 0.9999

    while time.time() < end_time:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta_i = 1.0 - h[i]
        max_delta_j = h[j]
        max_delta = min(max_delta_i, max_delta_j, step_max)
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        new_score = compute_c5(h_new)

        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            h = h_new
            cur_score = new_score

        T = max(T * decay_T, T_min)
        step_max = max(step_max * decay_step, 1e-4)

    return best_h, best_score

def thue_morse(n):
    """Return Thue–Morse binary sequence of length n as float array."""
    seq = np.fromiter(((bin(k).count('1') & 1) for k in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global rng
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target_n = 256

    # ----------------------- Build initial candidates -----------------------
    candidates = []

    if 'initial_h_values' in globals():
        h0 = upsample(np.asarray(initial_h_values, dtype=float), target_n)
        h0 = project_onto_box_sum(h0, target_n / 2.0)
        candidates.append(h0)

    candidates.append(thue_morse(target_n))
    for _ in range(3):
        candidates.append(random_binary_balanced(target_n, rng))
    for _ in range(3):
        v = rng.random(target_n)
        candidates.append(project_onto_box_sum(v, target_n / 2.0))

    # ----------------------- Momentum descent on each candidate -----------
    best_h = None
    best_score = float('inf')
    time_for_descent = budget_s * 0.7
    for base_h in candidates:
        if time.time() - start_time > time_for_descent:
            break
        # allocate iterations (more for the first candidate)
        iters = 6000 if best_h is None else 4000
        h_opt, score_opt = projected_momentum_descent(
            base_h, n_iter=iters, step0=0.5, momentum=0.9,
            start_time=start_time,
            time_limit=time_for_descent - (time.time() - start_time)
        )
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    # ----------------------- Simulated annealing refinement -------------
    remaining_time = budget_s - (time.time() - start_time)
    if best_h is not None and remaining_time > 5.0:
        end_time = time.time() + remaining_time * 0.9
        h_refined, score_refined = simulated_annealing_refine(best_h, end_time)
        if score_refined < best_score:
            best_score = score_refined
            best_h = h_refined.copy()

        # final short momentum polish
        end_time2 = time.time() + remaining_time * 0.05
        h_final, score_final = projected_momentum_descent(
            best_h, n_iter=2000, step0=0.3, momentum=0.95,
            start_time=start_time,
            time_limit=end_time2 - start_time
        )
        if score_final < best_score:
            best_score = score_final
            best_h = h_final.copy()

    # ----------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(target_n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, target_n / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, target_n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, target_n
