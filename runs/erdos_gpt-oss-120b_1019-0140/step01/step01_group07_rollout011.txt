import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral C5 for discretized h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Project vector v onto [0,1] and sum(v)=target (water-filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    # adjust excess or deficit iteratively
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # distribute need to largest capacities first
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _local_optimize(h0, target_sum, time_budget, cur_best, target_bound):

        # random perturbation around current best
        perturb = np.random.randn(n) * 0.02
        x0 = best_h + perturb
        x0 = _project_to_constraints(x0, target_sum)

        cons = {"type": "eq", "fun": lambda h, tgt=target_sum: np.sum(h) - tgt}
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                _compute_c5,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[cons],
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue
        if not res.success:
            continue
        h_opt = _project_to_constraints(res.x, target_sum)
        score = _compute_c5(h_opt)
        if score < best_score:
            best_score = float(score)
            best_h = h_opt.copy()
            if best_score < target_bound:
                break
    return best_h, best_score

def _anneal(h0, target_sum, time_budget, cur_best, target_bound):
    """Simple pairwise annealing that preserves sum exactly."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_score = cur_best
    cur_score = _compute_c5(h)
    n = h.size
    T = 0.05
    max_step = 0.1
    while time.time() - start < time_budget * 0.9:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * max_step
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        h_new = h.copy()
        h_new[i] = hi
        h_new[j] = hj
        new_score = _compute_c5(h_new)
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                if best_score < target_bound:
                    break
        T *= 0.9999
        if T < 1e-6:
            T = 1e-6
        # occasional step reduction
        if np.random.rand() < 0.001:
            max_step *= 0.99
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    target_bound = 0.3808

    # Initial vector
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        init_vec = np.array(init_vec, dtype=np.float64).ravel()
        n0 = init_vec.size
    else:
        n0 = 300
        init_vec = np.random.rand(n0)

    # Determine resolutions to explore
    n_vals = [n0]
    if budget_s > 500:
        n_vals.append(n0 * 2)  # double resolution if time permits

    best_h = None
    best_score = np.inf

    for n in n_vals:
        # Upscale or keep base resolution
        if n != init_vec.size:
            src = best_h if best_h is not None else init_vec
            x_src = np.linspace(0.0, 2.0, src.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_start = np.interp(x_new, x_src, src)
        else:
            h_start = init_vec.copy()

        target_sum = n / 2.0
        h_start = _project_to_constraints(h_start, target_sum)

        # Time allocation for this resolution
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= 0:
            break
        # Rough split: 60% local, 40% annealing
        local_time = remaining * 0.6
        anneal_time = remaining * 0.4

        # Local SLSQP optimization
        cur_best = _compute_c5(h_start) if best_score == np.inf else best_score
        h_opt, score_opt = _local_optimize(
            h_start, target_sum, local_time, cur_best, target_bound
        )
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
            if best_score < target_bound:
                break

        # Annealing refinement
        h_ref, score_ref = _anneal(
            best_h, target_sum, anneal_time, best_score, target_bound
        )
        if score_ref < best_score:
            best_score = score_ref
            best_h = h_ref.copy()
            if best_score < target_bound:
                break

    if best_h is None:
        # fallback to a feasible simple construction
        n = n0
        best_h = np.full(n, 0.5)
        best_h = _project_to_constraints(best_h, n / 2.0)
        best_score = _compute_c5(best_h)

    # Final evaluation using provided helper if present
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_h.size)
