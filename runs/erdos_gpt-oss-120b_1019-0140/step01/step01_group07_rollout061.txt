import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Compute max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clamp v to [0,1] and adjust sum(v) to target by water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 0:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 0:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _sa_optimize(h_init, target_sum, overall_start, budget_s, time_limit, best_score_target=0.3808):
    """Simulated annealing with sum‑preserving pair moves."""

    sa_start = time.time()
    iter_cnt = 0
    max_iters = 5_000_000   # safety cap
    while True:
        now = time.time()
        if now - sa_start > time_limit:
            break
        if now - overall_start > budget_s - 5.0:
            break
        if iter_cnt >= max_iters:
            break
        # linear cooling
        progress = iter_cnt / max_iters
        T = max(1e-12, init_T * (1.0 - progress))
        max_delta = max_delta0 * (1.0 - progress)
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        hi, hj = h_cur[i], h_cur[j]
        # feasible delta range that keeps both in [0,1]
        delta_low = max(-hi, hj - 1.0)
        delta_high = min(1.0 - hi, hj)
        if delta_low >= delta_high:
            iter_cnt += 1
            continue
        delta = np.random.uniform(delta_low, delta_high)
        if max_delta > 0.0 and abs(delta) > max_delta:
            delta = np.sign(delta) * max_delta
        h_new = h_cur.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
        new_score = _compute_c5(h_new)
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / T):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score < best_score_target:
                    break
        iter_cnt += 1
    return best_h, best_score

def _refine_with_slsqp(h_start, target_sum, overall_start, budget_s, time_limit):
    """Local refinement using SLSQP, respecting bounds and sum constraint."""
    if time.time() - overall_start > budget_s - 2.0:
        return h_start, _compute_c5(h_start)
    cons = {"type": "eq", "fun": lambda h, tgt=target_sum: np.sum(h) - tgt}
    bounds = [(0.0, 1.0)] * h_start.size
    try:
        res = minimize(
            _compute_c5,
            h_start,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": 500, "ftol": 1e-12, "disp": False},
        )
        h_opt = _project_to_constraints(res.x, target_sum)
        score = _compute_c5(h_opt)
    except Exception:
        h_opt = h_start
        score = _compute_c5(h_opt)
    return h_opt, score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    overall_start = time.time()

    # --- initial vector -------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        init_vec = np.array(init_vec, dtype=np.float64).ravel()
        n0 = init_vec.size
        target_sum = n0 / 2.0
        h0 = _project_to_constraints(init_vec, target_sum)
    else:
        n0 = 380
        target_sum = n0 / 2.0
        h0 = np.random.rand(n0)
        h0 = _project_to_constraints(h0, target_sum)

    best_h = h0.copy()
    best_score = _compute_c5(best_h)

    # --- resolution schedule --------------------------------------------
    resolutions = [n0]
    if budget_s > 400:
        resolutions.append(n0 * 2)
    if budget_s > 800:
        resolutions.append(n0 * 4)

    for n in resolutions:
        if n != best_h.size:
            # Upsample previous best
            x_old = np.arange(best_h.size) * (2.0 / best_h.size)
            x_new = np.arange(n) * (2.0 / n)
            h_start = np.interp(x_new, x_old, best_h)
            target_sum = n / 2.0
            h_start = _project_to_constraints(h_start, target_sum)
        else:
            h_start = best_h.copy()
            target_sum = n / 2.0

        # time check
        elapsed = time.time() - overall_start
        remaining = budget_s - elapsed
        if remaining < 30.0:
            break

        # allocate time for SA (≈60% of remaining)
        sa_time = remaining * 0.6
        cur_h, cur_score = _sa_optimize(
            h_start,
            target_sum,
            overall_start,
            budget_s,
            sa_time,
            best_score_target=0.3808,
        )
        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h.copy()
            if best_score < 0.3808:
                break

        # time check before refinement
        elapsed = time.time() - overall_start
        remaining = budget_s - elapsed
        if remaining < 15.0:
            break

        # refine with SLSQP (use the remaining time)
        refine_time = remaining * 0.9
        refined_h, refined_score = _refine_with_slsqp(
            best_h, target_sum, overall_start, budget_s, refine_time
        )
        if refined_score < best_score:
            best_score = refined_score
            best_h = refined_h.copy()
            if best_score < 0.3808:
                break

    final_score = best_score
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            pass
    return best_h.astype(np.float64), float(final_score), int(best_h.size)
