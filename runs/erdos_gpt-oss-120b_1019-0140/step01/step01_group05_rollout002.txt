def _simulated_annealing(h_start, time_limit, rng, T0=0.2, Tf=1e-4, max_eps=0.5):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * ((Tf / T0) ** frac)  # exponential cooling
        # pick two different indices
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi = h_cur[i]
        hj = h_cur[j]
        # Determine allowable epsilon range to keep within [0,1] after swap
        up = min(1.0 - hi, hj)  # we can increase hi (move mass from j to i)
        down = min(hi, 1.0 - hj) # we can decrease hi (move mass from i to j)
        # epsilon uniformly over [-down, up]
        if up + down == 0:
            continue
        eps = (rng.random() * (up + down)) - down
        # limit magnitude to max_eps
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        # propose
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # evaluate
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
    return best_h, best_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip -> [0,1] and scale to satisfy sum(v) = len(v)/2 exactly."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        return np.clip(v, 0.0, 1.0)
    factor = target / s
    v = np.clip(v * factor, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v = np.clip(v * factor2, 0.0, 1.0)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for discretised h."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float,
                         rng: np.random.Generator,
                         T0=0.3, Tf=1e-4, max_eps=0.5) -> tuple:
    """SA preserving sum and box constraints."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * ((Tf / T0) ** frac)  # exponential cooling
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)        # increase i, decrease j
        down = min(hi, 1.0 - hj)      # decrease i, increase j
        if up + down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        # apply move in‑place
        h_cur[i] = hi + eps
        h_cur[j] = hj - eps
        new_score = _c5_value(h_cur)
        delta = new_score - cur_score
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / (T + 1e-15)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            # revert
            h_cur[i] = hi
            h_cur[j] = hj
    return best_h, best_score

def _hill_climb(h_start: np.ndarray, time_limit: float,
                rng: np.random.Generator, max_eps=0.2) -> tuple:
    """Deterministic improvement (no uphill moves)."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while time.time() - start < time_limit:
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up + down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_cur[i] = hi + eps
        h_cur[j] = hj - eps
        new_score = _c5_value(h_cur)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            h_cur[i] = hi
            h_cur[j] = hj
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # initial construction
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        # fallback: random vector of length 100
        n_fallback = 100
        h0 = rng.random(n_fallback)
    h0 = _repair_vector(h0)
    n0 = h0.size

    start_time = time.time()
    best_h = h0.copy()
    best_score = _c5_value(best_h)

    # multiscale hierarchy
    max_n = 800
    levels = [n0]
    while levels[-1] * 2 <= max_n:
        levels.append(levels[-1] * 2)

    prev_h = best_h
    prev_n = n0

    for idx, n in enumerate(levels):
        # time check
        if time.time() - start_time >= budget_s - 2.0:
            break
        # interpolate to new resolution if needed
        if n != prev_n:
            x_prev = np.linspace(0.0, 2.0, prev_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_cur = np.interp(x_new, x_prev, prev_h)
        else:
            h_cur = prev_h.copy()
        h_cur = _repair_vector(h_cur)

        # allocate time for this level
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        remaining_levels = len(levels) - idx
        stage_time = time_left * 0.7 / remaining_levels
        if stage_time < 2.0:
            stage_time = min(2.0, time_left * 0.8)

        # main simulated annealing
        h_opt, score_opt = _simulated_annealing(
            h_cur, stage_time, rng, T0=0.3, Tf=1e-4, max_eps=0.5
        )

        # short hill‑climb refinement
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left > 1.0:
            hc_time = min(0.2 * time_left, 5.0)
            h_opt2, score_opt2 = _hill_climb(h_opt, hc_time, rng, max_eps=0.2)
            if score_opt2 < score_opt:
                h_opt, score_opt = h_opt2, score_opt2

        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

        prev_h, prev_n = h_opt, n

    # final random‑restart SA phase if time permits
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 8.0:
            break
        # give a modest chunk (up to 8 s) to a fresh run
        chunk = min(8.0, time_left * 0.6)
        h_rand = rng.random(prev_n)
        h_rand = _repair_vector(h_rand)
        h_opt, score_opt = _simulated_annealing(
            h_rand, chunk, rng, T0=0.3, Tf=1e-4, max_eps=0.5
        )
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
        # keep loop until budget exhausted

    # final repair and exact evaluation
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
