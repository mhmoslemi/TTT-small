def adjust_sum_to_target(h, target, rng):
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # detect indices that can be increased (h < 1) or decreased (h > 0)
    if diff > 0:
        # need to increase mass
        # find indices where h < 1
        idxs = np.where(h < 1.0)[0]
        if idxs.size == 0:
            # no capacity, cannot adjust
            return h
        # distribute diff uniformly among them, but not exceed 1
        while diff > 1e-12 and idxs.size > 0:
            # compute max increment possible per index
            max_inc = 1.0 - h[idxs]
            # total capacity sum
            cap = max_inc.sum()
            if cap < diff - 1e-12:
                # fill all to 1
                h[idxs] = 1.0
                diff -= cap
                idxs = np.where(h < 1.0)[0]
                continue
            # distribute proportionally
            inc = diff * (max_inc / cap)
            h[idxs] += inc
            diff = 0.0
    else:
        # need to decrease mass
        idxs = np.where(h > 0.0)[0]
        if idxs.size == 0:
            return h
        while diff < -1e-12 and idxs.size > 0:
            max_dec = h[idxs]
            cap = max_dec.sum()
            if cap < -diff - 1e-12:
                h[idxs] = 0.0
                diff += cap
                idxs = np.where(h > 0.0)[0]
                continue
            dec = -diff * (max_dec / cap)
            h[idxs] -= dec
            diff = 0.0
    # after loop, may still have tiny diff due to float, clip
    h = np.clip(h, 0.0, 1.0)
    # ensure exact sum by adding small uniform shift
    total = h.sum()
    diff = target - total
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Modify h in place (or return new array) so that sum(h) == target while 0<=h<=1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase mass
        while diff > 1e-12:
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            max_inc = 1.0 - h[idxs]
            cap = max_inc.sum()
            if cap <= diff + 1e-12:
                h[idxs] = 1.0
                diff -= cap
                continue
            inc = diff * (max_inc / cap)
            h[idxs] += inc
            diff = 0.0
    else:
        # need to decrease mass
        while diff < -1e-12:
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            max_dec = h[idxs]
            cap = max_dec.sum()
            if cap <= -diff + 1e-12:
                h[idxs] = 0.0
                diff += cap
                continue
            dec = -diff * (max_dec / cap)
            h[idxs] -= dec
            diff = 0.0
    # final tiny correction
    h = np.clip(h, 0.0, 1.0, out=h)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        h += residual / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.1  # safety margin

    # --- initialise ---
    try:

        n_points = init_h.size

        h0 = adjust_sum_to_target(init_h.copy(), target_sum, rng)

    except Exception:
        n_points = 200
        target_sum = n_points / 2.0
        h0 = rng.rand(n_points)
        h0 *= target_sum / h0.sum()
        h0 = adjust_sum_to_target(h0, target_sum, rng)
        np.clip(h0, 0.0, 1.0, out=h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    current_h = h0.copy()
    current_c5 = best_c5

    # annealing parameters
    step0 = 0.5
    min_step = 1e-4
    T0 = 0.05
    min_T = 1e-5

    while time.time() - start < max_time:
        elapsed = time.time() - start
        progress = elapsed / max_time
        step = step0 * (1.0 - progress) + min_step * progress
        T = T0 * (1.0 - progress) + min_T * progress

        if move_type < 0.2:
            # gradient move targeting the worst overlap shift
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_shift = int(np.argmax(corr_full)) - (n_points - 1)
            g = np.zeros_like(current_h)
            idx = np.arange(n_points)
            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n_points)

    final_c5 = compute_c5(best_h)
