def _repair_sum_brief(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # adjust uniformly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]

    h[i] = new_val

def _search_h(h0, rng, start_search, time_budget):
    import time, math, numpy as np
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    h = _repair_sum_brief(h, target)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    cur_c5 = best_c5

    max_step = 0.5
    min_step = 0.001
    max_temp = 0.01
    min_temp = 1e-6

    while True:
        now = time.time()
        if now - start_search >= time_budget:
            break
        frac = (now - start_search) / time_budget
        step = max_step * (1 - frac) + min_step * frac
        temp = max_temp * (1 - frac) + min_temp * frac

        # prob of using worst-lag move
        if rng.random() < 0.85:
            i, j, delta = _worst_pair_move(h, step)
            if delta <= 0:
                # random move fallback
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max <= 0:
                    continue
                delta = rng.random() * min(step, delta_max)
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 0:
                continue
            delta = rng.random() * min(step, delta_max)

        # apply move
        new_h = h.copy()
        new_h[i] = h[i] - delta
        new_h[j] = h[j] + delta

        new_c5 = _compute_c5(new_h)

        # acceptance
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
            h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        # else discard

    return best_h, best_c5

def _worst_pair_move(h, step):
    import numpy as np
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    L = idx - (n - 1)
    if L == 0:
        # Not useful; return dummy zero delta to trigger fallback
        return 0, 1, 0.0

    if L > 0:
        hi_slice = h[:n - L]
        hj_slice = h[L:]
        contrib = hi_slice * (1.0 - hj_slice)
        i_local = int(np.argmax(contrib))
        i = i_local
        j = i + L
    else:
        hi_slice = h[-L:]  # -L positive
        hj_slice = h[:n + L]
        contrib = hi_slice * (1.0 - hj_slice)
        i_local = int(np.argmax(contrib))
        i = i_local - L
        j = i_local
    delta_max = min(h[i], 1.0 - h[j])
    if delta_max <= 0:
        return i, j, 0.0
    delta = min(step, delta_max)
    return i, j, delta

    h[i] = new_val
    if abs(actual - diff) > 1e-12:

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum_brief(h: np.ndarray, target: float) -> np.ndarray:

    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum_brief(h, target)
    return h

def _worst_pair_move(h: np.ndarray, step: float):
    """
    Locate the lag with maximal correlation and return a feasible pair
    (i, j) together with a delta that decreases hi*(1-hj).
    Returns delta>0 meaning hi decreases by delta and hj increases.
    If no feasible move found (e.g. lag zero), returns delta=0.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    L = idx - (n - 1)

    # Lag zero cannot be used directly (i==j); fall back to random move.
    if L == 0:
        return 0, 1, 0.0

    if L > 0:
        hi_slice = h[: n - L]
        hj_slice = h[L:]
        contrib = hi_slice * (1.0 - hj_slice)
        i_local = int(np.argmax(contrib))
        i = i_local
        j = i + L
    else:  # L < 0
        hi_slice = h[-L:]          # -L > 0
        hj_slice = h[: n + L]      # n+L = n-(-L)
        contrib = hi_slice * (1.0 - hj_slice)
        i_local = int(np.argmax(contrib))
        i = i_local - L            # because L is negative
        j = i_local
    delta_max = min(h[i], 1.0 - h[j])
    if delta_max <= 0:
        return i, j, 0.0
    delta = min(step, delta_max)
    return i, j, delta

def _search_h(h0: np.ndarray, rng, start_search: float, time_budget: float):
    """
    Local stochastic search (simulated annealing) for a fixed discretisation.
    Returns the best (h, C5) found within the supplied time budget.
    """
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    h = _repair_sum_brief(h, target)

    best_h = h.copy()
    best_c5 = _compute_c5(h)
    cur_c5 = best_c5

    max_step = 0.5
    min_step = 0.001
    max_temp = 0.01
    min_temp = 1e-6

    while True:
        now = time.time()
        if now - start_search >= time_budget:
            break
        frac = (now - start_search) / time_budget
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # 85 % of the time we try a move targeting the worst lag
        if rng.random() < 0.85:
            i, j, delta = _worst_pair_move(h, step)
            if delta <= 0:
                # fallback to a simple random pair move
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max <= 0:
                    continue
                delta = rng.random() * min(step, delta_max)
        else:
            # pure random move
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 0:
                continue
            delta = rng.random() * min(step, delta_max)

        # apply move (preserves sum)
        new_h = h.copy()
        new_h[i] = h[i] - delta
        new_h[j] = h[j] + delta

        new_c5 = _compute_c5(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
            h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start_total = time.time()
    total_budget = budget_s

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ---- Use the supplied construction if it exists -----------------------
    try:
        init_h = np.asarray(initial_h_values, dtype=float)
        n0 = init_h.shape[0]
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = _repair_sum_brief(init_h, n0 / 2.0)

        elapsed = time.time() - start_total
        time_left = total_budget - elapsed
        if time_left > 0:
            alloc0 = min(total_budget * 0.3, time_left * 0.8)
            start0 = time.time()
            best_h0, best_c5_0 = _search_h(init_h, rng, start0, alloc0)
            best_h0 = _repair_sum_brief(best_h0, n0 / 2.0)
            best_c5_0 = _compute_c5(best_h0)
            if best_c5_0 < best_c5:
                best_c5 = best_c5_0
                best_h = best_h0
                best_n = n0
    except NameError:
        pass

    # ---- Larger, more flexible discretisation ----------------------------
    elapsed = time.time() - start_total
    time_left = total_budget - elapsed
    if time_left > 0:
        n_large = 200
        h_large = np.zeros(n_large, dtype=float)
        half = n_large // 2
        h_large[:half] = 1.0
        rng.shuffle(h_large)
        # small random perturbations to break symmetry
        for _ in range(5):
            idx = rng.integers(0, n_large)
            h_large[idx] = rng.random()
        h_large = _repair_sum_brief(h_large, n_large / 2.0)

        start_large = time.time()
        best_h_large, best_c5_large = _search_h(h_large, rng, start_large, time_left)
        best_h_large = _repair_sum_brief(best_h_large, n_large / 2.0)
        best_c5_large = _compute_c5(best_h_large)
        if best_c5_large < best_c5:
            best_c5 = best_c5_large
            best_h = best_h_large
            best_n = n_large

    # ---- Fallback simple construction if nothing else succeeded ----------
    if best_h is None:
        n = 200
        h = np.zeros(n, dtype=float)
        half = n // 2
        h[:half] = 1.0
        rng.shuffle(h)
        h = _repair_sum_brief(h, n / 2.0)
        best_h = h
        best_c5 = _compute_c5(h)
        best_n = n

    # final strict repair and recompute
    best_h = _repair_sum_brief(best_h, best_n / 2.0)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, int(best_n)
