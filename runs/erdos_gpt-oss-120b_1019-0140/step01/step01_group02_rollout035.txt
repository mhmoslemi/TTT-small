import time
import math
import numpy as np

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce sum == target, staying in [0,1]."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute evenly then clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _compute_c5_and_offset(h: np.ndarray):
    """Return (max_correlation, offset) for h with (1-h) using FFT."""
    n = h.shape[0]
    L = 2 * n
    # FFT of h and (1-h) with zero‑padding to length L
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[:2 * n - 1]
    max_idx = int(np.argmax(corr))
    max_corr = float(corr[max_idx])
    offset = max_idx - (n - 1)  # lag = offset * dx
    return max_corr, offset

def _gradient_for_offset(h: np.ndarray, offset: int):
    """Gradient of S(k)=sum h_i h_{i+offset} w.r.t. h (abs offset)."""
    n = h.shape[0]
    d = abs(int(offset))
    if d == 0:
        return 2.0 * h
    grad = np.zeros_like(h)
    # forward partner
    grad[:n - d] += h[d:]
    # backward partner
    grad[d:] += h[:n - d]
    return grad

def _random_pair_move(h: np.ndarray, step: float, rng: np.random.Generator):
    """Propose a feasible random transfer between two indices."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None, None, None
    hi, hj = h[i], h[j]
    delta_min = max(-hi, -(1.0 - hj))
    delta_max = min(1.0 - hi, hj)
    low = max(delta_min, -step)
    high = min(delta_max, step)
    if low >= high:
        return None, None, None
    delta = rng.random() * (high - low) + low
    return i, j, delta

    """
    Optimise a step function h on [0,2] (discretised) to lower the
    Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---------- initialise ----------
    try:
        init = np.asarray(initial_h_values, dtype=float)  # provided by harness

        current = np.interp(xt, x0, init)
        current = np.clip(current, 0.0, 1.0)
    except NameError:
        n_target = 300
        current = np.zeros(n_target, dtype=float)
        half = n_target // 2
        current[:half] = 1.0
        if n_target % 2:
            current[half] = 0.5
        rng.shuffle(current)
    target_sum = n_target / 2.0
    current = _repair_sum(current, target_sum)

    # ---------- bookkeeping ----------
    best_h = current.copy()
    max_corr, _ = _compute_c5_and_offset(best_h)
    best_c5 = max_corr * (2.0 / n_target)
    current_c5 = best_c5

    # hyper‑parameters
    max_step = 0.2
    min_step = 0.001
    max_temp = 0.02
    min_temp = 1e-6

    # ---------- optimisation loop ----------
    while True:

        if now - start >= budget_s:
            break
        elapsed = now - start
        frac = elapsed / budget_s

        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # evaluate current worst shift
        max_corr, offset = _compute_c5_and_offset(current)
        cur_c5 = max_corr * (2.0 / n_target)

    final_corr, _ = _compute_c5_and_offset(best_h)
    best_c5 = final_corr * (2.0 / n_target)

    return best_h, best_c5, n_target
