def optimize_h(h0, max_iters, time_limit):
    h = h0.copy()
    n = h.size
    s = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2   # initial step size (may be tuned)
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    for t in range(1, max_iters+1):
        if time.time() - start > time_limit:
            break
        # compute gradient for worst shift
        grad = compute_worst_grad(h)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = lr0 / np.sqrt(t)   # optionally decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        # maybe optional clamp to [0,1] inside projection
        h = h_new
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # occasional random perturbation if stuck
        if t % 500 == 0 and best_score - score > 1e-5:
            # optionally continue
            pass
```

    n = h.size
    s = n/2.0
    m = np.zeros_like(h)
    v_arr = np.zeros_like(h)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr0=0.2
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    for t in range(1, max_iters+1):
        if time.time() - start > time_budget:
            break
        # compute worst shift gradient
        grad = compute_worst_grad(h)
        # update moments
        m = beta1*m + (1-beta1)*grad
        v_arr = beta2*v_arr + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v_arr / (1 - beta2**t)
        lr = lr0 / np.sqrt(t)   # decaying
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, s)
        # evaluate
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # optional occasional random perturbation if stagnation
        # For simplicity, skip.
    return best_h, best_score

def compute_worst_grad(h):
    n = h.size
    # compute correlation vector corr = np.correlate(h, 1-h, mode='full')
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)  # shift = k (positive means h shifted right)
    # compute gradient as 1 - term1 - term2
    grad = np.ones_like(h)
    if shift >= 0:
        # term1: h shifted left: term1[i] = h[i+shift] for i < n-shift
        grad[:n - shift] -= h[shift:]
        # term2: shifted right: term2[i] = h[i-shift] for i >= shift
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        # term1: h[i - s] for i >= s yields term1[i] = h[i - s] (i - s >=0)
        grad[s:] -= h[:n - s]
        # term2: h[i + s] for i < n - s yields term2[i] = h[i + s]
        grad[:n - s] -= h[s:]
    return grad

import time
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """
    Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}.
    Returns an array of same shape as v.
    """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure brackets contain the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1 - h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_worst_grad(h):
    """Gradient of I_k = ∫ h(x)(1-h(x+k))dx for the shift attaining the max."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)        # integer shift (can be negative)
    grad = np.ones_like(h)

    if shift >= 0:
        # term1 = h[i+shift] for i < n-shift
        grad[:n - shift] -= h[shift:]
        # term2 = h[i-shift] for i >= shift
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        # term1 = h[i-s] for i >= s
        grad[s:] -= h[:n - s]
        # term2 = h[i+s] for i < n-s
        grad[:n - s] -= h[s:]
    return grad

def adam_optimize(h0, max_iters, time_budget):
    """
    Adam-style projected optimization of the max-overlap objective.
    Returns the best encountered (h, score).
    """
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5
    start = time.time()

    for t in range(1, max_iters + 1):
        if time.time() - start > time_budget:
            break
        grad = compute_worst_grad(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        # bias correction
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h = h - step
        h = project_onto_box_sum(h, target_sum)

        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def thue_morse(n):
    """Return the Thue–Morse binary sequence (0/1) of length n."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    arr[idx] = 1.0
    return arr
