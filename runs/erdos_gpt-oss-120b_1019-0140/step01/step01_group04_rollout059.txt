Pseudo:

def anneal(v, time_limit, max_step=0.2):
    start = time.time()
    best = v.copy()
    best_score = compute_c5(v)
    cur = v.copy()
    cur_score = best_score
    T0 = 0.1 # initial temperature relative to score scale
    while time.time() - start < time_limit:
        # gradually decrease temperature
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) # linear cooling
        # propose move: pick i != j; delta ~ uniform(-max_step, max_step)
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2 * max_step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        # ensure bounds
        cand = np.clip(cand, 0.0, 1.0)
        # repair sum
        # using iterative repair: adjust sum difference
        s = cand.sum()
        target = n/2.0
        diff = s - target
        if abs(diff) > 1e-12:
            # adjust proportionally across all free variables
            # For simplicity: linearly distribute diff equally across all entries not at bound
            mask = (cand > 0.0) & (cand < 1.0)
            if mask.sum() > 0:
                cand[mask] -= diff / mask.sum()
                cand = np.clip(cand, 0.0, 1.0)
            else:
                # fallback simple scale
                cand *= target / s if s > 0 else 0.0
                cand = np.clip(cand, 0.0, 1.0)
        # compute score
        score = compute_c5(cand)
        # acceptance
        if score < cur_score or np.random.rand() < np.exp(-(score - cur_score) / (T + 1e-12)):
            cur = cand
            cur_score = score
            if score < best_score:
                best = cand.copy()
                best_score = score
    return best, best_score

def _anneal(v, time_limit, max_step=0.2):
    n = len(v)
    cur = v.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    # parameters
    while time.time() - start < time_limit:
        # linear cooling schedule
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = 0.02 * (1 - frac) + 0.001  # small baseline
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * max_step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        # repair
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        # accept if better or with probability
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
    return best, best_score

import time
import math
import numpy as np

# ---- core utilities ---------------------------------------------------------

def _repair_vector(v, n, max_iter=20):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        if s > 0:
            v *= target / s
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        diff = target - s
        if abs(diff) > 1e-9:
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

# ---- deterministic initial shapes -------------------------------------------

def _cos_initial(n):
    """U‑shaped cosine: high at ends, low in the middle."""
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    v = 0.5 + 0.5 * np.cos(np.pi * xs)          # (1+cos(πx))/2
    return _repair_vector(v, n)

def _edge_initial(n):
    """V‑shaped linear shape: value 1 at both ends, 0 in the centre."""
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    v = np.abs(xs - 1.0)                         # 0 at centre, 1 at ends
    v = v / v.max() if v.max() != 0 else v
    return _repair_vector(v, n)

# ---- local optimisation helpers ---------------------------------------------

def _anneal(v, time_limit, max_step=0.2):
    """Simple simulated‑annealing style search."""
    n = len(v)
    cur = v.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    # temperature parameters
    T0, Tf = 0.03, 0.0005
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + Tf * frac
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * max_step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
    return best, best_score

def _local_search(v, time_limit, step=0.05):
    """Hill‑climbing by swapping small amounts."""
    n = len(v)
    best = v.copy()
    best_score = _compute_c5(best)
    start = time.time()
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        sc = _compute_c5(cand)
        if sc < best_score:
            best, best_score = cand, sc
    return best, best_score

def _optimize_at_resolution(n, init_v, time_limit):
    """SLSQP refinement (if scipy is available)."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return init_v, _compute_c5(init_v)
    obj = lambda x: _compute_c5(x)
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj,
            init_v,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": 250, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            v = _repair_vector(res.x, n)
            return v, _compute_c5(v)
    except Exception:
        pass
    return init_v, _compute_c5(init_v)

    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------ #
