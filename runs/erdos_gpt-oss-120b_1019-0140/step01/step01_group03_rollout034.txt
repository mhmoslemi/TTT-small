def smooth_pg_descent(h, max_iters, step0=1.0, tau0=0.1, tau_end=0.001, momentum=0.9, time_limit=None, start_time=None):
    n = h.size
    target_sum = n / 2.0
    v = np.zeros_like(h)  # momentum
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters+1):
        # time check
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # schedule tau decreasing linearly (or exponential) from tau0 to tau_end over max_iters
        tau = tau0 * (tau_end / tau0) ** ((it - 1) / max_iters)  # exponential decay
        # compute weights
        # w = np.exp((corr - max_corr) / tau)  # stable
        w = np.exp((corr - max_corr) / max(tau, 1e-12))
        # compute gradient weighted
        grad = weighted_gradient(h, w)
        # step size
        step = step0 / math.sqrt(it)
        # momentum update
        v = momentum * v - step * grad
        h_new = h + v
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate c5
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def weighted_gradient(h, w):
    n = h.shape[0]
    mid = n - 1
    total_w = w.sum()
    grad = np.full(n, total_w, dtype=float)
    for shift in range(-mid, mid+1):

        if weight == 0.0:
            continue
        s = shift
        if s >= 0:
            if s < n:
                # first term
                grad[:n - s] -= weight * h[s:]
                # second term
                grad[s:] -= weight * h[:n - s]
        else:
            s = -s
            # same code (since |shift|)
            grad[:n - s] -= weight * h[s:]
            grad[s:] -= weight * h[:n - s]
    return grad

def smooth_pg_descent(h0, max_iters, step0=0.5, tau0=0.1, tau_end=0.001, momentum=0.9, time_limit=None, start_time=None):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters+1):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # temperature schedule exponential
        tau = tau0 * (tau_end / tau0) ** ((it - 1) / max_iters)
        # compute weights
        if max_corr == float('-inf'):
            # no correlation? shouldn't happen
            continue
        w = np.exp((corr - max_corr) / max(tau, 1e-12))
        # gradient
        grad = weighted_gradient(h, w)
        # step size
        step = step0 / math.sqrt(it)
        # momentum
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

import time, math
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of arr to length new_n (can up- or down-sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def weighted_gradient(h, w):
    """
    Gradient of the soft‑max of correlations.
    h: vector (n,)
    w: weights for each shift, length 2n‑1 (aligned with np.correlate output)
    """
    n = h.size
    mid = n - 1
    total_w = w.sum()
    grad = np.full(n, total_w, dtype=float)

    for shift in range(-mid, mid + 1):
        weight = w[shift + mid]
        if weight == 0.0:
            continue
        s = abs(shift)
        if s == 0:
            grad[:] -= 2.0 * weight * h
        else:
            # subtract h[i+s] for i = 0..n-s-1
            grad[:n - s] -= weight * h[s:]
            # subtract h[i-s] for i = s..n-1
            grad[s:] -= weight * h[:n - s]
    return grad

def smooth_pg_descent(h0, max_iters, step0=0.5,
                     tau0=0.2, tau_end=0.001,
                     momentum=0.9,
                     time_limit=None, start_time=None):
    """
    Projected gradient descent on a smoothed max‑correlation objective.
    Returns best (h, score) seen during the run.
    """
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    v = np.zeros_like(h)               # momentum term
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iters + 1):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # exponential decay of temperature
        tau = tau0 * (tau_end / tau0) ** ((it - 1) / max_iters)
        # soft‑max weights (stable)
        w = np.exp((corr - max_corr) / max(tau, 1e-12))
        grad = weighted_gradient(h, w)

        step = step0 / math.sqrt(it)
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, target_sum)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def random_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    arr[idx] = 1.0
    return arr

def binary_local_search(h_bin, max_iters, rng):
    """
    Simple hill‑climbing on binary balanced vectors.
    Swaps a random 1 with a random 0; keeps change if it improves C5.
    """
    h = h_bin.copy()
    best_score = compute_c5(h)
    for _ in range(max_iters):
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        score = compute_c5(h)
        if score < best_score - 1e-12:
            best_score = score
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial construction (if supplied) or random balanced vector
    if 'initial_h_values' in globals():
        init_vec = np.asarray(initial_h_values, dtype=float)
    else:
        init_vec = random_balanced(256, rng)

    best_h = None
    best_score = float('inf')

    stage_sizes = [64, 128, 256]

    for n in stage_sizes:
        # start from previous best (upsampled) or from the global init
        if best_h is not None:
            h_start = upsample(best_h, n)
        else:
            h_start = upsample(init_vec, n)
        h_start = project_onto_box_sum(h_start, n / 2.0)

        # allocate iterations based on remaining time
        time_left = budget_s - (time.time() - start_time) - 2.0
        if time_left <= 0:
            break
        # rough estimate: 0.001 s per iter for n=256, scale linearly
        est_iter_time = 0.001 * (n / 256.0)
        max_iters = max(1, int(time_left / est_iter_time * 0.5))
        max_iters = min(max_iters, 20000)

        # smoothed projected gradient descent
        h_opt, score_opt = smooth_pg_descent(
            h_start, max_iters,
            step0=0.5, tau0=0.2, tau_end=0.001,
            momentum=0.9,
            time_limit=time_left, start_time=time.time()
        )
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

        # binary local search on a fresh random start at this resolution
        time_left = budget_s - (time.time() - start_time) - 1.0
        if time_left > 0.5:
            h_bin = random_balanced(n, rng)
            max_swaps = int(min(5000, time_left / 0.0002))
            h_bin_opt, score_bin_opt = binary_local_search(h_bin, max_swaps, rng)
            if score_bin_opt < best_score:
                best_score = score_bin_opt
                best_h = h_bin_opt.copy()

            # refine the binary solution with a short PGD run
            time_left = budget_s - (time.time() - start_time) - 0.5
            if time_left > 0.1:
                h_refined, score_refined = smooth_pg_descent(
                    h_bin_opt, max_iters=2000,
                    step0=0.5, tau0=0.1, tau_end=0.001,
                    momentum=0.9,
                    time_limit=time_left, start_time=time.time()
                )
                if score_refined < best_score:
                    best_score = score_refined
                    best_h = h_refined.copy()

    # final random‑restart phase until budget expires
    while time.time() - start_time < budget_s - 0.5:
        n = rng.choice([64, 128, 256])
        h_bin = random_balanced(n, rng)
        max_swaps = 1000
        h_bin_opt, score_bin_opt = binary_local_search(h_bin, max_swaps, rng)
        if score_bin_opt < best_score:
            best_score = score_bin_opt
            best_h = h_bin_opt.copy()

        time_left = budget_s - (time.time() - start_time) - 0.2
        if time_left <= 0:
            break
        # a few gradient steps from the current best
        h_refined, score_refined = smooth_pg_descent(
            best_h, max_iters=500,
            step0=0.5, tau0=0.05, tau_end=0.001,
            momentum=0.9,
            time_limit=time_left, start_time=time.time()
        )
        if score_refined < best_score:
            best_score = score_refined
            best_h = h_refined.copy()

    # final feasibility projection and score recomputation
    if best_h is None:
        best_h = np.full(256, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, 128.0)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
