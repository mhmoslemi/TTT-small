    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _gradient_move(h, shift, rng, max_eps=0.2):
    N = h.size
    # compute grad
    grad = np.zeros(N, dtype=np.float64)
    # shift positive (lag) can be negative; we need to compute according to definition: correlation index shift = shift; meaning term h[i] * (1 - h[i+shift]) for i+shift in [0,N-1].
    # Let's compute contributions:
    if shift >=0:
        # i from 0 to N-1-shift: term h[i] * (1 - h[i+shift])
        # derivative for these i: (1 - h[i+shift])
        grad[:N-shift] += (1.0 - h[shift:])
        # For i from shift to N-1: these i appear as second factor (when i = j+shift). derivative = -h[i-shift]
        grad[shift:] += -h[:N-shift]
    else:
        # shift negative => shift = -d where d>0
        d = -shift
        # For i from d to N-1: term h[i] * (1 - h[i-d]) (since i+shift = i-d)
        grad[d:] += (1.0 - h[:N-d])
        # For i from 0 to N-d-1: i appears as second factor in term h[i+d] * (1 - h[i]), derivative = -h[i+d]
        grad[:N-d] += -h[d:]
    # Now gradient inside shift
    # Choose i_high with max positive gradient (largest increase if increase h)
    # Choose i_low with min gradient (most negative; decreasing h would increase correlation so we may increase h ideally? Actually we want to reduce correlation: decreasing h where gradient positive reduces correlation, increasing where gradient negative reduces correlation).
    # So we want to move mass from i_high (positive gradient) to i_low (negative gradient).
    # i_high = argmax(grad)
    # i_low = argmin(grad)
    i_high = int(np.argmax(grad))
    i_low = int(np.argmin(grad))
    # If grad[i_low] >= grad[i_high] (i.e., all gradients equal or opposite sign), then no beneficial move; fallback to random.
    # Determine max transferable epsilon.
    eps_max = min(h[i_high], 1.0 - h[i_low])
    if eps_max <= 0:
        # cannot move, fallback random swap
        return None
    # Choose epsilon, not too large. Use limited max step.
    eps = min(eps_max, max_eps) * rng.random()
    if eps <= 0:
        return None
    h_new = h.copy()
    h_new[i_high] -= eps
    h_new[i_low] += eps
    # ensure bounds
    if h_new[i_high] < 0 or h_new[i_low] > 1:
        # due to rounding
        return None
    return h_new

def _guided_annealing(h_start, time_limit, rng):
    N = h_start.size
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    start = time.time()
    # Parameters:
    max_eps = 0.25 # maybe
    while time.time() - start < time_limit:
        # Determine worst shift
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        idx_max = int(np.argmax(corr))
        shift = idx_max - (N - 1)
        # Attempt gradient move
        h_new = _gradient_move(h_cur, shift, rng, max_eps)
        if h_new is None:
            # fallback: random swap as earlier
            i, j = rng.integers(0, N, size=2)
            while i == j:
                j = rng.integers(0,N)
            hi, hj = h_cur[i], h_cur[j]
            # transfer epsilon between i and j
            # compute up/down same as earlier
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up == 0 and down == 0:
                continue
            eps = rng.random() * (up + down) - down
            eps = max(-max_eps, min(max_eps, eps))
            if eps == 0:
                continue
            h_new = h_cur.copy()
            h_new[i] = hi + eps
            h_new[j] = hj - eps
        # Repair just in case (should stay within [0,1] and sum)
        # but due to floating errors sum may drift; call repair
        h_new = _repair_vector(h_new)
        new_score = _c5_value(h_new)

        # Accept with annealing schedule
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = 0.01 * frac  # temperature
        delta = new_score - cur_score
        # Accept if better or with prob e^{-delta/T}
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _binary_hill_climb(h, time_limit, rng):

    # Ensure h is binary (0/1), but might not be exactly half count if N odd.
    # With binary, sum = n1 maybe not exactly n/2; we need to adjust to meet sum constraint exactly via one fractional.
    # For simplicity, we restrict to even N.
    # Ensure N even: if odd, use fallback continuous.

        # fallback continuous search
        return _guided_annealing(h, time_limit, rng)
    # Make binary: set top N/2 entries to 1, rest 0.
    k = N // 2
    idx_sorted = np.argsort(h)[::-1]  # descending

    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()

    ones_idx = np.flatnonzero(cur_h == 1.0)
    zeros_idx = np.flatnonzero(cur_h == 0.0)
    while time.time() - start < time_limit:
        # Determine worst shift as before for guidance
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')

        # For binary values, gradient will be in -1,0,1 range maybe.
        # Choose swap pair that most decreases correlation:
        # Want i_high with high positive grad (remove 1 from there) and i_low with low (negative) grad (add 1 there)
        # So choose from ones_idx where grad high, and zeros_idx where grad low.
        # Find i_high = argmax(grad[ones_idx]), i_low = argmin(grad[zeros_idx])
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break

        grads_one = grad[ones_idx]
        grads_zero = grad[zeros_idx]
        i_high = ones_idx[int(np.argmax(grads_one))]
        i_low = zeros_idx[int(np.argmin(grads_zero))]
        # swap
        h_new = cur_h.copy()
        h_new[i_high] = 0.0
        h_new[i_low] = 1.0
        new_score = _c5_value(h_new)

                best_h = cur_h.copy()
        # optionally break if no improvement after many attempts? Not needed

import numpy as np
import time

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _gradient_move(h: np.ndarray, shift: int, rng, max_eps: float = 0.3):
    """Transfer a small amount of mass from a high‑gradient index to a low‑gradient one."""
    N = h.size
    grad = np.zeros(N, dtype=np.float64)

    if shift >= 0:
        grad[: N - shift] += (1.0 - h[shift:])
        grad[shift:] += -h[: N - shift]
    else:
        d = -shift
        grad[d:] += (1.0 - h[: N - d])
        grad[: N - d] += -h[d:]

    i_high = int(np.argmax(grad))
    i_low = int(np.argmin(grad))

    eps_max = min(h[i_high], 1.0 - h[i_low])
    if eps_max <= 0:
        return None
    eps = min(eps_max, max_eps) * (0.5 + 0.5 * rng.random())
    if eps <= 0:
        return None

    h_new = h.copy()
    h_new[i_high] -= eps
    h_new[i_low] += eps
    if not (0.0 <= h_new[i_high] <= 1.0 and 0.0 <= h_new[i_low] <= 1.0):
        return None
    return h_new

def _guided_annealing(h_start: np.ndarray, time_limit: float, rng):
    """Annealing using gradient‑directed moves, with random fallback."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    max_eps = 0.3
    T0 = 0.2

    while True:
        if time.time() - start >= time_limit:
            break

        # worst shift
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        idx_max = int(np.argmax(corr))
        shift = idx_max - (N - 1)

        # try a gradient‑guided move
        h_new = _gradient_move(h_cur, shift, rng, max_eps)
        if h_new is None:
            # random pair move
            i, j = rng.integers(0, N, size=2)
            while i == j:
                j = rng.integers(0, N)
            hi, hj = h_cur[i], h_cur[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up == 0 and down == 0:
                continue
            eps = rng.random() * (up + down) - down
            eps = max(-max_eps, min(max_eps, eps))
            if eps == 0:
                continue
            h_new = h_cur.copy()
            h_new[i] = hi + eps
            h_new[j] = hj - eps

        h_new = _repair_vector(h_new)
        new_score = _c5_value(h_new)

        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()

    return best_h, best_score

def _binary_hill_climb(h_start: np.ndarray, time_limit: float, rng):
    """Hill‑climbing on a binary (0/1) vector preserving exactly n/2 ones."""
    N = h_start.size
    if N % 2 != 0:
        # fallback to continuous annealing if size odd
        return _guided_annealing(h_start, time_limit, rng)

    # binarise by top‑half threshold
    k = N // 2
    idx = np.argsort(h_start)[::-1]
    h = np.zeros(N, dtype=np.float64)
    h[idx[:k]] = 1.0

    cur_score = _c5_value(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(np.argmax(corr))
        shift = idx_max - (N - 1)

        # gradient for this shift (same formula as in _gradient_move)
        grad = np.zeros(N, dtype=np.float64)
        if shift >= 0:
            grad[: N - shift] += (1.0 - h[shift:])
            grad[shift:] += -h[: N - shift]
        else:
            d = -shift
            grad[d:] += (1.0 - h[: N - d])
            grad[: N - d] += -h[d:]

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        # pick a 1 with largest positive gradient and a 0 with most negative gradient
        i_high = ones[int(np.argmax(grad[ones]))]
        i_low = zeros[int(np.argmin(grad[zeros]))]

        h_new = h.copy()
        h_new[i_high] = 0.0
        h_new[i_low] = 1.0
        new_score = _c5_value(h_new)

        if new_score < cur_score or rng.random() < 0.001:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Obtain parent if provided
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).copy()
        parent_n = parent.size
    except NameError:
        parent = None
        parent_n = 0

    # Base resolution (even, ≤1000)
    n_base = 200
    if n_base % 2:
        n_base += 1

    if parent is not None:
        x_old = np.linspace(0.0, 2.0, parent_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_base, endpoint=False)
        h0 = np.interp(x_new, x_old, parent)
    else:
        half = n_base // 2
        h0 = np.concatenate([np.ones(half), np.zeros(n_base - half)], dtype=np.float64)
        rng.shuffle(h0)

    h0 = _repair_vector(h0)

    start_total = time.time()
    time_left = budget_s - (time.time() - start_total)
    if time_left <= 0:
        return h0, _c5_value(h0), h0.size

    # Phase 1: guided annealing at base resolution
    phase1 = time_left * 0.45
    best_h, best_score = _guided_annealing(h0, phase1, rng)

    elapsed = time.time() - start_total
    time_left = budget_s - elapsed
    if time_left <= 5.0:
        best_h = _repair_vector(best_h)
        return best_h, _c5_value(best_h), best_h.size

    # Phase 2: upscale (if time permits)
    if time_left > budget_s * 0.25:
        n_up = min(n_base * 2, 800)
        if n_up != best_h.size:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = _repair_vector(h_up)
        else:
            h_up = best_h.copy()

        phase2 = time_left * 0.6
        h2, s2 = _guided_annealing(h_up, phase2, rng)
        if s2 < best_score:
            best_h, best_score = h2, s2
        elapsed = time.time() - start_total
        time_left = budget_s - elapsed

    # Phase 3: binary hill climbing on the current best (if enough time)
    if time_left > 5.0:
        h_bin, s_bin = _binary_hill_climb(best_h, time_left, rng)
        if s_bin < best_score:
            best_h, best_score = h_bin, s_bin

    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
