import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def hill_climb_binary(h0, best_val, start_time, budget_s):
    n = len(h0)
    dx = 2.0/n

    cur_c5 = best_val
    rng = np.random.default_rng()
    # Preliminary compute mismatched indexes?

def hill_climb_targeted(h0, start_c5, start_time, budget_s):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = start_c5
    cur_h = h0.copy()
    cur_c5 = start_c5
    rng = np.random.default_rng()
    while time.time() - start_time < budget_s - 0.1:
        # compute correlation
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        max_val = corr[max_idx] * (2.0/n)  # compute C5
        # compute mism one and candidate zero indices
        if k >= 0:
            mism_mask = (cur_h[:n-k] == 1) & (cur_h[k:] == 0)
            mism_idxs = np.flatnonzero(mism_mask)
            zeros = np.flatnonzero(cur_h == 0)
            cand_mask = (zeros >= n - k) | (cur_h[zeros + k] == 1)
            cand_idxs = zeros[cand_mask]
        else:
            kk = -k
            mism_mask = (cur_h[kk:] == 1) & (cur_h[:n-kk] == 0)
            mism_idxs = np.flatnonzero(mism_mask) + kk
            zeros = np.flatnonzero(cur_h == 0)
            cand_mask = (zeros < kk) | (cur_h[zeros - kk] == 1)
            cand_idxs = zeros[cand_mask]
        if len(mism_idxs) > 0 and len(cand_idxs) > 0:
            i = rng.choice(mism_idxs)
            j = rng.choice(cand_idxs)
            # perform swap
            cand_h = cur_h.copy()
            cand_h[i] = 0.0
            cand_h[j] = 1.0
            cand_c5 = compute_c5(cand_h)
            if cand_c5 + 1e-12 < cur_c5:
                cur_h, cur_c5 = cand_h, cand_c5
                if cand_c5 < best_c5:
                    best_h, best_c5 = cand_h.copy(), cand_c5
                continue
        # fallback: random swap
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand_h = cur_h.copy()
        cand_h[i] = 0.0
        cand_h[j] = 1.0
        cand_c5 = compute_c5(cand_h)
        # accept if improvement
        if cand_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = cand_h, cand_c5
            if cand_c5 < best_c5:
                best_h, best_c5 = cand_h.copy(), cand_c5

    return best_h, best_c5

                inc = min(diff, 1.0 - best_h[i])
                best_h[i] += inc
                diff -= inc
            elif diff < 0 and best_h[i] > 0.0:
                dec = min(-diff, best_h[i])
                best_h[i] -= dec
                diff += dec

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n

import time
import numpy as np

# ---------- Core evaluation ----------
def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


# ---------- Simple binary generator ----------
def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


# ---------- Targeted hill‑climbing on binary vectors ----------
def hill_climb_targeted(h0: np.ndarray, start_c5: float,
                       start_time: float, time_limit: float) -> tuple:
    """Swap‑based hill‑climbing that focuses on the worst shift."""
    n = h0.size
    rng = np.random.default_rng()
    best_h, best_val = h0.copy(), start_c5
    cur_h, cur_val = h0.copy(), start_c5

    while time.time() - start_time < time_limit - 0.1:
        # correlation & worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        # mismatching ones for this shift
        if k >= 0:
            mism_mask = (cur_h[:n - k] == 1) & (cur_h[k:] == 0)
            mism = np.flatnonzero(mism_mask)
            zeros = np.flatnonzero(cur_h == 0)
            cand_mask = (zeros >= n - k) | (cur_h[zeros + k] == 1)
            cand = zeros[cand_mask]
        else:
            kk = -k
            mism_mask = (cur_h[kk:] == 1) & (cur_h[:n - kk] == 0)
            mism = np.flatnonzero(mism_mask) + kk
            zeros = np.flatnonzero(cur_h == 0)
            cand_mask = (zeros < kk) | (cur_h[zeros - kk] == 1)
            cand = zeros[cand_mask]

        performed = False
        if mism.size and cand.size:
            i = rng.choice(mism)
            j = rng.choice(cand)
            trial = cur_h.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            val = compute_c5(trial)
            if val + 1e-12 < cur_val:
                cur_h, cur_val = trial, val
                performed = True
                if val < best_val:
                    best_h, best_val = trial.copy(), val

        if not performed:
            # generic random swap
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            trial = cur_h.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            val = compute_c5(trial)
            if val + 1e-12 < cur_val:
                cur_h, cur_val = trial, val
                if val < best_val:
                    best_h, best_val = trial.copy(), val

    return best_h, best_val


# ---------- Continuous gradient‑descent style refinement ----------
def continuous_refine(h0: np.ndarray, start_time: float,
                     total_budget: float, target_c5: float) -> tuple:
    """Projected gradient descent on the worst shift with a simple schedule."""
    n = h0.size
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h0.astype(float).copy()
    cur_val = compute_c5(h)
    rng = np.random.default_rng()

    while time.time() - start_time < total_budget - 0.1:
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        # gradient of the worst overlap
        grad = np.zeros_like(h)
        if k >= 0:
            grad[: n - k] += 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[kk:] += 1.0 - h[: n - kk]
            grad[: n - kk] -= h[kk:]

        # decreasing step size
        elapsed = time.time() - start_time
        remaining = total_budget - elapsed
        step = 0.5 * (remaining / total_budget)  # max step ≈0.5, shrinks to 0

        # take a step and project onto box
        h_new = h - step * grad
        np.clip(h_new, 0.0, 1.0, out=h_new)

        # enforce exact sum by fixing one coordinate if possible
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                cand = np.where(h_new < 1.0)[0]
            else:
                cand = np.where(h_new > 0.0)[0]
            if cand.size:
                i = rng.choice(cand)
                delta = min(abs(diff), (1.0 - h_new[i]) if diff > 0 else h_new[i])
                h_new[i] += np.sign(diff) * delta
                # any tiny remaining diff is ignored (will be corrected later)

        new_val = compute_c5(h_new)

        # accept if improvement, otherwise Metropolis‑style acceptance
        if new_val < cur_val - 1e-12:
            h, cur_val = h_new, new_val
        else:
            # annealing temperature proportional to remaining fraction
            T = max(1e-6, remaining / total_budget * 0.01)
            if rng.random() < np.exp(-(new_val - cur_val) / T):
                h, cur_val = h_new, new_val

        if cur_val <= target_c5:
            break

    return h, cur_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- grid size -----
    if 'initial_h_values' in globals():
        n = len(initial_h_values)          # keep the supplied resolution
        init_cont = np.array(initial_h_values, dtype=float)
        # binary start from the supplied continuous vector
        binary_start = np.zeros(n, dtype=float)
        binary_start[np.argsort(-init_cont)[: n // 2]] = 1.0
        best_h = binary_start
        best_c5 = compute_c5(best_h)
    else:
        n = 200                               # a modestly larger resolution
        best_h = random_binary(n, rng)
        best_c5 = compute_c5(best_h)

    target_c5 = 0.3808                       # we aim to be below this
    half_budget = budget_s * 0.7             # most time for discrete search

    # ----- discrete hill‑climbing with many random restarts -----
    while time.time() - start_time < half_budget - 2.0:
        h_start = random_binary(n, rng)
        c5_start = compute_c5(h_start)
        remaining = budget_s - (time.time() - start_time)
        h_opt, c5_opt = hill_climb_targeted(h_start, c5_start,
                                            start_time, remaining)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if best_c5 <= target_c5:
            break

    # ----- continuous refinement (projected gradient descent) -----
    elapsed = time.time() - start_time
    if elapsed < budget_s - 2.0 and best_c5 > target_c5:
        remaining = budget_s - elapsed
        h_ref, c5_ref = continuous_refine(best_h, start_time,
                                          remaining, target_c5)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    # ----- final projection onto feasible set -----
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        diff = target_sum - cur_sum
        # distribute diff greedily without breaking box constraints
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and best_h[i] < 1.0:
                inc = min(diff, 1.0 - best_h[i])
                best_h[i] += inc
                diff -= inc
            elif diff < 0 and best_h[i] > 0.0:
                dec = min(-diff, best_h[i])
                best_h[i] -= dec
                diff += dec

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
