def project_bounds_sum(x, target):
    # Project onto box [0,1] and sum constraint.
    # Use algorithm: clip to [0,1]; then adjust until sum close.
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Determine indices that are not at bounds.
    free = np.where((x > 0) & (x < 1))[0]
    # If no free variables (unlikely), distribute diff equally and clip again
    if free.size == 0:
        # Add diff uniformly then clip again
        x += diff / len(x)
        np.clip(x, 0, 1, out=x)
        # recursion maybe
        return project_bounds_sum(x, target)
    # distribute diff equally among free variables
    inc = diff / free.size
    # ensure not exceeding bounds
    # add inc, then clip again, then call recursively
    x[free] += inc
    # clip (due to possible overflow)
    np.clip(x, 0, 1, out=x)
    # recursion if still not satisfied
    if abs(target - x.sum()) > 1e-9:
        # recursion but limit depth
        return project_bounds_sum(x, target)
    return x

        diff = target - x.sum()

            break
        # Determine free indices

        if diff > 0:

        else:
            movable = free_down

            # can't adjust due to bounds, break (should not happen)
            break
        delta = diff / movable.size
        # ensure not overshoot bounds
        if diff > 0:
            max_inc = np.min(1.0 - x[movable])
            delta = min(delta, max_inc)
        else:

            delta = max(delta, -max_dec)  # delta negative

        # clamp

            for i in range(x.size):
                if 0 <= x[i] + diff <= 1:

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust array h (in-place) so that its sum equals target while respecting 0≤h_i≤1.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def project_bounds_sum(x: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Project onto the box [0,1] and the affine hyperplane sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            movable = np.where(x < 1.0)[0]
        else:
            movable = np.where(x > 0.0)[0]
        if movable.size == 0:
            break
        delta = diff / movable.size
        if diff > 0:
            # do not exceed upper bound
            max_inc = np.min(1.0 - x[movable])
            delta = min(delta, max_inc)
        else:
            # do not go below lower bound
            max_dec = np.min(x[movable])
            delta = max(delta, -max_dec)
        x[movable] += delta
        np.clip(x, 0.0, 1.0, out=x)
    # tiny residual correction (randomized but tiny)
    residual = target - x.sum()
    if abs(residual) > 1e-9:
        x += residual / x.size
        np.clip(x, 0.0, 1.0, out=x)
        # final fix if still off (unlikely)
        residual = target - x.sum()
        if abs(residual) > 1e-9:
            for i in range(x.size):
                if 0.0 <= x[i] + residual <= 1.0:
                    x[i] += residual
                    break
    return x

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimize a step function h to minimize the Erdős overlap constant C5."""
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # ----- initialise h -----
    try:
        init = np.array(initial_h_values, dtype=float)
        base_n = init.size
        n_points = 200  # target resolution (still modest)
        # linear interpolation to new grid
        x_old = np.linspace(0.0, 2.0, base_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_old, init)
    except Exception:
        n_points = 200
        h = rng.rand(n_points)

    target_sum = n_points / 2.0
    h = project_bounds_sum(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # Annealing / schedule parameters
    T0, T_end = 0.02, 0.0001
    alpha0, alpha_max = 0.5, 15.0
    lr0, lr_end = 0.20, 0.001

    # counters for stagnation handling
    stagnation = 0
    max_stagnation = 2000

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time

        # schedules
        T = T0 * (1 - progress) + T_end * progress
        alpha = alpha0 * (1 - progress) + alpha_max * progress
        lr = lr0 * (1 - progress) + lr_end * progress

        # ---- compute softmax weights for current h ----
        corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
        f = corr_full  # unscaled; scaling cancels in softmax
        max_f = np.max(f)
        w = np.exp(alpha * (f - max_f))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # ---- gradient via convolutions ----
        # w length = 2*n-1, midpoint index = n-1
        mid = w.size // 2
        conv_w = np.convolve(w, current_h, mode="full")
        conv_rev = np.convolve(w[::-1], current_h, mode="full")
        slice_w = conv_w[mid:mid + n_points]
        slice_rev = conv_rev[mid:mid + n_points]
        grad = 1.0 - (slice_w + slice_rev)

        # ---- propose candidate step ----
        cand = current_h - lr * grad

        # occasional random pairwise tweak for diversity
        if rng.rand() < 0.15:
            i, j = rng.randint(n_points, size=2)
            if i != j:
                hi, hj = cand[i], cand[j]
                # feasible delta s.t. both stay in [0,1]
                low = max(-hi, hj - 1.0)
                high = min(1.0 - hi, hj)
                if low < high:
                    delta = (rng.rand() * 2.0 - 1.0) * lr * 0.5
                    delta = np.clip(delta, low, high)
                    cand[i] = hi + delta
                    cand[j] = hj - delta

        cand = project_bounds_sum(cand, target_sum, rng)

        cand_c5 = compute_c5(cand)

        # ---- acceptance (simulated annealing style) ----
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = cand
            current_c5 = cand_c5
            stagnation = 0
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
        else:
            stagnation += 1

        # if stuck for a while, inject a random perturbation around best
        if stagnation >= max_stagnation:
            pert = best_h + rng.normal(scale=0.02, size=n_points)
            pert = project_bounds_sum(pert, target_sum, rng)
            current_h = pert
            current_c5 = compute_c5(current_h)
            stagnation = 0

    # final sanity projection
    best_h = project_bounds_sum(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
