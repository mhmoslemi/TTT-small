```python
def anneal_binary_vector(h0, time_limit, rng):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # initial temperature schedule: start T as maybe 0.005 * (max - min) or something
    # Since C5 in [0,2], but typical values near 0.4. We'll use T0 = 0.02
    T0 = 0.02
    # cooling schedule: exponential decay: T = T0 * 0.995 ** (iteration)
    it = 0
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    while time.time() - start < time_limit:
        # choose random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i], h[j] = h[j], h[i]  # swap values
        # update ones/zeros arrays
        ones[i_idx] = j
        zeros[j_idx] = i
        # evaluate
        new_score = compute_c5(h)
        delta = new_score - cur_score
        # temperature
        T = T0 * (0.995 ** it)
        it += 1
        if delta < 0 or rng.random() < math.exp(-delta / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # reject: swap back
            h[i], h[j] = h[j], h[i]
            # revert ones/zeros
            ones[i_idx] = i
            zeros[j_idx] = j
        # optional: maybe occasionally reheat or random restart but not needed
    return best_h, best_score
```

def projected_gradient_descent_smooth(h0, n_iter=2000, step0=0.5, beta=10.0):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()

    for it in range(1, n_iter + 1):

        eta = step0 / math.sqrt(it)

        h_new = project_onto_box_sum(h_new, s)

            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def gradient_softmax_max(h, beta):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # compute stable softmax weights
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    w = exps / np.sum(exps)
    grad = np.zeros_like(h)
    # shift index range from -(n-1) to (n-1)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        shift = idx - (n - 1)
        grad += weight * compute_gradient(h, shift)
    return grad

```python
import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    # clip extreme
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds (optional)
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def gradient_softmax_max(h, beta):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    w = exps / np.sum(exps)
    grad = np.zeros_like(h)
    # shifts from -(n-1) to n-1
    base = n - 1
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        shift = idx - base
        grad += weight * compute_gradient(h, shift)
    return grad

def projected_gradient_descent_smooth(h0, n_iter=2000, step0=0.5, beta=10.0):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        grad = gradient_softmax_max(h, beta)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_binary_vector(h0, time_limit, rng):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    n = h.size
    # maintain index arrays for ones and zeros
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    it = 0
    T0 = 0.02
    while time.time() - start < time_limit:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i], h[j] = h[j], h[i]
        # update index lists
        ones[i_idx] = j
        zeros[j_idx] = i
        new_score = compute_c5(h)
        delta = new_score - cur_score
        T = T0 * (0.995 ** it)
        it += 1
        if delta < 0 or rng.random() < math.exp(-delta / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # reject: swap back
            h[i], h[j] = h[j], h[i]
            ones[i_idx] = i
            zeros[j_idx] = j
    return best_h, best_score

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

    h[idx] = 1.0
    return h

import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Subgradient of the correlation term for a given integer shift."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
