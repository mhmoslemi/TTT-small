def project_sum_bounds(h, target):
    # ensure 0 <= h <= 1 and sum = target
    # clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    # Compute diff
    diff = target - h.sum()
    # tolerance
    eps = 1e-12
    if abs(diff) < eps:
        return h
    # iterative distribution
    while abs(diff) > eps:
        if diff > 0:
            # increase entries < 1
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            # total capacity
            cap = (1.0 - h[mask]).sum()
            if cap <= diff + 1e-12:  # fill to 1
                h[mask] = 1.0
                diff = target - h.sum()
                continue
            # distribute equally among mask
            inc = diff / mask.sum()
            # ensure inc does not exceed remaining capacity
            inc = min(inc, (1.0 - h[mask]).max())
            h[mask] += inc
            diff = target - h.sum()
        else:
            # decrease entries > 0
            mask = h > 1e-12
            if not np.any(mask):
                break
            cap = h[mask].sum()
            if cap <= -diff + 1e-12:
                h[mask] = 0.0
                diff = target - h.sum()
                continue
            dec = diff / mask.sum()  # diff negative, dec negative

            # compute max possible decrease per element = h[mask]
            # We want to subtract -dec from each
            # Actually dec = diff/num, negative; we can compute dec_clipped = max(dec, -h[mask].min())
            # Simpler: compute dec = diff / mask.sum(); if any h[mask] + dec < 0, adjust

def compute_softmax_gradient(h, alpha):
    n = h.shape[0]
    one_minus_h = 1.0 - h
    # compute f across shifts
    f = np.empty(2*n-1, dtype=float)
    # we can compute using np.convolve or correlation; but correlation gives f all in one line, but we want raw sum without dx.
    f = np.correlate(h, one_minus_h, mode='full')
    # Because np.correlate does sum_{i} h[i] * v[t-i]; but v=1-h. That yields same as our compute_f_all; okay.
    # Now for numeric stability compute softmax weights
    max_f = f.max()
    exp_f = np.exp(alpha * (f - max_f))
    w = exp_f / exp_f.sum()
    # compute gradient
    g = np.zeros_like(h)
    # iterate over shifts: index in f corresponds to shift = idx - (n-1)
    for idx in range(2*n-1):
        wk = w[idx]
        if wk == 0.0:
            continue
        shift = idx - (n-1)
        if shift >= 0:
            # i in [0, n-shift-1] -> add (1-h[i+shift]) to g[i]
            if shift == 0:
                # add (1-h[i]) to all i for first part, and subtract h[i] from all i for second part (since shift=0)
                g += wk * one_minus_h
                g -= wk * h
                # This results in g_i = wk * ((1 - h_i) - h_i) = wk * (1 - 2 h_i)
                # but we can combine.
            else:
                i_end = n - shift
                # part1
                g[:i_end] += wk * one_minus_h[shift:]
                # part2: subtract h at index i+shift? Actually need subtract h[i] from g[i+shift]
                g[shift:] -= wk * h[:i_end]
        else:
            p = -shift
            # i from p to n-1
            g[p:] += wk * one_minus_h[:n-p]
            g[:n-p] -= wk * h[p:]
    return g

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_sum_bounds(h, target):
    # Clip within [0,1]
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    eps = 1e-12
    # Tolerant iterative distribution
    while abs(diff) > eps:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                # cannot increase, break
                break
            capacity = (1.0 - h[mask]).sum()
            if capacity <= diff + eps:
                h[mask] = 1.0
                diff = target - h.sum()
                continue
            inc = diff / mask.sum()
            # inc may exceed remaining capacity for some entries; we clip after addition
            h[mask] += inc
            # Clip to <=1
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            capacity = h[mask].sum()
            if capacity <= -diff + eps:
                h[mask] = 0.0
                diff = target - h.sum()
                continue
            dec = diff / mask.sum()  # diff negative, dec negative
            h[mask] += dec
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
    return h

def worst_shift_gradient(h):
    n = h.shape[0]
    one_minus_h = 1.0 - h
    # correlation with one_minus_h
    corr = np.correlate(h, one_minus_h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (n - 1)
    g = np.zeros_like(h)
    if shift >= 0:
        i_end = n - shift
        if i_end > 0:
            g[:i_end] += one_minus_h[shift:]
            g[shift:] -= h[:i_end]
    else:
        p = -shift
        if p < n:
            g[p:] += one_minus_h[:n-p]
            g[:n-p] -= h[p:]
    return g

    start = time.time()

        target_sum = n / 2.0
        h = rng.rand(n)
        h = project_sum_bounds(h, target_sum)
    # Ensure shape
    h = h.astype(float)
    # Track best
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5
    # Parameters
    init_step = 0.5
    step_min = 1e-5
    step = init_step
    no_improve_iters = 0
    # Timing
    continuous_end = start + 0.75 * max_time
    while True:
        now = time.time()
        if now >= max_time:
            break
        if now >= continuous_end:
            # exit continuous phase
            break
        # Compute subgradient for worst shift
        g = worst_shift_gradient(current_h)
        # Update
        candidate = current_h - step * g
        candidate = np.clip(candidate, 0.0, 1.0)
        candidate = project_sum_bounds(candidate, target_sum)
        cand_c5 = compute_c5(candidate)
        # Accept if better or with small probability
        if cand_c5 < current_c5:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                no_improve_iters = 0
            else:
                no_improve_iters += 1
        else:
            # probabilistic acceptance (simulated annealing style)
            # temperature decreasing
            progress = (now - start) / max_time
            T = max(1e-3 * (1 - progress), 1e-6)
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            if rng.rand() < prob:
                current_h = candidate
                current_c5 = cand_c5
            no_improve_iters += 1

            current_h = candidate

            no_improve_iters = 0
            step = init_step

    idx_sorted = np.argsort(-best_h)  # descending
    binary_h = np.zeros(n, dtype=float)

    binary_c5 = compute_c5(binary_h)

    # Pre-compute indices of 1's and 0's
    ones_idx = np.where(binary_h > 0.5)[0]
    zeros_idx = np.where(binary_h < 0.5)[0]
    # Ensure we have correct counts
    # Local search loop
    while True:
        now = time.time()
        if now >= max_time:
            break
        # Randomly choose a 1 and a 0 to swap
        if len(ones_idx) == 0 or len(zeros_idx) == 0:

        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        binary_h[i], binary_h[j] = 0.0, 1.0
        new_c5 = compute_c5(binary_h)
        if new_c5 < binary_c5:
            binary_c5 = new_c5
            # update index sets
            ones_idx = np.where(binary_h > 0.5)[0]
            zeros_idx = np.where(binary_h < 0.5)[0]
            # update best if better than best_h
            if binary_c5 < best_c5:
                best_c5 = binary_c5
                best_h = binary_h.copy()
        else:
            # revert swap
            binary_h[i], binary_h[j] = 1.0, 0.0
            # optionally accept with small probability to escape
            # Use small temperature
            progress = (now - start) / max_time
            T = max(1e-4 * (1 - progress), 1e-6)
            if rng.rand() < np.exp(-(new_c5 - binary_c5) / T):
                binary_h[i], binary_h[j] = 0.0, 1.0
                binary_c5 = new_c5
                ones_idx = np.where(binary_h > 0.5)[0]
                zeros_idx = np.where(binary_h < 0.5)[0]
        # Continue loop
    # finalize best solution
    best_h = project_sum_bounds(best_h, target_sum)  # ensure exact sum
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def project_sum_bounds(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    eps = 1e-12
    diff = target - h.sum()
    # simple water-filling loop, converges quickly for moderate sizes
    while abs(diff) > eps:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            cap = (1.0 - h[mask]).sum()
            if cap <= diff + eps:
                h[mask] = 1.0
                diff = target - h.sum()
                continue
            inc = diff / mask.sum()
            h[mask] += inc
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            cap = h[mask].sum()
            if cap <= -diff + eps:
                h[mask] = 0.0
                diff = target - h.sum()
                continue
            dec = diff / mask.sum()   # diff is negative, so dec negative
            h[mask] += dec
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
    return h


def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    """Subgradient of the maximal overlap (worst shift)."""
    n = h.shape[0]
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    idx_max = np.argmax(corr)
    shift = idx_max - (n - 1)          # integer lag, can be negative
    g = np.zeros_like(h)
    if shift >= 0:
        i_end = n - shift
        if i_end > 0:
            g[:i_end] += one_minus_h[shift:]
            g[shift:] -= h[:i_end]
    else:
        p = -shift
        i_end = n - p
        if i_end > 0:
            g[p:] += one_minus_h[:i_end]
            g[:i_end] -= h[p:]
    return g

    start = time.time()
    max_time = budget_s - 0.2                   # safety margin

    # ----- initialise h -----
    try:

    except Exception:

        target = n / 2.0
        h = rng.rand(n)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # ----- continuous (subgradient) optimisation -----
    init_step = 0.5
    step_min = 1e-5
    step = init_step
    no_improve = 0
    continuous_cut = start + 0.75 * max_time

    while True:
        now = time.time()
        if now >= max_time or now >= continuous_cut:
            break

        # gradient of the worst shift
        g = worst_shift_gradient(current_h)

        # generate two candidate step sizes
        cand1 = project_sum_bounds(np.clip(current_h - step * g, 0.0, 1.0), target)
        cand2 = project_sum_bounds(np.clip(current_h - 0.5 * step * g, 0.0, 1.0), target)

        c1 = compute_c5(cand1)
        c2 = compute_c5(cand2)

        # pick the best among current, cand1, cand2
        if c1 < current_c5 and c1 <= c2:
            chosen_h, chosen_c5 = cand1, c1
        elif c2 < current_c5:
            chosen_h, chosen_c5 = cand2, c2
        else:
            chosen_h, chosen_c5 = current_h, current_c5

        # acceptance
        if chosen_c5 < current_c5:
            current_h, current_c5 = chosen_h, chosen_c5
            no_improve = 0
        else:
            # occasional Metropolis acceptance
            progress = (now - start) / max_time
            T = max(1e-3 * (1.0 - progress), 1e-6)

            if rng.rand() < prob:

            no_improve += 1

        # record best
        if current_c5 < best_c5:

        cand = binary_h.copy()
        cand[i] = 0.0

            if binary_c5 < best_c5:

                best_h = binary_h.copy()
        else:

            prog = (now - start) / max_time
            T = max(1e-4 * (1.0 - prog), 1e-6)
            if rng.rand() < np.exp(-(cand_c5 - binary_c5) / T):
                binary_h, binary_c5 = cand, cand_c5

    # final sanity projection
    best_h = project_sum_bounds(best_h, target)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n)
