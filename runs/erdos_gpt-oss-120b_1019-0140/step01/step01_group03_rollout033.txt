def compute_subgradient(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    # tolerance relative to max
    tol = 1e-8
    active_idxs = np.where(corr >= max_val - tol)[0]
    grad = np.zeros_like(h)
    n = h.size
    for idx in active_idxs:
        shift = idx - (n - 1)
        grad += compute_gradient(h, shift)
    grad /= len(active_idxs)
    return grad
```

def apgd(h0, max_iter, step0):
    h = h0.copy()
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter+1):
        grad = compute_subgradient(y)
        eta = step0 / np.sqrt(it)   # decaying
        h_new = y - eta * grad
        # project
        h_new = project_onto_box_sum(h_new, h_new.size / 2.0)
        # update t and y (Nesterov)
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = h_new + ((t - 1)/t_new)*(h_new - h)
        t = t_new
        h = h_new
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # optional early break if no improvement for many it
    return best_h, best_score
```

import time
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10.0
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10.0

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_gradient(h, shift):
    """Gradient of f_k = ∫h(1−h_shift) dx w.r.t. h for a given integer shift."""
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def compute_c5(h):
    """Exact C5 for discrete h (dx = 2/n)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_subgradient(h, top_k=3):
    """Average gradient of the top_k largest overlap terms."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    # indices of the top_k values (may contain duplicates if n small)
    if top_k >= corr.size:
        idxs = np.arange(corr.size)
    else:
        idxs = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)
        grad += compute_gradient(h, shift)
    grad /= float(len(idxs))
    return grad

def apgd(h0, max_iter=5000, step0=0.7, top_k=3):
    """Accelerated projected gradient descent with subgradient averaging."""
    h = h0.copy()
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    target_sum = n / 2.0
    for it in range(1, max_iter + 1):
        grad = compute_subgradient(y, top_k=top_k)
        eta = step0 / np.sqrt(it)
        h_new = y - eta * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) * 0.5
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        t = t_new
        h = h_new
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1 as float)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h
