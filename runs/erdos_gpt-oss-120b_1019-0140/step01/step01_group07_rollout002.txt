import numpy as np
import time
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust sum to target by water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-14:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-14:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx via FFT."""
    n = h.size
    dx = 2.0 / n
    nfft = 2 * n
    H = np.fft.fft(h, nfft)
    G = np.fft.fft(1.0 - h, nfft)
    conv = np.fft.ifft(H * np.conj(G)).real
    corr = conv[: 2 * n - 1]
    return np.max(corr) * dx

def _de_optimize(h_start, target_sum, global_end):
    """Differential Evolution with projection onto simplex constraints."""
    n = h_start.size
    popsize = max(20, int(30))
    F = 0.6
    CR = 0.8
    # initialise population
    pop = [h_start.copy()]
    for _ in range(popsize - 1):
        v = np.random.rand(n)
        pop.append(_project_to_constraints(v, target_sum))
    pop = np.array(pop)
    scores = np.array([_compute_c5(ind) for ind in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    generation = 0
    while time.time() < global_end and generation < 2000:
        generation += 1
        for i in range(popsize):
            # mutation
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = np.random.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = _project_to_constraints(mutant, target_sum)
            # crossover
            jrand = np.random.randint(n)
            trial = np.empty(n)
            rand_vals = np.random.rand(n)
            mask = (rand_vals < CR) | (np.arange(n) == jrand)
            trial[mask] = mutant[mask]
            trial[~mask] = pop[i, ~mask]
            trial = np.clip(trial, 0.0, 1.0)
            trial = _project_to_constraints(trial, target_sum)
            # selection
            trial_score = _compute_c5(trial)
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
                    if best_score < 0.3808:
                        return best_h, best_score
        # optional early stop if time is nearly exhausted
        if time.time() + 5.0 >= global_end:
            break
    return best_h, best_score

def _local_refine(h0, target_sum):
    """SLSQP local refinement."""
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * h0.size
    try:
        res = minimize(_compute_c5, h0, method="SLSQP",
                       bounds=bounds, constraints=[cons],
                       options={"maxiter": 200, "ftol": 1e-9, "disp": False})
        h_opt = _project_to_constraints(res.x, target_sum)
        return h_opt, _compute_c5(h_opt)
    except Exception:
        return h0, _compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    global_end = start_time + budget_s * 0.98

    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        init_vec = np.array(init_vec, dtype=np.float64).ravel()
        n0 = init_vec.size
    else:
        n0 = 200
        init_vec = np.random.rand(n0)

    n_vals = [n0]
    if budget_s >= 300:
        n_vals.append(n0 * 2)
    if budget_s >= 600:
        n_vals.append(n0 * 4)

    best_h = None
    best_score = float("inf")
    best_n = None

    for n in n_vals:
        if time.time() >= global_end:
            break
        target_sum = n / 2.0
        # interpolate previous best or initial vector
        if best_h is not None:
            src = best_h
        else:
            src = init_vec
        x_src = np.arange(src.size) * (2.0 / src.size)
        x_new = np.arange(n) * (2.0 / n)
        h_start = np.interp(x_new, x_src, src)
        h_start = _project_to_constraints(h_start, target_sum)

        # DE phase
        h_de, score_de = _de_optimize(h_start, target_sum, global_end)
        # Local refinement
        h_ref, score_ref = _local_refine(h_de, target_sum)

        # pick better of the two
        if score_ref < score_de:
            cur_h, cur_score = h_ref, score_ref
        else:
            cur_h, cur_score = h_de, score_de

        if cur_score < best_score:
            best_score = float(cur_score)
            best_h = cur_h.copy()
            best_n = n
            if best_score < 0.3808:
                break

    if best_h is None:
        # fallback to any feasible vector
        best_h = _project_to_constraints(init_vec[:n0], n0 / 2.0)
        best_score = _compute_c5(best_h)
        best_n = best_h.size

    # final evaluation via provided evaluator if available
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_n)
