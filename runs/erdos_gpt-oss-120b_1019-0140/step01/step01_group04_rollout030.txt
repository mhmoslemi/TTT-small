def _simulated_annealing(start_v, time_limit):
    best_v = start_v.copy()
    best_score = _compute_c5(best_v)
    current_v = start_v.copy()
    current_score = best_score
    start = time.time()
    T0 = 0.005
    Tf = 1e-6
    max_move_initial = 0.5  # max delta initial; maybe adjust.
    # We'll adapt max_move based on temperature? Use same T.
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        fraction = elapsed / time_limit
        # Temperature schedule
        T = T0 * (Tf / T0) ** fraction  # exponential decay

        # Determine max_move: maybe linearly decrease: max_move = 0.5 * (1 - fraction) + 0.01 * fraction
        max_move = 0.5 * (1 - fraction) + 0.01 * fraction

        # propose move
        i, j, delta = _random_exchange(current_v, max_move)
        if delta is None:
            continue
        cand_v = current_v.copy()
        cand_v[i] += delta
        cand_v[j] -= delta

        # compute score
        cand_score = _compute_c5(cand_v)

        # acceptance
        if cand_score < current_score or np.random.rand() < np.exp(-(cand_score - current_score) / (T + 1e-12)):
            current_v = cand_v
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_v = cand_v.copy()
    return best_v, best_score

def _repair_vector(v, n):
    """Project onto [0,1]^n with exact sum n/2."""
