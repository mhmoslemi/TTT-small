import time
import math
import numpy as np

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """
    Adjust one entry of h (in‑place) to enforce exact sum == target.
    """
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-15:
        return h
    # pick an index that can move in required direction
    if diff > 0:
        cand = np.where(h < 1.0 - 1e-15)[0]
    else:
        cand = np.where(h > 1e-15)[0]
    if cand.size == 0:
        # fallback: distribute uniformly (should be rare)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = int(cand[0])
    new_val = np.clip(h[i] + diff, 0.0, 1.0)
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _compute_c5(h: np.ndarray) -> float:
    """
    Compute C5 = max_k ∫ h(x)*(1‑h(x+k))dx using FFT for speed.
    The integral is discretised: dx = 2/n, n = len(h).
    """
    n = h.shape[0]
    L = 2 * n
    # correlation = convolution of reversed h with (1‑h)
    ha_rev = np.fft.fft(h[::-1], n=L)
    hb = np.fft.fft(1.0 - h, n=L)
    conv = np.fft.ifft(ha_rev * hb).real
    corr = conv[: 2 * n - 1]
    return float(np.max(corr) * (2.0 / n))

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ----- choose discretisation size ---------------------------------
    n = 500               # even, gives finer resolution while still fast
    target_sum = n / 2.0  # exact integral must be 1

    # ----- initialise a feasible candidate -----------------------------
    try:
        base = np.asarray(initial_h_values, dtype=float)
        repeats = int(np.ceil(n / base.size))
        h = np.tile(base, repeats)[:n].astype(float)
        np.clip(h, 0.0, 1.0, out=h)
        # randomise order to avoid pathological patterns
        rng.shuffle(h)
    except NameError:
        # no parent construction – start with half‑ones pattern
        h = np.zeros(n, dtype=float)
        half = n // 2
        h[:half] = 1.0
        rng.shuffle(h)

    # enforce exact sum constraint
    h = _repair_sum(h, target_sum)

    # ----- bookkeeping -------------------------------------------------
    current = h.copy()
    current_c5 = _compute_c5(current)
    best_h = current.copy()
    best_c5 = current_c5

    # annealing schedule parameters
    max_step = 0.5   # maximal delta early on
    min_step = 0.001 # final precision
    max_temp = 0.02
    min_temp = 1e-6

    # ----- optimisation loop -------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start
        frac = elapsed / budget_s

        # time‑dependent step size and temperature (linear decay)
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        hi = current[i]
        hj = current[j]

        # feasible delta interval keeping both entries in [0,1]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)

        # intersect with step magnitude limits
        low = max(delta_min, -step)
        high = min(delta_max, step)
        if low >= high:
            continue

        # propose a random delta within the feasible interval
        delta = rng.random() * (high - low) + low

        # apply tentative move
        current[i] = hi + delta
        current[j] = hj - delta

        # evaluate
        new_c5 = _compute_c5(current)
        delta_obj = new_c5 - current_c5

        # Metropolis acceptance
        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / temp):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current.copy()
                # early exit if we already beat the target
                if best_c5 <= 0.3808:
                    break
        else:
            # revert move
            current[i] = hi
            current[j] = hj

    # final repair to guarantee exact sum (tiny drift possible)
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
