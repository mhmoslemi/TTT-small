def compute_c5_and_lag(h):
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    lag = max_idx - (len(h)-1)
    return max_val * (2.0/len(h)), lag
```

import time
import numpy as np

def _compute_c5_and_lag(h):
    """Return C5 value and the lag that attains the maximum overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    lag = max_idx - (n - 1)
    return max_val * (2.0 / n), lag

def compute_c5(h):
    """C5 value for a sampled step function."""
    val, _ = _compute_c5_and_lag(h)
    return val

def _hill_climb_binary(h0, rng, start_time, time_budget):
    """Simple swap‑based hill climbing for binary vectors."""
    h = h0.copy()
    best_val, _ = _compute_c5_and_lag(h)
    n = h.shape[0]
    max_iters = 200  # hard limit per restart
    for _ in range(max_iters):
        if time.time() - start_time > time_budget * 0.98:
            break
        val, lag = _compute_c5_and_lag(h)

        # region where the lag overlaps
        if lag >= 0:
            i_start, i_end = 0, n - lag
        else:
            i_start, i_end = -lag, n
        idx = np.arange(i_start, i_end, dtype=int)

        # try to find a beneficial pair of swaps
        if idx.size:
            partner = idx + lag
            hi = h[idx]
            hp = h[partner]
            ones_mismatch = idx[(hi > 0.5) & (hp < 0.5)]
            zeros_mismatch = idx[(hi < 0.5) & (hp > 0.5)]
            if ones_mismatch.size and zeros_mismatch.size:
                i_swap = rng.choice(ones_mismatch)
                j_swap = rng.choice(zeros_mismatch)
                h_new = h.copy()
                h_new[i_swap] = 0.0
                h_new[j_swap] = 1.0
            else:
                # random 1↔0 swap
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i_swap = rng.choice(ones)
                j_swap = rng.choice(zeros)

                        flip = rng.choice(ones, size=-diff, replace=False)
                        h0[flip] = 0.0
            else:
                idx = rng.choice(n, size=n // 2, replace=False)
                h0 = np.zeros(n, dtype=float)
                h0[idx] = 1.0

            # hill‑climb from this start
            h_opt, val_opt = _hill_climb_binary(
                h0, rng, start_time, budget_s
            )

            if val_opt < best_val:
                best_val = val_opt
                best_h = h_opt.copy()
                best_n = n

            # small early‑exit shortcut if we already beat the target
            if best_val <= 0.3808:
                # continue a few more restarts to possibly improve further
                pass

    # fallback in the unlikely event that nothing was found
    if best_h is None:
        n = candidate_ns[0]
        h = np.zeros(n, dtype=float)
        h[::2] = 1.0
        best_h = h
        best_n = n
        best_val = compute_c5(best_h)

    # final safety checks / projection (binary already feasible)
    best_h = np.clip(best_h, 0.0, 1.0)
    # ensure the integral constraint holds exactly
    cur_sum = best_h.sum()
    target_sum = best_n / 2.0
    if abs(cur_sum - target_sum) > 1e-9:
        diff = target_sum - cur_sum
        # adjust the first element while staying within [0,1]
        best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    best_val = compute_c5(best_h)
    return best_h, best_val, best_n
