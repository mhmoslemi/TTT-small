def compute_c5(v):
    n = v.shape[0]
    w = 1.0 - v
    # Do correlation v with w
    # correlation = ifft( fft(v) * conj(fft(w)) )? No, for correlation we need fft(v) * conj(fft(w)) perhaps.
    # Actually numpy's correlate without reversal uses correlation as sum_i v[i] * w[i+k].
    # In frequency domain, correlation of a and b = ifft(fft(a) * conj(fft(b))) (like cross-correlation).
    # So compute:
    Va = np.fft.fft(v)
    Vb = np.fft.fft(w)
    corr = np.fft.ifft(Va * np.conj(Vb)).real
    # The result length is n (circular correlation). But we need linear correlation (full mode) of length 2n-1.
    # The linear correlation can be obtained by zero-padding arrays to length 2n.
    L = 2 * n
    Va = np.fft.fft(v, n=L)
    Vb = np.fft.fft(w, n=L)
    corr_full = np.fft.ifft(Va * np.conj(Vb)).real
    # This gives circular correlation length L; but since we padded zeros, the linear correlation at lags -(n-1) .. (n-1) appears in the first 2n-1 entries.
    # Usually we take corr_full[:2*n-1]
    corr = corr_full[:2*n-1]
    return np.max(corr) * (2.0 / n)

def repair(v):
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    # Parameter: while diff > tolerance adjust free entries
    for _ in range(10):
        s = v.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        free = (v > 0.0) & (v < 1.0) # entries not at bound
        n_free = free.sum()
        if n_free == 0:
            # if all at bounds, cannot adjust exactly; do scaling globally
            v = np.clip(v * (target / s), 0.0, 1.0)
            continue
        adjust = diff / n_free
        v[free] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    s = v.sum()
    if abs(s - target) > 1e-9:
        v += (target - s) / v.size
        v = np.clip(v, 0.0, 1.0)
    return v
```

def anneal(v, time_limit, start_T=0.01, step_max=0.1, cooling=0.9995):
    start = time.time()
    n = len(v)
    best_v = v.copy()
    best_score = compute_c5(best_v)
    curr_v = v.copy()
    curr_score = best_score
    T = start_T
    step = step_max
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2 * step
        cand = curr_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = repair(cand)
        cand_score = compute_c5(cand)
        # Accept if better or with probability
        if cand_score < curr_score or np.random.rand() < np.exp(-(cand_score - curr_score) / max(T, 1e-12)):
            curr_v = cand
            curr_score = cand_score
        # Update best
        if cand_score < best_score:
            best_v = cand.copy()
            best_score = cand_score
        # decay
        T *= cooling
        # optionally adjust step based on T?
    return best_v, best_score

def compute_c5(v):
    n = v.shape[0]
    w = 1.0 - v
    L = 2 * n
    # FFT of v and reversed w
    V = np.fft.fft(v, L)
    W = np.fft.fft(w[::-1], L)
    conv = np.fft.ifft(V * W).real[: 2 * n - 1]
    return float(np.max(conv) * (2.0 / n))

def repair(v):

    n = v.shape[0]
    target = n / 2.0

import time
import numpy as np

# ---------- feasibility utilities ----------
def _repair(v):
    """Project v onto the feasible set: 0≤v≤1 and sum(v)=len(v)/2."""
    v = np.clip(v, 0.0, 1.0)
    n = v.shape[0]
    target = n / 2.0
    for _ in range(20):
        s = v.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        n_free = int(free.sum())
        if n_free == 0:
            # nothing free, scale globally
            scale = target / s if s != 0 else 0.0
            v = np.clip(v * scale, 0.0, 1.0)
            continue
        adjust = diff / n_free
        v[free] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny fix
    s = v.sum()
    if abs(s - target) > 1e-9:
        v += (target - s) / n
        v = np.clip(v, 0.0, 1.0)
    return v

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair(v)

# ---------- objective ----------
def _compute_c5(v):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx for a discretised vector."""
    n = v.shape[0]
    w = 1.0 - v
    L = 2 * n
    Va = np.fft.fft(v, L)
    Vb = np.fft.fft(w[::-1], L)
    conv = np.fft.ifft(Va * Vb).real[: 2 * n - 1]
    return float(np.max(conv) * (2.0 / n))

# ---------- up‑sampling ----------
def _upsample(v, n_target):
    """Linear interpolation from len(v) to n_target points, then repair."""
    n_src = v.shape[0]
    if n_src == n_target:
        return v.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    v_new = np.interp(x_tgt, x_src, v)
    return _repair(v_new)

# ---------- simulated annealing ----------
def _anneal(v_init, time_limit, start_T=0.08, step_max=0.2, cooling=0.9995):
    """Simple simulated‑annealing optimiser for C5."""
    start = time.time()
    n = v_init.shape[0]
    cur = _repair(v_init.copy())
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    T = start_T
    step = step_max
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair(cand)
        cand_score = _compute_c5(cand)

        diff = cand_score - cur_score
        if diff <= 0 or np.random.rand() < np.exp(-min(diff / max(T, 1e-12), 700.0)):
            cur = cand
            cur_score = cand_score

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        T *= cooling
        # slowly shrink step with temperature
        step = step_max * (T / start_T) if T < start_T else step_max
        if T < 1e-6:
            # temperature exhausted – break early
            break
    return best, best_score

# ---------- multi‑run optimiser ----------
def _optimise(v0, budget):
    """Run several annealing runs within the allocated budget."""
    start = time.time()
    best_v = _repair(v0.copy())
    best_score = _compute_c5(best_v)
    n = best_v.shape[0]

    # keep launching short anneals until time runs out
    while True:
        elapsed = time.time() - start
        remaining = budget - elapsed
        if remaining < 2.0:
            break
        # allocate a slice (max 12 s) for a single anneal run
        slice_time = min(12.0, remaining * 0.6)

        # start either from current best or a fresh random point
        if np.random.rand() < 0.75:
            start_vec = best_v.copy()
        else:
            start_vec = _random_feasible(n)

        cand_v, cand_score = _anneal(start_vec, slice_time)
        if cand_score < best_score:
            best_score, best_v = cand_score, cand_v
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function for the Erdős C₅ problem."""
    np.random.seed(seed)
    start_time = time.time()

    # ----- initial construction -----

            best_n = best_v.shape[0]
        else:
            best_n = best_v.shape[0]
    else:
        best_n = best_v.shape[0]

    # ----- optional second up‑sampling to the 800‑point limit -----

        final_target = 800
        up2 = _upsample(best_v, final_target)
        remaining = budget_s - (time.time() - start_time) - 0.2
        if remaining > 1.0:
            best_v, best_score = _optimise(up2, remaining)
            best_n = best_v.shape[0]

    # ----- final repair and evaluation -----
    best_v = _repair(best_v)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(best_n)
