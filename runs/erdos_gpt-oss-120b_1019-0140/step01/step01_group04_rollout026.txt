def _repair_vector(v, n):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    total = v.sum()
    # If difference small, just return
    diff = total - target
    if abs(diff) < 1e-12:
        return v
    # Try to add diff uniformly across all entries that are not at bounds
    free_mask = (v > 0.0) & (v < 1.0)
    free_count = free_mask.sum()
    # If freeing enough, adjust
    if free_count > 0:
        adjust = diff / free_count
        v[free_mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
        # recursively call if not close
        return _repair_vector(v, n)
    else:
        # all at bounds, need to shift some from 1 to 0 or vice versa
        # We'll just scale.
        factor = target / total if total != 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        # final small fix
        total = v.sum()
        diff2 = total - target
        if abs(diff2) > 1e-9:
            inc = -diff2 / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
        return v

    start = time.time()
    cur = initial_bin.copy()
    cur_score = _compute_c5(cur.astype(float))
    best = cur.copy(); best_score = cur_score
    T = 0.1  # initial temperature
    # schedule: reduce T over time exponentially
    while time.time() - start < time_limit:
        # pick two indices i (0) and j (1) perhaps swapped
        idx0 = np.where(cur == 0)[0]
        idx1 = np.where(cur == 1)[0]
        if len(idx0) ==0 or len(idx1)==0:
            break
        i = idx0[np.random.randint(len(idx0))]
        j = idx1[np.random.randint(len(idx1))]

        cand = cur.copy()
        cand[i] = 1
        cand[j] = 0
        cand_score = _compute_c5(cand.astype(float))
        delta = cand_score - cur_score

            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
        # update temperature maybe based on elapsed time
        elapsed = time.time() - start
        # linear schedule: T = T0 * (1 - elapsed / time_limit)
        T = max(1e-6, T * 0.9995)

    n = len(v)
    dx = 2.0 / n
    # using FFT correlation
    v_fft = np.fft.rfft(v, n*2-1)   # Actually need zero padding length maybe 2*n-1
    w = 1.0 - v
    w_fft = np.fft.rfft(w, n*2-1)
    corr = np.fft.irfft(v_fft * w_fft, n*2-1)
    # The correlation shape full length = 2*n-1
    # Actually convolution (not correlation) yields length 2*n-1 but correlation is similar with reversed w.
    # But using convolution of v and w gives sum_{i} v[i] g[k-i] where g = w.
    # np.correlate(v,w,mode='full') = convolution(v[::-1], w) = conv(v_rev, w)
    # But cross-correlation we can compute using FFT of v and w and then real part:
    # Since both are real, correlation = ifft(fft(v) * conj(fft(w)))?? Wait truth:
    # cross-correlation (v ⋆ w)[k] = sum_i v[i] * w[i+k]; for zero pad beyond boundaries.
    # This equals convolution of v with reversed w: conv(v, rev(w)).

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        cnt = mask.sum()
        if cnt == 0:
            break
        adjust = diff / cnt
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """C5 = max_x ∫ h(x)(1−h(x+k))dx approximated by discrete correlation."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _make_shape_vector(a, n):
    """
    Continuous shape: plateau of length a, linear ramp of length b=2*(1-a),
    then zero. Returns sampled vector on n points.
    """
    b = 2.0 * (1.0 - a)
    xs = (np.arange(n) + 0.5) * (2.0 / n)   # mid‑point sampling
    v = np.zeros(n, dtype=np.float64)
    # plateau
    mask_plateau = xs < a
    v[mask_plateau] = 1.0
    # ramp
    if b > 0:
        mask_ramp = (xs >= a) & (xs < a + b)
        v[mask_ramp] = 1.0 - (xs[mask_ramp] - a) / b
    # remainder stays zero
    return v
