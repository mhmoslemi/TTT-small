def _repair_vector(v, n, max_iter=20):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # Determine indices that can move to offset diff
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final small correction:
    s = v.sum()
    if abs(s - target) > 1e-9:
        # simple scaling
        factor = target / s if s != 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            # uniform distribution of residual
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _upsample_vector(v, n_target):
    n_src = len(v)
    # linear interpolation using np.interp over positions
    src_x = np.linspace(0, 2, n_src, endpoint=False)
    tgt_x = np.linspace(0, 2, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

import time
import numpy as np

def _repair_vector(v, n, max_iter=20):
    """Project v onto the feasible set: 0≤v≤1, sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny fix if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        if s > 0:
            v *= target / s
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v


def _compute_c5(v):
    """Exact overlap bound using FFT (fast)."""
    n = v.shape[0]
    L = 2 * n
    H = np.fft.rfft(v, n=L)
    G = np.fft.rfft(np.flip(1.0 - v), n=L)
    conv = np.fft.irfft(H * G, n=L)
    corr = conv[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def _upsample_vector(v, n_target):
    """Linear interpolation from len(v) to n_target, then repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)


def _subgradient_descent(v, time_limit, initial_step=0.06):
    """Directed descent using gradient of the worst overlap shift."""
    n = v.shape[0]
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    cur_v = v.copy()
    cur_score = best_score
    step = initial_step
    start = time.time()
    stagnation = 0
    while time.time() - start < time_limit:
        # correlation & worst shift
        L = 2 * n
        H = np.fft.rfft(cur_v, n=L)
        G = np.fft.rfft(np.flip(1.0 - cur_v), n=L)
        conv = np.fft.irfft(H * G, n=L)
        corr = conv[: 2 * n - 1]
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)  # shift k
        # gradient for this shift
        k = max_shift
        pad = abs(k)
        if pad:
            vp = np.concatenate([np.zeros(pad), cur_v, np.zeros(pad)])
            idxc = np.arange(n) + pad
            v_ipk = vp[idxc + k]
            v_imk = vp[idxc - k]
        else:
            v_ipk = cur_v
            v_imk = cur_v
        grad = (1.0 - v_ipk) - v_imk
        # step opposite gradient
        new_v = cur_v - step * grad
        new_v = _repair_vector(new_v, n)
        new_score = _compute_c5(new_v)

        if new_score < cur_score:
            cur_v, cur_score = new_v, new_score
            stagnation = 0
            step = min(step * 1.02, 0.2)
        else:
            stagnation += 1
            # occasional random jitter to escape plateaus
            if stagnation % 500 == 0:
                jitter = (np.random.rand(n) - 0.5) * step
                cur_v = _repair_vector(cur_v + jitter, n)
                cur_score = _compute_c5(cur_v)
        if cur_score < best_score:
            best_score, best_v = cur_score, cur_v.copy()
        if stagnation > 2000:
            step = max(step * 0.5, 1e-4)
            stagnation = 0
    return best_v, best_score


def _local_search(v, n, time_limit):
    """Simple exchange‑based hill climbing."""
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.02
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = best_v[i], best_v[j]
        lo = max(-hi, -(1.0 - hj))
        hi_delta = min(1.0 - hi, hj)
        if lo >= hi_delta:
            continue
        delta = (np.random.rand() * 2 - 1) * step
        delta = np.clip(delta, lo, hi_delta)
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        sc = _compute_c5(cand)
        if sc < best_score:
            best_score, best_v = sc, cand
    return best_v, best_score
