def simulated_annealing(h_start, time_limit, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_score = c5_value(h)
    cur_score = best_score
    N = len(h)
    dx = 2.0/N
    start_time = time.time()
    # Set temperature schedule
    init_temp = 0.01  # could be larger
    min_temp = 1e-6
    while time.time() - start_time < time_limit:
        elapsed = time.time() - start_time
        frac = elapsed / time_limit
        # temperature decreasing linearly
        T = init_temp * (1 - frac) + min_temp * frac
        # max epsilon shrink
        max_eps = 0.2 * (1 - frac) + 0.001 * frac
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        hi = h[i]
        hj = h[j]
        # compute allowable epsilon
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up + down == 0: continue
        eps = rng.random() * (up + down) - down
        eps = np.clip(eps, -max_eps, max_eps)
        if eps == 0:
            continue
        h_new = h.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (T+1e-12)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score
```

    best_score = _c5_value(best_h)
    N = h_init.size

        if cand_score < best_score:
            best_score = cand_score
            best_h = h_candidate
    return best_h, best_score

def _simulated_annealing(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start_time = time.time()
    init_score = cur_score
    T0 = init_score * 0.1 + 1e-4
    max_eps0 = 0.2
    min_eps = 1e-5
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit  # 0..1
        T = T0 * (1 - frac) + 1e-6 * frac
        max_eps = max_eps0 * (1 - frac) + min_eps * frac
        # pick i, j
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        hi = h_cur[i]
        hj = h_cur[j]
        # compute eps bounds
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up + down == 0:
            continue
        eps = (rng.random() * (up + down) - down)
        # limit magnitude
        eps = np.clip(eps, -max_eps, max_eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # Evaluate
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    # Using the identity C5 = max_s (sum_h - sum_i h_i h_{i+s}) * dx
    target_sum = N / 2.0
    conv = np.correlate(h, h, mode='full')
    min_conv = np.min(conv)
    return float((target_sum - min_conv) * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Annealing that preserves sum and box constraints."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start_time = time.time()
    # temperature and move scaling based on initial score
    T0 = max(cur_score * 0.08, 1e-4)
    max_eps0 = 0.5
    min_eps = 1e-5
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit  # 0 -> 1
        T = T0 * (1.0 - frac) + 1e-8 * frac
        max_eps = max_eps0 * (1.0 - frac) + min_eps * frac
        # pick two distinct indices
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)      # max we can increase hi (decrease hj)
        down = min(hi, 1.0 - hj)    # max we can decrease hi (increase hj)
        if up + down == 0.0:
            continue
        eps = rng.random() * (up + down) - down  # uniform in [-down, up]
        eps = np.clip(eps, -max_eps, max_eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # evaluation
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _multi_start_sa(h_init: np.ndarray, time_budget: float, rng: np.random.Generator,
                    n_starts: int = 6) -> tuple[np.ndarray, float]:
    """Run several annealing runs with the given budget and pick the best."""
    best_h = h_init.copy()
    best_score = _c5_value(best_h)
    start = time.time()
    for s in range(n_starts):
        elapsed = time.time() - start
        remaining = time_budget - elapsed
        if remaining <= 0.0:
            break
        # allocate the rest equally among remaining starts
        sub_time = remaining / (n_starts - s)
        if s == 0:
            h_start = h_init
        else:
            # random start respecting constraints
            h_start = rng.random(best_h.size)
            h_start = _repair_vector(h_start)
        h_candidate, cand_score = _simulated_annealing(h_start, sub_time, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = h_candidate
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # initial vector
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        # fallback: random start with reasonable size
        n0 = 120
        h0 = rng.random(n0)
    h0 = _repair_vector(h0)
    total_start = time.time()
    total_budget = float(budget_s)

    # Phase 1 – work on the original resolution
    phase1_time = total_budget * 0.6
    best_h, best_score = _multi_start_sa(h0, phase1_time, rng, n_starts=8)

    # Remaining time after phase 1
    elapsed = time.time() - total_start
    remaining = total_budget - elapsed

    # Successive up‑sampling phases
    while remaining > total_budget * 0.05:
        cur_n = best_h.size
        if cur_n >= 2000:
            break
        next_n = min(cur_n * 2, 2000)

        # Linear interpolation to higher resolution
        x_old = np.linspace(0.0, 2.0, cur_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, next_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _repair_vector(h_up)

        # Allocate a fraction of the remaining budget to this level
        phase_time = remaining * 0.5
        # ensure we have a minimal amount of time to make progress
        if phase_time < 5.0:
            phase_time = remaining
        phase_time = min(phase_time, remaining)

        # Optimise at the new resolution
        best_h_new, best_score_new = _multi_start_sa(h_up, phase_time, rng, n_starts=4)
        if best_score_new < best_score:
            best_h, best_score = best_h_new, best_score_new

        # Update elapsed / remaining
        elapsed = time.time() - total_start
        remaining = total_budget - elapsed

    # Final repair and precise evaluation
    best_h = _repair_vector(best_h)
    try:
        final_score = float(evaluate_erdos_solution(best_h))
    except NameError:
        final_score = _c5_value(best_h)

    return best_h, final_score, best_h.size
