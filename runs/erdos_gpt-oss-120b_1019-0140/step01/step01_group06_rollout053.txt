def project_to_feasible(v, target_sum, tol=1e-9, max_iter=100):
    # v: np.ndarray
    # Want to find x such that sum x = target_sum, 0<=x_i<=1, minimizing squared distance to v.
    # Use bisection for delta.
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        delta = (lo + hi) / 2.0
        x = np.clip(v + delta, 0.0, 1.0)
        s = np.sum(x)
        if abs(s - target_sum) <= tol:
            break
        if s > target_sum:
            hi = delta
        else:
            lo = delta
    # After loop, maybe fine adjust with small iteration of redistribution
    # Ensure sum constraint
    x = np.clip(v + delta, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    # if leftover difference due to eps, adjust smallest distance from bounds
    if abs(diff) > 1e-8:
        # distribute diff among indices not at bounds
        free = np.where((x > 0) & (x < 1))[0]
        if free.size > 0:
            x[free] += diff / free.size
            # clip again
            x = np.clip(x, 0.0, 1.0)
    return x
```

```
def objective_and_grad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    # compute gradient
    grad = np.zeros(n, dtype=float)
    if lag >= 0:
        # plus term for i where i+lag < n => i in 0..n-lag-1
        # grad[i] = 1 - h[i+lag] for i in that range
        grad[:n - lag] = 1.0 - h[lag:]
        # minus term for i where i - lag >= 0 => i in lag..n-1
        grad[lag:] -= h[:n - lag]
    else:
        p = -lag
        # plus term for i where i+lag >=0 => i >= p
        grad[p:] = 1.0 - h[:n - p]
        # minus term for i where i - lag < n => i < n-p
        grad[:n - p] -= h[p:]
    obj = corr[idx] * (2.0 / n)
    grad = grad * (2.0 / n)
    return obj, grad
```

Check indices: In `lag >=0`, we used `grad[:n - lag] = 1 - h[lag:]` because for i in 0..n-lag-1, h[i+lag] = h[lag:] (starting at lag). Similarly minus term: for i in lag..n-1, where i - lag in 0..n-lag-1, with h[i-lag] = h[:n-lag]. So `grad[lag:] -= h[:n - lag]` correct.

When lag negative, p = -lag >0:

plus: for i >= p, h[i+lag] = h[i-p]; valid for i=p..n-1 with i-p from 0..n-p-1. So ++ term: grad[p:] = 1 - h[:n-p]; yes.

minus term: for i where i - lag < n => i + p < n => i < n-p, subtract h[i+ p] = h[p:] (since i from 0..n-p-1). So `grad[:n-p] -= h[p:]`.

def project_to_feasible(v, target, tol=1e-9, max_iter=100):
    # Clip and adjust using bisection for delta
    # v is numpy array
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        delta = (lo + hi) / 2.0
        x = np.clip(v + delta, 0.0, 1.0)
        s = np.sum(x)
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = delta
        else:
            lo = delta
    # after loop, adjust residual
    x = np.clip(v + delta, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-8:
        free = np.where((x > 0) & (x < 1))[0]
        if free.size > 0:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
    return x
```

    size = 2**int(np.ceil(np.log2(2*n-1)))
    a = h
    b_rev = (1.0 - h)[::-1]

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return C5 for sampled h using FFT-based full correlation."""
    n = h.shape[0]
    size = 2 ** int(np.ceil(np.log2(2 * n - 1)))
    a = h
    b_rev = (1.0 - h)[::-1]
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b_rev, size)
    conv = np.fft.irfft(A * B, size)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def project_to_feasible(v, target, tol=1e-9, max_iter=100):
    """Project v onto [0,1]^n ∩ {sum = target} by bisection on a uniform shift."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        delta = (lo + hi) * 0.5
        x = np.clip(v + delta, 0.0, 1.0)
        s = np.sum(x)
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = delta
        else:
            lo = delta
    # final fine‑adjustment if needed
    x = np.clip(v + delta, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-8:
        free = np.where((x > 0) & (x < 1))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
    return x


def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def objective_and_grad(h, n):
    """C5 and its gradient (subgradient) for the active lag."""
    size = 2 ** int(np.ceil(np.log2(2 * n - 1)))
    a = h
    b_rev = (1.0 - h)[::-1]
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b_rev, size)
    conv = np.fft.irfft(A * B, size)[: 2 * n - 1]
    idx = np.argmax(conv)
    max_corr = conv[idx]
    lag = idx - (n - 1)

    grad = np.zeros_like(h)
    if lag >= 0:
        grad[: n - lag] = 1.0 - h[lag:]
        grad[lag:] -= h[: n - lag]
    else:
        p = -lag
        grad[p:] = 1.0 - h[: n - p]
        grad[: n - p] -= h[p:]

    obj = max_corr * (2.0 / n)
    grad = grad * (2.0 / n)
    return obj, grad


def objective_fun(h, n):
    return objective_and_grad(h, n)[0]


def objective_jac(h, n):
    return objective_and_grad(h, n)[1]


def eq_sum_constraint(h, target):
    return np.sum(h) - target


def gradient_refine(h, n, budget_s, start_time):
    """Simple projected gradient descent to polish a solution."""
    step = 0.5
    max_iter = 800
    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        cur_obj, grad = objective_and_grad(h, n)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new, n / 2.0)
        new_obj = compute_c5(h_new)
        if new_obj < cur_obj:
            h = h_new
            step = min(step * 1.2, 1.0)
        else:
            step *= 0.5
            if step < 1e-7:
                break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    base_h = globals().get("initial_h_values")
    if base_h is not None:

    best_h = None
