    start = time.time()
    cur = v_start.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score

    # Get indices of ones and zeros
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur < 0.5)[0].tolist()
    # Temperature schedule
    # Estimate number of iterations maybe 100k
    # Use time-based loop
    while time.time() - start < time_limit:
        # progress fraction
        frac = (time.time() - start) / time_limit
        T = max(0.001, 0.02 * (1 - frac))  # linear cooling to near zero

        if not ones or not zeros:
            break

        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # swap
        cur[i] = 0.0
        cur[j] = 1.0
        # update ones/zeros lists
        # remove i from ones, add i to zeros
        # remove j from zeros, add j to ones
        # but we need to store old indices to revert if needed

        new_score = _compute_c5(cur)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = np.random.rand() < np.exp(-delta / T)

        if accept:
            # keep swap
            cur_score = new_score
            # update lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            # track best
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert swap
            cur[i] = 1.0
            cur[j] = 0.0

        # optionally check early exit if time left low
        # continue

    return best, best_score

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with the linear equality sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if still off
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _binary_round(v):
    """Return a binary (0/1) vector with exactly half entries = 1."""
    n = len(v)
    k = n // 2
    # indices of the k largest entries
    idx = np.argpartition(v, -k)[-k:]
    bin_vec = np.zeros(n, dtype=np.float64)
    bin_vec[idx] = 1.0
    return bin_vec

# Optimization primitives
# ----------------------------------------------------------------------
def _anneal_binary(v_start, n, time_limit):
    """Simulated annealing on binary vectors (swap 1↔0)."""
    start = time.time()
    cur = v_start.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    # temperature schedule parameters
    T0 = 0.02

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1.0 - frac)
        if T < 1e-6:
            T = 0.0

        # propose a swap (ensure different values)
        i = np.random.randint(n)
        j = np.random.randint(n)
        if cur[i] == cur[j]:
            continue
        # ensure i holds a 1 and j a 0
        if cur[i] < 0.5:
            i, j = j, i

        # perform swap
        cur[i] = 0.0
        cur[j] = 1.0

        new_score = _compute_c5(cur)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        elif T > 0.0 and np.random.rand() < np.exp(-delta / T):
            accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert swap
            cur[i] = 1.0
            cur[j] = 0.0

    return best, best_score

def _anneal_binary_multi(v_start, n, time_limit):
    """Run several short anneals (with occasional random restarts) within time_limit."""
    start = time.time()
    best = v_start.copy()
    best_score = _compute_c5(best)

    while True:
        elapsed = time.time() - start
        remaining = time_limit - elapsed
        if remaining < 5.0:
            break
        # allocate a sub‑run (max 20 s or a fraction of remaining)
        sub_time = min(20.0, remaining * 0.4)

        # random restart with probability 0.3
        if np.random.rand() < 0.3:
            rnd = np.random.rand(n)
            v0 = _binary_round(rnd)
        else:
            v0 = best.copy()

        cand, cand_score = _anneal_binary(v0, n, sub_time)
        if cand_score < best_score:
            best, best_score = cand, cand_score

    return best, best_score

def _local_search(v_start, n, time_limit):
    """Hill‑climbing with small exchanges (continuous values)."""
    start = time.time()
    cur = v_start.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
