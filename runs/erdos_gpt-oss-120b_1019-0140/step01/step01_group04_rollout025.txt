def gradient_step(v, alpha):
    n = len(v)
    # compute correlation via np.correlate
    corr = np.correlate(v, 1 - v, mode='full')
    max_idx = np.argmax(corr) # index of max correlation
    shift = max_idx - (n - 1) # shift relative offset
    # compute gradient for each i
    grad = np.zeros_like(v)
    # For each i where i+shift in range:
    if shift >= 0:
        # i from 0 to n-1-shift: term v[i]*(1 - v[i+shift])
        # derivative wrt v[i] = (1 - v[i+shift])
        # derivative wrt v[i+shift] = -v[i]
        i_range = np.arange(0, n - shift)
        grad[i_range] = (1 - v[i_range + shift])
        grad[i_range + shift] = grad[i_range + shift] - v[i_range]  # add -v[i] to grad at i+shift
    else:
        # shift negative: shift = -k <0 => offset left?
        k = -shift
        # term v[i+k] * (1 - v[i])
        i_range = np.arange(0, n - k)
        grad[i_range + k] = (1 - v[i_range])  # derivative wrt v[i+k]
        grad[i_range] = grad[i_range] - v[i_range + k]   # derivative wrt v[i]
    # Now grad is derivative of correlation sum (not multiplied by dx)
    # We'll use step = alpha * grad
    cand = v - alpha * grad
    cand = project(cand, target)
    cand_score = compute_c5(cand)
    return cand, cand_score

def project_onto_box_sum(v, target, max_iter=60):
    lo = -np.max(v) - 5  # ensure low enough
    hi = 1.0 - np.min(v) + 5
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    # final
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # fix small diff
    s = w.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # find any index where we can adjust
        # if diff >0, increase where w <1, else decrease where w>0
        if diff > 0:
            idxs = np.where(w < 1.0 - 1e-12)[0]
            if idxs.size>0:
                w[idxs[0]] += diff
        else:
            idxs = np.where(w > 1e-12)[0]
            if idxs.size>0:
                w[idxs[0]] += diff
        # clip again
        w = np.clip(w, 0.0, 1.0)
    # final check: if still off, maybe adjust further
    s = w.sum()
    if abs(s - target) > 1e-8:
        # distribute diff uniformly
        w += diff / len(w)
        w = np.clip(w, 0.0, 1.0)
        # final target enforcement: scale
        s = w.sum()
        if s != 0:
            w *= target / s
            w = np.clip(w, 0.0, 1.0)
    return w

def optimize_at_resolution(init_v, n, time_limit):
    target = n / 2.0
    cur_v = init_v.copy()
    cur_score = compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    start = time.time()
    # parameters
    init_temp = 0.02  # maybe 0.01? For objective around 0.38, diff ~0.001
    final_temp = 0.0002
    max_step = 0.2
    # pre-calc total time
    total = time_limit
    while True:
        elapsed = time.time() - start
        if elapsed >= total:
            break
        # temperature schedule exponential
        fraction = elapsed / total
        temp = init_temp * ((final_temp / init_temp) ** fraction)
        # step size adjust
        step = max_step * (temp / init_temp)  # linearly decreasing
        # choose two indices
        i, j = np.random.randint(n, size=2)
        if i == j:

        cand = cur_v.copy()

        cand[j] -= delta
        cand = project_onto_box_sum(cand, target)
        cand_score = compute_c5(cand)
        diff = cand_score - cur_score
        if diff < 0:
            cur_v, cur_score = cand, cand_score
        else:

            # avoid overflow

def project_feasible(v, target):
    # binary search for lambda
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # final correction
    s = w.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # find an index to adjust
        if diff > 0:
            # increase some not at 1
            idx = np.where(w < 1 - 1e-12)[0]
        else:
            idx = np.where(w > 1e-12)[0]
        if idx.size > 0:
            w[idx[0]] += diff
            w = np.clip(w, 0.0, 1.0)
    # after adjustment may still be off by tiny epsilon; scale if necessary
    s = w.sum()
    if abs(s - target) > 1e-9:
        # scale overall
        factor = target / s if s != 0 else 0.0
        w *= factor
        w = np.clip(w, 0.0, 1.0)
    return w

def anneal_optimize(init_v, time_limit):
    n = len(init_v)
    target = n/2.0
    cur_v = init_v.copy()
    cur_score = compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    start = time.time()
    # parameters
    init_temp = 0.012
    final_temp = 0.0002
    max_step = 0.2
    # frequency of gradient steps
    grad_prob = 0.03
    restart_prob = 0.001
    # loop
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        fraction = elapsed / time_limit
        temp = init_temp * (1 - fraction) + final_temp * fraction
        step = max_step * (temp / init_temp)  # linearly scale
        # random restart
        if np.random.rand() < restart_prob:
            cur_v = random_feasible(n)
            cur_score = compute_c5(cur_v)
            if cur_score < best_score:
                best_v, best_score = cur_v.copy(), cur_score
            continue
        # exchange move
        i, j = np.random.randint(n, size=2)
        delta = (np.random.rand() - 0.5) * 2 * step
        cand = cur_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = project_feasible(cand, target)
        cand_score = compute_c5(cand)
        diff = cand_score - cur_score
        if diff < 0 or np.random.rand() < np.exp(-diff / max(temp, 1e-12)):

        if np.random.rand() < grad_prob:

import time
import numpy as np

def _project_feasible(v, target):
    """Project vector onto [0,1]^n with sum(v)=target via binary search."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment to enforce exact sum
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(w < 1.0 - 1e-12)[0]
        else:
            idx = np.where(w > 1e-12)[0]
        if idx.size:
            w[idx[0]] += diff
            w = np.clip(w, 0.0, 1.0)
    s = w.sum()
    if abs(s - target) > 1e-9 and s > 0:
        w *= target / s
        w = np.clip(w, 0.0, 1.0)
    return w

def _compute_c5(v):
    """Return C5 bound for step vector v."""
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _project_feasible(v, n / 2.0)

def _upsample(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = v.shape[0]
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _project_feasible(new_v, n_target / 2.0)

def _max_shift_gradient(v):
    """Gradient of the maximal correlation term with respect to v."""
    n = v.shape[0]
    corr = np.correlate(v, 1.0 - v, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)  # shift can be negative
    grad = np.zeros_like(v)
    if shift >= 0:
        i_end = n - shift
        grad[:i_end] = 1.0 - v[shift:shift + i_end]
        grad[shift:shift + i_end] -= v[:i_end]
    else:
        k = -shift
        i_end = n - k
        grad[k:k + i_end] = 1.0 - v[:i_end]
        grad[:i_end] -= v[k:k + i_end]
    return grad

def _anneal_optimize(v0, time_limit):
    """Simulated annealing on feasible set."""
    n = v0.shape[0]
    target = n / 2.0
    cur = v0.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    init_temp = 0.012
    final_temp = 0.0002
    max_step = 0.20
    grad_prob = 0.03
    restart_prob = 0.001
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        temp = init_temp * (1 - frac) + final_temp * frac
        step = max_step * (temp / init_temp)
        # occasional full restart
        if np.random.rand() < restart_prob:
            cur = _random_feasible(n)
            cur_score = _compute_c5(cur)
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
            continue
        i, j = np.random.randint(n, size=2)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _project_feasible(cand, target)
        cand_score = _compute_c5(cand)
        diff = cand_score - cur_score
        if diff < 0.0 or np.random.rand() < np.exp(-diff / max(temp, 1e-12)):
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
        # occasional gradient step
        if np.random.rand() < grad_prob:
            grad = _max_shift_gradient(cur)
            alpha = step * 0.3
            cand2 = cur - alpha * grad
            cand2 = _project_feasible(cand2, target)
            cand2_score = _compute_c5(cand2)
            diff2 = cand2_score - cur_score
            if diff2 < 0.0 or np.random.rand() < np.exp(-diff2 / max(temp, 1e-12)):
                cur, cur_score = cand2, cand2_score
                if cur_score < best_score:
                    best, best_score = cur.copy(), cur_score
    return best, best_score

def _refine_local(v0, time_limit):
