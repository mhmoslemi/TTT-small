def optimize_h(h0, max_iter, step0=0.5, beta=0.9, restart_interval=2000, restart_frac=0.3):

   h = h0.copy()
   v = np.zeros_like(h)
   best_h = h.copy()
   best_score = compute_c5(h)
   last_improve_iter = 0
   for it in range(1, max_iter+1):
        # compute shift and gradient
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (h.size - 1)
        grad = compute_gradient(h, shift)
        step = step0 / np.sqrt(it)
        v = beta * v - step * grad
        h_candidate = h + v
        h_candidate = project_onto_box_sum(h_candidate, s)
        score = compute_c5(h_candidate)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_candidate.copy()
            last_improve_iter = it
        h = h_candidate
        # restart if no improvement for many steps
        if it - last_improve_iter > restart_interval:
            # add small random perturbation
            noise = rng.uniform(-restart_frac, restart_frac, size=h.shape)
            h = project_onto_box_sum(h + noise, s)
            v = np.zeros_like(h)
            last_improve_iter = it

   return best_h, best_score

def local_search_random_moves(h, best_score, n_moves=2000, move_eps=0.01):
    for _ in range(n_moves):
        # choose i to decrease, j to increase
        # Choose i with weight proportional to h[i] maybe
        i = rng.integers(h.size)
        # ensure we can move some epsilon
        max_decrease = min(move_eps, h[i])
        if max_decrease <= 0:
            continue
        j = rng.integers(h.size)
        max_increase = min(move_eps, 1 - h[j])
        eps = min(max_decrease, max_increase)
        if eps <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # sum unchanged
        score = compute_c5(h_new)
        if score < best_score:
            h = h_new
            best_score = score
    return h, best_score

def optimize_h(h0, rng, max_iter, step0=0.5, beta=0.8, restart_interval=1000, restart_frac=0.05):

    s = n / 2.0
    h = h0.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    last_improve = 0
    for it in range(1, max_iter+1):
        # compute worst shift and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        # step size
        step = step0 / np.sqrt(it)
        v = beta * v - step * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        # update best
        if score < best_score:
            best_score = score
            best_h = h_new.copy()

            h = project_onto_box_sum(h + noise, s)
            v.fill(0.0)
            last_improve = it
    return best_h, best_score

def compute_gradient(h, shift):
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def upsample(arr, new_n):
    m = arr.shape[0]

    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def optimize_h(h0, rng, max_iter, step0, beta, restart_interval,
               restart_frac, start_time, budget_s):
    n = h0.size
    s = n / 2.0
    h = h0.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    last_improve = 0
    for it in range(1, max_iter + 1):
        if time.time() - start_time > budget_s - 1.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        step = step0 / np.sqrt(it)
        v = beta * v - step * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            last_improve = it
        h = h_new
        if it - last_improve >= restart_interval:
            noise = rng.uniform(-restart_frac, restart_frac, size=h.shape)
            h = project_onto_box_sum(h + noise, s)
            v.fill(0.0)
            last_improve = it
    return best_h, best_score

def local_search_random_moves(h, best_score, n_moves, move_eps, rng,
                              start_time, budget_s):
    n = h.size
    for _ in range(n_moves):
        if time.time() - start_time > budget_s - 1.0:
            break
        i = rng.integers(n)
        if h[i] <= 1e-12:
            continue
        max_decr = min(move_eps, h[i])
        j = rng.integers(n)
        if h[j] >= 1.0 - 1e-12:
            continue
        max_incr = min(move_eps, 1.0 - h[j])
        eps = min(max_decr, max_incr)
        if eps <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            h = h_new
    return h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_score = float('inf')

    # Determine resolution
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        target_n = int(init_arr.size)
    else:
        target_n = 256
    if target_n % 2:
        target_n += 1

    # Build candidate list
    candidates = []
    if 'initial_h_values' in globals():
        h0 = upsample(np.asarray(initial_h_values, dtype=float), target_n)
        h0 = project_onto_box_sum(h0, target_n / 2.0)
        candidates.append(h0)

    n_tm = 64
    while n_tm <= target_n:
        if n_tm % 2 == 0:
            tm = thue_morse(n_tm)
            if n_tm != target_n:
                tm = upsample(tm, target_n)
            tm = project_onto_box_sum(tm, target_n / 2.0)
            candidates.append(tm)
        n_tm <<= 1

    for _ in range(4):
        candidates.append(random_binary_balanced(target_n, rng))
    for _ in range(4):
        v = rng.random(target_n)
        candidates.append(project_onto_box_sum(v, target_n / 2.0))

    # Optimize each base candidate
    for base_h in candidates:
        if time.time() - start_time > budget_s - 5.0:
            break
        remaining = budget_s - (time.time() - start_time)
        max_iters = int(min(50000, max(2000, remaining * 2500)))
        h_opt, score_opt = optimize_h(
            base_h, rng,
            max_iter=max_iters,
            step0=0.5,
            beta=0.8,
            restart_interval=max(500, max_iters // 5),
            restart_frac=0.05,
            start_time=start_time,
            budget_s=budget_s
        )
        h_opt, score_opt = local_search_random_moves(
            h_opt, score_opt,
            n_moves=max(500, max_iters // 10),
            move_eps=0.02,
            rng=rng,
            start_time=start_time,
            budget_s=budget_s
        )
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    # Random restarts until time runs out
    while time.time() - start_time < budget_s - 5.0:
        v = rng.random(target_n)
        h_start = project_onto_box_sum(v, target_n / 2.0)
        remaining = budget_s - (time.time() - start_time)
        max_iters = int(min(20000, max(1000, remaining * 1500)))
        if max_iters < 1000:
            break
        h_opt, score_opt = optimize_h(
            h_start, rng,
            max_iter=max_iters,
            step0=0.5,
            beta=0.8,
            restart_interval=max(300, max_iters // 4),
            restart_frac=0.05,
            start_time=start_time,
            budget_s=budget_s
        )
        h_opt, score_opt = local_search_random_moves(
            h_opt, score_opt,
            n_moves=max(300, max_iters // 15),
            move_eps=0.02,
            rng=rng,
            start_time=start_time,
            budget_s=budget_s
        )
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    # Final projection and evaluation
    if best_h is None:
        best_h = np.full(target_n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, target_n / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    return best_h, float(best_score), best_h.size
