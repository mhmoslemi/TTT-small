def _repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # find index that can adjust
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback distribute
        h += diff / h.size
        np.clip(h, 0, 1, out=h)
    else:
        i = idx[0]
        adjust = np.clip(h[i] + diff, 0.0, 1.0) - h[i]
        h[i] += adjust
        # recurs if not fixed
        if abs(adjust - diff) > 1e-12:
            return _repair_sum(h, target)
    return h
```

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that sum(h) == target (within tolerance)."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = np.clip(h[i] + diff, 0.0, 1.0)
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _worst_pair_and_lag(h: np.ndarray):
    """Return (source_index, dest_index, lag) for the pair that contributes most
    to the current maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)         # lag corresponding to the maximal correlation
    if k >= 0:
        i1 = np.arange(n - k)
        i2 = i1 + k
    else:
        i1 = np.arange(-k, n)
        i2 = i1 + k
    if i1.size == 0:
        return None, None, k
    contrib = h[i1] * (1.0 - h[i2])
    max_idx = int(np.argmax(contrib))
    src = int(i1[max_idx])
    dst = int(i2[max_idx])
    return src, dst, k
