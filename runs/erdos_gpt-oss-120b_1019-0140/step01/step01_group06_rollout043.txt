def compute_c5_fft(h):
    n = len(h)
    # compute correlation via FFT
    # pad to length 2*n maybe.
    N = 1 << (2*n-1).bit_length()
    # compute FFT on h and (1-h)
    # Actually correlation = ifft(fft(h, N) * np.conj(fft(1-h, N)))  (real).
    fh = np.fft.rfft(h, n=N)
    fg = np.fft.rfft(1 - h, n=N)
    corr = np.fft.irfft(fh * np.conj(fg), n=N)
    # corr length N, but we need full length 2n-1, central part
    # Take indices from 0 to 2n-2 inclusive: corr[:2*n-1]
    # Multiply by dx factor.
    return np.max(corr[:2*n-1]) * (2.0 / n)

def resample_h(base_h, n_target):

def optimize_sa_binary(n, init_h, time_limit, start_time, rng):
    h = init_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    # parameters
    max_iter = 300000
    T0 = 0.1
    T_end = 1e-6
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        # temperature schedule
        t = T0 * ((T_end / T0) ** (it / max_iter))
        # propose swap
        # pick random 1 index:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        new_val = compute_c5(h)
        delta = new_val - cur_val
        accept = delta <= 0 or rng.random() < math.exp(-delta / t)
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_val
```

def local_refine(h, n, maxiter=500):
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
    bnds = [(0.0, 1.0)] * n
    res = opt.minimize(lambda x: compute_c5(x), x0=h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if not res.success:
        return h
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n/2)

import time
import math
import numpy as np
import scipy.optimize as opt

def compute_corr_max_idx(h):
    """Return (max correlation value, index) of full correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=N)
    R = np.fft.rfft(rev, n=N)
    conv = np.fft.irfft(H * R, n=N)
    conv = conv[: 2 * n - 1]
    idx = int(np.argmax(conv))
    val = conv[idx]
    return val, idx


def compute_c5(h):
    """Overlap bound C5 for a sampled step function h."""
    max_val, _ = compute_corr_max_idx(h)
    return max_val * (2.0 / h.shape[0])


def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    """Project h onto box [0,1]^n and hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Linear interpolation of base_h to length n_target with feasibility projection."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def equality_constraint(x, target_sum):
    """Linear equality constraint: sum(x) == target_sum."""
    return np.sum(x) - target_sum


def objective_func(x):
    """Objective wrapper for optimizer."""
    return compute_c5(x)


def local_refine(h, target_sum, maxiter=400):
    """Run a brief SLSQP refinement starting from h."""
    n = h.shape[0]
    cons = {"type": "eq", "fun": equality_constraint, "args": (target_sum,)}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            objective_func,
            x0=h,
            method="SLSQP",
            bounds=bnds,
            constraints=[cons],
            options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = project_to_feasible(x, target_sum)
        return x
    except Exception:
        return h


def gradient_descent(h, target_sum, end_time, rng):
    """Subgradient descent on C5 with projection onto feasibility set."""
    h = project_to_feasible(h, target_sum)
    n = h.shape[0]
    max_val, max_idx = compute_corr_max_idx(h)
    cur_val = max_val * (2.0 / n)
    best_h = h.copy()
    best_val = cur_val
    step = 0.5

    while time.time() < end_time:
        # current worst lag
        lag = max_idx - (n - 1)

        # compute forward and backward shifted copies (zero where out of range)
        forward = np.zeros_like(h)
        backward = np.zeros_like(h)
        if lag >= 0:
            forward[: n - lag] = h[lag:]
            backward[lag:] = h[: n - lag]
        else:
            la = -lag
            forward[la:] = h[: n - la]
            backward[: n - la] = h[la:]

        # gradient of the active term
        grad = 1.0 - forward - backward

        # take step and re‑project
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new, target_sum)

        # evaluate new C5
        new_max_val, new_max_idx = compute_corr_max_idx(h_new)
        new_val = new_max_val * (2.0 / n)

        if new_val < cur_val:
            h = h_new
            cur_val = new_val
            max_idx = new_max_idx
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.5
            if step < 1e-6:
                break

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Main entry point: returns (h_values, c5_bound, n_points)."""
    start = time.time()
    end = start + budget_s * 0.97  # small safety margin
    rng = np.random.default_rng(seed)

    # Base resolution (use parent if available)
    n0 = 380
    parent = globals().get("initial_h_values")

    best_h = h0.copy()

    attempt = 0
    while time.time() < end and attempt < 30:
        attempt += 1

        # Generate a starting point
        if attempt == 1 and parent is not None:
            cand = best_h.copy()
        else:
            cand = best_h + rng.normal(scale=0.05, size=n0)
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_to_feasible(cand, n0 / 2.0)

        # Gradient descent on current resolution
        cand_h, cand_val = gradient_descent(cand, n0 / 2.0, end, rng)

        if cand_val < best_val:
            best_val = cand_val

            # Local refinement with SLSQP
            refined = local_refine(best_h, n0 / 2.0, maxiter=600)
            refined_val = compute_c5(refined)
