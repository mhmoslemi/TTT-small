def _simulated_annealing_adaptive(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    # initial temperature high enough: maybe set based on scale of scores. Use T0 = (max_score - min_score) * 0.1? But we need something.
    # we can compute an initial typical move delta magnitude: Use T0 = 0.1.
    T0 = 0.05 # covers small variations
    # max_eps initial maybe 0.5
    max_eps0 = 0.5
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * (frac ** 0.5) # slower cooling
        max_eps = max_eps0 * frac + 1e-6
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        # compute allowed epsilon range
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        # if zero move not possible, continue
        if up == 0.0 and down == 0.0:
            continue
        # sample eps uniform within [-down, up]
        eps = rng.random() * (up + down) - down
        # restrict magnitude
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        # create new vector
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # no need to check bounds because eps respects constraints
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) == len(v)/2."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if not np.isclose(s, target, atol=1e-12):
        if s > 0.0:
            v *= target / s
        else:
            v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        s2 = v.sum()
        if not np.isclose(s2, target, atol=1e-12):
            diff = target - s2
            v += diff / v.size
            np.clip(v, 0.0, 1.0, out=v)
            s3 = v.sum()
            if not np.isclose(s3, target, atol=1e-12):
                v *= target / s3
                np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing_adaptive(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Pairwise mass‑transfer SA with linear cooling and shrinking step size."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.05                # initial temperature
    max_eps0 = 0.5           # initial absolute move limit
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * (frac ** 0.5)          # slower cooling
        max_eps = max_eps0 * frac + 1e-12
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)          # positive bound
        down = min(hi, 1.0 - hj)        # negative bound
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # starting construction
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        h0 = rng.random(200)
    h0 = _repair_vector(h0)
    best_h = h0
    best_score = _c5_value(best_h)

    start_total = time.time()
    total_time = budget_s

    # resolution schedule: start size, then double until ≤1000
    res = [best_h.size]
    while res[-1] < 1000:
        nxt = min(res[-1] * 2, 1000)
        if nxt == res[-1]:
            break
        res.append(nxt)

    # allocate 30 % of time to the first (original) resolution,
    # remaining equally among the higher resolutions.
    time_initial = total_time * 0.30
    remaining_levels = len(res) - 1
    time_per_level = (total_time - time_initial) / max(1, remaining_levels)

    # Phase 0 – optimise at original resolution
    t0 = min(time_initial, total_time - (time.time() - start_total))
    if t0 > 0.0:
        h_opt, s_opt = _simulated_annealing_adaptive(best_h, t0, rng)
        if s_opt < best_score:
            best_h, best_score = h_opt, s_opt

    # Higher resolutions
    for level in range(1, len(res)):
        elapsed = time.time() - start_total
        time_left = total_time - elapsed
        if time_left <= 1.0:
            break
        t_alloc = min(time_per_level, time_left)
        n_new = res[level]
        # interpolate to new grid
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _repair_vector(h_up)
        h_opt, s_opt = _simulated_annealing_adaptive(h_up, t_alloc, rng)
        if s_opt < best_score:
            best_h, best_score = h_opt, s_opt

    # final repair and exact evaluation
    best_h = _repair_vector(best_h)
    final_score = _c5_value(best_h)
    return best_h, float(final_score), best_h.size
