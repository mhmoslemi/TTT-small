def smooth_max_descent(h0, n_iter, step0, alpha_start, alpha_end, beta=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    best_h_local = h.copy()
    best_score_local = compute_c5(h)
    for it in range(1, n_iter+1):
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        # compute smooth weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / n_iter)
        # avoid overflow: subtract max
        max_corr = np.max(corr)
        exp_vals = np.exp(alpha * (corr - max_corr))  # stable
        weights = exp_vals / exp_vals.sum()
        # compute term1 and term2 via convolution
        w_rev = weights[::-1]
        conv1 = np.convolve(w_rev, h, mode='full')
        conv2 = np.convolve(weights, h, mode='full')
        # slice conv results
        start = n-1
        term1 = conv1[start:start+n]
        term2 = conv2[start:start+n]
        grad = 1.0 - term1 - term2
        # update with momentum
        eta = step0 / np.sqrt(it)
        v = beta * v + (1 - beta) * grad
        h = h - eta * v
        # projection onto feasibility
        h = project_onto_box_sum(h, target_sum)
        # evaluate
        score = np.max(corr) * (2.0 / n)  # dx factor
        if score < best_score_local:
            best_score_local = score
            best_h_local = h.copy()
    return best_h_local, best_score_local

import time
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure bracket
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def thue_morse_seq(n):
    """Binary Thue–Morse sequence of length n (values 0.0 or 1.0)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                      dtype=float, count=n)
    return seq

def smooth_max_descent(h0, n_iter, step0=0.5,
                       alpha_start=1.0, alpha_end=30.0,
                       beta=0.9, start_time=None, budget_s=1e9):
    """Projected gradient descent using a smoothed max (softmax) objective."""
    h = h0.copy()
    vel = np.zeros_like(h)
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, n_iter + 1):
        if start_time is not None and time.time() - start_time > budget_s - 0.5:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        alpha = alpha_start + (alpha_end - alpha_start) * (it / n_iter)

        # stable softmax weights
        maxc = np.max(corr)
        exp_vals = np.exp(alpha * (corr - maxc))
        weights = exp_vals / np.sum(exp_vals)

        # weighted gradients via convolutions
        w_rev = weights[::-1]
        conv1 = np.convolve(w_rev, h, mode='full')
        conv2 = np.convolve(weights, h, mode='full')
        start = n - 1
        term1 = conv1[start:start + n]
        term2 = conv2[start:start + n]
        grad = 1.0 - term1 - term2

        eta = step0 / np.sqrt(it)
        vel = beta * vel + (1.0 - beta) * grad
        h = h - eta * vel

        h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

        v = rng.random(n)
        starts.append(project_onto_box_sum(v, target_sum))

    # 4. Random binary balanced starts
    for _ in range(2):
        idx = rng.choice(n, size=n // 2, replace=False)
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
        starts.append(h)

    max_iters_per_start = 30000

    # Optimize each deterministic/random start
    for h0 in starts:
        if time.time() - start_time > budget_s - 5:
            break
        time_left = budget_s - (time.time() - start_time) - 1.0
        est_iters = int(time_left * 800)          # rough 800 it/s estimate
        n_iter = min(max_iters_per_start, max(5000, est_iters))
        h_opt, sc_opt = smooth_max_descent(
            h0, n_iter=n_iter, step0=0.5,
            alpha_start=1.0, alpha_end=30.0, beta=0.9,
            start_time=start_time, budget_s=budget_s)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # Additional random restarts until time expires
    while time.time() - start_time < budget_s - 5:
        v = rng.random(n)
        h0 = project_onto_box_sum(v, target_sum)
        h_opt, sc_opt = smooth_max_descent(
            h0, n_iter=8000, step0=0.5,
            alpha_start=1.0, alpha_end=30.0, beta=0.9,
            start_time=start_time, budget_s=budget_s)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # Final refinement of the best candidate if time allows
    if best_h is not None:
        time_left = budget_s - (time.time() - start_time) - 1.0
        if time_left > 5:
            extra_iters = int(time_left * 800)
            best_h, best_score = smooth_max_descent(
                best_h, n_iter=extra_iters, step0=0.4,
                alpha_start=5.0, alpha_end=80.0, beta=0.9,
                start_time=start_time, budget_s=budget_s)

    # Safeguard feasibility and final evaluation
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
