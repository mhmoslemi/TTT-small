def _gradient_step(v, step):
    n = len(v)
    # compute correlation and find max shift
    corr = np.correlate(v, 1.0 - v, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)  # can be negative
    # compute gradient g
    # For each i, g_i = 1 - (v[i - shift] if valid else 0) - (v[i + shift] if valid else 0)
    # Use vectorized: create shifted arrays with padding zeros
    # Use np.take with mode='wrap'? No, we want zeros for out-of-range.
    # We'll create arrays left and right: left = zeros same shape, right zeros.
    # left[i] = v[i - shift] if 0 <= i-shift < n else 0
    # right[i] = v[i + shift] if 0 <= i+shift < n else 0
    # For shift positive: left shift reduces index; right shift increases.
    # Implement using slicing
    left = np.zeros_like(v)
    right = np.zeros_like(v)
    if shift >= 0:
        # left: i-shift >=0 -> i>=shift
        left[shift:] = v[:-shift] if shift != 0 else v.copy()
        # right: i+shift < n -> i < n-shift
        right[:n-shift] = v[shift:] if shift != 0 else v.copy()
    else:
        s = -shift
        # left: i - (-s) = i + s, indices < n -> i < n - s => left[:n-s] = v[s:]
        left[:n-s] = v[s:]  # shift <0 => left shift forward
        # right: i + (-s) = i - s, i >= s => right[s:] = v[:n-s]
        right[s:] = v[:n-s]
    # gradient
    g = 1.0 - left - right
    # update
    v_new = v - step * g
    # clip and repair
    v_new = _repair_vector(v_new, n)
    return v_new

import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * (2.0 / n)

def _gradient_step(v, step):
    """One projected subgradient step for the max correlation shift."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)  # integer lag
    if shift > 0:
        left = np.empty_like(v)
        left[:] = 0.0
        left[shift:] = v[:-shift]
        right = np.empty_like(v)
        right[:] = 0.0
        right[: n - shift] = v[shift:]
    elif shift < 0:
        s = -shift
        left = np.empty_like(v)
        left[:] = 0.0
        left[: n - s] = v[s:]
        right = np.empty_like(v)
        right[:] = 0.0
        right[s:] = v[: n - s]
    else:
        left = v.copy()
        right = v.copy()
    g = 1.0 - left - right
    v_new = v - step * g
    v_new = _repair_vector(v_new, n)
    return v_new

def _subgradient_opt(v, max_iter, time_limit, initial_step=0.2, decay=0.999):
    """Projected subgradient descent with time budget."""
    start = time.time()
    step = initial_step
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    for _ in range(max_iter):
        if time.time() - start > time_limit:
            break
        v = _gradient_step(v, step)
        sc = _compute_c5(v)
        if sc < best_score:
            best_score = sc
            best_v = v.copy()
        step *= decay
    return best_v, best_score

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _local_search(v, n, time_limit):
    """Simple exchange hill‑climbing."""
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        sc = _compute_c5(cand)
        if sc < best_score:
            best_score, best_v = sc, cand
    return best_v, best_score
