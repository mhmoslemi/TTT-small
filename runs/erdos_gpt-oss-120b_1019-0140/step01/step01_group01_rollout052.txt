def compute_c5(...): as earlier.

def adjust_sum_to_target(...): adapt but keep.

Then def run(...):

   rng = np.random.RandomState(seed)

   start = time.time()
   max_time = budget_s - 0.1

   # get n_points
   try:
        init_h_arr = np.array(initial_h_values, dtype=float)
        n_points = init_h_arr.size
        target_sum = n_points / 2.0
        # ensure values within bounds
        init_h_arr = np.clip(init_h_arr, 0.0, 1.0)
        # adjust sum
        init_h_arr = adjust_sum_to_target(init_h_arr.copy(), target_sum, rng)
   except Exception:
        n_points = 200
        target_sum = n_points / 2.0
        init_h_arr = rng.rand(n_points)
        init_h_arr = np.clip(init_h_arr, 0.0, 1.0)
        init_h_arr = adjust_sum_to_target(init_h_arr, target_sum, rng)

   best_h = init_h_arr.copy()
   best_c5 = compute_c5(best_h)
   current_h = init_h_arr.copy()
   current_c5 = best_c5

   # Initialize schedule parameters
   T_start = 0.02
   T_end = 5e-6
   step_start = 0.15
   step_end = 0.001

   # store last improvement time
   last_improve = start

   while True:
       now = time.time()
       if now - start >= max_time:
           break
       progress = (now - start) / max_time

       T = T_start * (1 - progress) + T_end * progress
       step = step_start * (1 - progress) + step_end * progress

       # Choose move
       if rng.rand() < 0.3:
           # gradient move
           corr_full = np.correlate(current_h, 1.0 - current_h, mode='full')
           # use integer shift of max correlation
           k_idx = int(np.argmax(corr_full))
           shift = k_idx - (n_points - 1)  # can be negative

           # compute gradient g
           g = np.zeros_like(current_h)
           idx = np.arange(n_points)
           if shift >= 0:
               plus = idx + shift
               mask_plus = plus < n_points
               g[mask_plus] += (1.0 - current_h[plus[mask_plus]])
               minus = idx - shift
               mask_minus = minus >= 0
               g[mask_minus] -= current_h[minus[mask_minus]]
           else:
               s = -shift
               plus = idx - s
               mask_plus = plus >= 0
               g[mask_plus] += (1.0 - current_h[plus[mask_plus]])
               minus = idx + s
               mask_minus = minus < n_points
               g[mask_minus] -= current_h[minus[mask_minus]]
           # Project gradient to sum-zero hyperplane
           g = g - g.mean()

           candidate = current_h - step * g
           np.clip(candidate, 0.0, 1.0, out=candidate)
           candidate = adjust_sum_to_target(candidate, target_sum, rng)
       else:
           # pairwise exchange move
           i, j = rng.randint(n_points, size=2)
           if i == j:
               continue
           hi, hj = current_h[i], current_h[j]
           low = max(-hi, hj - 1.0)
           high = min(1.0 - hi, hj)
           if low >= high:
               continue
           delta = (rng.rand() * 2.0 - 1.0) * step
           if delta < low:
               delta = low
           elif delta > high:
               delta = high
           candidate = current_h.copy()
           candidate[i] = hi + delta
           candidate[j] = hj - delta

       cand_c5 = compute_c5(candidate)

       # acceptance criteria
       if cand_c5 < current_c5:
           accept = True
       else:
           # if T is zero (should not happen) set zero chance
           prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
           accept = rng.rand() < prob

       if accept:
           current_h = candidate
           current_c5 = cand_c5
           if cand_c5 < best_c5:
               best_c5 = cand_c5
               best_h = candidate.copy()
               last_improve = now

       # occasional random restart if no improvement for awhile
       if now - last_improve > max_time * 0.25:  # 25% of total time without improvement
           # reinitialize from random
           h_new = rng.rand(n_points)
           h_new = np.clip(h_new, 0.0, 1.0)
           h_new = adjust_sum_to_target(h_new, target_sum, rng)
           cand_c5 = compute_c5(h_new)
           # Accept as current if better than current else just reset current to new
           if cand_c5 < current_c5:
               current_h = h_new
               current_c5 = cand_c5
           else:
               current_h = h_new
               current_c5 = cand_c5
           last_improve = now

   # final adjust
   best_h = adjust_sum_to_target(best_h, target_sum, rng)
   final_c5 = compute_c5(best_h)

   return (best_h, float(final_c5), n_points)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x) (1 - h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust h in-place so that its sum equals target while keeping 0 <= h_i <= 1.
    Small residuals are spread uniformly; finally the array is clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    n = h.size
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(n)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Heuristic optimisation for the Erdős minimum overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.1  # small safety margin

    # Initialise
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n_points = init_h.size
        target_sum = n_points / 2.0
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = adjust_sum_to_target(init_h.copy(), target_sum, rng)
    except Exception:
        n_points = 200
        target_sum = n_points / 2.0
        init_h = rng.rand(n_points)
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = adjust_sum_to_target(init_h, target_sum, rng)

    best_h = init_h.copy()
    best_c5 = compute_c5(best_h)
    current_h = init_h.copy()
    current_c5 = best_c5

    # Annealing schedule parameters
    T_start = 0.02
    T_end = 5e-6
    step_start = 0.15
    step_end = 0.001

    last_improve = start

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time

        T = T_start * (1.0 - progress) + T_end * progress
        step = step_start * (1.0 - progress) + step_end * progress

        # Choose move type
        if rng.rand() < 0.35:
            # Gradient move based on the current worst shift
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            shift = k_idx - (n_points - 1)   # shift can be negative

            g = np.zeros_like(current_h)
            idx = np.arange(n_points)
            if shift >= 0:
                plus = idx + shift
                mask_plus = plus < n_points
                g[mask_plus] += (1.0 - current_h[plus[mask_plus]])
                minus = idx - shift
                mask_minus = minus >= 0
                g[mask_minus] -= current_h[minus[mask_minus]]
            else:
                s = -shift
                plus = idx - s
                mask_plus = plus >= 0
                g[mask_plus] += (1.0 - current_h[plus[mask_plus]])
                minus = idx + s
                mask_minus = minus < n_points
                g[mask_minus] -= current_h[minus[mask_minus]]

            # Keep total mass unchanged
            g -= g.mean()
            candidate = current_h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        else:
            # Simple pairwise exchange of mass
            i, j = rng.randint(n_points, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            if delta < low:
                delta = low
            elif delta > high:
                delta = high
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        cand_c5 = compute_c5(candidate)

        # Metropolis acceptance
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                last_improve = now

        # Occasionally force binary discretisation in the late stage
        if progress > 0.75 and rng.rand() < 0.3:
            binary = (current_h > 0.5).astype(float)
            # Enforce exact half‑mass
            target_int = int(round(target_sum))
            cur_int = int(round(binary.sum()))
            diff = cur_int - target_int
            if diff > 0:
                ones_idx = np.where(binary == 1)[0]
                order = np.argsort(current_h[ones_idx])  # smallest original values become 0
                binary[ones_idx[order[:diff]]] = 0
            elif diff < 0:
                zeros_idx = np.where(binary == 0)[0]
                order = np.argsort(-current_h[zeros_idx])  # largest original values become 1
                binary[zeros_idx[order[: -diff]]] = 1
            cand_c5 = compute_c5(binary)
            if cand_c5 < current_c5:
                current_h = binary
                current_c5 = cand_c5
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = binary.copy()
                    last_improve = now

        # Restart if stuck for a while
        if now - last_improve > max_time * 0.25:
            new_h = rng.rand(n_points)
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = adjust_sum_to_target(new_h, target_sum, rng)
            new_c5 = compute_c5(new_h)
            current_h = new_h
            current_c5 = new_c5
            last_improve = now

    # Final correction and result
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
