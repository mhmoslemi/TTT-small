```python
def compute_gradients(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        term1 = np.zeros(n, dtype=h.dtype)
        term2 = np.zeros(n, dtype=h.dtype)
        if shift >= 0:
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad
```

def simulated_annealing(h_start, max_iters, rng, start_time, budget_s):
    n = h_start.size
    cur = h_start.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    # parameters
    T0 = 0.02
    T_end = 1e-6
    # linear schedule
    for it in range(max_iters):
        # time check
        if time.time() - start_time > budget_s - 0.2:
            break
        # choose i,j distinct
        i, j = rng.integers(n, size=2)
        while i == j:
            j = rng.integers(n)
        # choose delta
        max_delta = min(0.1, cur[i])  # cannot move more than cur[i] (down to 0)
        # also cannot exceed 1 - cur[j]
        max_delta = min(max_delta, 1.0 - cur[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # occasionally negative delta (i.e., move from j to i) - but symmetric choice already covers i and j roles.
        # Move from i to j
        new_h = cur.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # no need projection because we kept bounds controlled
        # compute new score
        new_score = compute_c5(new_h)
        # temperature schedule
        t_frac = it / max_iters
        T = T0 * (1 - t_frac) + T_end * t_frac
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new_h
            cur_score = new_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
    return best, best_score
```

def simulated_annealing(h_start, max_iters, rng, start_time, budget_s):
    n = h_start.size
    cur = h_start.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    # temperature schedule linear
    T0 = 0.02
    T_end = 1e-6
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 0.2:
            break
        # propose move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # Determine max delta we can move from i to j
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        # choose delta in (0, max_delta)
        delta = rng.random() * max_delta
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = compute_c5(new)
        # temperature
        t_frac = it / max_iters
        T = T0 * (1 - t_frac) + T_end * t_frac
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
    return best, best_score
```

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time, math
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_active_shifts_and_grad(h):
    """Return gradient of the active (max) correlation terms."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    idxs = np.where(corr >= max_corr - 1e-12)[0]
    shifts = idxs - (n - 1)
    num_shifts = len(shifts)
    grad = np.full_like(h, float(num_shifts))
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            sft = -shift
            grad[sft:] -= h[:n - sft]
            grad[:n - sft] -= h[sft:]
    return grad, max_corr

def projected_gradient_descent_momentum(h0, n_iter=3000, step0=0.8, beta=0.9, rng=None):
    """PGD with momentum on the max-correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        grad, _ = compute_active_shifts_and_grad(h)
        step = step0 / math.sqrt(it)
        v = beta * v + step * grad
        h_new = h - v
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                      dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def average_candidates(candidates):
    """Mean of a list of vectors, projected onto the feasible set."""
    avg = np.mean(candidates, axis=0)
    avg = project_onto_box_sum(avg, avg.size / 2.0)
    return avg

def simulated_annealing(h_start, max_iters, rng, start_time, budget_s):
    """Simple SA that moves mass between two coordinates while keeping sum fixed."""
    n = h_start.size
    cur = h_start.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    T0 = 0.02
    T_end = 1e-6
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 0.2:
            break
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        new_h = cur.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        t_frac = it / max_iters
        T = T0 * (1 - t_frac) + T_end * t_frac
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new_h
            cur_score = new_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    target_n = 256  # main resolution
    candidates = []

    # 1. Initial constructions
    if 'initial_h_values' in globals():
        h0 = upsample(np.asarray(initial_h_values, dtype=float), target_n)
        h0 = project_onto_box_sum(h0, target_n / 2.0)
        candidates.append(h0)

    if (target_n & (target_n - 1)) == 0:  # power of two
        candidates.append(thue_morse(target_n).astype(float))

    candidates.append(np.full(target_n, 0.5, dtype=float))

    for _ in range(5):
        candidates.append(random_binary_balanced(target_n, rng))
    for _ in range(5):
        v = rng.random(target_n)
        candidates.append(project_onto_box_sum(v, target_n / 2.0))

    best_h = None
    best_score = float('inf')
    best_list = []  # store top few candidates (score, vector)

    # 2. Local PGD refinement on each start
    for h0 in candidates:
        if time.time() - start > budget_s * 0.6:
            break
        h_opt, score_opt = projected_gradient_descent_momentum(
            h0, n_iter=3000, step0=0.8, beta=0.9, rng=rng)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
        # maintain a list of up to 10 best
        if len(best_list) < 10:
            best_list.append((score_opt, h_opt.copy()))
            best_list.sort(key=lambda x: x[0])
        else:
            if score_opt < best_list[-1][0]:
                best_list[-1] = (score_opt, h_opt.copy())
                best_list.sort(key=lambda x: x[0])

    # 3. Combine top solutions (averaging)
    if best_list:
        top_hs = [h for _, h in best_list[:5]]
        avg_h = average_candidates(top_hs)
        avg_score = compute_c5(avg_h)
        if avg_score < best_score:
            best_score = avg_score
            best_h = avg_h.copy()

    # 4. Simulated annealing refinement if time permits
    time_left = budget_s - (time.time() - start)
    if best_h is not None and time_left > 5:
        max_iters = int(min(50000, time_left * 30))
        h_sa, score_sa = simulated_annealing(
            best_h.copy(), max_iters, rng, start, budget_s)
        if score_sa < best_score:
            best_score = score_sa
            best_h = h_sa.copy()

    # 5. Final feasibility and return
    if best_h is None:
        best_h = np.full(target_n, 0.5, dtype=float)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
