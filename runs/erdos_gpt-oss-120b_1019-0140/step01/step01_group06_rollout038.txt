def hill_climb_binary(h, max_iter, rng, start_temp, end_temp, time_limit):
    best = h.copy()
    best_val = compute_c5(best)
    n = len(h)
    ones_idx = np.where(best == 1)[0]
    zeros_idx = np.where(best == 0)[0]
    cur = best.copy()
    cur_val = best_val
    iter = 0
    while iter < max_iter:
        # compute temperature given progress
        t = start_temp + (end_temp - start_temp) * (iter / max_iter)
        # pick random indices i in ones, j in zeros
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # propose swap
        cur[i] = 0
        cur[j] = 1
        # evaluate
        val = compute_c5(cur)
        # acceptance criteria
        if val < cur_val or rng.random() < np.exp((cur_val - val) / max(t, 1e-12)):
            # accept (improved or probabilistic)
            cur_val = val
            # update indices
            # i becomes zero, j becomes one
            ones_idx = np.append(ones_idx[ones_idx != i], j)
            zeros_idx = np.append(zeros_idx[zeros_idx != j], i)
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
        else:
            # revert swap
            cur[i] = 1
            cur[j] = 0
        iter += 1
        # time check
        if time.time() > time_limit:
            break
    return best, best_val

import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # interior indices
        interior = np.where((h > 0.0) & (h < 1.0))[0]
        if interior.size == 0:
            break
        delta = diff / interior.size
        h[interior] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust one element
        i = np.argmin(h)  # cheapest addition
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def project_to_binary(x, target_sum):
    idx = np.argpartition(-x, target_sum-1)[:target_sum]
    h = np.zeros_like(x, dtype=float)
    h[idx] = 1.0
    return h

def get_initial_h(n, rng):
    base = globals().get("initial_h_values")
    if base is not None and base.shape[0] == n:
        # use base but make binary
        arr = np.asarray(base, dtype=float)
        h = project_to_binary(arr, n // 2)
    elif base is not None:
        arr = np.asarray(base, dtype=float)
        n0 = arr.shape[0]
        x_base = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x_new, x_base, arr)
        h = project_to_binary(interp, n // 2)
    else:
        # random binary
        indices = rng.choice(n, n // 2, replace=False)
        h = np.zeros(n, dtype=float)
        h[indices] = 1.0
    return h

def local_optimize(h0, n, rng, maxiter=400):
    # SLSQP
    cons = {"type":"eq", "fun": lambda h: np.sum(h) - n/2.0}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            fun=lambda h: compute_c5(h),
            x0=h0,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-9, "disp": False}
        )
    except Exception:
        return None, None
    if not res.success:
        # still use the result
        pass
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n/2.0)
    c5 = compute_c5(h_opt)
    return h_opt, c5

    start = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = np.inf
    best_n = None
    # list of n to explore in increasing order
    n_list = [190, 380, 760]
    # restart counts per n (adjust)
    restart_dict = {190: 80, 380: 30, 760: 10}
    maxiter_dict = {190: 400, 380: 500, 760: 400}
    for n in n_list:
        target_sum = n / 2.0
        max_restarts = restart_dict.get(n, 10)
        for r in range(max_restarts):
            # time check
            if time.time() - start > budget_s * 0.97:
                break
            if r == 0:
                h0 = get_initial_h(n, rng)
            else:
                # random binary with some perturbation
                h0 = get_initial_h(n, rng)
                # optionally flip a few bits from best found at this n if available
                if best_h is not None and best_h.shape[0] == n:
                    # copy best and flip few random indices
                    h0 = best_h.copy()
                    # flip up to 3 pairs
                    for _ in range(rng.integers(1, 4)):
                        # pick a 1 and a 0 to swap
                        ones = np.where(h0 > 0.5)[0]
                        zeros = np.where(h0 < 0.5)[0]
                        if ones.size == 0 or zeros.size == 0:
                            break
                        i = rng.choice(ones)
                        j = rng.choice(zeros)
                        h0[i] = 0.0
                        h0[j] = 1.0
            # ensure feasibility
            h0 = project_to_feasible(h0, target_sum)
            # local optimization
            h_opt, c5_val = local_optimize(h0, n, rng, maxiter=maxiter_dict.get(n,400))
            if h_opt is None:
                continue
            if c5_val < best_c5 - 1e-9:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n
            # early exit if target met
            if best_c5 <= 0.3808:
                # still allow a few more restarts maybe but can break early if time low
                if time.time() - start > budget_s * 0.5:
                    # Enough time, stop early
                    break
        # after finishing restarts for this n, maybe try a refinement at higher n if we found good solution
        if best_h is not None and best_n == n and best_c5 <= 0.381:
            # try to upscale to next resolution(s) within remaining time
            pass
    # fallback if nothing
    if best_h is None:
        n = 190
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    # final projection
    best_h = project_to_feasible(best_h, best_n/2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        interior = np.where((h > 0.0) & (h < 1.0))[0]
        if interior.size == 0:
            break
        delta = diff / interior.size
        h[interior] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust a single entry
        i = np.argmin(h)  # most room to increase if need upward
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def project_to_binary(x, target_sum):
    # target_sum is integer number of ones
    target = int(target_sum)
    if target <= 0:
        return np.zeros_like(x, dtype=float)
    # argpartition to get indices of largest values
    idx = np.argpartition(-x, target - 1)[:target]
    h = np.zeros_like(x, dtype=float)
    h[idx] = 1.0
    return h

def get_initial_h(n, rng):
    base = globals().get("initial_h_values")
    target_sum = n // 2
    if base is not None and isinstance(base, np.ndarray) and base.shape[0] == n:
        arr = np.asarray(base, dtype=float)
        h = project_to_binary(arr, target_sum)
    elif base is not None and isinstance(base, np.ndarray):
        arr = np.asarray(base, dtype=float)
        n0 = arr.shape[0]
        x_base = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x_new, x_base, arr)
        h = project_to_binary(interp, target_sum)
    else:
        # random binary
        indices = rng.choice(n, target_sum, replace=False)
        h = np.zeros(n, dtype=float)
        h[indices] = 1.0
    return h

def interval_params_to_h(params, n):
    # params: [L0,L1,L2,G0,G1,G2]; G3 is implicit
    L = params[:3]
    G = params[3:]
    gap_sum = np.sum(G)
    G3 = 1.0 - gap_sum
    if G3 < 0.0:
        G3 = 0.0
    gaps = np.concatenate([G, [G3]])
    # Build starts and ends
    starts = np.empty(3)
    ends = np.empty(3)
    cur = gaps[0]  # first gap length
    for i in range(3):
        starts[i] = cur
        ends[i] = cur + L[i]
        cur = ends[i] + gaps[i+1]
    # sample points
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n, dtype=float)
    for i in range(3):
        mask = (xs >= starts[i]) & (xs < ends[i])
        h[mask] = 1.0
    return h

def objective_interval(p, n):
    h = interval_params_to_h(p, n)
    h = project_to_feasible(h, n/2.0)
    return compute_c5(h)

def optimize_intervals(n, rng, deadline, target_c5=0.3808):
    best_h = None
    best_c5 = np.inf
    # deterministic start: equal lengths and gaps
    L0 = np.full(3, 1.0/3.0)
    Gfull = np.full(4, 0.25)  # gaps including G3
    params0 = np.concatenate([L0, Gfull[:3]])

    h0 = project_to_feasible(h0, n/2.0)
    c5 = compute_c5(h0)
    if c5 < best_c5:

    # constraints
    eq_cons = {"type":"eq", "fun": lambda p: np.sum(p[:3]) - 1.0}
    ineq_cons = {"type":"ineq", "fun": lambda p: 1.0 - np.sum(p[3:])}
    cons = [eq_cons, ineq_cons]
    bounds = [(0.0, 1.0)] * 6
    # random restarts within time

        # random start
        L = rng.exponential(1.0, size=3)
        L = L / np.sum(L)
        Gfull = rng.exponential(1.0, size=4)
        Gfull = Gfull / np.sum(Gfull)

        return np.zeros_like(x, dtype=float)
    idx = np.argpartition(-x, target - 1)[:target]
    h = np.zeros_like(x, dtype=float)
    h[idx] = 1.0
    return h


def get_initial_h(n, rng):
    """Obtain a feasible binary start, using parent if present."""
    target = n // 2
    base = globals().get("initial_h_values")
    if isinstance(base, np.ndarray) and base.shape[0] == n:
        return project_to_binary(base.astype(float), target)
    if isinstance(base, np.ndarray):
        arr = base.astype(float)
        n0 = arr.shape[0]
        xb = np.linspace(0.0, 2.0, n0, endpoint=False)
        xn = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xn, xb, arr)
        return project_to_binary(interp, target)
    # random binary
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, target, replace=False)] = 1.0
    return h


def interval_params_to_h(params, n):
    """Build binary h from three interval lengths and three gaps."""
    L = params[:3]
    G = params[3:]            # three gaps, fourth (G3) implicit
    G3 = 1.0 - np.sum(G)
    if G3 < 0.0:
        G3 = 0.0
    gaps = np.concatenate([G, [G3]])  # total gaps sum to 1
    starts = np.empty(3)
    ends = np.empty(3)
    cur = gaps[0]               # first gap before first interval
    for i in range(3):
        starts[i] = cur
        ends[i] = cur + L[i]
        cur = ends[i] + gaps[i + 1]
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n, dtype=float)
    for i in range(3):
        mask = (xs >= starts[i]) & (xs < ends[i])
        h[mask] = 1.0
    return h


def objective_interval(p, n):
    h = interval_params_to_h(p, n)
    h = project_to_feasible(h, n / 2.0)
    return compute_c5(h)


def optimize_intervals(n, rng, deadline, target_c5=0.3808):
    """Search over three-interval families using SLSQP restarts."""
    best_h = None
    best_c5 = np.inf

    # deterministic equal‑length start
    L0 = np.full(3, 1.0 / 3.0)
    Gfull = np.full(4, 0.25)
    p0 = np.concatenate([L0, Gfull[:3]])
    h0 = interval_params_to_h(p0, n)
    h0 = project_to_feasible(h0, n / 2.0)
    c5 = compute_c5(h0)
    if c5 < best_c5:
        best_c5, best_h = c5, h0.copy()

    # constraints for SLSQP
    eq_cons = {"type": "eq", "fun": lambda p: np.sum(p[:3]) - 1.0}
    ineq_cons = {"type": "ineq", "fun": lambda p: 1.0 - np.sum(p[3:])}
    cons = [eq_cons, ineq_cons]
    bounds = [(0.0, 1.0)] * 6

    while time.time() < deadline:
        # random Dirichlet draws for lengths and gaps
        L = rng.exponential(1.0, size=3)
        L = L / L.sum()
        G = rng.exponential(1.0, size=4)
        G = G / G.sum()
        p = np.concatenate([L, G[:3]])
        try:
            res = opt.minimize(
                fun=lambda pp: objective_interval(pp, n),
                x0=p,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 200, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue
        # accept result regardless of success flag
        p_opt = res.x
        h_opt = interval_params_to_h(p_opt, n)
        h_opt = project_to_feasible(h_opt, n / 2.0)
        c5_val = compute_c5(h_opt)
        if c5_val < best_c5:
            best_c5, best_h = c5_val, h_opt.copy()
            if best_c5 <= target_c5:
                break
    return best_h, best_c5


def local_optimize(h0, n, maxiter=300):
    """Refine a feasible h with SLSQP."""
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - n / 2.0}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            fun=lambda h: compute_c5(h),
            x0=h0,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return None, None
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)


def binary_optim(n, rng, time_limit, cur_best):
    """Binary‑search fallback using random restarts + local SLSQP."""
    best_h = None
    best_c5 = cur_best
    end = time.time() + time_limit
    while time.time() < end:
        h0 = get_initial_h(n, rng)
        h_opt, c5_val = local_optimize(h0, n, maxiter=300)
        if h_opt is None:
            continue
        if c5_val < best_c5:
            best_c5, best_h = c5_val, h_opt.copy()
            if best_c5 <= 0.3808:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Main entry point – returns (h_values, c5_bound, n_points)."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # primary resolution (large enough to capture fine structure)
    n = 760
    interval_deadline = start + budget_s * 0.75
    best_h, best_c5 = optimize_intervals(n, rng, interval_deadline)

    # If time permits, run a binary‑search fallback
    now = time.time()
    remaining = start + budget_s * 0.97 - now
    if remaining > 5.0 and (best_h is None or best_c5 > 0.3808):
        h2, c5_2 = binary_optim(n, rng, remaining, best_c5 if best_c5 is not None else np.inf)
        if h2 is not None and c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Fallback simple construction if nothing succeeded
    if best_h is None:
        n = 190
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    # Final safety projection and recount
    best_h = project_to_feasible(best_h, best_h.shape[0] / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_h.shape[0]
