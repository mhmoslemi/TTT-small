def project(v, target, eps=1e-10, max_iter=1000):
    # iterative water-filling
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        total = v.sum()
        diff = target - total
        if abs(diff) < eps:
            break
        # indices that can be adjusted
        free = (v > eps) & (v < 1 - eps)
        if free.sum() == 0:
            # all at bounds; distribute to all but ensure not exceed bound
            # see method below
            # fallback: break
            break
        delta = diff / free.sum()
        v[free] += delta
        # Clip again
        np.clip(v, 0.0, 1.0, out=v)
    return v
```

def project_to_constraints(h, target_sum):
    # Clip to bounds
    np.clip(h, 0.0, 1.0, out=h)
    # Compute sum diff
    diff = target_sum - h.sum()
    # If diff is zero, done
    if abs(diff) < 1e-12:
        return h
    # Use water-filling algorithm: adjust free entries
    # Determine free indices not at bounds
    for _ in range(100):
        free = (h > 0.0) & (h < 1.0)
        nfree = free.sum()
        if nfree == 0:
            #all at bounds; we have to distribute to all (including some that will go out of bounds)
            #just add diff equally and clip
            h += diff / len(h)
            np.clip(h, 0.0, 1.0, out=h)
            diff = target_sum - h.sum()
            if abs(diff) < 1e-12:
                break
            continue
        delta = diff / nfree
        # apply delta only to free indices
        h[free] += delta
        # clip any that exceed bounds
        np.clip(h, 0.0, 1.0, out=h)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
    return h
```

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n intersected with sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # water‑filling style projection
    for _ in range(200):
        free = (v > 0.0) & (v < 1.0)
        nfree = free.sum()
        if nfree == 0:

        v[free] += delta
        np.clip(v, 0.0, 1.0, out=v)
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
    if abs(diff) > 1e-12:
        v += diff / v.size
        np.clip(v, 0.0, 1.0, out=v)
    return v

    # annealing schedule
    init_T = 0.6
    final_T = 1e-5
    init_step = 0.2
    final_step = 1e-5

    iter_cnt = 0
    last_improve = 0

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        T = init_T * (1.0 - progress) + final_T * progress
        step = init_step * (1.0 - progress) + final_step * progress

        r = rng.rand()
        if r < 0.45:
            # ---- gradient step on the worst shift ----
            full_corr = np.correlate(current_h, 1.0 - current_h, mode="full")
            lag_idx = int(np.argmax(full_corr))
            k_shift = lag_idx - (n_points - 1)

            g = np.zeros_like(current_h)
            idx = np.arange(n_points)

            plus = idx + k_shift
            mask_plus = (plus >= 0) & (plus < n_points)
            if mask_plus.any():
                g[mask_plus] += (1.0 - current_h[plus[mask_plus]])

            minus = idx - k_shift
            mask_minus = (minus >= 0) & (minus < n_points)
            if mask_minus.any():
                g[mask_minus] -= current_h[minus[mask_minus]]

            norm = np.linalg.norm(g)
            if norm > 0:
                g = g / norm
                candidate = current_h - step * g
            else:
                candidate = current_h.copy()
            candidate = project_to_constraints(candidate, target_sum)

        elif r < 0.75:
            # ---- pairwise exchange ----
            i, j = rng.randint(n_points, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            if delta < low:
                delta = low
            elif delta > high:
                delta = high
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta
            candidate = project_to_constraints(candidate, target_sum)

        else:
            # ---- global jitter ----
            noise = (rng.rand(n_points) - 0.5) * 2.0 * step
            candidate = current_h + noise
            candidate = project_to_constraints(candidate, target_sum)

        cand_c5 = compute_c5(candidate)

        # Simulated‑annealing acceptance
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / max(T, 1e-12))
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                last_improve = iter_cnt

        iter_cnt += 1

        # occasional restart if stuck
        if iter_cnt - last_improve > 50000 and (now - start) < max_time * 0.9:
            jitter = (rng.rand(n_points) - 0.5) * 0.2
            current_h = best_h + jitter
            current_h = project_to_constraints(current_h, target_sum)
            current_c5 = compute_c5(current_h)
            last_improve = iter_cnt

    best_h = project_to_constraints(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
