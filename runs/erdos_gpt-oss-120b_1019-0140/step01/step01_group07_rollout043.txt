def compute_c5(b):
    n = b.size
    total_ones = b.sum()
    L = 1 << ((2*n-1).bit_length())
    fb = np.fft.rfft(b, L)
    # reversed b: b[::-1]
    fbr = np.fft.rfft(b[::-1], L)
    conv = np.fft.irfft(fb * fbr, L)
    # conv length L
    # take indices n-1 to n-1 + n-1 (i.e., 2n-2)
    start = n-1
    M = conv[start:start+n]  # length n
    # Round to nearest integer due to floating errors
    M = np.rint(M).astype(np.int64)
    O = total_ones - M
    # ignore shift 0 (O[0]=0). compute max over rest
    max_O = O[1:].max()
    dx = 2.0 / n
    return max_O * dx

def _compute_c5(h):
    n = h.shape[0]
    target = n // 2
    dx = 2.0 / n

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

    return float(corr.max() * dx)

import numpy as np
import time
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust so that sum(v) == target."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    """Discrete C5: max_k ∫ h(x)*(1‑h(x+k))dx approximated on a grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _random_vector(n, target):
    v = np.random.rand(n)
    return _project_to_constraints(v, target)

    np.random.seed(seed)

    # ---------- initial vector ----------
    init_vec = globals().get('initial_h_values', None)

            init_vec = init_vec[:-1]         # force even length
        n = init_vec.size
        target_sum = n / 2.0
        init_h = _project_to_constraints(init_vec, target_sum)
    else:
        n = 380
        target_sum = n / 2.0
        init_h = _random_vector(n, target_sum)

    # ---------- GA parameters ----------
    pop_size = 30
    tournament_k = 3
    max_generations = 20000
    sigma_initial = 0.12
    sigma_final = 0.015

    population = [init_h.copy()]

    ramp_vec = np.linspace(0.0, 1.0, n)
