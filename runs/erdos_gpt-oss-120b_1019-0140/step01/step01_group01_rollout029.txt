def gradient_logsumexp(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    # compute weights
    max_corr = np.max(corr)  # for numerical stability may subtract...
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)
    # gradient init with ones
    g = np.ones_like(h)
    # loop
    for idx, weight in enumerate(w):
        shift = idx - (n - 1)
        if shift >= 0:
            if shift < n:
                # forward term: for i where i+shift < n
                g[:n - shift] -= weight * h[shift:]
                # backward term: for i where i-shift >=0 => i>=shift
                g[shift:] -= weight * h[:n - shift]
        else:
            # negative shift
            k = -shift
            # forward term if shift negative => i+shift = i - k; for valid i where i - k >=0 => i>=k
            # Equivalent to backward contribution for positive shift k?

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Scale / tweak h in-place so that sum(h) == target while staying in [0,1]."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # try to fix the whole deficit/surplus using a few random indices
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_smooth(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of smooth-approximation of max_k corr_k, with zero-mean."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    # stabilise exponentials
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    w = exp_vals / exp_vals.sum()
    g = np.ones_like(h, dtype=float)  # start with constant 1 term
    # loop over all shifts
    for idx, weight in enumerate(w):
        shift = idx - (n - 1)
        if shift >= 0:
            if shift < n:
                # forward contribution: i -> i+shift
                g[: n - shift] -= weight * h[shift:]
                # backward contribution: i -> i-shift
                g[shift:] -= weight * h[: n - shift]
        else:
            k = -shift
            # forward (i+shift) = i - k
            g[k:] -= weight * h[: n - k]
            # backward (i-shift) = i + k
            g[: n - k] -= weight * h[k:]
    # gradient (zero-sum) multiplied by alpha
    g *= alpha
    # normalise magnitude for stable step size
    max_abs = np.max(np.abs(g))
    if max_abs > 0:
        g /= max_abs
    return g

    except Exception:
        base_len = 70
        target_base = base_len / 2.0
        base = rng.rand(base_len)
        base = base / base.sum() * target_base
        base = adjust_sum_to_target(base, target_base, rng)

    # upscale to allow finer adjustments (repeat each entry twice)
    repeat = 2
    n_points = base_len * repeat
    h = np.repeat(base, repeat)
    target_sum = n_points / 2.0
    h = adjust_sum_to_target(h, target_sum, rng)

    # best-so-far
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # annealing schedules
    T0, T_end = 0.1, 1e-6
    step0, step_end = 0.5, 1e-5
    alpha0, alpha_end = 0.5, 10.0

    # early‑stop helper
    it_without_improve = 0

    while True:

            current_c5 = compute_c5(cand)

    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
