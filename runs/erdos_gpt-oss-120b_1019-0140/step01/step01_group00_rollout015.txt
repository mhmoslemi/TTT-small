    T0 = 0.001  # maybe start high

            break

def simulated_annealing_binary(h, cur_val, start_time, budget_s):
    n = len(h)
    cur_h = h.copy()
    cur_best_h = cur_h.copy()
    cur_best_val = cur_val
    rng = np.random.default_rng()
    # define max iterations adaptively based on time left
    while True:
        if time.time() - start_time >= budget_s - 0.5:
            break
        elapsed = time.time() - start_time
        t_frac = elapsed / budget_s
        # temperature schedule: start at 0.01, decay to 0.0001
        T = 0.01 * (1.0 - t_frac) + 0.0001
        # random swap
        # choose one index from ones and one from zeros
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_val = compute_c5(cur_h)[0]
        delta = new_val - cur_val
        accept = False
        if delta < 0:
            accept = True
        else:
            # acceptance probability
            if np.exp(-delta / T) > rng.random():
                accept = True
        if accept:
            cur_val = new_val
            if cur_val < cur_best_val - 1e-12:
                cur_best_val = cur_val
                cur_best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        # occasionally try steepest descent to refine local optimum
        # e.g., if t_frac > 0.7 maybe skip
        if t_frac > 0.6 and rng.random() < 0.01:
            # try local steepest descent from current configuration
            cur_h, cur_val = steepest_descent_binary(cur_h, start_time, budget_s)
    return cur_best_h, cur_best_val

import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def binary_random(n, rng):
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1
    return h

def binary_from_continuous(h_cont):
    n = len(h_cont)
    idx = np.argsort(-h_cont)[: n // 2]
    h = np.zeros(n, dtype=int)
    h[idx] = 1
    return h

def steepest_descent_binary(h, start_time, budget_s):

    best_h = h.copy()
    best_val = compute_c5(best_h)

            break

        zeros = np.flatnonzero(1 - best_h)
        best_swap_val = best_val
        best_swap = None
        for i in ones:
            for j in zeros:
                cand = best_h.copy()
                cand[i] = 0
                cand[j] = 1
                val = compute_c5(cand)

                    best_swap_val = val

            best_h[i] = 0
            best_h[j] = 1
            best_val = best_swap_val

            break
