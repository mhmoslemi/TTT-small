def _simulated_annealing(h_start, time_limit, rng, T0=0.5, max_eps=0.3):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    attempts_without_improve = 0
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        frac = 1.0 - elapsed / time_limit
        T = T0 * max(frac, 0.01)   # keep min temperature
        # randomly pick two indices
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi, hj = h_cur[i], h_cur[j]
        # compute allowable epsilon
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        eps = (rng.random() * (up + down) - down) * T / T0 # scale eps with temperature
        # limit magnitude
        max_move = max_eps * T / T0
        if eps > max_move:
            eps = max_move
        elif eps < -max_move:
            eps = -max_move
        if np.abs(eps) < 1e-12:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # clipping not needed if eps bounds ensure within [0,1]
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h_cur = h_new
            cur_score = new_score
            attempts_without_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            attempts_without_improve += 1
            if attempts_without_improve >= 10000:
                # reheat
                attempts_without_improve = 0
                # maybe random perturb
                # not essential
                pass
    return best_h, best_score

def _hill_climb(h, rng, max_iter=10000):
    N = h.size
    best = h.copy()
    best_score = _c5_value(best)
    for _ in range(max_iter):
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi, hj = best[i], best[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0 and down == 0:

        eps = rng.uniform(-down, up) * 0.1  # small

        if eps == 0.0:

        new = best.copy()
        new[i] = hi + eps
        new[j] = hj - eps
        new_score = _c5_value(new)
        if new_score < best_score:
            best = new
            best_score = new_score
    return best, best_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) = len(v)/2."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v.fill(target / v.size)
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    for _ in range(5):
        s = v.sum()
        if abs(s - target) <= 1e-12:
            break
        factor = target / s
        v *= factor
        np.clip(v, 0.0, 1.0, out=v)
    return v


def _c5_value(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _simulated_annealing(h_start: np.ndarray, time_limit: float,
                         rng: np.random.Generator,
                         T0: float = 0.5, max_eps: float = 0.5):
    """Plain SA that preserves the total mass exactly."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * max(frac, 0.01)               # never fully freeze
        i, j = rng.integers(0, N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)                # how much we can increase i
        down = min(hi, 1.0 - hj)              # how much we can decrease i
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        # scale step by temperature (larger T → larger moves)
        eps *= T / T0
        if eps > max_eps:
            eps = max_eps
        elif eps < -max_eps:
            eps = -max_eps
        if abs(eps) < 1e-12:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # the construction ensures bounds, but a final clip guards numeric drift
        np.clip(h_new, 0.0, 1.0, out=h_new)
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score


def _hill_climb(h_start: np.ndarray, time_limit: float,
                rng: np.random.Generator):
    """Greedy descent (always accept improving moves)."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while time.time() - start < time_limit:
        i, j = rng.integers(0, N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        eps *= 0.1                                   # small fine‑tune step
        if abs(eps) < 1e-12:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        np.clip(h_new, 0.0, 1.0, out=h_new)
        new_score = _c5_value(h_new)
        if new_score < cur_score:
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score


def _sinusoid_prototype(n: int) -> np.ndarray:
    """Zero‑at‑ends bump h(x)=a*sin(pi*x/2) with ∫h=1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    a = math.pi / 4.0                       # gives integral 1 before repair
    h = a * np.sin(math.pi * x / 2.0)
    return _repair_vector(h)


def _interpolate_to(v: np.ndarray, new_n: int) -> np.ndarray:
    old_n = v.size
    if old_n == new_n:
        return v.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    v_new = np.interp(x_new, x_old, v)
    return _repair_vector(v_new)
