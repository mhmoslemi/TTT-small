```python
import numpy as np, time
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    # clip
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) <= 1e-9:
        return v
    if cur < target:
        need = target - cur
        # distribute to indices where v < 1
        free = np.where(v < 1)[0]
        # capacities
        caps = 1.0 - v[free]
        # sort by capacity descending to fill the biggest first (or any order)
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    # final clip and scaling if still off
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v
```

```python
def _c5_score(h):
    n = h.size
    dx = 2.0 / n
    # using FFT for full correlation
    n_fft = 1 << ((2*n - 1).bit_length())
    a = np.fft.rfft(h, n=n_fft)
```

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust sum(v) to be exactly target."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        if free.size > 0:
            caps = 1.0 - v[free]
            order = np.argsort(caps)[::-1]
            for idx in order:
                i = free[idx]
                add = min(need, caps[idx])
                v[i] += add
                need -= add
                if need <= 1e-12:
                    break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        if free.size > 0:
            caps = v[free]
            order = np.argsort(caps)[::-1]
            for idx in order:
                i = free[idx]
                sub = min(excess, caps[idx])
                v[i] -= sub
                excess -= sub
                if excess <= 1e-12:
                    break
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

        v = _project_to_constraints(v, target_sum)
        pop.append(v)
    return np.array(pop, dtype=np.float64)

def _tournament(pop, scores, size):
    idx = np.random.choice(len(pop), size=size, replace=False)
    best_idx = idx[np.argmin(scores[idx])]
    return pop[best_idx]

def _crossover(p1, p2):
    """Uniform crossover."""
    mask = np.random.rand(p1.size) < 0.5
    return np.where(mask, p1, p2)

def _mutate(v, sigma, prob):
    mask = np.random.rand(v.size) < prob
    noise = np.random.randn(v.size) * sigma
    v_new = v.copy()
    v_new[mask] += noise[mask]
    return v_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Prepare initial parent if available
    init_parent = globals().get("initial_h_values", None)
    if init_parent is not None:
        init_parent = np.asarray(init_parent, dtype=np.float64).ravel()
        base_n = init_parent.size
    else:
        base_n = 380
        init_parent = None

    # Resolution schedule (doubling)
    resolutions = [base_n]
    for _ in range(3):
        resolutions.append(resolutions[-1] * 2)

    best_h = None
    best_score = float("inf")

    def time_left():
        return budget_s - (time.time() - start_time)

    # ----------------------------------------------------------------------
    # Main loop over resolutions
    for n in resolutions:
        if time_left() < 30:
            break  # not enough time for another stage

        target_sum = n / 2.0

        # Upsample previous best or the initial parent
        if best_h is not None:
            x_old = np.arange(best_h.size) * (2.0 / best_h.size)
            x_new = np.arange(n) * (2.0 / n)
            start_vec = np.interp(x_new, x_old, best_h)
        elif init_parent is not None and init_parent.size == n:
            start_vec = init_parent.copy()
        elif init_parent is not None:
            x_old = np.arange(init_parent.size) * (2.0 / init_parent.size)
            x_new = np.arange(n) * (2.0 / n)
            start_vec = np.interp(x_new, x_old, init_parent)
        else:
            start_vec = np.random.rand(n)

        start_vec = _project_to_constraints(start_vec, target_sum)

        # ------------------------------------------------------------------
        # GA hyper‑parameters (scaled with remaining time)
        remaining = time_left()
        # allocate most time to the first (coarsest) resolution
        if n == base_n:
            pop_size = min(80, max(30, int(remaining // 10)))
            max_gens = min(300, max(80, int(remaining // (pop_size * 0.05))))
        else:
            pop_size = min(60, max(20, int(remaining // 20)))
            max_gens = min(200, max(30, int(remaining // (pop_size * 0.07))))

        elitism = max(2, pop_size // 20)
        tournament_sz = 3
        sigma = 0.12
        mut_prob = 0.25

        # ------------------------------------------------------------------
        # Initialise population
        population = _initialize_population(pop_size, n, target_sum, base_solution=start_vec)
        scores = np.array([_c5_score(ind) for ind in population])
        order = np.argsort(scores)
        population = population[order]
        scores = scores[order]

        best_local_h = population[0].copy()
        best_local_score = scores[0]

        # ------------------------------------------------------------------
        # Evolutionary loop
        for gen in range(max_gens):
            if time_left() < 5:
                break

            new_pop = population[:elitism].tolist()  # keep elites

            while len(new_pop) < pop_size:
                p1 = _tournament(population, scores, tournament_sz)
                p2 = _tournament(population, scores, tournament_sz)
                child = _crossover(p1, p2)
                child = _mutate(child, sigma, mut_prob)
                child = _project_to_constraints(child, target_sum)
                new_pop.append(child)

            population = np.array(new_pop, dtype=np.float64)
            scores = np.array([_c5_score(ind) for ind in population])
            order = np.argsort(scores)
            population = population[order]
            scores = scores[order]

            if scores[0] + 1e-12 < best_local_score:
                best_local_score = scores[0]
                best_local_h = population[0].copy()
                if best_local_score < 0.3808:
                    break

            # decay mutation magnitude
            sigma = max(0.02, sigma * 0.97)

        # ------------------------------------------------------------------
        # Keep the best found at this resolution
        if best_local_score < best_score:
            best_score = best_local_score
            best_h = best_local_h.copy()

        # Local refinement with gradient solver if time permits
        if time_left() > 15:
            def eq_sum(x):
                return np.sum(x) - target_sum
            cons = {"type": "eq", "fun": eq_sum}
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(_c5_score, best_h, method="SLSQP",
                               bounds=bounds, constraints=[cons],
                               options={"maxiter": 250, "ftol": 1e-9, "disp": False})
                if res.success:
                    refined = _project_to_constraints(res.x, target_sum)
                    refined_score = _c5_score(refined)
                    if refined_score < best_score:
                        best_score = refined_score
                        best_h = refined.copy()
            except Exception:
                pass

        if best_score < 0.3808:
            break  # target reached, stop further refinements

    # ----------------------------------------------------------------------
    # Fallback safety
    if best_h is None:
        # simple half‑ones construction
        best_h = np.zeros(base_n, dtype=np.float64)
        best_h[: base_n // 2] = 1.0
        best_h = _project_to_constraints(best_h, base_n / 2.0)
        best_score = _c5_score(best_h)

    # Evaluate with provided external verifier if present
    final_score = best_score
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            pass

    n_final = int(best_h.size)
    return best_h.astype(np.float64), float(final_score), n_final
