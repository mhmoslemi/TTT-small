def compute_gradient_smooth(h, w):
    n = h.size
    grad = np.full(n, 1.0, dtype=h.dtype)
    # shift range - (n-1) .. (n-1)
    for offset, weight in enumerate(w):
        s = offset - (n - 1)
        if weight == 0:
            continue
        if s >= 0:
            # forward shift: h[i+s] term
            grad[:n - s] -= weight * h[s:]
            # backward shift: h[i-s] term
            grad[s:] -= weight * h[:n - s]
        else:
            sneg = -s
            # forward shift for negative s: h[i+s] = h[i - sneg]
            grad[sneg:] -= weight * h[:n - sneg]
            # backward shift: h[i-s] = h[i + sneg]
            grad[:n - sneg] -= weight * h[sneg:]
    return grad

def smooth_pgd(h0, n_iter, step0, alpha0, alpha_max):
    h = h0.copy()
    n = len(h)
    s_target = n/2
    best_h = h.copy()
    best_score = compute_c5(h)  # actual max.

    for it in range(1, n_iter+1):
        # compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        # compute smoothed max
        alpha = min(alpha_max, alpha0 + (alpha_max - alpha0) * it / n_iter)
        w = np.exp(alpha * (corr - np.max(corr)))
        w /= np.sum(w)
        # compute gradient
        grad = compute_gradient_weighted(h, w)  # using function above.
        # step size (maybe diminishing)
        eta = step0 / np.sqrt(it)
        # gradient descent: we want to minimize smooth max => h = h - eta * grad
        h_new = h - eta * grad
        # project to box [0,1] and sum constraint
        h_new = project_onto_box_sum(h_new, s_target)
        # evaluate actual max correlation
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()

    for it in range(1, n_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # smoothing weight
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (n_iter - 1)
        # To avoid overflow, subtract max before exp
        max_corr = np.max(corr)
        exp_weights = np.exp(alpha * (corr - max_corr))

        grad = np.full(n, 1.0, dtype=h.dtype)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    # Euclidean projection onto intersection of hyperplane sum(x)=s and box [lo, hi].
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # The solution is x_i = clip(v_i - lam, lo, hi) where lam chosen so sum = s.
    # Binary search lam.
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # define f(lam) = sum clip(v - lam, lo, hi)

        x = np.clip(v - lam_mid, lo, hi)
        sum_x = x.sum()
        if np.abs(sum_x - s) < 1e-12:
            return x
        if sum_x > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time, math
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam_mid, lo, hi)
        cur = x.sum()
        if abs(cur - s) < 1e-12:
            return x
        if cur > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Exact max overlap (without smoothing)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def smooth_pgd(h0, n_iter=2000, step0=0.5,
               alpha_start=1.0, alpha_end=30.0, time_limit=None):
    """Projected gradient descent on a smooth approximation of the max overlap."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        if time_limit is not None and time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # smoothing weight (softmax on correlation values)
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        max_corr = np.max(corr)
        w = np.exp(alpha * (corr - max_corr))
        w /= np.sum(w)
        # weighted gradient
        grad = np.full(n, 1.0, dtype=h.dtype)
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            s = idx - (n - 1)
            if s >= 0:
                if n - s > 0:
                    grad[:n - s] -= weight * h[s:]
                if s < n:
                    grad[s:] -= weight * h[:n - s]
            else:
                s = -s
                if n - s > 0:
                    grad[s:] -= weight * h[:n - s]
                if s < n:
                    grad[:n - s] -= weight * h[s:]
        eta = step0 / math.sqrt(it)
        h = h - eta * grad
        h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score
