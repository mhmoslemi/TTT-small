import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_smooth_c5(h, alpha=10.0):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use max of exp; to avoid overflow we subtract max:
    m = np.max(alpha * corr)
    sm = np.log(np.sum(np.exp(alpha * corr - m))) / alpha + m / alpha
    return sm * (2.0 / n)

def project_to_feasible(h, target_sum):
    # clip to [0,1]; then adjust to meet target sum via simple scheme
    h = np.clip(h, 0.0, 1.0)
    # Compute deficit/excess
    cur = np.sum(h)
    # iterative adjustment: adjust free variables (not at bounds)
    # We'll do simple water-filling:
    for _ in range(10):
        diff = target_sum - cur
        if abs(diff) < 1e-10:
            break
        # free indices: those not at bounds
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
    # If still off due to small rounding, fix the element furthest from bound
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-7:
        # adjust the element with most slack
        slack = np.where(h < 0.5, h, 1.0 - h)  # distance to nearest bound
        i = np.argmax(slack)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.size
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def local_optimize(h0, maxiter=400, use_smooth=True):

    h_opt = np.clip(res.x, 0.0, 1.0)

    resolutions = [95, 190, 380, 760, 1520]

        if time.time() - start > budget_s * 0.96:
            break

        cur_h = resample_h(best_h, n)
        target_sum = n / 2.0

        # choose number of restarts based on size
        if n <= 190:
            restarts = 30
        elif n <= 380:
            restarts = 20
        elif n <= 760:
            restarts = 12
        else:
            restarts = 6

        for r in range(restarts):
            if time.time() - start > budget_s * 0.96:
                break
            if r == 0:
                h_start = cur_h
            else:
                noise = rng.normal(scale=0.07, size=n)
                h_start = np.clip(cur_h + noise, 0.0, 1.0)
                h_start = project_to_feasible(h_start, target_sum)

            # smooth phase (helps navigate the max)
            h_opt, val = local_optimize(h_start, maxiter=800, use_smooth=True)
            # final refinement on the true objective
            h_opt, val = local_optimize(h_opt, maxiter=400, use_smooth=False)

            if val < best_val:
                best_val = val
                best_h = h_opt.copy()
                best_n = n

        # hill climbing on the current resolution (use best of this size if available)
        hc_start = best_h if best_h.shape[0] == n else cur_h
        steps = 2500 if n <= 380 else 1000
        scale = 0.04 if n <= 380 else 0.03
        h_hc, val_hc = random_hill_climb(hc_start, steps, scale, n / 2.0, rng)
        if val_hc < best_val:
            best_val = val_hc
            best_h = h_hc.copy()
            best_n = n

    # final safety projection and evaluation
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_val = compute_c5(best_h)
    return best_h, best_val, best_n
