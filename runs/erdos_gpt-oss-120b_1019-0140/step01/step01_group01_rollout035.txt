import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def correlation_counts_int(h):
    # h binary float array 0/1
    hi = h.astype(np.int64)
    return np.correlate(hi, 1 - hi, mode='full')

def binary_initial(n, target_sum, rng):
    # try to use global initial_h_values
    try:
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            # adjust size
            if init.size > n:
                init = init[:n]
            else:
                # pad with zeros
                init = np.concatenate([init, np.zeros(n - init.size)])
        # pick top target_sum values
        idx = np.argpartition(init, -target_sum)[-target_sum:]
        h = np.zeros(n, dtype=np.float64)
        h[idx] = 1.0
        return h
    except Exception:
        idx = rng.choice(n, target_sum, replace=False)
        h = np.zeros(n, dtype=np.float64)
        h[idx] = 1.0
        return h

def greedy_optimize(h, rng, time_deadline):
    n = h.shape[0]
    while time.time() < time_deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        if max_corr == 0:
            break
        k_idx = int(np.argmax(corr))
        k_shift = k_idx - (n - 1)
        if k_shift >= 0:
            i_range = np.arange(n - k_shift)
            mask_i = (h[i_range] == 1) & (h[i_range + k_shift] == 0)
            mask_j = (h[i_range] == 0) & (h[i_range + k_shift] == 1)
            i_candidates = i_range[mask_i]
            j_candidates = i_range[mask_j]
        else:
            shift = -k_shift
            i_range = np.arange(shift, n)
            mask_i = (h[i_range] == 1) & (h[i_range - shift] == 0)
            mask_j = (h[i_range] == 0) & (h[i_range - shift] == 1)
            i_candidates = i_range[mask_i]
            j_candidates = i_range[mask_j]
        if i_candidates.size == 0 or j_candidates.size == 0:
            # cannot improve this shift, break out
            break
        i = int(rng.choice(i_candidates))
        j = int(rng.choice(j_candidates))
        # swap
        h[i] = 0.0
        h[j] = 1.0
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.1  # safety margin

    # Determine problem size
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n_points = init_arr.size
    except Exception:
        n_points = 70  # fallback size
    # enforce even n
    if n_points % 2 != 0:
        n_points += 1

    target_sum = n_points // 2

    # Initial solution
    current_h = binary_initial(n_points, target_sum, rng)
    current_val = compute_c5(current_h)
    best_h = current_h.copy()
    best_val = current_val

    # annealing parameters
    T0 = 0.5
    Tmin = 1e-6
    restart_interval = 7.0  # seconds without improvement before restart
    last_improve = time.time()

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        # update temperature
        progress = (now - start) / max_time
        T = T0 * (1.0 - progress) + Tmin * progress

        # propose a swap
        ones = np.flatnonzero(current_h > 0.5)
        zeros = np.flatnonzero(current_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            # reset
            current_h = binary_initial(n_points, target_sum, rng)
            current_val = compute_c5(current_h)
            continue
        i = rng.choice(ones)
        j = rng.choice(zeros)
        candidate = current_h.copy()
        candidate[i] = 0.0
        candidate[j] = 1.0
        cand_val = compute_c5(candidate)

        if cand_val < current_val or rng.rand() < np.exp(-(cand_val - current_val) / T):
            current_h = candidate
            current_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = candidate.copy()
                last_improve = now

        # restart if stagnation
        if now - last_improve > restart_interval:
            current_h = binary_initial(n_points, target_sum, rng)
            current_val = compute_c5(current_h)
            last_improve = now

    # post‑processing greedy refinement on best solution
    deadline = start + max_time
    best_h = greedy_optimize(best_h, rng, deadline)

    # final evaluation
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)

import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős overlap bound C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def binary_initial(n: int, target_sum: int, rng: np.random.RandomState) -> np.ndarray:
    """Create a binary start vector of length n with exactly target_sum ones."""
    try:
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            if init.size > n:
                init = init[:n]
            else:
                init = np.concatenate([init, np.zeros(n - init.size, dtype=float)])
        idx = np.argpartition(init, -target_sum)[-target_sum:]
        h = np.zeros(n, dtype=np.float64)
        h[idx] = 1.0
        return h
    except Exception:
        idx = rng.choice(n, target_sum, replace=False)
        h = np.zeros(n, dtype=np.float64)
        h[idx] = 1.0
        return h

def greedy_optimize(h: np.ndarray, rng: np.random.RandomState, deadline: float) -> np.ndarray:
    """
    Greedy sweeps that repeatedly pick the current worst shift and swap a mismatched
    (1,0) pair with a complementary (0,1) pair, reducing the maximal overlap.
    Stops when no improving swap exists or time runs out.
    """
    n = h.shape[0]
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        if max_corr == 0:
            break
        k_idx = int(np.argmax(corr))
        k_shift = k_idx - (n - 1)

        if k_shift >= 0:
            rngange = np.arange(n - k_shift)
            mask_i = (h[rngange] == 1) & (h[rngange + k_shift] == 0)
            mask_j = (h[rngange] == 0) & (h[rngange + k_shift] == 1)
            i_cand = rngange[mask_i]
            j_cand = rngange[mask_j]
        else:
            shift = -k_shift
            rngange = np.arange(shift, n)
            mask_i = (h[rngange] == 1) & (h[rngange - shift] == 0)
            mask_j = (h[rngange] == 0) & (h[rngange - shift] == 1)
            i_cand = rngange[mask_i]
            j_cand = rngange[mask_j]

        if i_cand.size == 0 or j_cand.size == 0:
            break

        i = int(rng.choice(i_cand))
        j = int(rng.choice(j_cand))
        h[i] = 0.0
        h[j] = 1.0
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.1  # safety margin

    # ---- problem size -------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n_points = init_arr.size
    except Exception:
        n_points = 70
    if n_points % 2 != 0:
        n_points += 1
    target_sum = n_points // 2

    # ---- initial solution ---------------------------------------------
    current_h = binary_initial(n_points, target_sum, rng)
    current_val = compute_c5(current_h)
    best_h = current_h.copy()
    best_val = current_val

    # ---- simulated annealing parameters --------------------------------
    T0 = 0.5
    Tmin = 1e-6
    restart_gap = 7.0               # seconds without improvement before restart
    last_improve = time.time()

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        T = T0 * (1.0 - progress) + Tmin * progress

        # propose a random 1↔0 swap
        ones = np.flatnonzero(current_h > 0.5)
        zeros = np.flatnonzero(current_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            current_h = binary_initial(n_points, target_sum, rng)
            current_val = compute_c5(current_h)
            continue
        i = rng.choice(ones)
        j = rng.choice(zeros)
        candidate = current_h.copy()
        candidate[i] = 0.0
        candidate[j] = 1.0
        cand_val = compute_c5(candidate)

        if cand_val < current_val or rng.rand() < np.exp(-(cand_val - current_val) / T):
            current_h = candidate
            current_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = candidate.copy()
                last_improve = now

        # restart if stuck for a while
        if now - last_improve > restart_gap:
            current_h = binary_initial(n_points, target_sum, rng)
            current_val = compute_c5(current_h)
            last_improve = now

    # ---- greedy refinement on the best candidate -----------------------
    deadline = start + max_time
    best_h = greedy_optimize(best_h, rng, deadline)

    # ---- final result --------------------------------------------------
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
