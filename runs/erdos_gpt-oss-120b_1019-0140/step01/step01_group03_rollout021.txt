def simulated_annealing_binary(h_init, time_limit, start_time):
    h = h_init.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.shape[0]
    # create list of indices with 1s and 0s
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    # base parameters
    T0 = 0.1
    T_end = 1e-4
    max_iters = 500000  # but break early on time
    for it in range(1, max_iters+1):
        if time.time() - start_time > time_limit:
            break
        # temperature schedule
        t_frac = it / max_iters
        T = T0 * (T_end/T0)**t_frac
        # pick random swap
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        idx_one = ones[i]
        idx_zero = zeros[j]
        # swap
        h[idx_one] = 0
        h[idx_zero] = 1
        # compute new score
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta/T):
            # accept
            cur_score = new_score
            # update ones/zeros lists
            ones[i] = idx_zero
            zeros[j] = idx_one
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # reject, swap back
            h[idx_one] = 1
            h[idx_zero] = 0
        # maybe occasional shuffle recompute indices? Not needed.
    return best_h, best_score
```

def proj_subgradient_descent(h0, n_iter, lr0=0.5, momentum=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    s = h.shape[0] / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter+1):
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (h.size - 1)
        grad = compute_gradient(h, shift)
        # subgradient step
        lr = lr0 / np.sqrt(it)
        v = momentum * v - lr * grad
        h_new = h + v
        # projection
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def adam_descent(h0, n_iter, lr0=0.1):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    s = h.size / 2.0
    best_h = h.copy()

def resample(arr, new_n):
    if arr.shape[0] == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, arr.shape[0], endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

```
def compute_gradient(h, shift):
    n = h.shape[0]
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2
```

def simulated_annealing_binary(h_init, deadline, rng):
    h = h_init.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.shape[0]
    # maintain indices of ones and zeros as numpy arrays
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    # schedule: max_iters set to something large, break if time exceeded
    max_iters = 1000000
    T0 = 0.1
    T_end = 1e-5
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # temperature
        t_frac = it / max_iters
        T = T0 * (T_end / T0) ** t_frac
        # pick random pair
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.integers(ones.size)
        j = rng.integers(zeros.size)
        idx_one = ones[i]
        idx_zero = zeros[j]
        # swap
        h[idx_one] = 0.0
        h[idx_zero] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            # swap indices in arrays
            ones[i] = idx_zero
            zeros[j] = idx_one
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # revert swap
            h[idx_one] = 1.0
            h[idx_zero] = 0.0
    return best_h, best_score

import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k)) dx using discrete samples."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Subgradient for the shift that attains the max correlation."""
    n = h.shape[0]
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def momentum_subgradient_descent(h0, n_iter=15000, lr0=0.5, momentum=0.9):
    """Projected momentum subgradient descent for the max‑overlap objective."""
    h = h0.copy()
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    dx = 2.0 / n
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        lr = lr0 / math.sqrt(it)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, s)
        score = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    """Generate Thue–Morse binary sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Balanced random binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def resample(arr, new_n):
    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def simulated_annealing_binary(h_init, deadline, rng):
    """Binary SA to reduce C5 while preserving exactly half ones."""
    h = h_init.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.shape[0]
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    max_iters = 800000
    T0, T_end = 0.12, 1e-5
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        if ones.size == 0 or zeros.size == 0:
            break
        t_frac = it / max_iters
        T = T0 * (T_end / T0) ** t_frac
        i = rng.integers(ones.size)
        j = rng.integers(zeros.size)
        idx_one = ones[i]
        idx_zero = zeros[j]
        # propose swap
        h[idx_one] = 0.0
        h[idx_zero] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            ones[i] = idx_zero
            zeros[j] = idx_one
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # revert
            h[idx_one] = 1.0
            h[idx_zero] = 0.0
    return best_h, best_score

def hill_climbing_binary(h_init, max_attempts, rng):
    """Pure descent: accept swaps only if they improve C5."""
    h = h_init.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.shape[0]
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _ in range(max_attempts):
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.integers(ones.size)
        j = rng.integers(zeros.size)
        idx_one = ones[i]
        idx_zero = zeros[j]
        h[idx_one] = 0.0
        h[idx_zero] = 1.0
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            ones[i] = idx_zero
            zeros[j] = idx_one
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            h[idx_one] = 1.0
            h[idx_zero] = 0.0
    return best_h, best_score

    start = time.time()
    margin = 2.0
    deadline = start + budget_s - margin

    # target resolution
    target_n = 256

    # Prepare initial candidates
    candidates = []

    for _ in range(5):

    best_h = None
    best_score = float('inf')

    # Stage 1: brief momentum descent on each candidate
    for base in candidates:

        h_opt, sc_opt = momentum_subgradient_descent(base, n_iter=12000, lr0=0.5, momentum=0.9)
        if sc_opt < best_score:
