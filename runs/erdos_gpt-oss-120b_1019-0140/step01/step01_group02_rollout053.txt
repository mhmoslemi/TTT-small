import time, math, numpy as np

def _compute_c5(h):
    n=len(h)
    dx=2.0/n
    corr=np.correlate(h,1.0 - h,mode='full')
    return float(np.max(corr)*dx)

def _repair_sum(h,target):
    cur=h.sum()
    diff=target-cur
    if abs(diff)<1e-12:
        return h
    # Add diff to first non-boundary element
    if diff>0:
        idx=np.where(h<1.0-1e-12)[0]
    else:
        idx=np.where(h>1e-12)[0]
    if idx.size==0:
        h+=diff/h.size
        np.clip(h,0.0,1.0,out=h)
        return h
    i=idx[0]
    new_val=np.clip(h[i]+diff,0.0,1.0)
    actual=new_val-h[i]
    h[i]=new_val
    if abs(actual-diff)>1e-12:
        return _repair_sum(h,target)
    return h

import time

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h
