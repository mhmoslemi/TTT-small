def mip_solve(n, time_limit):
    h = cp.Variable(boolean=True, shape=n)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n/2]
    # allocate w as dict: w[(i,k)] is variable bool.
    # For memory and speed, we can allocate w as cp.Variable(boolean=True, shape=(n, 2*n-1))
    w = cp.Variable(boolean=True, shape=(n, 2*n-1))
    # For each shift.
    for k in range(-n+1, n):
        if k == 0: continue
        idx = k + n - 1
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                # constraints for w[i, idx] = h[i] * (1 - h[j]) as described
                constraints.append(w[i, idx] <= h[i])
                constraints.append(w[i, idx] <= 1 - h[j])
                constraints.append(w[i, idx] >= h[i] - h[j])
        # sum constraint
        constraints.append(cp.sum(w[:, idx]) <= t)
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)
    except Exception as e:
        # fallback
        return None, None
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # Might be not solved to optimal but feasible; we can still try to check values
        # Actually cp returns a solution if status is "optimal" or "optimal_inaccurate"
        # Some solvers may return "optimal" only
        pass
    # Extract h
    if h.value is None:
        return None, None
    h_arr = np.array(h.value).astype(float).flatten()
    # Clip to 0/1 due to solver tolerance
    h_arr = np.rint(h_arr)  # round
    # Ensure sum = n/2
    # Might adjust small rounding errors
    cur_sum = np.sum(h_arr)
    target = n/2
    if cur_sum != target:
        # if rounding pops sum not correct, adjust by flipping some entries
        # Slightly adjust: if sum > target, set some 1's to 0 (choose smallest index)
        diff = int(cur_sum - target)
        if diff > 0:
            idx1 = np.where(h_arr == 1)[0][:diff]
            h_arr[idx1] = 0
        elif diff < 0:
            idx0 = np.where(h_arr == 0)[0][: -diff]
            h_arr[idx0] = 1
    # Compute C5
    dx = 2.0 / n
    max_corr = np.max(np.correlate(h_arr, 1 - h_arr, mode='full')) * dx
    return h_arr, max_corr

import cvxpy as cp

def solve_mip(n, time_limit):
    h = cp.Variable(boolean=True, shape=n)
    t = cp.Variable(integer=True)
    # w matrix
    w = cp.Variable(boolean=True, shape=(n, 2*n-1))
    constraints = [cp.sum(h) == n/2]
    # Or use target_sum = n//2; but n even ensures integer.

    for k in range(-n+1, n):
        if k == 0:
            continue
        idx = k + n - 1
        # sum over valid i
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                constraints.append(w[i, idx] <= h[i])
                constraints.append(w[i, idx] <= 1 - h[j])
                constraints.append(w[i, idx] >= h[i] - h[j])
        # sum bound
        constraints.append(cp.sum(w[:, idx]) <= t)

    constraints.append(t >= 0)

    # Define objective minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    # solve
    try:
        prob.solve(solver=cp.CBC, cbc={'seconds': time_limit}, warm_start=False, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, glpk={'tm_lim': int(time_limit*1000)}, verbose=False)
        except Exception:
            return None, None

    if h.value is None:
        return None, None
    # convert to binary
    h_arr = np.rint(np.array(h.value).flatten()).astype(int)
    # Ensure sum
    target = n//2
    cur = int(h_arr.sum())
    if cur != target:
        # adjust
        diff = cur - target
        if diff > 0:
            # change diff ones to zero
            ones_idx = np.where(h_arr == 1)[0][:diff]
            h_arr[ones_idx] = 0
        elif diff < 0:
            zeros_idx = np.where(h_arr == 0)[0][: -diff]
            h_arr[zeros_idx] = 1
    # compute C5
    dx = 2.0 / n
    c5 = np.max(np.correlate(h_arr, 1 - h_arr, mode='full')) * dx
    return h_arr.astype(float), c5

def _compute_c5(arr):
    """Return C5 for a binary (or float) array."""
    n = len(arr)
    dx = 2.0 / n

def _solve_mip(n, time_limit):
    """Solve the exact MILP: minimise max overlap t for binary h of length n."""
    # Variables
    h = cp.Variable(boolean=True, shape=n)
    t = cp.Variable(integer=True)
    w = cp.Variable(boolean=True, shape=(n, 2 * n - 1))

    constraints = [cp.sum(h) == n // 2]

    # Build product linearisation constraints and per‑shift bounds
    for k in range(-n + 1, n):
        if k == 0:
            continue
        idx = k + n - 1
        # valid i where i+k in [0,n)
        i_start = max(0, -k)
        i_end = min(n, n - k)  # exclusive
        for i in range(i_start, i_end):
            j = i + k
            constraints.append(w[i, idx] <= h[i])
            constraints.append(w[i, idx] <= 1 - h[j])
            constraints.append(w[i, idx] >= h[i] - h[j])
        constraints.append(cp.sum(w[:, idx]) <= t)

    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # Try a few solvers with a time cap
    for solver, opts in [
        (cp.CBC, {"seconds": time_limit}),
        (cp.GLPK_MI, {"tm_lim": int(time_limit * 1000)}),
        (cp.ECOS_BB, {}),
    ]:
        try:
            prob.solve(solver=solver, **opts, verbose=False)
            if h.value is not None:
                break
        except Exception:
            continue

    if h.value is None:
        return None, None

    # Extract a binary vector (rounding tolerances)
    h_arr = np.rint(np.array(h.value).flatten()).astype(int)

    # Repair possible rounding drift
    target = n // 2
    cur = int(h_arr.sum())
    if cur != target:
        diff = cur - target
        if diff > 0:
            idx = np.where(h_arr == 1)[0][:diff]
            h_arr[idx] = 0
        else:
            idx = np.where(h_arr == 0)[0][: -diff]
            h_arr[idx] = 1

    c5 = _compute_c5(h_arr.astype(float))
    return h_arr.astype(float), float(c5)


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 80

    best_h = None
    best_c5 = np.inf

    # ---------- Phase 1: exact MILP (if time permits) ----------
    elapsed = time.time() - start
    time_left = budget_s - elapsed - 5.0   # keep a small safety margin
    if time_left > 30.0:
        # allocate at most 300 s to the MILP
        mip_time = min(300.0, time_left)
        h_sol, c5_sol = _solve_mip(n, mip_time)
        if h_sol is not None:
            best_h = h_sol
            best_c5 = c5_sol

    # ---------- Phase 2: Simulated‑annealing refinement ----------
    elapsed = time.time() - start
    time_left = budget_s - elapsed - 0.5
    if time_left > 0.0:
        # starting point
        if best_h is not None:
            cur_h = np.rint(best_h).astype(int)
        else:
            cur_h = np.zeros(n, dtype=int)
            cur_h[rng.choice(n, n // 2, replace=False)] = 1

        # ensure exact half‑ones
        target_sum = n // 2
        cur_sum = cur_h.sum()
        if cur_sum != target_sum:
            diff = cur_sum - target_sum
            if diff > 0:
                idx = np.where(cur_h == 1)[0][:diff]
                cur_h[idx] = 0
            else:
                idx = np.where(cur_h == 0)[0][: -diff]
                cur_h[idx] = 1

        cur_val = _compute_c5(cur_h.astype(float))
        best_local_h = cur_h.copy()
        best_local_val = cur_val

        # annealing loop
        while time.time() - start < budget_s - 0.2:
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

            cand = cur_h.copy()
            cand[i] = 0
            cand[j] = 1
            cand_val = _compute_c5(cand.astype(float))

            if cand_val < cur_val:
                accept = True
            else:
                # linear cooling schedule
                frac = (time.time() - start) / budget_s
                T = max(1e-4, 1.0 - frac)   # temperature in [~0,1]
                if rng.random() < np.exp(-(cand_val - cur_val) / T):
                    accept = True
                else:
                    accept = False

            if accept:
                cur_h = cand
                cur_val = cand_val
                if cur_val < best_local_val:
                    best_local_h = cur_h.copy()
                    best_local_val = cur_val

        if best_local_val < best_c5:
            best_h = best_local_h.astype(float)
            best_c5 = best_local_val

    # ---------- Final sanity checks ----------
    if best_h is None:
        # fallback: simple random balanced vector
        best_h = np.zeros(n, dtype=float)
        best_h[rng.choice(n, n // 2, replace=False)] = 1.0
        best_c5 = _compute_c5(best_h)

    # enforce box constraints and exact integral
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    target = n / 2.0
    if abs(cur_sum - target) > 1e-9:
        best_h += (target - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, float(best_c5), n
