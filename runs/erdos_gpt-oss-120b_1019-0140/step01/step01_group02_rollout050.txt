import time
import math
import numpy as np

def _compute_c5_corr(h: np.ndarray):
    """Return C5 value and full correlation array."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = float(corr.max())
    return max_corr * dx, corr

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust one entry to make sum exactly target (within tolerance)."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # distribute uniformly as fallback
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

    start = time.time()
