def compute_weighted_gradient(h, w):
    n = h.size
    grad = np.zeros_like(h)
    # w is length 2n-1, zero-centered at n-1
    for idx, wt in enumerate(w):
        if wt == 0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            # left part: i from 0 to n-1-shift
            i_end = n - shift
            # left indices (starting part)
            grad[:i_end] += wt * (1 - h[shift:])
            grad[shift:] -= wt * h[:i_end]
        else:
            s = -shift
            i_end = n - s
            grad[s:] += wt * (1 - h[:i_end])
            grad[:i_end] -= wt * h[s:]
    return grad

Implement smoothing gradient descent function:

def smoothing_descent(h0, n_iter, step0, alpha0, alpha_inc, beta=0.9) -> (h_opt, best_score)

Pseudo:

h = h0.copy()
best_h = h.copy()
best_score = compute_c5(h)
v = np.zeros_like(h) # momentum

for it in range(1, n_iter+1):
    # compute correlation raw (no dx)
    corr = np.correlate(h, 1 - h, mode='full')
    # compute alpha (increase)
    alpha = alpha0 + (alpha_inc * (it-1) / n_iter)  # linearly increase to final alpha
    # compute weights w
    # we can scale by max to avoid overflow
    maxc = np.max(alpha * corr)
    w = np.exp(alpha * corr - maxc)
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    # compute weighted gradient
    grad = compute_weighted_gradient(h, w)
    # optionally, add momentum:
    step = step0 / np.sqrt(it) # or step0 * (0.9)**it
    v = beta * v - step * grad
    h_new = h + v
    # project onto constraints
    h_new = project_onto_box_sum(h_new, h.size/2.0)
    # evaluate score
    score = compute_c5(h_new)
    if score < best_score:
        best_score = score
        best_h = h_new.copy()
    h = h_new

Return best_h, best_score.

import time, numpy as np

def upsample(arr, new_n):
    ...

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    ...

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        end = n - shift
        grad[:end] += (1.0 - h[shift:])
        grad[shift:] -= h[:end]
    else:
        s = -shift
        end = n - s
        grad[s:] += (1.0 - h[:end])
        grad[:end] -= h[s:]
    return grad

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_balanced(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def pgd_optimize(h_start, n_iter, step0, momentum, top_k, rng):
    h = h_start.copy()
    n = h.size
    target_sum = n / 2.0
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # pick shift
        if top_k >= corr.size:
            idx = np.argmax(corr)
        else:
            # get top_k indices
            # argpartition to get k largest
            part = np.argpartition(corr, -top_k)[-top_k:]
            idx = rng.choice(part)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        step = step0 / np.sqrt(it)
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    # extreme cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand brackets if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) + 10 * lo

    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the correlation term for a given integer shift."""
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        end = n - shift
        grad[:end] += 1.0 - h[shift:]
        grad[shift:] -= h[:end]
    else:
        s = -shift
        end = n - s
        grad[s:] += 1.0 - h[:end]
        grad[:end] -= h[s:]
    return grad

def thue_morse(n):
    """Binary Thue–Morse sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_balanced(n, rng):
    """Random continuous vector projected onto box and sum constraints."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def pgd_optimize(h_start, n_iter, step0, momentum, top_k, rng):
    """Projected sub‑gradient descent with momentum."""
    h = h_start.copy()
    n = h.size
    target_sum = n / 2.0
    v = np.zeros_like(h)
    best_h = h.copy()
    # initial score
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = np.max(corr) * (2.0 / n)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # choose a shift among the top_k correlations
        if top_k >= corr.size:
            idx = np.argmax(corr)
        else:
            cand = np.argpartition(corr, -top_k)[-top_k:]
            idx = rng.choice(cand)
        shift = idx - (n - 1)

        grad = compute_gradient(h, shift)
        step = step0 / np.sqrt(it)
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, target_sum)

        # evaluate
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        score = np.max(corr_new) * (2.0 / n)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
