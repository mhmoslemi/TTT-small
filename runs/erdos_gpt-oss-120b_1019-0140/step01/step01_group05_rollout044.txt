def c5_score(h):
    N = len(h)
    dx = 2.0/N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def simulated_annealing(h_start, time_limit, rng):
    h = h_start.copy()
    cur = c5_score(h)
    best = h.copy()
    best_score = cur
    N = len(h)
    start = time.time()
    # schedule: temperature T = T0 * (1 - elapsed/limit)^alpha
    T0 = 0.01 # or set adaptively maybe 0.1
    # max_eps initial large (0.5) and decays.
    max_eps0 = 0.5
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = 1 - elapsed / time_limit
        T = T0 * (frac ** 2)  # quadratic cooling (slower initially)
        max_eps = max_eps0 * frac # shrink max_eps linearly

        # propose a move:
        i, j = rng.integers(0, N, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        # compute possible epsilon bounds:
        up = min(1 - hi, hj)  # max epsilon positive to increase hi, decrease hj
        down = min(hi, 1 - hj)  # max negative epsilon
        # if both zero, skip.
        if up + down == 0:
            continue
        eps = rng.uniform(-down, up)
        eps = np.clip(eps, -max_eps, max_eps)
        if eps == 0:
            continue

        h_new = h.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # no violation: the clipping ensures, but we still check maybe rounding error
        new_score = c5_score(h_new)
        delta = new_score - cur
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            h = h_new
            cur = new_score
            if new_score < best_score:
                best = h_new.copy()
                best_score = new_score
    return best, best_score

def repair(v):
    np.clip(v, 0, 1, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0:
        v[:] = target / v.size
        np.clip(v, 0, 1, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0, 1, out=v)

    # second scaling:

        v *= factor2
        np.clip(v, 0, 1, out=v)

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to n/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_score(h: np.ndarray) -> float:
    """Maximum overlap integral for discretised step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """SA preserving sum and box constraints. Returns best vector and its score."""
    h_cur = h_start.copy()
    cur_score = _c5_score(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    n = h_cur.size
    start = time.time()
    T0 = 0.05          # initial temperature
    max_eps0 = 0.5     # maximal absolute change at start
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * (frac ** 2)          # quadratic cooling
        max_eps = max_eps0 * frac      # linear decay of move size
        # propose a two‑index exchange preserving the sum
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)        # max positive epsilon
        down = min(hi, 1.0 - hj)      # max negative epsilon
        if up + down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        eps = np.clip(eps, -max_eps, max_eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_score(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C5 constant."""
    rng = np.random.default_rng(seed)
    # initialise from provided construction if present
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 100
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)], dtype=np.float64)
    h0 = _repair_vector(h0)

    start_total = time.time()
    best_h = h0
    best_score = _c5_score(best_h)

    # Phase 1 – optimisation at initial resolution
    remaining = budget_s - (time.time() - start_total)
    phase1_time = max(5.0, remaining * 0.35)
    best_h, best_score = _simulated_annealing(best_h, phase1_time, rng)

    # Subsequent phases – increase resolution iteratively
    n_current = best_h.size
    while True:
        elapsed = time.time() - start_total
        if elapsed >= budget_s * 0.97:
            break
        n_next = min(n_current * 2, 1000)
        if n_next == n_current:
            break
        # upsample by linear interpolation
        x_old = np.linspace(0.0, 2.0, n_current, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_next, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _repair_vector(h_up)

        # allocate time for this level
        remaining = budget_s - (time.time() - start_total)
        if remaining < 10.0:
            break
        level_time = remaining * 0.3
        # run SA at the new resolution
        h_opt, sc_opt = _simulated_annealing(h_up, level_time, rng)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
        n_current = n_next

    # Final repair and exact evaluation
    best_h = _repair_vector(best_h)
    best_score = _c5_score(best_h)

    return best_h, float(best_score), best_h.size
