def compute_c5_fft(h, dx):
    n = len(h)
    # compute convolution of h and reversed (1-h)
    b = 1.0 - h
    L = 1 << ((2 * n - 1).bit_length())
    fft_h = np.fft.rfft(h, n=L)
    fft_b_rev = np.fft.rfft(b[::-1], n=L)
    conv = np.fft.irfft(fft_h * fft_b_rev, n=L)[:2*n-1]
    # conv is cross-correlation values for lags from -(n-1) to (n-1); non-negative (maybe negative due to numeric)
    return np.max(conv) * dx

import numpy as np
import time
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust sum(v) to target by water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_fft(h):
    """Fast C5 evaluation using FFT (full linear cross-correlation)."""
    n = h.shape[0]
    dx = 2.0 / n
    b = 1.0 - h
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    # convolution of h with reversed b
    conv = np.fft.irfft(
        np.fft.rfft(h, n=L) * np.fft.rfft(b[::-1], n=L), n=L
    )[: 2 * n - 1]
    return np.max(conv) * dx

def _tournament_select(pop, fit, k=3):
    idxs = np.random.choice(len(pop), k, replace=False)
    best = idxs[0]
    best_score = fit[best]
    for i in idxs[1:]:
        if fit[i] < best_score:
            best = i
            best_score = fit[i]
    return pop[best]

def _refine_continuous(start, target_sum, max_iter=200, restarts=5, sigma=0.05):
    """Run several SLSQP refinements starting from `start` (float vector)."""
    n = start.size
    best_x = start.copy()
    best_score = _c5_fft(best_x)
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    for _ in range(restarts):
        x0 = start + sigma * np.random.randn(n)
        x0 = _project_to_constraints(x0, target_sum)
        try:
            res = minimize(_c5_fft, x0, method="SLSQP", bounds=bounds,
                           constraints=[cons], options=opts)
        except Exception:
            continue
        x = _project_to_constraints(res.x, target_sum)
        score = _c5_fft(x)
        if score < best_score:
            best_score = float(score)
            best_x = x.copy()
    return best_x, best_score

                        child[i] = 0
                        child[j] = 1
                # ensure exact weight
                cur = child.sum()
                if cur != target_sum:
                    diff = int(cur - target_sum)
                    if diff > 0:

                    else:
