import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_binary(n, rng):
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1
    return h

def hill_climb_swap(h, current_val, end_time, rng):
    best_h = h.copy()
    best_val = current_val
    while time.time() < end_time:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        best_h[i] = 0
        best_h[j] = 1
        new_val = compute_c5(best_h.astype(float))
        if new_val + 1e-12 < best_val:
            best_val = new_val
        else:
            # revert
            best_h[i] = 1
            best_h[j] = 0
    return best_h, best_val

def tournament_select(pop, scores, k, rng):
    # return index of selected individual
    candidates = rng.choice(len(pop), k, replace=False)
    best = candidates[0]
    best_score = scores[best]
    for idx in candidates[1:]:
        s = scores[idx]
        if s < best_score:   # lower C5 is better
            best = idx
            best_score = s
    return best

def crossover(p1, p2, n, rng):
    # union based selection
    union = np.where((p1 == 1) | (p2 == 1))[0]
    half = n // 2
    if len(union) >= half:
        chosen = rng.choice(union, half, replace=False)
    else:
        rest = np.where((p1 == 0) & (p2 == 0))[0]
        needed = half - len(union)
        extra = rng.choice(rest, needed, replace=False)
        chosen = np.concatenate([union, extra])
    child = np.zeros(n, dtype=int)
    child[chosen] = 1
    return child

def mutate_swap(h, n, rng):
    # perform one random swap with prob 1 (just always swap once)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h
    i = rng.choice(ones)
    j = rng.choice(zeros)
    h[i], h[j] = 0, 1
    return h

    rng = np.random.default_rng(seed)

    # Determine n and initial best
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = len(init)
        # convert to binary by taking top half entries
        binary = np.zeros(n, dtype=int)
        top = np.argpartition(-init, n//2-1)[:n//2]
        binary[top] = 1
        best_h = binary.astype(float)
        best_c5 = compute_c5(best_h)
    else:
        n = 80
        binary = random_binary(n, rng)
        best_h = binary.astype(float)
        best_c5 = compute_c5(best_h)

    # Parameters for GA
    pop_size = 40
    tournament_k = 5
    crossover_rate = 0.9
    mutation_rate = 0.3
    # Initialize population
    population = [random_binary(n, rng) for _ in range(pop_size)]
    # Ensure at least one is the current best
    population[0] = (best_h > 0.5).astype(int)

    # Evaluate initial population
    scores = np.empty(pop_size)
    for i, ind in enumerate(population):
        scores[i] = compute_c5(ind.astype(float))

    # Update global best
    best_idx = int(np.argmin(scores))
    if scores[best_idx] < best_c5 - 1e-12:
        best_h = population[best_idx].astype(float)
        best_c5 = scores[best_idx]

    # GA main loop
    while time.time() - start < budget_s - 2.0:
        new_pop = []
        new_scores = []
        # Elitism: keep best individual
        new_pop.append(population[best_idx].copy())
        new_scores.append(best_c5)

        while len(new_pop) < pop_size:
            # selection
            p1_idx = tournament_select(population, scores, tournament_k, rng)
            p2_idx = tournament_select(population, scores, tournament_k, rng)
            p1 = population[p1_idx]
            p2 = population[p2_idx]

            else:

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1
    return h

def tournament_select(scores, k, rng):
    # returns index of selected individual (lower score better)
    candidates = rng.choice(len(scores), k, replace=False)
    best = candidates[0]
    best_score = scores[best]
    for idx in candidates[1:]:
        s = scores[idx]
        if s < best_score:
            best = idx
            best_score = s
    return best

def crossover(p1, p2, n, rng):
    half = n // 2
    union = np.where((p1 == 1) | (p2 == 1))[0]
    if len(union) >= half:
        chosen = rng.choice(union, half, replace=False)
    else:

        needed = half - len(union)
        extra = rng.choice(rest, needed, replace=False)
        chosen = np.concatenate([union, extra])

    child[chosen] = 1
    return child

    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1
    return h

def tournament_select(scores, k, rng):
    """Select index of best (lowest) score among k random contestants."""
    cand = rng.choice(len(scores), k, replace=False)
    best = cand[0]
    best_score = scores[best]
    for i in cand[1:]:
        s = scores[i]
        if s < best_score:
            best = i
            best_score = s
    return best

def crossover(p1, p2, n, rng):
    """Union‑based crossover preserving exact half‑ones constraint."""
    half = n // 2
    union = np.where((p1 == 1) | (p2 == 1))[0]
    if len(union) >= half:
        chosen = rng.choice(union, half, replace=False)
    else:
        rest = np.where((p1 == 0) & (p2 == 0))[0]
        needed = half - len(union)
        extra = rng.choice(rest, needed, replace=False)
        chosen = np.concatenate([union, extra])
    child = np.zeros(n, dtype=int)
    child[chosen] = 1
    return child

def mutate_swap(h, rng):
    """Swap a random 1 with a random 0."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h
    i = rng.choice(ones)
    j = rng.choice(zeros)
    h[i], h[j] = 0, 1
    return h

def hill_climb_swap(h_int, cur_val, until, rng):
    """Improvement by random swaps until time limit."""
    best = h_int.copy()
    best_val = cur_val
    while time.time() < until:
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best[i] = 0
        best[j] = 1
        val = compute_c5(best.astype(float))
        if val + 1e-12 < best_val:
            best_val = val
        else:
            best[i] = 1
            best[j] = 0
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initial solution -----
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = len(init)
        bin_vec = np.zeros(n, dtype=int)
        top = np.argpartition(-init, n // 2 - 1)[: n // 2]
        bin_vec[top] = 1
        best_h_int = bin_vec
    else:
        n = 80
        best_h_int = random_binary(n, rng)
    best_h = best_h_int.astype(float)
    best_c5 = compute_c5(best_h)

    # ----- GA parameters -----
    pop_size = 40
    tour_k = 5
    cx_rate = 0.9
    mut_rate = 0.3

    # ----- initialise population -----
    population = [random_binary(n, rng) for _ in range(pop_size)]
    population[0] = best_h_int.copy()
    scores = np.empty(pop_size)
    for i, ind in enumerate(population):
        scores[i] = compute_c5(ind.astype(float))

    best_idx = int(np.argmin(scores))
    if scores[best_idx] < best_c5 - 1e-12:
        best_idx = best_idx
        best_h_int = population[best_idx].copy()
        best_h = best_h_int.astype(float)
        best_c5 = scores[best_idx]

    # ----- evolutionary loop -----
    while time.time() - start < budget_s - 5.0:
        new_pop = [population[best_idx].copy()]  # elitism
        new_scores = [best_c5]
        while len(new_pop) < pop_size:
            p1 = population[tournament_select(scores, tour_k, rng)]
            p2 = population[tournament_select(scores, tour_k, rng)]
            child = crossover(p1, p2, n, rng) if rng.random() < cx_rate else p1.copy()
            if rng.random() < mut_rate:
                child = mutate_swap(child, rng)
            cscore = compute_c5(child.astype(float))
            new_pop.append(child)
            new_scores.append(cscore)
        population = new_pop
        scores = np.array(new_scores)
        best_idx = int(np.argmin(scores))
        if scores[best_idx] < best_c5 - 1e-12:
            best_c5 = scores[best_idx]
            best_h_int = population[best_idx].copy()
            best_h = best_h_int.astype(float)

    # ----- final hill‑climbing refinement -----
    remaining = budget_s - (time.time() - start)
    if remaining > 0.5:
        end_time = time.time() + remaining - 0.1
        best_h_int, best_c5 = hill_climb_swap(best_h_int, best_c5, end_time, rng)
        best_h = best_h_int.astype(float)

    # ----- enforce exact constraints -----
    half = n // 2
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - half) > 1e-9:
        best_h = np.rint(best_h)               # force binary
        diff = half - best_h.sum()
        if diff > 0:
            zeros = np.where(best_h == 0)[0]
            if len(zeros) >= diff:
                best_h[zeros[:int(diff)]] = 1.0
        elif diff < 0:
            ones = np.where(best_h == 1)[0]
            if len(ones) >= -diff:
                best_h[ones[:int(-diff)]] = 0.0

    return best_h, best_c5, n
