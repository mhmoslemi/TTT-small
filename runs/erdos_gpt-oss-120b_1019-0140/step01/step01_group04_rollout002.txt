def simulated_annealing(v0, time_limit, rng):
    v = v0.copy()
    best_v = v.copy()
    best_score = compute_c5(v)
    cur_score = best_score
    n = len(v)
    step = 0.2  # max delta
    # initial temperature relative to score variance
    T = 0.01  # maybe 0.01
    # cooling schedule: exponential
    iteration = 0
    start = time.time()
    while time.time() - start < time_limit:
        # adapt step based on iteration
        # sample i, j
        i, j = rng.integers(0, n, size=2, endpoint=False)
        while i == j:
            j = rng.integers(0, n)

        cand[j] -= delta

def _repair_vector(v, n):
    v = np.clip(v, 0.0, 1.0)
    target = n/2.0
    s = v.sum()
    if s == 0:
        # set uniformly target/n
        v[:] = target / n
    else:
        v = v * (target / s)
        v = np.clip(v, 0.0, 1.0)
    # after clipping might deviate

    if abs(diff) > 1e-9:

def _gradient_descend(v0, time_limit, rng):
    n = len(v0)
    v = _repair_vector(v0.copy(), n)
    best_v = v.copy()
    best_score = _compute_c5(v)
    start = time.time()
    alpha = 0.01  # step size
    while time.time() - start < time_limit:
        # compute correlation
        corr = np.correlate(v, 1 - v, mode='full')
        max_idx = np.argmax(corr)
        # lag k
        k = max_idx - (n - 1)
        # gradient
        grad = np.ones(n)  # derivative of sum(v)
        if k >= 0:
            # subtract v[i+k] from i < n - k
            grad[: n - k] -= v[k:]
            # subtract v[i - k] from i >= k
            grad[k:] -= v[: n - k]
        else:
            kk = -k
            grad[kk:] -= v[: n - kk]
            grad[: n - kk] -= v[kk:]
        # gradient descent step
        cand = v - alpha * grad
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_v = cand.copy()
        # accept if improved else maybe revert depending
        if cand_score < best_score + 1e-12:
            v = cand
        # reduce step size
        alpha *= 0.995
        if alpha < 1e-5:
            alpha = 1e-5
    return best_v, best_score
```

import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project vector onto [0,1]^n with sum(v) = n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        if s > 0:
            v *= target / s
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Compute C5 for a discrete step function."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation from current grid to a finer grid."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _sa_optimize(v0, time_limit, rng):
    """Simulated annealing with simple exchange moves."""
    n = len(v0)
    v = _repair_vector(v0.copy(), n)
    best_v = v.copy()
    best_score = _compute_c5(v)
    cur_score = best_score
    start = time.time()
    step = 0.2
    T = 0.005
    decay = 0.9995
    while time.time() - start < time_limit:
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = (rng.random() - 0.5) * 2.0 * step
        cand = v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        dscore = cand_score - cur_score
        if dscore < 0 or rng.random() < np.exp(-dscore / max(T, 1e-12)):
            v = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_v = cand.copy()
        T *= decay
        step *= decay
        if T < 1e-8:
            T = 0.001
        if step < 1e-4:
            step = 0.001
    return best_v, best_score

def _gradient_descent(v0, time_limit, rng):
    """Projected gradient descent guided by the worst overlap lag."""
    n = len(v0)
    v = _repair_vector(v0.copy(), n)
    best_v = v.copy()
    best_score = _compute_c5(v)
    start = time.time()
    alpha = 0.01
    while time.time() - start < time_limit:
        corr = np.correlate(v, 1.0 - v, mode="full")
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)  # lag
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= v[k:]
            grad[k:] -= v[: n - k]
        else:
            kk = -k
            grad[kk:] -= v[: n - kk]
            grad[: n - kk] -= v[kk:]
        cand = v - alpha * grad
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_v = cand.copy()
        if cand_score < best_score + 1e-12:
            v = cand
        alpha *= 0.995
        if alpha < 1e-5:
            alpha = 1e-5
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function for the Erdős C5 problem."""
    rng = np.random.default_rng(seed)
    start = time.time()
    # Acquire an initial construction if provided.
    parent = globals().get('initial_h_values')
    if parent is not None:
        parent = np.array(parent, dtype=np.float64)
        n0 = len(parent)
        parent = _repair_vector(parent, n0)
    else:
        n0 = 200
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0 // 2] = 1.0
        parent = _repair_vector(parent, n0)
    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0
    # Build a list of resolutions, doubling while ≤800.
    resolutions = [best_n]
    while resolutions[-1] * 2 <= 800:
        resolutions.append(resolutions[-1] * 2)
    # Main optimization loop over resolutions.
    for idx, n in enumerate(resolutions):
        elapsed = time.time() - start
        remaining = budget_s - elapsed - 0.5  # keep a safety margin
        if remaining <= 0:
            break
        remaining_res = len(resolutions) - idx
        time_this = remaining / remaining_res
        # Prepare initial guess at this resolution.
        if best_n != n:
            init_guess = _upsample_vector(best_v, n)
        else:
            init_guess = best_v.copy()
        # Allocate time between SA and gradient descent.
        time_sa = max(1.0, time_this * 0.6)
        time_gd = max(0.5, time_this * 0.2)
        # Simulated annealing refinement.
        v_sa, sc_sa = _sa_optimize(init_guess, time_sa, rng)
        if sc_sa < best_score:
            best_score, best_v, best_n = sc_sa, v_sa, n
        # Gradient descent refinement.
        v_gd, sc_gd = _gradient_descent(v_sa, time_gd, rng)
        if sc_gd < best_score:
            best_score, best_v, best_n = sc_gd, v_gd, n
    # Use any leftover time for a final SA pass at the best resolution.
    leftover = budget_s - (time.time() - start) - 0.1
    if leftover > 2.0:
        v_final, sc_final = _sa_optimize(best_v, leftover, rng)
        if sc_final < best_score:
            best_score, best_v = sc_final, v_final
    # Final repair and evaluation.
    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)
    return best_v.astype(np.float64), float(best_score), int(best_n)
