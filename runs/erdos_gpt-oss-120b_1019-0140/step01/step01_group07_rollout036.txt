def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    fft_len = 1 << ((2*n - 1).bit_length())  # next power of 2 for performance
    H = np.fft.fft(h, n=fft_len)

    corr = np.fft.ifft(H * np.conj(G)).real
    # Full correlation length = 2n-1; after ifft length = fft_len; we need appropriate segment.
    # The correlation via ifft returns circular correlation. To get linear correlation, we need zero padding to size >= 2n-1.
    # With zero padding to fft_len >= 2n, the circular correlation equals linear correlation at first 2n-1 elements.
    corr = corr[: 2*n - 1]
    return np.max(corr) * dx

def project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-10:
        return v
    if cur > target:
        # need to decrease sum by cur - target
        excess = cur - target
        # compute capacities to decrease: v_i - 0
        caps = v.copy()
        # sort caps descending
        sorted_idx = np.argsort(caps)[::-1]
        # iterate
        for idx in sorted_idx:
            if excess <= 0:
                break
            dec = min(caps[idx], excess)
            v[idx] -= dec
            excess -= dec
    else:
        # need to increase sum
        need = target - cur
        caps = 1.0 - v
        sorted_idx = np.argsort(caps)[::-1]
        for idx in sorted_idx:
            if need <= 0:
                break
            inc = min(caps[idx], need)
            v[idx] += inc
            need -= inc
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # choose size for FFT: next power of two >= 2*n
    size = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=size)   # using real FFT for efficiency
    G = np.fft.rfft(1.0 - h, n=size)
    # rfft returns n/2+1 length representing positive freqs; product: H * conj(G)
    # using real valued sequences, correlation can be obtained via irfft(H * conj(G), n=size)
    corr = np.fft.irfft(H * np.conj(G), n=size)
    # take first 2*n-1 values
    corr = corr[:2*n-1]
    # maybe due to numeric errors produce slightly negative values; fine.
    return np.max(corr) * dx

import numpy as np
import time
from scipy.optimize import minimize

# ---------- Helper functions ----------

def _fast_project(v, target):
    """Quick projection: clip to [0,1] then scale to meet the sum."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if cur > 0:
        v = v * (target / cur)
    v = np.clip(v, 0.0, 1.0)
    # second pass to correct tiny drift
    cur = v.sum()
    if abs(cur - target) > 1e-8:
        v = v * (target / cur)
    return np.clip(v, 0.0, 1.0)

def _precise_project(v, target):
    """Exact water‑filling projection onto [0,1] with prescribed sum."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:  # need to add mass
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:  # need to remove mass
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5_fft(h):
    """Maximum overlap integral using FFT‑based full correlation."""
    n = h.size
    dx = 2.0 / n
    size = 2 * n
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * np.conj(G), n=size)
    corr = corr[: 2 * n - 1]
    return np.max(corr) * dx

def _interp_to(v, new_n):
    """Interpolate vector v (length N) to length new_n preserving domain [0,2]."""
    N = v.size
    x_old = np.linspace(0.0, 2.0, N, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, v)

# ---------- Differential Evolution optimizer ----------

def _de_optimize(start_vec, target_sum, time_limit):
    """Simple DE that respects box & sum constraints via fast projection."""
    n = start_vec.size
    # population size adapts to dimension
    pop_size = max(10, min(50, n // 10))
    rng = np.random.default_rng()
    # initialise population
    pop = np.empty((pop_size, n), dtype=np.float64)
    for i in range(pop_size):
        pop[i] = _fast_project(rng.random(n), target_sum)
    # force one member to be a perturbed start vector
    pop[0] = _fast_project(start_vec + rng.normal(scale=0.02, size=n), target_sum)

    fitness = np.empty(pop_size, dtype=np.float64)
    for i in range(pop_size):
        fitness[i] = _compute_c5_fft(pop[i])

    best_idx = int(np.argmin(fitness))
    best_vec = pop[best_idx].copy()
    best_score = float(fitness[best_idx])

    F = 0.8
    CR = 0.9
    start = time.time()
    while time.time() - start < time_limit:
        for i in range(pop_size):
            idxs = np.arange(pop_size)
            idxs = idxs[idxs != i]
            a, b, c = rng.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = _fast_project(mutant, target_sum)  # clip+scale
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = _fast_project(trial, target_sum)
            trial_score = _compute_c5_fft(trial)
            if trial_score < fitness[i]:
                pop[i] = trial
                fitness[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    if best_score < 0.3808:
                        return best_vec, best_score
        # occasional random immigration to keep diversity
        if rng.random() < 0.05:
            worst = int(np.argmax(fitness))
            pop[worst] = _fast_project(rng.random(n), target_sum)
            fitness[worst] = _compute_c5_fft(pop[worst])
    return best_vec, best_score

# ---------- Local refinement (optional) ----------

def _local_refine(vec, target_sum, remaining_time):
    """Brief SLSQP refinement if time permits."""
    if remaining_time < 4.0:
        return vec, _compute_c5_fft(vec)
    # SLSQP on the exact objective (still using FFT for speed)
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * vec.size
    try:
        res = minimize(_compute_c5_fft, vec, method="SLSQP",
                       bounds=bounds, constraints=[cons],
                       options={"maxiter": 200, "ftol": 1e-9, "disp": False})
    except Exception:
        return vec, _compute_c5_fft(vec)
    refined = _precise_project(res.x, target_sum)
    return refined, _compute_c5_fft(refined)

# ---------- Main entry point ----------

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    start_time = time.time()

    # --- initial vector -------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        init_vec = np.array(init, dtype=np.float64).ravel()
        n0 = init_vec.size
    else:
        n0 = 200
        init_vec = np.random.rand(n0)
    init_vec = _precise_project(init_vec, n0 / 2.0)

    best_h = init_vec.copy()
    best_score = _compute_c5_fft(best_h)

    # --- resolution schedule --------------------------------------------
    n_vals = [n0]
    # Add higher resolutions if we have enough budget left (empirical thresholds)
    if budget_s >= 300:
        n_vals.append(n0 * 2)
    if budget_s >= 600:
        n_vals.append(n0 * 4)

    for n in n_vals:
        # time check
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # build start vector for this resolution
        if n == n0:
            h_start = best_h.copy()
        else:
            h_start = _interp_to(best_h, n)

        target_sum = n / 2.0
        h_start = _precise_project(h_start, target_sum)

        # allocate most of the remaining budget to this resolution
        alloc = remaining * 0.7
        h_opt, score_opt = _de_optimize(h_start, target_sum, alloc)

        # local refinement if still time
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        h_opt, score_opt = _local_refine(h_opt, target_sum, remaining)

        if score_opt < best_score:
            best_score = float(score_opt)
            best_h = h_opt.copy()

        if best_score < 0.3808:
            break  # target achieved

    # final exact projection
    final_n = best_h.size
    best_h = _precise_project(best_h, final_n / 2.0)

    # final bound using provided evaluator when available
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(_compute_c5_fft(best_h))
    else:
        final_score = float(_compute_c5_fft(best_h))

    return best_h.astype(np.float64), final_score, int(final_n)
