def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5
    
    # Determine n_points and target_sum
    if 'initial_h_values' in globals():
        try:
            init_h = np.array(initial_h_values, dtype=float)
            n_points = init_h.size
            target_sum = n_points / 2.0
            h0 = adjust_sum_to_target(init_h.copy(), target_sum, rng)
        except Exception:
            init_h = None
    if init_h is None:
        n_points = 200
        target_sum = n_points / 2.0
        h0 = rng.rand(n_points)
        h0 = h0 / h0.sum() * target_sum
        h0 = adjust_sum_to_target(h0, target_sum, rng)
    
    # DE population
    pop_size = 20 if n_points <= 200 else 40
    # if n_points small (e.g., 70), still pop_size=20
    population = np.empty((pop_size, n_points))
    # first individual is h0
    population[0] = h0
    for i in range(1, pop_size):
        # generate random vector
        v = rng.rand(n_points)
        v = v / v.sum() * target_sum
        v = adjust_sum_to_target(v, target_sum, rng)
        population[i] = v
    
    # compute fitness
    fitness = np.empty(pop_size)
    for i in range(pop_size):
        fitness[i] = compute_c5(population[i])
    best_idx = fitness.argmin()
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    F = 0.5
    Cr = 0.9
    
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        for i in range(pop_size):
            # choose distinct a,b,c indices not i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            # clip and project
            np.clip(mutant, 0.0, 1.0, out=mutant)
            mutant = adjust_sum_to_target(mutant, target_sum, rng)
            # crossover
            trial = population[i].copy()
            jrand = rng.randint(n_points)
            for j in range(n_points):
                if rng.rand() < Cr or j == jrand:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0.0, 1.0)
            trial = adjust_sum_to_target(trial, target_sum, rng)
            trial_c5 = compute_c5(trial)
            # selection
            if trial_c5 < fitness[i]:
                population[i] = trial
                fitness[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
        # Optional: adapt F based on time (decrease)
    
    # Ensure final best_h exact sum
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Scale/adjust vector h in-place so that its sum equals target while
    respecting 0 <= h_i <= 1.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    # Distribute the difference by fixing individual entries one at a time
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # Small residual: distribute uniformly and re‑clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Differential‑Evolution style optimizer for the Erdős C₅ problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # ---------- initialise ----------
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n_points = init_arr.size
        target_sum = n_points / 2.0
        h0 = adjust_sum_to_target(init_arr.copy(), target_sum, rng)
    except Exception:
        # No parent solution – fall back to a modest size
        n_points = 200
        target_sum = n_points / 2.0
        h0 = rng.rand(n_points)
        h0 = h0 / h0.sum() * target_sum
        h0 = adjust_sum_to_target(h0, target_sum, rng)

    # population size – a small constant works well for n≤200
    pop_size = 20 if n_points <= 200 else 40
    # ---------- create initial population ----------
    population = np.empty((pop_size, n_points), dtype=float)
    population[0] = h0
    for i in range(1, pop_size):
        cand = rng.rand(n_points)
        cand = cand / cand.sum() * target_sum
        cand = adjust_sum_to_target(cand, target_sum, rng)
        population[i] = cand

    # initial fitness
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(population[i])

    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # DE parameters
    F = 0.5          # differential weight
    Cr = 0.9         # crossover probability

    generations_without_improve = 0
    prev_best = best_c5

    # ---------- main DE loop ----------
    while True:
        now = time.time()
        if now - start >= max_time:
            break

        # one generation
        for i in range(pop_size):
            # select three distinct indices different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)

            # mutation
            mutant = population[a] + F * (population[b] - population[c])
            np.clip(mutant, 0.0, 1.0, out=mutant)
            mutant = adjust_sum_to_target(mutant, target_sum, rng)

            # binomial crossover
            trial = population[i].copy()
            jrand = rng.randint(n_points)
            mask = rng.rand(n_points) < Cr
            mask[jrand] = True
            trial[mask] = mutant[mask]
            np.clip(trial, 0.0, 1.0, out=trial)
            trial = adjust_sum_to_target(trial, target_sum, rng)

            trial_c5 = compute_c5(trial)

            # selection
            if trial_c5 < fitness[i]:
                population[i] = trial
                fitness[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()

        # progress monitoring for stagnation
        if best_c5 < prev_best - 1e-8:
            prev_best = best_c5
            generations_without_improve = 0
        else:
            generations_without_improve += 1

        # occasional rejuvenation if stuck
        if generations_without_improve >= 200:
            # replace the worst half of the population with fresh random candidates
            worst_idxs = np.argsort(fitness)[-pop_size // 2 :]
            for idx in worst_idxs:
                cand = rng.rand(n_points)
                cand = cand / cand.sum() * target_sum
                cand = adjust_sum_to_target(cand, target_sum, rng)
                population[idx] = cand
                fitness[idx] = compute_c5(cand)
            generations_without_improve = 0

    # final cleanup – enforce exact mass one more time
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
