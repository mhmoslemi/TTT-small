def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5

    # Determine n_points
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    except Exception:
        n = 200
        init_h = rng.rand(n)
    target_sum = n / 2.0

    # Adjust init to constraints
    init_h = np.clip(init_h, 0.0, 1.0)
    init_h = adjust_sum_to_target(init_h, target_sum, rng)
    current_h = init_h.copy()
    current_c5 = compute_c5(current_h)

    best_h = current_h.copy()
    best_c5 = current_c5

    # SA parameters
    T_initial = 0.02
    T_final = 1e-6
    step_initial = 0.5
    step_final = 1e-4

    # main loop
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        T = T_initial * (1.0 - progress) + T_final * progress
        step = step_initial * (1.0 - progress) + step_final * progress

        move_type = rng.rand()
        if move_type < 0.3:
            # gradient move
            corr = np.correlate(current_h, 1.0 - current_h, mode='full')
            max_idx = int(np.argmax(corr))
            shift = max_idx - (n - 1)
            # compute gradient
            g = np.zeros_like(current_h)
            idx = np.arange(n)
            plus = idx + shift
            valid_plus = (plus >= 0) & (plus < n)
            g[valid_plus] += (1.0 - current_h[plus[valid_plus]])
            minus = idx - shift
            valid_minus = (minus >= 0) & (minus < n)
            g[valid_minus] -= current_h[minus[valid_minus]]
            # projected gradient (zero sum)
            g -= g.mean()
            candidate = current_h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        elif move_type < 0.9:
            # pairwise exchange
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = np.clip(delta, low, high)
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta
        else:
            # multi-index jitter: pick 3 indices, randomly redistribute mass among them preserving sum
            idxs = rng.choice(n, size=3, replace=False)
            # generate random perturbation vector of size 3 with zero sum
            rand_vec = rng.randn(3)
            # project to zero sum
            rand_vec -= rand_vec.mean()
            # scale magnitude to step
            max_abs = np.max(np.abs(rand_vec))
            if max_abs > 0:
                rand_vec = rand_vec / max_abs * step
            # apply and clip then adjust sum
            candidate = current_h.copy()
            candidate[idxs] += rand_vec
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)

        cand_c5 = compute_c5(candidate)

        if cand_c5 < current_c5:
            accept = True
        else:
            # compute acceptance probability
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()

    # gradient refinement
    # Continue until time out or no improvement after several attempts
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        # compute correlation and gradients for worst shifts
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = corr.max()
        # find all indices close to max (within 1e-12)
        max_idxs = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        # gradient sum
        g = np.zeros_like(best_h)
        idx = np.arange(n)
        for max_idx in max_idxs:
            shift = max_idx - (n - 1)
            plus = idx + shift
            valid_plus = (plus >= 0) & (plus < n)
            g[valid_plus] += (1.0 - best_h[plus[valid_plus]])
            minus = idx - shift
            valid_minus = (minus >= 0) & (minus < n)
            g[valid_minus] -= best_h[minus[valid_minus]]
        # project onto zero-sum hyperplane
        g -= g.mean()
        # descent direction
        d = -g
        # line search: try several step sizes decreasing geometrically
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step_refine = step_initial * factor * 0.1  # smaller base step
            candidate = best_h + step_refine * d
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
            cand_c5 = compute_c5(candidate)
            if cand_c5 < best_c5 - 1e-9:  # significant improvement
                best_h = candidate
                best_c5 = cand_c5
                improved = True
                break
        if not improved:
            # if no improvement, reduce step and perhaps break after several attempts
            # We'll just reduce step_initial gradually
            step_initial *= 0.5
            if step_initial < 1e-8:
                break

    # final adjust to exact sum
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Project h onto the feasible set:
        0 ≤ h_i ≤ 1  and  sum(h) = target.
    Simple iterative adjustment; fast for moderate dimensions.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Try to fix the deficit/surplus using a few random indices
    for _ in range(500):
        if abs(diff) < 1e-12:
            break
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
    if abs(diff) > 1e-12:
        # Distribute any tiny residue uniformly and clip again
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # One more uniform fix if needed (very small)
        total = h.sum()
        diff = target - total
        if abs(diff) > 1e-12:
            h += diff / h.size
            np.clip(h, 0.0, 1.0, out=h)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """

    start = time.time()
    total_budget = float(budget_s)
    # reserve a small margin for finalization
    end_time = start + total_budget - 0.1
    # allocate ~80% of the time for the simulated-annealing phase
    sa_end = start + total_budget * 0.80

    # ---- initialise h ----
    try:
        init_h = np.array(initial_h_values, dtype=float)  # may be undefined

    except Exception:
        n = 200
        target_sum = n / 2.0
        h = rng.rand(n)

    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()

    # ---- simulated annealing parameters ----
    T_initial = 0.02

            plus = idx + shift

        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
