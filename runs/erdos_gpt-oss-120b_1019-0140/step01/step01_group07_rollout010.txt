def _optimize_slsqp(h_start, target_sum, time_remaining):

    # compute max restarts based on time, maybe 20

    while max_restarts > 0 and (time.time() - start_time) < time_remaining:
        max_restarts -= 1
        pert = np.random.randn(h_start.size) * 0.02
        x0 = h_start + pert
        x0 = _project_to_constraints(x0, target_sum)
        cons = {"type": "eq", "fun": lambda h, tgt=target_sum: np.sum(h) - tgt}

            continue

            best_score = score
            best_h = h_opt.copy()
            if best_score <= 0.3808:
                break

import numpy as np
import time
import math
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral C5 for discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    N = 2 * n - 1
    M = int(2 ** np.ceil(np.log2(N)))  # power‑of‑two length for FFT
    H = np.fft.rfft(h, n=M)
    B = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(B), n=M)[:N].real
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clamp v to [0,1] and adjust to satisfy sum(v)=target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0.0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _anneal_binary(h0, max_iter, time_limit):
    """Simple simulated‑annealing on binary vectors (swap 0↔1)."""
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_score = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    T = 0.01
    decay = 0.9995
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # pick two opposite bits
        i = np.random.randint(n)
        j = np.random.randint(n)
        while cur_h[i] == cur_h[j]:
            j = np.random.randint(n)
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = _compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                if best_score <= 0.3808:
                    break
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]  # revert
        T = max(T * decay, 1e-12)
    return best_h, best_score

def _optimize_slsqp(h_start, target_sum, time_budget):
    """Repeated SLSQP refinements under the given time budget."""
    best_h = h_start.copy()
    best_score = _compute_c5(best_h)
    start = time.time()
    restarts = 0
    while time.time() - start < time_budget and restarts < 30:
        restarts += 1
        pert = np.random.randn(best_h.size) * 0.02
        x0 = best_h + pert
        x0 = _project_to_constraints(x0, target_sum)
        cons = {"type": "eq", "fun": lambda h, tgt=target_sum: np.sum(h) - tgt}
        bounds = [(0.0, 1.0)] * best_h.size
        try:
            res = minimize(_compute_c5, x0, method="SLSQP",
                           bounds=bounds, constraints=[cons],
                           options={"maxiter": 300, "ftol": 1e-9, "disp": False})
        except Exception:
            continue
        h_opt = _project_to_constraints(res.x, target_sum)
        score = _compute_c5(h_opt)
        if score < best_score:
            best_score = score
            best_h = h_opt.copy()
            if best_score <= 0.3808:
                break
    return best_h, best_score
