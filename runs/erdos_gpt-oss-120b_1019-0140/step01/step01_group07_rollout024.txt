def _compute_c5_from_h(h):
    # compute C5 via FFT
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    # compute FFT of h padded with zeros
    H = np.fft.rfft(np.concatenate([h, np.zeros(L - n, dtype=h.dtype)]))
    autocorr = np.fft.irfft(H * np.conj(H), n=L)[:2*n - 1]
    min_ac = np.min(autocorr)
    # C5 = 1 - (2 * min_ac) / n
    c5 = 1.0 - (2.0 * min_ac) / n
    return c5

def _subgradient_optimize(h_init, target_sum, time_limit):

    n = h.size
    # step schedule

    # initial step

import time
import numpy as np

def _project_to_constraints(v, target):
    """Clip to [0,1] and enforce sum(v)=target by water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_from_h(h):
    """Return C₅ given step values h using FFT (fast linear autocorrelation)."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, n=L)                  # zero‑padded FFT
    autocorr = np.fft.irfft(H * np.conj(H), n=L)[: 2 * n - 1]
    min_ac = np.min(autocorr)
    if min_ac < 0:
        min_ac = 0.0
    # C₅ = 1 - (2 * min_autocorr) / n   (since target = n/2, dx = 2/n)
    return 1.0 - (2.0 * min_ac) / n

def _subgradient_optimize(h0, target, timelimit):
    """Projected sub‑gradient ascent on min_autocorr (i.e. minimise C₅)."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_score = _c5_from_h(h)
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    step0 = 0.3
    it = 0
    while time.time() - start < timelimit:
        # autocorrelation and most violated lag
        H = np.fft.rfft(h, n=L)
        autocorr = np.fft.irfft(H * np.conj(H), n=L)[: 2 * n - 1]
        min_idx = np.argmin(autocorr)
        lag = min_idx - (n - 1)

        # gradient of the autocorrelation at this lag
        grad = np.zeros_like(h)
        if lag >= 0:
            s = lag
            grad[: n - s] += h[s:]
            grad[s:] += h[: n - s]
        else:
            s = -lag
            grad[s:] += h[: n - s]
            grad[: n - s] += h[s:]

        # diminishing step size
        alpha = step0 / np.sqrt(it + 1.0)
        h_new = h + alpha * grad
        h_new = _project_to_constraints(h_new, target)

        score = _c5_from_h(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score < 0.3808:
                break
        h = h_new
        it += 1
    return best_h, best_score

def _local_search_optimize(h0, target, timelimit):
    """Simple random pairwise updates with hill climbing."""
    start = time.time()
    n = h0.size
    cur = h0.copy()
    cur_score = _c5_from_h(cur)
    best_h = cur.copy()
    best_score = cur_score
    while time.time() - start < timelimit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move mass from j to i (keep sum constant)
        max_inc = 1.0 - cur[i]
        max_dec = cur[j]
        if max_inc <= 0 or max_dec <= 0:
            continue
        delta = np.random.rand() * min(max_inc, max_dec) * 0.5
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_score = _c5_from_h(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand.copy()
            cur = cand
            cur_score = cand_score
            if best_score < 0.3808:
                break
        elif cand_score < cur_score:
            cur = cand
            cur_score = cand_score
    return best_h, best_score
