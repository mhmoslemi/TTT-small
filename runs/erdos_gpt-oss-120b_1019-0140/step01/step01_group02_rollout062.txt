def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    import math
    start = time.time()
    rng = np.random.default_rng(seed)
    
    # helper functions
    def _compute_c5(h):
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * dx)
    
    def _repair_sum(h, target):
        cur = float(h.sum())
        diff = target - cur
        if abs(diff) < 1e-12:
            return h
        # find indices that can adjust
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            h += diff / h.size
            np.clip(h, 0.0, 1.0, out=h)
            return h
        i = idx[0]
        new = np.clip(h[i] + diff, 0.0, 1.0)
        h[i] = new
        if abs(new - h[i] - diff) > 1e-12:
            return _repair_sum(h, target)
        return h

def _repair_sum(h, target):
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # adjust a single entry
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # spread evenly (fallback)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new = np.clip(h[i] + diff, 0.0, 1.0)
    h[i] = new
    if abs(new - h[i] - diff) > 1e-12:
        return _repair_sum(h, target)
    return h
