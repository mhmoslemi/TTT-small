import time
import numpy as np

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust h in place so that its sum equals target while keeping 0 <= h <= 1.
    Small residual errors are distributed uniformly and the result clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # try to fix the whole deficit/surplus using a few random indices
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_corr_full(h: np.ndarray) -> np.ndarray:
    """
    Full linear correlation of h with (1-h) using FFT.
    Returns array of length 2*len(h)-1 containing the correlation values.
    """
    n = h.shape[0]
    v_rev = (1.0 - h)[::-1]
    n_fft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    V = np.fft.rfft(v_rev, n=n_fft)
    conv = np.fft.irfft(H * V, n=n_fft)
    return conv[: 2 * n - 1]

def compute_c5(h: np.ndarray) -> float:
    """Return the discrete approximation of C5 for vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = compute_corr_full(h)
    return float(dx * np.max(corr))

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Heuristic minimisation of the Erdős overlap constant C5.
    Returns (best_h, best_c5, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # ---- choose resolution -------------------------------------------------
    n_points = 256                     # power of two => fast FFT
    target_sum = n_points / 2.0

    # ---- initialise --------------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        if init_arr.size != n_points:
            # linear interpolation to required size
            init_arr = np.interp(
                np.linspace(0, 1, n_points),
                np.linspace(0, 1, init_arr.size),
                init_arr,
            )
        h0 = adjust_sum_to_target(init_arr.copy(), target_sum, rng)
    except Exception:
        # random binary start (exactly half ones)
        h0 = np.zeros(n_points, dtype=float)
        ones = rng.choice(n_points, size=int(target_sum), replace=False)
        h0[ones] = 1.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    current_h = h0.copy()
    current_c5 = best_c5

    # ---- annealing parameters ---------------------------------------------
    T0, T_final = 2e-2, 1e-6
    step0, step_final = 1.5e-1, 1e-4
    Kgrad = 5  # number of top shifts used for sub‑gradient

    while time.time() - start < max_time:
        elapsed = time.time() - start
        progress = elapsed / max_time

        # linear cooling
        T = T0 * (1.0 - progress) + T_final * progress
        step = step0 * (1.0 - progress) + step_final * progress

        # correlation of current state
        corr_full = compute_corr_full(current_h)
        # choose a few worst shifts (largest correlation values)
        K = min(Kgrad, corr_full.size)
        top_idx = np.argpartition(-corr_full, K)[:K]

        # sub‑gradient for the selected shifts
        n = n_points
        g = np.zeros(n, dtype=float)
        for idx in top_idx:
            k_shift = idx - (n - 1)  # integer lag
            if k_shift >= 0:
                g[: n - k_shift] += 1.0 - current_h[k_shift:]
                g[k_shift:] -= current_h[: n - k_shift]
            else:
                kp = -k_shift
                g[kp:] += 1.0 - current_h[: n - kp]
                g[: n - kp] -= current_h[kp:]

        # project gradient to the hyperplane sum(g)=0
        g -= g.mean()

        # scale step so that max change per entry is about `step`
        max_abs = np.max(np.abs(g)) + 1e-12
        delta = step / max_abs
        cand = current_h - delta * g
        np.clip(cand, 0.0, 1.0, out=cand)
        cand = adjust_sum_to_target(cand, target_sum, rng)
        cand_c5 = compute_c5(cand)

        # occasional discrete swap move
        if rng.rand() < 0.2:
            zeros = np.where(current_h <= 0.5)[0]
            ones = np.where(current_h >= 0.5)[0]
            if zeros.size and ones.size:
                i = rng.choice(zeros)
                j = rng.choice(ones)
                cand_swap = current_h.copy()
                cand_swap[i], cand_swap[j] = cand_swap[j], cand_swap[i]
                cand_swap_c5 = compute_c5(cand_swap)
                if cand_swap_c5 < cand_c5:
                    cand, cand_c5 = cand_swap, cand_swap_c5

        # Metropolis acceptance
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = cand
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

        # infrequent random restart to escape stagnation
        if rng.rand() < 0.001 and (time.time() - start) < max_time - 5.0:
            h_restart = np.zeros(n_points, dtype=float)
            idx = rng.choice(n_points, size=int(target_sum), replace=False)
            h_restart[idx] = 1.0
            current_h = h_restart
            current_c5 = compute_c5(current_h)

    # final sanity correction
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return best_h, float(final_c5), n_points
