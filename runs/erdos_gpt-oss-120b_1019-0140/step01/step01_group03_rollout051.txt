def subgradient_momentum_opt(h0, n_iter, step0=0.5, momentum=0.9, top_k=3, rng=None):

    h = h0.copy()
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # get top_k indices
        # find indices in descending order
        # idxs = np.argpartition(-corr, range(top_k))[:top_k] does not guarantee sorted; but we just need selection.
        # We'll use np.argpartition for speed.
        k = min(top_k, corr.size)
        # topk unsorted
        top_idxs = np.argpartition(-corr, k-1)[:k]
        # pick one at random
        idx = int(rng.choice(top_idxs))
        shift = idx - (h.size - 1)
        grad = compute_gradient(h, shift)
        # step size schedule
        eta = step0 / math.sqrt(it)
        v = momentum * v - eta * grad
        h_new = h + v
        # project
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time, math
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    lo_tot = lo * n
    hi_tot = hi * n
    if s <= lo_tot:
        return np.full_like(v, lo)
    if s >= hi_tot:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds until the target sum is bracketed
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10.0
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10.0

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the active max‑correlation term."""
    n = h.size
    grad = np.ones_like(h)
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:]    -= h[:n - shift]
    else:
        s = -shift
        grad[s:]       -= h[:n - s]
        grad[:n - s]   -= h[s:]
    return grad

def subgradient_momentum_opt(h0, n_iter=4000, step0=0.5,
                             momentum=0.9, top_k=3, rng=None):
    """Projected sub‑gradient descent with momentum and top‑k random shift."""
    if rng is None:
        rng = np.random.default_rng()
    h = h0.copy()
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k = min(top_k, corr.size)
        top_idxs = np.argpartition(-corr, k - 1)[:k]
        idx = int(rng.choice(top_idxs))
        shift = idx - (h.size - 1)
        grad = compute_gradient(h, shift)

        eta = step0 / math.sqrt(it)
        v = momentum * v - eta * grad
        h_new = project_onto_box_sum(h + v, s)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    """Return Thue‑Morse binary sequence of length n (0/1 values)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                      dtype=float, count=n)
    return seq

def random_balanced_binary(n, rng):

    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h
