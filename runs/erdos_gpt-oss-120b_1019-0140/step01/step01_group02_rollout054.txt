def _targeted_move(h, step, rng):
    n = h.shape[0]
    # compute correlation and find shift with max correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)  # integer shift (positive-> shift right? We'll treat as defined)
    # Determine overlap range
    if shift >= 0:
        i_start = 0
        i_end = n - shift
        i = np.arange(i_start, i_end)
        j = i + shift
    else:
        # negative shift
        i_start = -shift
        i_end = n
        i = np.arange(i_start, i_end)
        j = i + shift  # shift negative so j < i
    # contributions array:
    contrib = h[i] * (1.0 - h[j])
    # find index with max contribution
    idx = np.argmax(contrib)
    i_idx = i[idx]
    j_idx = j[idx]
    # possible delta: move from i_idx to j_idx (decrease h[i], increase h[j])
    max_delta = min(step, h[i_idx], 1.0 - h[j_idx])
    if max_delta <= 1e-12:
        # not possible; fallback random
        return None, None
    delta = rng.random() * max_delta
    # apply
    new_h = h.copy()
    new_h[i_idx] = h[i_idx] - delta
    new_h[j_idx] = h[j_idx] + delta
    return new_h, (i_idx, j_idx, delta)

def _targeted_move(h, step, rng):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    if shift == 0:
        # find index i with largest hi*(1 - hi)
        contrib = h * (1.0 - h)
        i = int(np.argmax(contrib))
        # decide direction: if h[i] > 0.5, we want to increase it; else decrease it.
        if h[i] > 0.5:
            # need a donor j with h[j] > 0 and <0.5 ideally
            # candidate donors: indices where h < 0.5
            donor_idx = np.where(h < 0.5 - 1e-12)[0]
            if donor_idx.size == 0:
                return None
            # choose j with minimal h (most low)
            j = int(donor_idx[np.argmin(h[donor_idx])])
            # delta limited by step, 1 - h[i], and h[j]
            max_delta = min(step, 1.0 - h[i], h[j])
            if max_delta <= 1e-12:
                return None
            delta = rng.random() * max_delta
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        else:
            # h[i] <=0.5, we want to decrease it
            # donor j to increase, choose j with h >0.5
            donor_idx = np.where(h > 0.5 + 1e-12)[0]
            if donor_idx.size == 0:
                return None
            j = int(donor_idx[np.argmax(h[donor_idx])])  # highest
            max_delta = min(step, h[i], 1.0 - h[j])
            if max_delta <= 1e-12:
                return None
            delta = rng.random() * max_delta
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            return new_h
    else:
        # compute overlapping indices
        if shift > 0:
            i_range = np.arange(0, n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        # contributions hi * (1 - hj)
        contrib = h[i_range] * (1.0 - h[j_range])
        max_k = int(np.argmax(contrib))
        i_idx = i_range[max_k]
        j_idx = j_range[max_k]
        # move delta from i to j to reduce product: hi down, hj up
        max_delta = min(step, h[i_idx], 1.0 - h[j_idx])
        if max_delta <= 1e-12:
            return None
        delta = rng.random() * max_delta
        new_h = h.copy()
        new_h[i_idx] = h[i_idx] - delta
        new_h[j_idx] = h[j_idx] + delta
        return new_h
```

def _repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # choose index with slack
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback distribute evenly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    add = np.clip(diff, -h[i], 1.0 - h[i])
    h[i] += add
    # Recursively fix if leftover
    if abs(diff - add) > 1e-12:
        return _repair_sum(h, target)
    return h

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust entries to satisfy sum constraint exactly."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    add = np.clip(diff, -h[i], 1.0 - h[i])
    h[i] += add
    if abs(diff - add) > 1e-12:
        return _repair_sum(h, target)
    return h

def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary (0/1) vector with sum = n/2 (half 0.5 if odd)."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    if n % 2:
        remaining = np.setdiff1d(np.arange(n), ones)
        h[rng.choice(remaining)] = 0.5
    return h

def _targeted_move(h: np.ndarray, step: float, rng: np.random.Generator):
    """Perform a move aimed at decreasing the current maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)  # integer lag
    # shift zero case – push values toward extremes
    if shift == 0:
        contrib = h * (1.0 - h)
        i = int(np.argmax(contrib))
        if h[i] > 0.5:
            donor = np.where(h < 0.5 - 1e-12)[0]
            if donor.size == 0:
                return None
            j = int(donor[np.argmin(h[donor])])
            max_delta = min(step, 1.0 - h[i], h[j])
            if max_delta <= 1e-12:
                return None
            delta = rng.random() * max_delta
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        else:
            donor = np.where(h > 0.5 + 1e-12)[0]
            if donor.size == 0:
                return None
            j = int(donor[np.argmax(h[donor])])
            max_delta = min(step, h[i], 1.0 - h[j])
            if max_delta <= 1e-12:
                return None
            delta = rng.random() * max_delta
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            return new_h
    # non‑zero shift – move mass from i to j to reduce hi*(1‑hj)
    if shift > 0:
        i_range = np.arange(0, n - shift)
        j_range = i_range + shift
    else:
        i_range = np.arange(-shift, n)
        j_range = i_range + shift
    contrib = h[i_range] * (1.0 - h[j_range])
    k = int(np.argmax(contrib))
    i_idx = int(i_range[k])
    j_idx = int(j_range[k])
    max_delta = min(step, h[i_idx], 1.0 - h[j_idx])
    if max_delta <= 1e-12:
        return None
    delta = rng.random() * max_delta
    new_h = h.copy()
    new_h[i_idx] = h[i_idx] - delta
    new_h[j_idx] = h[j_idx] + delta
    return new_h
