def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    from scipy.optimize import minimize

    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # function to compute c5
    def compute_c5_arr(h):
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # list of n to try: maybe [80, 90, 100, 110] up to 200
    ns = [80, 90, 100, 110, 120]

    for n in ns:
        # check time
        if time.time() - start_time > budget_s - 5:
            break
        # if n == len(initial_h_values) use that vector else random
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
        else:
            # random binary vector of n/2 ones
            h0 = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h0[idx] = 1.0

        # Run continuous optimization using SLSQP a few times
        # Define constraints and objective similar to previous program.
        # Use variable t as last entry.
        # number of restarts per n: maybe 4.
        # Use minimal budget for each.
        target_sum = n / 2.0
        dx = 2.0 / n

        # Pre-define constraints
        def sum_constraint(x):
            return np.sum(x[:n]) - target_sum

        # Overlap constraints using lambda capturing k
        overlap_constraints = []
        for k in range(-n+1, n):
            if k == 0:
                continue
            def make_con(k):
                def con(x):
                    h = x[:n]
                    if k >= 0:
                        overl = np.dot(h[:n - k], 1.0 - h[k:])
                    else:
                        kk = -k
                        overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                    return x[-1] - overl * dx
                return con
            overlap_constraints.append({'type': 'ineq', 'fun': make_con(k)})

        eq_cons = {'type': 'eq', 'fun': sum_constraint}
        # bounds: h in [0,1], t >= 0
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        # multiple random restarts
        restarts = 6
        for r in range(restarts):
            if time.time() - start_time > budget_s - 5:
                break
            # random binary start
            h_start = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h_start[idx] = 1.0
            # compute initial t = compute_c5
            t_start = compute_c5_arr(h_start)
            x0 = np.concatenate([h_start, [t_start]])

            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=[eq_cons] + overlap_constraints,
                options={'maxiter': 400, 'ftol': 1e-9, 'disp': False}
            )
            if not res.success:
                continue
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            # enforce exact sum
            sum_h = np.sum(h_opt)
            if abs(sum_h - target_sum) > 1e-8:
                h_opt += (target_sum - sum_h) / n
                h_opt = np.clip(h_opt, 0.0, 1.0)
            c5_val = compute_c5_arr(h_opt)
            if c5_val < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_val
                best_n = n

            # after continuous, do binary hill climbing from rounded version
            # create binary from top half
            if time.time() - start_time > budget_s - 5:
                break
            binary_h = np.zeros(n, dtype=int)
            top_idx = np.argsort(-h_opt)[: n // 2]
            binary_h[top_idx] = 1
            bin_c5 = compute_c5_arr(binary_h.astype(float))
            # hill climbing via simple swap
            # limit steps by time budget
            # define hillclimb function
            def hill_climb_bin(current_h, current_c5, time_limit):
                best_h_local = current_h.copy()
                best_val = current_c5
                while time.time() < time_limit:
                    # pick random 1 and 0 to swap
                    ones = np.flatnonzero(best_h_local == 1)
                    zeros = np.flatnonzero(best_h_local == 0)
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    cand = best_h_local.copy()
                    cand[i] = 0
                    cand[j] = 1
                    cand_val = compute_c5_arr(cand.astype(float))
                    if cand_val + 1e-12 < best_val:
                        best_h_local = cand
                        best_val = cand_val
                return best_h_local, best_val

            hill_end_time = start_time + budget_s - 5
            bin_h2, bin_c5_2 = hill_climb_bin(binary_h, bin_c5, hill_end_time)
            if bin_c5_2 < best_c5:
                best_h = bin_h2.astype(float)
                best_c5 = bin_c5_2
                best_n = n

    # After all loops, ensure best_h satisfies constraints (clip, sum)
    if best_h is None:

import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Return the C5 value for a (binary or continuous) vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _hill_climb_binary(start_h, start_val, rng, time_limit):
    """Simple swap hill‑climbing for a binary vector."""
    best_h = start_h.copy()
    best_val = start_val
    n = best_h.size
    while time.time() < time_limit:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = _compute_c5(cand.astype(float))
        if cand_val + 1e-12 < best_val:
            best_h = cand
            best_val = cand_val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # ------------------------------------------------------------------ #
    # Helper to enforce exact sum after any modification.
    def _enforce_sum(vec, n):
        target = n / 2.0
        s = np.sum(vec)
        if abs(s - target) > 1e-8:
            vec = vec + (target - s) / n
            vec = np.clip(vec, 0.0, 1.0)
        return vec

    # ------------------------------------------------------------------ #
    # List of discretisations to explore (increase resolution gradually).
    n_list = [80, 90, 100, 110, 120]

    for n in n_list:
        # stop if almost out of time
        if time.time() - start_time > budget_s - 5:
            break
        dx = 2.0 / n
        target_sum = n / 2.0

        # -------------------------------------------------------------- #
        # Phase A (only for n==80): continuous SLSQP + binary refinement.
        if n == 80:
            # Build constraints once.
            sum_con = {'type': 'eq',
                       'fun': (lambda x, n=n: np.sum(x[:n]) - n / 2.0)}
            constraints = [sum_con]

            for k in range(-n + 1, n):
                if k == 0:
                    continue
                def con(x, k=k, n=n, dx=dx):
                    h = x[:n]
                    if k >= 0:
                        overl = np.dot(h[:n - k], 1.0 - h[k:])
                    else:
                        kk = -k
                        overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                    return x[-1] - overl * dx
                constraints.append({'type': 'ineq', 'fun': con})

            bounds = [(0.0, 1.0)] * n + [(0.0, None)]

            # first consider the provided initial construction, if any
            init_candidates = []
            if 'initial_h_values' in globals() and len(initial_h_values) == n:
                init = np.array(initial_h_values, dtype=float)
                init = np.clip(init, 0.0, 1.0)
                init = _enforce_sum(init, n)
                init_candidates.append(init)

            # a few random binary starts
            for _ in range(6):
                cand = np.zeros(n, dtype=float)
                idx = rng.choice(n, n // 2, replace=False)
                cand[idx] = 1.0
                init_candidates.append(cand)

            for h_start in init_candidates:
                if time.time() - start_time > budget_s - 5:
                    break
                t_start = _compute_c5(h_start)
                x0 = np.concatenate([h_start, [t_start]])

                try:
                    res = minimize(
                        fun=lambda x: x[-1],
                        x0=x0,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=constraints,
                        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
                    )
                except Exception:
                    continue
                if not res.success:
                    continue

                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = _enforce_sum(h_opt, n)
                c5_opt = _compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5, best_n = h_opt.copy(), c5_opt, n

                # ----- binary refinement from continuous solution -----
                # turn into a binary vector by taking the largest n/2 entries
                binary_h = np.zeros(n, dtype=int)
                top_idx = np.argsort(-h_opt)[: n // 2]
                binary_h[top_idx] = 1
                bin_c5 = _compute_c5(binary_h.astype(float))

                # hill‑climb binary version
                time_limit = start_time + budget_s - 2
                bin_h2, bin_c5_2 = _hill_climb_binary(
                    binary_h, bin_c5, rng, time_limit

                if bin_c5_2 < best_c5:
                    best_h, best_c5, best_n = bin_h2.astype(float).copy(), bin_c5_2, n

        # -------------------------------------------------------------- #
        # Phase B (all n): pure binary stochastic search.
        else:

            restarts = 12
            for _ in range(restarts):
                if time.time() - start_time > budget_s - 5:

                # create random binary start
                h_bin = np.zeros(n, dtype=int)
                idx = rng.choice(n, n // 2, replace=False)
                h_bin[idx] = 1
                c5_val = _compute_c5(h_bin.astype(float))

                # hill‑climb this start
                time_limit = start_time + budget_s - 2
                bin_h2, bin_c5_2 = _hill_climb_binary(
                    h_bin, c5_val, rng, time_limit

    # Final polishing on the globally best candidate if time permits.
    if best_h is not None and time.time() - start_time < budget_s - 2:
        n = best_h.shape[0]
        # ensure binary representation
        if not np.all(np.logical_or(best_h == 0, best_h == 1)):
            binary_h = np.zeros(n, dtype=int)
            top_idx = np.argsort(-best_h)[: n // 2]
            binary_h[top_idx] = 1
        else:
            binary_h = best_h.astype(int)

        final_c5 = _compute_c5(binary_h.astype(float))
        time_limit = start_time + budget_s - 1
        binary_h3, final_c5_3 = _hill_climb_binary(
            binary_h, final_c5, rng, time_limit
        )
        if final_c5_3 < best_c5:
            best_h = binary_h3.astype(float).copy()
            best_c5 = final_c5_3
            best_n = binary_h3.shape[0]

    # ------------------------------------------------------------------ #
    # Guarantee constraints exactly.
    if best_h is None:
        # fallback: simple random binary vector at n=80
        n = 80
        best_h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        best_h[idx] = 1.0
        best_c5 = _compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _enforce_sum(best_h, best_n)

    return best_h, best_c5, int(best_n)
