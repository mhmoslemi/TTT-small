def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        # Use simple start
        n0 = 100
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)], dtype=np.float64)
    # Determine final resolution
    n_initial = h0.size
    target_n = min(800, max(200, n_initial*2)) # may tune
    # Ensure even for sum target = target_n/2.
    if target_n % 2 != 0:
        target_n += 1
    # Interpolate
    x_old = np.linspace(0.0, 2.0, n_initial, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h = np.interp(x_new, x_old, h0)
    # repair vector to target sum
    h = _repair_vector(h)
    # Global best trackers
    best_h = h.copy()
    best_score = _c5_value(best_h)
    start_time = time.time()
    # allocate time for SA
    while time.time() - start_time < budget_s * 0.8:
        remaining = budget_s - (time.time() - start_time)
        # allocate a slice: maybe min(30, remaining/5) seconds
        slice_time = min(30.0, remaining * 0.3)
        # pick start vector: either best_h with some random perturbation or random
        if rng.random() < 0.2:
            # random start
            h_start = rng.random(target_n)
        else:
            # perturb best
            perturb = rng.normal(scale=0.2, size=target_n)
            h_start = best_h + perturb
        h_start = _repair_vector(np.clip(h_start, 0.0, 1.0))
        # run SA
        h_local, score_local = _simulated_annealing(h_start, slice_time, rng)
        if score_local < best_score:
            best_h = h_local
            best_score = score_local
    # After SA, optional refinement using gradient descent
    # Simple projected gradient descent
    # Number of steps limited by remaining time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5.0:
        best_h, best_score = _gradient_refine(best_h, best_score, remaining_time, rng)
    # final repair and compute exact value
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size

    best_h = h.copy()
    best_score = cur_score

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        # determine step size: linearly decreasing
        eta = 0.2 * (1.0 - elapsed / time_limit) + 1e-4   # ensure >0
        # compute corr and find worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (N - 1)
        # compute gradient
        g = np.zeros_like(h)
        if shift >= 0:
            s = shift
            # first term
            g[:N - s] += 1.0 - h[s:]
            # second term
            g[s:] -= h[:N - s]

            g[s:] += 1.0 - h[:N - s]

def _simulated_annealing(h_start, time_limit, rng):
    N = len(h_start)
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    start = time.time()
    # schedule parameters
    T0 = 0.02  # initial temperature
    T_end = 1e-6
    # for dynamic step size:
    max_eps = 0.5
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = 1.0 - elapsed / time_limit
        T = T0 * max(frac, 0.0) + T_end
        # select i,j
        i, j = rng.integers(0, N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        # max possible epsilon to keep within bounds:
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0 and down == 0:
            continue
        # magnitude scaling: limit to max_eps * frac
        limit = max_eps * max(frac, 0.1)  # up to max_eps early, lower later
        # Random eps uniform [-down, up] but limited by limit
        eps_range = up + down
        eps = rng.random() * eps_range - down
        if abs(eps) > limit:

        if eps == 0.0:
            continue
        # apply move
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # check mask maybe not needed due to eps validity
        new_score = _c5_value(h_new)

                best_h = h_new.copy()

import numpy as np
import time
import math

def _c5_value(h: np.ndarray) -> float:
    """Return max overlap integral C5 for discretized h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v)=len(v)/2 exactly (within tolerance)."""
    n = v.size
    target = n / 2.0
    np.clip(v, 0.0, 1.0, out=v)
    s = v.sum()
    if s == 0.0:
        v[:] = target / n
        np.clip(v, 0.0, 1.0, out=v)
        s = v.sum()
    if abs(s - target) > 1e-12:
        # Scale then clip, repeat if needed
        factor = target / s
        v *= factor
        np.clip(v, 0.0, 1.0, out=v)
        s2 = v.sum()
        if abs(s2 - target) > 1e-12:
            diff = target - s2
            v += diff / n
            np.clip(v, 0.0, 1.0, out=v)
            s3 = v.sum()
            if abs(s3 - target) > 1e-12:
                v *= target / s3
                np.clip(v, 0.0, 1.0, out=v)
    return v

                     time_limit: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on the worst shift."""
    h = h_start.copy()
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    dx = 2.0 / n
    start = time.time()
    target = 0.3808
    while True:

        g = np.zeros_like(h)

            g[s:] -= h[: n - s]
