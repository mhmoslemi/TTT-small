import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Clip to [0,1] and adjust sum to target while staying in bounds."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    # greedy fill/empty until sum matches (or attempts exhausted)
    while abs(diff) > 1e-12 and attempts < 5000:
        if diff > 0.0:
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # tiny residual: distribute uniformly, then clip and renormalise once
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        total = h.sum()
        if abs(total - target) > 1e-12:
            h *= target / total
            np.clip(h, 0.0, 1.0, out=h)
    return h
