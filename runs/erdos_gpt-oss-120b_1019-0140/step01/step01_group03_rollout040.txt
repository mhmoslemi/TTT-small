def optimizer(h0, budget_time, start_time, max_iter, step0=1.0, momentum=0.9, top_k=5):
    h = h0.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    dx = 2.0 / n
    offset = n - 1
    for it in range(1, max_iter+1):
        # time check
        if time.time() - start_time > budget_time - 0.1:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        f = corr * dx
        # Determine top_k indices
        if top_k >= f.size:
            top_indices = np.arange(f.size)
        else:
            # use argpartition for max k
            idx = np.argpartition(f, -top_k)[-top_k:]
            top_indices = idx
        # randomly pick one shift weighted by f values
        probs = f[top_indices] - f.min()  # to get non-negative? maybe use just f values
        if probs.sum() <= 0:
            probs = np.ones_like(probs)
        probs = probs / probs.sum()
        chosen_idx = rng.choice(top_indices, p=probs)
        shift = chosen_idx - offset
        grad = compute_gradient(h, shift)  # shape (n,)
        # momentum update
        v = momentum * v + (1 - momentum) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * v
        # projection
        h_new = project_onto_box_sum(h_new, n/2.0)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def binary_hill_climb(h, O, time_end, rng):
    n = h.shape[0]
    if O is None:
        O = np.correlate(h, h, mode='full')[n-1:]
    # Ensure O[0] = n/2
    best_O = O
    best_min = O.min()
    best_h = h.copy()
    # We'll attempt many swaps
    # Let's set a loop with a max iteration count but break on time.
    while time.time() < time_end:
        # pick random i where h[i]==1 and j where h[j]==0
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)  # alternative
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Compute delta O
        delta = np.zeros(n, dtype=int)
        # i (1->0)
        # first index contributions
        if i < n - 1:
            delta[1:n - i] += -h[i + 1:]
        if i > 0:
            delta[1:i + 1] += -h[i - 1::-1]
        # j (0->1)
        if j < n - 1:
            delta[1:n - j] += h[j + 1:]
        if j > 0:
            delta[1:j + 1] += h[j - 1::-1]
        # adjust for pair
        dist = abs(i - j)
        if dist > 0:
            delta[dist] -= 1  # since delta_i*delta_j = -1
        # compute new O and min
        O_new = best_O + delta
        # Ensure O_new[0] stays n/2 (should not change)
        # compute new min
        new_min = O_new.min()
        if new_min > best_min:
            # accept swap
            h[i] = 0
            h[j] = 1
            best_O = O_new
            best_min = new_min
            best_h = h.copy()
        # else reject and continue
    return best_h, best_O

    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

        iter_count += 1
        # correlation
        corr = np.correlate(h, 1 - h, mode='full')
        f = corr * (2.0 / n)  # dx factor
        # find top_k shifts
        if top_k >= f.size:
            idxs = np.arange(f.size)
        else:
            idxs = np.argpartition(f, -top_k)[-top_k:]
        # compute weights
        scores = f[idxs]
        # shift scores to non-negative
        # we could use softmax-like but simple weighting
        rel = scores - scores.min() + 1e-12
        w = rel / rel.sum()
        grad = np.zeros_like(h)
        for idx, weight in zip(idxs, w):
            shift = idx - offset
            grad += weight * compute_gradient(h, shift)
        # momentum update
        v = momentum * v + (1 - momentum) * grad
        eta = step0 / math.sqrt(iter_count)
        h_new = h - eta * v
        # projection onto constraints
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_gradient(h, shift):
    """Gradient of one shift term f_k = ∫ h(1-h_shift)."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def compute_c5(h):
    """C5 = max_k ∫ h (1-h_shift) dx."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def swap_delta_O(i, j, h):
    """Delta of overlap vector O when swapping h[i]=1->0 with h[j]=0->1."""
    n = h.size
    delta = np.zeros(n, dtype=int)
    # i : 1->0, delta_i = -1
    if i < n - 1:
        delta[1:n - i] += -h[i + 1:]
    if i > 0:
        delta[1:i + 1] += -h[i - 1::-1]
    # j : 0->1, delta_j = +1
    if j < n - 1:
        delta[1:n - j] += h[j + 1:]
    if j > 0:
        delta[1:j + 1] += h[j - 1::-1]
    # correct double-counted pair (i,j)
    dist = abs(i - j)
    if dist > 0:
        delta[dist] -= 1   # delta_i * delta_j = -1
    return delta

def binary_hill_climb(h, O, time_end, rng):
    """Improve binary h by swapping to increase min overlap."""
    n = h.shape[0]
    best_h = h.copy()
    best_O = O.copy()
    best_min = best_O.min()
    while time.time() < time_end:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        delta = swap_delta_O(i, j, best_h)
        new_O = best_O + delta
        new_min = new_O.min()
        if new_min > best_min:
            best_h[i] = 0
            best_h[j] = 1
            best_O = new_O
            best_min = new_min
    return best_h, best_O

def gradient_optimize(h_start, time_end, rng):
    """Projected gradient descent with momentum, using top‑k shift weighting."""
    h = h_start.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    v = np.zeros_like(h)
    momentum = 0.9
    step0 = 0.5
    top_k = 5
    offset = n - 1
    it = 0
    while time.time() < time_end:
        it += 1
        corr = np.correlate(h, 1 - h, mode='full')
        f = corr * (2.0 / n)   # dx factor applied
        if top_k >= f.size:
            idxs = np.arange(f.size)
        else:
            idxs = np.argpartition(f, -top_k)[-top_k:]
        scores = f[idxs]
        rel = scores - scores.min() + 1e-12
        w = rel / rel.sum()
        grad = np.zeros_like(h)
        for idx, weight in zip(idxs, w):
            shift = idx - offset
            grad += weight * compute_gradient(h, shift)
        v = momentum * v + (1.0 - momentum) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * v
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
