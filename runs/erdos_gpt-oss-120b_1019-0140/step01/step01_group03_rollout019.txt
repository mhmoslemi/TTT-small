import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient(h, shift):
    """Gradient of the active shift term."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2


def projected_gradient_descent(h0, n_iter=2000, step0=0.5, deadline=None):
    """Subgradient descent on max-correlation objective with Euclidean projection."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        if deadline is not None and time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def thue_morse(n):
    """Generate binary Thue–Morse sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq


def random_balanced_binary(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample(arr, new_n):
    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.astype(float).copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr).astype(float)


def sine_pattern(n, k):
    """Smooth sinusoidal pattern with k half‑periods over [0,2]."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.5 * np.sin(np.pi * k * x / 2.0)  # period = 4/k -> k half‑cycles
    return np.clip(h, 0.0, 1.0)
