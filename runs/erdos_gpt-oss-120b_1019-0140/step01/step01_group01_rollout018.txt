import time
import numpy as np

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Scale/redistribute h so that sum(h) == target while keeping 0 ≤ h_i ≤ 1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # If the needed correction is small, distribute uniformly
    if abs(diff) < 1e-6:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    # Try to fix it by moving mass between random indices
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    # If still not exact, do a final uniform adjustment and clip
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def max_feasible_step(h: np.ndarray, direction: np.ndarray) -> float:
    """
    Largest scalar s ≥ 0 such that 0 ≤ h + s*direction ≤ 1 componentwise.
    Assumes direction has zero sum (so total mass is conserved).
    """
    pos = direction > 1e-12
    neg = direction < -1e-12
    step_pos = np.min((1.0 - h[pos]) / direction[pos]) if np.any(pos) else np.inf
    step_neg = np.min((-h[neg]) / direction[neg]) if np.any(neg) else np.inf
    return min(step_pos, step_neg)

        move_type = rng.rand()
        if move_type < 0.35:
            # ----- gradient move on worst lag -----
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            lag = int(np.argmax(corr_full)) - (n_points - 1)

            g = np.ones_like(current_h)          # gradient of the worst overlap term
            if lag == 0:
                g -= 2.0 * current_h
            else:
                idx = np.arange(n_points)
                plus = idx + lag
                mask = (plus >= 0) & (plus < n_points)
                g[mask] -= current_h[plus[mask]]
                minus = idx - lag
                mask = (minus >= 0) & (minus < n_points)
                g[mask] -= current_h[minus[mask]]

            # direction that decreases the worst overlap, projected to sum=0
            direction = -(g - g.mean())
            max_step = max_feasible_step(current_h, direction)
            if max_step > 0.0:
                actual_step = min(step, max_step * 0.9)
                candidate = current_h + direction * actual_step
                np.clip(candidate, 0.0, 1.0, out=candidate)
                candidate = adjust_sum_to_target(candidate, target_sum, rng)
            else:
                continue

        elif move_type < 0.7:
            # ----- pairwise exchange move -----
            i, j = rng.randint(n_points, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = np.clip(delta, low, high)
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        else:
            # ----- global random perturbation -----
            noise = rng.randn(n_points) * step * 0.5
            candidate = current_h + noise
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)

        cand_c5 = compute_c5(candidate)

        # acceptance rule (simulated annealing)
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                last_improve = now

        # occasional restart if stagnation > 30 s and still enough time left
        if (now - last_improve) > 30.0 and progress < 0.9:
            sigma = max(step, 0.02)
            current_h = best_h + rng.randn(n_points) * sigma
            np.clip(current_h, 0.0, 1.0, out=current_h)
            current_h = adjust_sum_to_target(current_h, target_sum, rng)
            current_c5 = compute_c5(current_h)
            last_improve = now

    # ----- final deterministic refinement on the best solution -----
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        # compute worst lag for the current best
        corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
        lag = int(np.argmax(corr_full)) - (n_points - 1)

        g = np.ones_like(best_h)
        if lag == 0:
            g -= 2.0 * best_h
        else:
            idx = np.arange(n_points)
            plus = idx + lag
            mask = (plus >= 0) & (plus < n_points)
            g[mask] -= best_h[plus[mask]]
            minus = idx - lag
            mask = (minus >= 0) & (minus < n_points)
            g[mask] -= best_h[minus[mask]]

        direction = -(g - g.mean())
        max_step = max_feasible_step(best_h, direction)
        if max_step <= 1e-9:
            break
        step_size = min(1e-4, max_step * 0.9)
        candidate = best_h + direction * step_size
        np.clip(candidate, 0.0, 1.0, out=candidate)
        candidate = adjust_sum_to_target(candidate, target_sum, rng)
        cand_c5 = compute_c5(candidate)
        if cand_c5 < best_c5 - 1e-9:
            best_c5 = cand_c5
            best_h = candidate.copy()
        else:
            # if no improvement for a few attempts, break
            break

    # final sanity
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
