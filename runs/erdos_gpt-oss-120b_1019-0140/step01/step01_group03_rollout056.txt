def optimize_initial_h(start_h, max_iters, max_time):
    h = start_h.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.2
    best_local_h = h.copy()
    best_local_score = compute_c5(h)
    t = 1
    while t <= max_iters and (time.time() - start_time) < max_time:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        # compute near-opt shifts
        # tolerance absolute maybe 1e-4 or relative.
        tol = max(1e-4, 1e-3 * max_corr)
        near_idx = np.where(corr >= max_corr - tol)[0]
        grad = np.zeros_like(h)
        for idx in near_idx:
            shift = idx - (len(h) - 1)
            grad += compute_gradient(h, shift)
        grad /= len(near_idx)  # average
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, len(h)/2)
        score = compute_c5(h_new)
        if score < best_local_score:
            best_local_score = score
            best_local_h = h_new.copy()
        h = h_new
        t += 1
    return best_local_h, best_local_score
```

import time
import math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=30):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    # Define f(lambda) = sum clip(v - lambda, lo, hi)
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    # Ensure brackets contain the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linearly interpolate arr to length new_n (preserving total sum)."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    interp = np.interp(x_new, x_old, arr)
    # after interpolation enforce exact sum
    return project_onto_box_sum(interp, new_n / 2.0)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient(h, shift):
    """Gradient of correlation term for a given integer shift."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[: n - shift] = h[shift:]
        term2[shift:] = h[: n - shift]
    else:
        s = -shift
        term1[s:] = h[: n - s]
        term2[: n - s] = h[s:]
    return 1.0 - term1 - term2


def thue_morse(n):
    """Binary Thue–Morse sequence of length n (float 0/1)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq


def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def binarize_by_threshold(h):
    """Return a balanced binary vector by thresholding the top n/2 entries."""
    n = h.size
    thresh = np.partition(h, -n // 2)[-n // 2]
    binary = (h > thresh).astype(float)
    # in case of ties adjust
    cur = binary.sum()
    if cur < n // 2:
        # add missing ones at positions equal to thresh
        need = n // 2 - int(cur)
        ties = np.where(h == thresh)[0]
        binary[ties[:need]] = 1.0
    elif cur > n // 2:
        # remove excess ones
        need = int(cur) - n // 2
        ties = np.where(h == thresh)[0]
        binary[ties[:need]] = 0.0
    return binary


def optimize_h_adam(h0, max_iters, start_time, budget, step0=0.2):
    """Projected Adam optimizer for the max-correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)

    for t in range(1, max_iters + 1):
        if time.time() - start_time > budget - 0.1:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # tolerance for near-maximum shifts
        tol = max(1e-3, 0.001 * max_corr)
        idxs = np.where(corr >= max_corr - tol)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            shift = idx - (n - 1)
            grad += compute_gradient(h, shift)
        grad /= len(idxs)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)

        h = h - step
        h = project_onto_box_sum(h, s)

        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            h = h_new
    return best_h, best_score


def simulated_annealing_binary(h_bin, max_iters, start_time, budget, rng):
    """Simulated annealing on balanced binary vectors."""
    n = h_bin.size
    h = h_bin.copy()
    current_score = compute_c5(h)
    best_score = current_score
    best_h = h.copy()
    T0 = 0.02  # initial temperature
    for it in range(1, max_iters + 1):
        if time.time() - start_time > budget - 0.1:
            break
        T = T0 * (1.0 - it / max_iters)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        score = compute_c5(h_new)
        delta = score - current_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            h = h_new
            current_score = score
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target_n = 256
    target_bound = 0.3808

    # ------------------------------------------------------------------
    # Helper to monitor remaining time
    # ------------------------------------------------------------------
    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Build initial candidate pool
    # ------------------------------------------------------------------
    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != target_n:
            init = upsample(init, target_n)
        init = project_onto_box_sum(init, target_n / 2.0)
        candidates.append(init)

    # Thue–Morse
    tm = thue_morse(target_n).astype(float)
    candidates.append(tm)

    # Random binary & fractional starts
    for _ in range(4):
        candidates.append(random_binary_balanced(target_n, rng))
        v = rng.random(target_n)
        candidates.append(project_onto_box_sum(v, target_n / 2.0))

    best_h = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optimize each base candidate with a relatively long run
    # ------------------------------------------------------------------
    for base_h in candidates:
        if time_left() <= 1.0:
            break
        iters = 8000 if base_h.size == target_n else 4000
        h_opt, score_opt = optimize_h_adam(
            base_h,
            max_iters=iters,
            start_time=start_time,
            budget=time_left(),
            step0=0.2,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
            if best_score <= target_bound:
                break

    # ------------------------------------------------------------------
    # Random restarts until time runs out
    # ------------------------------------------------------------------
    while time_left() > 1.0:
        # pick start type randomly
        if rng.random() < 0.5:
            start_h = random_binary_balanced(target_n, rng)
        else:
            start_h = project_onto_box_sum(rng.random(target_n), target_n / 2.0)

        h_opt, score_opt = optimize_h_adam(
            start_h,
            max_iters=2000,
            start_time=start_time,
            budget=time_left(),
            step0=0.2,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
            if best_score <= target_bound:
                break

    # ------------------------------------------------------------------
    # If best solution looks nearly binary, enforce binary and refine
    # ------------------------------------------------------------------
    if best_h is not None:
        # Create a binary version by thresholding
        binary_h = binarize_by_threshold(best_h)
        binary_score = compute_c5(binary_h)
        if binary_score < best_score:
            best_h = binary_h
            best_score = binary_score

        # Apply hill climbing on the binary version
        if time_left() > 1.0:
            h_hc, score_hc = hill_climb_binary_swap(
                best_h,
                max_iters=5000,
                start_time=start_time,
                budget=time_left(),
                rng=rng,
            )
            if score_hc < best_score:
                best_h, best_score = h_hc, score_hc

        # Apply a short simulated annealing pass
        if time_left() > 1.0:
            h_sa, score_sa = simulated_annealing_binary(
                best_h,
                max_iters=4000,
                start_time=start_time,
                budget=time_left(),
                rng=rng,
            )
            if score_sa < best_score:
                best_h, best_score = h_sa, score_sa

    # ------------------------------------------------------------------
    # Final safety projection and compute final score
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(target_n, 0.5, dtype=float)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
