import time
import numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    v = np.asarray(v, dtype=float)
    n = v.size
    # Quick extreme cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def clipped_sum(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure bounds
    if clipped_sum(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if clipped_sum(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        sm = clipped_sum(lam_mid)
        if abs(sm - s) < 1e-12:
            break
        if sm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.size
    grad = np.ones_like(h)
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad

def subgradient_descent(h0, max_iter=5000, step0=0.3):
    h = h0.copy()
    n = h.size
    s = n * 0.5
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / np.sqrt(it)
        h = project_onto_box_sum(h - eta * grad, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def hill_climb_swap(h_start, score_start, rng, start_time, budget_s, max_iters=200000, target=0.3808):
    h = h_start.copy()
    cur_score = score_start
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    no_improve = 0
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 0.5:
            break
        # random pair
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # determine direction of moving mass
        if h[i] > 0 and h[j] < 1:
            src, dst = i, j
        elif h[j] > 0 and h[i] < 1:
            src, dst = j, i
        else:
            continue
        delta_max = min(h[src], 1.0 - h[dst])
        if delta_max <= 0:
            continue
        # occasionally make a small fractional move
        if rng.random() < 0.3:
            delta = delta_max * rng.random()
        else:
            delta = delta_max
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_score = compute_c5(h_new)
        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_h = h_new.copy()
            h = h_new
            cur_score = new_score
            no_improve = 0

                break

        if no_improve >= 20000:
            # random perturbation
            h = project_onto_box_sum(rng.random(n), n * 0.5)
            cur_score = compute_c5(h)
            no_improve = 0
    return best_h, best_score

import time
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo<=x_i<=hi, sum x_i = s}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def clipped_sum(lam):
        return np.clip(v - lam, lo, hi).sum()

    if clipped_sum(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if clipped_sum(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        sm = clipped_sum(lam_mid)
        if abs(sm - s) < 1e-12:
            break
        if sm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1 - h(x+k)) dx on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    else:
        s = -shift
        grad[s:] -= h[:n - s]

    return grad

    h = h0.copy()
    n = h.size
    s = n * 0.5
    best_h = h.copy()

    for it in range(1, max_iter + 1):

        idx = np.argmax(corr)

        score = compute_c5(h)

            best_h = h.copy()
