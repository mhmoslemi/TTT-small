def _compute_c5(v):
    n = v.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length()) # next power of two.
    # Compute convolution
    v_fft = np.fft.rfft(v, L)
    w_rev = (1.0 - v)[::-1]
    w_fft = np.fft.rfft(w_rev, L)
    conv = np.fft.irfft(v_fft * w_fft, L)[:2 * n - 1]
    return np.max(conv) * dx

def _simulated_annealing(v, time_budget):
    # v initial feasible
    n = len(v)
    cur_v = v.copy()
    cur_score = _compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    start = time.time()
    # parameters
    T0 = 0.01  # initial temperature
    # step size: max delta proportion of allowed (maybe 0.2)
    max_step = 0.2
    # schedule to decay T linear to near 0 at end
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # compute temperature based on progress
        # Using linear schedule: T = T0 * (1 - elapsed / time_budget)
        T = T0 * (1.0 - elapsed / time_budget)
        # If T is very small, set to small epsilon to avoid division by zero
        if T < 1e-9:
            T = 1e-9
        # propose move
        i, j = np.random.choice(n, 2, replace=False)
        # direction random: try transferring from i to j
        max_fwd = min(max_step, cur_v[i], 1.0 - cur_v[j])
        max_rev = min(max_step, cur_v[j], 1.0 - cur_v[i])
        if max_fwd <= 0 and max_rev <= 0:
            continue
        # choose direction
        if max_fwd > 0 and max_rev > 0:
            if np.random.rand() < 0.5:
                max_transfer = max_fwd
                src, dst = i, j
            else:
                max_transfer = max_rev
                src, dst = j, i
        elif max_fwd > 0:
            max_transfer = max_fwd
            src, dst = i, j
        else:
            max_transfer = max_rev
            src, dst = j, i
        # random delta uniform in (0, max_transfer)
        delta = np.random.rand() * max_transfer
        # apply move
        cur_v[src] -= delta
        cur_v[dst] += delta
        # compute new_score
        new_score = _compute_c5(cur_v)
        delta_score = new_score - cur_score
        accept = False
        if delta_score <= 0:
            accept = True
        else:
            # acceptance probability
            if np.random.rand() < np.exp(-delta_score / T):
                accept = True
        if accept:

            if new_score < best_score:
                best_score = new_score
                best_v = cur_v.copy()
        else:
            # revert
            cur_v[src] += delta

def _greedy_refine(v, time_budget):
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5_fft(best_v)
    start = time.time()
    step_max = 0.02
    while time.time() - start < time_budget:
        i, j = np.random.choice(n, 2, replace=False)
        max_transfer = min(step_max, best_v[i], 1 - best_v[j])
        # also consider reverse
        max_rev = min(step_max, best_v[j], 1 - best_v[i])
        if max_transfer <= 0 and max_rev <= 0:
            continue
        # pick direction randomly among feasible
        if max_transfer > 0 and max_rev > 0:
            if np.random.rand() < 0.5:
                max_tr = max_transfer; src, dst = i, j
            else:
                max_tr = max_rev; src, dst = j, i
        elif max_transfer > 0:
            max_tr = max_transfer; src, dst = i, j
        else:
            max_tr = max_rev; src, dst = j, i
        delta = np.random.rand() * max_tr
        best_v[src] -= delta
        best_v[dst] += delta
        new_score = _compute_c5_fft(best_v)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best_v[src] += delta
            best_v[dst] -= delta
    return best_v, best_score

def _compute_c5_fft(v):
    n = v.shape[0]
    dx = 2.0 / n
    L = _next_pow2(2 * n - 1)
    v_fft = np.fft.rfft(v, n=L)
    w_rev = (1.0 - v)[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(v_fft * w_fft, n=L)[:2 * n - 1]
    return np.max(conv) * dx

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _repair_vector(v: np.ndarray, n: int, max_iter: int = 15) -> np.ndarray:
    """Project v onto the feasible set: 0<=v<=1 and sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _random_feasible(n: int) -> np.ndarray:
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _upsample_vector(v: np.ndarray, n_target: int) -> np.ndarray:
    """Linearly interpolate v to length n_target and repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _compute_c5_fft(v: np.ndarray) -> float:
    """Fast evaluation of C5 using FFT."""
    n = v.shape[0]
    dx = 2.0 / n
    L = _next_pow2(2 * n - 1)
    v_fft = np.fft.rfft(v, n=L)
    w_rev = (1.0 - v)[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(v_fft * w_fft, n=L)[: 2 * n - 1]
    return np.max(conv) * dx

def _compute_c5_brute(v: np.ndarray) -> float:
    """Exact (O(n^2)) evaluation of C5 using np.correlate."""
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _simulated_annealing(v0: np.ndarray, time_budget: float) -> tuple:
    """SA that preserves the sum (transfer between two coordinates)."""
    n = v0.shape[0]
    cur_v = v0.copy()
    cur_score = _compute_c5_fft(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    start = time.time()
    max_step = 0.2
    T0 = 0.01
    final_T = 1e-9
    restart_prob = 0.001

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # temperature schedule (linear decay)
        frac = elapsed / time_budget
        T = T0 * (1.0 - frac) + final_T

        # occasional restart from best with small jitter
        if np.random.rand() < restart_prob:
            cur_v = best_v.copy()
            cur_v += (np.random.rand(n) - 0.5) * 0.02
            cur_v = _repair_vector(cur_v, n)
            cur_score = _compute_c5_fft(cur_v)

        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # possible transfers
        max_fwd = min(max_step, cur_v[i], 1.0 - cur_v[j])   # i -> j
        max_rev = min(max_step, cur_v[j], 1.0 - cur_v[i])   # j -> i

        if max_fwd <= 0 and max_rev <= 0:
            continue

        if max_fwd > 0 and max_rev > 0:
            if np.random.rand() < 0.5:
                max_tr, src, dst = max_fwd, i, j
            else:
                max_tr, src, dst = max_rev, j, i
        elif max_fwd > 0:
            max_tr, src, dst = max_fwd, i, j
        else:
            max_tr, src, dst = max_rev, j, i

        delta = np.random.rand() * max_tr
        cur_v[src] -= delta
        cur_v[dst] += delta

        new_score = _compute_c5_fft(cur_v)
        delta_score = new_score - cur_score

        accept = False
        if delta_score <= 0:
            accept = True
        else:
            if np.random.rand() < np.exp(-delta_score / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_v = cur_v.copy()
        else:
            # revert
            cur_v[src] += delta
            cur_v[dst] -= delta

    return best_v, best_score

    n = v0.shape[0]

        cur_v[src] -= delta

        new_score = _compute_c5_fft(cur_v)
