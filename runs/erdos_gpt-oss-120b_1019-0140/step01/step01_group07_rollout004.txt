def _grad_smoothed(h, alpha):
    n = h.size
    # raw correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    # smoothing weights
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    w = exp_vals / np.sum(exp_vals)  # shape (2n-1)
    # compute conv using FFT
    w_rev = w[::-1]
    # convolution via FFT
    L = len(h) + len(w) - 1
    # use real FFT
    H = np.fft.rfft(h, L)
    W = np.fft.rfft(w, L)
    W_rev = np.fft.rfft(w_rev, L)
    conv_bwd = np.fft.irfft(H * W, L)
    conv_fwd = np.fft.irfft(H * W_rev, L)
    # slice central region: indices n-1 : n-1+n
    start = n - 1
    conv_slice = conv_fwd[start:start+n] + conv_bwd[start:start+n]
    grad = np.ones(n) - conv_slice
    return grad, corr

Now correctness: Because we used convolution of h with w, we get sum_k w_k * h_{j - k}: correct. The sum of both gives total shift contributions.

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if np.abs(cur - target) < 1e-9:
        return v
    # Use water-filling approach to adjust sum while staying in [0, 1]
    # Equivalent to solving: minimize ||v - v0||^2 subject to sum(v)=target, 0<=v<=1.
    # Can be done via sorting.
    # We'll implement generic algorithm:
    # We need to solve: v_i = clip(u_i + lambda, 0, 1) where u_i are original v (maybe after clipping).
    # Find lambda such that sum(clip(u_i + lambda,0,1)) = target.
    # Use binary search on lambda.
    # This is efficient.
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        v_adj = np.clip(v + lam, 0.0, 1.0)
        s = np.sum(v_adj)
        if s > target:
            hi = lam
        else:
            lo = lam
    v = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # ensure exact sum by scaling small difference
    cur = np.sum(v)
    if np.abs(cur - target) > 1e-8:
        # distribute residual proportionally to non-bound entries
        mask = (v > 1e-12) & (v < 1 - 1e-12)
        if np.any(mask):
            v[mask] *= target / cur
        else:
            # fallback scaling of all entries
            v *= target / cur
    v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _pgd_optimize(h0, target_sum, budget_s, global_start_time):

    # hyperparams

def _compute_c5(h):
    """Maximum overlap integral for a sampled step function."""
