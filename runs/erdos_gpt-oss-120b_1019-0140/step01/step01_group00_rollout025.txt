from scipy.optimize import minimize

class SumConstraint: ...
class OverlapConstraint: ...

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def objective_t(x):
    return x[-1]

def generate_initial_h(n, M, rng):
    # random choice of generation method
    method = rng.integers(3)
    if method == 0:
        # random binary
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:
        # random continuous uniform then project
        h = rng.random(n)
        # scale to sum M
        # project using simple algorithm: h = h / sum(h) * M
        h = h / np.sum(h) * M
        # clip to [0,1] and adjust
        h = np.clip(h, 0, 1)
        cur = np.sum(h)
        if cur < M:
            # add to smallest values
            deficit = M - cur
            # iterate over sorted indices
            for i in np.argsort(h):
                inc = min(deficit, 1 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    else:
        # linear decreasing shape
        lin = np.linspace(1, 0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0, 1)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    return h

def optimize_for_n(n, total_time, rng):

    dx = 2.0 / n
    target_sum = M

    sum_con = SumConstraint(n)  # equality constraint sum(h) == n/2
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type':'eq','fun': sum_con}]
    constraints += [{'type':'ineq','fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf

    # we allocate remaining time for this n as the min of (total_time - elapsed) and some per-n limit.
    while time.time() - start < total_time:

                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 800, 'ftol': 1e-10, 'disp': False})
        except Exception:
            continue
        if not res.success:
            continue

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Constraint helpers ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        # x includes h (first n entries) and auxiliary t at the end
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ---------- Objective ----------
def compute_c5(h):
    """Maximum overlap integral for a given (continuous) h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_t(x):
    """Auxiliary variable t to be minimised."""
    return x[-1]

# ---------- Initial vector generators ----------
def generate_initial_h(n, M, rng):
    """Return a feasible h (sum = M, 0≤h≤1) using a random scheme."""
    method = rng.integers(3)
    if method == 0:  # binary random
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    elif method == 1:  # uniform random then project to sum M
        h = rng.random(n)
        h = h / np.sum(h) * M
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:
                    break
    else:  # linear decreasing shape
        lin = np.linspace(1.0, 0.0, n)
        scaling = M / np.sum(lin)
        h = lin * scaling
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if cur < M:
            deficit = M - cur
            for i in np.argsort(h):
                inc = min(deficit, 1.0 - h[i])
                h[i] += inc
                deficit -= inc
                if deficit <= 1e-12:
                    break
        elif cur > M:
            excess = cur - M
            for i in np.argsort(-h):
                dec = min(excess, h[i])
                h[i] -= dec
                excess -= dec
                if excess <= 1e-12:

def optimise_for_n(n, time_budget, rng):
    """Run SLSQP restarts for a given n within time_budget seconds."""
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
