def project_to_feasible(h, target_sum, tol=1e-9, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # all at bounds, cannot adjust; break
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction if still off
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-8:
        # adjust a single entry with capacity
        if diff > 0:
            # need to increase sum: find index with h < 1
            idx = np.argmin(h)  # smallest value (most capacity)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            # need to decrease sum: find index with h > 0
            idx = np.argmax(h)  # largest value
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-9, max_iter=15):
    """Project h onto [0,1]^n while enforcing sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if still off
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-8:
        if diff > 0:
            idx = np.argmin(h)           # most room to increase
        else:
            idx = np.argmax(h)           # most room to decrease
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def hill_climb_binary(h, n, max_swaps, rng, start_time, time_budget):
    """Perform swap-based hill climbing on a binary vector."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_swaps):
        if time.time() - start_time > time_budget * 0.97:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        cand = best_h.copy()
        cand[i1] = 0
        cand[i0] = 1
        c5cand = compute_c5(cand)
        if c5cand < best_c5 - 1e-12:
            best_h = cand
            best_c5 = c5cand
    return best_h, best_c5


def local_refine(h, n, rng, start_time, time_budget):
    """Small random perturbations followed by projection."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    # geometric schedule of perturbation scales
    scales = np.geomspace(0.05, 0.001, num=8)

            c5cand = compute_c5(cand)

                bnds = [(0.0, 1.0)] * n
                try:
                    res = opt.minimize(
                        fun=lambda h: compute_c5(h),
                        x0=h_cont,
                        method="SLSQP",
                        bounds=bnds,
                        constraints=cons,
                        options={"maxiter": 500, "ftol": 1e-9, "disp": False},
                    )
                except Exception:
                    continue
                if res.success:
                    h_opt = np.clip(res.x, 0.0, 1.0)
                    h_opt = project_to_feasible(h_opt, target_sum)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        best_n = n

            # Odd-indexed restarts: pure continuous SLSQP from random start
            else:
                if base_h is not None and restart == 1:
                    h0 = base_h.copy()
                else:
                    h0 = rng.random(n)
                    h0 = project_to_feasible(h0, target_sum)

                # Add small noise on later restarts to diversify
                if restart > 2:
                    h0 = np.clip(h0 + rng.normal(scale=0.07, size=n), 0.0, 1.0)
                    h0 = project_to_feasible(h0, target_sum)

                cons = {"type": "eq", "fun": lambda h, nd=n: np.sum(h) - nd / 2.0}
                bnds = [(0.0, 1.0)] * n
                try:
                    res = opt.minimize(
                        fun=lambda h: compute_c5(h),
                        x0=h0,
                        method="SLSQP",
                        bounds=bnds,
                        constraints=cons,
                        options={"maxiter": 800, "ftol": 1e-9, "disp": False},
                    )
                except Exception:
                    continue
                if not res.success:
                    continue
                h_opt = np.clip(res.x, 0.0, 1.0)
                h_opt = project_to_feasible(h_opt, target_sum)
                c5_opt = compute_c5(h_opt)

                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n

                # Fine‑grained refinement after SLSQP
                h_ref, c5_ref = local_refine(
                    h_opt, n, rng, start_time, budget_s
                )
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
                    best_n = n

            # Early exit if we already beat the target
            if best_c5 <= 0.3808:
                # continue a few more iterations for possible further gains
                pass

    # Fallback if nothing succeeded (should not happen)
    if best_h is None:
        best_n = 380
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)

    # Final safety projection and recompute
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
