def compute_gradient(h, shift):
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def compute_multi_gradient(h, top_k=3):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    # find indices of top_k largest values
    # note: that's the correlation values themselves; we need shift values
    # (largest corresponds to worst overlap)
    # take argsort descending etc.
    flat_idx = np.argpartition(corr, -top_k)[-top_k:]  # unsorted
    # compute shifts: shift = idx - (n - 1)
    shifts = flat_idx - (n - 1)
    grads = np.zeros((top_k, n), dtype=h.dtype)
    for i, shift in enumerate(shifts):
        grads[i] = compute_gradient(h, shift)
    # average
    return grads.mean(axis=0)

def projected_gradient_descent_multi(h0, max_iters=5000, step0=1.0, top_k=3):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters+1):
        # compute gradient using top_k worst shifts
        grad = compute_multi_gradient(h, top_k)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # early exit if time limit flagged? Not in this function; the caller will handle.
    return best_h, best_score

def anneal_binary(h0, start_time, budget_s, rng):
    # h0 binary (0/1) with sum = n/2
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    # temperature schedule
    T0 = 0.02
    # max iterations at most some fixed; we'll loop until time left small
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s - 0.1:
            break
        # pick random indices i (where h[i]==1) and j (where h[j]==0)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_score = compute_c5(h_new)
        delta = new_score - best_score
        # temperature falls linearly with elapsed time
        fraction = elapsed / budget_s
        T = T0 * max(0.01, 1 - fraction)
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h = h_new
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()

import time, math
import numpy as np

def resample(arr, new_n):
    """Linear interpolation from arr to length new_n (periodic on [0,2))."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60, tol=1e-12):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets contain the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) + 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < tol:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the overlap term for a single integer shift."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def compute_multi_gradient(h, top_k=3):
    """Average gradient over the top_k worst shifts (subgradient of max)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    # indices of the largest correlations (worst overlaps)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    shifts = idx - (n - 1)
    grads = np.empty((top_k, n), dtype=h.dtype)
    for i, s in enumerate(shifts):
        grads[i] = compute_gradient(h, s)
    return grads.mean(axis=0)

def projected_gradient_descent_multi(h0, max_iters=5000, step0=1.0, top_k=3):
    """PGD using multi‑gradient on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters + 1):
        grad = compute_multi_gradient(h, top_k)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_binary(h0, time_limit, rng):
    """Simple simulated annealing on a balanced binary vector."""
    start = time.time()
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    T0 = 0.02
    it = 0
    max_iter = 20000
    while it < max_iter and (time.time() - start) < time_limit:
        it += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_score = compute_c5(h_new)
        delta = new_score - best_score
        # linear temperature decay
        frac = (time.time() - start) / time_limit
        T = T0 * max(0.01, 1.0 - frac)
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h = h_new
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score
