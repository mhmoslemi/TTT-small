import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _compute_c5_and_shift(h: np.ndarray):
    """Return C5 and the shift (lag) attaining the maximum."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    c5 = float(corr[idx] * dx)
    return c5, shift

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that sum(h) == target, respecting bounds."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # distribute evenly (rare)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _gradient_wrt_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the overlap sum for a given shift (lag)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # + (1 - h[i+s]) for i in [0, n-s)
        grad[:n - s] += 1.0 - h[s:]
        # - h[i-s] for i in [s, n)
        grad[s:] -= h[:n - s]
    else:
        s = -shift
        # + (1 - h[i-s]) for i in [s, n)
        grad[s:] += 1.0 - h[:n - s]
        # - h[i+s] for i in [0, n-s)
        grad[:n - s] -= h[s:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that lowers the Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise a feasible candidate -----
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = int(init.shape[0])
        target_sum = n / 2.0
        init = np.clip(init, 0.0, 1.0)
        init = _repair_sum(init, target_sum)
        current = init.copy()
    except NameError:
        # No pre‑supplied construction; start with half‑ones pattern.
        n = 200
        target_sum = n / 2.0
        current = np.zeros(n, dtype=float)
        half = n // 2
        current[:half] = 1.0
        if n % 2:
            current[half] = 0.5
        rng.shuffle(current)
        current = _repair_sum(current, target_sum)

    # Ensure feasibility
    current = _repair_sum(current, target_sum)

    # ----- bookkeeping -----
    current_c5 = _compute_c5(current)
    best_h = current.copy()
    best_c5 = current_c5
    last_improve = start_time

    # Annealing / step‑size schedule (linear decay)
    max_step = 0.5
    min_step = 0.001
    max_temp = 0.01
    min_temp = 1e-6

    # ----- main optimisation loop -----
    while True:
        now = time.time()
        if now - start_time >= budget_s:
            break

        # time‑dependent parameters
        frac = (now - start_time) / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # compute worst shift and its gradient
        c5, shift = _compute_c5_and_shift(current)
        grad = _gradient_wrt_shift(current, shift)

        # candidate indices for improving the worst shift
        inc_idx = np.where((grad < -1e-12) & (current < 1.0 - 1e-12))[0]
        dec_idx = np.where((grad > 1e-12) & (current > 1e-12))[0]

        moved = False
        if inc_idx.size and dec_idx.size:

                            last_improve = now

                        current[j] = hj

            perturb = rng.normal(scale=step, size=n)

            current_c5 = _compute_c5(current)
            last_improve = now

    # Final repair and evaluation
    best_h = _repair_sum(best_h, n / 2.0)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
