def project_to_sum_box(v, target, tol=1e-8, max_iter=100):
    low = -np.max(v) - 1.0  # ensure low enough that all v+λ <=0
    high = 1.0 - np.min(v) + 1.0  # ensure high enough that all v+λ >=1
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s < target:
            low = mid
        else:
            high = mid
    # after loop, return w
    return np.clip(v + (low+high)/2.0, 0.0, 1.0)

def hill_climb_binary(init_h, target_sum, time_budget, start_time):
    best_h = init_h.copy()
    best_score = compute_c5(best_h)
    h = best_h.copy()
    cur_score = best_score
    n = h.size
    # track ones/zeros indices
    ones_idx = np.flatnonzero(h > 0.5).tolist()
    zeros_idx = np.flatnonzero(h < 0.5).tolist()
    # main loop
    attempts = 0
    max_no_improve = 5000
    while time.time() - start_time < time_budget:
        # propose swap
        i = random choice from ones_idx
        j = random choice from zeros_idx
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
            # update indices: swap
            ones_idx.remove(i)
            zeros_idx.append(i)
            zeros_idx.remove(j)
            ones_idx.append(j)
            attempts = 0
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            attempts += 1
            # occasionally accept worse? maybe we skip.

        if attempts >= max_no_improve:
            # restart random binary solution (maybe from scratch)

            ones_idx = np.flatnonzero(h > 0.5).tolist()

def hill_climb_binary(initial_h, target_sum, time_budget, start_time):
    n = len(initial_h)
    h = initial_h.copy()  # binary int array (0/1)
    # Compute autocorr positive l using np.correlate.
    autocorr = np.correlate(h, h, mode='full')
    autocorr = autocorr[n-1:]  # length n
    autocorr = autocorr.astype(np.int32)
    cur_min = autocorr.min()
    cur_score = (target_sum - cur_min) * (2.0 / n)  # S = target_sum
    best_h = h.copy()
    best_score = cur_score
    best_min = cur_min

    # sets/lists
    ones = list(np.flatnonzero(h).astype(int))
    zeros = list(np.flatnonzero(1 - h).astype(int))

    # parameters for simulated annealing
    T = 0.02  # temperature
    if cur_score > 0.5:
        T = 0.1
    # cooling
    cooling = 0.99995

    while time.time() - start_time < time_budget:
        # random indices
        i_idx = np.random.randint(len(ones))
        j_idx = np.random.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # compute delta
        delta = np.zeros(n, dtype=np.int32)
        # removal i contributions
        if i + 1 < n:
            delta[1 : n - i] -= h[i+1 : ]
        if i > 0:
            delta[1 : i+1] -= h[i-1::-1]
        # addition j contributions
        if j + 1 < n:
            delta[1 : n - j] += h[j+1 : ]
        if j > 0:
            delta[1 : j+1] += h[j-1::-1]
        # compute new min
        new_autocorr = autocorr + delta
        new_min = new_autocorr.min()
        new_score = (target_sum - new_min) * (2.0 / n)
        # acceptance
        accept = False
        if new_min > cur_min:
            accept = True
        else:
            # probability
            prob = np.exp((new_min - cur_min) / max(T, 1e-12))  # new_min - cur_min <=0, negative
            if np.random.rand() < prob:
                accept = True
        if accept:
            # accept move
            # update h, autocorr, cur_min, cur_score
            h[i] = 0
            h[j] = 1
            autocorr = new_autocorr
            cur_min = new_min
            cur_score = new_score
            # update lists
            # remove i from ones using swap-pop
            last_one = ones[-1]
            ones[i_idx] = last_one
            ones.pop()
            zeros.append(i)
            # remove j from zeros
            last_zero = zeros[-1]
            zeros[j_idx] = last_zero
            zeros.pop()
            ones.append(j)

            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                best_min = cur_min
        # else reject: no changes
        # cooling
        T *= cooling
        if T < 1e-8:
            T = 1e-8
        # early exit if best_score below target
        if best_score < 0.3808:
            break

    return best_h, best_score

def hill_climb_binary(h0, target_sum, time_budget, start_time):
    n = len(h0)
    # Internal helper to initialize from a binary vector
    def init_state(vec):
        h = vec.copy()
        autocorr = np.correlate(h, h, mode='full')[n-1:].astype(np.int32)
        cur_min = autocorr.min()
        cur_score = (target_sum - cur_min) * (2.0 / n)
        ones = list(np.flatnonzero(h).astype(int))
        zeros = list(np.flatnonzero(1 - h).astype(int))
        return h, autocorr, cur_min, cur_score, ones, zeros

    # start with given
    h, autocorr, cur_min, cur_score, ones, zeros = init_state(h0)
    best_h = h.copy()
    best_score = cur_score
    best_min = cur_min

    # parameters
    T = 0.02
    cooling = 0.99997
    no_improve_counter = 0
    max_no_improve = 20000

    while time.time() - start_time < time_budget:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = np.random.randint(len(ones))
        j_idx = np.random.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # compute delta as earlier
        delta = np.zeros(n, dtype=np.int16)
        if i + 1 < n:
            delta[1:n - i] -= h[i+1:].astype(np.int16)
        if i > 0:
            delta[1:i+1] -= h[i-1::-1].astype(np.int16)
        if j + 1 < n:
            delta[1:n - j] += h[j+1:].astype(np.int16)
        if j > 0:
            delta[1:j+1] += h[j-1::-1].astype(np.int16)
        new_autocorr = autocorr + delta
        new_min = new_autocorr.min()
        new_score = (target_sum - new_min) * (2.0 / n)
        accept = False
        if new_min > cur_min:
            accept = True
        else:
            # probabilistic acceptance for worsening moves
            prob = np.exp((new_min - cur_min) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            # update state
            h[i] = 0
            h[j] = 1
            autocorr = new_autocorr
            cur_min = new_min
            cur_score = new_score
            # update lists
            # remove i from ones
            ones[i_idx] = ones[-1]
            ones.pop()
            zeros.append(i)
            # remove j from zeros
            zeros[j_idx] = zeros[-1]
            zeros.pop()
            ones.append(j)
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                best_min = cur_min
                no_improve_counter = 0
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1
        T *= cooling
        if T < 1e-8:
            T = 1e-8
        # restart if stuck
        if no_improve_counter >= max_no_improve:
            # reinitialize random binary
            idx = np.random.choice(n, target_sum, replace=False)
            h_new = np.zeros(n, dtype=np.float64)
            h_new[idx] = 1.0
            h, autocorr, cur_min, cur_score, ones, zeros = init_state(h_new)
            no_improve_counter = 0
            # optionally increase T to initial
            T = 0.02
        # early exit
        if best_score < 0.3808:
            break
    return best_h, best_score

        init_vec = np.asarray(init_vec, dtype=np.float64).ravel()
        n = init_vec.size
    else:
        n = 380
        init_vec = np.random.rand(n)

    if n % 2 != 0:
        n -= 1
        init_vec = init_vec[:n]

    target_sum = n // 2

    # binary start: top half entries become 1
    idx = np.argpartition(-init_vec, target_sum - 1)[:target_sum]
    h0 = np.zeros(n, dtype=np.int8)
    h0[idx] = 1

    elapsed = time.time() - start
    remaining = budget_s - elapsed
    hc_time = remaining * 0.8
    best_h, best_score = _hill_climb_binary(h0, target_sum, hc_time, start)

    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        refined_h, refined_score = _refine_with_slsqp(best_h, target_sum, remaining, start)
        if refined_score < best_score:
            best_h, best_score = refined_h, refined_score

    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_h.size)
