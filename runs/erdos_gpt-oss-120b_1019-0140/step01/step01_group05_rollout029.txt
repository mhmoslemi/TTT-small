import numpy as np, time, math

def compute_c5_fft(h):
    N = h.shape[0]
    dx = 2.0 / N
    a = h
    b = 1.0 - h
    nfft = 1
    while nfft < 2*N:
        nfft <<= 1
    # compute convolution via FFT
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(A * B, n=nfft)[:2*N-1]
    return np.max(conv) * dx
```

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0:
        # set uniform
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    # scale to target
    v *= target / s
    # clip again
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        # small residual scaling
        v *= target / s2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def simulate_annealing_binary(h_start, time_limit, rng):
    h = h_start.copy()
    N = h.size
    target = N // 2
    # no need to repair each time as sum remain same after swaps
    # ensure it's binary floats
    # Precompute nfft and dx
    dx = 2.0 / N
    nfft = 1 << ((2*N - 1).bit_length())
    # initial score
    cur_score = compute_c5_fft(h, nfft, dx)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    max_iters = int(1e7) # large, but we break on time
    # set temperature schedule
    T0 = 0.1
    T_end = 1e-6
    for it in range(max_iters):
        if time.time() - start >= time_limit:
            break
        # sample i, j opposite
        i = rng.integers(N)
        hi = h[i]
        # find opposite j
        if hi > 0.5:
            # 1, need zero
            j = rng.integers(N)
            while h[j] > 0.5:
                j = rng.integers(N)
        else:
            # zero, need one
            j = rng.integers(N)
            while h[j] < 0.5:
                j = rng.integers(N)
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5_fft(h, nfft, dx)
        delta = new_score - cur_score
        if delta <= 0:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # acceptance
            frac = it / max_iters
            T = T0 * ((T_end / T0) ** frac)
            if rng.random() < math.exp(-delta / (T + 1e-15)):
                cur_score = new_score
            else:
                # revert
                h[i], h[j] = h[j], h[i]
    return best_h, best_score
```

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) == len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        v *= target / s2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def compute_c5_fft(h: np.ndarray) -> float:
    """Maximum overlap integral using FFT for speed."""
    N = h.shape[0]
    dx = 2.0 / N
    nfft = 1 << ((2 * N - 1).bit_length())
    a = h
    b = 1.0 - h
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    conv = np.fft.irfft(A * np.conj(B), nfft)[: 2 * N - 1]
    return float(np.max(conv) * dx)

def generate_legendre_sequence(p: int) -> np.ndarray:
    """Binary Legendre sequence of length p (quadratic residues -> 1)."""
    seq = np.zeros(p, dtype=np.float64)
    for i in range(1, p):
        if pow(i, (p - 1) // 2, p) == 1:
            seq[i] = 1.0
    return seq

def adjust_binary_to_target(v: np.ndarray, target: int, rng: np.random.Generator) -> np.ndarray:
    """Flip bits so that sum(v) == target (v is binary 0/1)."""
    cur = int(v.sum())
    if cur > target:
        ones = np.flatnonzero(v)
        to_zero = rng.choice(ones, size=cur - target, replace=False)
        v[to_zero] = 0.0
    elif cur < target:
        zeros = np.flatnonzero(1.0 - v)
        to_one = rng.choice(zeros, size=target - cur, replace=False)
        v[to_one] = 1.0
    return v

def simulated_annealing_binary(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Binary simulated annealing (swap 0/1) preserving exactly N/2 ones."""
    h = h_start.copy()
    N = h.size
    target = N // 2
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.1
    T_end = 1e-6
    max_iters = 2_000_000  # safety bound
    target_bound = 0.3808

    for it in range(max_iters):
        now = time.time()
        if now - start >= time_limit:
            break
        # select a random index and opposite type index
        i = rng.integers(N)
        hi = h[i]
        if hi > 0.5:
            j = rng.integers(N)
            while h[j] > 0.5:
                j = rng.integers(N)
        else:
            j = rng.integers(N)
            while h[j] < 0.5:
                j = rng.integers(N)
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        if delta <= 0:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target_bound:
                    return best_h, best_score
        else:
            # temperature based on elapsed fraction
            elapsed = now - start
            frac = elapsed / time_limit
            T = T0 * ((T_end / T0) ** frac)
            if rng.random() < math.exp(-delta / (T + 1e-15)):
                cur_score = new_score
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]
    return best_h, best_score
