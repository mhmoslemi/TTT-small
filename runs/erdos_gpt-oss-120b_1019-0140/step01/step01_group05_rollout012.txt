def simulated_annealing(start_h, time_limit, rng, init_temp=0.02, max_eps=0.5):
    h = start_h.copy()
    cur = c5_value(h)
    best_h = h.copy()
    best = cur
    start = time.time()
    while time.time() - start < time_limit:
        # compute temperature decaying to near 0
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = init_temp * frac
        # choose random move
        N = h.size
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        # optionally choose direction based on gradient for max shift? skipping
        # compute allowable epsilon: up, down as before
        hi, hj = h[i], h[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up + down == 0:
            continue
        # scale max_eps by frac maybe: limit = max_eps * frac?
        eps_range = up + down
        # random eps uniform
        eps = rng.random() * eps_range - down
        # optionally limit magnitude
        eps = max(-max_eps, min(max_eps, eps))
        # apply move
        h_new = h.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # ensure bounds (already)
        new_score = c5_value(h_new)
        delta = new_score - cur
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h = h_new
            cur = new_score
            if new_score < best:
                best = new_score
                best_h = h_new.copy()
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # get initial vector
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        # fallback: binary half ones, half zeros length 100
        n_fallback = 100
        h0 = np.concatenate([np.ones(n_fallback // 2), np.zeros(n_fallback - n_fallback // 2)], dtype=np.float64)
    # repair it
    h0 = _repair_vector(h0)
    n0 = h0.size
    # target resolution: if n0 < 200, upsample to n_start = 200 else use n0
    n_start = max(n0, 200)
    if n_start != n0:
        # interpolate
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_start, endpoint=False)
        h_start = np.interp(x_new, x_old, h0)
        h_start = _repair_vector(h_start)
    else:
        h_start = h0
    # Phase 1
    time_start = time.time()
    remaining_budget = budget_s
    phase1_time = remaining_budget * 0.6
    best_h, best_score = _simulated_annealing(h_start, phase1_time, rng, init_temp=0.03, max_eps=0.5)
    elapsed = time.time() - time_start
    remaining_budget = budget_s - elapsed
    # Phase 2: increase resolution if enough time
    if remaining_budget > budget_s * 0.2 and n_start < 800:
        n2 = min(n_start * 2, 800)
        # interpolate best_h
        x_old = np.linspace(0, 2, n_start, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _repair_vector(h_up)
        phase2_time = remaining_budget * 0.85
        best_h2, best_score2 = _simulated_annealing(h_up, phase2_time, rng, init_temp=0.02, max_eps=0.4)
        if best_score2 < best_score:
            best_h, best_score = best_h2, best_score2
        # Keep for possible final refinement
        remaining_budget = budget_s - (time.time() - time_start)
    # Phase 3: final small SA with small max_eps to fine-tune
    if remaining_budget > 0.1 * budget_s:
        phase3_time = remaining_budget
        best_h, best_score = _simulated_annealing(best_h, phase3_time, rng, init_temp=0.005, max_eps=0.1)
    # Final repair and recompute exact score
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _interpolate_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a vector defined on [0,2)."""
    old_n = old_h.size
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    return new_h

def _simulated_annealing(h_start: np.ndarray, time_limit: float,
                         rng: np.random.Generator,
                         init_temp: float = 0.02,
                         max_eps: float = 0.5,
                         final_temp: float = 1e-4) -> tuple[np.ndarray, float]:
    """Simple simulated annealing preserving sum and box constraints."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # linearly decreasing temperature
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = final_temp + (init_temp - final_temp) * frac
        # pick two distinct indices
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)   # amount we can move from j to i
        down = min(hi, 1.0 - hj) # amount we can move from i to j
        if up + down == 0.0:
            continue
        # propose epsilon uniformly in feasible interval
        eps = rng.random() * (up + down) - down
        # limit magnitude adaptively by current temperature
        eps = max(-max_eps, min(max_eps, eps))
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score
