    N = v.shape[0]
    target = N / 2.0
    np.clip(v, 0.0, 1.0, out=v)
    # iterate
    for _ in range(20):
        s = v.sum()
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        # scaling factor
        if s == 0:
            # distribute equally
            v[:] = target / N
        else:
            factor = target / s
            v *= factor
        np.clip(v, 0.0, 1.0, out=v)
    # final adjust tiny diff if needed
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # add diff to a random index within bounds
        idx = np.random.randint(N)
        # ensure we don't exceed bounds
        if diff > 0:
            v[idx] = min(1.0, v[idx] + diff)
        else:
            v[idx] = max(0.0, v[idx] + diff)
        # final clip
        np.clip(v, 0.0, 1.0, out=v)

def _simulated_annealing(h_start, time_limit, rng):
    h = h_start.copy()
    cur_score = _c5_value(h)
    best_h = h.copy()
    best_score = cur_score
    N = h.size
    start = time.time()
    # parameters
    T0 = cur_score * 0.02  # maybe a fraction of score
    T_end = 1e-6
    max_move0 = 0.5
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        # temperature schedule linear decay
        T = T0 * (1 - frac) + T_end * frac
        max_move = max_move0 * (1 - frac) + 1e-4 * frac
        # select i, j
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        # compute allowable epsilon
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up + down == 0:
            continue
        eps = rng.random() * (up + down) - down
        # limit magnitude
        if abs(eps) > max_move:
            eps = math.copysign(max_move, eps)
        # apply move
        h_new = h.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # no need to clip cause eps respects bounds
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
    return best_h, best_score
```

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """
    Clip values to [0,1] and adjust the sum to exactly N/2 (where N = len(v)).
    The adjustment preserves the box constraints.
    """
    N = v.size
    target = N / 2.0
    # clip to bounds
    np.clip(v, 0.0, 1.0, out=v)
    # iterative scaling+clipping
    for _ in range(20):
        s = v.sum()
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        if s == 0.0:
            v[:] = target / N
        else:
            factor = target / s
            v *= factor
        np.clip(v, 0.0, 1.0, out=v)
    # final tiny correction (deterministic)
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # add/subtract diff to a non‑saturated entry
        if diff > 0:
            idxs = np.where(v < 1.0)[0]
        else:
            idxs = np.where(v > 0.0)[0]
        if idxs.size > 0:
            i = idxs[0]
            v[i] = min(1.0, max(0.0, v[i] + diff))
        else:
            # fallback: distribute evenly (should not happen)
            v += diff / N
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """
    Compute C5 = max_k ∫ h(x) (1 - h(x+k)) dx for a discretised step function.
    """
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    best_h = h_cur.copy()

    N = h_cur.size
    start = time.time()
    # temperature parameters (linear schedule)
    T0 = max(0.001, cur_score * 0.02)   # a small fraction of the current score
    T_end = 1e-6
    # move magnitude schedule
    max_move0 = max_move_initial
    for _ in range(10_000_000):  # safety bound; will break by time
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1.0 - frac) + T_end * frac
        max_move = max_move0 * (1.0 - frac) + 1e-4 * frac
        # pick two distinct indices
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        # feasible epsilon range that respects box constraints
        up = min(1.0 - hi, hj)      # positive direction
        down = min(hi, 1.0 - hj)    # negative direction
        if up + down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        # limit magnitude
        if abs(eps) > max_move:
            eps = math.copysign(max_move, eps)
        if eps == 0.0:
            continue
        # propose new vector
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score

            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
