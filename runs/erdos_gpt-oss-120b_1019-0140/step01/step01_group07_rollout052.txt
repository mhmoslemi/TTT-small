def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # sort caps decreasing
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = caps[idx] if caps[idx] < need else need
            v[i] += add
            need -= add
            if need <= 1e-15:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = caps[idx] if caps[idx] < excess else excess
            v[i] -= sub
            excess -= sub
            if excess <= 1e-15:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-9 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

    remaining = budget_s - elapsed
    if remaining > 250:
        resolutions.append(min(int(n0 * 2), 2000))
    if remaining > 600:
        resolutions.append(min(int(n0 * 4), 3000))

    # ------------------------------------------------------------
    for n in resolutions:
        if time.time() - start >= budget_s * 0.95:
            break

        target_sum = n / 2.0

        # interpolate best known solution (or the provided one)
        src = best_h if best_h is not None else init_vec
        x_src = np.linspace(0.0, 2.0, src.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_start = np.interp(x_new, x_src, src)

        # blend with a triangle shape for diversity
        h_start = 0.5 * h_start + 0.5 * _triangle_shape(n)
        h_start = _project_to_constraints(h_start, target_sum)

        max_restarts = 120
        max_iter = 250

        for restart in range(max_restarts):
            if time.time() - start >= budget_s * 0.95:
                break

            pert = np.random.randn(n) * 0.02
            x0 = h_start + pert
            x0 = _project_to_constraints(x0, target_sum)

            cons = {"type": "eq", "fun": lambda h, tgt=target_sum: np.sum(h) - tgt}
            bounds = [(0.0, 1.0)] * n

            try:
                res = minimize(
                    _compute_c5,
                    x0,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=[cons],
                    options={"maxiter": max_iter, "ftol": 1e-10, "disp": False},
                )
            except Exception:
                continue

            h_opt = _project_to_constraints(res.x, target_sum)
            score = _compute_c5(h_opt)

            if score < best_score:
                best_score = float(score)
                best_h = h_opt.copy()
                best_n = n
                if best_score < 0.3808:
                    break

        if best_score < 0.3808:
            break

        # optional smoothing refinement at current resolution
        if best_h is not None and best_h.size == n:
            for _ in range(4):
                sm = _smooth_h(best_h, kernel_size=5)
                sc = _compute_c5(sm)
                if sc < best_score:
                    best_score = sc
                    best_h = sm.copy()
                else:
                    break

    # ------------------------------------------------------------
    if best_h is None:
        best_h = h_start
        best_score = _compute_c5(best_h)
        best_n = best_h.size

    # final evaluation (may use external verifier)
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_n)
