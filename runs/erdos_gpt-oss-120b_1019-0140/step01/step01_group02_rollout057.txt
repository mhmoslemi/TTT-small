import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to satisfy the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Choose a flexible index
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute evenly and clip again
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h


def _worst_shift_and_gradient(h: np.ndarray):
    """Return (cmax, lag, gradient) for the current worst shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
