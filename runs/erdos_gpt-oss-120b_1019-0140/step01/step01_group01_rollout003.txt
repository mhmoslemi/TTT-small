import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)*(1 - h(x+k)) dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust h in-place so that sum(h) == target while keeping 0 <= h_i <= 1.
    Small residuals are spread uniformly and clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_move(h: np.ndarray, target_sum: float, step: float,
                  rng: np.random.RandomState) -> np.ndarray:
    """Perform a deterministic gradient step for the worst overlap shift."""
    n = h.shape[0]
    # correlation without scaling (dx factor irrelevant for gradient direction)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_idx = int(np.argmax(corr))
    shift = k_idx - (n - 1)   # integer lag, can be negative

    g = np.zeros_like(h)

    if shift >= 0:
        length = n - shift
        idx = np.arange(length)
        g[idx] += (1.0 - h[idx + shift])
        g[idx + shift] -= h[idx]
    else:
        s = -shift
        length = n - s
        idx = np.arange(s, n)
        g[idx] += (1.0 - h[idx - s])
        g[idx - s] -= h[idx]

    # keep sum of h unchanged
    g -= g.mean()

    cand = h - step * g
    np.clip(cand, 0.0, 1.0, out=cand)
    cand = adjust_sum_to_target(cand, target_sum, rng)
    return cand

    """

    safe_margin = 0.5

    target_bound = 0.3808

    pop_size = 50

    pop = []
