def compute_c5_and_shift(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    return max_val, shift

def compute_gradient_worst(h, shift):

    grad = np.ones(n)
    if shift >= 0:

    n = initial_h.shape[0]
    target = n / 2.0
    h = initial_h.copy()
    current_h = h.copy()

    best_c5, cur_shift = compute_c5_and_shift(best_h)

    # algorithm parameters
    # We'll use iteration loop with while time left.
    # Schedule:
    # step_factor decays linearly from 0.5 to 0.01.
    # temperature decays from T0 = 0.02 to 1e-6.
    # p_grad = 0.6 (fraction of gradient moves).
    # We track no_improve_counter.
    # We'll also incorporate occasional random disturbance if stuck.

    T0 = 0.02
    step_start = 0.5
    step_end = 0.01
    p_grad = 0.6
    no_improve_iter = 0
    max_no_improve = int(5000) # we can adjust.

    iter_count = 0
    while True:
        now = time.time()
        elapsed = now - start_time
        if now - start_time >= max_time:
            break
        progress = elapsed / max_time
        step_factor = step_start * (1 - progress) + step_end * progress
        T = T0 * (1 - progress) + 1e-6 * progress

        # Choose move type
        if rng.rand() < p_grad:
            # gradient move
            # compute worst shift
            cur_c5, cur_k = compute_c5_and_shift(current_h)
            grad = compute_gradient_worst(current_h, cur_k)

            i_inc = np.argmin(grad)  # most negative

                candidate[i_inc] += delta

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Scale or shift h so that sum(h) == target while staying in [0,1]."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Try to fix the whole deficit/surplus using a few random indices.
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # Distribute the remainder uniformly (may slightly violate bounds)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_c5_and_shift(h: np.ndarray):
    """Return (C5 value, integer lag achieving it) for given h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (n - 1)


def compute_gradient_worst(h: np.ndarray, shift: int):
    """Gradient of the worst overlap term w.r.t. h."""
    n = h.shape[0]
    grad = np.ones(n)
    if shift >= 0:
        k = shift
        if k < n:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
    else:
        k = -shift
        if k < n:
            grad[k:] -= h[: n - k]
            grad[: n - k] -= h[k:]
    return grad


def run_optim(initial_h: np.ndarray, rng: np.random.RandomState, end_time: float):
    """Projected‑gradient / annealing optimizer for a fixed n."""
    start_local = time.time()
    n = initial_h.shape[0]
    target = n / 2.0

    current_h = initial_h.copy()
    best_h = current_h.copy()
    best_c5, _ = compute_c5_and_shift(best_h)
    current_c5 = best_c5

    # hyper‑parameters
    T0 = 0.02                     # initial temperature
    step_start = 0.5              # initial step fraction
    step_end = 0.01               # final step fraction
    p_grad = 0.6                  # probability of gradient‑guided move
    max_no_improve = 2000        # iterations before a perturbation
    no_improve = 0

    while time.time() < end_time:
        elapsed = time.time() - start_local
        total_allowed = end_time - start_local
        if total_allowed <= 0:
            break
        progress = elapsed / total_allowed
        step_factor = step_start * (1.0 - progress) + step_end * progress
        T = T0 * (1.0 - progress) + 1e-6 * progress

        # decide move type
        candidate = None
        if rng.rand() < p_grad:
            # gradient‑guided exchange
            cur_c5, cur_shift = compute_c5_and_shift(current_h)
            grad = compute_gradient_worst(current_h, cur_shift)
            i_dec = int(np.argmax(grad))   # decreasing this reduces the worst term
            i_inc = int(np.argmin(grad))   # increasing this reduces the worst term
            if i_dec != i_inc:
                allowed = min(current_h[i_dec], 1.0 - current_h[i_inc])
                if allowed > 0.0:
                    delta = allowed * step_factor * rng.rand()
                    cand = current_h.copy()
                    cand[i_dec] -= delta
                    cand[i_inc] += delta
                    candidate = cand
        else:
            # random pairwise exchange
            i, j = rng.randint(n, size=2)
            if i != j:
                hi, hj = current_h[i], current_h[j]
                low = max(-hi, hj - 1.0)
                high = min(1.0 - hi, hj)
                if low < high:
                    delta = (rng.rand() * 2.0 - 1.0) * step_factor
                    delta = np.clip(delta, low, high)
                    cand = current_h.copy()
                    cand[i] = hi + delta
                    cand[j] = hj - delta
                    candidate = cand

        if candidate is None:
            continue

        cand_c5, _ = compute_c5_and_shift(candidate)

        # Metropolis acceptance
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0.0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional perturbation to escape stagnation
        if no_improve >= max_no_improve:
            pert = best_h.copy()
            mask = rng.rand(n) < 0.1
            if np.any(mask):
                noise = (rng.rand(mask.sum()) - 0.5) * step_factor * 0.5
                pert[mask] += noise
                np.clip(pert, 0.0, 1.0, out=pert)
                pert = adjust_sum_to_target(pert, target, rng)
                current_h = pert
                current_c5, _ = compute_c5_and_shift(current_h)
                no_improve = 0

    # final projection and evaluation
    best_h = adjust_sum_to_target(best_h, target, rng)
    final_c5, _ = compute_c5_and_shift(best_h)
    return best_h, final_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_end = start + max(0.0, budget_s - 0.5)   # tiny safety margin

    # ---------- initialise ----------
    init_h = None
    if 'initial_h_values' in globals():
        try:
            init_h = np.array(initial_h_values, dtype=float)
        except Exception:
            init_h = None

    if init_h is not None:
        n0 = init_h.size
        target0 = n0 / 2.0
        h0 = adjust_sum_to_target(init_h.copy(), target0, rng)
    else:
        n0 = 200
        target0 = n0 / 2.0
        h0 = rng.rand(n0)
        h0 = h0 / h0.sum() * target0
        h0 = adjust_sum_to_target(h0, target0, rng)

    # primary optimisation
    best_h, best_c5 = run_optim(h0, rng, max_end)

    # optional higher‑resolution refinement if time permits
    remaining = max_end - time.time()
    if remaining > 30.0:
        n2 = best_h.size * 2
        if n2 <= 500:  # keep size reasonable
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
            h2 = np.interp(x_new, x_old, best_h)
            np.clip(h2, 0.0, 1.0, out=h2)
            h2 = adjust_sum_to_target(h2, n2 / 2.0, rng)
            end2 = time.time() + max(0.0, remaining - 0.5)
            h2_opt, c5_2 = run_optim(h2, rng, end2)
            if c5_2 < best_c5:
                best_h, best_c5 = h2_opt, c5_2

    # final sanity check
    best_h = adjust_sum_to_target(best_h, best_h.size / 2.0, rng)
    final_c5, _ = compute_c5_and_shift(best_h)

    return (best_h, float(final_c5), best_h.size)
