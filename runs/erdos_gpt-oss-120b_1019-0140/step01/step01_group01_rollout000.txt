def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Scale/clip h in-place so that sum(h) == target while staying in [0,1]."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts, max_attempts = 0, 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])

                diff -= inc

    return h

        cur_c5 = compute_c5(child)

            step_sz = base_step * (1.0 - progress()) + 1e-5 * progress()
            cand = gradient_step(child, step_sz)
            cand = adjust_sum_to_target(cand, target_sum, rng)
            cand_c5 = compute_c5(cand)
            if cand_c5 < cur_c5:
                child, cur_c5 = cand, cand_c5

        child_c5 = cur_c5

        # insert if better than worst in population
        if child_c5 < population[-1][1]:
            population[-1] = (child, child_c5)
            population.sort(key=lambda x: x[1])
            if child_c5 < best_c5:
                best_h = child.copy()
                best_c5 = child_c5

    # ----- final polishing ---------------------------------------------------
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
