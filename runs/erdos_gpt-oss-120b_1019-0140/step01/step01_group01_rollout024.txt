import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # Use np.correlate
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * corr.max())

def adjust_sum_to_target(h, target, rng):
    total = h.sum()
    diff = target - total
    # If diff zero within tolerance, just return.
    if abs(diff) < 1e-12:
        return h
    # Randomly adjust entries
    attempts = 0
    max_attempts = 5000  # increase attempts
    n = h.size
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(n)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # distribute small error uniformly and clip
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        # recompute diff and correct iteratively maybe once more
        total = h.sum()
        diff = target - total
        if abs(diff) > 1e-12:
            # final correction by adding diff to random index within bounds
            idx = rng.randint(n)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def gradient_worst_lag(h):
    n = h.shape[0]
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    k_idx = int(np.argmax(corr_full))
    k_shift = k_idx - (n - 1)
    g = np.zeros_like(h)
    # compute contributions
    if k_shift >= 0:
        i1 = np.arange(n - k_shift)
        g[i1] += (1.0 - h[i1 + k_shift])
        i2 = np.arange(k_shift, n)
        g[i2] -= h[i2 - k_shift]
    else:
        shift = -k_shift
        i1 = np.arange(shift, n)
        g[i1] += (1.0 - h[i1 - shift])
        i2 = np.arange(0, n - shift)
        g[i2] -= h[i2 + shift]
    # project to zero sum
    g -= g.mean()
    return g, k_shift

def pairwise_move(h, step, rng):
    n = h.shape[0]
    i, j = rng.randint(n, size=2)
    if i == j:
        return None  # indicating no move
    hi, hj = h[i], h[j]
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if low >= high:
        return None
    delta = (rng.rand() * 2.0 - 1.0) * step
    # clamp
    if delta < low:
        delta = low
    elif delta > high:
        delta = high
    candidate = h.copy()
    candidate[i] = hi + delta
    candidate[j] = hj - delta
    return candidate

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Adjust h (in-place) so that sum(h) == target while staying in [0,1]."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    n = h.size
    attempts = 0
    max_attempts = 5000
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(n)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        # final tiny correction
        total = h.sum()
        diff = target - total
        if abs(diff) > 1e-12:
            idx = rng.randint(n)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def gradient_worst_lag(h: np.ndarray):
    """Gradient of the worst overlap lag (subgradient). Returns (g, k_shift)."""
    n = h.shape[0]
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    k_idx = int(np.argmax(corr_full))
    k_shift = k_idx - (n - 1)
    g = np.zeros_like(h)
    if k_shift >= 0:
        i1 = np.arange(n - k_shift)
        g[i1] += (1.0 - h[i1 + k_shift])
        i2 = np.arange(k_shift, n)
        g[i2] -= h[i2 - k_shift]
    else:
        shift = -k_shift
        i1 = np.arange(shift, n)
        g[i1] += (1.0 - h[i1 - shift])
        i2 = np.arange(0, n - shift)
        g[i2] -= h[i2 + shift]
    g -= g.mean()
    return g, k_shift

def pairwise_move(h: np.ndarray, step: float, rng: np.random.RandomState):
    """Return a new candidate by exchanging mass between two coordinates."""
    n = h.shape[0]
    i, j = rng.randint(n, size=2)
    if i == j:
        return None
    hi, hj = h[i], h[j]
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if low >= high:
        return None
    delta = (rng.rand() * 2.0 - 1.0) * step
    if delta < low:
        delta = low
    elif delta > high:
        delta = high
    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    return cand

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # --- initialise ---
    init_arr = None
    if "initial_h_values" in globals():
        try:
            tmp = np.array(initial_h_values, dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                init_arr = tmp
        except Exception:
            init_arr = None

    default_n = 200
    if init_arr is not None:
        n_points = max(default_n, init_arr.size)
    else:
        n_points = default_n
    target_sum = n_points / 2.0

    if init_arr is not None:
        if init_arr.size == n_points:
            h = init_arr.copy()
        else:
            # linear interpolation to desired size
            old_x = np.linspace(0.0, 2.0, init_arr.size, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h = np.interp(new_x, old_x, init_arr)
        np.clip(h, 0.0, 1.0, out=h)
        h = adjust_sum_to_target(h, target_sum, rng)
    else:
        h = rng.rand(n_points)
        h = h / h.sum() * target_sum
        np.clip(h, 0.0, 1.0, out=h)
        h = adjust_sum_to_target(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # annealing parameters
    T0, T_final = 1e-3, 1e-6
    step0, step_final = 0.2, 1e-4
    prob_gradient = 0.6
    no_improve = 0

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        elapsed = now - start
        progress = elapsed / max_time

        T = T0 * (1.0 - progress) + T_final * progress
        step = step0 * (1.0 - progress) + step_final * progress

        if rng.rand() < prob_gradient:
            g, _ = gradient_worst_lag(current_h)
            candidate = current_h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        else:
            cand = pairwise_move(current_h, step, rng)
            if cand is None:
                continue
            candidate = cand

        cand_c5 = compute_c5(candidate)

        # acceptance
        if cand_c5 < current_c5:
            accept = True
        else:
            accept = rng.rand() < np.exp(-(cand_c5 - current_c5) / T) if T > 0 else False

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5 - 1e-12:
                best_c5 = cand_c5
                best_h = candidate.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional perturbation if stuck
        if no_improve > 2000 and rng.rand() < 0.1:
            random_vec = rng.rand(n_points)
            random_vec = random_vec / random_vec.sum() * target_sum
            mix = rng.rand()
            candidate = mix * current_h + (1.0 - mix) * random_vec
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
            current_h = candidate
            current_c5 = compute_c5(candidate)
            no_improve = 0

    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
