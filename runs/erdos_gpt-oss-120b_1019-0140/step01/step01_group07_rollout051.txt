def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if cur == target:
        return v
    # if cur < target, add to entries not at 1.
    # use water-filling:
    # compute deficit
    if cur < target:
        deficit = target - cur
        free = np.where(v < 1.0)[0]
        if free.size > 0:
            caps = 1.0 - v[free]
            # allocate proportionally? We can just fill equally until capacity.
            # Use sorted caps descending to allocate to largest capacity first.
            order = np.argsort(caps)[::-1]  # largest capacity first
            for idx in order:
                i = free[idx]
                add = min(deficit, caps[idx])
                v[i] += add
                deficit -= add
                if deficit <= 1e-12:
                    break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        if free.size > 0:
            caps = v[free]
            order = np.argsort(caps)[::-1]  # largest value first
            for idx in order:
                i = free[idx]
                sub = min(excess, caps[idx])
                v[i] -= sub
                excess -= sub
                if excess <= 1e-12:
                    break
    # final clip
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    # if still off due to rounding, rescale
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):

    # With dx factor.
    return np.max(corr) * dx

def _compute_c5(h):
    """Maximum overlap integral for step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust sum to target via water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        deficit = target - cur
        free = np.where(v < 1.0)[0]
        if free.size:
            caps = 1.0 - v[free]
            order = np.argsort(caps)[::-1]  # largest capacity first
            for idx in order:
                i = free[idx]
                add = min(deficit, caps[idx])
                v[i] += add
                deficit -= add
                if deficit <= 1e-12:
                    break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        if free.size:
            caps = v[free]
            order = np.argsort(caps)[::-1]  # largest value first
            for idx in order:
                i = free[idx]
                sub = min(excess, caps[idx])
                v[i] -= sub
                excess -= sub
                if excess <= 1e-12:
                    break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-9 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

            best_h = pop[cur_best].copy()
            if best_score < target_bound:
                break
        gen += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    init_h = globals().get("initial_h_values", None)
    if init_h is not None:
        init_h = np.array(init_h, dtype=np.float64).ravel()
        n0 = init_h.size
    else:
        n0 = 380
        init_h = np.random.rand(n0)
    target_sum0 = n0 / 2.0
    init_h = _project_to_constraints(init_h, target_sum0)

    target_bound = 0.3808

    # Stage 1: GA at base resolution
    best_h, best_score = _ga_optimize(
        init_h, target_sum0, start, budget_s, target_bound, max_gen=300
    )

    # Stage 2: local SLSQP refinement if time permits
    if budget_s - (time.time() - start) > 30:
        cons = {"type": "eq", "fun": lambda h, tgt=target_sum0: np.sum(h) - tgt}
        bounds = [(0.0, 1.0)] * n0
        try:
            res = minimize(_compute_c5, best_h, method="SLSQP",
                           bounds=bounds, constraints=[cons],
                           options={"maxiter": 200, "ftol": 1e-9, "disp": False})
            if res.success:
                h_ref = _project_to_constraints(res.x, target_sum0)
                sc = _compute_c5(h_ref)
                if sc < best_score:
                    best_h, best_score = h_ref, sc
        except Exception:
            pass

    # Stage 3: optional higher resolution if time remains
    elapsed = time.time() - start
    if (budget_s - elapsed) > 200 and n0 <= 760:
        n1 = n0 * 2
        h_up = _upsample(best_h, n1)
        target_sum1 = n1 / 2.0
        h_up = _project_to_constraints(h_up, target_sum1)

        best_h2, best_score2 = _ga_optimize(
            h_up, target_sum1, start, budget_s, target_bound, max_gen=150
        )

        if budget_s - (time.time() - start) > 30:
            cons2 = {"type": "eq", "fun": lambda h, tgt=target_sum1: np.sum(h) - tgt}
            bounds2 = [(0.0, 1.0)] * n1
            try:
                res2 = minimize(_compute_c5, best_h2, method="SLSQP",
                                bounds=bounds2, constraints=[cons2],
                                options={"maxiter": 200, "ftol": 1e-9, "disp": False})
                if res2.success:
                    h_ref2 = _project_to_constraints(res2.x, target_sum1)
                    sc2 = _compute_c5(h_ref2)
                    if sc2 < best_score2:
                        best_h2, best_score2 = h_ref2, sc2
            except Exception:
                pass
        if best_score2 < best_score:
            best_h, best_score = best_h2, best_score2

    # Final evaluation via harness helper if present
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, best_h.size
