def _repair_vector(v, n, max_iter=50):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # Determine mask of adjustable entries: if diff>0 need to reduce sum, else increase.
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        # scale uniformly
        factor = target / s if s > 0 else 0.0
        v *= factor
        v = np.clip(v, 0.0, 1.0)
        # If still off due to clipping, distribute diff uniformly
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v
```

import time
import numpy as np

def _repair_vector(v, n, max_iter=50):
    """Project v onto [0,1]^n with sum(v) = n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v *= factor
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Maximum overlap integral for discrete vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation to new length and repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v)
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Uniform random vector repaired to feasibility."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _local_search(v, n, time_limit):
    """Simple exchange hill‑climbing within time limit."""
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score, best_v = score, cand
    return best_v, best_score

def _sa_optimize(v0, time_limit):
    """Simulated annealing on vector v0 for given time."""
    start = time.time()
    n = len(v0)
    cur_v = v0.copy()
    cur_score = _compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    step = 0.05
    T0 = 0.02
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (0.5 ** (frac * 5.0))  # fast cooling

        # worst shift (max overlap)
        corr_full = np.correlate(cur_v, 1.0 - cur_v, mode='full')
        idx_max = np.argmax(corr_full)
        s = idx_max - (n - 1)

        # propose move
        if s != 0:
            if s > 0:
                i_start, i_end = 0, n - s
            else:
                i_start, i_end = -s, n
            mask = (cur_v[i_start:i_end] < 1.0) & (cur_v[i_start + s:i_end + s] < 1.0)
            feas = np.where(mask)[0]
            if feas.size > 0:
                i_local = np.random.choice(feas)
                i = i_start + i_local
                # choose a donor index j
                dec_cands = np.where(cur_v > 1e-12)[0]
                dec_cands = dec_cands[~np.isin(dec_cands, [i, i + s])]
                if dec_cands.size > 0:
                    j = np.random.choice(dec_cands)
                    max_inc_i = 1.0 - cur_v[i]
                    max_inc_is = 1.0 - cur_v[i + s]
                    max_dec_j = cur_v[j]
                    max_delta = min(max_inc_i, max_inc_is, max_dec_j / 2.0, step)
                    if max_delta > 0:
                        delta = max_delta * np.random.rand()
                        new_v = cur_v.copy()
                        new_v[i] += delta
                        new_v[i + s] += delta
                        new_v[j] -= 2.0 * delta
                    else:
                        continue
                else:
