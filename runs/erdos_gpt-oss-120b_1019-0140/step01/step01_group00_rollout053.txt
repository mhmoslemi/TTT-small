    lo = 0.0

    # find lambda such that sum(clip(x + lam, lo, hi)) == target_sum

        if s < target_sum:
            lam_low = lam
        else:

import time, math, numpy as np

def compute_c5(h):
    """Maximum overlap integral C5 for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_h(x, target):
    """
    Project vector x onto the box [0,1]^n while enforcing sum(x) == target.
    Finds λ such that sum(clip(x+λ,0,1)) = target via binary search.
    """
    lo, hi = 0.0, 1.0
    lam_low = lo - np.max(x)          # makes all entries ≤ lo
    lam_high = hi - np.min(x)         # makes all entries ≥ hi
    for _ in range(60):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(x + lam, lo, hi)
        s = proj.sum()
        if s < target:
            lam_low = lam
        else:
            lam_high = lam
        if abs(s - target) < 1e-12:
            break
    return proj

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise h -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 80
        h = rng.random(n)
    target_sum = n / 2.0
    h = project_h(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ---- optimisation parameters --------------------------------------
    p_grad = 0.7            # probability to take a gradient step
    init_alpha = 0.3        # initial step size for gradient descent
    T_max = 0.012           # initial temperature for stochastic moves
    T_min = 1e-5            # final temperature

    # ---- main optimisation loop ---------------------------------------
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > budget_s - 0.5:
            break
        # temperature schedule (linear)
        T = T_max * (1.0 - elapsed / budget_s) + T_min * (elapsed / budget_s)

        if rng.random() < p_grad:
            # ---- gradient step on the worst shift(s) --------------------
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            max_val = corr.max()
            max_idxs = np.flatnonzero(corr == max_val)
            ks = max_idxs - (n - 1)        # convert index to shift
            g = np.zeros(n, dtype=float)
            for k in ks:
                if k >= 0:
                    g[:n - k] += 1.0 - cur_h[k:]
                    g[k:]    -= cur_h[:n - k]
                else:
                    kk = -k
                    g[kk:]    += 1.0 - cur_h[:n - kk]
                    g[:n - kk] -= cur_h[kk:]
            g = g / len(ks)                # average over all maximal shifts
            alpha = init_alpha * (1.0 - elapsed / budget_s)
            if alpha < 1e-5:
                alpha = 1e-5
            # gradient descent step with projection
            cur_h = project_h(cur_h - alpha * g, target_sum)
            cur_c5 = compute_c5(cur_h)
        else:
            # ---- random two‑coordinate perturbation --------------------
            i, j = rng.choice(n, size=2, replace=False)
            # Compute allowable delta preserving bounds and sum
            delta_min = max(-0.2, -cur_h[i], cur_h[j] - 1.0)
            delta_max = min(0.2, 1.0 - cur_h[i], cur_h[j])
            if delta_max > delta_min:
                delta = rng.uniform(delta_min, delta_max)
                cand = cur_h.copy()
                cand[i] += delta
                cand[j] -= delta
                # (sum unchanged by construction, bounds satisfied)
                cand_c5 = compute_c5(cand)
                delta_c5 = cand_c5 - cur_c5
                if delta_c5 <= 0.0 or rng.random() < math.exp(-delta_c5 / T):
                    cur_h = cand
                    cur_c5 = cand_c5

        # ---- record best ------------------------------------------------
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()
            if best_c5 < 0.3808:
                break

    # final projection to guarantee feasibility
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h.astype(float), best_c5, n
