def project_to_feasible(h, target_sum):
    # Clip then adjust mean exactly as before but can be simpler.
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # If diff is zero, return.
    # Distribute diff equally among not-at-bound entries.
    # similar to previous.
    # Use small tolerance.
    for _ in range(10):
        free = (h > 0) & (h < 1)
        if not free.any():
            break
        delta = diff / free.sum()
        h[free] += delta
        # Clip again.
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-9:
            break
    # final correction
    cur = h.sum()
    if abs(cur - target_sum) > 1e-6:
        # adjust single element
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        # adjust within bounds
        change = np.clip(target_sum - cur, -h[idx], 1 - h[idx])
        h[idx] += change
    return h

def compute_c5(h):
    n = h.shape[0]
    g = 1.0 - h
    # reverse g for correlation:
    g_rev = g[::-1]
    # compute convolution via FFT:
    N = 1 << ((2*n - 1).bit_length())  # next power of 2
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(g_rev, n=N)
    conv = np.fft.irfft(H * G, n=N)[:2*n-1]
    return conv.max() * (2.0 / n)

import time
import numpy as np

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx for sampled h."""
    n = h.shape[0]
    g_rev = (1.0 - h)[::-1]
    # Linear convolution via FFT (next power of 2 length)
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(g_rev, n=N)
    conv = np.fft.irfft(H * G, n=N)[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-9, max_iter=15):
    """Clip to [0,1] and adjust to satisfy sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = (h > 0.0) & (h < 1.0)
        if not free.any():
            break
        delta = diff / free.sum()
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-6:
        # adjust the element with most slack
        if cur < target_sum:
            idx = np.argmin(h)  # most room to increase
            inc = min(target_sum - cur, 1.0 - h[idx])
            h[idx] += inc
        else:
            idx = np.argmax(h)  # most room to decrease
            dec = min(cur - target_sum, h[idx])
            h[idx] -= dec
    return h


def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def random_binary_h(n, rng):
    """Generate a feasible binary (or half‑binary) vector with sum = n/2."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        # one extra entry with value 0.5
        remaining = np.setdiff1d(np.arange(n), idx)
        extra_idx = rng.choice(remaining, size=1)
        h[extra_idx] = 0.5
    h = project_to_feasible(h, n / 2.0)
    return h


def anneal_optimize(h0, n, rng, time_budget):
    """
    Simple simulated‑annealing optimizer.
    Maintains the exact sum constraint by swapping a delta between two positions.
    """
    start = time.time()
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    # Temperature schedule
    T0 = 1e-2
    T_end = 1e-6
    max_iter = 300000

    for it in range(max_iter):
        if time.time() - start > time_budget:
            break
        # Linear cooling
        T = T0 + (T_end - T0) * (it / max_iter)

        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        # feasible delta range so that both stay in [0,1]
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue
        delta = rng.random() * (delta_max - delta_min) + delta_min

        # propose move
        h_new = h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta

        new_val = compute_c5(h_new)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / max(T, 1e-12)):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()

    return best_h, best_val
