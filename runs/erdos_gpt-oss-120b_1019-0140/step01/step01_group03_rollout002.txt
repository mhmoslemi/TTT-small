def optimize_h(h0, max_iter, start_time, budget_s):
    n = h0.size
    s = n/2.0
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    h = h0.copy()
    for it in range(1, max_iter+1):
        # maybe every 100 steps check time
        if it % 200 == 0:
            if time.time() - start_time > budget_s - 0.5:
                break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12 + 1e-9 * abs(maxc)
        idxs = np.where(corr >= maxc - eps)[0]
        shifts = idxs - (n - 1)
        # compute gradient
        grad = np.full_like(h, len(shifts), dtype=float)
        for sh in shifts:
            if sh >= 0:
                grad[:n - sh] -= h[sh:]
                grad[sh:] -= h[:n - sh]
            else:
                sft = -sh
                grad[sft:] -= h[:n - sft]
                grad[:n - sft] -= h[sft:]
        grad /= len(shifts)
        eta = 0.5 / math.sqrt(it)
        h = project_onto_box_sum(h - eta * grad, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

import time, math

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    # clip extreme
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    s = n / 2.0

    best_h = h.copy()

        if it % 200 == 0 and time.time() - start_time > budget_s - 0.5:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
