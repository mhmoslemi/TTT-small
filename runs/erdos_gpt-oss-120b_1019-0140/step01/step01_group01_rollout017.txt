import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Rescale h in-place so that sum(h) == target while keeping 0<=h_i<=1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # greedy redistribution
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # small uniform correction, then clip and fix any residue again
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # final single‑index fix if needed
        residual = target - h.sum()
        if abs(residual) > 1e-12:
            idx = np.argmin(np.abs(h - (0.0 if residual < 0 else 1.0)))
            adjust = np.clip(residual, -h[idx], 1.0 - h[idx])
            h[idx] += adjust
    return h

def gradient_move(h: np.ndarray, target_sum: float, step: float, rng: np.random.RandomState) -> np.ndarray:
    """Move in direction of gradient of the currently worst overlap."""
    n = h.shape[0]
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    k_idx = int(np.argmax(corr_full))
    k_shift = k_idx - (n - 1)

    g = np.zeros_like(h)
    idx = np.arange(n)

    plus = idx + k_shift
    valid_plus = (plus >= 0) & (plus < n)
    g[valid_plus] += (1.0 - h[plus[valid_plus]])

    minus = idx - k_shift
    valid_minus = (minus >= 0) & (minus < n)
    g[valid_minus] -= h[minus[valid_minus]]

    # project to hyperplane sum(g)=0
    g -= g.mean()

    cand = h - step * g
    np.clip(cand, 0.0, 1.0, out=cand)
    cand = adjust_sum_to_target(cand, target_sum, rng)
    return cand

def perturb_move(h: np.ndarray, target_sum: float, step: float, rng: np.random.RandomState):
    """Random two‑index exchange preserving the total mass."""
    n = h.shape[0]
    i, j = rng.randint(n, size=2)
    if i == j:
        return None
    hi, hj = h[i], h[j]
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if low >= high:
        return None
    delta = (rng.rand() * 2.0 - 1.0) * step
    delta = np.clip(delta, low, high)
    if abs(delta) < 1e-12:
        return None
    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    return cand

        n_points = init_h.size
        target_sum = n_points / 2.0
        h = adjust_sum_to_target(init_h.copy(), target_sum, rng)
    except Exception:
        n_points = 200
        target_sum = n_points / 2.0
        h = rng.rand(n_points)
        h = h / h.sum() * target_sum
        h = adjust_sum_to_target(h, target_sum, rng)

    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # annealing schedule parameters
    T_start, T_end = 1e-2, 1e-7
    step_start, step_end = 0.3, 1e-4
    last_improve = start
    restart_interval = 45.0   # seconds without improvement before a gentle restart

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        T = T_start * (1 - progress) + T_end * progress
        step = step_start * (1 - progress) + step_end * progress

        # decide move type
        if rng.rand() < 0.4:
            candidate = gradient_move(current_h, target_sum, step, rng)
        else:
            cand = perturb_move(current_h, target_sum, step, rng)
            if cand is None:
                continue
            candidate = cand

        cand_c5 = compute_c5(candidate)

        # Metropolis acceptance
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / max(T, 1e-12))
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                last_improve = now

        # gentle restart if stuck
        if now - last_improve > restart_interval:
            # add a modest random perturbation to the best known state
            perturb = rng.normal(scale=step * 5.0, size=n_points)
            cand = best_h + perturb
            np.clip(cand, 0.0, 1.0, out=cand)
            cand = adjust_sum_to_target(cand, target_sum, rng)
            current_h = cand
            current_c5 = compute_c5(current_h)
            last_improve = now

    # final correction of mass
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
