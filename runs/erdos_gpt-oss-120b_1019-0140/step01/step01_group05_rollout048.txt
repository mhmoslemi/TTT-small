    h = h_start.copy()
    best = h.copy()
    best_score = c5_value(h)
    cur_score = best_score
    N = len(h)
    dx = 2.0 / N
    start = time.time()
    # set schedule parameters
    T0 = 0.1
    # maybe use exponential cooling: T = T0 * (0.95**iteration)
    iteration = 0
    while time.time() - start < time_limit:
        # adapt temperature
        T = T0 * (0.995 ** iteration)
        # pick two indices i,j
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi, hj = h[i], h[j]
        # compute max epsilon positive that can be added to hi & sub from hj:
        # hi can increase up to 1 - hi, hj can decrease down to -hj (i.e., new hj >=0)
        max_up = min(1.0 - hi, hj)  # we can transfer from j to i up to this
        max_down = min(hi, 1.0 - hj)  # we can transfer from i to j
        # choose direction randomly: pick epsilon uniformly within [-max_down, max_up]
        eps = rng.random() * (max_up + max_down) - max_down
        # perhaps allow bigger steps: ups to max_eps?
        max_eps = 0.2 # could tune based on T maybe
        if eps > max_eps: eps = max_eps
        elif eps < -max_eps: eps = -max_eps
        if eps == 0: continue
        # apply move
        h[i] = hi + eps
        h[j] = hj - eps
        # evaluate new score
        new_score = c5_value(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = h.copy()
        else:
            # revert move
            h[i] = hi
            h[j] = hj
        iteration += 1
    return best, best_score

def repair_vector(v):
    np.clip(v, 0, 1, out=v)
    N = v.size
    target = N/2
    s = v.sum()
    if s == 0:
        # fill uniform
        v[:] = target / N
        np.clip(v, 0, 1, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0, 1, out=v)
    # after clipping, might deviate, adjust again
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0, 1, out=v)
    # final check
    # maybe add small correction to first element to compensate rounding
    diff = target - v.sum()

import numpy as np
import time
import math

def repair_vector(v):
    # Ensure v is a copy? Usually we modify in place; but return v.
    np.clip(v, 0.0, 1.0, out=v)
    N = v.size
    target = N / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / N
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    # final adjustment if needed
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        # distribute diff to non-bound entries if any
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size > 0:
            # split diff equally among free indices
            alloc = diff / free.size
            v[free] += alloc
            np.clip(v, 0.0, 1.0, out=v)
        else:
            # fallback: uniform adjustment
            v[:] += diff / N
            np.clip(v, 0.0, 1.0, out=v)
    return v
```

def c5_value(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def simulated_annealing(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.1
    max_move = 0.2
    # Precompute target sum for possible repair (though sum preserved)
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        # Linear cooling schedule
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)
        # select two distinct indices
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        # compute epsilon bounds
        max_up = min(1.0 - hi, hj)   # move from j to i
        max_down = min(hi, 1.0 - hj) # move from i to j
        if max_up + max_down == 0.0:
            continue
        eps = rng.random() * (max_up + max_down) - max_down
        # clip magnitude
        if eps > max_move:
            eps = max_move
        elif eps < -max_move:
            eps = -max_move
        if eps == 0.0:
            continue
        # apply move
        h_cur[i] = hi + eps
        h_cur[j] = hj - eps
        new_score = c5_value(h_cur)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            # revert move
            h_cur[i] = hi
            h_cur[j] = hj
    return best_h, best_score
```

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to N/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    N = v.size
    target = N / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / N
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    # final tiny adjustment
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size > 0:
            inc = diff / free.size
            v[free] += inc
            np.clip(v, 0.0, 1.0, out=v)
        else:
            v[:] += diff / N
            np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
