def project_onto_box_sum(v, target, max_iter=100):
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        mid = (lo+hi)/2
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0.0, 1.0)

def _gradient_descent(h_start, time_limit, rng):
    N = h_start.size
    target = N/2.0
    h = h_start.copy()
    # ensure sum correct
    h = _project_onto_box_sum(h, target)
    best_h = h.copy()
    best_val = _c5_value(best_h)
    start = time.time()
    iteration = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # compute correlation c
        c = np.correlate(h, 1 - h, mode='full')
        # schedule alpha and step size gradually
        # Compute fraction of time elapsed (0->1)
        frac = elapsed / time_limit
        # alpha from 5 to 200
        alpha = 5.0 + 195.0 * frac
        # compute weights
        c_max = c.max()
        exp_vals = np.exp(alpha * (c - c_max))
        p = exp_vals / exp_vals.sum()
        # compute forward and backward
        p_rev = p[::-1]
        forward = np.convolve(p_rev, h, mode='full')[N-1:N-1+N]
        backward = np.convolve(p, h, mode='full')[N-1:N-1+N]
        grad = np.full(N, 1.0) - forward - backward
        # step size decay
        step = 0.5 * (1 - frac) + 0.01

            best_h = h_new.copy()
        # maybe accept new if val <= current? We always go to new (projected) anyway.
        h = h_new
        iteration += 1
    return best_h, best_val

import numpy as np
import time
import math

# ----- Utility functions -----
def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce exact sum = len(v)/2."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Exact overlap integral C5 for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_onto_box_sum(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    proj = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return proj

# ----- Gradient descent with smooth max -----
def _gradient_descent(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    N = h_start.size
    target = N / 2.0
    h = _project_onto_box_sum(h_start, target)
    best_h = h.copy()
    best_val = _c5_value(best_h)
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # Correlation (no dx scaling needed for optimisation)
        c = np.correlate(h, 1.0 - h, mode='full')
        # Smooth‑max parameters
        frac = elapsed / time_limit
        alpha = 5.0 + 195.0 * frac            # 5 → 200
        c_max = c.max()
        exp_vals = np.exp(alpha * (c - c_max))
        p = exp_vals / exp_vals.sum()        # weights, sum to 1
        # Gradient of smooth‑max: 1 - Σ p_k h_{j+k} - Σ p_k h_{j-k}
        p_rev = p[::-1]
        forward = np.convolve(p_rev, h, mode='full')[N-1:N-1+N]
        backward = np.convolve(p, h, mode='full')[N-1:N-1+N]
        grad = np.full(N, 1.0) - forward - backward
        # Step size schedule
        step = 0.5 * (1 - frac) + 0.01       # 0.5 → 0.01
        h_new = h - step * grad
        h_new = _project_onto_box_sum(h_new, target)
        val = _c5_value(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val

# ----- Simulated annealing (fallback / refinement) -----
def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.005                # more aggressive than before
    max_eps = 0.3
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)          # positive move bound
        down = min(hi, 1.0 - hj)        # negative move bound
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score
