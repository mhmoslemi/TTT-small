import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def deterministic_local_search(h):
    n = len(h)
    while True:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_improve = 0.0
        best_swap = None
        for i in ones:
            for j in zeros:
                # swap
                h_candidate = h.copy()
                h_candidate[i] = 0
                h_candidate[j] = 1
                c5_candidate = compute_c5(h_candidate.astype(float))
                improv = current_c5 - c5_candidate
                if improv > best_improve + 1e-12:
                    best_improve = improv
                    best_swap = (i,j,c5_candidate)
        if best_swap is None:
            break
        i,j,new_c5 = best_swap
        h[i] = 0
        h[j] = 1
        current_c5 = new_c5
    return h, current_c5
```

import time
import numpy as np
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    # Determine n
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
        k = n // 2
        # convert to binary by top entries
        idx = np.argpartition(-init, k-1)[:k]
        h0 = np.zeros(n, dtype=float)
        h0[idx] = 1.0
    else:
        n = 80
        k = n // 2
        h0 = np.zeros(n, dtype=float)
        h0[:k] = 1.0
        rng.shuffle(h0)
    target_sum = n / 2.0
    # ensure exact sum by adjusting tiny diff if needed
    cur_sum = np.sum(h0)
    if abs(cur_sum - target_sum) > 1e-12:
        # adjust uniformly
        diff = target_sum - cur_sum
        h0 += diff / n
        # clip and fix residual
        h0 = np.clip(h0, 0.0, 1.0)
        # recompute sum diff
        diff2 = target_sum - np.sum(h0)
        if abs(diff2) > 1e-12:
            # adjust first element
            adjust = np.clip(h0[0] + diff2, 0.0, 1.0) - h0[0]
            h0[0] += adjust
    # Compute initial C5
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    # Simulated annealing parameters
    T0 = best_c5  # initial temperature
    Tmin = 1e-6
    delta_initial = 0.5  # initial max delta change (portion of values)
    delta_final = 1e-4
    # Main SA loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 1.0:  # keep a margin
            break
        # progress ratio
        progress = elapsed / budget_s
        T = max(T0 * (1.0 - progress), Tmin)
        delta_max = max(delta_initial * (1.0 - progress), delta_final)
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # randomly decide direction: increase i, decrease j
        if rng.random() < 0.5:
            inc_idx, dec_idx = i, j
        else:
            inc_idx, dec_idx = j, i
        # compute feasible delta
        max_inc = 1.0 - cur_h[inc_idx]
        max_dec = cur_h[dec_idx]
        feasible = min(max_inc, max_dec, delta_max)
        if feasible <= 0:
            continue
        delta = rng.random() * feasible  # uniform [0, feasible)
        # propose new h
        h_new = cur_h.copy()
        h_new[inc_idx] += delta
        h_new[dec_idx] -= delta
        # ensure bounds (should be within but clip for safety)
        h_new[inc_idx] = min(1.0, h_new[inc_idx])
        h_new[dec_idx] = max(0.0, h_new[dec_idx])
        # compute C5
        new_c5 = compute_c5(h_new)
        # acceptance
        if new_c5 < cur_c5:
            cur_h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        else:
            # metropolis acceptance
            if rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                cur_h = h_new
                cur_c5 = new_c5
        # occasionally reproject sum to exact target (to avoid drift)
        if (elapsed % 10) < 0.001:  # roughly every 10 seconds
            diff_sum = target_sum - np.sum(cur_h)
            if abs(diff_sum) > 1e-9:
                # adjust a random index within bounds
                # find indices that can accommodate diff_sum
                if diff_sum > 0:
                    candidates = np.where(cur_h < 1.0)[0]
                else:
                    candidates = np.where(cur_h > 0.0)[0]
                if candidates.size > 0:
                    idx_adj = rng.choice(candidates)
                    adjust = np.clip(cur_h[idx_adj] + diff_sum, 0.0, 1.0) - cur_h[idx_adj]
                    cur_h[idx_adj] += adjust
        # optional early break if we beat target bound
        if best_c5 < 0.3808:
            # we reached target, can stop
            break
    # final correction for best_h sum
    sum_best = np.sum(best_h)
    if abs(sum_best - target_sum) > 1e-9:
        diff = target_sum - sum_best
        # adjust first element that can accommodate diff
        if diff > 0:
            idx = np.where(best_h < 1.0)[0]
        else:
            idx = np.where(best_h > 0.0)[0]
        if idx.size > 0:
            i0 = idx[0]
            best_h[i0] = np.clip(best_h[i0] + diff, 0.0, 1.0)
        # reclip entire vector
        best_h = np.clip(best_h, 0.0, 1.0)
    # final clip
    best_h = np.clip(best_h, 0.0, 1.0)
    # compute final bound for best_h (in case corrections changed it)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n

import time
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise ----
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        half = n // 2
        # take the half largest entries to obtain a feasible binary start
        idx = np.argpartition(-init, half - 1)[:half]
        h0 = np.zeros(n, dtype=float)
        h0[idx] = 1.0
    else:
        n = 80
        half = n // 2
        h0 = np.zeros(n, dtype=float)
        h0[:half] = 1.0
        rng.shuffle(h0)

    target_sum = n / 2.0
    # enforce exact sum (tiny uniform adjustment)
    cur_sum = np.sum(h0)
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        h0 += diff / n
        h0 = np.clip(h0, 0.0, 1.0)
        # final tiny correction
        residual = target_sum - np.sum(h0)
        if abs(residual) > 1e-12:
            # adjust first element that can absorb the residual
            if residual > 0:
                cand = np.where(h0 < 1.0)[0]
            else:
                cand = np.where(h0 > 0.0)[0]
            if cand.size:
                i0 = cand[0]
                h0[i0] = np.clip(h0[i0] + residual, 0.0, 1.0)

    # ---- current state ----

    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ---- Simulated annealing parameters ----
    T0 = best_c5                     # initial temperature
    Tmin = 1e-6
    delta_initial = 1.0              # initial max delta for a move
    delta_final = 1e-4               # final max delta

    # ---- main SA loop ----
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1.0:      # keep a little margin

        # linear cooling and delta schedule
        progress = elapsed / budget_s
