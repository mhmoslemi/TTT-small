def c5_value(h):
    N = len(h)
    w = N/2
    # compute convolution self
    corr = np.correlate(h, h, mode='full')
    min_corr = corr.min()
    dx = 2.0/N
    return dx * (w - min_corr)  # equals (2/N)*(w - min_corr)

import numpy as np
import time

def _c5_score(h):
    N = h.size
    w = N/2
    # Correlate with itself
    corr = np.correlate(h, h, mode='full')
    min_corr = corr.min()
    # C5 = dx * (w - min_corr) where dx = 2/N
    return (2.0/N)*(w - min_corr)
```

```
def _simulated_annealing_binary(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_score(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = max(0.01, cur_score * 0.1)   # dynamic maybe
    # do while time
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # compute temperature
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        # try a random swap
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h_cur[i], h_cur[j] = 0, 1
        new_score = _c5_score(h_cur)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            # accept swap
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            # reject, revert
            h_cur[i], h_cur[j] = 1, 0
    return best_h, best_score
```

import numpy as np
import time
import math

def _c5_score(h: np.ndarray) -> float:
    """Return C5 = max shift integral for a discretised step function."""
    N = h.size
    w = N / 2.0                       # total integral = w*dx = 1
    corr = np.correlate(h, h, mode='full')
    min_corr = corr.min()             # worst autocorrelation
    return (2.0 / N) * (w - min_corr)

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to exactly len(v)/2."""
    v = np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    # one more correction if needed
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        v *= target / s2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """SA that preserves sum and box constraints; works for continuous values."""
    h_cur = h_start.copy()
    cur_score = _c5_score(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.02            # higher initial temperature
    max_eps = 0.5        # allow larger moves
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)          # positive epsilon bound
        down = min(hi, 1.0 - hj)        # negative epsilon bound
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_score(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # ---- initial construction -------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
    except NameError:
        init = None

    base_N = 200
    if init is not None and init.size > 0:
        base_N = max(base_N, init.size)
    if base_N % 2:
        base_N += 1
    base_N = min(base_N, 800)   # keep modest

    if init is not None and init.size == base_N:
        h0 = _repair_vector(init.copy())
    elif init is not None:
        # interpolate to desired size
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, base_N, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
        h0 = _repair_vector(h0)
    else:
        h0 = np.zeros(base_N, dtype=np.float64)
        h0[rng.choice(base_N, base_N // 2, replace=False)] = 1.0

    # ---- Phase 1 : optimise at base resolution -------------------------------
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    phase1_time = remaining * 0.7
    best_h, best_score = _simulated_annealing(h0, phase1_time, rng)

    # ---- Phase 2 : increase resolution if time permits -----------------------
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining > 30 and best_h.size * 2 <= 800:
        N2 = best_h.size * 2
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _repair_vector(h_up)
        phase2_time = remaining * 0.9
