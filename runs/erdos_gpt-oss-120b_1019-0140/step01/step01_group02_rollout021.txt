def _compute_corr_and_c5(h: np.ndarray):
    """Return correlation array and C5 value for a given h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c5 = float(np.max(corr) * dx)
    return corr, c5

def _compute_c5(h: np.ndarray) -> float:
    """Simple wrapper to compute only C5."""
    _, c5 = _compute_corr_and_c5(h)
    return c5

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to hit the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-14)[0]
    else:
        idx = np.where(h > 1e-14)[0]
    if idx.size == 0:
        # fallback: distribute evenly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-14:
        return _repair_sum(h, target)
    return h

def _apply_delta_to_corr(corr: np.ndarray, h: np.ndarray, idx: int, delta: float):
    """In‑place update of correlation array for a change h[idx] → h[idx]+delta."""
    n = h.shape[0]
    offset = n - 1
    # loop over all possible shifts
    for shift in range(-n + 1, n):
        pos = shift + offset
        contrib = 0.0
        j = idx + shift
        if 0 <= j < n:
            contrib += delta * (1.0 - h[j])
        j2 = idx - shift
        if 0 <= j2 < n:
            contrib -= delta * h[j2]
        if contrib != 0.0:
            corr[pos] += contrib

                best_h = h.copy()
        else:
            steps_since_best += 1

        steps += 1

        # periodic full recompute to avoid drift
        if steps % recompute_interval == 0:
            corr, current_c5 = _compute_corr_and_c5(h)

        # stagnation handling: re‑heat around the best solution
        if steps_since_best > 120000:
            # perturb best and restart from it
            h = best_h.copy()
            # introduce modest random noise respecting bounds
            noise = (rng.random(n) - 0.5) * step * 2.0
            h += noise
            np.clip(h, 0.0, 1.0, out=h)
            target_sum = n / 2.0
            h = _repair_sum(h, target_sum)
            corr, current_c5 = _compute_corr_and_c5(h)
            steps_since_best = 0

    # final repair & evaluation of the best solution
    best_h = _repair_sum(best_h, n / 2.0)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
