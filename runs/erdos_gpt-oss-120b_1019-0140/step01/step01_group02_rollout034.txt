def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

Define _repair_sum:

def _repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return
    # find indices that can be adjusted
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # distribute evenly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return
    i = idx[0]
    # adjust
    adjust = np.clip(diff, -h[i], 1.0 - h[i])
    h[i] += adjust
    # recurse if needed:
    if abs(diff - adjust) > 1e-12:
        _repair_sum(h, target)
```

import time, math, numpy as np

def _compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> None:
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return
    # try to adjust a single element
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # distribute uniformly as fallback
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return
    i = idx[0]
    adj = np.clip(diff, -h[i], 1.0 - h[i])
    h[i] += adj
    # recurse if not fully fixed
    if abs(diff - adj) > 1e-12:
        _repair_sum(h, target)

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> None:
    """Adjust a single entry to enforce the exact integral constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return
    # pick a mutable entry that can move in needed direction
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute uniformly then clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return
    i = idx[0]
    adj = np.clip(diff, -h[i], 1.0 - h[i])
    h[i] += adj
    if abs(diff - adj) > 1e-12:
        _repair_sum(h, target)
