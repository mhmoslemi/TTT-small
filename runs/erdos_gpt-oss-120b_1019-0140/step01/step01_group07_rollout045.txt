import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    # water-filling
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # fill in order of largest capacity to reduce loop
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _tournament_select(pop, fit, k=3):
    size = pop.shape[0]
    best_idx = np.random.randint(size)
    best_fit = fit[best_idx]
    for _ in range(k-1):
        i = np.random.randint(size)
        if fit[i] < best_fit:
            best_idx = i
            best_fit = fit[i]
    return pop[best_idx]

def _ga_optimize(start_vec, n, target_sum, start_time, budget_s):
    # hyperparameters (may adapt to budget)
    pop_size = 30 if n <= 800 else 20
    crossover_prob = 0.7
    mutation_prob = 0.3
    mutation_sigma = 0.08
    elite_num = 2
    max_generations = 200

    # Build initial population
    pop = []
    # Include interpolated start_vec (projected)
    vec0 = _project_to_constraints(start_vec.copy(), target_sum)
    pop.append(vec0)
    # Fill rest random
    while len(pop) < pop_size:
        v = np.random.rand(n)
        v = _project_to_constraints(v, target_sum)
        pop.append(v)
    pop = np.array(pop)

    # Evaluate
    fitness = np.array([_compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_score = float(fitness[best_idx])

    generation = 0
    while generation < max_generations:
        # time check
        if time.time() - start_time > budget_s * 0.97:
            break

        # Elite preservation
        elite_idxs = fitness.argsort()[:elite_num]
        new_pop = pop[elite_idxs].tolist()

        # Generate offspring
        while len(new_pop) < pop_size:
            parent1 = _tournament_select(pop, fitness)
            child = parent1.copy()
            if np.random.rand() < crossover_prob:
                parent2 = _tournament_select(pop, fitness)
                mask = np.random.rand(n) < 0.5
                child = np.where(mask, parent1, parent2)
            if np.random.rand() < mutation_prob:
                child = child + mutation_sigma * np.random.randn(n)
            child = _project_to_constraints(child, target_sum)
            new_pop.append(child)

        pop = np.array(new_pop)
        fitness = np.array([_compute_c5(ind) for ind in pop])

        # Update best
        gen_best_idx = np.argmin(fitness)
        gen_best_score = fitness[gen_best_idx]
        if gen_best_score < best_score:
            best_score = float(gen_best_score)
            best_h = pop[gen_best_idx].copy()
            if best_score < 0.3808:
                break

        generation += 1

    return best_h, best_score

def _local_refine(h, target_sum, start_time, budget_s):
    # try a few gradient refinements while time permits
    remaining = budget_s - (time.time() - start_time)
    if remaining < 5.0:
        return h, _compute_c5(h)
    try:
        cons = {"type":"eq","fun":lambda x, tgt=target_sum: np.sum(x) - tgt}
        bounds = [(0.0,1.0)] * len(h)
        opts = {"maxiter":200, "ftol":1e-9}
        res = minimize(_compute_c5, h, method="SLSQP", bounds=bounds,
                       constraints=[cons], options=opts)
        if not res.success:
            return h, _compute_c5(h)
        h_opt = _project_to_constraints(res.x, target_sum)
        return h_opt, _compute_c5(h_opt)

import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Discrete C5 evaluation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and enforce sum(v)=target by water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _tournament_select(pop, fit, k=3):
    """Select index via tournament."""
    size = pop.shape[0]
    best_idx = np.random.randint(size)
    best_fit = fit[best_idx]
    for _ in range(k - 1):
        i = np.random.randint(size)
        if fit[i] < best_fit:
            best_idx = i
            best_fit = fit[i]
    return pop[best_idx]

def _ga_optimize(start_vec, n, target_sum, start_time, budget_s):
    """Simple GA for a fixed resolution."""
    pop_size = 30 if n <= 800 else 20
    elite_num = 2
    crossover_prob = 0.7
    mutation_prob = 0.3
    mutation_sigma = 0.08
    max_generations = 200

    # initialise population
    pop = [_project_to_constraints(start_vec.copy(), target_sum)]
    while len(pop) < pop_size:
        v = np.random.rand(n)
        pop.append(_project_to_constraints(v, target_sum))
    pop = np.array(pop)

    fitness = np.array([_compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_score = float(fitness[best_idx])

    generation = 0
    while generation < max_generations:
        if time.time() - start_time > budget_s * 0.97:
            break

        elite_idxs = fitness.argsort()[:elite_num]
        new_pop = pop[elite_idxs].tolist()

        while len(new_pop) < pop_size:
            parent1 = _tournament_select(pop, fitness)
            child = parent1.copy()
            if np.random.rand() < crossover_prob:
                parent2 = _tournament_select(pop, fitness)
                mask = np.random.rand(n) < 0.5
                child = np.where(mask, parent1, parent2)
            if np.random.rand() < mutation_prob:
                child = child + mutation_sigma * np.random.randn(n)
            child = _project_to_constraints(child, target_sum)
            new_pop.append(child)

        pop = np.array(new_pop)
        fitness = np.array([_compute_c5(ind) for ind in pop])

        gen_best_idx = np.argmin(fitness)
        gen_best_score = fitness[gen_best_idx]
        if gen_best_score < best_score:
            best_score = float(gen_best_score)
            best_h = pop[gen_best_idx].copy()
            if best_score < 0.3808:
                break

        generation += 1

    return best_h, best_score

def _local_refine(h, target_sum, start_time, budget_s):
    """One SLSQP refinement, respecting remaining time."""
    if budget_s - (time.time() - start_time) < 5.0:
        return h, _compute_c5(h)
    try:
        cons = {"type": "eq", "fun": lambda x, tgt=target_sum: np.sum(x) - tgt}
        bounds = [(0.0, 1.0)] * len(h)
        res = minimize(_compute_c5, h, method="SLSQP", bounds=bounds,
                       constraints=[cons], options={"maxiter": 200, "ftol": 1e-9})
        if not res.success:
            return h, _compute_c5(h)
        h_opt = _project_to_constraints(res.x, target_sum)
        return h_opt, _compute_c5(h_opt)
    except Exception:
        return h, _compute_c5(h)
