def _repair_binary(ind, target_sum):
   # Input: binary 0/1 array but may have sum not equal target.
   current = int(np.sum(ind))
   if current == target_sum:
       return ind
   elif current > target_sum:
       # need to turn (current-target) ones to zeros
       # get indices where ind ==1
       ones = np.where(ind==1)[0]
       # randomly choose diff ones
       to_zero = np.random.choice(ones, size=current-target_sum, replace=False)
       ind[to_zero] = 0
   else: # current < target
       zeros = np.where(ind==0)[0]
       to_one = np.random.choice(zeros, size=target_sum-current, replace=False)
       ind[to_one] = 1
   return ind

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        # free indices where v < 1
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # handle total capacity
        total_cap = caps.sum()
        if total_cap < need:
            # cannot fill, set all to 1
            v[free] = 1.0
            # adjust scaling after
        else:
            # water-filling: fill proportionally? but simplest: fill in descending caps order:
            order = np.argsort(caps)[::-1]
            for idx in order:
                i = free[idx]
                add = min(need, caps[idx])
                v[i] += add
                need -= add
                if need <= 1e-12:
                    break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    # final scaling if still off
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

import numpy as np
import time
from scipy.optimize import minimize

class SumConstraint:
    def __init__(self, target):
        self.target = target
    def __call__(self, h):
        return np.sum(h) - self.target

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _real_to_binary(v, target_sum):
    idx = np.argsort(v)[-int(target_sum):]
    bin_vec = np.zeros_like(v, dtype=np.float64)
    bin_vec[idx] = 1.0
    return bin_vec

def _generate_random_binary(n, target_sum):
    ind = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, size=int(target_sum), replace=False)
    ind[idx] = 1.0
    return ind

def _repair_binary(ind, target_sum):
    cur = int(ind.sum())
    if cur == target_sum:
        return ind
    diff = cur - target_sum
    if diff > 0:
        ones = np.where(ind == 1)[0]
        if diff >= len(ones):
            ind[:] = 0.0
        else:
            to_zero = np.random.choice(ones, size=diff, replace=False)
            ind[to_zero] = 0.0
    else:
        zeros = np.where(ind == 0)[0]
        need = -diff
        if need >= len(zeros):
            ind[:] = 1.0
        else:
            to_one = np.random.choice(zeros, size=need, replace=False)
            ind[to_one] = 1.0
    return ind

def _uniform_crossover(p1, p2):
    mask = np.random.rand(p1.size) < 0.5
    child = np.where(mask, p1, p2).astype(np.float64)
    return child

def _mutate_swap(ind, mutation_rate):
    if np.random.rand() < mutation_rate:
        ones = np.where(ind == 1)[0]
        zeros = np.where(ind == 0)[0]
        if len(ones) and len(zeros):
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            ind[i] = 0.0
            ind[j] = 1.0
    return ind

def _tournament_select(pop, scores, k=3):
    idxs = np.random.choice(len(pop), size=k, replace=True)
    best_idx = idxs[0]
    best_score = scores[best_idx]
    for idx in idxs[1:]:
        if scores[idx] < best_score:
            best_idx = idx
            best_score = scores[idx]
    return pop[best_idx].copy()
