def adjust_sum_to_target(h, target, rng):
    # Ensure values in [0,1] and sum equals target.
    # Clip to bounds
    np.clip(h, 0.0, 1.0, out=h)
    total = h.sum()
    diff = target - total
    # while diff significant (>1e-12) limit loops
    # Add or subtract uniformly across indices with capacity
    # For diff>0 (need to increase), find indices where h < 1
    # For diff<0 (need to decrease), indices where h > 0
    max_iter=1000
    iter_count=0
    while abs(diff) > 1e-12 and iter_count < max_iter:
        if diff > 0:
            # capacity to increase
            cap = 1.0 - h
            total_cap = cap.sum()
            if total_cap < 1e-12:
                break
            # distribute proportionally
            # respond by increment = diff * cap/total_cap
            inc = diff * cap / total_cap
            h += inc
        else:
            # need to decrease
            cap = h
            total_cap = cap.sum()
            if total_cap < 1e-12:
                break
            dec = diff * cap / total_cap  # diff negative
            h += dec  # dec is negative
        total = h.sum()
        diff = target - total
        iter_count += 1
    # final clip again
    np.clip(h, 0.0, 1.0, out=h)
    # If still diff non-zero (tiny), adjust uniformly and clip again
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h
```

Proportionally distributes diff across all capacity; this converges quickly.

    pop_size = 30

    CR = 0.9

    population = []
    for _ in range(pop_size):
        h = rng.rand(n_points)
        # adjust to sum target and bounds
        h = adjust_sum_to_target(h, target_sum, rng)
        population.append(h)
    population = np.array(population)
    
    # Optionally replace one individual with initial if provided
    if 'initial_h_values' in globals():
        try:
            init_vec = np.array(initial_h_values, dtype=float)
            # ensure length matches n_points; if not, resample to n_points via interpolation
            if init_vec.size != n_points:
                # interpolate using linear scaling
                x_old = np.linspace(0, 2, init_vec.size, endpoint=False)
                x_new = np.linspace(0, 2, n_points, endpoint=False)
                init_vec = np.interp(x_new, x_old, init_vec)
            init_vec = adjust_sum_to_target(init_vec, target_sum, rng)
            population[0] = init_vec
        except Exception:
            pass
    
    # Evaluate initial fitness
    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = int(fitness.argmin())
    best_h = population[best_idx].copy()
    best_c5 = float(fitness[best_idx])
    
    # DE main loop
    while time.time() - start < max_time:
        # generate new population
        for i in range(pop_size):
            # select three distinct indices != i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])

            mutant = np.clip(mutant, 0.0, 1.0)

            cross = rng.rand(n_points) < CR

            # enforce constraints
            trial = np.clip(trial, 0.0, 1.0)
            trial = adjust_sum_to_target(trial, target_sum, rng)

            trial_f = compute_c5(trial)
            # selection

                fitness[i] = trial_f
                if trial_f < best_c5:
                    best_c5 = trial_f
                    best_h = trial.copy()

        # check time again
        if time.time() - start >= max_time:
            break
    # final ensure sum
    best_h = adjust_sum_to_target(best_h, target_sum, rng)

    return (best_h, float(final_c5), n_points)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def adjust_sum_to_target(h, target, rng):
    # ensure bounds
    np.clip(h, 0.0, 1.0, out=h)
    total = h.sum()
    diff = target - total
    # distribute diff proportionally to capacity
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            capacity = 1.0 - h
        else:
            capacity = h
        cap_sum = capacity.sum()
        if cap_sum <= 0:
            break
        # distribute proportionally
        delta = diff * capacity / cap_sum
        h += delta
        total = h.sum()
        diff = target - total
    # final clipping and tiny fix
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute uniformly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)

            if init_h.size != n_points:
                x_old = np.linspace(0, 2, init_h.size, endpoint=False)
                x_new = np.linspace(0, 2, n_points, endpoint=False)
                init_h = np.interp(x_new, x_old, init_h)
            init_h = adjust_sum_to_target(init_h, target_sum, rng)
            population[0] = init_h
        except Exception:
            pass

    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = float(fitness[best_idx])

    # Differential Evolution main loop
    while time.time() - start < max_time:
        for i in range(pop_size):
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = adjust_sum_to_target(mutant, target_sum, rng)

            cross = rng.rand(n_points) < CR
            if not np.any(cross):
                cross[rng.randint(n_points)] = True
            trial = np.where(cross, mutant, population[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = adjust_sum_to_target(trial, target_sum, rng)

            trial_f = compute_c5(trial)
            if trial_f <= fitness[i]:
                population[i] = trial
                fitness[i] = trial_f
                if trial_f < best_c5:
                    best_c5 = trial_f
                    best_h = trial.copy()
        if time.time() - start >= max_time:
            break

    # Gradient refinement on the best individual
    current_h = best_h.copy()
    current_c5 = best_c5
    T0 = 1e-3
    step0 = 0.2
    while time.time() - start < max_time:
        elapsed = time.time() - start
        progress = elapsed / max_time
        step = step0 * (1.0 - progress) + 1e-4 * progress
        T = T0 * (1.0 - progress) + 1e-6 * progress

        corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
        k_idx = int(np.argmax(corr_full))
        shift = k_idx - (n_points - 1)

        g = np.zeros_like(current_h)
        n = n_points
        if shift >= 0:
            i1 = slice(0, n - shift)
            i2 = slice(shift, n)
            g[i1] += (1.0 - current_h[i2])
            g[i2] -= current_h[i1]
        else:
            k = -shift
            i1 = slice(k, n)
            i2 = slice(0, n - k)
            g[i1] += (1.0 - current_h[i2])
            g[i2] -= current_h[i1]
        g -= g.mean()  # keep total mass unchanged

        candidate = current_h - step * g
        np.clip(candidate, 0.0, 1.0, out=candidate)
        candidate = adjust_sum_to_target(candidate, target_sum, rng)

        cand_c5 = compute_c5(candidate)
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob
        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()

    # Final exact-sum correction
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1 - 1e-12))[0]
        if free.size == 0:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)
        else:
            best_h[free[0]] = np.clip(best_h[free[0]] + diff, 0.0, 1.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
