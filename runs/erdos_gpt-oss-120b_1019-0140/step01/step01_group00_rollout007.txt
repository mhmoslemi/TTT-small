def simulated_annealing(h0, init_val, start_time, budget_s):
    h = h0.copy()
    cur_val = init_val
    n = len(h)
    rng = np.random.default_rng()
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    it = 0
    while time.time() - start_time < budget_s - 0.1:
        T = max(1e-4, (budget_s - (time.time() - start_time)) / budget_s) # linearly decreasing
        # propose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand)
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = cand
            cur_val = cand_val
            # update ones zeros
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
        it += 1
    return h, cur_val

import time
import numpy as np

def compute_c5(h):
    """Return C5 for binary or real array h (values in [0,1])."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return np.max(corr) * dx

def tournament_select(pop, fitness, k, rng):
    """Select one individual from pop using k‑tournament."""
    idx = rng.choice(len(pop), size=k, replace=False)
    best = idx[0]
    best_f = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_f:
            best = i
            best_f = fitness[i]
    return pop[best]

def mutate(ind, rng):
    """Return a mutated copy of binary individual ind (swap 1‑3 pairs)."""
    h = ind.copy()
    swaps = rng.integers(1, 4)  # 1,2 or 3 swaps
    for _ in range(swaps):
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
    return h

def hill_climb_swap(h_start, start_val, start_time, budget_s):
    """Simple greedy hill‑climb using single‑swap moves."""
    n = len(h_start)
    best_h = h_start.copy()
    best_val = start_val
    rng = np.random.default_rng()
    while time.time() - start_time < budget_s - 0.2:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        val = compute_c5(cand)
        if val + 1e-12 < best_val:
            best_h = cand
            best_val = val
    return best_h, best_val
