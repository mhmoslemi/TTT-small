def propose_move(h, step):
   i, j = rng.integers(0, n, size=2)
   hi, hj = h[i], h[j]
   # Determine max delta such that hi+delta in [0,1] and hj - delta in [0,1]
   delta_min = max(-hi, -(1 - hj))
   delta_max = min(1 - hi, hj)
   # bound by step magnitude
   low = max(delta_min, -step)
   high = min(delta_max, step)
   if low >= high: return None
   delta = rng.uniform(low, high)
   return i, j, delta
```

def repair_sum(h, target):
    diff = target - h.sum()
    # tolerance 1e-12
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0: break
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0: break
        i = idx[0]
        delta = np.clip(diff, -(h[i]), 1 - h[i])
        h[i] += delta
        diff = target - h.sum()
    return h
```

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust entries so that sum(h) == target, preserving bounds."""
    # Simple iterative redistribution
    diff = target - h.sum()
    eps = 1e-12
    while abs(diff) > eps:
        if diff > 0:
            idx = np.where(h < 1.0 - eps)[0]
        else:
            idx = np.where(h > eps)[0]
        if idx.size == 0:
            # All entries are at a bound; break to avoid infinite loop
            break
        i = idx[0]
        delta = np.clip(diff, -h[i], 1.0 - h[i])
        h[i] += delta
        diff = target - h.sum()
    return h

def _upsample(h: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of a coarse vector to a finer grid."""
    n0 = len(h)
    if n0 == n_target:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x1, x0, h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise ----------------------------------------------------
    try:
        init0 = np.asarray(initial_h_values, dtype=float)  # may be provided
        n0 = init0.shape[0]
    except NameError:
        init0 = None
        n0 = None

    # target discretisation size – a compromise between quality and speed
    n_target = 200

    if init0 is not None:
        h = _upsample(init0, n_target)
    else:
        # random start in [0,1]
        h = rng.random(n_target)

    n = h.shape[0]
    target_sum = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    _repair_sum(h, target_sum)

    current = h.copy()
    current_c5 = _compute_c5(current)
    best_h = current.copy()
    best_c5 = current_c5

    # ----- annealing schedule --------------------------------------------
    max_step = 0.5          # early‑stage maximal delta
    min_step = 0.001        # final precision
    max_temp = 0.05         # early temperature
    min_temp = 1e-6         # final temperature

    # ----- main simulated‑annealing loop ---------------------------------
    while time.time() - start < budget_s * 0.78:
        elapsed = time.time() - start
        frac = elapsed / budget_s  # 0 → 1
        # exponential cooling for step and temperature
        step = max_step * (min_step / max_step) ** frac
        temp = max_temp * (min_temp / max_temp) ** frac

        # ------- basic pair‑move -----------------------------------------
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1  # ensure i != j

        hi, hj = current[i], current[j]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        low = max(delta_min, -step)

            if new_c5 < current_c5 or rng.random() < math.exp(-(new_c5 - current_c5) / max(temp, 1e-12)):

        delta_max = min(1.0 - hi, hj)

        low = max(delta_min, -step_hc)

        best_h[j] = hj - delta
        new_c5 = _compute_c5(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            best_h[i] = hi
            best_h[j] = hj

    # ----- final sanity check --------------------------------------------
    _repair_sum(best_h, n / 2.0)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
