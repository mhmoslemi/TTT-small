        v = np.clip(v, 0.0, 1.0)

import numpy as np
import time

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust sum(v) to equal target (water-filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    diff = target - cur
    if diff > 0.0:  # need to add mass
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # fill largest capacities first (order does not matter for feasibility)
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(diff, caps[idx])
            v[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:  # need to remove mass
        diff = -diff
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(diff, caps[idx])
            v[i] -= sub
            diff -= sub
            if diff <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-9 and cur > 0.0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _conv_full(a, b):
    """Linear convolution via FFT (real inputs)."""
    L = len(a) + len(b) - 1
    n_fft = 1 << ((L - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)
    return conv[:L]

def _correlation_full(h):
    """Full correlation r_s = sum_i h[i]*(1-h[i+s]) using FFT."""
    n = h.shape[0]
    g = 1.0 - h
    L = 2 * n - 1
    n_fft = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    G_rev = np.fft.rfft(g[::-1], n=n_fft)
    conv = np.fft.irfft(H * G_rev, n=n_fft)
    conv = conv[:L]
    return conv[::-1]   # correlation order matches np.correlate

def _compute_c5_from_corr(r, n):
    """Convert raw correlation to C5 value."""
    dx = 2.0 / n
    return np.max(r) * dx
