import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    # trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # initial bounds for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure that f(lam_low) >= s >= f(lam_high)
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10
    # binary search
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the active overlap term for given integer shift."""
    n = h.size
    grad = np.ones_like(h)
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad

def active_shift_and_grad(h):
    """Return the shift attaining the max overlap and its gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = compute_gradient(h, shift)
    return shift, grad, corr[idx] * (2.0 / n)

def adam_optimize(h0, max_iters, lr, beta1, beta2, eps, start_time, budget_s):
    """Adam optimizer on the max‑overlap objective with projection."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    while t < max_iters:
        t += 1
        if time.time() - start_time > budget_s - 0.2:
            break
        shift, grad, _ = active_shift_and_grad(h)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    """Linear interpolation of arr to new_n points."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    rng = np.random.default_rng(seed)
    start = time.time()
    # primary resolution (use 256 if possible)
    target_n = 256
    # build initial candidates
    candidates = []

        init_arr = np.asarray(initial_h_values, dtype=float)

    # 2. Simple balanced binary patterns
    for _ in range(3):
        v = np.zeros(target_n, dtype=float)

        v[idx] = 1.0

    # 3. Random fractional starts
    for _ in range(4):
        v = rng.random(target_n)
        v = project_onto_box_sum(v, target_n / 2.0)

    best_h = None
    best_score = float('inf')
    # Adam hyper‑parameters (tuned for stability)
    lr = 0.08
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Optimize the seed candidates
    # ------------------------------------------------------------------
    for h0 in candidates:

    # ------------------------------------------------------------------

    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)
