def project_to_feasible(h, target_sum, tol=1e-8, max_iter=20):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()

        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:

def compute_c5(h):
    """Compute the Erdős C5 value for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=20):
    """Project onto the box [0,1]^n intersected with the hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h
