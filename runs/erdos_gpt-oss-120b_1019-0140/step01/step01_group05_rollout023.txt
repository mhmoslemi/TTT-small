def _subgradient_optimize(h_start, time_limit, rng):
    h_cur = h_start.copy()
    n = h_cur.size
    target = n/2.0
    dx = 2.0/n
    best_h = h_cur.copy()
    best_score = _c5_value(h_cur)
    start = time.time()
    # parameters
    step0 = 0.5  # initial step size (aggressive)
    iter = 0
    while True:
        if time.time() - start >= time_limit:
            break
        # compute correlation
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        idx = np.argmax(corr)
        cur_score = float(corr[idx] * dx)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h_cur.copy()
        tau = idx - (n - 1)  # shift integer
        # gradient
        grad = np.zeros(n, dtype=np.float64)
        if tau >= 0:
            shift = tau
            if shift < n:
                grad[:n-shift] += 1 - h_cur[shift:]
                grad[shift:] -= h_cur[:n-shift]
        else:
            shift = -tau
            # tau negative
            if shift < n:
                grad[shift:] += 1 - h_cur[:n-shift]
                grad[:n-shift] -= h_cur[shift:]

        step = step0 / np.sqrt(iter + 1)

        h_new = h_cur - step * grad
        # add small random noise to encourage exploration

def _subgradient_optimize(h_start, time_limit, rng):
    h_cur = h_start.copy()
    n = h_cur.size
    dx = 2.0 / n
    best_h = h_cur.copy()
    best_score = _c5_value(h_cur)
    start = time.time()
    iter = 0
    # constants
    step0 = 0.5
    while True:
        if time.time() - start >= time_limit:
            break
        # Compute correlation and identify worst shift
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        idx = np.argmax(corr)
        cur_score = float(corr[idx] * dx)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h_cur.copy()
        tau = idx - (n - 1)
        # compute gradient of the worst shift
        grad = _gradient_for_shift(h_cur, tau)
        # step size schedule
        step = step0 / np.sqrt(iter + 1)
        # move
        h_new = h_cur - step * grad
        # add small perturbation
        h_new += step * 0.05 * rng.standard_normal(n)
        # repair to constraints
        h_new = _repair_vector(h_new)
        # evaluate new
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_score = float(np.max(corr_new) * dx)
        # acceptance
        delta = new_score - cur_score
        temp = step0 / np.sqrt(iter + 1)
        if delta < 0 or rng.random() < math.exp(-delta / (temp + 1e-15)):
            h_cur = h_new
            # cur_score updated next loop
        # else keep h_cur unchanged
        iter += 1
    return best_h, best_score

import numpy as np

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip entries to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    for _ in range(5):
        s = v.sum()
        if abs(s - target) <= 1e-12:
            break
        if s == 0.0:
            v[:] = target / v.size
        else:
            v *= target / s
        v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment if still off
    s = v.sum()
    if abs(s - target) > 1e-12:
        diff = target - s
        free = (v > 0.0) & (v < 1.0)
        cnt = free.sum()
        if cnt > 0:
            v[free] += diff / cnt
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Exact C5 value for a discretized h."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _gradient_for_shift(h: np.ndarray, tau: int) -> np.ndarray:
    """Subgradient of g_tau(h)=∑ h_i (1-h_{i+tau}) w.r.t. h."""
    n = h.size
    grad = np.zeros_like(h)
    if tau >= 0:
        s = tau
        if s < n:
            # (1 - h_{i+tau}) for i = 0 .. n-s-1
            grad[:n - s] += 1.0 - h[s:]
            # -h_{i-tau} for i = s .. n-1
            grad[s:] -= h[:n - s]
    else:
        s = -tau
        if s < n:
            # (1 - h_{i+tau}) for i = s .. n-1 (i+tau = i-s)
            grad[s:] += 1.0 - h[:n - s]
            # -h_{i-tau} for i = 0 .. n-s-1 (i-tau = i+s)
            grad[:n - s] -= h[s:]
    return grad

        h_new += step * 0.05 * rng.standard_normal(n)
        h_new = _repair_vector(h_new)
        # evaluate proposal
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_score = float(np.max(corr_new) * dx)
        delta = new_score - cur_score
        temp = step0 / math.sqrt(it + 1)
        if delta < 0.0 or rng.random() < math.exp(-delta / (temp + 1e-15)):
            h_cur = h_new
        it += 1
        # occasional restart from best + noise to escape stagnation
        if it % 300 == 0 and time.time() - start < time_limit * 0.9:
            h_cur = best_h + step0 * 0.1 * rng.standard_normal(n)
            h_cur = _repair_vector(h_cur)
    return best_h, best_score

def _tent_shape(n: int) -> np.ndarray:
    """Symmetric tent (piecewise linear) shape on [0,2] with integral 1."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.where(x <= 1.0, x, 2.0 - x)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # initial construction
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        # fall back to simple tent shape of size 120
        h0 = _tent_shape(120)
    h0 = _repair_vector(h0)
    start_total = time.time()
    total_budget = float(budget_s)

    # Phase 1: optimise at original resolution
    phase1_time = total_budget * 0.45
    best_h, best_score = _subgradient_optimize(h0, phase1_time, rng)

    elapsed = time.time() - start_total
    remaining = total_budget - elapsed

    # Phase 2: increase resolution if time permits
    if remaining > total_budget * 0.3:
        n_up = min(best_h.size * 2, 800)
        if n_up != best_h.size:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = _repair_vector(h_up)
            phase2_time = remaining * 0.6
            h2, s2 = _subgradient_optimize(h_up, phase2_time, rng)
            if s2 < best_score:
                best_h, best_score = h2, s2
            elapsed = time.time() - start_total
            remaining = total_budget - elapsed

    # Phase 3: random restarts at current best resolution
    while remaining > 30.0:
        restart_time = min(remaining * 0.5, 60.0)
        rand_vec = rng.random(best_h.size)
        rand_vec = _repair_vector(rand_vec)
        h_r, s_r = _subgradient_optimize(rand_vec, restart_time, rng)
        if s_r < best_score:
            best_h, best_score = h_r, s_r
        elapsed = time.time() - start_total
        remaining = total_budget - elapsed

    # final repair and evaluation
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
