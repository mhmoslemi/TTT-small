def search_binary(n, budget, start_time, rng):
    best_h = None; best_val = np.inf
    while time.time() - start_time < budget:
        # generate random vector with weight n/2
        h = np.zeros(n, dtype=int)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1
        cur_val = compute_c5(h)
        # maybe local improvements
        stagnation = 0
        while time.time() - start_time < budget:
            # propose random swap
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)  # or where h==0
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = h[j], h[i]
            new_val = compute_c5(h)
            if new_val + 1e-12 < cur_val:
                cur_val = new_val
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_h = h.copy()
                stagnation = 0
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]
                stagnation += 1
                if stagnation >= 5000:
                    break
        if best_val < 0.3808: break
    return best_h, best_val

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
