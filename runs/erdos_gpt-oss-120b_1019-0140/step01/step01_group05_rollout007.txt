def _simulated_annealing(h_start, time_limit, rng, T0=0.02, max_eps=0.4):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    # Use linear schedule: fraction = (remaining time)/time_limit
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = 1.0 - elapsed / time_limit
        T = T0 * frac

        # pick random i, j
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        hi = h_cur[i]
        hj = h_cur[j]
        # compute bounds
        # moving epsilon from i to j: we increase hi by eps, decrease hj by eps (preserve sum)
        # So eps positive => hi increases, hj decreases
        # eps must satisfy hi+eps <= 1 and hi+eps >= 0 etc.
        # Also hj-eps between 0 and 1.
        # So eps <= 1- hi (positive bound)
        # eps <= hj (negative bound magnitude)
        # For negative eps: hi+eps >=0 => eps >= -hi
        # and hj-eps <=1 => eps >= -(1-hj)
        # So we can get lower bound (down) = min(hi, 1-hj) for negative epsilon magnitude
        # The previous code computed up/min and down accordingly.
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        # randomly choose eps uniformly -down..up
        eps = rng.random() * (up + down) - down
        # limit to max_eps
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        # create new vector
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # Guarantee bounds after clip
        # Actually due to eps clamping should be fine
        # Evaluate
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    N = v.size
    target = N / 2.0
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if s == 0.0:
        v[:] = target / N
        return np.clip(v, 0.0, 1.0)
    # scale, clip, possibly iterate a bit
    for _ in range(3):
        factor = target / s
        v *= factor
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) <= 1e-12:
            break
    # tiny correction if still off
    delta = target - v.sum()
    if abs(delta) > 1e-12:
        free = (v > 0.0) & (v < 1.0)
        cnt = free.sum()
        if cnt > 0:
            v[free] += delta / cnt
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised function."""
    N = h.size
    f = 1.0 - h
    L = 2 * N
    H = np.fft.fft(h, n=L)
    F = np.fft.fft(f, n=L)
    corr = np.fft.ifft(H * np.conj(F)).real
    corr = corr[:2 * N - 1]          # linear correlation (same values as np.correlate)
    dx = 2.0 / N
    return float(np.max(corr) * dx)

def _upsample_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger resolution."""
    old_n = v.size
    if new_n == old_n:
        return v.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, v)

def _simulated_annealing(h_start: np.ndarray,
                         time_limit: float,
                         rng: np.random.Generator,
                         T0: float = 0.02,
                         max_eps: float = 0.5) -> tuple[np.ndarray, float]:
    """Simple SA that respects the box and sum constraints."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    restart_prob = 0.0008

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # linear temperature schedule
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac

        # occasional random restart to escape stagnation
        if rng.random() < restart_prob:
            h_cur = _repair_vector(rng.random(N))
            cur_score = _c5_value(h_cur)
            continue

        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]

        # feasible epsilon bounds while preserving box constraints
        up = min(1.0 - hi, hj)          # positive epsilon (increase hi, decrease hj)
        down = min(hi, 1.0 - hj)        # magnitude of negative epsilon
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue

        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps

        # safety clip (should already be inside bounds)
        if not (0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0):
            continue

        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

    rng = np.random.default_rng(seed)
