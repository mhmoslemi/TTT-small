def _greedy_descent(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    N = h_cur.size
    dx = 2.0/N
    start = time.time()
    max_eps = 0.1
    while time.time() - start < time_limit:
        i, j = rng.integers(0, N, size=2)
        if i == j:
            continue
        hi = h_cur[i]; hj = h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0 and down == 0:
            continue
        # choose epsilon sign uniformly?
        eps = (rng.random() * (up + down)) - down
        # limit

        if eps == 0:

        new_h = h_cur.copy()

        # enforce bounds: already ensured

        if new_score < cur_score - 1e-12:  # improvement

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    total = v.sum()
    if total == 0:
        # fill uniformly
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / total
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    # If still off, adjust again
    total2 = v.sum()
    if abs(total2 - target) > 1e-12:
        factor2 = target / total2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v
```

    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()

    N = h_cur.size
    dx = 2.0 / N  # not used within but okay
    start = time.time()
    T0 = 0.01
    max_eps_initial = 0.2
    max_eps_final = 0.02
    p_target = 0.2
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = 1.0 - elapsed / time_limit
        T = T0 * frac
        max_eps = max_eps_initial * frac + max_eps_final * (1 - frac)  # linear between init and final?
        # Actually we want large max_eps initially, small later:
        max_eps = max_eps_final + (max_eps_initial - max_eps_final) * frac
        # decide move type
        if rng.random() < p_target:
            # targeted move
            # compute correlation to get max shift
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            idx_max = np.argmax(corr)
            k = idx_max - (N - 1)
            # Determine possible i range
            if k >= 0:
                i_vals = rng.integers(0, N - k)  # uniform
                i = int(i_vals)
                j = i + k
            else:
                i_vals = rng.integers(-k, N)
                i = int(i_vals)
                j = i + k
            hi = h_cur[i]; hj = h_cur[j]
            # we want to move from hi to hj (decrease hi, increase hj) to reduce product hi*(1 - hj)
            # epsilon negative direction: epsilon = -delta, delta>0
            down = min(hi, 1.0 - hj)  # amount we can transfer
            if down <= 0:
                continue
            eps = -rng.random() * min(max_eps, down)
            if eps == 0.0:
                continue
        else:
            # random pair
            i, j = rng.integers(0, N, size=2)
            while i == j:
                j = rng.integers(0, N)
            hi, hj = h_cur[i], h_cur[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up == 0.0 and down == 0.0:
                continue
            # random direction:
            eps_raw = rng.random() * (up + down) - down
            # limit magnitude
            if abs(eps_raw) > max_eps:
                eps_raw = math.copysign(max_eps, eps_raw)
            eps = eps_raw
        # apply move
        # ensure after move stays within bounds; it should
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # compute new score
        new_score = _c5_value(h_new)

            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()

import numpy as np

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) == len(v)/2 exactly."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        v = np.clip(v, 0.0, 1.0)
        return v
    factor = target / s
    v *= factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simulated annealing that respects box and sum constraints."""
    h_cur = _repair_vector(h_start.copy())
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.02               # initial temperature
    max_eps_init = 0.2
    max_eps_fin = 0.02
    p_target = 0.25         # probability to use a targeted move
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)   # goes from 1 to 0
        T = T0 * frac
        max_eps = max_eps_fin + (max_eps_init - max_eps_fin) * frac
        # decide move type
        if rng.random() < p_target:
            # targeted move based on current worst shift
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            idx_max = np.argmax(corr)
            k = idx_max - (N - 1)
            if k == 0:
                continue
            if k > 0:
                limit = N - k
                if limit <= 0:
                    continue
                i = int(rng.integers(0, limit))
                j = i + k
            else:
                limit = N + k
                if limit <= 0:
                    continue
                i = int(rng.integers(-k, N))
                j = i + k
            hi, hj = h_cur[i], h_cur[j]
            down = min(hi, 1.0 - hj)   # amount we can transfer from i to j
            if down <= 0.0:
                continue
            eps = -rng.random() * min(max_eps, down)   # negative: move from i to j
        else:
            # random pair move
            i, j = rng.integers(0, N, size=2)
            while i == j:
                j = rng.integers(0, N)
            hi, hj = h_cur[i], h_cur[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up == 0.0 and down == 0.0:
                continue
            eps_raw = rng.random() * (up + down) - down
            if abs(eps_raw) > max_eps:
                eps_raw = math.copysign(max_eps, eps_raw)
            eps = eps_raw
        # apply move
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # bounds already ensured by construction
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _greedy_descent(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Deterministic hill climbing that only accepts improvements."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    N = h_cur.size
    start = time.time()
    max_eps_init = 0.1
    max_eps_fin = 0.01
    p_target = 0.25
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        max_eps = max_eps_fin + (max_eps_init - max_eps_fin) * frac
        if rng.random() < p_target:
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            idx_max = np.argmax(corr)
            k = idx_max - (N - 1)
            if k == 0:
                continue
            if k > 0:
                limit = N - k
                if limit <= 0:
                    continue
                i = int(rng.integers(0, limit))
                j = i + k
            else:
                limit = N + k
                if limit <= 0:
                    continue
                i = int(rng.integers(-k, N))
                j = i + k
            hi, hj = h_cur[i], h_cur[j]
            down = min(hi, 1.0 - hj)
            if down <= 0.0:
                continue
            eps = -rng.random() * min(max_eps, down)
        else:
            i, j = rng.integers(0, N, size=2)
            if i == j:
                continue
            hi, hj = h_cur[i], h_cur[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up == 0.0 and down == 0.0:
                continue
            eps_raw = rng.random() * (up + down) - down
            if abs(eps_raw) > max_eps:
                eps_raw = math.copysign(max_eps, eps_raw)
            eps = eps_raw
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        if new_score < cur_score - 1e-12:
            h_cur = h_new
            cur_score = new_score
    return h_cur, cur_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    # initial construction
    try:
        h_init = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 120
        h_init = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)], dtype=np.float64)
    h_init = _repair_vector(h_init)
    n_cur = h_init.size
    best_h = h_init.copy()
    best_score = _c5_value(best_h)

    while True:
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 8.0:
            break
        # allocate time for this resolution round
        round_time = min(remaining * 0.6, 180.0)   # at most 3 minutes per level
        restarts = 3
        time_per_restart = round_time / (restarts + 1e-9)
        round_best_h = best_h
        round_best_score = best_score
        for r in range(restarts):
            if time.time() - start_total > budget_s - 5.0:
                break
            if r == 0:
                start_vec = best_h.copy()
            else:
                rand_vec = rng.random(n_cur)
                start_vec = _repair_vector(rand_vec)
            # Simulated annealing
            sa_time = time_per_restart * 0.7
            gd_time = time_per_restart * 0.3
            h_sa, _ = _simulated_annealing(start_vec, sa_time, rng)
            h_gd, _ = _greedy_descent(h_sa, gd_time, rng)
            score = _c5_value(h_gd)
            if score < round_best_score:
                round_best_score = score
                round_best_h = h_gd.copy()
        # update global best
        if round_best_score < best_score:
            best_score = round_best_score
            best_h = round_best_h.copy()
        # attempt to increase resolution if time permits
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 12.0:
            break
        if n_cur * 2 <= 1000:
            new_n = min(n_cur * 2, 1000)
            x_old = np.linspace(0.0, 2.0, n_cur, endpoint=False)
            x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = _repair_vector(h_up)
            # prepare for next loop
            best_h = h_up.copy()
            best_score = _c5_value(best_h)
            n_cur = new_n
        else:
            break

    best_h = _repair_vector(best_h)
    final_score = _c5_value(best_h)
    return best_h, float(final_score), best_h.size
