import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust sum(v)=target via water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-14:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-14:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _interpolate_solution(src, new_n):
    """Linear interpolation of a step function to a finer grid."""
    m = src.size
    x_src = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_src, src)

def _penalized_objective(h, target):
    """C5 with a quadratic penalty enforcing the integral constraint."""
    penalty = 1e5 * (np.sum(h) - target) ** 2
    return _compute_c5(h) + penalty

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    TARGET = 0.3808  # target bound to beat

    # ------------------------------------------------------------------
    # Initial vector (use supplied parent if present)
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        init_vec = np.array(init_vec, dtype=np.float64).ravel()
        n0 = init_vec.size
    else:
        n0 = 200
        init_vec = np.random.rand(n0)

    target0 = n0 / 2.0
    init_vec = _project_to_constraints(init_vec, target0)

    # ------------------------------------------------------------------
    # Resolution schedule (doubling until time runs out)
    resolutions = [n0, n0 * 2, n0 * 4, n0 * 8]

    best_h = None
    best_score = float("inf")

    def time_left():
        return budget_s - (time.time() - start_time)

    for n in resolutions:
        if time_left() < 5.0:
            break

        # ----- build start vector for this resolution -------------------
        if n == n0:
            h_start = init_vec.copy()
        else:
            src = best_h if best_h is not None else init_vec
            h_start = _interpolate_solution(src, n)
        target_sum = n / 2.0
        h_start = _project_to_constraints(h_start, target_sum)

        # ----- determine how many local optimizations we can afford ------
        # Rough time budget per optimisation (empirical)
        per_opt = 0.12 * (n / n0)  # scale with size
        max_restarts = max(20, int(min(500, time_left() // (per_opt * 1.5))))
        max_iter = 500

        # ----- local optimisation loop ----------------------------------
        for restart in range(max_restarts):
            if time_left() < 1.0:
                break

            # random start: sometimes completely random, otherwise perturb
            if restart % 9 == 0:
                x0 = np.random.rand(n)
            else:
                perturb = np.random.randn(n) * 0.03
                x0 = h_start + perturb
            x0 = _project_to_constraints(x0, target_sum)

            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(
                    _penalized_objective,
                    x0,
                    args=(target_sum,),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={"maxiter": max_iter, "ftol": 1e-9},
                )
            except Exception:
                continue

            h_opt = _project_to_constraints(res.x, target_sum)
            score = _compute_c5(h_opt)

            if score < best_score:
                best_score = float(score)
                best_h = h_opt.copy()
                # early exit if we beat target
                if best_score < TARGET:
                    break

        # ----- small hill‑climb refinement --------------------------------
        if best_h is not None and best_h.size == n:
            h_ref = best_h.copy()
            score_ref = best_score
            for _ in range(800):
                if time_left() < 0.8:
                    break
                i, j = np.random.randint(n, size=2)
                if i == j:
                    continue
                cand = h_ref.copy()
                # simple swap (preserves sum)
                cand[i], cand[j] = cand[j], cand[i]
                # ensure bounds then project
                cand = _project_to_constraints(cand, target_sum)
                cand_score = _compute_c5(cand)
                if cand_score < score_ref:
                    h_ref = cand
                    score_ref = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_h = cand.copy()
                    if cand_score < TARGET:
                        break
            # use refined solution as start for next resolution
            best_h = h_ref.copy()
            best_score = score_ref

        if best_score < TARGET:
            break

    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    if best_h is None:
        best_h = init_vec
        best_score = _compute_c5(best_h)

    # Evaluate with the external helper when available
    final_score = best_score
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            pass

    return best_h.astype(np.float64), float(final_score), int(best_h.size)
