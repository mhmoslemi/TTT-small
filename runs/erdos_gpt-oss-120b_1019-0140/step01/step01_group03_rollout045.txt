def optimize_h(h0, max_iter, step0, beta, s, rng, time_limit, start_time):
    h = h0.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    for it in range(1, max_iter+1):
        # time check
        if time.time() - start_time > time_limit:
            break
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find active shifts
        tol = 1e-12
        # maybe relative tolerance
        idxs = np.where(np.abs(corr - max_corr) <= tol)[0]
        # pick random shift among idxs
        idx = rng.choice(idxs)
        shift = idx - (n - 1)
        # gradient
        grad = compute_gradient(h, shift)
        # momentum
        v = beta * v + (1 - beta) * grad
        # step size
        eta = step0 / np.sqrt(it)  # could be step0/(1+it*decay)
        h_new = h - eta * v
        # project
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        # if improvement
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def optimizer(h_start, max_iter, step0, beta, s, rng, start_time, time_budget):
    # h_start shape (n,)
    n = h_start.size
    h = h_start.copy()
    v = np.zeros_like(h)
    best_local_h = h.copy()
    best_local_score = compute_c5(h)
    for it in range(1, max_iter+1):
        # time check
        if time.time() - start_time > time_budget:
            break
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find active indices within tolerance
        tol = 1e-12
        idxs = np.where(np.abs(corr - max_corr) <= tol)[0]
        idx = rng.choice(idxs)
        shift = idx - (n - 1)
        # gradient
        grad = compute_gradient(h, shift)
        # momentum update
        v = beta * v + (1 - beta) * grad
        # step size schedule
        eta = step0 / np.sqrt(it)
        # update and project
        h_new = h - eta * v
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_local_score:
            best_local_score = score
            best_local_h = h_new.copy()
        h = h_new
    return best_local_h, best_local_score

import time, math
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    # trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search on lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on uniform grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of correlation term for a given integer shift."""
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def thue_morse(n):
    """Generate Thue–Morse binary sequence of length n (float 0/1)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def optimizer(h_start, max_iter, step0, beta, s, rng, start_time, time_budget):
    """Projected gradient descent with momentum on the max-correlation objective."""
    n = h_start.size
    h = h_start.copy()
    v = np.zeros_like(h)  # momentum accumulator
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        if time.time() - start_time > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # pick a shift among those attaining the max (to handle ties)
        tol = 1e-12
        idxs = np.where(np.abs(corr - max_corr) <= tol)[0]
        idx = rng.choice(idxs)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        # momentum update
        v = beta * v + grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * v
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
