import numpy as np, time, math
from scipy.optimize import minimize

def _repair_vector(v):
    # Clip and adjust sum to len(v)/2 exactly.
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _compute_overlaps(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _max_overlap(h):
    return np.max(_compute_overlaps(h))

def _simulated_annealing(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _max_overlap(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    n = h_cur.size
    start = time.time()
    T0 = 0.001
    max_eps = 0.2
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps

import numpy as np

import math
from scipy.optimize import minimize

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _compute_overlaps(h: np.ndarray) -> np.ndarray:
    """Return array of overlap integrals for all shifts."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _max_overlap(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretized step function."""
    return float(np.max(_compute_overlaps(h)))

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simple SA preserving sum and box constraints."""

    T0 = 0.001
    max_eps = 0.2
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(0, n, size=2)
        if i == j:

        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
