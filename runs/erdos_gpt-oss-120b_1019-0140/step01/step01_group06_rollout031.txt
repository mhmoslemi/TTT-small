def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def compute_c5(h):
    """Maximum overlap integral for sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=15):
    """Clip to [0,1] and adjust to meet the integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = (h > 0.0) & (h < 1.0)
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        deficit = target_sum - cur
        if deficit > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        idx = np.where(mask)[0]
        if idx.size:
            h[idx] += deficit / idx.size
            h = np.clip(h, 0.0, 1.0)

    base_h = np.asarray(base_h, dtype=float)

            cur_h = resample_h(parent, n)
        elif best_h is not None and n != best_n:
            cur_h = resample_h(best_h, n)
        else:
            cur_h = rng.random(n)
            cur_h = project_to_feasible(cur_h, n / 2.0)

        cur_c5 = compute_c5(cur_h)

        # ---------- SLSQP local optimization ----------
        max_restarts = 15 if n <= 380 else 8
        for restart in range(max_restarts):
            if time.time() - start > budget_s * 0.90:
                break
            if restart == 0:
                h0 = cur_h.copy()
            else:
                h0 = cur_h + rng.normal(scale=0.04, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_to_feasible(h0, n / 2.0)

            cons = {"type": "eq", "fun": lambda x, target=n / 2.0: np.sum(x) - target}
            bnds = [(0.0, 1.0)] * n
            try:
                res = opt.minimize(
                    lambda x: compute_c5(x),
                    x0=h0,
                    method="SLSQP",
                    bounds=bnds,
                    constraints=cons,
                    options={"maxiter": 800, "ftol": 1e-9, "disp": False},
                )
            except Exception:
                continue
            if not res.success:
                continue
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_to_feasible(h_opt, n / 2.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < cur_c5:
                cur_c5 = c5_opt
                cur_h = h_opt.copy()
            if c5_opt < best_c5:
                best_c5, best_h, best_n = c5_opt, h_opt.copy(), n
            if best_c5 <= 0.3808:
                break

        # ---------- Sub‑gradient descent ----------
        step = 5.0
        decay = 0.9999
        grad_iters = 20000
        it = 0
        while it < grad_iters and time.time() - start < budget_s * 0.96:
            new_h = gradient_step(cur_h, step)
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5:
                cur_c5, cur_h = new_c5, new_h.copy()
                if new_c5 < best_c5:
                    best_c5, best_h, best_n = new_c5, new_h.copy(), n
                if best_c5 <= 0.3808:
                    break
            step *= decay
            it += 1
            if step < 1e-6:
                break

        # ---------- Simulated annealing ----------
        T0 = 0.018
        while time.time() - start < budget_s * 0.99:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta
            c5_cand = compute_c5(cand)
            elapsed = time.time() - start
            T = T0 * (1 - elapsed / (budget_s * 0.99))
            if c5_cand < cur_c5 or rng.random() < np.exp(-(c5_cand - cur_c5) / max(T, 1e-12)):
                cur_h, cur_c5 = cand, c5_cand
                if c5_cand < best_c5:
                    best_c5, best_h, best_n = c5_cand, cand.copy(), n
                if best_c5 <= 0.3808:
                    break

        # store best for possible up‑sampling
        if cur_c5 < best_c5:
            best_c5, best_h, best_n = cur_c5, cur_h.copy(), n
        if best_c5 <= 0.3808:
            break

    # fallback if nothing succeeded
    if best_h is None:
        n = 380
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        best_h = project_to_feasible(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
