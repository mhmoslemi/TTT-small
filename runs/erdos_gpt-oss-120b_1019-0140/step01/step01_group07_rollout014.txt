def optimize_h_slsqp(start_h, target_sum, time_limit, start_time):
    n = start_h.size
    best_h = None
    best_score = float('inf')
    # Determine number of restarts based on time: allow many restarts if time left.
    # Use while loop
    restart = 0
    while time.time() - start_time < time_limit * 0.7:  # use part for SLSQP
        restart += 1
        # perturb start_h
        perturb = np.random.randn(n) * 0.02
        x0 = start_h + perturb
        x0 = _project_to_constraints(x0, target_sum)

        cons = {"type": "eq", "fun": lambda h, tgt=target_sum: np.sum(h) - tgt}
        bounds = [(0.0, 1.0)] * n

        try:
            res = minimize(_compute_c5, x0, method="SLSQP", bounds=bounds, constraints=[cons],
                           options={"maxiter": 200, "ftol": 1e-9, "disp": False})
        except Exception:
            continue
        h_opt = _project_to_constraints(res.x, target_sum)
        score = _compute_c5(h_opt)
        if score < best_score:
            best_score = score
            best_h = h_opt
            # early exit if good
            if best_score < 0.3808:
                break
        # early break if time low
        if time.time() - start_time > time_limit * 0.9:
            break
    # After restarts, hill climb refinement
    time_for_hc = time_limit - (time.time() - start_time)
    if time_for_hc > 0.1 and best_h is not None:
        hc_h, hc_score = hill_climb(best_h, target_sum, time_for_hc, start_time)
        if hc_score < best_score:
            best_h = hc_h
            best_score = hc_score
    return best_h, best_score

def refine_trust_constr(h, target_sum, time_limit, start_time):
    n = h.size
    # create LinearConstraint
    linear_constraint = LinearConstraint(np.ones((1, n)), [target_sum], [target_sum])
    bounds = Bounds(0.0, 1.0)

    def fun_and_grad(x):
        # compute correlation and gradient
        n = x.size
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        # shift functions defined earlier
        term1 = 1.0 - shift(x, lag)
        term2 = shift(x, -lag)

    try:

        return h
    # after optimization, ensure projection onto constraints (should be satisfied)
    # but due to numeric tolerance may be slightly off; project
    final_h = _project_to_constraints(res.x, target_sum)
    return final_h

import numpy as np, time
from scipy.optimize import minimize, Bounds, LinearConstraint

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clamp to [0,1] and enforce sum(v)=target by simple water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    # simple proportional scaling as fallback
    if cur > 0:
        v *= target / cur
    v = np.clip(v, 0.0, 1.0)
    # final correction of any residual error
    err = np.sum(v) - target
    if abs(err) > 1e-9:
        # distribute error on the largest free entries
        free = np.where((v > 0) & (v < 1))[0]
        if free.size:
            delta = -err / free.size
            v[free] = np.clip(v[free] + delta, 0.0, 1.0)
            # renormalize once more
            v *= target / np.sum(v)
    return v

def _grad_subgrad(h):
    """Return (value, subgradient) of max_{k} ∫ h(x)(1−h(x+k))."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    i = np.arange(n)
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    valid1 = (i + lag >= 0) & (i + lag < n)
    term1[valid1] = 1.0 - h[i[valid1] + lag]
    valid2 = (i - lag >= 0) & (i - lag < n)
    term2[valid2] = h[i[valid2] - lag]
    grad = (term1 - term2) * dx
    val = corr[idx] * dx
    return val, grad

def _obj_scalar(x):
    return _grad_subgrad(x)[0]

def _obj_grad(x):
    return _grad_subgrad(x)[1]

def _optimize_slsqp(start, target_sum, time_limit, start_time):
    n = start.size
    best_h = None
    best_val = float("inf")
    # equality constraint: sum == target_sum
    cons = {"type": "eq", "fun": lambda h, tgt=target_sum: np.sum(h) - tgt}
    bounds = [(0.0, 1.0)] * n
    # allocate restarts while time permits
    while time.time() - start_time < time_limit * 0.7:
        pert = np.random.randn(n) * 0.02
        x0 = _project_to_constraints(start + pert, target_sum)
        try:
            res = minimize(_obj_scalar, x0, method="SLSQP", jac=_obj_grad,
                           bounds=bounds, constraints=[cons],
                           options={"maxiter": 250, "ftol": 1e-9, "disp": False})
        except Exception:
            continue
        h = _project_to_constraints(res.x, target_sum)
        val = _compute_c5(h)
        if val < best_val:
            best_val = float(val)
            best_h = h.copy()
            if best_val < 0.3808:
                break
    # local random refinement (hill‑climbing)
    if best_h is not None:
        rem = time_limit - (time.time() - start_time)
        if rem > 0.1:
            best_h, best_val = _hill_climb(best_h, target_sum, rem, start_time)
    return best_h, best_val

def _hill_climb(h, target_sum, time_limit, start_time):
    n = h.size
    best_h = h.copy()
    best_val = _compute_c5(best_h)
    sigma = 0.08
    while time.time() - start_time < time_limit:
        k = max(1, int(0.02 * n))
        idx = np.random.choice(n, k, replace=False)
        new_h = best_h.copy()
        new_h[idx] += np.random.randn(k) * sigma
        new_h = _project_to_constraints(new_h, target_sum)
        val = _compute_c5(new_h)
        if val < best_val:
            best_h, best_val = new_h, val
            sigma = max(sigma * 0.99, 0.001)
        else:
            if np.random.rand() < 0.005:
                sigma = min(sigma * 1.05, 0.12)
    return best_h, best_val

def _refine_trust(h, target_sum, time_limit, start_time):
    n = h.size
    lin = LinearConstraint(np.ones((1, n)), [target_sum], [target_sum])
    bnd = Bounds(0.0, 1.0)
    def fun_grad(x):
        return _grad_subgrad(x)
    try:
        res = minimize(fun_grad, h, method="trust-constr", jac=True,
                       constraints=[lin], bounds=bnd,
                       options={"maxiter": 500, "verbose": 0})
    except Exception:
        return h
    refined = _project_to_constraints(res.x, target_sum)
    return refined

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # initial vector
    init = globals().get("initial_h_values", None)
    if init is not None:
        h0 = np.array(init, dtype=np.float64).ravel()
    else:
        n0 = 200
        h0 = np.random.rand(n0)
    n0 = h0.size
    target0 = n0 / 2.0
    h0 = _project_to_constraints(h0, target0)

    # resolution schedule (keep under ~2000 points)
    n_vals = [n0]
    if budget_s > 300 and n0 * 2 <= 2000:
        n_vals.append(n0 * 2)
    if budget_s > 600 and n0 * 4 <= 2000:
        n_vals.append(n0 * 4)

    best_h = None
    best_val = float("inf")

    for n in n_vals:
        # time check
        if time.time() - start_time > budget_s * 0.97:
            break
        # upsample / interpolate
        if n == n0:
            start_h = h0.copy()
        else:
            src = best_h if best_h is not None else h0
            xs = np.arange(src.size) * (2.0 / src.size)
            xn = np.arange(n) * (2.0 / n)
            start_h = np.interp(xn, xs, src)
        target = n / 2.0
        start_h = _project_to_constraints(start_h, target)

        # allocate time slice for this resolution
        remain = budget_s - (time.time() - start_time)
        if remain < 5:
            break
        # main optimization
        h_opt, val_opt = _optimize_slsqp(start_h, target, remain * 0.7, start_time)
        if h_opt is not None and val_opt < best_val:
            best_h, best_val = h_opt.copy(), float(val_opt)
        # optional trust‑constr refinement if time permits
        if best_h is not None:
            rem2 = budget_s - (time.time() - start_time)
            if rem2 > 2:
                refined = _refine_trust(best_h, target, rem2 * 0.3, start_time)
                refined_val = _compute_c5(refined)
                if refined_val < best_val:
                    best_h, best_val = refined, float(refined_val)
        # early exit if target already met
        if best_val < 0.3808:
            break

    if best_h is None:
        best_h = h0
        best_val = _compute_c5(best_h)

    # final evaluation using provided evaluator if any
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_val)
    else:
        final_score = float(best_val)

    return best_h.astype(np.float64), final_score, int(best_h.size)
