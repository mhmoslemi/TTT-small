import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust one entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute evenly (rare)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _gradient_worst_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Subgradient of the worst‑lag overlap w.r.t. h."""
    n = h.shape[0]
    if lag >= 0:
        # shift to the right by lag (zero padding on left)
        plus = np.concatenate([h[lag:], np.zeros(lag)])
        # shift to the left by lag (zero padding on right)
        minus = np.concatenate([np.zeros(lag), h[: n - lag]])
    else:
        k = -lag
        plus = np.concatenate([np.zeros(k), h[: n - k]])
        minus = np.concatenate([h[k:], np.zeros(k)])
    return 1.0 - (plus + minus)

def _guided_move(h: np.ndarray, grad: np.ndarray, step: float, rng: np.random.Generator) -> tuple:
    """Propose a feasible move using the gradient sign (mass transfer)."""
    n = h.shape[0]
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]

    # try a directed transfer from a positive‑gradient entry to a negative‑gradient entry
    if pos.size and neg.size:
        i = pos[np.argmax(grad[pos])]   # decrease this entry
        j = neg[np.argmin(grad[neg])]   # increase this entry
        hi, hj = h[i], h[j]
        delta_max_i = hi                 # how much we can subtract from i
        delta_max_j = 1.0 - hj           # how much we can add to j
        delta_max = min(delta_max_i, delta_max_j, step)
        if delta_max > 1e-12:
            delta = rng.random() * delta_max
            h_new = h.copy()
            h_new[i] = hi - delta
            h_new[j] = hj + delta
            return h_new, True

    # fallback: random two‑index transfer respecting bounds and step
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h, False
    hi, hj = h[i], h[j]
    delta_min = max(-hi, -(1.0 - hj))
    delta_max = min(1.0 - hi, hj)
    low = max(delta_min, -step)
    high = min(delta_max, step)
    if low >= high:
        return h, False
    delta = rng.random() * (high - low) + low
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new, True

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to lower the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- choose discretisation size ------------------------------------------------
    n_points = 200                     # moderate size for flexibility
    target_sum = n_points / 2.0

    # ---- initialise feasible candidate --------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        # interpolate to desired resolution
        m = init.shape[0]
        x_src = np.arange(m) * (2.0 / m)
        x_tgt = np.arange(n_points) * (2.0 / n_points)
        current = np.interp(x_tgt, x_src, init)
    except NameError:
        # no parent construction: start from a simple half‑ones pattern
        current = np.zeros(n_points, dtype=float)
        half = n_points // 2
        current[:half] = 1.0
        if n_points % 2:
            current[half] = 0.5
        rng.shuffle(current)

    current = np.clip(current, 0.0, 1.0)
    current = _repair_sum(current, target_sum)

    best_h = current.copy()
    best_c5 = _compute_c5(best_h)

    # ---- optimisation schedules ----------------------------------------------------
    max_step = 0.4
    min_step = 0.001
    max_temp = 0.01
    min_temp = 1e-6

    # ---- main loop ----------------------------------------------------------------
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s:
            break

        frac = elapsed / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # evaluate current worst lag
        corr = np.correlate(current, 1.0 - current, mode="full")
        max_idx = int(np.argmax(corr))
        cur_c5 = float(corr[max_idx] * (2.0 / n_points))
        lag = max_idx - (n_points - 1)

        # gradient w.r.t. this lag
        grad = _gradient_worst_lag(current, lag)

        # propose a move
        h_new, moved = _guided_move(current, grad, step, rng)
        if not moved:
            continue

        # evaluate candidate
        new_corr = np.correlate(h_new, 1.0 - h_new, mode="full")
        new_c5 = float(np.max(new_corr) * (2.0 / n_points))

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
            current = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

    # final repair (tiny numerical drift possible)
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n_points
