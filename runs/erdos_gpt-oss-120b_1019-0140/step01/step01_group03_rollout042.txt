def smooth_obj_and_grad(h, beta):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # compute log-sum-exp for smooth max
    beta_corr = beta * corr
    maxb = np.max(beta_corr)
    exp_shifted = np.exp(beta_corr - maxb)
    sum_exp = np.sum(exp_shifted)
    smooth_max = (maxb + np.log(sum_exp)) / beta
    # compute weights
    w = exp_shifted / sum_exp
    # compute weighted sum of left+right contributions (agg)
    agg = np.zeros_like(h)
    n_shifts = 2 * n - 1
    # index offset = n-1
    offset = n - 1
    for idx in range(n_shifts):
        weight = w[idx]
        s = idx - offset
        if s >= 0:
            if weight == 0:
                continue
            # left contribution
            agg[s:] += weight * h[:n - s]
            # right contribution
            agg[:n - s] += weight * h[s:]
        else:
            t = -s
            agg[:n - t] += weight * h[t:]
            agg[t:] += weight * h[:n - t]
    # gradient = dx * (1 - agg)
    grad = dx * (1.0 - agg)
    return smooth_max, grad

import time
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of a 1‑D array to a new length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x : lo≤x_i≤hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(l):
        return np.clip(v - l, lo, hi).sum()

    # ensure brackets contain the target sum
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Exact discretized C5 for a given h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def smooth_obj_and_grad(h, beta):
    """Smooth approximation of max correlation and its gradient."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx          # length 2n-1
    beta_corr = beta * corr
    maxb = np.max(beta_corr)
    exp_shift = np.exp(beta_corr - maxb)
    sum_exp = np.sum(exp_shift)
    smooth_max = (maxb + np.log(sum_exp)) / beta

    w = exp_shift / sum_exp                                   # weights, sum to 1
    agg = np.zeros_like(h)                                    # weighted sum of left+right terms
    offset = n - 1
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - offset
        if s >= 0:
            # left contribution (h shifted right by s)
            agg[s:] += weight * h[:n - s]
            # right contribution (h shifted left by s)
            agg[:n - s] += weight * h[s:]
        else:
            t = -s
            agg[:n - t] += weight * h[t:]
            agg[t:] += weight * h[:n - t]
    grad = dx * (1.0 - agg)                                   # ∇_h smooth_max
    return smooth_max, grad

def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                      dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Binary vector of length n with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def optimize_h(h0, budget_s, start_time,
               total_iters=12000,
               beta_start=2.0, beta_end=30.0,
               step0=5.0):
    """Projected (smooth) gradient descent with a β‑continuation schedule."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)

    half_iters = max(1, total_iters // 2)
    for it in range(1, total_iters + 1):
        if time.time() - start_time > budget_s - 0.5:
            break

        # β ramps up linearly over the first half of the schedule
        if it <= half_iters:
            beta = beta_start + (beta_end - beta_start) * (it - 1) / (half_iters - 1)
        else:
            beta = beta_end

        _, grad = smooth_obj_and_grad(h, beta)
        step = step0 / np.sqrt(it)
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, n / 2.0)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score
