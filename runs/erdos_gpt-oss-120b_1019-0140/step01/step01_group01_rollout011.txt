def project(y, target):
    # Clip to [0,1]
    x = np.clip(y, 0, 1)
    cur_sum = x.sum()
    diff = target - cur_sum
    # Distribute diff uniformly across free variables (not at bound)
    # while diff not small:
    free = (x > 0) & (x < 1)
    # if no free but diff not zero, then can't satisfy maybe due to rounding; adjust small.
    # Add diff/num_free to free entries and clip again.
    while abs(diff) > 1e-9 and free.any():
        add = diff / free.sum()
        x[free] += add
        # clip again
        np.clip(x, 0, 1, out=x)
        # recompute diff
        cur_sum = x.sum()
        diff = target - cur_sum
        free = (x > 0) & (x < 1)
    # If diff remains, apply small uniform correction may slightly violate bounds but okay.
    if abs(diff) > 1e-9:
        # tiny adjust across all entries
        x += diff/x.size
        np.clip(x, 0, 1, out=x)
    return x

    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5
    n_points = 200
    target_sum = n_points/2.0

    # create initial population
    popsize = 30
    pop = np.empty((popsize, n_points))
    for i in range(popsize):
        # random uniform then project
        candidate = rng.rand(n_points)
        candidate = project(candidate, target_sum)
        pop[i] = candidate

    # evaluate
    fitness = np.array([compute_c5(ind) for ind in pop])

    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    # DE parameters
    F = 0.5
    CR = 0.9

    generation = 0
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        generation += 1

        # produce new population
        for i in range(popsize):
            # select three distinct indices a,b,c ≠ i
            indices = list(range(popsize))
            indices.remove(i)
            a,b,c = rng.choice(indices, size=3, replace=False)

def project_onto_bounds_and_sum(v, target, lower=0.0, upper=1.0):
    x = np.clip(v, lower, upper)
    cur = x.sum()
    diff = target - cur
    # tolerance
    tol = 1e-9
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        free_count = free.sum()
        if free_count == 0:
            # cannot distribute further, break; adjust rest uniformly may violate bounds
            break
        delta = diff / free_count
        x[free] += delta
        # clip again
        np.clip(x, lower, upper, out=x)
        cur = x.sum()
        diff = target - cur
    # final fallback: adjust uniformly all entries if still off
    if abs(diff) > tol:
        x += diff / x.size
        np.clip(x, lower, upper, out=x)
    return x

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the exact sum target."""
    x = np.clip(v, 0.0, 1.0)
    diff = target - x.sum()
    tol = 1e-12
    # iterative water‑filling on free variables
    for _ in range(1000):
        if abs(diff) <= tol:
            break
        free = (x > 0.0 + 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(free.sum())
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        np.clip(x, 0.0, 1.0, out=x)
        diff = target - x.sum()
    if abs(diff) > tol:
        x += diff / x.size
        np.clip(x, 0.0, 1.0, out=x)
    return x

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # ----- problem parameters -----
    n_points = 200
    target_sum = n_points / 2.0

    # ----- population initialization (Differential Evolution) -----
    popsize = 30
    pop = np.empty((popsize, n_points), dtype=float)
    for i in range(popsize):
        cand = rng.rand(n_points)
        pop[i] = _project_to_bounds_and_sum(cand, target_sum)

    # use parent construction if available
    if "initial_h_values" in globals():
        try:
            init = np.array(initial_h_values, dtype=float)
            if init.size != n_points:
                # linear interpolation to required length
                x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
                init = np.interp(x_new, x_old, init)
            pop[0] = _project_to_bounds_and_sum(init, target_sum)
        except Exception:
            pass

    fitness = np.empty(popsize, dtype=float)
    for i in range(popsize):
        fitness[i] = _compute_c5(pop[i])

    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

    # ----- DE parameters -----
    F = 0.7
    CR = 0.9
    n = n_points

    # ----- Differential Evolution main loop -----
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        for i in range(popsize):
            if time.time() - start >= max_time:
                break
            # choose distinct a,b,c ≠ i
            idxs = list(range(popsize))
            idxs.remove(i)

            jrand = rng.randint(n)
            for j in range(n):

                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]

            trial = _project_to_bounds_and_sum(trial, target_sum)
            trial_f = _compute_c5(trial)

            if trial_f <= fitness[i]:
                pop[i] = trial
                fitness[i] = trial_f
                if trial_f < best_f:
                    best_f = trial_f
                    best_h = trial.copy()

    # ----- Local gradient descent on the worst shift -----
    while True:
        if time.time() - start >= max_time:
            break
        # worst shift for current best
        corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_shift = int(np.argmax(corr_full)) - (n - 1)
