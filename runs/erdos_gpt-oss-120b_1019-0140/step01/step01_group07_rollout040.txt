import numpy as np

from scipy.optimize import minimize

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    # water-filling similar as before but better: adjust by scaling
    # We will first adjust via proportional scaling, then clip
    if cur != 0:
        v = v * (target / cur)
    v = np.clip(v, 0.0, 1.0)
    # If still not exact due to clipping, fix by adding/subtracting small amount
    cur = np.sum(v)
    diff = target - cur
    if abs(diff) > 1e-9:
        # distribute diff uniformly across entries not at bounds
        free = np.where((v > 0) & (v < 1))[0]
        if free.size > 0:
            v[free] += diff / free.size
            v = np.clip(v, 0.0, 1.0)
    return v

def _to_binary(v):
    n = v.size
    target = n // 2
    bin_arr = (v > 0.5).astype(np.float64)
    cur = int(bin_arr.sum())
    if cur < target:
        zeros = np.where(bin_arr == 0)[0]
        add_idx = np.random.choice(zeros, size=target - cur, replace=False)
        bin_arr[add_idx] = 1.0
    elif cur > target:
        ones = np.where(bin_arr == 1)[0]
        sub_idx = np.random.choice(ones, size=cur - target, replace=False)
        bin_arr[sub_idx] = 0.0
    return bin_arr

def _hill_climb_binary(start_h, target_sum, time_limit, start_time, target_score):
    n = start_h.size
    h = start_h.copy()
    # lists of indices
    ones = list(np.where(h == 1)[0].tolist())
    zeros = list(np.where(h == 0)[0].tolist())
    best_h = h.copy()
    best_score = _compute_c5(best_h)
    cur_score = best_score
    no_improve = 0
    iter_count = 0
    # loop
    while True:
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        # random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        # perform swap in copy
        h[i] = 0.0
        h[j] = 1.0
        # evaluate
        new_score = _compute_c5(h)
        if new_score < cur_score - 1e-12:
            # accept improvement
            cur_score = new_score
            # update lists
            # replace i with j in ones list, replace j with i in zeros list
            idx_i = ones.index(i)  # O(len) but small
            idx_j = zeros.index(j)
            ones[idx_i] = j
            zeros[idx_j] = i
            # maybe update best
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score < target_score:
                    break
            no_improve = 0
        else:
            # reject: revert swap
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        iter_count += 1
        # occasional restart
        if no_improve >= 2000:
            # random binary reinit
            idx_one = np.random.choice(n, size=int(target_sum), replace=False)
            h[:] = 0.0
            h[idx_one] = 1.0
            ones = list(idx_one.tolist())
            zeros = [i for i in range(n) if i not in ones]
            cur_score = _compute_c5(h)
            no_improve = 0
    return best_h, best_score

def _to_binary(v):
    """Convert a feasible vector to a binary one with exactly n/2 ones."""
    n = v.size
    target = n // 2
    bin_arr = (v > 0.5).astype(np.float64)
    cur = int(bin_arr.sum())
    if cur < target:
        zeros = np.where(bin_arr == 0)[0]
        add = np.random.choice(zeros, size=target - cur, replace=False)
        bin_arr[add] = 1.0
    elif cur > target:
        ones = np.where(bin_arr == 1)[0]
        sub = np.random.choice(ones, size=cur - target, replace=False)
        bin_arr[sub] = 0.0
    return bin_arr

def _hill_climb_binary(start_h, target_sum, time_limit, start_time, target_score):
    """Simple hill‑climbing on binary vectors (swap a 1 with a 0)."""
    h = start_h.copy()
    n = h.size
    # maintain index lists for fast random access
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    best_h = h.copy()
    best_score = _compute_c5(best_h)
    cur_score = best_score
    no_improve = 0

    while True:
        if time.time() - start_time > time_limit:
            break
        # pick a random 1 and a random 0
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        # try the swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = _compute_c5(h)
        if new_score < cur_score - 1e-12:
            # accept improvement
            cur_score = new_score
            # update index lists
            idx_i = ones.index(i)
            idx_j = zeros.index(j)
            ones[idx_i] = j
            zeros[idx_j] = i
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score < target_score:
                    break
            no_improve = 0
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

        if no_improve >= 2000:
            # random restart
            idx_one = np.random.choice(n, size=int(target_sum), replace=False)
            h[:] = 0.0
            h[idx_one] = 1.0
            ones = list(idx_one.tolist())
            zeros = [i for i in range(n) if i not in ones]
            cur_score = _compute_c5(h)
            no_improve = 0

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        init_vec = np.array(init_vec, dtype=np.float64).ravel()
    else:
        init_vec = np.random.rand(200)

    n = init_vec.size
    target_sum = n / 2.0
    init_vec = _project_to_constraints(init_vec, target_sum)

    # ------------------------------------------------------------------
    # Binary hill‑climbing (only if n is even)
    best_h = None
    best_score = float("inf")
    if n % 2 == 0:
        h_bin = _to_binary(init_vec)
        # allocate about 80 % of the budget for the discrete phase
        binary_deadline = start_time + budget_s * 0.8
        hill_h, hill_score = _hill_climb_binary(
            h_bin, target_sum, binary_deadline, start_time, target_score=0.3808
        )
        best_h = hill_h.copy()
        best_score = hill_score

    # ------------------------------------------------------------------
    # If binary failed to produce a feasible vector, fall back to the
    # continuous start vector.
    if best_h is None:
        best_h = init_vec.copy()
        best_score = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Continuous refinement with many SLSQP restarts (remaining budget)
    time_remaining = start_time + budget_s - time.time()
    max_restarts = 80
    maxiter = 300
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s * 0.97:
            break
        # random perturbation around current best
        perturb = np.random.randn(n) * 0.02
        x0 = best_h + perturb
        x0 = _project_to_constraints(x0, target_sum)

        cons = {"type": "eq", "fun": lambda h, tgt=target_sum: np.sum(h) - tgt}
        bounds = [(0.0, 1.0)] * n

        try:
            res = minimize(
                _compute_c5,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[cons],
                options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue

        h_opt = _project_to_constraints(res.x, target_sum)
        score = _compute_c5(h_opt)
        if score < best_score - 1e-12:
            best_score = float(score)
            best_h = h_opt.copy()
            if best_score < 0.3808:
                break

    # ------------------------------------------------------------------
    # Final evaluation (use the provided evaluator if present)
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_h.size)
