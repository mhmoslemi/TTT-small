import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Maximum overlap integral for sampled step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=20):
    """Clip to [0,1] and adjust to hit target_sum by distributing residual uniformly."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # add/subtract the remaining amount to the element with most slack
        if cur < target_sum:
            idx = np.argmax(1.0 - h)
            h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = np.clip(h[idx] - (cur - target_sum), 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    """Linear interpolation of base_h to length n_target and feasibility projection."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def triangular_initial(n):
    """Triangular function h(x)=min(x,2-x) sampled at n points; then projected."""
    dx = 2.0 / n
    # sample at cell centers for better integral approximation
    x = (np.arange(n) + 0.5) * dx
    h = np.minimum(x, 2.0 - x)
    h = project_to_feasible(h, n / 2.0)
    return h

def hill_climb(h0, rng, max_iter, start_time, budget_s):
    """Simple improve‑only hill climbing preserving the sum constraint."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.99:
            break
        if rng.random() < 0.5:
            # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # transfer a small amount between two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # randomly pick direction
            if rng.random() < 0.5:
                # move from i to j
                max_transfer = min(h[i], 1.0 - h[j])
                if max_transfer <= 0:
                    continue
                delta = rng.random() * max_transfer * 0.2
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
            else:
                # move from j to i
                max_transfer = min(h[j], 1.0 - h[i])
                if max_transfer <= 0:
                    continue
                delta = rng.random() * max_transfer * 0.2
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
        # evaluate
        new_c5 = compute_c5(h_new)
        if new_c5 < c5:
            h = h_new
            c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        # occasional re‑projection to fix floating drift
        if (it & 63) == 0:  # every 64 iterations
            h = project_to_feasible(h, target)
            c5 = compute_c5(h)
    return best_h, best_c5

def refine_local(h_start, n, start_time, budget_s):
    """One round of SLSQP refinement from a given start point."""
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            fun=lambda x: compute_c5(x),
            x0=h_start,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": 500, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return h_start, compute_c5(h_start)
    if not res.success:
        return h_start, compute_c5(h_start)
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, target)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low Erdős overlap constant C₅."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # candidate resolutions (small enough for many evaluations)
    candidate_ns = [95, 190, 380]

    for n in candidate_ns:
        target_sum = n / 2.0
        # Determine number of restarts and iterations per restart based on n
        if n <= 100:
            max_restarts = 30
            iter_per = 2000
        elif n <= 200:
            max_restarts = 15
            iter_per = 2500
        else:
            max_restarts = 8
            iter_per = 3000

        for restart in range(max_restarts):
            if time.time() - start_time > budget_s * 0.97:
                break

            # Initial vector
            if base_h is not None and restart == 0:
                h0 = resample_h(base_h, n)
            elif restart == 1:
                # deterministic triangular start
                h0 = triangular_initial(n)
            else:
                h0 = rng.random(n)
                h0 = project_to_feasible(h0, target_sum)

            # Local hill climbing
            h_opt, c5_opt = hill_climb(h0, rng, iter_per, start_time, budget_s)

            # Keep the best found so far
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n

            # If we already beat the target, we can continue a few more iterations
            if best_c5 <= 0.3808:
                # allow a couple more restarts for possible improvement
                pass

        # After all restarts for this resolution, perform a final SLSQP refinement
        if best_h is not None and best_n == n:
            h_ref, c5_ref = refine_local(best_h, n, start_time, budget_s)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n

    # Fallback in the unlikely case no solution was discovered
    if best_h is None:
        # simple uniform solution
        best_n = 95
        best_h = np.full(best_n, 0.5)
        best_h = project_to_feasible(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    # Final safety projection and recompute exact bound
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
