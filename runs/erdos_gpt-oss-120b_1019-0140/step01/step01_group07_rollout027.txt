def project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    # If cur == target: return
    if abs(cur - target) <= 1e-12:
        return v
    # We'll adjust iteratively:
    # Use vector al = v.copy()
    diff = target - cur
    # Determine free indices (where we can increase if diff>0 else decrease)
    # We'll loop while diff nonzero:
    while True:
        if diff > 0:
            free = np.where(v < 1.0)[0]
            if free.size == 0:
                break
            # max possible increase:
            caps = 1.0 - v[free]
            # total capacity:
            cap_sum = np.sum(caps)
            if cap_sum < diff - 1e-12:
                # Not enough capacity -> just fill all to 1, continue?
                v[free] = 1.0
                diff -= cap_sum
                continue
            # Distribute diff proportionally to caps (e.g., fill uniformly over free)
            # We'll add diff * caps / cap_sum to each free element.
            inc = diff * caps / cap_sum
            v[free] += inc
            diff = target - np.sum(v)
            break
        else: # diff < 0 need to decrease
            free = np.where(v > 0.0)[0]
            if free.size == 0:
                break
            caps = v[free] # amount we can reduce
            cap_sum = np.sum(caps)
            if cap_sum < -diff - 1e-12:
                v[free] = 0.0
                diff += cap_sum
                continue
            dec = -diff * caps / cap_sum
            v[free] -= dec
            diff = target - np.sum(v)
            break
    # Ensure bounds again
    v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment:
    cur = np.sum(v)
    if abs(cur - target) > 1e-8:
        # small scaling fix
        v *= target / cur
        v = np.clip(v, 0.0, 1.0)
    return v
```

def de_optimize(h_init, n, budget_left, target_sum):
    # Determine max_iter based on budget_left
    # Approx each DE iteration runs popsize evaluations
    # popsize = 15 -> 15 evaluations per iteration
    # Suppose each evaluation costs O(n) ~ n * constant. For n up to 760, each evaluation maybe 0.0002 s. So we can allocate about 200 iterations.
    max_iter = 200
    # Use callback to early stop if score < target_bound or time out
    start_local = time.time()
    best_local_score = compute_c5(h_init) # maybe
    best_local_h = h_init

    def callback(xk, convergence):
        nonlocal best_local_score, best_local_h, start_local
        # xk is best vector found so far (raw), need to project
        h_proj = project_to_constraints(xk, target_sum)
        score = compute_c5(h_proj)
        if score < best_local_score:
            best_local_score = score
            best_local_h = h_proj.copy()
        if time.time() - start_local > budget_left * 0.95:
            return True
        if best_local_score < target_bound:
            return True
        return False

    result = differential_evolution(
        lambda v: compute_c5(project_to_constraints(v, target_sum)),
        bounds,
        strategy='best1bin',
        maxiter=max_iter,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        polish=False,
        init=pop,
        callback=callback,
        updating='deferred',
        disp=False,

    h_best = project_to_constraints(result.x, target_sum)

def _de_optimize(h_initial, n, max_iter, time_limit, target_bound, seed):
    # start timer
    start = time.time()
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h = None
    best_score = float('inf')
    # callback function
    def _callback(xk, convergence):
        nonlocal best_h, best_score, start
        h = _project_to_constraints(xk, target_sum)
        score = _compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        if time.time() - start > time_limit * 0.95:
            return True
        if score < target_bound:
            return True
        return False
    # objective
    def _obj(v):
        h = _project_to_constraints(v, target_sum)
        return _compute_c5(h)
    # run DE
    from scipy.optimize import differential_evolution
    # Use a small popsize
    popsize = 5  # ensures popsize*n individuals
    # Use seed to create reproducible
    result = differential_evolution(
        _obj,
        bounds,
        strategy='best1bin',
        maxiter=max_iter,
        popsize=popsize,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        polish=False,
        callback=_callback,
        disp=False,
        updating='deferred',
    )
    # After DE, result.x may give some solution; compare
    h_res = _project_to_constraints(result.x, target_sum)
    score_res = _compute_c5(h_res)
    if score_res < best_score:
        best_score = score_res
        best_h = h_res.copy()
    if best_h is None:
        best_h = h_initial
        best_score = _compute_c5(best_h)
    return best_h, best_score

def _local_refine(h_start, n, maxiter=200):
    target_sum = n / 2.0
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_compute_c5, h_start, method="SLSQP", bounds=bounds,
                       constraints=[cons], options={"maxiter": maxiter, "ftol": 1e-9, "disp": False})
        h_opt = _project_to_constraints(res.x, target_sum)
    except Exception:
        h_opt = _project_to_constraints(h_start, target_sum)
    score = _compute_c5(h_opt)
    return h_opt, score

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    # Determine direction
    if cur < target:
        # Need to increase some elements not at 1
        deficit = target - cur
        free = np.where(v < 1.0)[0]

        # Allocate proportionally to remaining capacity

            v[free] = 1.0
