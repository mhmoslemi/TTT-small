**Compute C5**:

```
def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)
```

def local_refine(h, n, rng):
    # define objective points
    def obj(x):
        return compute_c5(x)
    # linear equality constraint on sum
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n/2.0}
    # bounds
    bnds = [(0.0, 1.0)] * n
    # Run SLSQP
    try:
        res = opt.minimize(obj, h, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 300, 'ftol': 1e-9})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_to_feasible(h_opt, n/2.0)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h, compute_c5(h)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # Find interior points not at bounds
        interior = np.where((h > 0.0) & (h < 1.0))[0]
        if interior.size == 0:

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(x, target_sum, tol=1e-8, max_iter=10):
    """Clip to [0,1] and adjust to hit the required sum."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(x)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        interior = np.where((x > 0.0) & (x < 1.0))[0]
        if interior.size == 0:
            break
        delta = diff / interior.size
        x[interior] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction if needed
    cur = np.sum(x)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(x)  # most room to increase
        x[i] = np.clip(x[i] + (target_sum - cur), 0.0, 1.0)
    return x

def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def hill_climb_binary(initial_h, max_iters, rng, start_time, time_limit):
    """Simulated‑annealing style hill‑climb on binary vectors."""
    h = initial_h.astype(float)
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    init_temp = 0.02
    n = h.size
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        T = max(init_temp * (1.0 - it / max_iters), 1e-9)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_val = compute_c5(h)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_val

def local_refine(h_start, n):
    """Polish a solution with SLSQP."""
    def obj(x):
        return compute_c5(x)
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            fun=obj,
            x0=h_start,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": 300, "ftol": 1e-9, "disp": False},
        )
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_to_feasible(h_opt, n / 2.0)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h_start, compute_c5(h_start)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    time_limit = start_time + budget_s * 0.98  # safety margin
    rng = np.random.default_rng(seed)

    # Global parent construction, if any.
    base_h_global = globals().get("initial_h_values")
    if base_h_global is not None:
        base_h_global = np.asarray(base_h_global, dtype=float)

    candidate_ns = [190, 380, 760]  # even sizes for half‑mass constraint
    # Weight towards cheaper resolutions
    probs = np.array([0.5, 0.35, 0.15])

    best_h = None
    best_val = np.inf
    best_n = None

    while time.time() < time_limit:
        n = rng.choice(candidate_ns, p=probs)

        # Build an initial binary vector.
        if base_h_global is not None and base_h_global.shape[0] == n:
            h_float = base_h_global.copy()
        elif base_h_global is not None:
            h_float = resample_h(base_h_global, n)
        else:

        # Threshold to binary (exactly n/2 ones)

    if best_h is None:

        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
        best_val = compute_c5(best_h)

    # Final safety projection
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_val = compute_c5(best_h)

    return best_h, best_val, best_n
