def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

        try:
            cand = binary_from(np.array(initial_h_values, dtype=float))
            c5 = compute_c5(cand)
            if c5 < best_c5:
                best_c5, best_h = c5, cand.copy()
        except Exception:
            pass

    # ----- random binary generator ----------------------------------------
    def random_binary():
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, target_sum, replace=False)
        h[ones] = 1.0
        return h

    # start from best found or fresh random
    if best_h is not None:
        current_h = best_h.copy()
        current_c5 = compute_c5(current_h)
    else:
        current_h = random_binary()
        current_c5 = compute_c5(current_h)
        best_h, best_c5 = current_h.copy(), current_c5

    # ----- SA parameters -------------------------------------------------
    T_start = 2e-2
    T_end = 1e-6
    restart_limit = 6000   # moves without global improvement before restart
    moves_since_best = 0

    # ----- main optimisation loop ----------------------------------------
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        T = T_start * (1.0 - progress) + T_end * progress

        # worst shift for current solution
        corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
        max_idx = int(np.argmax(corr_full))
        k_shift = max_idx - (n - 1)

        # masks for targeted swap
        if k_shift >= 0:
            idx = np.arange(n - k_shift)
            h_i = current_h[idx]
            h_j = current_h[idx + k_shift]
        else:
            ks = -k_shift
            idx = np.arange(ks, n)
            h_i = current_h[idx]
            h_j = current_h[idx - ks]

        mask1 = idx[(h_i == 1) & (h_j == 0)]
        mask2 = idx[(h_i == 0) & (h_j == 1)]

        candidates = []
        cand_scores = []

        # targeted swap if possible
        if mask1.size and mask2.size and rng.rand() < 0.8:
            i_abs = rng.choice(mask1)
            j_abs = rng.choice(mask2)
            cand = current_h.copy()
            cand[i_abs] = 0.0
            cand[j_abs] = 1.0
            candidates.append(cand)
            cand_scores.append(compute_c5(cand))

        # a few random swaps
        ones_idx = np.flatnonzero(current_h == 1)
        zeros_idx = np.flatnonzero(current_h == 0)
        if ones_idx.size and zeros_idx.size:
            for _ in range(3):
                i_abs = rng.choice(ones_idx)
                j_abs = rng.choice(zeros_idx)
                cand = current_h.copy()
                cand[i_abs] = 0.0
                cand[j_abs] = 1.0
                candidates.append(cand)
                cand_scores.append(compute_c5(cand))

        # select best candidate among generated
        best_local_idx = int(np.argmin(cand_scores))
        cand_h = candidates[best_local_idx]
        cand_c5 = cand_scores[best_local_idx]

        # acceptance
        if cand_c5 < current_c5 or rng.rand() < np.exp(-(cand_c5 - current_c5) / (T + 1e-12)):
            current_h = cand_h
            current_c5 = cand_c5

        # update global best
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h.copy()
            moves_since_best = 0
        else:
            moves_since_best += 1

        # restart if stuck
        if moves_since_best >= restart_limit:
            current_h = random_binary()
            current_c5 = compute_c5(current_h)
            moves_since_best = 0

    # ----- final correction (ensure exact mass) -------------------------
    # best_h should be binary, sum may already equal target_sum.
    # If a tiny deviation occurs, fix it.
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            zeros = np.flatnonzero(best_h == 0)
            if zeros.size:
                take = rng.choice(zeros, int(diff), replace=False)
                best_h[take] = 1.0
        else:
            ones = np.flatnonzero(best_h == 1)
            if ones.size:
                take = rng.choice(ones, int(-diff), replace=False)
                best_h[take] = 0.0

    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n)
