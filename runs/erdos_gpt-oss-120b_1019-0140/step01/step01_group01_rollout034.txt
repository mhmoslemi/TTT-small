def gradient_move(h, step, target_sum, rng):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_idx = int(np.argmax(corr))
    k_shift = k_idx - (n - 1)
    idx = np.arange(n)
    g = np.zeros_like(h)
    # forward part
    plus = idx + k_shift
    valid_plus = (plus >= 0) & (plus < n)
    g[valid_plus] += 1.0 - h[plus[valid_plus]]
    # backward part
    minus = idx - k_shift
    valid_minus = (minus >= 0) & (minus < n)
    g[valid_minus] -= h[minus[valid_minus]]
    # ensure sum(g)=0? Not necessary as we project later.
    # also project to sum zero direction to preserve total mass: subtract mean
    g = g - g.mean()
    candidate = h - step * g
    np.clip(candidate, 0.0, 1.0, out=candidate)
    candidate = adjust_sum_to_target(candidate, target_sum, rng)
    return candidate

def pairwise_move(h, step, rng):
    n = h.shape[0]
    i, j = rng.randint(n, size=2)
    if i == j: return h.copy() # no change
    hi, hj = h[i], h[j]
    low = max(-hi, hj - 1.0)  # delta can be negative (transfer from i to j)
    high = min(1.0 - hi, hj)
    if low >= high:
        return h.copy()
    delta = (rng.rand() * 2.0 - 1.0) * step
    delta = np.clip(delta, low, high)
    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    # sum unchanged automatically
    # clip again and adjust sum (should be sum correct)
    np.clip(cand, 0.0, 1.0, out=cand)
    return cand

Note: low for delta means hi must stay >=0 and <=1 and hj must stay >=0 and <=1 after transfer: h[i] -> hi + delta, h[j] -> hj - delta; change of mass balances.

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Project h onto {0 ≤ h_i ≤ 1, sum(h)=target}.
    Small residuals are spread uniformly; final clipping ensures bounds.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def gradient_move(h: np.ndarray, step: float, target_sum: float, rng: np.random.RandomState) -> np.ndarray:
    """Follow subgradient of the worst shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_idx = int(np.argmax(corr))
    k_shift = k_idx - (n - 1)  # lag in [-n+1, n-1]

    idx = np.arange(n)
    g = np.zeros_like(h)

    plus = idx + k_shift
    valid_plus = (plus >= 0) & (plus < n)
    g[valid_plus] += 1.0 - h[plus[valid_plus]]

    minus = idx - k_shift
    valid_minus = (minus >= 0) & (minus < n)
    g[valid_minus] -= h[minus[valid_minus]]

    # enforce zero-sum direction (mass preservation)
    g -= g.mean()

    cand = h - step * g
    np.clip(cand, 0.0, 1.0, out=cand)
    cand = adjust_sum_to_target(cand, target_sum, rng)
    return cand


def pairwise_move(h: np.ndarray, step: float, rng: np.random.RandomState) -> np.ndarray:
    """Exchange δ mass between two coordinates while respecting bounds."""
    n = h.shape[0]
    i, j = rng.randint(n, size=2)
    if i == j:
        return h.copy()
    hi, hj = h[i], h[j]
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if low >= high:
        return h.copy()
    delta = (rng.rand() * 2.0 - 1.0) * step
    delta = np.clip(delta, low, high)
    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    # sum unchanged, just clip to be safe
    np.clip(cand, 0.0, 1.0, out=cand)
    return cand


def random_perturb(h: np.ndarray, step: float, target_sum: float, rng: np.random.RandomState) -> np.ndarray:
    """Add small uniform noise to a random subset of entries."""
    n = h.shape[0]
    k = max(1, int(0.05 * n))
    idx = rng.choice(n, size=k, replace=False)
    cand = h.copy()
    noise = (rng.rand(k) * 2.0 - 1.0) * step
    cand[idx] += noise
    np.clip(cand, 0.0, 1.0, out=cand)
    cand = adjust_sum_to_target(cand, target_sum, rng)
    return cand


def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D step function to a finer grid."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old_h)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Heuristic optimisation for the Erdős minimum overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.3  # safety margin

    # ----- initialise -------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n0 = init_h.size
        n_points = max(200, n0)   # work on at least 200 samples
        if n_points != n0:
            h0 = upsample_h(init_h, n_points)
        else:
            h0 = init_h.copy()
    except Exception:
        n_points = 200
        h0 = rng.rand(n_points)

    target_sum = n_points / 2.0
    # normalise to correct total mass
    h0 = h0 / max(h0.sum(), 1e-12) * target_sum
    h0 = adjust_sum_to_target(h0, target_sum, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # parameters for annealing
    n_restarts = 4
    T0, T_end = 0.12, 1e-6
    step0, step_end = 0.35, 1e-4
    threshold = 0.3808

    elapsed = time.time() - start
    remaining = max_time - elapsed
    if remaining <= 0:
        return (best_h, float(best_c5), n_points)

    time_per_restart = remaining / n_restarts

    for rs in range(n_restarts):
        # initialise this restart
        if rs == 0:
            cur_h = best_h.copy()
        else:
            # perturb best solution slightly
            cur_h = best_h.copy()
            cur_h += rng.randn(n_points) * 0.04
            np.clip(cur_h, 0.0, 1.0, out=cur_h)
            cur_h = adjust_sum_to_target(cur_h, target_sum, rng)

        cur_c5 = compute_c5(cur_h)

        restart_start = time.time()
        while True:
            now = time.time()
            if now - restart_start > time_per_restart * 0.96:
                break

            # linearly decay step and temperature inside this restart
            prog = min((now - restart_start) / time_per_restart, 1.0)
            T = T0 * (1.0 - prog) + T_end * prog
            step = step0 * (1.0 - prog) + step_end * prog

            move_type = rng.rand()
            if move_type < 0.4:
                cand = gradient_move(cur_h, step, target_sum, rng)
            elif move_type < 0.8:
                cand = pairwise_move(cur_h, step, rng)
            else:
                cand = random_perturb(cur_h, step, target_sum, rng)

            cand_c5 = compute_c5(cand)

            # Metropolis acceptance
            if cand_c5 < cur_c5:
                accept = True
            else:
                prob = np.exp(-(cand_c5 - cur_c5) / T) if T > 0 else 0.0
                accept = rng.rand() < prob

            if accept:
                cur_h = cand
                cur_c5 = cand_c5
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = cand.copy()
                    if best_c5 <= threshold:
                        # achievement; abort all loops
                        return (best_h, float(best_c5), n_points)

        # end of this restart – keep best so far
        # continue with next restart if time permits

    # final projection and optional binary rounding
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    # try a pure binary version if n_points even
    if n_points % 2 == 0:
        M = n_points // 2
        idx_desc = np.argsort(-best_h)
        bin_h = np.zeros_like(best_h)
        bin_h[idx_desc[:M]] = 1.0
        bin_c5 = compute_c5(bin_h)
        if bin_c5 < final_c5:
            best_h = bin_h
            final_c5 = bin_c5

    return (best_h, float(final_c5), n_points)
