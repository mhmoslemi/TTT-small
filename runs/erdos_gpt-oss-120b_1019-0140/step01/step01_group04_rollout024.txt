def _gradient_of_max(v):
    n = len(v)
    corr = np.correlate(v, 1 - v, mode='full')
    i_max = np.argmax(corr)
    k = i_max - (n - 1)  # shift
    g = np.zeros_like(v)
    if k >= 0:
        offset = k
        # linear term: add 1 for indices 0..n-1 - offset
        g[: n - offset] += 1.0
        # subtract v[j+offset] on same range
        g[: n - offset] -= v[offset:]
        # subtract v[j - offset] on indices offset..n-1
        g[offset:] -= v[: n - offset]
    else:
        offset = -k
        # linear term: add 1 for indices offset..n-1
        g[offset:] += 1.0
        # subtract v[j - offset] for j >= offset
        g[offset:] -= v[: n - offset]
        # subtract v[j + offset] for j <= n-1 - offset
        g[: n - offset] -= v[offset:]
    return g

import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project vector onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Maximum overlap integral for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _grad_and_score(v):
    """Gradient of the worst‐case shift and its score."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode="full")
    i_max = int(np.argmax(corr))
    max_corr = corr[i_max]
    k = i_max - (n - 1)  # shift causing the maximum
    g = np.zeros_like(v)
    if k >= 0:
        offset = k
        # linear term contribution
        g[: n - offset] += 1.0
        # product term contributions
        g[: n - offset] -= v[offset:]
        g[offset:] -= v[: n - offset]
    else:
        offset = -k
        g[offset:] += 1.0
        g[offset:] -= v[: n - offset]
        g[: n - offset] -= v[offset:]
    score = max_corr * (2.0 / n)
    return g, score

def _symmetrize(v):
    """Make vector symmetric around the centre."""
    return (v + v[::-1]) / 2.0

def _gradient_descent(v0, time_limit, sym=False):
    """Projected gradient descent on the max‑overlap objective."""
    start = time.time()
    v = v0.copy()
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(v)
    step0 = 0.5
    it = 0
    while time.time() - start < time_limit:
        it += 1
        g, cur_score = _grad_and_score(v)
        step = step0 / np.sqrt(it)
        if step < 1e-6:
            break
        cand = v - step * g
        cand = _repair_vector(cand, n)
        if sym:
            cand = _symmetrize(cand)
            cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_v = cand.copy()
        # acceptance rule: accept improving moves always, occasionally accept worse
        if cand_score <= cur_score or np.random.rand() < 0.1:
            v = cand
    return best_v, best_score

def _random_feasible(n):
    """Generate a random feasible vector."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _local_search(v0, time_limit):
    """Simple pairwise exchange hill‑climbing."""
    start = time.time()
    n = len(v0)
    best_v = v0.copy()
    best_score = _compute_c5(best_v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score = score
            best_v = cand
    return best_v, best_score

def _upsample_vector(v, n_target):
    """Linear interpolation to a finer grid."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)
