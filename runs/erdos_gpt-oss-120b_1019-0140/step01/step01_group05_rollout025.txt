def _binary_repair(v, rng):
    # v is float with values 0 or 1 possibly plus some fractional; convert to binary
    # threshold 0.5: v_bin = (v >= 0.5).astype(int)
    v_bin = (v >= 0.5).astype(np.int8)
    target = v_bin.size // 2
    current = v_bin.sum()
    if current > target:
        # need to turn some ones to zeros
        idx_one = np.flatnonzero(v_bin == 1)
        to_flip = rng.choice(idx_one, size=current - target, replace=False)
        v_bin[to_flip] = 0
    elif current < target:
        idx_zero = np.flatnonzero(v_bin == 0)
        to_flip = rng.choice(idx_zero, size=target - current, replace=False)
        v_bin[to_flip] = 1
    return v_bin.astype(np.float64)

def _c5_value(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _binary_repair(v, rng):
    # ensures v binary and sum = n/2 exactly.
    # v is float array.
    n = v.size
    target = n // 2
    # if n even, target is integer; if n odd, target = n//2? but we ensure n even.
    v_bin = (v >= 0.5).astype(np.int8)
    current = int(v_bin.sum())
    if current > target:
        idx_one = np.flatnonzero(v_bin == 1)
        to_flip = rng.choice(idx_one, size=current - target, replace=False)
        v_bin[to_flip] = 0
    elif current < target:
        idx_zero = np.flatnonzero(v_bin == 0)
        to_flip = rng.choice(idx_zero, size=target - current, replace=False)
        v_bin[to_flip] = 1
    return v_bin.astype(np.float64)

def _initialize_binary_vector(n, rng):
    h = np.zeros(n, dtype=np.float64)
    ones_idx = rng.choice(n, size=n//2, replace=False)
    h[ones_idx] = 1.0
    return h

import numpy as np
import time
import math

def _c5_value(h: np.ndarray) -> float:
    """Compute the C5 value for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _binary_repair(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Clip to {0,1} and enforce exactly n/2 ones (n must be even)."""
    n = v.size
    target = n // 2
    vb = (v >= 0.5).astype(np.int8)
    cur = int(vb.sum())
    if cur > target:
        idx = np.flatnonzero(vb == 1)
        to_flip = rng.choice(idx, size=cur - target, replace=False)
        vb[to_flip] = 0
    elif cur < target:
        idx = np.flatnonzero(vb == 0)
        to_flip = rng.choice(idx, size=target - cur, replace=False)
        vb[to_flip] = 1
    return vb.astype(np.float64)

def _initialize_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def _simulated_annealing_binary(h_start: np.ndarray, time_limit: float,
                                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """SA on binary vectors preserving the exact sum constraint."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.05  # initial temperature

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = 1.0 - elapsed / time_limit
        T = T0 * frac

        # pick a 1 and a 0 to swap
        ones = np.flatnonzero(h_cur > 0.5)
        zeros = np.flatnonzero(h_cur <= 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h_new = h_cur.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0

        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _hill_climb(h_start: np.ndarray, time_limit: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Deterministic hill climbing (only accepting improvements)."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.flatnonzero(h_cur > 0.5)
        zeros = np.flatnonzero(h_cur <= 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h_cur.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = _c5_value(h_new)
        if new_score < cur_score - 1e-12:
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    total_budget = float(budget_s)

    # Choose an even resolution; if a parent is present, use its length (adjusted to even)
    try:
        init_len = len(initial_h_values)
        n_base = init_len if init_len % 2 == 0 else init_len + 1
    except NameError:
        n_base = 200
    n_base = max(100, n_base)
    if n_base % 2 != 0:
        n_base += 1

    # Phase 1: binary SA at base resolution
    phase1_time = total_budget * 0.5
    h0 = _initialize_binary_vector(n_base, rng)
    best_h, best_score = _simulated_annealing_binary(h0, phase1_time, rng)

    elapsed = time.time() - start_total
    remaining = total_budget - elapsed

    # Phase 2: increase resolution if enough time remains
    if remaining > total_budget * 0.25:
        n2 = n_base * 2
        # up‑sample current best and re‑binary‑repair
        x_old = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_up = _binary_repair(h_interp, rng)

        phase2_time = min(total_budget * 0.3, remaining * 0.9)
        h2, score2 = _simulated_annealing_binary(h_up, phase2_time, rng)
        if score2 < best_score:
            best_h, best_score = h2, score2
        remaining = total_budget - (time.time() - start_total)

    # Phase 3: final greedy polishing
    if remaining > 2.0:
        h_final, score_final = _hill_climb(best_h, remaining, rng)
        if score_final < best_score:
            best_h, best_score = h_final, score_final

    # Ensure feasibility and compute final exact C5
    best_h = _binary_repair(best_h, rng)
    final_score = _c5_value(best_h)

    return best_h, float(final_score), best_h.size
