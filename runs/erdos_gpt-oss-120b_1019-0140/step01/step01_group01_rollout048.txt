def adjust_sum_to_target(h, target, rng):
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # iterate a limited number of attempts
    max_iters = 1000
    for _ in range(max_iters):
        if abs(diff) < 1e-12:
            break
        # choose random index
        idx = rng.randint(h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0:
                h[idx] -= dec
                diff += dec
    if abs(diff) > 1e-12:
        # distribute remaining diff uniformly and clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # after clipping, may still have tiny error; recalc
        # adjust again recursively but limited
        residual = target - h.sum()
        if abs(residual) > 1e-12:
            # fix by adding residual to a random free spot
            free = np.where((h > 0) & (h < 1))[0]
            if free.size == 0:
                # all at bounds, just set one element arbitrarily within bounds
                idx = rng.randint(h.size)
                h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)

                idx = rng.choice(free)
                h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
            # final clipping
            np.clip(h, 0.0, 1.0, out=h)
    return h

import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Adjust h in-place so that sum(h) == target while keeping 0≤h_i≤1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
    if abs(diff) > 1e-12:
        # distribute residual uniformly and clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        residual = target - h.sum()
        if abs(residual) > 1e-12:
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size == 0:
                idx = rng.randint(h.size)
                h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
            else:
                idx = rng.choice(free)
                h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
            np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """

        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_old, base)
        h0 = np.clip(h0, 0.0, 1.0)
        h = adjust_sum_to_target(h0, target_sum, rng)
    except Exception:
        # random start
        h = rng.rand(n_points)
        h = h / h.sum() * target_sum
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum_to_target(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # annealing schedule parameters
    T0, T_end = 1e-2, 1e-6
    step0, step_end = 0.3, 1e-5

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        T = T0 * (1.0 - progress) + T_end * progress
        step = step0 * (1.0 - progress) + step_end * progress

        move_type = rng.rand()
        # ---------- gradient move ----------
        if move_type < 0.5:
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n_points - 1)

            g = np.zeros_like(current_h)
            idx = np.arange(n_points)

            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n_points)
            g[valid_plus] += (1.0 - current_h[plus[valid_plus]])

            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n_points)
            g[valid_minus] -= current_h[minus[valid_minus]]

            # project to hyperplane sum(g)=0
            g -= g.mean()

            cand = current_h - step * g
            np.clip(cand, 0.0, 1.0, out=cand)
            cand = adjust_sum_to_target(cand, target_sum, rng)

        # ---------- biased pairwise exchange ----------
        elif move_type < 0.75:
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n_points - 1)

            g = np.zeros_like(current_h)
            idx = np.arange(n_points)
            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n_points)
            g[valid_plus] += (1.0 - current_h[plus[valid_plus]])
            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n_points)
            g[valid_minus] -= current_h[minus[valid_minus]]

            pos_idxs = np.where(g > 1e-12)[0]
            neg_idxs = np.where(g < -1e-12)[0]

            if pos_idxs.size == 0 or neg_idxs.size == 0:
                # fallback to a simple random exchange
                i, j = rng.randint(n_points, size=2)
                hi, hj = current_h[i], current_h[j]
                low = max(-hi, hj - 1.0)
                high = min(1.0 - hi, hj)
                if low >= high:
                    continue
                delta = (rng.rand() * 2.0 - 1.0) * step
                delta = np.clip(delta, low, high)
                cand = current_h.copy()
                cand[i] = hi + delta
                cand[j] = hj - delta
            else:
                i = rng.choice(pos_idxs)
                j = rng.choice(neg_idxs)
                hi, hj = current_h[i], current_h[j]
                max_decrease = hi
                max_increase = 1.0 - hj
                max_delta = min(max_decrease, max_increase, step)
                if max_delta <= 0.0:
                    continue
                delta = rng.rand() * max_delta
                cand = current_h.copy()
                cand[i] = hi - delta
                cand[j] = hj + delta

        # ---------- plain random pairwise exchange ----------
        else:
            i, j = rng.randint(n_points, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = np.clip(delta, low, high)
            cand = current_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta

        cand_c5 = compute_c5(cand)

        # acceptance
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = cand
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

    # final sanity check
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
