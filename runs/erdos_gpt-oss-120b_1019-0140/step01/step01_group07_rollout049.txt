def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    # use np.correlate, order reversed? Should be identical
    # But np.correlate returns convolution: sum_i a[i]*b[i+k] for k...
    # With 'full' we get length 2n-1
    # correlation = np.correlate(h, (1-h), mode='full')
    # max returns max correlation (including negative shifts)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust to satisfy sum(v)=target (water-filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    # Simple iterative scaling with clipping
    for _ in range(10):
        if cur > target:
            # need to reduce sum -> scale down
            ratio = target / cur
            v *= ratio
        else:
            # need to increase sum -> distribute to free capacity
            need = target - cur
            free = 1.0 - v
            total_free = free.sum()
            if total_free > 0:
                v += need * free / total_free
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        if abs(cur - target) < 1e-12:
            break
    # final renorm if still off
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
