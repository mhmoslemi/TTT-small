import time, numpy as np
def _compute_c5(h):
    n = h.shape[0]; dx = 2.0/n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h, target):
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute evenly, clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # recalc recursively
        return _repair_sum(h, target)
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def crossover(p1, p2, rng):
    n = p1.shape[0]
    if n <= 1:
        return p1.copy()

    child = np.empty_like(p1)
    child[:cut] = p1[:cut]
    child[cut:] = p2[cut:]
    return child

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Compute the discrete C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """
    Adjust a single entry (recursively if needed) so that sum(h) == target.
    Values stay in [0,1].
    """
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0.0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return _repair_sum(h, target)
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _tournament_select(pop, fitness, rng, tournament_size):
    """Return index of the best individual among a random subset."""
    size = len(pop)
    sz = min(tournament_size, size)

    best = idxs[0]
    best_fit = fitness[best]
    for i in idxs[1:]:
        if fitness[i] < best_fit:
            best = i
            best_fit = fitness[i]
    return best
