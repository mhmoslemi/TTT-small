def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_gradient(h, shift):
    n = h.shape[0]
    grad = np.ones_like(h)
    if shift >= 0:
        grad[:n-shift] -= h[shift:]
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        grad[s:] -= h[:n-s]
        grad[:n-s] -= h[s:]
    return grad

def projected_gradient_descent(h0, max_iter=5000, step0=0.5, time_limit=None, start_time=None):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter+1):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break

        idx = np.argmax(corr)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand brackets if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient(h, shift):
    """Gradient of the active correlation term for integer shift."""
    n = h.size
    grad = np.ones_like(h)
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad


def projected_gradient_descent(h0, max_iter=20000, step0=0.5,
                               time_limit=None, start_time=None):
    """Projected gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    h = np.zeros(n, dtype=float)

def thue_morse(n):
    """Generate Thue–Morse binary sequence of length n (requires n power of two)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
