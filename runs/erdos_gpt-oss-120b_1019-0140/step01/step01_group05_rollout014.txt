def _repair_vector(v):
    # Clip
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    # Might not meet exactly due to clipping: adjust remaining difference by distributing tiny amount
    s2 = v.sum()
    diff = target - s2
    if abs(diff) > 1e-12:
        # distribute diff equally over all entries that are not at the bounds or all?
        # A simple approach: add diff/v.size to each entry then clip again.
        v += diff / v.size
        np.clip(v, 0.0, 1.0, out=v)
        # final scaling maybe close enough
        # after that, correct sum by scaling again
        s3 = v.sum()
        if abs(s3 - target) > 1e-12:
            v *= target / s3
            np.clip(v,0.0,1.0,out=v)
    return v

def _c5_value(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v


def _c5_value(h: np.ndarray) -> float:
    """Exact C5 value for a discretised h."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _targeted_move(h: np.ndarray, k_offset: int, rng: np.random.Generator, max_eps: float):
    """Increase h[i] and h[i+k] while decreasing a donor to keep the sum fixed."""
    N = h.size
    if k_offset == 0:
        return None
    # viable i range for the chosen shift
    if k_offset > 0:
        i_start, i_end = 0, N - 1 - k_offset
    else:
        i_start, i_end = -k_offset, N - 1
    if i_end < i_start:
        return None
    i = rng.integers(i_start, i_end + 1)
    j = i + k_offset
    # try a few donors
    for _ in range(5):
        l = rng.integers(0, N)
        if l == i or l == j:
            continue
        up_i = 1.0 - h[i]
        up_j = 1.0 - h[j]
        down_l = h[l] / 2.0
        eps_max = min(up_i, up_j, down_l, max_eps)
        if eps_max <= 0.0:
            continue
        eps = rng.random() * eps_max
        if eps < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] += eps
        h_new[j] += eps
        h_new[l] -= 2.0 * eps
        if 0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0 and 0.0 <= h_new[l] <= 1.0:
            return h_new
    return None


def _random_move(h: np.ndarray, rng: np.random.Generator, max_eps: float):
    """Transfer epsilon from one coordinate to another while respecting bounds."""
    N = h.size
    i, j = rng.integers(0, N, size=2)
    while i == j:
        j = rng.integers(0, N)
    hi, hj = h[i], h[j]
    # amount we can move from i to j
    up = min(1.0 - hj, hi)
    if up <= 0.0:
        return None
    eps = rng.random() * up
    if eps < 1e-12:
        return None
    h_new = h.copy()
    h_new[i] = hi - eps
    h_new[j] = hj + eps
    return h_new


def _simulated_annealing_targeted(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """SA that focuses on the worst (largest) correlation shift."""
    h_cur = h_start.copy()
    N = h_cur.size
    corr_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    cur_score = float(np.max(corr_cur) * (2.0 / N))
    best_h = h_cur.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.05              # initial temperature
    eps_start = 0.20
    eps_end = 0.02

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        max_eps = eps_end + (eps_start - eps_end) * frac

        # decide move type
        if rng.random() < 0.7:
            # target move based on a few worst shifts
            if corr_cur.size > 3:
                top_idxs = np.argpartition(corr_cur, -3)[-3:]
            else:
                top_idxs = np.arange(corr_cur.size)
            k_idx = int(rng.choice(top_idxs))
            k_offset = k_idx - (N - 1)
            h_new = _targeted_move(h_cur, k_offset, rng, max_eps)
            if h_new is None:
                h_new = _random_move(h_cur, rng, max_eps)
                if h_new is None:
                    continue
        else:
            h_new = _random_move(h_cur, rng, max_eps)
            if h_new is None:
                continue

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_score = float(np.max(corr_new) * (2.0 / N))
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            corr_cur = corr_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # obtain initial vector
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        # fallback: uniform 0.5 which satisfies the integral exactly
        N0 = 200
        h0 = np.full(N0, 0.5, dtype=np.float64)
    h0 = _repair_vector(h0)
    n0 = h0.size

    start_total = time.time()
    phase1_time = budget_s * 0.60
    best_h, best_score = _simulated_annealing_targeted(h0, phase1_time, rng)

    elapsed = time.time() - start_total
    remaining = budget_s - elapsed

    # Phase 2: increase resolution if time permits
    if remaining > budget_s * 0.15:
        n1 = min(n0 * 2, 800)
        if n1 != n0:
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n1, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = _repair_vector(h_up)
            phase2_time = remaining * 0.90
            h2, s2 = _simulated_annealing_targeted(h_up, phase2_time, rng)
            if s2 < best_score:
                best_h, best_score = h2, s2

    # final repair and exact evaluation
    best_h = _repair_vector(best_h)
    final_score = _c5_value(best_h)
    return best_h, float(final_score), best_h.size
