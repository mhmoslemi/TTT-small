def local_optimize_binary(h, deadline):
    n = h.shape[0]
    dx = 2.0 / n
    # compute autocorrelation a = np.correlate(h, h, mode='full')
    a = np.correlate(h, h, mode='full')
    # compute current min
    cur_min = int(a.min())
    # identify zeros and ones positions
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    # loop
    while True:
        best_new_min = None
        best_swap_i0 = None
        best_swap_i1 = None

        for i0 in zeros:
            for i1 in ones:
                new_min = swap_min_a(i0, i1, h, a, cur_min)

    n = h.shape[0]

    delta0 = 1
    delta1 = -1

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def binary_from_real(arr: np.ndarray) -> np.ndarray:
    """Convert a real-valued vector to a binary one with exactly half ones."""
    n = arr.size
    target = n // 2
    idx = np.argsort(arr)[::-1]          # descending
    bin_vec = np.zeros(n, dtype=int)
    bin_vec[idx[:target]] = 1
    return bin_vec

def swap_min_a(i0: int, i1: int, h: np.ndarray, a: np.ndarray, cur_min: int) -> int | None:
    """
    Evaluate the new minimum autocorrelation after swapping
    a zero at i0 with a one at i1.
    Returns the new minimum if it is strictly larger than cur_min,
    otherwise None.
    """
    n = h.shape[0]
    offset = n - 1
    # delta0 = +1, delta1 = -1
    new_min = None
    for ell in range(-n + 1, n):
        idx = ell + offset
        val = int(a[idx])
        # contributions of i0 (0 -> 1)
        ip = i0 + ell
        if 0 <= ip < n:
            val += h[ip]                # delta0 * h[ip]
        im = i0 - ell
        if 0 <= im < n:
            val += h[im]                # delta0 * h[im]
        # contributions of i1 (1 -> 0)
        jp = i1 + ell
        if 0 <= jp < n:
            val -= h[jp]                # delta1 * h[jp]
        jm = i1 - ell
        if 0 <= jm < n:
            val -= h[jm]                # delta1 * h[jm]
        # self terms
        if ell == 0:
            val += 2
        elif ell == i1 - i0:
            val -= 1
        elif ell == i0 - i1:
            val -= 1
        # track minimum
        if new_min is None or val < new_min:
            new_min = val
        # early exit: cannot beat cur_min
        if new_min <= cur_min:
            return None
    return new_min  # guaranteed > cur_min

def local_optimize_binary(h: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy local optimization on a binary vector.
    Repeatedly applies the best improving swap (if any) until no improvement.
    Returns the improved vector and its C5 value.
    """
    n = h.shape[0]
    dx = 2.0 / n
    a = np.correlate(h, h, mode="full")          # integer autocorrelation
    cur_min = int(a.min())
    while True:
        # locate zeros and ones
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_new_min = None
        best_i0 = best_i1 = None
        # examine all possible swaps
        for i0 in zeros:
            for i1 in ones:
                new_min = swap_min_a(i0, i1, h, a, cur_min)
                if new_min is not None:
                    if best_new_min is None or new_min > best_new_min:
                        best_new_min = new_min
                        best_i0, best_i1 = i0, i1
        if best_i0 is None:                     # no improving swap
            break
        # perform the chosen swap
        h[best_i0] = 1
        h[best_i1] = 0
        # recompute autocorrelation and current minimum
        a = np.correlate(h, h, mode="full")
        cur_min = int(a.min())
        if time.time() > deadline:
            break
    c5 = 1.0 - dx * cur_min
    return h, c5

def generate_pattern_h(n: int) -> np.ndarray:
    """
    Simple deterministic pattern (1100 repeated) with exactly n/2 ones.
    """
    pattern = np.tile([1, 1, 0, 0], (n // 4) + 1)[:n]
    # Ensure exact half ones; adjust if needed
    target = n // 2
    current = int(pattern.sum())
    if current > target:
        # turn some ones to zero
        excess = current - target
        ones_idx = np.where(pattern == 1)[0]
        pattern[ones_idx[:excess]] = 0
    elif current < target:
        deficit = target - current
        zeros_idx = np.where(pattern == 0)[0]
        pattern[zeros_idx[:deficit]] = 1
    return pattern.astype(int)

            h_pert = best_h.copy()
            # perform a few random swaps regardless of improvement
            for _ in range(max(1, target_ones // 10)):
                zeros = np.where(h_pert == 0)[0]
                ones = np.where(h_pert == 1)[0]
                if zeros.size == 0 or ones.size == 0:
                    break
                i0 = rng.choice(zeros)
                i1 = rng.choice(ones)
                h_pert[i0] = 1
                h_pert[i1] = 0
            h_opt2, c5_opt2 = local_optimize_binary(h_pert, deadline)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                if best_c5 <= 0.3808:
                    break

    # Final sanity check
    if best_h is None:
        # fallback: uniform split
        best_h = np.zeros(n_points, dtype=int)
        best_h[:target_ones] = 1
    final_c5 = compute_c5(best_h.astype(float))

    return (best_h.astype(float), float(final_c5), n_points)
