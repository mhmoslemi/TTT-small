    while time.time() - start < time_limit:

import time
import numpy as np
from scipy.optimize import minimize

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if still slightly off
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = v.shape[0]
    dx = 2.0 / n
    size = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=size)
    fb = np.fft.rfft(1.0 - v, n=size)
    corr = np.fft.irfft(fv * np.conj(fb), n=size)[: 2 * n - 1]
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _simulated_anneal(v0, time_limit, max_step=0.2):
    """Simple SA on the feasible simplex (sum fixed, bounds [0,1])."""
    start = time.time()
    n = len(v0)
    cur_v = v0.copy()
    cur_score = _compute_c5(cur_v)
    best_v, best_score = cur_v.copy(), cur_score

    T_start, T_end = 1.0, 1e-4
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T_start * (T_end / T_start) ** frac

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # pick direction
        if np.random.rand() < 0.5:
            max_delta = min(max_step, 1.0 - cur_v[i], cur_v[j])
            if max_delta <= 1e-12:
                continue
            delta = np.random.rand() * max_delta
            cand = cur_v.copy()
            cand[i] += delta
            cand[j] -= delta
        else:
            max_delta = min(max_step, cur_v[i], 1.0 - cur_v[j])
            if max_delta <= 1e-12:
                continue
            delta = np.random.rand() * max_delta
            cand = cur_v.copy()
            cand[i] -= delta
            cand[j] += delta

        cand_score = _compute_c5(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur_v, cur_score = cand, cand_score
            if cand_score < best_score:
                best_v, best_score = cand.copy(), cand_score
    return best_v, best_score
