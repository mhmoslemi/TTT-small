def hill_climb(h0, n, rng, time_limit):
    h = h0.copy()
    best = h.copy()
    best_val = compute_c5(best)
    cur_val = best_val
    start = time.time()
    # parameters
    step = 0.2  # initial max delta
    # track non-improve count
    no_improve = 0
    # while time budget not exceeded
    while time.time() - start < time_limit:
        # adapt step: e.g., step = max(1e-4, step * 0.9999)
        # random mutation: pick two indices i, j diff
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 2 * step  # uniform [-step, step]
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # clip and project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new, n/2.0)
        new_val = compute_c5(h_new)
        if new_val < cur_val:
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best = h_new.copy()
                best_val = new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            # maybe accept with small probability
            # acceptance probability = exp(-(new_val-cur_val)/temp)
            # temperature decays
            # compute temperature based on elapsed fraction
            frac = (time.time() - start) / time_limit
            temp = (1 - frac) * 0.001  # small temperature
            if rng.random() < np.exp(-(new_val - cur_val) / (temp + 1e-12)):
                # accept uphill move
                h = h_new
                cur_val = new_val
                # no improve count increases?
            else:
                no_improve += 1
        # reduce step gradually
        step *= 0.9999
        if no_improve > 5000:
            # random restart near best
            h = best.copy()
            cur_val = best_val
            no_improve = 0
            step = 0.2
    return best, best_val

Implementation details for project_to_feasible:

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final adjustment if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust one element
        # find element with most slack
        slack_up = 1.0 - h
        slack_down = h
        # we can pick index where slack_up is max if need increase
        if cur < target_sum:
            i = np.argmax(slack_up)
            h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
        else:
            i = np.argmax(slack_down)
            h[i] = np.clip(h[i] - (cur - target_sum), 0.0, 1.0)
    return h

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x) (1 - h(x+k)) dx for sampled h."""
    n = h.shape[0]
    # np.correlate with mode='full' computes discrete correlation with zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    """Project vector h onto the box [0,1]^n and the affine hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        if cur < target_sum:
            # need to increase somewhere
            slack = 1.0 - h
            i = np.argmax(slack)
            h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
        else:
            # need to decrease somewhere
            slack = h
            i = np.argmax(slack)
            h[i] = np.clip(h[i] - (cur - target_sum), 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def hill_climb(initial_h, rng, time_limit):
    """Simple stochastic hill‑climbing with occasional uphill moves."""
    n = initial_h.shape[0]
    best_h = initial_h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    start = time.time()
    # parameters controlling exploration
    step = 0.2               # initial magnitude of delta
    no_improve = 0
    # main loop
    while True:
        now = time.time()
        if now - start >= time_limit:
            break

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = (rng.random() - 0.5) * 2.0 * step  # uniform in [-step, step]
        trial = cur_h.copy()
        trial[i] += delta
        trial[j] -= delta
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_to_feasible(trial, n / 2.0)

        trial_val = compute_c5(trial)

        if trial_val < cur_val:
            cur_h, cur_val = trial, trial_val
            if trial_val < best_val:
                best_h, best_val = trial.copy(), trial_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            # occasional uphill acceptance (simulated annealing style)
            elapsed = now - start
            frac = elapsed / time_limit
            temperature = max(1e-5, (1.0 - frac) * 5e-4)  # decays to near zero
            prob = np.exp(-(trial_val - cur_val) / (temperature + 1e-12))
            if rng.random() < prob:
                cur_h, cur_val = trial, trial_val
            no_improve += 1

        # gradually shrink step size
        step *= 0.9999
        if step < 1e-5:
            step = 1e-5

        # if stuck for a while, restart from the best found so far
        if no_improve > 5000:
            cur_h = best_h.copy()
            cur_val = best_val
            no_improve = 0
            step = 0.2

    return best_h, best_val


def slsqp_refine(h0, rng, time_limit):
    """A short SLSQP refinement; respects the time budget."""
    n = h0.shape[0]
    # Allocate a small slice of the remaining budget
    start = time.time()
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            fun=lambda x: compute_c5(x),
            x0=h0,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": 300, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return h0
    elapsed = time.time() - start
    if elapsed > time_limit:
        return h0
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Candidate sample counts – keep them small for fast evaluation.
    candidate_ns = [95, 190, 380]

    # Grab parent construction if supplied.
    parent = globals().get("initial_h_values", None)

    best_h = None
    best_val = float("inf")
    best_n = None

    # Iterate over resolutions, giving each a fair share of the budget.
    for idx, n in enumerate(candidate_ns):
        # Break if overall budget is exhausted.
        if time.time() - start_time > budget_s * 0.97:
            break

        # Build an initial feasible vector for this n.
        if parent is not None:
            h0 = resample_h(parent, n)
        else:
            h0 = rng.random(n)
            h0 = project_to_feasible(h0, n / 2.0)

        # Allocate time slice for this resolution.
        remaining = budget_s - (time.time() - start_time)
        # Reserve a small fraction for final safety work.
        slice_time = max(5.0, remaining * 0.6 / (len(candidate_ns) - idx))

        # Perform stochastic hill climbing.
        h_best, val_best = hill_climb(h0, rng, slice_time * 0.85)

        # Optional quick SLSQP polish – use any leftover slice time.
        leftover = slice_time - (time.time() - start_time)
        if leftover > 2.0:
            h_best = slsqp_refine(h_best, rng, leftover)

        val_best = compute_c5(h_best)

        if val_best < best_val:
            best_h, best_val, best_n = h_best.copy(), val_best, n

        # Early exit if we already beat the target.
        if best_val <= 0.3808:
            break

    # Fallback: if nothing succeeded, use a uniform function.
    if best_h is None:
        best_n = candidate_ns[0]
        best_h = np.full(best_n, 0.5)
        best_val = compute_c5(best_h)

    # Final projection to guarantee feasibility.
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_val = compute_c5(best_h)

    return best_h, best_val, best_n
