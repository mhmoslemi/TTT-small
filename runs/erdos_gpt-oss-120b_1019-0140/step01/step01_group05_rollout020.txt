def compute_c5_value(h):
    N = h.shape[0]
    dx = 2.0 / N
    b = 1.0 - h
    b_rev = b[::-1]
    L = 1 << ((2 * N - 1).bit_length())  # next power of two for speed?
    # robust: use L = 2*N
    L = 2 * N
    # But to avoid aliasing, need L >= len(a) + len(b_rev) -1 = N+N-1=2N-1 <= L.
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(H * B, n=L)
    # conv length = L
    # full convolution result is first N+N-1 entries
    full = conv[:2*N-1]
    return float(np.max(full) * dx)

    Tmax = T0
    Tmin = 1e-6

    # Determine number of cooling steps based on time evaluation
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        # linear cooling fraction
        frac = 1.0 - elapsed / time_limit
        T = T0 * frac + Tmin
        max_move = max_move_initial * frac + 1e-6

        # pick two distinct indices

def _c5_value(h):
    N = h.shape[0]
    dx = 2.0 / N
    b_rev = (1.0 - h)[::-1]
    L = 1 << ((2 * N - 1).bit_length())  # next power of 2
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(H * B, n=L)
    full = conv[:2 * N - 1]
    return float(np.max(full) * dx)

import numpy as np
import time
import math

# ----------------------------------------------------------------------
def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Project v onto 0≤v≤1 and enforce sum(v)=len(v)/2 exactly."""
    target = v.size / 2.0
    np.clip(v, 0.0, 1.0, out=v)

    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        s = v.sum()

    # scale to hit target, then clip again
    v *= target / s
    np.clip(v, 0.0, 1.0, out=v)

    # iterative residual correction
    for _ in range(5):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        if mask.any():
            v[mask] += diff / mask.sum()
            np.clip(v, 0.0, 1.0, out=v)
        else:
            v[0] = np.clip(v[0] + diff, 0.0, 1.0)
    return v

# ----------------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    """Fast evaluation of C5 = max_k dx * Σ h[i]*(1-h[i+k]) using FFT."""
    N = h.size
    dx = 2.0 / N
    b_rev = (1.0 - h)[::-1]                     # reversed complement
    L = 1 << ((2 * N - 1).bit_length())        # next power of two ≥2N-1
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(H * B, n=L)
    full = conv[: 2 * N - 1]
    return float(np.max(full) * dx)

# ----------------------------------------------------------------------
def _simulated_annealing(h_start: np.ndarray,
                         time_limit: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """SA that preserves the linear constraints via pairwise epsilon moves."""
    h_cur = _repair_vector(h_start.copy())
    cur_score = _c5_value(h_cur)

    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size

    start = time.time()
    T0 = 0.04                     # initial temperature
    max_move0 = 0.5               # initial max epsilon
    min_T = 1e-7

    no_improve = 0

    while True:
        if time.time() - start >= time_limit:
            break

        elapsed = time.time() - start
        frac = 1.0 - elapsed / time_limit
        T = min_T + (T0 - min_T) * max(frac, 0.0)
        max_move = max_move0 * max(frac, 0.0) + 1e-9

        # propose a pairwise epsilon move
        i, j = rng.integers(0, N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]

        up = min(1.0 - hi, hj)      # how much we can increase i / decrease j
        down = min(hi, 1.0 - hj)    # how much we can decrease i / increase j
        if up == 0.0 and down == 0.0:
            continue
        up = min(up, max_move)
        down = min(down, max_move)

        eps = rng.random() * (up + down) - down
        if eps == 0.0:
            continue

        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps

        # evaluate
        new_score = _c5_value(h_new)
        delta = new_score - cur_score

        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            no_improve += 1

        # if stuck, perturb around current best
        if no_improve >= 2500:
            h_cur = best_h + rng.normal(scale=0.03, size=N)
            h_cur = _repair_vector(h_cur)
            cur_score = _c5_value(h_cur)
            no_improve = 0

    return best_h, best_score

# ----------------------------------------------------------------------

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    max_N = 800
    levels = [h0.size]
    while levels[-1] < max_N:

        if nxt == levels[-1]:
            break
        levels.append(nxt)

    best_h = h0
    best_score = _c5_value(best_h)

    start_total = time.time()
    total_budget = budget_s

    # ------------------------------------------------------------------
    for idx, N in enumerate(levels):
        # up‑sample if needed
        if best_h.size != N:

            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            best_h = np.interp(x_new, x_old, best_h)
            best_h = _repair_vector(best_h)

        elapsed = time.time() - start_total
        remaining = total_budget - elapsed
        if remaining <= 0.0:

        weight_sum = sum(levels[idx:])          # remaining weight
        time_limit = remaining * (N / weight_sum) * 0.94   # keep modest slack
        time_limit = max(5.0, min(time_limit, remaining * 0.9))

        # run SA at this resolution
        best_h, best_score = _simulated_annealing(best_h, time_limit, rng)

        # final repair/evaluation for this level
        best_h = _repair_vector(best_h)
        best_score = _c5_value(best_h)

    # ------------------------------------------------------------------
    # final clean‑up
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
