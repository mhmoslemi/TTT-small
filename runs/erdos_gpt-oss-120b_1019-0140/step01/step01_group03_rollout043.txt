    gbest_idx = np.argmin(scores)
    gbest_pos = pbest_pos[gbest_idx].copy()
    gbest_score = pbest_score[gbest_idx]

    # PSO parameters: inertia weight w gradually decreasing; cognitive c1=2.0, social c2=2.0; possibly vary.

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    # Use FFT size at least 2*n maybe for linear convolution (zero padding)
    size = 2*n
    H = np.fft.rfft(h, n=size)
    L = np.fft.rfft(1.0 - h, n=size)   # same
    conv = np.fft.irfft(H * L, n=size)  # length 2n

    # It also includes maybe near zero beyond 0..2n-2.
    # compute max.
    max_val = np.max(conv) * dx
    return max_val

def binary_projection(v):
    """Return binary vector of length n with exactly n/2 ones, choosing largest entries."""
    n = v.shape[0]
    # find threshold: we can use np.argpartition to get indices of top n//2.
    idx = np.argpartition(v, - (n//2))[-(n//2):]  # indices of top half.
    result = np.zeros_like(v, dtype=float)
    result[idx] = 1.0
    return result

    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    s = h.size / 2.0
    best_h = h.copy()

    best_score = compute_c5(best_h)

    # We'll treat them as python list for random choice.
    ones = list(np.flatnonzero(best_h > 0.5))
    zeros = list(np.flatnonzero(best_h < 0.5))
    it = 0
    while it < max_iter and (time.time() - start) < time_limit:
        if not ones or not zeros:
            break

        j = rng.choice(zeros)
        # swap
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0

            ones.remove(i)
            zeros.remove(j)
            # add swapped indices
            ones.append(j)
            zeros.append(i)
        it += 1

import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    # trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure bounds contain solution
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on a uniform grid."""
    n = h.size
    dx = 2.0 / n
    size = 2 * n
    H = np.fft.rfft(h, n=size)
    L = np.fft.rfft(1.0 - h, n=size)
    conv = np.fft.irfft(H * L, n=size)[:2 * n - 1]
    return np.max(conv) * dx


def compute_gradient(h, shift):
    """Gradient of the active correlation term."""
    n = h.size
    grad = np.ones_like(h)
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad


def projected_gradient_descent(h0, n_iter=500, step0=0.5):
    """Local refinement of a feasible h."""
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        n = h.size
        size = 2 * n
        H = np.fft.rfft(h, n=size)
        L = np.fft.rfft(1.0 - h, n=size)
        conv = np.fft.irfft(H * L, n=size)[:2 * n - 1]
        idx = np.argmax(conv)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq


def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    arr[idx] = 1.0
    return arr


def binary_projection(v):
    """Project a vector onto the set of binary vectors with exactly n/2 ones."""
    n = v.size
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    h_bin = np.zeros_like(v, dtype=float)
    h_bin[idx] = 1.0
    return h_bin


def hill_climb_binary(h_bin, max_iter, rng, time_budget):
    """Swap‑based local search on the binary feasible set."""
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    ones = list(np.flatnonzero(best_h > 0.5))
    zeros = list(np.flatnonzero(best_h < 0.5))
    start = time.time()
    it = 0
    while it < max_iter and (time.time() - start) < time_budget:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_score = new_score
            best_h = new_h
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        it += 1
    return best_h, best_score

    positions = np.stack(candidates[:pop_size])
    velocities = np.zeros_like(positions)
    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        scores[i] = compute_c5(positions[i])
    pbest_pos = positions.copy()
    pbest_score = scores.copy()
    gbest_idx = int(np.argmin(pbest_score))
    gbest_pos = pbest_pos[gbest_idx].copy()
    gbest_score = pbest_score[gbest_idx]

    w_max, w_min = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    vmax = 0.5

    pso_deadline = start + budget_s * 0.6
    iteration = 0
    while time.time() < pso_deadline - 0.4:
        elapsed = time.time() - start
        w = w_max - (w_max - w_min) * (elapsed / budget_s)
        r1 = rng.random(positions.shape)
        r2 = rng.random(positions.shape)
        velocities = w * velocities + c1 * r1 * (pbest_pos - positions) + c2 * r2 * (gbest_pos - positions)
        velocities = np.clip(velocities, -vmax, vmax)
        positions = positions + velocities
        for i in range(pop_size):
            positions[i] = project_onto_box_sum(positions[i], target_sum)
        for i in range(pop_size):
            sc = compute_c5(positions[i])
            if sc < pbest_score[i]:
                pbest_score[i] = sc
                pbest_pos[i] = positions[i].copy()
                if sc < gbest_score:
                    gbest_score = sc
                    gbest_pos = positions[i].copy()
        iteration += 1
        if iteration % 200 == 0:
            # brief local refinement of current global best
            if time.time() < pso_deadline - 1.0:
                refined, ref_score = projected_gradient_descent(gbest_pos, n_iter=200, step0=0.5)
                if ref_score < gbest_score:
                    gbest_score = ref_score
                    gbest_pos = refined.copy()

    # ---------- Gradient refinement ----------
    grad_deadline = start + budget_s * 0.75
    if time.time() < grad_deadline:
        refined, ref_score = projected_gradient_descent(gbest_pos, n_iter=800, step0=0.5)
        if ref_score < gbest_score:
            gbest_score = ref_score
            gbest_pos = refined.copy()

    # ---------- Binary projection & hill climbing ----------
    time_left = start + budget_s - time.time()
    if time_left > 0.5:
        # binary candidate from best continuous solution
        bin_cand = binary_projection(gbest_pos)
        bin_score = compute_c5(bin_cand)
        if bin_score < gbest_score:
            gbest_score = bin_score
            gbest_pos = bin_cand.copy()
        # hill climbing on binary feasible set
        hill_time = time_left * 0.6
        max_swaps = int(hill_time * 2500)  # heuristic swaps per second
        hc_h, hc_score = hill_climb_binary(gbest_pos, max_swaps, rng, hill_time)
        if hc_score < gbest_score:
            gbest_score = hc_score
            gbest_pos = hc_h.copy()

    # Ensure final feasibility
    gbest_pos = project_onto_box_sum(gbest_pos, target_sum)
    final_score = compute_c5(gbest_pos)
    return gbest_pos, final_score, n
