    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    # simple scaling if small difference
    if cur != 0:

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _local_search(start_h, target_sum, start_time, budget_s):
    h = start_h.copy()
    n = h.size
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    it = 0
    max_iters = 500000  # can be large
    # Simulated annealing schedule
    T = cur_score * 0.01 + 1e-6
    decay = 0.99995
    last_improve = 0
    # continue until time budget
    while it < max_iters and (time.time() - start_time) < budget_s * 0.99:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n-1)
        if j >= i:  # ensure j != i
            j += 1
        hi = h[i]
        hj = h[j]
        max_inc = min(1.0 - hi, hj)
        max_dec = min(hi, 1.0 - hj)
        if max_inc <= 0 and max_dec <= 0:
            it += 1
            continue
        # decide direction
        if max_inc > 0 and (max_dec <= 0 or np.random.rand() < max_inc/(max_inc+max_dec)):
            delta = np.random.rand() * max_inc
            h[i] = hi + delta
            h[j] = hj - delta
            direction = 'inc'
        else:
            delta = np.random.rand() * max_dec
            h[i] = hi - delta
            h[j] = hj + delta
            direction = 'dec'
        new_score = _compute_c5(h)
        # accept if better or with probability
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Monte Carlo acceptance
            if T > 0:
                prob = np.exp(-(new_score - cur_score) / (T + 1e-12))
                accept = np.random.rand() < prob
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                last_improve = it
                if best_score < 0.3808:
                    break
        else:
            # revert
            h[i] = hi
            h[j] = hj
        it += 1
        T = T * decay
        # jitter if stagnated
        if it - last_improve > 20000:
            # Apply small Gaussian noise to the whole vector
            sigma = 0.01
            h_noisy = h + np.random.randn(n) * sigma
            h_noisy = _project_to_constraints(h_noisy, target_sum)
            # compute new score
            new_score2 = _compute_c5(h_noisy)
            # accept if improves, else keep old
            if new_score2 < cur_score:
                h = h_noisy
                cur_score = new_score2
                # not resetting best_h unless new_score2 < best_score? we can update
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
                    last_improve = it
            else:
                # maybe accept with some probability
                if T > 0 and np.random.rand() < np.exp(-(new_score2 - cur_score) / (T + 1e-12)):
                    h = h_noisy
                    cur_score = new_score2
                # else keep old h (no change)
        # optional early break if time near end
        if (time.time() - start_time) > budget_s * 0.99:
            break
    return best_h, best_score

def _local_search(start_h, target_sum, start_time, budget_s):
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    it = 0
    max_iters = 2000000 # large
    T = cur_score * 0.01 + 1e-6
    decay = 0.9999
    last_improve = 0
    while it < max_iters and (time.time() - start_time) < budget_s * 0.99:
        # compute correlation for direction
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        # compute shifted vector
        shifted = np.zeros_like(h)
        if lag >= 0:
            shifted[:n - lag] = h[lag:]
        else:
            shifted[-lag:] = h[:n + lag]
        # contributions
        contrib = h * (1 - shifted)
        # i candidate
        i_candidates = np.where(h > 1e-12)[0]
        if i_candidates.size == 0:
            break
        i = i_candidates[np.argmax(contrib[i_candidates])]
        # j candidate
        j_candidates = np.where(h < 1 - 1e-12)[0]
        if j_candidates.size == 0:
            break
        # compute derivative values as (1 - shifted) -> we want minimal = max shifted
        j = j_candidates[np.argmax(shifted[j_candidates])]
        # ensure i != j
        if i == j:
            # fallback to random pair move
            i, j = np.random.choice(np.arange(n), size=2, replace=False)
        # compute delta
        delta_max = min(h[i], 1 - h[j])
        if delta_max <= 0:
            # fallback small random move
            i, j = np.random.choice(np.arange(n), size=2, replace=False)
            delta_max = min(h[i], 1 - h[j])
            if delta_max <= 0:
                # cannot move
                it += 1
                continue
        delta = delta_max * np.random.rand() * 0.5  # up to half of max possible
        # store old
        old_i = h[i]; old_j = h[j]
        h[i] = old_i - delta
        h[j] = old_j + delta
        # Evaluate new score
        new_score = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                last_improve = it
                if best_score < 0.3808:
                    break
        else:
            # revert
            h[i] = old_i
            h[j] = old_j
        # temperature decay
        T *= decay
        it += 1
        # jitter if stagnated
        if it - last_improve > 50000:
            sigma = 0.01
            h_candidate = h + np.random.randn(n) * sigma
            h_candidate = _project_to_constraints(h_candidate, target_sum)
            cand_score = _compute_c5(h_candidate)

                    best_score = cur_score
                    best_h = h.copy()
                    last_improve = it
            # reset last_improve maybe
            last_improve = it
    return best_h, best_score

import numpy as np
import time

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust sum(v) to target by water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    """Return max_k ∫ h(x)(1-h(x+k))dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _local_search(start_h, target_sum, start_time, budget_s):
    """Simple directed local search with occasional stochastic jumps."""
    h = start_h.copy()
    n = h.size
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    it = 0
    max_iters = 2000000  # safe upper bound; actual stop by time
    T = cur_score * 0.01 + 1e-6
    decay = 0.9999
    last_improve = 0

    while it < max_iters and (time.time() - start_time) < budget_s * 0.99:
        # correlation to identify worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)

        # build shifted version of h according to this lag
        shifted = np.zeros_like(h)
        if lag >= 0:
            shifted[: n - lag] = h[lag:]
        else:
            shifted[-lag:] = h[: n + lag]

        # contribution of each index to the current worst correlation
        contrib = h * (1.0 - shifted)

        # choose i to decrease (largest contribution, must have h[i] > 0)
        i_cand = np.where(h > 1e-12)[0]
        if i_cand.size == 0:
            break
        i = i_cand[np.argmax(contrib[i_cand])]

        # choose j to increase (where shifted is largest, must have h[j] < 1)
        j_cand = np.where(h < 1.0 - 1e-12)[0]
        if j_cand.size == 0:
            break
        j = j_cand[np.argmax(shifted[j_cand])]

        if i == j:
            # fallback to random pair if selection collides
            i, j = np.random.choice(np.arange(n), size=2, replace=False)

        # feasible delta respecting bounds
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0:
            # fallback random move
            i, j = np.random.choice(np.arange(n), size=2, replace=False)
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 0:
                it += 1
                continue
        delta = delta_max * np.random.rand() * 0.5  # move at most half of allowed

        old_i, old_j = h[i], h[j]
        h[i] = old_i - delta
        h[j] = old_j + delta

        new_score = _compute_c5(h)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                last_improve = it
                if best_score < 0.3808:
                    break
        else:
            h[i] = old_i
            h[j] = old_j

        T *= decay
        it += 1

        # occasional stochastic jump if stagnation detected
        if it - last_improve > 20000:
            sigma = 0.01
            cand = h + np.random.randn(n) * sigma
            cand = _project_to_constraints(cand, target_sum)
            cand_score = _compute_c5(cand)
            accept_jump = False
            if cand_score < cur_score:
                accept_jump = True
            else:
                prob = np.exp(-(cand_score - cur_score) / (T + 1e-12))
                if np.random.rand() < prob:
                    accept_jump = True
            if accept_jump:
                h = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
                    last_improve = it
            else:
                last_improve = it  # reset stagnation counter

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # initial vector
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:

        n0 = init_vec.size
    else:
        n0 = 200
        init_vec = np.random.rand(n0)
    target0 = n0 / 2.0

    # a few random starters to improve initial best

        rnd = _project_to_constraints(rnd, target0)
        sc = _compute_c5(rnd)
        if sc < best_score:
            best_score = sc
            best_h = rnd.copy()
            if best_score < 0.3808:

    n = n0
    max_n = 1520  # limit to keep runtime reasonable
    while True:
        if (time.time() - start_time) > budget_s * 0.95:

        target = n / 2.0

        # several local‑search restarts at current resolution
        for _ in range(3):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            start_h = best_h + np.random.randn(n) * 0.02
            start_h = _project_to_constraints(start_h, target)
            h_opt, sc_opt = _local_search(start_h, target, start_time, budget_s)
            if sc_opt < best_score:

                best_h = h_opt.copy()
                if best_score < 0.3808:
                    break

            break

        # prepare next resolution if time permits
        if n >= max_n:
            break
        if (budget_s - (time.time() - start_time)) < 200:
            break
        n_new = n * 2
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
        best_h = np.interp(x_new, x_old, best_h)

        best_score = _compute_c5(best_h)

    # final bound evaluation

        try:
            bound = float(evaluate_erdos_solution(best_h))
        except Exception:
            bound = float(best_score)
    else:
        bound = float(best_score)

    return best_h.astype(np.float64), bound, best_h.size
