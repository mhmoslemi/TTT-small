    n = h.shape[0]

                    adj = min(abs(diff), (1.0 - cur_h[idx_adj]) if diff > 0 else cur_h[idx_adj])
                    cur_h[idx_adj] += np.sign(diff) * adj
                    cur_val = compute_c5(cur_h)
                    if cur_val < best_val:
                        best_val, best_h = cur_val, cur_h.copy()

    # ------------------------------------------------------------------
    # Final binary hill‑climbing refinement
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    if elapsed < budget_s - 0.2:
        # create binary version by taking the largest entries
        binary = np.zeros(n, dtype=int)
        top_idx = np.argpartition(-best_h, n // 2 - 1)[: n // 2]
        binary[top_idx] = 1
        bin_val = compute_c5(binary.astype(float))
        binary, bin_val = hill_climb_binary(binary, bin_val, start_time, budget_s, rng)
        if bin_val + 1e-12 < best_val:
            best_h = binary.astype(float)
            best_val = bin_val

    # ------------------------------------------------------------------
    # safety clip and exact‑sum enforcement
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    if abs(s - target_sum) > 1e-12:
        best_h += (target_sum - s) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    best_val = compute_c5(best_h)
    return best_h, best_val, n
