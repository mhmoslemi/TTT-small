def _simulated_annealing(v, time_limit):
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    cur_v = v.copy()
    cur_score = best_score
    start = time.time()
    T0 = 0.1
    Tmin = 1e-5
    # schedule: exponential cooling
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = max(Tmin, T0 * (0.99 ** (elapsed * 100))) # adjust speed
        # propose move
        i, j = np.random.choice(len(v), 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2 * T * 5 # scale with T maybe
        cand = cur_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, len(v))
        new_score = _compute_c5(cand)
        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < np.exp(-dE / max(T, 1e-12)):
            cur_v, cur_score = cand, new_score
            if new_score < best_score:
                best_v, best_score = cand.copy(), new_score
    return best_v, best_score

def _compute_c5(v):
    n = len(v)
    dx = 2.0 / n
    # Use fft of v and w = 1 - v, need cross-correlation.
    # Convolution via FFT: corr = ifft(fft(v) * conj(fft(w)))
    # However correlation (full) is conv with reversed w.
    # correlation of v with w = conv(v, w_rev)
    w = 1.0 - v
    # compute convolution via fft:
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()  # next power of 2
    fv = np.fft.rfft(v, n=N)
    fw = np.fft.rfft(w[::-1], n=N)  # reversed w for correlation
    conv = np.fft.irfft(fv * fw, n=N)
    # conv contains length N, but the first (n-1) elements correspond to negative lags?
    # For full correlation, we need indices from 0 to 2*n-2.
    corr_full = conv[:2*n-1]
    return np.max(corr_full) * dx
```

def _simulated_annealing(v, time_limit):
    n = len(v)
    cur_v = v.copy()
    cur_score = _compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = 0.2 * (1 - frac) + 0.001 * frac
        # propose exchange
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2 * step
        cand = cur_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        new_score = _compute_c5(cand)
        dE = new_score - cur_score
        # temperature schedule for acceptance
        temperature = 0.1 * (1 - frac) + 0.001 * frac
        if dE < 0 or np.random.rand() < np.exp(-dE / max(temperature, 1e-12)):
            cur_v = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_v = cand.copy()
    return best_v, best_score

def _dct_random_search(init_v, n, time_limit, N_basis=20):
    # Build DCT basis matrix
    x = np.arange(n) * (2.0 / n)
    B = np.empty((n, N_basis), dtype=np.float64)
    B[:,0] = 1.0
    for k in range(1, N_basis):
        B[:,k] = np.cos(k * np.pi * x / 2.0)
    # Normalize columns to unit norm
    norms = np.linalg.norm(B, axis=0, keepdims=True)
    B = B / norms
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)
    start = time.time()
    while time.time() - start < time_limit:
        # random coefficients: Gaussian
        a = np.random.randn(N_basis) * 0.2
        cand = B @ a
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score = score
            best_v = cand.copy()
    return best_v, best_score

        if res.success:
            v = _repair_vector(res.x, n)
            sc = _compute_c5(v)
            if sc < best_score:
                best_score, best_v = sc, v
    except Exception:
        pass

    # random restarts
    while time.time() - start < time_limit:
        rnd = _random_feasible(n)
        try:
            res = minimize(_compute_c5, rnd, method="SLSQP",
                           bounds=bounds, constraints=cons,
                           options={"maxiter": 300, "ftol": 1e-9, "disp": False})
            if res.success:
                v = _repair_vector(res.x, n)
                sc = _compute_c5(v)
                if sc < best_score:
                    best_score, best_v = sc, v
        except Exception:
            pass
        if time.time() - start > time_limit * 0.9:
            break
    return best_v, best_score

def _simulated_annealing(v, time_limit):
    n = len(v)
    cur_v = v.copy()
    cur_score = _compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = 0.2 * (1 - frac) + 0.001 * frac
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        new_score = _compute_c5(cand)
        dE = new_score - cur_score
        temperature = 0.1 * (1 - frac) + 0.001 * frac
        if dE < 0 or np.random.rand() < np.exp(-dE / max(temperature, 1e-12)):
            cur_v, cur_score = cand, new_score
            if new_score < best_score:
                best_score, best_v = new_score, cand.copy()
    return best_v, best_score

def _dct_random_search(init_v, n, time_limit, N_basis=20):
    x = np.arange(n) * (2.0 / n)
    B = np.empty((n, N_basis), dtype=np.float64)
    B[:, 0] = 1.0
    for k in range(1, N_basis):
        B[:, k] = np.cos(k * np.pi * x / 2.0)
    norms = np.linalg.norm(B, axis=0, keepdims=True)
    B /= norms
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)
    start = time.time()
    while time.time() - start < time_limit:
        a = np.random.randn(N_basis) * 0.2
        cand = B @ a
        cand = _repair_vector(cand, n)
        sc = _compute_c5(cand)
        if sc < best_score:
            best_score, best_v = sc, cand.copy()
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function for the Erdős C₅ problem."""
    np.random.seed(seed)
    start = time.time()

    # acquire parent construction if available
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 200
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0 // 2] = 1.0
    parent = _repair_vector(parent, n0)

    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    # schedule of resolutions
    resolutions = [n0]
    if n0 * 2 <= 800:
        resolutions.append(n0 * 2)
    if n0 * 4 <= 800:
        resolutions.append(n0 * 4)

    for idx, n in enumerate(resolutions):
        if time.time() - start > budget_s - 2.0:
            break

        # upsample or keep current best for this resolution
        if len(best_v) != n:
            init_guess = _upsample_vector(best_v, n)
        else:
            init_guess = best_v.copy()

        # time budgeting per resolution
        elapsed = time.time() - start
        time_left = max(0.0, budget_s - elapsed - 2.0)
        remaining_res = len(resolutions) - idx
        per_res = time_left / remaining_res if remaining_res else 0.0

        # SLSQP optimization
        t_opt = per_res * 0.4
        if t_opt > 1.0:
            v_opt, sc_opt = _optimize_at_resolution(n, init_guess, t_opt)
            if sc_opt < best_score:
                best_score, best_v, best_n = sc_opt, v_opt.copy(), n

        # Simulated annealing
        if time.time() - start < budget_s - 1.0:
            cur_v = best_v if best_n == n else init_guess
            t_ann = per_res * 0.4
            v_ann, sc_ann = _simulated_annealing(cur_v, t_ann)
            if sc_ann < best_score:
                best_score, best_v, best_n = sc_ann, v_ann.copy(), n

        # DCT random search
        if time.time() - start < budget_s - 0.5:
            cur_v = best_v if best_n == n else init_guess
            t_dct = per_res - t_opt - t_ann
            if t_dct > 0.5:
                v_dct, sc_dct = _dct_random_search(cur_v, n, t_dct, N_basis=20)
                if sc_dct < best_score:
                    best_score, best_v, best_n = sc_dct, v_dct.copy(), n

    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)
    return best_v.astype(np.float64), float(best_score), int(best_n)
