def _compute_c5_fft(v):
    n = len(v)
    w = 1.0 - v
    N = 1 << (2*n-1).bit_length() # next power of 2 >= 2n-1, maybe 2**ceil(log2(2*n))
    # Using real FFT
    a_fft = np.fft.rfft(v, N)
    b_fft = np.fft.rfft(w[::-1], N)
    conv = np.fft.irfft(a_fft * b_fft, N)
    # conv length N, but full convolution length = 2*n-1, taken from indices 0 to 2*n-2
    conv = conv[:2*n-1]  # full correlation
    # maybe need to scale (no scaling)
    # multiply dx
    dx = 2.0 / n
    return np.max(conv) * dx

def _repair_vector(v, n, max_iter=50):
    # ensure v is numpy array
    v = np.asarray(v, dtype=np.float64)
    target = n / 2.0
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # compute adjustment per free variable (those not at bounds)
        free = (v > 0.0) & (v < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds, adjust by scaling
            if s > 0:
                v *= target / s
                v = np.clip(v, 0.0, 1.0)
            break
        # distribute diff equally among free variables
        adjust = diff / n_free
        v[free] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction
    s = v.sum()
    if abs(s - target) > 1e-9:
        # scale all to match target, then clip again
        factor = target / s if s != 0 else 0.0
        v *= factor
        v = np.clip(v, 0.0, 1.0)
        # If still off, adjust uniformly
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

import time
import numpy as np
from scipy.optimize import minimize

def _repair_vector(v, n, max_iter=15):
    """Project v onto the feasible set: 0<=v<=1, sum(v)=n/2."""
    v = np.asarray(v, dtype=np.float64)
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # identify variables that can move (not at a bound)
        free = (v > 0.0) & (v < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds: scale globally
            if s != 0:
                v *= target / s
                v = np.clip(v, 0.0, 1.0)
            break
        adjust = diff / n_free
        v[free] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        # simple scaling then clipping
        factor = target / s if s != 0 else 0.0
        v *= factor
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx for the discrete representation."""
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of a vector to a finer grid, then repair feasibility."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

    """Simple simulated annealing on the C5 objective."""
    n = v.shape[0]
    cur = v.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    step = 0.2            # magnitude of exchange
    T0 = 0.02             # initial temperature
    Tmin = 1e-5
    it = 0
    while time.time() - start < time_limit:
        # pick two distinct indices without allocation overhead
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        dscore = cand_score - cur_score
        # exponential cooling schedule
        T = max(Tmin, T0 * (0.995 ** it))
        if dscore < 0 or np.random.rand() < np.exp(-dscore / max(T, 1e-12)):
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
        it += 1
        if it % 2000 == 0:
            step *= 0.9
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function for the Erdős C₅ problem."""
    np.random.seed(seed)
    start_time = time.time()

    # -------------------------------------------------------------
    # Acquire initial construction if provided.
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        # fallback: uniform 0.5 function (feasible)
        parent = np.full(200, 0.5, dtype=np.float64)
    n0 = parent.shape[0]
    parent = _repair_vector(parent, n0)

    # -------------------------------------------------------------
    # Build resolution schedule (powers of two up to a modest size).
    max_n = 1520
    resolutions = [n0]
    while resolutions[-1] * 2 <= max_n:
        resolutions.append(resolutions[-1] * 2)

    best_v = parent
    best_score = _compute_c5(best_v)

    # -------------------------------------------------------------
    for idx, n in enumerate(resolutions):
        # time guard
        if time.time() - start_time >= budget_s - 2.0:
            break

        # upscale / downscale to current resolution
        if best_v.shape[0] != n:
            best_v = _upsample_vector(best_v, n)
        best_v = _repair_vector(best_v, n)
        best_score = _compute_c5(best_v)

        # allocate time for this resolution (80% of remaining split equally)
        elapsed = time.time() - start_time
        remaining_total = budget_s - elapsed
        remaining_res = len(resolutions) - idx
        time_this_res = max(0.0, remaining_total * 0.8 / remaining_res)

        # ---------------------------------------------------------
        # Stage 1: Simulated annealing (dominant effort)
        anneal_time = time_this_res * 0.85
        cand_v, cand_score = _simulated_anneal(best_v, anneal_time)
        if cand_score < best_score:
            best_v, best_score = cand_v, cand_score

        # ---------------------------------------------------------
        # Stage 2: Brief local SLSQP refinement
        refine_time = time_this_res - anneal_time
        if refine_time > 1.0:
            try:
                cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
                bounds = [(0.0, 1.0)] * n
                res = minimize(
                    lambda x: _compute_c5(x),
                    best_v,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 200, "ftol": 1e-9, "disp": False},
                )
                if res.success:
                    v_opt = _repair_vector(res.x, n)
                    sc_opt = _compute_c5(v_opt)
                    if sc_opt < best_score:
                        best_v, best_score = v_opt, sc_opt
            except Exception:
                pass

    # -------------------------------------------------------------
    # Use any remaining budget for a final annealing pass.
    leftover = budget_s - (time.time() - start_time)
    if leftover > 1.0:
        best_v, best_score = _simulated_anneal(best_v, leftover)

    best_v = _repair_vector(best_v, best_v.shape[0])
    best_score = _compute_c5(best_v)
    return best_v.astype(np.float64), float(best_score), int(best_v.shape[0])
