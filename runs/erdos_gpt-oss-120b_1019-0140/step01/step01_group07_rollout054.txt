def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    # compute difference
    diff = target - cur
    if diff > 0:
        # Need to add diff to entries where v < 1
        # distribute proportionally to remaining capacity.
        while diff > 1e-12:
            free = np.where(v < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            caps = 1.0 - v[free]
            total_cap = caps.sum()
            if total_cap < 1e-12:
                break
            # amount to add proportionally
            add = diff * caps / total_cap
            # limit to capacity
            add = np.minimum(add, caps)
            v[free] += add
            diff = target - v.sum()

        diff = -diff

            free = np.where(v > 1e-12)[0]

    if abs(cur - target) < 1e-12:
        return v
    # Add or remove mass until the target sum is reached.
    if cur < target:
        need = target - cur
        while need > 1e-12:
            free = np.where(v < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            caps = 1.0 - v[free]
            add = min(need, caps.max())
            idx = free[np.argmax(caps)]
            inc = min(add, caps[np.argmax(caps)])
            v[idx] += inc
            need -= inc
    else:
        excess = cur - target
        while excess > 1e-12:
            free = np.where(v > 1e-12)[0]
            if free.size == 0:
                break
            caps = v[free]
            sub = min(excess, caps.max())
            idx = free[np.argmax(caps)]
            dec = min(sub, caps[np.argmax(caps)])
            v[idx] -= dec
            excess -= dec
    v = np.clip(v, 0.0, 1.0)
    # Final scaling if still off slightly.
    cur = np.sum(v)
    if cur > 0 and abs(cur - target) > 1e-9:
        v *= target / cur
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    """Maximum correlation ∫ h(x)*(1-h(x+k)) dx via FFT."""
    n = h.size
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    return np.max(corr) * (2.0 / n)

def _refine_with_slsqp(h, target):
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * h.size
    try:
        res = minimize(
            _compute_c5,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": 200, "ftol": 1e-10, "disp": False},
        )
        h_opt = _project_to_constraints(res.x, target)
        return h_opt, _compute_c5(h_opt)
    except Exception:
        return h, _compute_c5(h)

def _simulated_annealing(h0, target, max_iters, start_time, budget_s):
    """Simple SA preserving sum via pairwise transfers."""
    h = h0.copy()
    n = h.size
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T0, Tf = 0.02, 1e-6
    eps_limit = 0.2

    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.95:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        # try to find a feasible pair
        for _ in range(5):
            i = np.random.randint(0, n)
            if h[i] <= 1e-12:
                continue
            j = np.random.randint(0, n)
            if j == i or h[j] >= 1.0 - 1e-12:
                continue
            eps_max = min(eps_limit, h[i], 1.0 - h[j])
            if eps_max <= 1e-12:
                continue
            epsilon = np.random.rand() * eps_max
            h[i] -= epsilon
            h[j] += epsilon
            new_score = _compute_c5(h)
            delta = new_score - cur_score
            if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
                break
            # reject move
            h[i] += epsilon
            h[j] -= epsilon
        # end trial loop
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial vector
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h0 = np.array(init_vec, dtype=np.float64).ravel()
        n0 = h0.size
    else:
        n0 = 380
        h0 = np.random.rand(n0)

    target0 = n0 / 2.0
    h0 = _project_to_constraints(h0, target0)
    best_h = h0.copy()
    best_score = _compute_c5(best_h)

    # Work through increasing resolutions while time permits
    current_h = best_h
    current_n = n0
    max_n = 3040

    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 30 or current_n >= max_n:
            break

        # Determine next resolution
        n_next = min(max_n, current_n * 2)
        if n_next == current_n:
            break

        # Upsample via linear interpolation
        x_old = np.linspace(0.0, 2.0, current_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_next, endpoint=False)
        h_up = np.interp(x_new, x_old, current_h)
        target_next = n_next / 2.0
        h_up = _project_to_constraints(h_up, target_next)

        # Estimate iterations based on size and remaining time
        base_iters = 80000  # for n=380
        est_iters = int(base_iters * 380 / n_next)
        est_iters = max(2000, min(est_iters, int(remaining * 1500)))
        # Simulated annealing
        cand_h, cand_score = _simulated_annealing(
            h_up, target_next, est_iters, start_time, budget_s
        )
        # Local refinement with SLSQP
        cand_h, cand_score = _refine_with_slsqp(cand_h, target_next)

        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        # Prepare for possible further refinement
        current_h = cand_h
        current_n = n_next

    # Final projection to guarantee constraints exactly
    best_h = _project_to_constraints(best_h, best_h.size / 2.0)

    # Evaluate final bound using external evaluator if available
    final_score = best_score
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_h.size)
