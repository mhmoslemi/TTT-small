    # Round to nearest integer (0 or 1)
    bin_h = (initial_h > 0.5).astype(np.float64)
    # Adjust count to match target_sum

import time
import numpy as np

def compute_max_overlap_int(h: np.ndarray) -> int:
    """Return the maximal overlap count ∫h(x)(1-h(x+k))dx approximated by correlation."""
    # np.correlate computes sum_i a[i]*v[i+lag] for each lag
    corr = np.correlate(h, 1.0 - h, mode="full")
    return int(np.max(corr))


def init_binary_vector(n: int, target_sum: int, rng: np.random.RandomState) -> np.ndarray:
    """Create a binary vector of length n with exactly target_sum ones."""
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, target_sum, replace=False)
    h[idx] = 1.0
    return h


def binary_from_initial(initial: np.ndarray, target_sum: int, rng: np.random.RandomState) -> np.ndarray:
    """Convert any initial vector to a binary one with the correct sum."""
    bin_h = (initial > 0.5).astype(np.float64)
    cur = int(bin_h.sum())
    diff = target_sum - cur
    if diff > 0:
        zeros = np.where(bin_h == 0)[0]
        if zeros.size >= diff:
            chosen = rng.choice(zeros, diff, replace=False)
            bin_h[chosen] = 1.0
        else:
            # fallback: set all zeros to 1 then trim excess ones
            bin_h[zeros] = 1.0
            excess = int(bin_h.sum()) - target_sum
            ones = np.where(bin_h == 1)[0]
            drop = rng.choice(ones, excess, replace=False)
            bin_h[drop] = 0.0
    elif diff < 0:
        ones = np.where(bin_h == 1)[0]
        chosen = rng.choice(ones, -diff, replace=False)
        bin_h[chosen] = 0.0
    return bin_h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.1  # safety margin

    # Determine problem size
    n = None
    try:
        init_vals = np.array(initial_h_values, dtype=np.float64)
        if init_vals.size % 2 == 0 and init_vals.size >= 20:
            n = init_vals.size
    except Exception:
        n = None
    if n is None:
        n = 70  # default size
    target_sum = n // 2

    # Prepare a list of initial candidates
    init_candidates = []

    # candidate derived from the provided initial_h_values (if any)
    try:
        init_vals = np.array(initial_h_values, dtype=np.float64)
        if init_vals.size == n:
            cand = binary_from_initial(init_vals, target_sum, rng)
            init_candidates.append(cand)
    except Exception:
        pass

    # add a purely random candidate
    init_candidates.append(init_binary_vector(n, target_sum, rng))

    # parameters for simulated annealing
    T0 = 5.0          # initial temperature
    Tf = 0.001        # final temperature
    rotate_prob = 0.05
    max_no_improve = 5000  # steps without improvement before restart

    best_h = None
    best_int = np.iinfo(np.int64).max

    for start_h in init_candidates:
        cur_h = start_h.copy()
        cur_int = compute_max_overlap_int(cur_h)

        # update global best
        if cur_int < best_int:
            best_int = cur_int
            best_h = cur_h.copy()

        steps_since_best = 0

        while True:
            now = time.time()
            if now - start >= max_time:
                break
            progress = (now - start) / max_time
            T = T0 * (1.0 - progress) + Tf * progress

            # decide move type
            if rng.rand() < rotate_prob:
                # rotate the whole vector
                offset = rng.randint(1, n)
                cand = np.roll(cur_h, offset)
            else:
                # swap a random 1 with a random 0
                ones = np.where(cur_h == 1)[0]
                zeros = np.where(cur_h == 0)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                cand = cur_h.copy()
                cand[i] = 0.0
                cand[j] = 1.0

            cand_int = compute_max_overlap_int(cand)
            delta = cand_int - cur_int

            if delta <= 0:
                accept = True
            else:
                prob = np.exp(-delta / (T + 1e-12))
                accept = rng.rand() < prob

            if accept:
                cur_h = cand
                cur_int = cand_int
                steps_since_best = 0
                if cur_int < best_int:
                    best_int = cur_int
                    best_h = cur_h.copy()
            else:
                steps_since_best += 1
                if steps_since_best >= max_no_improve:
                    # restart from a fresh random vector
                    cur_h = init_binary_vector(n, target_sum, rng)
                    cur_int = compute_max_overlap_int(cur_h)
                    steps_since_best = 0

        if time.time() - start >= max_time:
            break

    if best_h is None:
        # fallback – should never happen
        best_h = init_binary_vector(n, target_sum, rng)
        best_int = compute_max_overlap_int(best_h)

    c5_bound = best_int * (2.0 / n)
    return (best_h, float(c5_bound), n)
