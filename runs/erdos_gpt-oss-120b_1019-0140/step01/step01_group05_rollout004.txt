def hill_climb(h, time_limit, rng):
    best = h.copy()
    best_score = _c5_value(best)
    N = len(h)
    start = time.time()
    # Use gradient-like moves that increase min correlation for the worst shift.
    # Loop for time.
    while time.time() - start < time_limit:
        # Compute correlation of h with itself (full)
        corr = np.correlate(h, h, mode='full')
        # find shift index with minimal correlation (worst)
        idx = np.argmin(corr)
        min_corr = corr[idx]
        # Compute shift k (index offset)
        k = idx - (N - 1)
        # Compute shifted version of h (with zeros out-of-range)
        if k >= 0:
            shifted = np.concatenate([h[k:], np.zeros(k)])
        else:
            shifted = np.concatenate([np.zeros(-k), h[:N+k]])
        # Decide move
        # Determine potential donor indices (high h to decrease) and recipient (low h to increase)
        # Actually we want to increase corr at shift k, which needs increasing product sum.
        # The product sum term is sum_i h_i * shifted[i].
        # Moving epsilon from i to j changes sum by epsilon * (shifted[j] - shifted[i]).
        # So we need i where shifted[i] is small, and j where shifted[j] is large.
        # Choose i = argmin shifted among positions where h[i] > 0 (can decrease)
        # Choose j = argmax shifted among positions where h[j] < 1 (can increase)
        # Use small epsilon = min(lambda* (some) ...)
        possible_i = np.where(h > 1e-12)[0]  # donor
        possible_j = np.where(h < 1 - 1e-12)[0]  # recipient
        if possible_i.size == 0 or possible_j.size == 0:
            break
        # Among i candidates, pick i with smallest shifted[i]
        i_vals = shifted[possible_i]
        i = possible_i[np.argmin(i_vals)]
        # Among j candidates, pick j with largest shifted[j]
        j_vals = shifted[possible_j]
        j = possible_j[np.argmax(j_vals)]
        if shifted[j] <= shifted[i] + 1e-12:
            # No improvement possible for this shift; maybe random inertia
            # Perform random pair move as SA fallback
            i, j = rng.choice(N, size=2, replace=False)
        # Determine max possible epsilon
        max_eps = min(h[i], 1 - h[j])
        # Also limit step size to some factor, e.g., 0.2 (like before)
        max_step = 0.2
        eps = min(max_eps, max_step) * rng.random()
        if eps <= 0:
            continue
        # Apply move
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # clip just in case
        # compute score
        new_score = _c5_value(h_new)
        # Accept if better (or with SA temperature)
        if new_score < best_score:
            h = h_new
            best = h_new
            best_score = new_score
        else:
            # With probability maybe accept?
            # Use simple temperature schedule
            elapsed = time.time() - start
            frac = max(0.0, 1.0 - elapsed / time_limit)
            T = 0.01 * frac
            if rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                h = h_new
                best_score = new_score
    return best, best_score

    n = h.shape[0]
    if k >= 0:
        out = np.empty_like(h)
        out[:n-k] = h[k:]
        out[n-k:] = 0
    else:
        out = np.empty_like(h)
        out[-k:] = h[:n+k]

    return out

def hill_climb(h, time_limit, rng):
    N = h.size
    best_h = h.copy()
    best_score = _c5_value(best_h)
    start = time.time()
    while time.time() - start < time_limit:
        # compute self-correlation
        corr = np.correlate(h, h, mode='full')
        idx = np.argmin(corr)
        k = idx - (N - 1)
        # compute shifted array
        # use slicing to avoid copying entire array? but okay.
        if k >= 0:
            shifted = np.empty_like(h)
            shifted[:N - k] = h[k:]
            shifted[N - k:] = 0.0
        else:
            shifted = np.empty_like(h)
            shifted[-k:] = h[:N + k]
            shifted[: -k] = 0.0
        # donor-possible indices
        donor_mask = h > 1e-12
        recipient_mask = h < 1 - 1e-12
        if not donor_mask.any() or not recipient_mask.any():
            # no possible move
            break
        # compute candidate i (min shifted)
        i_candidates = np.where(donor_mask)[0]
        i_shift_vals = shifted[i_candidates]
        i = i_candidates[np.argmin(i_shift_vals)]
        # compute candidate j (max shifted)
        j_candidates = np.where(recipient_mask)[0]
        j_shift_vals = shifted[j_candidates]
        j = j_candidates[np.argmax(j_shift_vals)]
        if i == j or shifted[j] <= shifted[i] + 1e-12:
            # fallback random pair
            i, j = rng.integers(0, N, size=2, endpoint=False)
            while i == j:

    N = h.size
    start = time.time()

    T0 = 0.01  # adjust

        if elapsed >= time_limit:
            break

        corr = np.correlate(h, h, mode='full')
        idx_min = int(np.argmin(corr))

        # compute shifted version for this k
        if k_shift >= 0:

            shifted[:N - k_shift] = h[k_shift:]

            shifted[-k_shift:] = h[:N + k_shift]

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce exact integral constraint sum(v)=len(v)/2."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _hill_climb(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Improvement targeting the worst shift using pairwise mass transfers."""
    h = h_start.copy()
    best_h = h.copy()
    best_score = _c5_value(h)
    N = h.size
    start = time.time()
    T0 = 0.01          # initial temperature for occasional uphill steps
    max_step = 0.05    # maximal epsilon per move
    while True:
        if time.time() - start >= time_limit:
            break
        # self‑correlation (full) to locate the worst shift
        corr = np.correlate(h, h, mode='full')
        idx_min = int(np.argmin(corr))
        k = idx_min - (N - 1)               # shift corresponding to min correlation
        # shifted version of h for this k (zero‑padded)
        if k >= 0:
            shifted = np.empty_like(h)
            shifted[:N - k] = h[k:]
            shifted[N - k:] = 0.0
        else:
            shifted = np.empty_like(h)
            shifted[-k:] = h[:N + k]
            shifted[: -k] = 0.0
        donor_mask = h > 1e-12
        recipient_mask = h < 1.0 - 1e-12
        if not donor_mask.any() or not recipient_mask.any():
            break
        donor_idxs = np.where(donor_mask)[0]
        donor_shift = shifted[donor_idxs]
        i = donor_idxs[int(np.argmin(donor_shift))]
        rec_idxs = np.where(recipient_mask)[0]

        j = rec_idxs[int(np.argmax(rec_shift))]

            continue

        if eps <= 0.0:
            continue

        h_new[j] += eps

            best_h = h_new
