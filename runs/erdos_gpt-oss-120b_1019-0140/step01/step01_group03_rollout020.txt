def compute_grad_smooth(h, alpha):
    n = h.size
    # compute correlation
    cvec = np.correlate(h, 1 - h, mode='full')
    # weight
    a_c = alpha * cvec
    max_a_c = np.max(a_c)
    exp_a = np.exp(a_c - max_a_c)
    w = exp_a / np.sum(exp_a)
    # gradient accumulation
    grad = np.zeros_like(h)
    # Loop over shifts
    # shift index l = -(n-1)...(n-1)
    # pos = l + n - 1
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        l = idx - (n - 1)
        if l >= 0:
            # term1: (1 - h[i + l]) for i=0..n-1-l?

        else:
            s = -l
            grad[s:] += weight * (1 - h[:n - s])
            grad[:n - s] += weight * (-h[s:])
    return grad

def gradient_smooth(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    a_corr = alpha * corr
    max_a = np.max(a_corr)
    exp_a = np.exp(a_corr - max_a)
    w = exp_a / np.sum(exp_a)
    grad = np.zeros_like(h)
    # shift from -(n-1) to n-1; index range in corr: idx 0..2n-2

            grad[s:] += weight * (1 - h[:n - s])
            grad[:n - s] += weight * (-h[s:])
    return grad

    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # broaden brackets if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Exact C5 value for a discretised h (max correlation times dx)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the max-correlating shift term w.r.t. h."""
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[:n - shift] = 1.0 - h[shift:]
        grad[shift:] += -h[:n - shift]
    else:
        s = -shift
        grad[s:] = 1.0 - h[:n - s]
        grad[:n - s] += -h[s:]
    return grad

def gradient_smooth(h, alpha):
    """Gradient of the smoothed max (log-sum-exp) correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * corr
    a_max = np.max(a)
    weights = np.exp(a - a_max)
    weights /= np.sum(weights)

    grad = np.zeros_like(h)
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - h[shift:])
            grad[shift:] += w * (-h[:n - shift])
        else:
            s = -shift
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] += w * (-h[s:])
    return grad

def exact_pgd_refine(h0, n_iter=400, step0=0.5):
    """Short exact projected gradient descent using the active shift."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h_new = project_onto_box_sum(h - eta * grad, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    """Binary Thue–Morse sequence (0/1 values)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Random {0,1} vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1. Build initial candidates at several resolutions
    # ------------------------------------------------------------------
    target_sizes = [180, 256, 128, 64]
    candidates = []

    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        for n in target_sizes:
            h = upsample(init_arr, n)
            h = project_onto_box_sum(h, n / 2.0)
            candidates.append(h)

    # Thue–Morse binary seeds (exact half ones for powers of two)
    for n in (64, 128, 256):
        if n in target_sizes:
            tm = thue_morse(n)
            candidates.append(project_onto_box_sum(tm, n / 2.0))

    # Random starts (continuous) for each size
    for n in target_sizes:
        for _ in range(2):
            v = rng.random(n)
            candidates.append(project_onto_box_sum(v, n / 2.0))
        # Random binary balanced seeds
        for _ in range(1):
            candidates.append(random_binary_balanced(n, rng))

    # ------------------------------------------------------------------
    # 2. Optimize each candidate with smoothed gradient descent
    # ------------------------------------------------------------------
    for h0 in candidates:
        if time.time() - start > budget_s - 2.0:
            break
        n = h0.size
        s = n / 2.0

        # iteration budgets per size (more for larger n)
        if n >= 256:
            total_iters = 2500
            alpha_end = 30.0
        elif n >= 180:
            total_iters = 3000
            alpha_end = 30.0
        elif n >= 128:
            total_iters = 2000
            alpha_end = 20.0
        else:
            total_iters = 1500
            alpha_end = 15.0
        alpha_start = 1.0
        factor = (alpha_end / alpha_start) ** (1.0 / max(total_iters - 1, 1))
        alpha = alpha_start

        h = h0.copy()
        # local best tracking
        local_best_h = h.copy()
        local_best_score = compute_c5(h)

        for it in range(1, total_iters + 1):
            grad = gradient_smooth(h, alpha)
            step = 0.4 / math.sqrt(it)
            h = project_onto_box_sum(h - step * grad, s)

            if it % 30 == 0 or it == total_iters:
                score = compute_c5(h)
                if score < local_best_score:
                    local_best_score = score
                    local_best_h = h.copy()
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
            alpha *= factor

        # Short exact PGD refinement
        refined_h, refined_score = exact_pgd_refine(local_best_h, n_iter=400, step0=0.5)
        if refined_score < best_score:
            best_score = refined_score
            best_h = refined_h.copy()
        if refined_score < local_best_score:
            local_best_score = refined_score
            local_best_h = refined_h.copy()

    # ------------------------------------------------------------------
    # 3. Random-restart loop for remaining time
    # ------------------------------------------------------------------
    while time.time() - start < budget_s - 2.0:
        n = rng.choice([180, 180, 256, 128, 64])
        s = n / 2.0
        h = project_onto_box_sum(rng.random(n), s)

        total_iters = 800
        alpha_start = 1.0
        alpha_end = 15.0
        factor = (alpha_end / alpha_start) ** (1.0 / max(total_iters - 1, 1))
        alpha = alpha_start

        for it in range(1, total_iters + 1):
            grad = gradient_smooth(h, alpha)
            step = 0.4 / math.sqrt(it)
            h = project_onto_box_sum(h - step * grad, s)
            alpha *= factor

            if it % 100 == 0:
                score = compute_c5(h)
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
        # final check
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()

    # ------------------------------------------------------------------
    # 4. Final projection and return
    # ------------------------------------------------------------------
    if best_h is None:
        # fallback uniform half-filled construction
        n = 180
        best_h = np.full(n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)
        n_points = n
    else:
        n_points = best_h.size
        best_h = project_onto_box_sum(best_h, n_points / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, n_points
