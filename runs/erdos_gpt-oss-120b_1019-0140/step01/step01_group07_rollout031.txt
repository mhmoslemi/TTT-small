def compute_corr_and_grad(h):
    n = len(h)
    r = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(r)
    k = idx_max - (n - 1)
    # compute grad
    # shift right by k: h_shift_right[i] = h[i + k] if i + k < n else 0
    if k >= 0:
        right = np.concatenate((h[k:], np.zeros(k)))
        left = np.concatenate((np.zeros(k), h[:n - k]))

def _subgradient_descent(h, target_sum, max_iters, time_end):
    n = h.size
    dx = 2.0 / n
    best_local_h = h.copy()
    best_local_score = _compute_c5(h)
    for it in range(1, max_iters+1):
        if time.time() > time_end:
            break
        # compute correlation spectrum and find worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(np.argmax(corr))
        lag = idx_max - (n - 1)
        # compute gradient
        if lag >= 0:
            right = np.concatenate((h[lag:], np.zeros(lag, dtype=h.dtype)))
            left = np.concatenate((np.zeros(lag, dtype=h.dtype), h[:n - lag]))
        else:
            k_abs = -lag
            right = np.concatenate((np.zeros(k_abs, dtype=h.dtype), h[:n - k_abs]))
            left = np.concatenate((h[k_abs:], np.zeros(k_abs, dtype=h.dtype)))
        grad = 1.0 - right - left
        # step size schedule
        alpha = 0.5 / math.sqrt(it)  # step0 0.5
        # update and project
        h_new = h - alpha * grad
        h_new = _project_to_constraints(h_new, target_sum)
        # compute score
        score = _compute_c5(h_new)
        # update best local
        if score < best_local_score:
            best_local_score = score
            best_local_h = h_new.copy()
        # early stop if below target
        if best_local_score < 0.3808:
            return best_local_h, best_local_score
        # set h = h_new for next iteration
        h = h_new
    return best_local_h, best_local_score

def random_swap_descent(h, target_sum, max_iters, step_scale, time_end):
    n = len(h)
    best_h = h.copy()
    best_score = _compute_c5(h)
    for t in range(max_iters):
        if time.time() > time_end:
            break
        # pick i, j
        # choose i where h_i < 1 and j where h_j > 0
        # we can randomly sample i, j and adjust if possible else skip
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # Determine direction: transfer delta from j to i
        # ensure h_i < 1 and h_j > 0
        if h[i] >= 1.0 or h[j] <= 0.0:
            continue
        max_delta = min(1.0 - h[i], h[j])
        # choose delta
        delta = max_delta * step_scale * np.random.rand()
        if delta <= 1e-12:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        score = _compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            h = h_new
        # Optionally, accept with some probability to escape
        # else maybe keep original h unchanged
    return best_h, best_score

import numpy as np
import time
import math

def _compute_c5(h):
    """Maximum overlap integral using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust to meet sum(v)=target by water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # fill the largest capacities first
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    # final safeguard scaling if still off
    cur = np.sum(v)
    if cur > 0 and abs(cur - target) > 1e-8:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _interpolate(src, n_new):
    """Linear interpolation of a 1‑D array onto a new grid of length n_new."""
    n_src = src.size
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_src, src)

def _random_move_descent(h0, target_sum, max_iters, step_scale, time_end):
    """Improves h by transferring mass between two coordinates."""
    h = h0.copy()
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    for it in range(1, max_iters + 1):
        if time.time() > time_end:
            break
        # linear decay of step magnitude
        step = step_scale * (1.0 - it / max_iters)
        inc_idx = np.where(h < 1.0)[0]
        dec_idx = np.where(h > 0.0)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = inc_idx[np.random.randint(inc_idx.size)]
        j = dec_idx[np.random.randint(dec_idx.size)]
        if i == j:
            continue
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * step * np.random.rand()
        if delta <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_score = _compute_c5(h_new)
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score

                if best_score < 0.3808:
