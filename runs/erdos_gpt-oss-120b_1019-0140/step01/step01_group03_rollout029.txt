import time, math
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of a 1‑D array to a new length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    # extreme cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Exact discretized C5 for the current step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the current worst‑shift objective."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def projected_gradient_descent(h0, n_iter=2000, step0=0.5):
    """PGD on the original max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h = h - eta * grad
        h = project_onto_box_sum(h, s)
        if it % 50 == 0:
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    return best_h, best_score

def smooth_gradient(h, beta):
    """Gradient of the soft‑max approximation to the max‑overlap objective."""
    # quadratic terms Q_k = sum_i h_i h_{i+k}
    Q = np.correlate(h, h, mode='full')
    # soft‑max weights over -beta*Q (focus on smallest Q)
    scores = -beta * Q
    m = np.max(scores)
    exp_shift = np.exp(scores - m)
    w = exp_shift / np.sum(exp_shift)               # length 2n‑1, symmetric
    # convolution with symmetric kernel gives Σ_k w_k h_{i+k}
    conv = np.convolve(h, w, mode='same')
    # gradient of the smooth objective (to be subtracted)
    return -2.0 * conv

def optimize_smooth(h0, time_allow):
    """Projected gradient descent on the smooth surrogate."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    step0 = 0.5
    beta = 5.0
    beta_inc_iter = 2000
    max_iters = 30000
    for it in range(1, max_iters + 1):
        if time.time() - start > time_allow:
            break
        if it % beta_inc_iter == 0:
            beta = min(beta * 2.0, 200.0)
        grad = smooth_gradient(h, beta)
        eta = step0 / math.sqrt(it)
        h = h - eta * grad
        h = project_onto_box_sum(h, s)
        if it % 100 == 0:
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    return best_h, best_score

def random_feasible(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

    rng = np.random.default_rng(seed)
    start_time = time.time()
    target_n = 256                     # primary resolution (<1k samples)
    half_sum = target_n / 2.0

    # ------------------------------------------------------------------
    # 1. Assemble a small pool of start points
    # ------------------------------------------------------------------
    starts = []

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != target_n:
            h0 = upsample(h0, target_n)
        h0 = project_onto_box_sum(h0, half_sum)
        starts.append(h0)

    # a few random feasible vectors
    for _ in range(6):
        starts.append(random_feasible(target_n, rng))

    best_h = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 2. Optimize each start point with the smooth surrogate
    # ------------------------------------------------------------------
    for h_start in starts:
        remaining = budget_s - (time.time() - start_time) - 0.5
        if remaining <= 0:
            break
        h_opt, sc = optimize_smooth(h_start, remaining)
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()

    # ------------------------------------------------------------------
    # 3. Random restarts until time runs out
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1.0:
            break
        h_rand = random_feasible(target_n, rng)
        remaining = budget_s - elapsed - 0.5
        h_opt, sc = optimize_smooth(h_rand, remaining)
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()

    # ------------------------------------------------------------------
    # 4. Final refinement with exact PGD
    # ------------------------------------------------------------------
    if best_h is not None:
        h_ref, sc_ref = projected_gradient_descent(best_h, n_iter=3000, step0=0.5)
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref

    # Ensure feasibility one last time
    best_h = project_onto_box_sum(best_h, half_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
