import time, numpy as np
from scipy.optimize import minimize, differential_evolution

def _repair_vector(v, n, max_iter=15):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final fix
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * (2.0 / n)

def _random_feasible(n):
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _local_search(v, time_limit):
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.1
    start = time.time()
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score, best_v = score, cand
    return best_v, best_score

def _parameter_opt(n, M, time_limit):
    # Build basis matrix
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    B = np.empty((n, M), dtype=np.float64)
    for m in range(1, M+1):
        col = np.cos(np.pi * m * x)
        col = col - col.mean()  # zero mean
        B[:, m-1] = col
    # objective for DE
    def obj(params):
        v = 0.5 + B @ np.asarray(params)
        v = _repair_vector(v, n)
        corr = np.correlate(v, 1.0 - v, mode='full')
        return np.max(corr) * (2.0 / n)
    # bounds
    bounds = [(-0.5, 0.5)] * M
    # allocate limited iterations based on time
    # Use relatively small popsize and maxiter to stay within time.
    # We set maxiter = max(10, int(time_limit // 0.3)) maybe.
    # but DE doesn't accept time limit directly; we just fine-tune.
    maxiter = max(10, int(time_limit / 0.5))  # rough estimate
    maxiter = min(maxiter, 200)  # caps
    popsize = 10
    # run DE
    result = differential_evolution(obj, bounds, strategy='best1bin',
                                    maxiter=maxiter, popsize=popsize,
                                    tol=1e-7, polish=False, init='latinhypercube')
    best_params = result.x
    best_v = 0.5 + B @ best_params
    best_v = _repair_vector(best_v, n)
    best_score = _compute_c5(best_v)
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):

import time
import numpy as np
from scipy.optimize import differential_evolution

def _repair_vector(v, n, max_iter=15):
    """Project v onto the box [0,1]^n while enforcing sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final corrective scaling / uniform adjustment
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Maximum overlap C5 for a given vector v."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * (2.0 / n)

def _random_feasible(n):
    """Generate a random feasible vector (box + sum constraint)."""
    return _repair_vector(np.random.rand(n), n)

def _local_search(v, time_limit):
    """Simple hill‑climbing: exchange small amounts between two coordinates."""
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.1
    start = time.time()
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score, best_v = score, cand
    return best_v, best_score

def _parameter_opt(n, M, time_limit):
    """
    Optimize a low‑dimensional Fourier‑type representation.
    h(x) = 0.5 + Σ_m a_m * (cos(mπx) - mean) + b * (x - mean_x)
    """
    # grid points on [0,2)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    # build zero‑mean basis columns
    B = np.empty((n, M + 1), dtype=np.float64)
    for m in range(1, M + 1):
        col = np.cos(np.pi * m * x)
        col -= col.mean()
        B[:, m - 1] = col
    # linear term (centred)
    lin = x - x.mean()
    B[:, M] = lin

    def obj(params):
        v = 0.5 + B @ np.asarray(params)
        v = _repair_vector(v, n)
        corr = np.correlate(v, 1.0 - v, mode='full')
        return np.max(corr) * (2.0 / n)

    bounds = [(-0.5, 0.5)] * (M + 1)

    # estimate a modest number of generations from the time budget
    est_iter = max(20, int(time_limit / 0.5))
    maxiter = min(est_iter, 200)
    popsize = 10

    result = differential_evolution(
        obj,
        bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=popsize,
        tol=1e-7,
        polish=False,
        init='latinhypercube',
        updating='deferred',
        disp=False,
    )
    best_params = result.x
    best_v = 0.5 + B @ best_params
    best_v = _repair_vector(best_v, n)
    best_score = _compute_c5(best_v)
    return best_v, best_score
