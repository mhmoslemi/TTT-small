def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)  # dx

def binarize_to_target(arr, target_sum):
    # arr is any vector; pick top target entries as 1 else 0
    # target_sum is integer number of ones needed.
    # returns binary 0/1 array length n.
    idx = np.argsort(arr)[-target_sum:]
    h = np.zeros_like(arr, dtype=np.float64)
    h[idx] = 1.0
    return h

def improve_at_resolution(h_init, n, target_ones, max_time):
    start = time.time()
    current = h_init.copy()
    cur_score = compute_c5(current)
    best = current.copy()
    best_score = cur_score
    # temperature schedule: start higher, decays to small
    T_start = 0.005
    T_end = 1e-6
    # maintain number of steps
    steps = 0
    while time.time() - start < max_time:
        # propose swap
        # find random i_one
        i_one = np.random.randint(n)
        while current[i_one] < 0.5:
            i_one = np.random.randint(n)
        i_zero = np.random.randint(n)
        while current[i_zero] > 0.5:
            i_zero = np.random.randint(n)
        # generate candidate
        cand = current.copy()
        cand[i_one] = 0.0
        cand[i_zero] = 1.0
        cand_score = compute_c5(cand)
        # acceptance
        elapsed = time.time() - start
        T = T_start * (1 - elapsed / max_time) + T_end
        # if better or accept worse
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / T):
            # accept
            current = cand
            cur_score = cand_score
            # update best if better
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                # early exit if target reached
                if best_score < 0.3808:
                    break
        # loop
        steps += 1
        # optional occasional random restart if many steps without improvement (e.g., 5000)
        # Not needed maybe.
    return best, best_score

import numpy as np

def compute_c5(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def binarize_to_target(arr, target):
    # flatten
    flat = arr.ravel()
    # get indices of largest target values (handle target 0/ len)
    if target <= 0:
        return np.zeros_like(flat, dtype=np.float64)
    if target >= flat.size:
        return np.ones_like(flat, dtype=np.float64)
    idx = np.argpartition(flat, -target)[-target:]
    bin_arr = np.zeros_like(flat, dtype=np.float64)
    bin_arr[idx] = 1.0
    return bin_arr

def improve_at_resolution(start_h, max_time):
    """Hill climbing with simulated annealing using swaps.
    Returns best solution and its score within max_time."""
    n = start_h.shape[0]
    target = n // 2
    current = start_h.copy()
    cur_score = compute_c5(current)
    best = current.copy()
    best_score = cur_score
    start = time.time()
    # Temperature schedule parameters
    T_start = 5e-3
    T_end = 1e-6
    while time.time() - start < max_time:
        # propose a swap
        # ensure we have both 0 and 1
        if target == 0 or target == n:
            break
        i_one = np.random.randint(n)
        while current[i_one] < 0.5:  # find a 1
            i_one = np.random.randint(n)
        i_zero = np.random.randint(n)
        while current[i_zero] > 0.5:  # find a 0
            i_zero = np.random.randint(n)
        cand = current.copy()
        cand[i_one] = 0.0
        cand[i_zero] = 1.0
        cand_score = compute_c5(cand)
        # cooling based on elapsed time
        elapsed = time.time() - start
        T = T_start * (1 - elapsed / max_time) + T_end
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # accept worse with probability
            if T > 0:
                if np.random.rand() < np.exp(-(cand_score - cur_score) / T):
                    accept = True
        if accept:
            current = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                if best_score < 0.3808:
                    # early exit
                    break
    return best, best_score

import numpy as np, time

def compute_c5(h):
    """Discrete C₅ for a step vector h on [0,2]."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def binarize_to_target(arr, target):
    """Return a binary {0,1} array of the same shape as arr with exactly
    `target` entries equal to 1 (chosen from the largest values of arr)."""
    flat = arr.ravel()
    if target <= 0:
        return np.zeros_like(flat, dtype=np.float64)
    if target >= flat.size:
        return np.ones_like(flat, dtype=np.float64)
    idx = np.argpartition(flat, -target)[-target:]
    bin_arr = np.zeros_like(flat, dtype=np.float64)
    bin_arr[idx] = 1.0
    return bin_arr

def improve_at_resolution(start_h, max_time):

    Returns the best vector seen within `max_time` seconds."""
