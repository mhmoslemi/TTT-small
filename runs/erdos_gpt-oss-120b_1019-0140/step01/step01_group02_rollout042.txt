def swap_update(h, corr, i, j):
    a = h[i]
    b = h[j]
    if a == b:
        # no change
        return np.max(corr) * (2.0 / h.shape[0])
    delta = b - a
    n = h.shape[0]
    lag0 = n - 1

    # first index i
    start_i = lag0 - i
    corr[start_i:start_i + (n - i)] += delta * (1.0 - h[i:])
    corr[lag0:lag0 + i + 1] -= delta * h[i::-1]

    # second index j (delta_j = -delta)
    delta_j = -delta
    start_j = lag0 - j
    corr[start_j:start_j + (n - j)] += delta_j * (1.0 - h[j:])
    corr[lag0:lag0 + j + 1] -= delta_j * h[j::-1]

    delta2 = delta * delta
    corr[lag0] -= 2.0 * delta2
    s = j - i
    if s != 0:
        corr[lag0 + s] += delta2
        corr[lag0 - s] += delta2

    # perform swap
    h[i] = b
    h[j] = a

    # return new C5
    c5 = float(np.max(corr) * (2.0 / n))
    return c5

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def _compute_c5(h):
    return _c5_and_corr(h)[0]

def _make_binary_feasible(n, rng):
    h = np.zeros(n, dtype=float)
    h[: n//2] = 1.0
    # if n odd, add 0.5 at middle
    if n % 2 == 1:
        half_idx = n//2
        h[half_idx] = 0.5
    rng.shuffle(h)
    return h

def _repair_sum(h, target):
    # adjust a single entry to match target sum
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # pick a mutable entry
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if idx.size == 0:
        # distribute uniformly as fallback
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    # if not fixed, recurse (unlikely)
    if abs(h.sum() - target) > 1e-12:
        return _repair_sum(h, target)
    return h

    delta_j = -delta
    start_j = lag0 - j
    corr[start_j : start_j + (n - j)] += delta_j * (1.0 - h[j:])
    corr[lag0 : lag0 + j + 1] -= delta_j * h[j::-1]

    # quadratic corrections
    delta2 = delta * delta
    corr[lag0] -= 2.0 * delta2           # self‑pair (i,i) and (j,j)
    s = j - i
    if s != 0:
        corr[lag0 + s] += delta2        # pair (i,j)
        corr[lag0 - s] += delta2        # pair (j,i)

    # perform the swap
    h[i] = b
    h[j] = a

    return float(np.max(corr) * (2.0 / n))

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf

    # ------------------------------------------------------------------
    # 1.  Use any supplied construction as a baseline.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        init = np.clip(init, 0.0, 1.0)
        init = _repair_sum(init, init.shape[0] / 2.0)
        c5_init = _compute_c5(init)
        best_h, best_c5 = init.copy(), c5_init
    except NameError:
        init = None

    # ------------------------------------------------------------------
    # 2.  Small integer programme (binary, even length) – fast and exact.
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > 30.0:
        n_milp = 48  # even, modest size
        y = cp.Variable(n_milp, boolean=True)
        t = cp.Variable()
        constraints = [cp.sum(y) == n_milp / 2.0, t >= 0]

        # overlap variables for each shift
        for s in range(n_milp):
            length = n_milp - s
            z = cp.Variable(length, boolean=True)
            constraints += [
                z <= y[:length],
                z <= y[s : s + length],
                z >= y[:length] + y[s : s + length] - 1,
                t <= cp.sum(z),
            ]

        prob = cp.Problem(cp.Maximize(t), constraints)
        try:
            prob.solve(solver=cp.CBC, verbose=False, max_iters=200000)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK_MI, verbose=False, max_iters=200000)
            except Exception:
                prob = None

        if prob is not None and prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
            y_val = np.rint(y.value).astype(float)
            y_val = _repair_sum(y_val, n_milp / 2.0)
            c5_milp = _compute_c5(y_val)
            if c5_milp < best_c5:
                best_h, best_c5 = y_val.copy(), c5_milp

    # ------------------------------------------------------------------
    # 3.  Swap‑based simulated annealing (fast O(n) updates).
    # ------------------------------------------------------------------
    if best_h is not None:
        h = best_h.copy()
    else:
        h = _make_binary_feasible(200, rng)

    n = h.shape[0]
    dx = 2.0 / n
    _, corr = _c5_and_corr(h)
    current_c5 = float(np.max(corr) * dx)

    best_local_h = h.copy()
    best_local_c5 = current_c5

    max_temp = 0.02
    min_temp = 1e-6
    max_no_improve = 20000
    no_improve = 0

    while True:
        now = time.time()
        if now - start_time >= budget_s:
            break
        frac = (now - start_time) / budget_s
        temp = max_temp * (1.0 - frac) + min_temp * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        new_c5 = _swap_update(h, corr, i, j)
        delta = new_c5 - current_c5

        if delta < 0 or rng.random() < math.exp(-delta / temp):
            # accept move
            current_c5 = new_c5
            if new_c5 < best_local_c5:
                best_local_c5 = new_c5
                best_local_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # reject – undo by swapping again
            _swap_update(h, corr, i, j)
            no_improve += 1

        if no_improve >= max_no_improve:
            # restart from current best to diversify
            h = best_local_h.copy()
            _, corr = _c5_and_corr(h)
            current_c5 = best_local_c5
            no_improve = 0

    # final safe‑guard repair
    final_h = _repair_sum(best_local_h, n / 2.0)
    final_c5 = _compute_c5(final_h)

    # Return the best of MILP / SA / initial
    if final_c5 < best_c5:
        return final_h, final_c5, n
    else:
        return best_h, best_c5, best_h.shape[0]
