def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # load parent if available
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        # fallback: simple half ones.
        n0 = 200
        parent = np.zeros(n0)
        parent[: n0//2] = 1.0
    n0 = len(parent)
    parent = _repair_vector(parent, n0)
    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    # define resolution schedule
    # Ensure max size <= 800 for speed
    res = [n0]
    if n0*2 <= 800:
        res.append(n0*2)
    if n0*4 <= 800:
        res.append(n0*4)

    # allocate total time minus small buffer
    total_budget = max(0.0, budget_s - 3.0)
    per_res = total_budget / len(res) if res else 0.0

    for n in res:
        # time check break if insufficient
        elapsed = time.time() - start
        if elapsed >= budget_s - 1.0: break

        # upscale init guess if needed
        if len(best_v) != n:
            init = _upsample_vector(best_v, n)
        else:
            init = best_v.copy()

        # allocate time for this resolution
        remaining = budget_s - elapsed - 1.0
        time_for_res = min(per_res, remaining)

        # SLSQP phase (maybe one call)
        slsqp_time = time_for_res * 0.4
        if slsqp_time > 0.1:
            v_opt, sc_opt = _optimize_at_resolution(n, init, slsqp_time)
            if sc_opt < best_score:
                best_score, best_v = sc_opt, v_opt.copy()
                best_n = n

        # Simulated annealing phase
        sa_time = time_for_res - slsqp_time
        if sa_time > 0.2:
            v_sa, sc_sa = _simulated_annealing(best_v, sa_time)
            if sc_sa < best_score:
                best_score, best_v = sc_sa, v_sa.copy()
                best_n = n

        # final local hill climbing short phase
        lc_time = max(0.0, time_for_res - slsqp_time - sa_time)
        if lc_time > 0.1:
            v_lc, sc_lc = _local_search(best_v, n, lc_time)
            if sc_lc < best_score:
                best_score, best_v = sc_lc, v_lc.copy()
                best_n = n

    # final repair and evaluation
    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)
    return best_v.astype(np.float64), float(best_score), int(best_n)

    current_v = v0.copy()
    current_score = _compute_c5(current_v)
    best_v = current_v.copy()
    best_score = current_score

import time
import math
import numpy as np
from scipy.optimize import minimize

def _repair_vector(v, n, max_iter=15):
    """Project vector onto [0,1]^n with exact sum n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Overlap bound for a discrete step function."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation to a finer grid, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Uniform random feasible vector (continuous)."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _random_bin_feasible(n):
    """Random half‑ones binary vector."""
    v = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, n // 2, replace=False)
    v[ones] = 1.0
    return v

def _optimize_at_resolution(n, init_v, time_limit):
    """SLSQP refinement with occasional random restarts."""
    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _compute_c5(x)

    # primary run
    try:
        res = minimize(
            obj,
            init_v,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 500, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            v = _repair_vector(res.x, n)
            sc = _compute_c5(v)
            if sc < best_score:
                best_score, best_v = sc, v
    except Exception:
        pass

    # random restarts while time permits
    while time.time() - start < time_limit:
        rnd = _random_feasible(n)
        try:
            res = minimize(
                obj,
                rnd,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n)
                sc = _compute_c5(v)
                if sc < best_score:
                    best_score, best_v = sc, v
        except Exception:
            pass
        if time.time() - start > 0.9 * time_limit:
            break
    return best_v, best_score

def _local_search(v, n, time_limit):
    """Hill climbing by exchanging small amounts between two coordinates."""
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        sc = _compute_c5(cand)
        if sc < best_score:
            best_score, best_v = sc, cand
    return best_v, best_score

def _simulated_annealing(v0, time_limit):
    """Simple SA with a mix of delta moves and swaps."""
    start = time.time()
    current_v = v0.copy()
    current_score = _compute_c5(current_v)
    best_v = current_v.copy()
    best_score = current_score
    n = len(v0)
    T = 0.01  # initial temperature
    step_max = 0.2
    decay = 0.9995
    iter_count = 0
    while time.time() - start < time_limit:
        iter_count += 1
        if np.random.rand() < 0.3:
            # swap move
            i, j = np.random.choice(n, 2, replace=False)
            cand = current_v.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 2.0 * step_max
            cand = current_v.copy()
            cand[i] += delta
            cand[j] -= delta
            cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        dE = cand_score - current_score
        if dE < 0 or np.random.rand() < math.exp(-dE / max(T, 1e-12)):
            current_v, current_score = cand, cand_score
            if cand_score < best_score:
                best_v, best_score = cand, cand_score
        T *= decay
        if iter_count % 1000 == 0:
            step_max = max(0.01, step_max * 0.99)
    return best_v, best_score

    """Optimize a step function for the Erdős C₅ problem."""
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # Load parent construction if present
    try:
        parent = np.array(initial_h_values, dtype=np.float64)  # pre‑imported
        n0 = parent.shape[0]
    except Exception:
        # Fallback: simple balanced binary vector
        n0 = 200
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0 // 2] = 1.0
    parent = _repair_vector(parent, n0)

    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    # ------------------------------------------------------------------
    # Resolution schedule (stay ≤800 points for speed)
    resolutions = [n0]
    if n0 * 2 <= 800:
        resolutions.append(n0 * 2)
    if n0 * 4 <= 800:
        resolutions.append(n0 * 4)

    total_budget = max(0.0, budget_s - 5.0)  # reserve a few seconds for final steps
    per_res = total_budget / max(1, len(resolutions))

    for n in resolutions:
        # Check remaining time
        elapsed = time.time() - start
        if elapsed >= budget_s - 1.0:
            break

        # Initialise guess at this resolution
        if len(best_v) != n:
            init_guess = _upsample_vector(best_v, n)
        else:
            init_guess = best_v.copy()

        # Time allocation for this resolution
        remaining = budget_s - elapsed - 1.0
        time_for_res = min(per_res, remaining)

        # ---------- SLSQP phase ----------
        slsqp_time = time_for_res * 0.4
        if slsqp_time > 0.2:
            v_opt, sc_opt = _optimize_at_resolution(n, init_guess, slsqp_time)
            if sc_opt < best_score:
                best_score, best_v, best_n = sc_opt, v_opt, n

        # ---------- Simulated annealing ----------
        sa_time = max(0.0, time_for_res - slsqp_time)
        if sa_time > 0.3:
            v_sa, sc_sa = _simulated_annealing(best_v, sa_time)
            if sc_sa < best_score:
                best_score, best_v, best_n = sc_sa, v_sa, n

        # ---------- Short local refinement ----------
        lc_time = max(0.0, time_for_res - slsqp_time - sa_time)
        if lc_time > 0.2:
            v_lc, sc_lc = _local_search(best_v, best_n, lc_time)
            if sc_lc < best_score:
                best_score, best_v, best_n = sc_lc, v_lc, best_n

    # Final repair and evaluation
    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(best_n)
