def _repair_vector(v, n, max_iter=15):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

import time
import numpy as np

def _repair_vector(v, n):
    """Clip to [0,1] and adjust sum to n/2 (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # tiny uniform shift keeps values within bounds
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        diff = target - s
        # if clipping altered the sum, do a few projection steps
        if abs(diff) > 1e-12:
            for _ in range(10):
                mask = (v > 0) & (v < 1)
                free = mask.sum()
                if free == 0:
                    break
                adj = diff / free
                v[mask] += adj
                v = np.clip(v, 0.0, 1.0)
                s = v.sum()
                diff = target - s
                if abs(diff) < 1e-12:
                    break
    return v

def _cosine_step(n):
    """Construct h_i = (1+cos(2π i / n))/2 on [0,2]."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / n        # points in [0,2)
    return (1.0 + np.cos(np.pi * x)) / 2.0

def _compute_c5(v):
    """Compute C5 = max_k dx * sum_i v_i (1 - v_{i+k}) using full correlation."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full") * dx
    return float(np.max(corr))
