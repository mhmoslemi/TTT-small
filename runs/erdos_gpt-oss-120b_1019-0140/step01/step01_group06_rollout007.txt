def compute_c5(h):
    n = h.shape[0]
    # compute correlation of h with (1-h) using FFT
    m = 2*n - 1
    f_h = np.fft.rfft(h, n=m)
    f_one_minus_h = np.fft.rfft(1.0 - h, n=m)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    # linear correlation via FFT
    m = 2*n - 1
    # Use rfft for real input
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * np.conj(G), n=m)
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=200):
    # Simple projection onto box [0,1] and hyperplane sum = target_sum
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        # active set: indices not at bounds
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # distribute to the variable with most slack
            # find index with largest distance to bound in direction of diff
            if diff > 0:
                i = np.argmax(1.0 - h)
                delta = min(diff, 1.0 - h[i])
                h[i] += delta
            else:
                i = np.argmin(h)
                delta = max(diff, -h[i])
                h[i] += delta
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    s = np.sum(h)
    if abs(s - target_sum) > 1e-8:
        # adjust a single interior element if possible
        i = np.argmin(np.abs(h - 0.5))  # interior roughly
        h[i] = np.clip(h[i] + (target_sum - s), 0.0, 1.0)
    return h

def resample_h(old_h, n_new):
    old_h = np.asarray(old_h, dtype=float)
    n_old = old_h.shape[0]
    if n_new == n_old:
        return project_to_feasible(old_h.copy(), n_new/2.0)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    return project_to_feasible(new_h, n_new/2.0)

def random_lowfreq_h(n, rng, num_freq=5, scale=0.3):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for k in range(1, num_freq+1):
        amp = rng.normal(scale=scale)
        phase = rng.uniform(0, 2*np.pi)
        h += amp * np.sin(2*np.pi*k*x/2 + phase)
    # shift to positive
    h = h - np.min(h)
    # normalize to [0,1] then project to feasible
    h = np.clip(h, 0.0, None)
    return project_to_feasible(h, n/2.0)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Maximum overlap integral C5 for a sampled step function."""
    n = h.shape[0]
    m = 2 * n - 1
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * np.conj(G), n=m)
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v, target_sum, tol=1e-9, max_iter=200):
    """Project onto box [0,1]^n and hyperplane sum(v)=target_sum."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(v)
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            # No free variable: push the one with most slack
            if diff > 0:
                i = np.argmax(1.0 - v)
                delta = min(diff, 1.0 - v[i])
            else:
                i = np.argmin(v)
                delta = max(diff, -v[i])
            v[i] += delta
            v = np.clip(v, 0.0, 1.0)
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny fix
    s = np.sum(v)
    if abs(s - target_sum) > 1e-8:
        i = np.argmin(np.abs(v - 0.5))
        v[i] = np.clip(v[i] + (target_sum - s), 0.0, 1.0)
    return v


def resample_h(old_h, n_new):
    """Interpolate old_h to length n_new and re‑project to feasibility."""
    old_h = np.asarray(old_h, dtype=float)
    n_old = old_h.shape[0]
    if n_new == n_old:
        return project_to_feasible(old_h.copy(), n_new / 2.0)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    return project_to_feasible(new_h, n_new / 2.0)


def random_lowfreq_h(n, rng, num_freq=5, scale=0.3):
    """Generate a smooth random shape by summing a few low‑frequency sines."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for k in range(1, num_freq + 1):
        amp = rng.normal(scale=scale)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * k * x / 2 + phase)
    h -= h.min()                     # shift to non‑negative
    return project_to_feasible(h, n / 2.0)


def binary_from_h(h):
    """Return a binary vector with exactly n/2 ones, taking the largest entries of h."""
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(-h, k - 1)[:k]
    b = np.zeros(n)
    b[idx] = 1.0
    return b


def improve_binary(b, max_iter, rng):
    """Simple hill‑climbing: swap a 1 with a 0 if it lowers C5."""
    n = b.shape[0]
    best = b.copy()
    best_val = compute_c5(best)
    ones = np.where(best == 1)[0]
    zeros = np.where(best == 0)[0]
    for _ in range(max_iter):
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = compute_c5(cand)
        if val < best_val:
            best = cand
            best_val = val
            # update index sets
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
    return best, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h minimizing the Erdős C5 constant."""
    start = time.time()
    rng = np.random.default_rng(seed)

    base_h_global = globals().get("initial_h_values")
    n_list = [190, 380, 760, 1520]          # increasing resolutions

    best_h = None
    best_c5 = float("inf")
    best_n = None

    for n in n_list:
        if time.time() - start > budget_s * 0.97:
            break
        target = n / 2.0

        # initialise at this resolution
        if best_h is not None:
            init0 = resample_h(best_h, n)
        elif base_h_global is not None and base_h_global.shape[0] == n:
            init0 = project_to_feasible(np.asarray(base_h_global, dtype=float), target)
        elif base_h_global is not None:
            init0 = resample_h(base_h_global, n)
        else:
            init0 = None

        time_left = budget_s - (time.time() - start)
        # heuristic number of restarts (at least 5, at most 30)
        max_restarts = max(5, min(30, int(time_left / (0.15 * n))))

        for r in range(max_restarts):
            if time.time() - start > budget_s * 0.97:
                break

            if r == 0 and init0 is not None:
                h0 = init0.copy()
                h0 += rng.normal(scale=0.01, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_to_feasible(h0, target)
            else:
                if rng.random() < 0.5:
                    h0 = rng.random(n)
                else:
                    h0 = random_lowfreq_h(n, rng, num_freq=6, scale=0.4)
                h0 = project_to_feasible(h0, target)

            cons = {"type": "eq", "fun": lambda h, nd=n: np.sum(h) - nd / 2.0}
            bnds = [(0.0, 1.0)] * n

            try:
                res = opt.minimize(
                    fun=compute_c5,
                    x0=h0,
                    method="SLSQP",
                    bounds=bnds,
                    constraints=cons,
                    options={"maxiter": 1200, "ftol": 1e-12, "disp": False},
                )
            except Exception:
                continue

            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_to_feasible(h_opt, target)
            c5_val = compute_c5(h_opt)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

    # ---------- binary post‑processing ----------
    if best_h is not None:
        # deterministic binary from current best
        b0 = binary_from_h(best_h)
        c5b = compute_c5(b0)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b0
            best_n = b0.shape[0]

        # hill‑climb while time permits
        remaining = budget_s - (time.time() - start)
        if remaining > 0.5:
            # estimate ~0.001 s per iteration
            iter_limit = int(min(5000, remaining / 0.001))
            cand, cand_val = improve_binary(best_h.copy(), iter_limit, rng)
            if cand_val < best_c5:
                best_h = cand
                best_c5 = cand_val
                best_n = cand.shape[0]

    # Fallback if nothing was found
    if best_h is None:
        n = 190
        best_h = project_to_feasible(np.full(n, 0.5), n / 2.0)
        best_n = n
        best_c5 = compute_c5(best_h)

    # Final safety projection and recompute
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
