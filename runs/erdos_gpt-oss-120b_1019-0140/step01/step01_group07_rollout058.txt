def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    f = 1.0 - h
    # linear convolution: pad to length 2*n
    size = 2*n
    H = np.fft.rfft(h, n=size)
    F = np.fft.rfft(f[::-1], n=size) # reversed
    corr = np.fft.irfft(H * F, n=size)
    # result length size, but convolution length is 2*n-1, take relevant part:
    corr = corr[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * dx

def compute_correlation(h):
    n = h.size
    f = 1.0 - h
    # Determine FFT length as next power of two >= 2*n-1
    m = 2 * n - 1
    fft_len = 1 << (m - 1).bit_length()
    H = np.fft.rfft(h, n=fft_len)
    F_rev = np.fft.rfft(f[::-1], n=fft_len)
    corr = np.fft.irfft(H * F_rev, n=fft_len)
    # Slice to length 2*n-1
    return corr[:m]

def project_to_constraints(v, target):
    # Clip v to [0,1]
    v = np.clip(v, 0, 1)
    current = v.sum()
    if abs(current - target) < 1e-12:
        return v
    # adjust residual linearly: we need to add or subtract same total from all entries that are not at bounds
    free_idx = np.where((v > 0) & (v < 1))[0]
    # If no free entries, we adjust all proportionally
    if free_idx.size == 0:
        # scale all (since sum is off)
        v = v * (target / current)
        return np.clip(v, 0, 1)
    # Determine needed amount
    diff = target - current
    # Distribute diff equally among free entries, but ensure bounds not violated.
    while True:
        if free_idx.size == 0:
            break
        delta = diff / free_idx.size
        # compute potential new values
        new_vals = v[free_idx] + delta
        # Identify values that would exceed bounds after delta
        over = new_vals > 1
        under = new_vals < 0
        # For those exceed, set to bound and adjust diff accordingly, remove from free set
        if np.any(over):
            over_idx = free_idx[over]
            diff -= np.sum(1 - v[over_idx])
            v[over_idx] = 1.0
            free_idx = np.setdiff1d(free_idx, over_idx, assume_unique=True)
            continue
        if np.any(under):
            under_idx = free_idx[under]
            diff -= np.sum(-v[under_idx])
            v[under_idx] = 0.0
            free_idx = np.setdiff1d(free_idx, under_idx, assume_unique=True)
            continue
        # No bounds violations
        v[free_idx] = new_vals
        break
    # Ensure sum close
    return v

import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project(v, target):
    """Clip to [0,1] and adjust sum(v) to target by water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    # simple iterative redistribution
    for _ in range(10):
        if cur < target:
            need = target - cur
            free = np.where(v < 1.0)[0]
            if free.size == 0:
                break
            caps = 1.0 - v[free]
            total = np.sum(caps)
            if total <= 0:
                break
            add = np.minimum(caps, need * caps / total)
            v[free] += add
        else:
            excess = cur - target
            free = np.where(v > 0.0)[0]
            if free.size == 0:
                break
            caps = v[free]
            total = np.sum(caps)
            if total <= 0:
                break
            sub = np.minimum(caps, excess * caps / total)
            v[free] -= sub
        cur = np.sum(v)
        if abs(cur - target) < 1e-12:
            break
    # final scaling if needed
    cur = np.sum(v)
    if abs(cur - target) > 1e-8:
        v = v * (target / cur)
    return np.clip(v, 0.0, 1.0)

def _eq_sum_constraint(x, target):
    return np.sum(x) - target
