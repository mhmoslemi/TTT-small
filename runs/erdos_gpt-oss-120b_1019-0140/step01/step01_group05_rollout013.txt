def simulated_annealing_targeted(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = len(h_cur)
    start = time.time()
    T0 = 0.05
    eps0 = 0.5
    eps_min = 1e-4
    p_target = 0.3
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        progress = 1.0 - elapsed / time_limit
        T = T0 * progress
        max_eps = eps0 * progress + eps_min
        # decide move type
        if rng.random() < p_target:
            # target move
            # compute corr and worst shift
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            idx_max = np.argmax(corr)
            ell = idx_max - (N-1)
            # compute a array
            if ell >= 0:
                partner = np.empty(N)
                partner[:ell] = 0.
                partner[ell:] = h_cur[:N-ell]
            else:
                ell_abs = -ell
                partner = np.empty(N)
                partner[:N-ell_abs] = h_cur[ell_abs:]
                partner[N-ell_abs:] = 0.
            a = 1.0 - partner
            # select source i
            src_mask = h_cur > 1e-12
            if not np.any(src_mask):
                continue
            src_indices = np.where(src_mask)[0]
            src_a = a[src_indices]
            i = src_indices[np.argmax(src_a)]
            # dest j
            dst_mask = h_cur < 1.0 - 1e-12
            if not np.any(dst_mask):
                continue
            dst_indices = np.where(dst_mask)[0]
            dst_a = a[dst_indices]
            j = dst_indices[np.argmin(dst_a)]
            # ensure beneficial
            if a[i] <= a[j] + 1e-12:
                # not beneficial: fallback to random move
                use_random = True
            else:
                # compute epsilon
                max_transfer = min(h_cur[i], 1.0 - h_cur[j])
                if max_transfer <= 0:
                    continue
                eps = max_eps * rng.random()
                eps = min(eps, max_transfer)
                if eps <= 0:
                    continue
                # apply move
                h_new = h_cur.copy()
                h_new[i] -= eps
                h_new[j] += eps
                # evaluate
                new_score = c5_value(h_new)
                delta = new_score - cur_score
                if delta < 0 or rng.random() < math.exp(-delta/(T+1e-15)):
                    h_cur = h_new
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h_new.copy()
                continue
        # random move
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        # max epsilon
        max_transfer = up + down
        eps = rng.random() * max_transfer - down  # uniform in [-down, up]
        # clip magnitude
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # ensure bounds
        if not (0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0):
            continue
        # evaluate
        new_score = c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta/(T+1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def repair_vec(v):
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        v = np.clip(v, 0.0, 1.0)
        return v
    # First scale
    v = v * (target / s)
    v = np.clip(v, 0.0, 1.0)
    # Might need second scaling
    s2 = v.sum()
    if abs(s2 - target) > 1e-9:
        v = v * (target / s2)
        v = np.clip(v, 0.0, 1.0)
    return v

import numpy as np

def repair_vec(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to exactly len(v)/2."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        return np.clip(v, 0.0, 1.0)
    # first scaling
    v *= target / s
    v = np.clip(v, 0.0, 1.0)
    # second scaling if needed
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        v *= target / s2
        v = np.clip(v, 0.0, 1.0)
    return v

def c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _target_move(h_cur: np.ndarray, max_eps: float, rng: np.random.Generator):
    """Perform a move guided by the current worst shift."""
    N = h_cur.size
    corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    idx = np.argmax(corr)
    ell = idx - (N - 1)                     # shift index
    # partner values h[i-ell] with zero padding outside range
    if ell >= 0:
        partner = np.empty(N)
        partner[:ell] = 0.0
        partner[ell:] = h_cur[:N - ell]
    else:
        ell_abs = -ell
        partner = np.empty(N)
        partner[:N - ell_abs] = h_cur[ell_abs:]
        partner[N - ell_abs:] = 0.0
    a = 1.0 - partner                       # contribution factor for this shift
    src_mask = h_cur > 1e-12
    dst_mask = h_cur < 1.0 - 1e-12
    if not (np.any(src_mask) and np.any(dst_mask)):
        return None, None, None
    src_idx = np.where(src_mask)[0]
    dst_idx = np.where(dst_mask)[0]
    i = src_idx[np.argmax(a[src_idx])]
    j = dst_idx[np.argmin(a[dst_idx])]
    if a[i] <= a[j] + 1e-12:
        return None, None, None            # not beneficial for this shift
    max_transfer = min(h_cur[i], 1.0 - h_cur[j])
    if max_transfer <= 0.0:
        return None, None, None
    eps = max_eps * rng.random()
    eps = min(eps, max_transfer)
    if eps <= 0.0:
        return None, None, None
    h_new = h_cur.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new, i, j

    h_cur = h_start.copy()

    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.05               # initial temperature
    eps0 = 0.5              # initial move magnitude
    eps_min = 1e-4
    p_target = 0.3          # probability of using a guided move
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        prog = 1.0 - elapsed / time_limit
        T = T0 * prog
        max_eps = eps0 * prog + eps_min

        if rng.random() < p_target:
            h_new, i, j = _target_move(h_cur, max_eps, rng)
            if h_new is not None:
                new_score = c5_value(h_new)
                delta = new_score - cur_score
                if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
                    h_cur = h_new
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h_new.copy()
                continue  # skip random move this iteration

        #--- random pair move -------------------------------------------------
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)       # max positive epsilon
        down = min(hi, 1.0 - hj)     # max negative epsilon
        if up == 0.0 and down == 0.0:

        eps = rng.random() * (up + down) - down   # uniform in [-down, up]

            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # safety bounds (should already hold)
        if not (0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0):
            continue
        new_score = c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
