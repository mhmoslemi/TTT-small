def solve_milp_binary(n, target, time_limit):
    import cvxpy as cp
    x = cp.Variable(n, boolean=True)
    # Y variable only for pairs i<j
    y = {}
    constraints = []
    for i in range(n):
        for j in range(i+1, n):
            y_ij = cp.Variable(boolean=True)
            y[(i,j)] = y_ij
            constraints += [y_ij <= x[i],
                            y_ij <= x[j],
                            y_ij >= x[i] + x[j] - 1]
    # sum constraint
    constraints.append(cp.sum(x) == target)

    # variable C
    C = cp.Variable()
    constraints.append(C >= 0)

    # shift constraints
    for k in range(1, n):
        # t_pos = sum_{i=0}^{n-1-k} x[i]
        t_pos = cp.sum(x[:n - k])
        # s_k = sum_{i=0}^{n-1-k} y_{i,i+k}
        s_k = cp.sum([ y[(i,i+k)] for i in range(n - k) ])
        constraints.append(C >= t_pos - s_k)
        # t_neg = sum_{i=k}^{n-1} x[i]
        t_neg = cp.sum(x[k:])
        constraints.append(C >= t_neg - s_k)
    # objective
    prob = cp.Problem(cp.Minimize(C), constraints)
    # solve with time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_iters=200000, time_limit=time_limit)
    except Exception as e:
        # fallback solve without time limit?
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        # may still have feasible solution
        pass
    # retrieve solution
    x_val = np.array(x.value).flatten()
    if x_val is None:
        return None, None
    # round to 0/1
    x_bin = np.rint(x_val).astype(int)
    # ensure sum correct if rounding errors
    current_sum = x_bin.sum()
    diff = target - current_sum
    if diff != 0:
        # adjust
        if diff > 0:
            # need to set diff zeros to 1
            zero_idxs = np.where(x_bin == 0)[0]
            if len(zero_idxs) >= diff:
                x_bin[np.random.choice(zero_idxs, diff, replace=False)] = 1
        else:
            one_idxs = np.where(x_bin == 1)[0]
            if len(one_idxs) >= -diff:
                x_bin[np.random.choice(one_idxs, -diff, replace=False)] = 0
    C_val = C.value
    return x_bin.astype(float), float(C_val)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max()) * dx

def make_binary_vector(n: int, target: int, rng: np.random.RandomState) -> np.ndarray:
    """Random binary vector of length n with exactly target ones."""
    vec = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=target, replace=False)
    vec[ones] = 1.0
    return vec

def max_corr_and_shift(h: np.ndarray):
    """Return (max_corr, shift) where shift = index - (n-1)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (n - 1), corr

def improve_vector(h: np.ndarray, rng: np.random.RandomState,
                   time_limit: float, init_c5: float):
    """
    Greedy repair: repeatedly target the worst shift and swap a mismatched
    1 with a mismatched 0 to lower the maximal correlation.
    Returns the best vector and its C5 seen during the allotted time.
    """
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = init_c5
    start = time.time()
    while time.time() - start < time_limit:
        max_corr, shift, _ = max_corr_and_shift(h)
        cur_c5 = max_corr * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # stop if already optimal (cannot be improved below zero)
        if max_corr == 0:
            break
        s = shift
        if s >= 0:
            # forward shift
            onezero = np.where((h[:n - s] == 1) & (h[s:] == 0))[0]
            zerone = np.where((h[:n - s] == 0) & (h[s:] == 1))[0]
        else:
            s = -s
            # backward shift
            onezero = np.where((h[s:] == 1) & (h[:n - s] == 0))[0] + s
            zerone = np.where((h[s:] == 0) & (h[:n - s] == 1))[0] + s
        if onezero.size == 0:
            break
        if zerone.size > 0:
            i = rng.choice(onezero)
            j = rng.choice(zerone)
        else:
            i = rng.choice(onezero)
            zeros = np.where(h == 0)[0]
            if zeros.size == 0:
                break
            j = rng.choice(zeros)
            if i == j:
                continue
        # swap values (keeps total sum unchanged)
        tmp = h[i]
        h[i] = h[j]
        h[j] = tmp
    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    max_time = budget_s - 0.5         # safety margin

    # Choose problem size
    n_init = None
    if 'initial_h_values' in globals():
        try:
            n_init = int(len(initial_h_values))
        except Exception:
            n_init = None
    n_points = 200
    if n_init is not None and n_init > n_points:
        n_points = n_init
    if n_points % 2 != 0:
        n_points += 1
    target_sum = n_points // 2

    # Generate initial candidates
    candidates = []

    # Sturmian style sequences with several irrationals
    alphas = [(np.sqrt(5) - 1) / 2, np.sqrt(2) - 1, 0.6180339887498949]
    for alpha in alphas:
        seq = ((np.mod(np.arange(n_points) * alpha, 1) < 0.5).astype(float))
        diff = int(target_sum - seq.sum())
        if diff > 0:
            zeros = np.where(seq == 0)[0]
            seq[rng.choice(zeros, diff, replace=False)] = 1
        elif diff < 0:
            ones = np.where(seq == 1)[0]
            seq[rng.choice(ones, -diff, replace=False)] = 0
        candidates.append(seq)

    # A few random binary starts
    for _ in range(2):
        candidates.append(make_binary_vector(n_points, target_sum, rng))

    # Use provided parent solution if compatible
    if 'initial_h_values' in globals():
        try:
            init = np.array(initial_h_values, dtype=float)
            if init.size == n_points:
                bin_vec = (init >= 0.5).astype(float)
                diff = int(target_sum - bin_vec.sum())
                if diff > 0:
                    zeros = np.where(bin_vec == 0)[0]
                    bin_vec[rng.choice(zeros, diff, replace=False)] = 1
                elif diff < 0:
                    ones = np.where(bin_vec == 1)[0]
                    bin_vec[rng.choice(ones, -diff, replace=False)] = 0
                candidates.append(bin_vec)
        except Exception:
            pass

    # Track global best
    best_h = None
    best_c5 = float('inf')

    # Main loop over candidates, each gets a slice of remaining time
    cand_idx = 0
    while time.time() - start_time < max_time:
        # allocate a small time slice for each candidate
        remaining = max_time - (time.time() - start_time)
        if remaining <= 0:

        slice_time = min(0.2, remaining)  # seconds per improvement round
        h = candidates[cand_idx]

        candidates[cand_idx] = improved_h  # keep improved version for next round
        if improved_c5 < best_c5:
            best_c5 = improved_c5
            best_h = improved_h.copy()
        cand_idx = (cand_idx + 1) % len(candidates)

    # If still no solution (unlikely), fall back to a simple random binary vector
    if best_h is None:
        best_h = make_binary_vector(n_points, target_sum, rng)
        best_c5 = compute_c5(best_h)

    # Final sanity: ensure exact mass
    cur_sum = best_h.sum()
    diff = int(target_sum - cur_sum)
    if diff > 0:
        zeros = np.where(best_h == 0)[0]
        best_h[rng.choice(zeros, diff, replace=False)] = 1
    elif diff < 0:
        ones = np.where(best_h == 1)[0]
        best_h[rng.choice(ones, -diff, replace=False)] = 0

    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
