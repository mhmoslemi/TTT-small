    n = len(h)
    best_h = h.copy()
    best_c = cur_c
    curr_h = h.copy()
    curr_c = cur_c
    # Determine schedule
    total_time = end_time - start_time
    # set max_temp
    max_temp = 0.001  # adjust as needed
    # use geometric cooling per iteration
    iter_count = 0
    while time.time() < end_time:
        elapsed = time.time() - start_time
        # linear schedule
        T = max_temp * (1 - elapsed / total_time)
        if T <= 0:
            T = 1e-12
        # pick random swap
        ones = np.flatnonzero(curr_h == 1)
        zeros = np.flatnonzero(curr_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        curr_h[i], curr_h[j] = 0, 1
        new_c = compute_c5_val(curr_h)
        delta = new_c - curr_c
        accept = False
        if delta < 0:
            accept = True
        else:
            # acceptance probability
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            curr_c = new_c
            # update best if improved
            if curr_c < best_c:
                best_c = curr_c
                best_h = curr_h.copy()
            # continue with current
        else:
            # revert
            curr_h[i], curr_h[j] = 1, 0
        iter_count += 1
    return best_h, best_c

        candidates.append(h_bin)
    # add some random seeds
    for _ in range(5):
        candidates.append(rand_bin())
    best_h = None
    best_c = float('inf')
    # hill climbing function
    def hill_climb(h):
        cur_h = h.copy()
        cur_c = c5_of(cur_h)
        # parameters
        no_improve = 0
        max_no_improve = 5000
        while time.time() < end_time and no_improve < max_no_improve:
            # choose random ones and zeros
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # swap
            cur_h[i], cur_h[j] = 0.0, 1.0
            new_c = c5_of(cur_h)
            if new_c + 1e-12 < cur_c:
                cur_c = new_c
                no_improve = 0
            else:
                # revert
                cur_h[i], cur_h[j] = 1.0, 0.0
                no_improve += 1
        return cur_h, cur_c
    # simulated annealing function
    def anneal(h, init_c):
        cur_h = h.copy()
        cur_c = init_c
        best_local_h = cur_h.copy()
        best_local_c = cur_c
        total_budget = end_time - time.time()
        max_temp = 0.01
        # start time for schedule
        t0 = time.time()
        while time.time() < end_time:
            elapsed = time.time() - t0
            # linear cooling
            T = max_temp * max(0.0, 1 - elapsed / total_budget)
            # pick random swap
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # swap
            cur_h[i], cur_h[j] = 0.0, 1.0
            new_c = c5_of(cur_h)
            delta = new_c - cur_c
            accept = False
            if delta < 0:
                accept = True
            else:
                if T > 0.0 and rng.random() < math.exp(-delta / T):
                    accept = True
            if accept:
                cur_c = new_c
                if cur_c < best_local_c:
                    best_local_c = cur_c
                    best_local_h = cur_h.copy()
            else:
                # revert
                cur_h[i], cur_h[j] = 1.0, 0.0
        return best_local_h, best_local_c
    # Main loop over candidates and restarts
    while time.time() < end_time:
        # pick a start candidate: either from list or random new
        if candidates:
            start_h = candidates.pop(0)
        else:
            start_h = rand_bin()
        # hill climb
        h_hc, c_hc = hill_climb(start_h)
        if c_hc < best_c:
            best_c = c_hc
            best_h = h_hc.copy()
            if best_c < 0.3808:
                break
        # annealing from hill-climb result
        if time.time() < end_time:
            h_ann, c_ann = anneal(h_hc, c_hc)
            if c_ann < best_c:
                best_c = c_ann
                best_h = h_ann.copy()
                if best_c < 0.3808:
                    break
        # optional: add a few more random candidates memory?
        # stop condition if time low
        if time.time() + 5 > end_time:
            break
    # Ensure best_h defined
    if best_h is None:
        best_h = rand_bin()
        best_c = c5_of(best_h)
    # guarantee constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    total = np.sum(best_h)
    target = n / 2.0
    if abs(total - target) > 1e-9:
        # adjust a random index
        diff = target - total
        # find a index with space to adjust (0 or 1) accordingly
        # we will add diff to one element, clamped
        idx_mod = rng.integers(n)
        best_h[idx_mod] += diff
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c, n

def compute_c5_val(h):
    """Return C₅ for a binary (or 0‑1) step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def hill_climb_swap(h, rng, time_limit):
    """Greedy swap hill‑climbing on a binary vector."""
    cur_h = h.copy()
    cur_c = compute_c5_val(cur_h)
    max_no_improve = 20000
    no_improve = 0
    while time.time() < time_limit and no_improve < max_no_improve:
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c = compute_c5_val(cur_h)
        if new_c + 1e-12 < cur_c:
            cur_c = new_c
            no_improve = 0
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
            no_improve += 1
    return cur_h, cur_c

def anneal_swap(h, rng, start_time, end_time):
