def _gradient_of_shift(v, shift):
    n = len(v)
    idx = np.arange(n)
    grad = np.zeros_like(v)
    # term1: (1 - v[i+shift]) where i+shift in range
    loc1 = idx + shift
    mask1 = (loc1 >= 0) & (loc1 < n)
    grad[mask1] = 1.0 - v[loc1[mask1]]
    # term2: remove v[i-shift] where i-shift in range
    loc2 = idx - shift
    mask2 = (loc2 >= 0) & (loc2 < n)
    grad[mask2] -= v[loc2[mask2]]
    return grad

    target = n * 0.5
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        # decide which coordinates can be moved
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the C5 value for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _structured_initial(n):
    """Sinusoidal starting pattern."""
    idx = np.arange(n)
    v = 0.5 * (1.0 + np.sin(2.0 * np.pi * idx / n))
    return _repair_vector(v, n)

def _gradient_of_shift(v, shift):
    """Gradient of the correlation term for a given integer shift."""
    n = len(v)
    idx = np.arange(n)
    grad = np.zeros_like(v)

    loc1 = idx + shift
    mask1 = (loc1 >= 0) & (loc1 < n)
    grad[mask1] = 1.0 - v[loc1[mask1]]

    loc2 = idx - shift
    mask2 = (loc2 >= 0) & (loc2 < n)
    grad[mask2] -= v[loc2[mask2]]
    return grad

def _gd_opt(v_start, time_limit):
    """Projected gradient descent on the max‑correlation objective."""
    start = time.time()
    v = v_start.copy()
    n = len(v)
    best_score = _compute_c5(v)
    step = 0.1
    stagnation = 0

    while time.time() - start < time_limit:
        # current worst shift
        corr = np.correlate(v, 1.0 - v, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)

        grad = _gradient_of_shift(v, shift)
        cand = v - step * grad
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)

        if cand_score < best_score:
            v, best_score = cand, cand_score
            step = min(step * 1.2, 0.5)
            stagnation = 0
        else:
            step *= 0.5
            stagnation += 1
            if step < 1e-6:
                step = 0.01
            if stagnation >= 10:
                # occasional random jitter to escape plateaus
                jitter = (np.random.rand(n) - 0.5) * step * 2.0
                cand2 = v + jitter
                cand2 = _repair_vector(cand2, n)
                cand2_score = _compute_c5(cand2)
                if cand2_score < best_score:
                    v, best_score = cand2, cand2_score
                    step *= 1.5
                stagnation = 0
        # safeguard against infinite loop
        if time.time() - start >= time_limit:
            break
    return v, best_score

def _local_search(v_start, n, time_limit):
    """Simple hill‑climbing using pairwise exchanges."""
    start = time.time()
    v = v_start.copy()
    best_score = _compute_c5(v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < best_score:
            v, best_score = cand, cand_score
    return v, best_score

def _slsqp_refine(v_start, time_limit):
    """Short SLSQP polish."""
    n = len(v_start)
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n * 0.5}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _compute_c5,
            v_start,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 300, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            v_opt = _repair_vector(res.x, n)
            return v_opt, _compute_c5(v_opt)
    except Exception:
        pass
    return v_start, _compute_c5(v_start)

# Main entry point

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function that yields a low C5 bound."""
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Acquire a parent construction if present.
    # ------------------------------------------------------------------
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        # fallback: balanced random start
        n0 = 190
        parent = _random_feasible(n0)

    parent = _repair_vector(parent, n0)

    # ------------------------------------------------------------------
    # Define the multi‑resolution schedule (keep size <= 800)
    # ------------------------------------------------------------------
    resolutions = [n0]
    if n0 * 2 <= 800:
        resolutions.append(n0 * 2)
    if n0 * 4 <= 800:
        resolutions.append(n0 * 4)
    # ensure monotonic order
    resolutions = sorted(set(resolutions))

    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    # ------------------------------------------------------------------
    # Loop over resolutions
    # ------------------------------------------------------------------
    for target_n in resolutions:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:
            break
        remaining = budget_s - elapsed - 0.5  # keep a small reserve

        # Prepare initial guess at this resolution
        if len(best_v) != target_n:
            init_guess = _upsample_vector(best_v, target_n)
        else:
            init_guess = best_v.copy()

        # Determine per‑restart budget (allow a few restarts)
        n_restarts = 4
        per_restart = max(0.05, remaining / (n_restarts + 1))

        # ------------------------------------------------------------------
        # Primary optimization from the upsampled guess
        # ------------------------------------------------------------------
        v_opt, sc_opt = _gd_opt(init_guess, per_restart * 0.9)
        if sc_opt < best_score:
            best_v, best_score, best_n = v_opt, sc_opt, target_n

        # ------------------------------------------------------------------
        # Additional random restarts (including a sinusoidal seed)
        # ------------------------------------------------------------------
        seeds = [
            _structured_initial(target_n),
        ] + [_random_feasible(target_n) for _ in range(2)]
        for seed_vec in seeds:
            elapsed = time.time() - start_time
            if elapsed >= budget_s - 0.5:
                break
            v_opt, sc_opt = _gd_opt(seed_vec, per_restart * 0.9)
            if sc_opt < best_score:
                best_v, best_score, best_n = v_opt, sc_opt, target_n

        # ------------------------------------------------------------------
        # Local refinement with pairwise moves
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        if elapsed < budget_s - 0.3:
            extra = min(0.2, budget_s - elapsed - 0.3)
            v_ls, sc_ls = _local_search(best_v, best_n, extra)
            if sc_ls < best_score:
                best_v, best_score = v_ls, sc_ls

        # ------------------------------------------------------------------
        # Very short SLSQP polish if time permits
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        if elapsed < budget_s - 0.2:
            extra = min(0.1, budget_s - elapsed - 0.2)
            v_pol, sc_pol = _slsqp_refine(best_v, extra)
            if sc_pol < best_score:
                best_v, best_score = v_pol, sc_pol

    # ------------------------------------------------------------------
    # Final safety repair and evaluation
    # ------------------------------------------------------------------
    best_v = _repair_vector(best_v, best_n)
    final_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(final_score), int(best_n)
