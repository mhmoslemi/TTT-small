import numpy as np
import time
import math

def _c5_score(v: np.ndarray) -> float:
    n = v.size
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return float(dx * np.max(corr))

def _repair_binary(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = v.size
    target = n // 2
    cur = int(v.sum())
    if cur > target:
        # too many ones
        ones = np.where(v == 1)[0]
        to_zero = rng.choice(ones, size=cur - target, replace=False)
        v[to_zero] = 0
    elif cur < target:
        zeros = np.where(v == 0)[0]
        to_one = rng.choice(zeros, size=target - cur, replace=False)
        v[to_one] = 1
    return v

def _simulated_annealing_binary(v_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    v = v_start.copy()
    cur_score = _c5_score(v)
    best_v = v.copy()
    best_score = cur_score
    n = v.size
    start = time.time()
    T0 = 0.01   # initial temperature
    max_eps = None   # not used for binary swap
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        # propose swap of a 1 and a 0
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if v[i] == v[j]:
            continue
        # swap
        v[i], v[j] = v[j], v[i]
        new_score = _c5_score(v)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_v = v.copy()
        else:
            # revert swap
            v[i], v[j] = v[j], v[i]
    return best_v, best_score

import numpy as np
import time
import math

def _c5_score(v: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = v.size
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return float(dx * np.max(corr))

def _repair_binary(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force a 0/1 vector to contain exactly n/2 ones."""
    n = v.size
    target = n // 2
    cur = int(v.sum())
    if cur > target:
        ones = np.where(v == 1)[0]
        to_zero = rng.choice(ones, size=cur - target, replace=False)
        v[to_zero] = 0
    elif cur < target:
        zeros = np.where(v == 0)[0]
        to_one = rng.choice(zeros, size=target - cur, replace=False)
        v[to_one] = 1
    return v

def _simulated_annealing_binary(v_start: np.ndarray,
                                time_limit: float,
                                rng: np.random.Generator):
    """Simple SA that swaps a 1 with a 0 while preserving the sum."""
    v = v_start.copy()
    cur_score = _c5_score(v)
    best_v = v.copy()
    best_score = cur_score
    n = v.size
    start = time.time()

        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if v[i] == v[j]:
            continue  # need a 1↔0 swap

        # perform swap
        v[i], v[j] = v[j], v[i]
