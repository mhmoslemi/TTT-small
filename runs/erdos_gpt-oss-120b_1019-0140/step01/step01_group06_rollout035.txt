def compute_gradient_max_lag(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (n - 1)  # shift ∈ [-(n-1), n-1]

    # shifted_fwd: h shifted by +shift (i+shift)
    if shift >= 0:
        fwd = np.concatenate((h[shift:], np.zeros(shift)))
    else:
        fwd = np.concatenate((np.zeros(-shift), h[:n+shift]))

    # shifted_back: h shifted by -shift (i-shift)
    if shift <= 0:
        back = np.concatenate((h[-shift:], np.zeros(-shift)))  # since -shift >=0
    else:
        back = np.concatenate((np.zeros(shift), h[:n-shift]))

    grad = 1.0 - (fwd + back)
    return grad, idx_max

def project_to_feasible(x, target_sum):

    n = x.shape[0]
    lo = -1.0
    hi = 2.0

def binary_annealing(h_init, max_iters, rng):
    h = h_init.copy()
    n = len(h)
    ones_idx = np.flatnonzero(h)
    zeros_idx = np.flatnonzero(1 - h)
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score

    # schedule parameters
    T0 = 0.1  # initial temperature
    for it in range(max_iters):
        # linear cooling
        T = T0 * (1 - it / max_iters)
        # propose swap
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i], h[j] = h[j], h[i]
        # update idx arrays quickly: swap location in arrays
        # But we can recompute after each iteration for simplicity.
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            current_score = new_score
            # update idx sets reflect swap
            # modify ones_idx and zeros_idx:
            # i is now zero, j is now one
            # replace i with new zero in zero set, j with new one...
            # Simpler: recompute after each swap:
            ones_idx = np.flatnonzero(h)
            zeros_idx = np.flatnonzero(1 - h)
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]  # swap back
        if current_score < best_score:
            best_score = current_score
            best_h = h.copy()
    return best_h, best_score

def binary_annealing(h_init, time_limit, rng):
    start = time.time()
    h = h_init.copy()
    n = h.shape[0]
    # index lists
    ones = list(np.flatnonzero(h).tolist())
    zeros = list(np.flatnonzero((1 - h).astype(int)).tolist())
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score

    # Temperature schedule parameters
    T0 = 0.2
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # Temperature decays
        T = T0 * (0.995 ** it)
        # choose random indices
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap in h
        h[i], h[j] = h[j], h[i]
        # update lists
        ones[i_idx] = j
        zeros[j_idx] = i

        new_score = compute_c5(h)

                best_h = h.copy()

            # revert h
            h[i], h[j] = h[j], h[i]
            # revert lists
            ones[i_idx] = i
            zeros[j_idx] = j
        # Possibly break early if best_score already extremely low threshold (< 0.3808)
        # but continue maybe for better.
    return best_h, best_score

def compute_gradient_max_lag(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (n - 1)  # shift can be negative

    # compute shifted_fwd
    if shift >= 0:
        fwd = np.concatenate((h[shift:], np.zeros(shift, dtype=h.dtype)))
    else:
        fwd = np.concatenate((np.zeros(-shift, dtype=h.dtype), h[:n+shift]))
    # compute shifted_back = shift by -shift
    if shift <= 0:
        back = np.concatenate((h[-shift:], np.zeros(-shift, dtype=h.dtype))) # shift <=0; -shift >=0
    else:
        back = np.concatenate((np.zeros(shift, dtype=h.dtype), h[:n-shift]))
    grad = 1.0 - (fwd + back)
    return grad, idx_max

def project_to_feasible(x, target_sum, tol=1e-9, max_iter=60):
    # Clip to bounds and adjust linearly via lambda
    lo = -1.0
    hi = 2.0
    # binary search for lambda such that sum(clipped(x - lam,0,1)) = target_sum
    for _ in range(max_iter):

        s = y.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(x, target_sum, max_iter=60):
    lo, hi = -1.0, 2.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_gradient_max_lag(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    shift = idx_max - (n - 1)
    if shift >= 0:
        fwd = np.concatenate((h[shift:], np.zeros(shift, dtype=h.dtype)))
    else:
        fwd = np.concatenate((np.zeros(-shift, dtype=h.dtype), h[:n + shift]))
    if shift <= 0:
        back = np.concatenate((h[-shift:], np.zeros(-shift, dtype=h.dtype)))
    else:
        back = np.concatenate((np.zeros(shift, dtype=h.dtype), h[:n - shift]))
    grad = 1.0 - (fwd + back)
    return grad, idx_max

def binary_annealing(h_init, time_limit, rng):
    start = time.time()
    h = h_init.copy()
    n = h.shape[0]
    ones = list(np.flatnonzero(h).tolist())
    zeros = list(np.flatnonzero((1 - h).astype(int)).tolist())
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    T0 = 0.2
    it = 0
    while time.time() - start < time_limit:
        it += 1
        T = T0 * (0.995 ** it)
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i, j = ones[i_idx], zeros[j_idx]
        h[i], h[j] = h[j], h[i]
        ones[i_idx], zeros[j_idx] = j, i
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            ones[i_idx], zeros[j_idx] = i, j
    return best_h, best_score

def gradient_descent_refine(h_start, target_sum, time_limit, rng):
    start = time.time()
    h = h_start.astype(float).copy()
    kernel = np.array([0.2, 0.6, 0.2])
    h = np.convolve(h, kernel, mode="same")
    h = np.clip(h, 0.0, 1.0)
    h = project_to_feasible(h, target_sum)
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    step = 0.4
    while time.time() - start < time_limit:
        grad, _ = compute_gradient_max_lag(h)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new, target_sum)
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score, best_h = new_score, h_new.copy()
        if new_score < cur_score:
            h, cur_score = h_new, new_score
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.5
            if step < 1e-5:
                h = np.clip(h + rng.normal(scale=0.02, size=h.shape), 0.0, 1.0)
                h = project_to_feasible(h, target_sum)
                cur_score = compute_c5(h)
                step = 0.4
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    base_h = globals().get("initial_h_values")
    if base_h is not None:

    else:
        n0 = 380
        base_h = None

    candidates = [n0]
    if n0 * 2 <= 1000:
        candidates.append(n0 * 2)

    best_h, best_c5, best_n = None, np.inf, None

    for idx, n in enumerate(candidates):
        if time.time() - start_time >= budget_s * 0.98:
            break
        remaining = len(candidates) - idx
        sub_budget = (budget_s - (time.time() - start_time)) / remaining

        if base_h is not None:
            if base_h.shape[0] != n:
                x_old = np.linspace(0.0, 2.0, base_h.shape[0], endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h0 = np.interp(x_new, x_old, base_h)
            else:

        else:

            h0[rng.choice(n, size=n // 2, replace=False)] = 1.0

        target = n / 2.0

        if not np.all((h0 == 0) | (h0 == 1)):

    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5), int(best_n)
