def compute_gradient_for_shift(v, s):
    n = len(v)
    g = np.zeros_like(v)
    start = max(0, -s)
    end = min(n, n - s)  # exclusive bound for i
    # Equivalent: i in [start, end)
    # If s >= 0: start=0, end=n-s
    # If s <0: start=-s, end=n
    # We'll use slicing to vectorize.
    i = np.arange(start, end)
    ip = i + s
    # ip within [0,n)
    g[i] += (1 - v[ip])
    g[ip] -= v[i]
    return g

import time
import numpy as np
from scipy.optimize import minimize

# --- helper functions ---------------------------------------------------------

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if still off
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v


def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx


def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)


def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)


def _gradient_for_shift(v, shift):
    """Sub‑gradient of the worst‑shift correlation w.r.t. v."""
    n = len(v)
    g = np.zeros_like(v)
    start = max(0, -shift)
    end = min(n, n - shift)
    i = np.arange(start, end, dtype=np.intp)
    ip = i + shift
    # contribution from the term v[i]*(1 - v[i+shift])
    g[i] += (1.0 - v[ip])
    # contribution from the same term w.r.t. v[i+shift]
    g[ip] -= v[i]
    return g


def _gd_optimize(v0, time_limit):
    """Projected gradient descent on the worst‑shift objective."""
    start = time.time()
    v = v0.copy()
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(v)
    iter_cnt = 0
    # simple diminishing step schedule
    while time.time() - start < time_limit and iter_cnt < 5000:
        corr = np.correlate(v, 1.0 - v, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        g = _gradient_for_shift(v, shift)
        # step size ~0.5 / sqrt(iter+1)
        alpha = 0.5 / np.sqrt(iter_cnt + 1)
        v_new = v - alpha * g
        v_new = _repair_vector(v_new, n)
        new_score = _compute_c5(v_new)
        if new_score < best_score:
            best_score, best_v = new_score, v_new.copy()
            v = v_new
        else:
            # occasional random jitter to escape plateaus
            if np.random.rand() < 0.02:
                jitter = (np.random.rand(n) - 0.5) * 0.02
                v = _repair_vector(v + jitter, n)
        iter_cnt += 1
    return best_v, best_score


def _local_search(v0, time_limit):
    """Hill‑climbing by exchanging small amounts between two coordinates."""
    start = time.time()
    v = v0.copy()
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(v)
    step_mag = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step_mag
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score, best_v = score, cand
    return best_v, best_score
