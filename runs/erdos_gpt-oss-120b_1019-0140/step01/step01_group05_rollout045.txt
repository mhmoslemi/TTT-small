def differential_evolution_opt(h_initial, time_limit, rng, popsize=15, F=0.8, CR=0.9):
   N = h_initial.size
   # Initialize population
   pop = []
   scores = []
   for _ in range(popsize):
       v = rng.random(N)
       v = repair_vector(v)
       pop.append(v)
       scores.append(_c5_value(v))
   pop = np.array(pop)
   scores = np.array(scores)
   best_idx = np.argmin(scores)
   best = pop[best_idx].copy()
   best_score = scores[best_idx]
   start = time.time()
   while time.time() - start < time_limit:
       for i in range(popsize):
           # choose three distinct indices not equal to i
           idxs = list(range(popsize))
           idxs.remove(i)
           a,b,c = rng.choice(idxs, size=3, replace=False)
           mutant = pop[a] + F * (pop[b] - pop[c])
           # clip and repair
           mutant = np.clip(mutant, 0.0, 1.0)
           mutant = repair_vector(mutant)
           # crossover
           cross = rng.random(N) < CR
           # ensure at least one component from mutant
           if not np.any(cross):
               cross[rng.integers(0,N)] = True
           trial = np.where(cross, mutant, pop[i])
           trial = repair_vector(trial)
           trial_score = _c5_value(trial)
           if trial_score < scores[i]:
               pop[i] = trial
               scores[i] = trial_score
               if trial_score < best_score:
                   best_score = trial_score
                   best = trial.copy()
   return best, best_score

def repair_vector(v):
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    n = v.size
    target = n * 0.5
    # Clip
    np.clip(v, 0.0, 1.0, out=v)
    # Compute sum
    s = v.sum()
    if s == 0:
        v[:] = target / n
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    # After scaling, may still not meet target due to clipping

    if abs(s2 - target) > 1e-12:

        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    # final check:
    # optionally adjust small residual by adding to first element not at bound.

    return v

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    target = n * 0.5
    np.clip(v, 0.0, 1.0, out=v)
    s = v.sum()
    if s == 0.0:
        v[:] = target / n
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simple SA that preserves sum and box constraints."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T_start = 0.1
    T_end = 1e-5
    max_eps_initial = 0.5
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        frac = elapsed / time_limit
        T = T_start * (T_end / T_start) ** frac
        max_move = max_eps_initial * (1 - frac) + 0.01 * frac
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        max_possible = up + down
        eps_bound = min(max_move, max_possible)
        eps = rng.random() * (up + down) - down
        if abs(eps) > eps_bound:
            eps = math.copysign(eps_bound, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        if not (0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0):
            continue
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _differential_evolution_opt(h_initial: np.ndarray, time_limit: float, rng: np.random.Generator,
                                popsize=12, F=0.8, CR=0.9):
    """Small DE with repair to enforce constraints."""
    N = h_initial.size
    pop = np.empty((popsize, N), dtype=np.float64)
    scores = np.empty(popsize, dtype=np.float64)
    # seed first individual with the provided start
    pop[0] = _repair_vector(h_initial.copy())
    scores[0] = _c5_value(pop[0])
    # remaining random individuals
    for i in range(1, popsize):
        v = rng.random(N)
        v = _repair_vector(v)
        pop[i] = v
        scores[i] = _c5_value(v)
    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]
    start = time.time()
    while time.time() - start < time_limit:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            np.clip(mutant, 0.0, 1.0, out=mutant)
            mutant = _repair_vector(mutant)
            cross = rng.random(N) < CR
            if not np.any(cross):

            trial = np.where(cross, mutant, pop[i])
            trial = _repair_vector(trial)
            trial_score = _c5_value(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
    return best, best_score
