def _hill_climb(h, target, time_left, start):
    best = h.copy()
    best_score = _compute_c5(best)
    n = len(h)
    while time.time() - start < time_left:
        # propose random perturb:
        idx = np.random.randint(n)
        # small delta in [-d, d]
        d = 0.05 # maybe adapt
        delta = np.random.uniform(-d, d)
        # adjust idx by delta, adjust another idx2 to keep sum constant
        idx2 = np.random.randint(n)
        while idx2 == idx:
            idx2 = np.random.randint(n)
        # Build candidate
        cand = best.copy()
        cand[idx] = np.clip(cand[idx] + delta, 0, 1)
        # to keep sum, adjust idx2 opposite amount
        delta2 = - (cand[idx] - best[idx])
        cand[idx2] = np.clip(cand[idx2] + delta2, 0, 1)
        # after adjusting idx2, sum may drift; project fully
        cand = _project_to_constraints(cand, target)
        score = _compute_c5(cand)
        if score < best_score:
            best = cand
            best_score = score
    return best, best_score

    np.random.seed(seed)
    start_time = time.time()
    time_check = lambda: time.time() - start_time

    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        h0 = np.array(init_vec, dtype=np.float64).ravel()
        n0 = h0.size

        h0 = np.random.rand(n0)

    best_h = None
    best_score = float('inf')
    target_bound = 0.3808

import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral C5 for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust it so that sum(v)=target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]  # largest capacity first
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]  # largest amount first
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-9 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _hill_climb(h, target, time_left, start_time):
    """Simple random hill‑climbing with Gaussian steps."""
    best = h.copy()
    best_score = _compute_c5(best)
    n = best.size
    deadline = start_time + time_left
    iter_cnt = 0
    max_iter = 5000
    sigma = 0.01
    while iter_cnt < max_iter and time.time() < deadline:
        cand = best + np.random.randn(n) * sigma
        cand = _project_to_constraints(cand, target)
        score = _compute_c5(cand)
        if score < best_score:
            best = cand
            best_score = score
        iter_cnt += 1
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_elapsed = lambda: time.time() - start_time

    # ----------------------------------------------------------------------
    # Initial vector (may be supplied by the harness)
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        h0 = np.array(init_vec, dtype=np.float64).ravel()
        n0 = h0.size
    else:
        n0 = 380
        h0 = np.random.rand(n0)

    best_h = None
    best_score = float("inf")
    target_bound = 0.3808

    # Resolution schedule – grow only if time permits
    resolutions = [n0]
    if budget_s >= 300:
        resolutions.append(n0 * 2)
    if budget_s >= 600:
        resolutions.append(n0 * 4)

    for n in resolutions:
        if time_elapsed() > budget_s * 0.94:
            break

        # ------------------------------------------------------------------
        # Build start vector for this resolution
        if best_h is not None:
            src = best_h
            n_src = src.size
            x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_start = np.interp(x_new, x_src, src)
        else:
            if n == n0:
                h_start = h0.copy()
            else:
                x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h_start = np.interp(x_new, x0, h0)

        target_sum = n / 2.0
        h_start = _project_to_constraints(h_start, target_sum)

        # ------------------------------------------------------------------
        # Determine how many SLSQP restarts we can afford
        remaining = budget_s - time_elapsed()
        # Rough estimate: 0.30 s per restart for n≈380, more for larger n
        est_per = 0.30 * (n / 380.0)
        max_restarts = max(0, int(remaining / est_per) - 3)
        # Cap according to size (more restarts for smaller problems)
        if n <= 500:
            max_restarts = min(max_restarts, 250)
        elif n <= 1000:
            max_restarts = min(max_restarts, 120)
        else:
            max_restarts = min(max_restarts, 50)

        maxiter = 300

        # ------------------------------------------------------------------
        # Main restart loop (SLSQP + occasional binary candidates)
        for r in range(max_restarts):
            if time_elapsed() > budget_s * 0.94:
                break

            # 20 % of the time try a purely binary candidate
            if np.random.rand() < 0.2:
                cand = np.zeros(n)
                ones = np.random.choice(n, n // 2, replace=False)
                cand[ones] = 1.0
                cand = _project_to_constraints(cand, target_sum)  # safety
                score = _compute_c5(cand)
                if score < best_score:
                    best_h = cand.copy()
                    best_score = float(score)
                    if best_score < target_bound:
                        break
                continue

            # otherwise start from h_start + Gaussian noise
            sigma = 0.05 if r < max_restarts // 2 else 0.015
            perturb = np.random.randn(n) * sigma
            x0 = h_start + perturb
            x0 = _project_to_constraints(x0, target_sum)

            # Equality constraint for the integral
            cons = {"type": "eq", "fun": lambda h, tgt=target_sum: np.sum(h) - tgt}
            bounds = [(0.0, 1.0)] * n

            try:
                res = minimize(
                    _compute_c5,
                    x0,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=[cons],
                    options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
                )
            except Exception:
                continue

            h_opt = _project_to_constraints(res.x, target_sum)
            score = _compute_c5(h_opt)
            if score < best_score:
                best_score = float(score)
                best_h = h_opt.copy()
                # early exit if we already beat the target bound
                if best_score < target_bound:
                    break

        # ------------------------------------------------------------------
        # Optional local refinement via hill climbing
        if best_h is not None and time_elapsed() < budget_s * 0.94:
            time_left = budget_s - time_elapsed()
            best_h, best_score = _hill_climb(best_h, target_sum, time_left, start_time)

        # Stop if target achieved
        if best_score < target_bound:
            break

    # ----------------------------------------------------------------------
    # Fallback if nothing succeeded (should not happen)
    if best_h is None:
        best_h = h_start
        best_score = _compute_c5(best_h)

    n_final = int(best_h.size)

    # Use the harness evaluator if available for the final bound
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, n_final
