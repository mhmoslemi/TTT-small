def project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    # if cur < target, need to add mass
    if cur < target:
        deficit = target - cur
        # indices where v < 1
        idx = np.where(v < 1.0)[0]
        while deficit > 1e-12 and idx.size > 0:
            # amount we can add to each without exceeding bound
            caps = 1.0 - v[idx]
            # equally allocate
            add_per = deficit / idx.size
            # allocate as min(add_per, caps)
            add = np.minimum(add_per, caps)
            v[idx] += add
            deficit -= add.sum()
            # remove those that reached 1
            idx = idx[v[idx] < 1.0 - 1e-12]
        # after loop, may still have deficit due to precision; apply scaling
    else:
        excess = cur - target
        idx = np.where(v > 0.0)[0]
        while excess > 1e-12 and idx.size > 0:
            caps = v[idx]  # can remove at most this much
            remove_per = excess / idx.size
            sub = np.minimum(remove_per, caps)
            v[idx] -= sub
            excess -= sub.sum()
            idx = idx[v[idx] > 1e-12]
    # final clip
    v = np.clip(v, 0.0, 1.0)
    # small scaling if needed
    cur = v.sum()
    if abs(cur - target) > 1e-8:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def project_to_constraints(v, target):
    # Clip
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    # If sum close -> return
    if abs(cur - target) < 1e-10:
        return v
    # Use simple approach: while loop adjusting:
    # Use tolerance = 1e-8.
    # In each iteration adjust uniformly across free variables (not at bounds).
    max_iter = 100
    for _ in range(max_iter):
        if abs(cur - target) < 1e-8:
            break
        if cur < target:
            deficit = target - cur
            free_idx = np.where(v < 1.0 - 1e-12)[0]
            if free_idx.size == 0:
                break
            add = deficit / free_idx.size
            v[free_idx] += add
            v[free_idx] = np.minimum(v[free_idx], 1.0)
        else:
            excess = cur - target
            free_idx = np.where(v > 1e-12)[0]
            if free_idx.size == 0:
                break
            sub = excess / free_idx.size
            v[free_idx] -= sub
            v[free_idx] = np.maximum(v[free_idx], 0.0)
        cur = v.sum()
    # Final clip and scaling
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-6 and cur > 0:
        v *= target/cur
        v = np.clip(v,0,1)
    return v

def project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    eps = 1e-10
    # If already close, return
    if abs(cur - target) <= eps:
        return v
    # Use water-filling style
    for _ in range(100):
        if abs(cur - target) <= eps:
            break
        if cur < target:
            deficit = target - cur
            free = np.where(v < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            add = deficit / free.size
            v[free] += add
            v[free] = np.minimum(v[free], 1.0)
        else:
            excess = cur - target
            free = np.where(v > 1e-12)[0]
            if free.size == 0:
                break
            sub = excess / free.size
            v[free] -= sub
            v[free] = np.maximum(v[free], 0.0)
        cur = v.sum()
    # final clip and scaling
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-6 and cur > 0:
        v = v * (target / cur)
    v = np.clip(v, 0.0, 1.0)
    return v

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def de_optimize_best(init_vec, n, target, start_time, total_budget):
    # Build initial population
    popsize = max(8, min(20, n // 30))
    # ensure at least 8
    pop = [init_vec]
    # add random binary vectors
    while len(pop) < popsize:
        v = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, int(round(target)), replace=False)
        v[idx] = 1.0
        pop.append(v)
    pop = np.array(pop)
    scores = np.array([compute_c5(vec) for vec in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # DE parameters
    F = 0.5
    CR = 0.9
    # generation loop
    max_gen = 2000  # high bound
    for gen in range(max_gen):
        # check time
        if time.time() - start_time > total_budget * 0.99:
            break
        for i in range(popsize):
            # random distinct indices a,b,c not i
            idxs = np.arange(popsize)
            idxs = idxs[idxs != i]
            a,b,c = np.random.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])

                    best_h = trial.copy()
                    if best_score < 0.3808:

import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v)=target by simple water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    eps = 1e-10
    if abs(cur - target) <= eps:
        return v
    # iterative water‑filling
    for _ in range(100):
        cur = v.sum()
        if abs(cur - target) <= eps:
            break
        if cur < target:
            deficit = target - cur
            free = np.where(v < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            add = deficit / free.size
            v[free] += add
            v[free] = np.minimum(v[free], 1.0)
        else:
            excess = cur - target
            free = np.where(v > 1e-12)[0]
            if free.size == 0:
                break
            sub = excess / free.size
            v[free] -= sub
            v[free] = np.maximum(v[free], 0.0)
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-6 and cur > 0:
        v = v * (target / cur)
    return np.clip(v, 0.0, 1.0)


def _binary_hill_climb(h: np.ndarray, n: int, target: float, max_seconds: float):
    """Greedy 1‑swap hill climbing on a binary version of h."""
    start = time.time()
    target_int = int(round(target))
    # initialise a binary vector: set the largest target_int entries to 1
    order = np.argsort(-h)
    best = np.zeros(n, dtype=np.float64)
    best[order[:target_int]] = 1.0
    best_score = _compute_c5(best)
    while time.time() - start < max_seconds:
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_score = _compute_c5(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            if best_score < 0.3808:
                break
    return best, best_score


def _de_optimize(init_vec: np.ndarray, n: int, target: float, max_seconds: float):
    """Simple Differential Evolution with projection onto the linear constraints."""
    start = time.time()
    # population size scales modestly with dimension
    popsize = max(8, min(20, n // 30))
    # build initial population
    pop = [init_vec.copy()]
    while len(pop) < popsize:
        v = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, int(round(target)), replace=False)
        v[idx] = 1.0
        pop.append(v)
    pop = np.array(pop)
    scores = np.array([_compute_c5(ind) for ind in pop])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    F = 0.5
    CR = 0.9
    # main DE loop
    while time.time() - start < max_seconds:
        for i in range(popsize):
            # time guard
            if time.time() - start >= max_seconds:
                break
            # distinct indices a,b,c != i
            idxs = np.arange(popsize)
            idxs = idxs[idxs != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = _project_to_constraints(mutant, target)
            # crossover
            cross = np.random.rand(n) < CR
            jrand = np.random.randint(n)
            cross[jrand] = True
            trial = np.where(cross, mutant, pop[i])
            trial = _project_to_constraints(trial, target)
            trial_score = _compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
                    if best_score < 0.3808:
                        return best_h, best_score
        # optional early stop if no improvement for many generations could be added
    return best_h, best_score

    start_time = time.time()
