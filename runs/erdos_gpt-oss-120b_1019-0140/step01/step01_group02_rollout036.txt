def perturb(h, magnitude, steps):
    for _ in range(steps):
        i, j = rng.integers(0, n, size=2)
        if i==j:
            continue
        hi=h[i]; hj=h[j]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        if delta_min >= delta_max:
            continue
        delta = rng.uniform(-magnitude, magnitude)
        # Clip to allowed range
        delta = max(delta, delta_min)
        delta = min(delta, delta_max)
        h[i] = hi + delta
        h[j] = hj - delta
    return h
```

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

```
def _repair_sum(h, target):
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h
```

def tournament(population, fitness, size, rng):
    n = len(population)
    idx = rng.choice(n, size=size, replace=False)
    best = idx[0]
    best_fit = fitness[best]
    for i in idx[1:]:
        f = fitness[i]
        if f < best_fit:
            best = i
            best_fit = f
    return population[best]
```

def mutate(h, max_step, rng):
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    hi = h[i]
    hj = h[j]
    delta_min = max(-hi, -(1.0 - hj))
    delta_max = min(1.0 - hi, hj)
    low = max(delta_min, -max_step)
    high = min(delta_max, max_step)
    if low >= high:
        return h
    delta = rng.random() * (high - low) + low
    h[i] = hi + delta
    h[j] = hj - delta
    return h
```

def run(seed=42, budget_s=1100, **kwargs):
    import time
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine problem size
    try:
        init0 = np.asarray(initial_h_values, dtype=float)
        n0 = init0.shape[0]
        # Choose n >= max(200, n0) but not > 800 to keep runtime reasonable
        n = max(200, n0)
    except NameError:
        n = 200
        init0 = None

    target_sum = n / 2.0
    dx = 2.0 / n   # not used directly, just for compute C5
    # Build initial vector from init0 if provided
    if init0 is not None:
        if init0.shape[0] != n:
            # interpolate to length n
            x_src = np.linspace(0, 2, init0.shape[0], endpoint=False) + 0.5 * (2.0 / init0.shape[0])
            x_tgt = np.linspace(0, 2, n, endpoint=False) + 0.5 * dx
            init = np.interp(x_tgt, x_src, init0, left=init0[0], right=init0[-1])
        else:
            init = init0.astype(float).copy()
        np.clip(init, 0.0, 1.0, out=init)
        init = _repair_sum(init, target_sum)
    else:
        init = None

    popsize = 40
    population = []
    # Add init as first individual if exists
    if init is not None:
        population.append(init.copy())
    # Fill rest of population
    while len(population) < popsize:
        # mix binary and random
        if rng.random() < 0.5:
            h = generate_binary_vector(n, target_sum, rng)
        else:
            h = generate_random_vector(n, target_sum, rng)
        # optionally perturb a bit
        # apply a few random mass transfers
        for _ in range(3):
            h = mutate(h, max_step=0.3, rng=rng)
        population.append(h)

    # Ensure all have exact sum
    population = [_repair_sum(h, target_sum) for h in population]

    # Evaluate fitness
    fitness = np.empty(len(population), dtype=float)
    for i, h in enumerate(population):
        fitness[i] = _compute_c5(h)

    # Track best
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = float(fitness[best_idx])

    # Hyperparameters
    tournament_size = 5
    crossover_prob = 0.9
    mutation_prob = 0.7
    max_step_start = 0.5
    max_step_end = 0.001
    # Counters
    no_improve_gens = 0
    restart_limit = 200

    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        # compute schedule
        elapsed = now - start
        frac = elapsed / budget_s
        max_step = max_step_start * (1.0 - frac) + max_step_end * frac

        # Build next generation
        new_pop = []
        # elitism: keep current best
        new_pop.append(best_h.copy())
        # Fill rest
        while len(new_pop) < popsize:
            # selection
            p1 = tournament(population, fitness, tournament_size, rng)
            p2 = tournament(population, fitness, tournament_size, rng)
            # crossover
            if rng.random() < crossover_prob:
                child = crossover(p1, p2, rng)
            else:
                child = p1.copy()
            # clip and repair sum
            np.clip(child, 0.0, 1.0, out=child)

            delta = rng.random() * (delta_max - delta_min) + delta_min

            cur_h[j] = hj - delta
            new_c5 = _compute_c5(cur_h)
            if new_c5 < cur_c5:
                cur_c5 = new_c5
            else:
                # revert
                cur_h[i] = hi
                cur_h[j] = hj
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()
            no_improve_gens = 0

        # possibly restart if stuck
        if no_improve_gens >= restart_limit:
            # replace half of population with fresh random individuals
            population = [best_h.copy()]
            while len(population) < popsize:
                if rng.random() < 0.5:
                    h = generate_binary_vector(n, target_sum, rng)
                else:
                    h = generate_random_vector(n, target_sum, rng)
                population.append(h)
            population = [_repair_sum(h, target_sum) for h in population]
            for i, h in enumerate(population):
                fitness[i] = _compute_c5(h)
            # reset counters and continue
            no_improve_gens = 0
            continue  # skip rest processing for this loop

        fitness = new_fitness

    # final repair/verification
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n

import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a vector so that its sum equals target while staying in [0,1]."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # choose direction
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h


def tournament(population, fitness, size, rng):
    """Select the best individual among a random subset."""
    idx = rng.choice(len(population), size=size, replace=False)
    best = idx[0]
    best_fit = fitness[best]
    for i in idx[1:]:
        f = fitness[i]
        if f < best_fit:
            best = i
            best_fit = f
    return population[best]


def generate_random_vector(n, target_sum, rng):
    """Random real vector in [0,1] with exact sum target_sum."""
    h = rng.random(n)
    s = h.sum()
    if s == 0.0:
        h[:] = target_sum / n
    else:
        h *= (target_sum / s)
    np.clip(h, 0.0, 1.0, out=h)
    h = _repair_sum(h, target_sum)
    return h


def generate_binary_vector(n, target_sum, rng):
    """Binary (0/1) vector with sum target_sum (fractional part allowed)."""
    h = np.zeros(n, dtype=float)
    int_part = int(np.floor(target_sum))
    if int_part > 0:
        idx = rng.choice(n, size=int_part, replace=False)
        h[idx] = 1.0
    frac = target_sum - int_part
    if frac > 1e-12:
        remaining = np.where(h == 0.0)[0]
        if remaining.size:
            i = rng.choice(remaining)
            h[i] = frac
    return h


def crossover(p1, p2, rng):
    """One‑point crossover."""
    n = p1.shape[0]
    if n <= 1:
        return p1.copy()
    cut = rng.integers(1, n)
    child = np.concatenate((p1[:cut], p2[cut:])).astype(float)
    return child


def mutate(h, max_step, rng):
    """Transfer a small amount of mass between two entries."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    hi, hj = h[i], h[j]
    delta_min = max(-hi, -(1.0 - hj))
    delta_max = min(1.0 - hi, hj)
    low = max(delta_min, -max_step)
    high = min(delta_max, max_step)
    if low >= high:
        return h
    delta = rng.random() * (high - low) + low
    h[i] = hi + delta
    h[j] = hj - delta
    return h
