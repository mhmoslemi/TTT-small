def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    # clip
    h = np.clip(h, 0.0, 1.0)
    # adjust sum by shifting all interior values
    # Use water-filling method: 
    # If sum > target, we need to reduce some values; reduce from those >0 proportionally.
    # If sum < target, increase some values <1 proportionally.
    # Implementation using while loop distributing diff among all non-boundaries.
    cur = np.sum(h)
    diff = target_sum - cur
    for _ in range(max_iter):
        if abs(diff) < tol:
            break
        if diff > 0:
            # need to increase
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            # ensure not exceed bound
            inc = np.minimum(delta, 1.0 - h[free])
            h[free] += inc
            cur = np.sum(h)
            diff = target_sum - cur
        else:
            # need to decrease
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            delta = diff / free.size  # negative
            dec = np.maximum(delta, -h[free])
            h[free] += dec
            cur = np.sum(h)
            diff = target_sum - cur
    # final adjustment small: if still off, adjust one index within bounds
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h) if cur < target_sum else np.argmax(h)
        # adjust i within bound
        req = target_sum - cur
        h[i] = np.clip(h[i] + req, 0.0, 1.0)
    return h

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # Determine indices that can move in the direction needed
        if diff > 0:
            # increase some entries that are <1
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size == 0:
            break
        # distribute diff equally
        delta = diff / idx.size
        # adjust with clipping
        if diff > 0:
            inc = np.minimum(delta, 1.0 - h[idx])
            h[idx] += inc
        else:
            dec = np.maximum(delta, -h[idx])
            h[idx] += dec
    # final tiny fix
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            inc = np.minimum(delta, 1.0 - h[idx])
            h[idx] += inc
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            dec = np.maximum(delta, -h[idx])
            h[idx] += dec
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Linear interpolation to a new length followed by feasibility projection."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def random_feasible(n, rng):
    """Uniform random feasible vector."""
    h = rng.random(n)
    return project_to_feasible(h, n / 2.0)


def search_at_resolution(h, n, time_limit):
    """Targeted hill‑climbing using the lag that attains the current max."""
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.default_rng()
    no_imp = 0
    max_no_imp = 5000
    while time.time() - start < time_limit:
        # correlation and dominant lag
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)

        improved = False
        if lag >= 0:
            max_i = n - lag
            if max_i > 0:
                i = rng.integers(0, max_i)
                j = i + lag
                dec = best_h[i]
                inc = 1.0 - best_h[j]
                trans = min(dec, inc)
                if trans > 1e-12:
                    delta = rng.random() * min(trans, 0.05)
                    nh = best_h.copy()
                    nh[i] -= delta
                    nh[j] += delta
                    c5_new = compute_c5(nh)
                    if c5_new < best_c5:
                        best_h, best_c5 = nh, c5_new
                        no_imp = 0
                        improved = True
        else:
            min_i = -lag
            if min_i < n:
                i = rng.integers(min_i, n)
                j = i + lag
                dec = best_h[i]
                inc = 1.0 - best_h[j]
                trans = min(dec, inc)
                if trans > 1e-12:
                    delta = rng.random() * min(trans, 0.05)
                    nh = best_h.copy()
                    nh[i] -= delta
                    nh[j] += delta
                    c5_new = compute_c5(nh)
                    if c5_new < best_c5:
                        best_h, best_c5 = nh, c5_new
                        no_imp = 0
                        improved = True

        if not improved and rng.random() < 0.02:
            nh = best_h + rng.normal(scale=0.02, size=n)
            nh = project_to_feasible(nh, n / 2.0)
            c5_new = compute_c5(nh)
            if c5_new < best_c5:
                best_h, best_c5 = nh, c5_new
                no_imp = 0
                continue

        if not improved:
            no_imp += 1
            if no_imp > max_no_imp:
                best_h = random_feasible(n, rng)
                best_c5 = compute_c5(best_h)
                no_imp = 0
    return best_h, best_c5


def refine_with_slsqp(h, n, max_iter=200):
    """Brief SLSQP refinement."""
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            lambda x: compute_c5(x),
            x0=h,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return h
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n / 2.0)
    if compute_c5(h_opt) < compute_c5(h):
        return h_opt
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h to minimize the Erdős overlap constant C5."""
    start = time.time()
    rng = np.random.default_rng(seed)

    base = globals().get("initial_h_values")
    if base is not None:
        base = np.asarray(base, dtype=float)
        n = base.shape[0]
        base = project_to_feasible(base, n / 2.0)
    else:
        n = 380
        base = rng.random(n)
        base = project_to_feasible(base, n / 2.0)

    best_h, best_n = base, n
    best_c5 = compute_c5(best_h)

    max_n = 1520  # keep < 2000 for speed

    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # spend up to half of remaining time on local search at this resolution
        search_time = min(remaining * 0.5, 600.0)
        best_h, best_c5 = search_at_resolution(best_h, best_n, search_time)

        # optional SLSQP polish if time permits
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining > 10.0:
            best_h = refine_with_slsqp(best_h, best_n, max_iter=200)
            best_c5 = compute_c5(best_h)

        # decide whether to upscale resolution
        if best_n >= max_n:
            break
        if remaining < 60.0:  # not enough time for another level
            break
        new_n = best_n * 2
        best_h = resample_h(best_h, new_n)
        best_n = new_n
        best_c5 = compute_c5(best_h)

    # final safety projection
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
