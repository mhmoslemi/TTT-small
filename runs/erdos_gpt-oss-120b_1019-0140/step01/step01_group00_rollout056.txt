def hill_descend(h, rng, start_val, start_time, budget_s):
    best = h
    best_val = start_val
    n = len(h)
    attempts_without_improvement = 0
    max_no_improve = 2000
    while attempts_without_improvement < max_no_improve:
        if time.time() - start_time > budget_s - 0.5:
            break
        # random swap
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if ones.size ==0 or zeros.size ==0:
            break

        cand[j] = 1
        val = compute_c5(cand)
        if val + 1e-12 < best_val:
            best, best_val = cand, val
            attempts_without_improvement = 0
        else:
            attempts_without_improvement += 1

def slsqp_refine(h_start, t_start, budget_remaining, start_time):
    n = len(h_start)
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    constraints = [{'type':'eq', 'fun': sum_con}]
    constraints += [{'type':'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_start, [t_start]])
    # set maxiter based on remaining time maybe 200?
    try:
        res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
    except Exception:
        return None
    if not res.success:
        return None
    h_opt = res.x[:n]
    h_opt = np.clip(h_opt, 0.0, 1.0)
    # enforce sum exactly
    cur_sum = h_opt.sum()
    target_sum = n/2.0
    if abs(cur_sum - target_sum) > 1e-8:
        h_opt += (target_sum - cur_sum) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def local_search_directed(h, rng, start_time, budget_s, max_no_improve=2000):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    no_improve = 0
    while time.time() - start_time < budget_s - 0.5:
        # worst shift
        raw = np.correlate(best, 1.0 - best, mode='full')
        k_idx = raw.argmax()
        k = k_idx - (n - 1)
        # mismatched ones where h[i]=1, h[i+k]=0
        if k >= 0:
            mism = np.where((best[:n - k] == 1) & (best[k:] == 0))[0]
        else:
            kk = -k
            mism = np.where((best[kk:] == 1) & (best[:n - kk] == 0))[0] + kk
        if mism.size == 0:
            ones = np.flatnonzero(best)
            i = rng.choice(ones)
        else:
            i = rng.choice(mism)
        zeros = np.flatnonzero(1 - best)
        j = rng.choice(zeros)
        if i == j:
            continue
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        val = compute_c5(cand)
        if val + 1e-12 < best_val:
            best, best_val = cand, val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    return best, best_val

def slsqp_refine(h, t_start, start_time, budget_s):
    n = len(h)
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h.astype(float), [t_start]])
    maxiter = 300
    # ensure we have time for SLSQP, else skip
    if time.time() - start_time > budget_s - 5:
        return None
    try:
        res = minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    except Exception:
        return None
    if not res.success:
        return None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce exact sum
    target_sum = n / 2.0
    cur = h_opt.sum()
    if abs(cur - target_sum) > 1e-8:
        h_opt += (target_sum - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = len(init_arr)
        # convert to binary (threshold 0.5)
        init_bin = (init_arr >= 0.5).astype(int)
    else:
        n = 80
        init_bin = None

    target_ones = n // 2

    # initial best
    if init_bin is not None:
        best_h = init_bin.astype(float)
        best_c5 = compute_c5(best_h)
    else:
        # random binary start
        start_idx = rng.choice(n, target_ones, replace=False)
        best_h = np.zeros(n, dtype=float)
        best_h[start_idx] = 1.0
        best_c5 = compute_c5(best_h)

    # Phase 1: many directed local searches
    # allocate up to 70% of budget
    while time.time() - start_time < budget_s * 0.7:
        # generate random start
        vec = np.zeros(n, dtype=int)
        vec[rng.choice(n, target_ones, replace=False)] = 1
        cand, cand_val = local_search_directed(vec, rng, start_time, budget_s)
        if cand_val < best_c5 - 1e-12:
            best_h = cand.astype(float)
            best_c5 = cand_val
            # early exit if already below target
            if best_c5 < 0.3808:
                break

    # Phase 2: refine with SLSQP if time permits
    if time.time() - start_time < budget_s - 5:
        refined = slsqp_refine(best_h, best_c5, start_time, budget_s)
        if refined is not None:
            h_opt, c5_opt = refined
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # final safety clip and ensure sum
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    target_sum = n / 2.0
    if abs(cur - target_sum) > 1e-8:
        best_h += (target_sum - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on a grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k)·dx  (=> t - overlap·dx >= 0)."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def local_search_directed(h, rng, start_time, budget_s, max_no_improve=2000):
    """Hill‑climbing that always swaps a 1 participating in the worst shift."""
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    no_improve = 0
    while time.time() - start_time < budget_s - 0.5:
        # shift giving the maximal overlap (raw counts, before scaling)
        raw = np.correlate(best, 1.0 - best, mode='full')
        k_idx = raw.argmax()
        k = k_idx - (n - 1)

        # indices of mismatched ones for that shift
        if k >= 0:
            mism = np.where((best[:n - k] == 1) & (best[k:] == 0))[0]
        else:
            kk = -k
            mism = np.where((best[kk:] == 1) & (best[:n - kk] == 0))[0] + kk

        if mism.size == 0:
            i = rng.choice(np.flatnonzero(best))
        else:
            i = rng.choice(mism)

        zeros = np.flatnonzero(1 - best)
        j = rng.choice(zeros)

        if i == j:
            continue
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        val = compute_c5(cand)

        if val + 1e-12 < best_val:
            best, best_val = cand, val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    return best, best_val

def slsqp_refine(h, t_start, start_time, budget_s):
    """Continuous refinement of a binary start via SLSQP on auxiliary variable t."""
    n = len(h)
    # build constraints t >= overlap(k)·dx
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    if time.time() - start_time > budget_s - 5:
        return None
    x0 = np.concatenate([h.astype(float), [t_start]])
    try:
        res = minimize(lambda x: x[-1],
                       x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-9, 'disp': False})
    except Exception:
        return None
    if not res.success:
        return None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    target = n / 2.0
    cur = h_opt.sum()
    if abs(cur - target) > 1e-8:
        h_opt += (target - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)

                break

    # ---- phase 2: continuous refinement (SLSQP) -----------------------
    if time.time() - start_time < budget_s - 5:
        refined = slsqp_refine(best_h, best_c5, start_time, budget_s)
        if refined is not None:
            h_opt, c5_opt = refined
            if c5_opt + 1e-12 < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # ---- final safety adjustments --------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    target = n / 2.0
    if abs(cur - target) > 1e-8:
        best_h += (target - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
