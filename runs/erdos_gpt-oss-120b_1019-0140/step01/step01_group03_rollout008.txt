def compute_gradient_shift(h, shift):
    n = h.shape[0]
    grad = np.ones(n, dtype=h.dtype)
    if shift >=0:
        grad[:n-shift] -= h[shift:]
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        grad[s:] -= h[:n - s]

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure brackets
    # Since f is decreasing, we need f(lam_low) >= s >= f(lam_high)
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def worst_shift(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (n - 1)

def compute_gradient_shift(h, shift):
    n = h.size
    grad = np.ones(n, dtype=h.dtype)
    if shift >= 0:
        if shift < n:
            grad[:n - shift] -= h[shift:]
            grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            grad[s:] -= h[:n - s]
            grad[:n - s] -= h[s:]
    return grad

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:

    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure brackets (f decreasing in lam)
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def worst_shift(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (n - 1)

def compute_gradient_shift(h, shift):
    n = h.size
    grad = np.ones(n, dtype=h.dtype)
    if shift >= 0:
        if shift < n:
            grad[:n - shift] -= h[shift:]
            grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            grad[s:] -= h[:n - s]
            grad[:n - s] -= h[s:]
    return grad

def softmax_gradient(h, tau):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    w = np.exp((corr - max_corr) / tau)
    w_sum = w.sum()
    w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)
    grad = np.ones(n, dtype=h.dtype)
    for idx, w_s in enumerate(w):
        if w_s == 0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            if shift < n:
                grad[:n - shift] -= w_s * h[shift:]
                grad[shift:] -= w_s * h[:n - shift]
        else:
            s = -shift
            if s < n:
                grad[s:] -= w_s * h[:n - s]
                grad[:n - s] -= w_s * h[s:]
    return grad

def local_search(h, cur_score, s, rng, start_time, budget_s):

            new_h[i] += delta
            new_h[j] -= delta

            new_h[j] += delta

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
