def _random_move(h, rng):
    n = h.shape[0]
    # decide move type
    move_type = rng.integers(0,2)  # 0 = swap 0 and 1; 1 = move half
    if move_type == 0:
        # find 0 and 1 positions
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            return None  # no move possible
        i = rng.choice(zeros)
        j = rng.choice(ones)
        # perform swap
        h[i], h[j] = 1.0, 0.0
        return (i, j, 'swap')
    else:
        # move half
        half_idx = np.where(np.isclose(h, 0.5))[0]
        if half_idx.size == 0:
            # maybe no half if n even - then we can treat half as 0?
            # but we only use half when odd n.
            return None
        a = half_idx[0]
        # pick any other index b (except a)
        b = rng.integers(0, n-1)
        if b >= a:
            b += 1
        # swap half with b
        val_b = h[b]
        # b can be 0 or 1
        if np.isclose(val_b, 0.0):
            h[a] = 0.0
            h[b] = 0.5
        elif np.isclose(val_b, 1.0):
            h[a] = 1.0
            h[b] = 0.5
        else:
            # other half? not possible
            return None
        return (a, b, 'move_half')

def binary_sa(n, rng, time_limit):
    target_sum = n / 2.0
    # initial solution:
    h = _binary_initial_state(n, rng)
    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    max_temp = 0.5
    min_temp = 1e-6
    while time.time() - start < time_limit:
        # compute fraction
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = max_temp * (1 - frac) + min_temp * frac
        # choose move type
        move_type = rng.integers(0, 2)  # 0 swap 0-1, 1 move half
        if move_type == 0:
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if zeros.size == 0 or ones.size == 0:
                continue
            i = rng.choice(zeros)
            j = rng.choice(ones)
            # apply swap
            h[i] = 1.0
            h[j] = 0.0
            new_c5 = _compute_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < math.exp(-delta / temp):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                # revert
                h[i] = 0.0
                h[j] = 1.0
        else:
            # move half
            half_idx = np.where(np.isclose(h, 0.5))[0]
            if half_idx.size == 0:
                # if n even, we don't have half; do a different move
                continue
            a = half_idx[0]
            # choose random b != a
            b = rng.integers(0, n-1)
            if b >= a:
                b += 1
            val_b = h[b]
            # store old values for revert
            old_a = h[a]
            old_b = h[b]
            # swap half with b
            h[a] = val_b
            h[b] = 0.5
            # ensure bounds [0,1] automatically satisfied because val_b is 0 or 1
            new_c5 = _compute_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < math.exp(-delta / temp):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[a] = old_a
                h[b] = old_b
    # after loop, ensure sum exactly
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5
```

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that sum(h) == target (within floating error)."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute evenly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _binary_sa(n: int, rng: np.random.Generator, time_limit: float):

            break
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]

        # perform the swap on the candidate
        h[i] = 1.0
        h[j] = 0.0

        new_c5 = _compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0.0:
            accept = True
        elif temp > 0.0:
            if rng.random() < math.exp(-delta / temp):
                accept = True

        if accept:
            cur_c5 = new_c5
            # update the index lists permanently
            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)
            ones.append(i)

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                it_without_improve = 0
            else:
                it_without_improve += 1
        else:
            # revert the swap
            h[i] = 0.0
            h[j] = 1.0
            it_without_improve += 1

        # occasional restart to escape stagnation
        if it_without_improve >= reset_limit:
            perm = rng.permutation(n)
            h[:] = 0.0
            h[perm[: n // 2]] = 1.0
            cur_c5 = _compute_c5(h)
            zeros = list(np.where(h == 0)[0])
            ones = list(np.where(h == 1)[0])
            it_without_improve = 0

    # final repair (should be exact for even n) and final evaluation
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h that lowers the Erdős overlap constant C₅.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: achieved C₅ value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Candidate from the supplied initial_h_values (if any)
    # ------------------------------------------------------------------
    best_h = None
    best_c5 = float("inf")
    best_n = None
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n_init = init.shape[0]
        target_init = n_init / 2.0
        init = np.clip(init, 0.0, 1.0)
        init = _repair_sum(init, target_init)
        c5_init = _compute_c5(init)
        best_h = init.copy()
        best_c5 = c5_init
        best_n = n_init
    except NameError:
        # no external construction supplied
        pass

    # ------------------------------------------------------------------
    # 2. Main binary simulated‑annealing search (even length, n=200)
    # ------------------------------------------------------------------
    # allocate the majority of the remaining budget to this search
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed)
    # ensure we leave a tiny margin for the final return
    sa_time = remaining * 0.95

    n_target = 200
    h_sa, c5_sa = _binary_sa(n_target, rng, sa_time)

    if c5_sa < best_c5:
        best_h = h_sa
        best_c5 = c5_sa
        best_n = n_target

    # ------------------------------------------------------------------
    # 3. Final safety repair and return
    # ------------------------------------------------------------------
    if best_h is None:
        # fallback: a simple half‑ones construction
        n_target = 200
        best_h = np.zeros(n_target, dtype=float)
        best_h[: n_target // 2] = 1.0
        best_h = _repair_sum(best_h, n_target / 2.0)
        best_c5 = _compute_c5(best_h)
        best_n = n_target

    # final exactness of the sum constraint
    best_h = _repair_sum(best_h, best_n / 2.0)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, best_n
