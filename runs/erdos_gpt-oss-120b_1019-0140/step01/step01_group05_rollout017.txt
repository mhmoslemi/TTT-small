import numpy as np

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip v to [0,1] and adjust its sum to exactly len(v)/2."""
    N = v.size
    target = N / 2.0
    np.clip(v, 0.0, 1.0, out=v)
    s = v.sum()
    if s == 0.0:
        v[:] = target / N
        np.clip(v, 0.0, 1.0, out=v)
        return v
    # scale to meet the sum, then re‑clip
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    # water‑filling fix‑up if clipping changed the sum
    for _ in range(5):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            free = np.where(v < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            v[free] += inc
            np.clip(v, 0.0, 1.0, out=v)
        else:
            free = np.where(v > 0.0)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            v[free] += dec
            np.clip(v, 0.0, 1.0, out=v)
    # final tiny scaling if needed
    s = v.sum()
    if abs(s - target) > 1e-12:
        v *= target / s
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a discretised step function h."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * (2.0 / N))

        if now - start >= time_limit:

        # current worst lag
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
