def _subgradient_max_corr(h):
    N = h.size
    g = np.zeros(N, dtype=np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    # tolerance to include multiple shifts close to max
    tol = max_corr * 1e-12 + 1e-14  # small.
    # Find shift indices where corr within tol of max_corr
    ks = np.where(np.abs(corr - max_corr) <= tol)[0]
    # convert to shift values
    for idx in ks:
        k = idx - (N - 1)  # shift
        if k >= 0:
            # first term: grad[0:N-k] += (1 - h[k:])
            g[:N - k] += (1.0 - h[k:])
            # second term: grad[k:] += -h[:N - k]
            g[k:] += -h[:N - k]
        else:
            kp = -k
            # first term: grad[kp:] += (1 - h[:N - kp])
            g[kp:] += (1.0 - h[:N - kp])
            # second term: grad[:N - kp] += -h[kp:]
            g[:N - kp] += -h[kp:]
    return g

def _guided_move(h, grad, max_eps, rng):
    # find indices with max positive gradient and max negative gradient
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        # fallback to random move
        return None
    # ensure we can move from i to j (mass from i to j)
    # If grad[i] <= 0 or grad[j] >= 0, maybe opposite direction is better.
    # Actually we want to move mass from i (high positive grad) to j (high negative)
    # Because decreasing h[i] reduces objective, increasing h[j] reduces objective.
    # So we assume grad[i] > 0 > grad[j]
    # If not, fallback:
    if not (grad[i] > 0 and grad[j] < 0):
        # fallback
        # try to find any pair satisfying condition
        pos_candidates = np.where(grad > 0)[0]
        neg_candidates = np.where(grad < 0)[0]
        if pos_candidates.size > 0 and neg_candidates.size > 0:
            i = int(rng.choice(pos_candidates))
            j = int(rng.choice(neg_candidates))
        else:
            return None
    # max epsilon allowed:
    eps_max = min(h[i], 1.0 - h[j], max_eps)
    if eps_max <= 0:
        # try reverse direction if possible
        eps_max = min(h[j], 1.0 - h[i], max_eps)
        if eps_max <= 0:
            return None
        i, j = j, i  # swap direction
    eps = rng.random() * eps_max
    if eps <= 0:
        return None
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def _random_swap_move(h, max_eps, rng):
    N = h.size
    i, j = rng.integers(0, N, size=2)
    while i == j:
        j = rng.integers(0, N)
    hi, hj = h[i], h[j]
    up = min(1.0 - hi, hj)
    down = min(hi, 1.0 - hj)
    if up == 0.0 and down == 0.0:
        return h.copy()
    eps = rng.random() * (up + down) - down
    eps = max(-max_eps, min(eps, max_eps))
    if eps == 0.0:
        return h.copy()
    h_new = h.copy()
    h_new[i] = hi + eps
    h_new[j] = hj - eps
    return h_new

import numpy as np, time, math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    # first scaling
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        # second scaling if needed
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _subgradient_from_corr(h: np.ndarray, corr: np.ndarray) -> np.ndarray:
    """Subgradient of max_k corr_k w.r.t. h."""
    N = h.size
    g = np.zeros(N, dtype=np.float64)
    max_corr = corr.max()
    # tolerance to include numerically equal peaks
    tol = max_corr * 1e-12 + 1e-14
    ks = np.where(np.abs(corr - max_corr) <= tol)[0]
    offset = N - 1
    for idx in ks:
        k = idx - offset
        if k >= 0:
            # grad contribution from h[i] * (1 - h[i+k])
            g[:N - k] += (1.0 - h[k:])
            # contribution from -(h[i] * (1 - h[i+k])) via second factor
            g[k:] += -h[:N - k]
        else:
            kp = -k
            g[kp:] += (1.0 - h[:N - kp])
            g[:N - kp] += -h[kp:]
    return g

def _random_swap_move(h: np.ndarray, max_eps: float, rng: np.random.Generator) -> np.ndarray:
    """Propose a feasible move by transferring epsilon between two indices."""
    N = h.size
    i, j = rng.integers(0, N, size=2)
    while i == j:
        j = rng.integers(0, N)
    hi, hj = h[i], h[j]
    up = min(1.0 - hi, hj)          # max positive epsilon
    down = min(hi, 1.0 - hj)        # max negative epsilon (as positive value)
    if up == 0.0 and down == 0.0:
        return h.copy()
    eps = rng.random() * (up + down) - down   # uniform in [-down, up]
    if abs(eps) > max_eps:
        eps = math.copysign(max_eps, eps)
    if eps == 0.0:
        return h.copy()
    h_new = h.copy()
    h_new[i] = hi + eps
    h_new[j] = hj - eps
    return h_new

def _guided_move(h: np.ndarray, grad: np.ndarray, max_eps: float, rng: np.random.Generator) -> np.ndarray | None:
    """Move mass from a high‑gradient index to a low‑gradient index."""
    pos = np.argmax(grad)
    neg = np.argmin(grad)
    if grad[pos] <= 0 or grad[neg] >= 0 or pos == neg:
        # search for any pair satisfying the sign condition
        pos_cands = np.where(grad > 0)[0]
        neg_cands = np.where(grad < 0)[0]
        if pos_cands.size == 0 or neg_cands.size == 0:
            return None
        pos = int(rng.choice(pos_cands))
        neg = int(rng.choice(neg_cands))
        if pos == neg:
            return None
    # Transfer from pos (decrease) to neg (increase)
    eps_max = min(h[pos], 1.0 - h[neg], max_eps)
    if eps_max <= 0:
        # try opposite direction
        eps_max = min(h[neg], 1.0 - h[pos], max_eps)
        if eps_max <= 0:
            return None
        pos, neg = neg, pos
    eps = rng.random() * eps_max
    if eps <= 0.0:
        return None
    h_new = h.copy()
    h_new[pos] -= eps
    h_new[neg] += eps
    return h_new

def _advanced_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Annealing with gradient‑guided moves."""
    h_cur = h_start.copy()
    N = h_cur.size
    dx = 2.0 / N
    cur_corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    cur_score = float(cur_corr.max() * dx)
    best_h = h_cur.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.02            # initial temperature
    max_eps0 = 0.2       # initial max epsilon
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        max_eps = max_eps0 * (0.1 + 0.9 * frac)   # shrink slowly
        # gradient for current state
        grad = _subgradient_from_corr(h_cur, cur_corr)
        # decide move type
        if rng.random() < 0.78:
            h_new = _guided_move(h_cur, grad, max_eps, rng)
            if h_new is None:
                h_new = _random_swap_move(h_cur, max_eps, rng)
        else:
            h_new = _random_swap_move(h_cur, max_eps, rng)
        # repair (ensures feasibility)
        h_new = _repair_vector(h_new)
        new_corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_score = float(new_corr.max() * dx)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_corr = new_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score
