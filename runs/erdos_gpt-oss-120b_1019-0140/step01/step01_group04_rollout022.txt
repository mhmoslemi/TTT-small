def _repair_vector(v, n, max_iter=100, tol=1e-9):
    target = n/2.0
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # If sum already equal target within tol, return.
    s = v.sum()
    if abs(s - target) <= tol:
        return v
    # Determine lower bound for λ: maybe -1? Actually we need to find λ such that sum(min(max(v_i - λ, 0), 1)) = target.
    # Since v_i ∈[0,1], λ can be negative to increase sum or positive to decrease.
    # We'll search λ in [-max(v), max(v)] maybe.
    # Use bisection:
    lo = -1.0  # lower bound (decrease λ increases sum)
    hi = 1.0   # upper bound (increase λ decreases sum)
    # adjust bounds to guarantee sum(lo) >= target and sum(hi) <= target
    # compute sum at lo:
    def proj_sum(lambda_):
        return np.clip(v - lambda_, 0.0, 1.0).sum()
    lo_sum = proj_sum(lo)
    hi_sum = proj_sum(hi)
    # Expand bounds if needed
    while lo_sum < target:
        lo -= 1.0
        lo_sum = proj_sum(lo)
    while hi_sum > target:
        hi += 1.0
        hi_sum = proj_sum(hi)
    for _ in range(max_iter):

        mid_sum = proj_sum(mid)
        if abs(mid_sum - target) <= tol:
            break
        if mid_sum > target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - mid, 0.0, 1.0)
    # Might still be slight off due to float; adjust if needed
    s = v.sum()
    if abs(s - target) > 1e-7:

        diff = target - s
        # add diff equally to all entries that are not at bounds
        free = (v > 0.0 + 1e-12) & (v < 1.0 - 1e-12)
        free_sum = free.sum()
        if free_sum > 0:
            v[free] += diff / free_sum
            v = np.clip(v, 0.0, 1.0)
        else:
            # fallback scaling
            factor = target / s if s>0 else 0.0
            v = np.clip(v * factor, 0.0, 1.0)
    return v

import time
import numpy as np

# ---- helper functions -----------------------------------------------------

def _repair_vector(v, n, max_iter=15):
    """Project v onto the box [0,1]^n and enforce sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adj = diff / n_free
        v[mask] -= adj
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Maximum overlap integral for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _max_corr_shift(v):
    """Return (max_corr, shift) where shift = index - (n-1)."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode="full")
    max_idx = int(np.argmax(corr))
    return float(corr[max_idx]), max_idx - (n - 1), corr

def _gradient_at_shift(v, shift):
    """Gradient of the active correlation (unscaled) w.r.t v."""
    n = len(v)
    g = np.ones(n)
    if shift >= 0:
        d = shift
        g[:n - d] -= v[d:]          # subtract h[i+shift]
        g[d:] -= v[:n - d]          # subtract h[i-shift]
    else:
        d = -shift
        g[:n - d] -= v[d:]          # subtract h[i+shift] (i+shift = i-d)
        g[d:] -= v[:n - d]          # subtract h[i-shift] (i-shift = i+d)
    return g

def _random_feasible(n):
    """Uniform random feasible vector."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _upsample_vector(v, n_target):
    """Linear interpolation to a new length, then repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _gradient_descent(v0, n, time_limit):
    """Projected gradient descent on max correlation."""
    start = time.time()
    dx = 2.0 / n
    cur = _repair_vector(v0.copy(), n)
    cur_corr = np.correlate(cur, 1.0 - cur, mode="full")
    cur_max_idx = int(np.argmax(cur_corr))
    cur_shift = cur_max_idx - (n - 1)
    cur_score = float(cur_corr[cur_max_idx]) * dx
    best = cur.copy()
    best_score = cur_score

    step0 = 0.5
    it = 0
    last_improve = 0

    while time.time() - start < time_limit:
        g = _gradient_at_shift(cur, cur_shift)
        step = step0 / np.sqrt(it + 1.0)
        cand = _repair_vector(cur - step * g, n)
        cand_corr = np.correlate(cand, 1.0 - cand, mode="full")
        cand_max_idx = int(np.argmax(cand_corr))
        cand_shift = cand_max_idx - (n - 1)
        cand_score = float(cand_corr[cand_max_idx]) * dx

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            last_improve = it

        if cand_score < cur_score:
            cur = cand
            cur_corr = cand_corr
            cur_shift = cand_shift
            cur_score = cand_score

        # stagnation handling: occasional random restart
        if it - last_improve > 5000:
            cur = _random_feasible(n)
            cur_corr = np.correlate(cur, 1.0 - cur, mode="full")
            cur_max_idx = int(np.argmax(cur_corr))
            cur_shift = cur_max_idx - (n - 1)
            cur_score = float(cur_corr[cur_max_idx]) * dx
            last_improve = it

        it += 1
        if it > 2000000:  # safety guard
            break
    return best, best_score

def _local_search(v0, n, time_limit):
    """Simple exchange hill‑climb."""
    start = time.time()
    cur = v0.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    step = 0.1
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
    return best, best_score
