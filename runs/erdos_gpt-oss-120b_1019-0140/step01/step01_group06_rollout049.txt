from scipy.optimize import differential_evolution

def de_optimize(n, rng):
    bounds = [(0.0, 1.0)] * n
    target_sum = n / 2.0
    # penalty parameter large
    penalty = 1e6
    def obj(x):
        # x is array of length n
        s = np.sum(x)
        penalty_term = penalty * (s - target_sum) ** 2
        return compute_c5(x) + penalty_term
    # custom init: we can provide initial population using rng.random
    # SciPy DE default uses random uniform within bounds; we can also supply 'init' as 'latinhypercube'
    result = differential_evolution(obj, bounds, updating='deferred', workers=1,
                                   maxiter=200, popsize=15, 
                                   seed=rng, polish=False, 
                                   mutation=(0.5, 1), recombination=0.7,
                                   tol=1e-6,
                                   callback=lambda x, convergence: None)
    x = np.clip(result.x, 0.0, 1.0)
    x = project_to_feasible(x, target_sum)
    return x, compute_c5(x)

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def binary_local_search(h, start_time, budget_s, rng, target_bound=0.3808):
    """
    Simple hill‑climbing on binary vectors (exactly half 1's).
    Swaps a random 1 with a random 0; keeps swap only if it improves C5.
    """
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    while time.time() - start_time < budget_s * 0.95:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h.copy()
            if best_c5 <= target_bound:
                break
        if c5_new >= best_c5:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)

    best_h = None
    best_c5 = float("inf")
    best_n = None
    target_bound = 0.3808

    # Sample sizes to try (larger sizes give finer resolution)
    candidate_ns = [380, 760]

    for n in candidate_ns:
        if time.time() - start > budget_s * 0.97:
            break
        target_sum = n / 2.0

        # Prepare a resampled parent if available
        parent_resampled = None
        if base_h is not None:
            parent_resampled = resample_h(base_h, n)

        # ---------- Global search with Differential Evolution ----------
        # Determine generous but safe maxiter (balance quality vs time)
        maxiter = 80
        popsize = 15
        # Initial population: include parent if present
        if parent_resampled is not None:
            init_pop = rng.random((popsize - 1, n))
            init = np.vstack([parent_resampled, init_pop])
        else:
            init = 'latinhypercube'

        penalty = 1e6

        def de_obj(x):
            return compute_c5(x) + penalty * (np.sum(x) - target_sum) ** 2

        try:
            de_res = differential_evolution(
                de_obj,
                [(0.0, 1.0)] * n,
                maxiter=maxiter,
                popsize=popsize,
                init=init,
                polish=False,
                seed=seed,
                strategy='best1bin',
                mutation=(0.5, 1.0),
                recombination=0.7,
                tol=1e-7,
                disp=False,
            )
            h_candidate = np.clip(de_res.x, 0.0, 1.0)
            h_candidate = project_to_feasible(h_candidate, target_sum)
        except Exception:
            # fallback random feasible vector
            h_candidate = rng.random(n)
            h_candidate = project_to_feasible(h_candidate, target_sum)

        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5, best_h, best_n = c5_candidate, h_candidate.copy(), n

        # ---------- Local refinement with SLSQP ----------
        if time.time() - start < budget_s * 0.93:
            cons = {"type": "eq", "fun": lambda h, ts=target_sum: np.sum(h) - ts}
            bnds = [(0.0, 1.0)] * n
            try:
                res = opt.minimize(
                    compute_c5,
                    h_candidate,
                    method="SLSQP",
                    bounds=bnds,
                    constraints=cons,
                    options={"maxiter": 600, "ftol": 1e-9, "disp": False},
                )
                if res.success:
                    h_opt = np.clip(res.x, 0.0, 1.0)
                    h_opt = project_to_feasible(h_opt, target_sum)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5, best_h, best_n = c5_opt, h_opt.copy(), n
            except Exception:
                pass

        # ---------- Binary hill‑climbing if still above target ----------
        if best_c5 > target_bound and time.time() - start < budget_s * 0.9:
            # Prepare a binary version of the current best vector
            ones_needed = int(target_sum)
            idx = np.argpartition(best_h, -ones_needed)[-ones_needed:]
            binary_start = np.zeros(n)
            binary_start[idx] = 1.0

            # Perform several restarts of the hill‑climber
            for attempt in range(5):
                if attempt == 0:
                    h_bin = binary_start.copy()
                else:
                    rnd = rng.random(n)
                    idx = np.argpartition(rnd, -ones_needed)[-ones_needed:]
                    h_bin = np.zeros(n)
                    h_bin[idx] = 1.0
                h_bin_best, c5_bin = binary_local_search(
                    h_bin, start, budget_s, rng, target_bound=target_bound
                )
                if c5_bin < best_c5:
                    best_c5, best_h, best_n = c5_bin, h_bin_best.copy(), n
                if best_c5 <= target_bound:
                    break

        if best_c5 <= target_bound:
            break

    # Fallback if nothing was found
    if best_h is None:
        if base_h is not None:
            best_h = base_h.copy()
            best_n = best_h.shape[0]
        else:
            best_n = 95
            best_h = np.full(best_n, 0.5)
        best_h = project_to_feasible(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    # Final safety projection and evaluation
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
