        elapsed = now - start
        progress = elapsed / max_time
        T = T0 * (1 - progress) + Tf * progress
        step = step0 * (1 - progress) + step_min * progress

        move_type = rng.rand()
        if move_type < 0.6:
            # pairwise exchange move
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:

            delta = (rng.rand() * 2 - 1) * step
            delta = np.clip(delta, low, high)
            candidate = current_h.copy()

            candidate[j] = hj - delta

def adjust_sum_to_target(h, target, rng):
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Determine indices that can be increased (h < 1) and decreased (h > 0)
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # pick random eligible index
        if diff > 0:
            # need increase
            inc_idxs = np.where(h < 1.0)[0]
            if inc_idxs.size == 0:
                break
            idx = rng.choice(inc_idxs)
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
        else:
            dec_idxs = np.where(h > 0.0)[0]

def adjust_sum_to_target(h, target, rng):
    # h is np.ndarray float, we will attempt to adjust in-place
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # We will try to add/subtract diff in small increments across random indices.
    # Build list of indices that can move in required direction.
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need increase
            inc_candidates = np.where(h < 1.0)[0]
            if inc_candidates.size == 0:
                break
            idx = rng.choice(inc_candidates)
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
        else:
            dec_candidates = np.where(h > 0.0)[0]
            if dec_candidates.size == 0:
                break
            idx = rng.choice(dec_candidates)
            dec = min(-diff, h[idx])
            h[idx] -= dec
            diff += dec
    # If still not perfect, distribute residual uniformly and clip
    if abs(diff) > 1e-10:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # Finally scale to exact target by adjusting a single element
        total = h.sum()
        residual = target - total
        if abs(residual) > 1e-12:
            # add residual to first element within bounds else adjust second, etc.
            for idx in range(h.size):
                if residual > 0 and h[idx] < 1.0:
                    change = min(residual, 1.0 - h[idx])
                    h[idx] += change
                    residual -= change
                elif residual < 0 and h[idx] > 0.0:
                    change = min(-residual, h[idx])
                    h[idx] -= change
                    residual += change
                if abs(residual) < 1e-12:
                    break
        # after this residual should be near 0
    return h

def compute_c5(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)·(1‑h(x+k)) dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust a vector in‑place so that its sum equals target while keeping 0≤h_i≤1.
    Uses a simple greedy redistribution; any tiny residual is spread uniformly.
    """
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Greedy redistribution
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            inc_idx = np.where(h < 1.0)[0]
            if inc_idx.size == 0:
                break
            i = rng.choice(inc_idx)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            dec_idx = np.where(h > 0.0)[0]
            if dec_idx.size == 0:
                break
            i = rng.choice(dec_idx)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    if abs(diff) > 1e-12:
        # distribute remaining error uniformly and snap back into bounds
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # final tiny correction on a feasible entry
        residual = target - h.sum()
        if abs(residual) > 1e-12:
            if residual > 0:
                idx = np.where(h < 1.0)[0]
            else:
                idx = np.where(h > 0.0)[0]
            if idx.size:
                j = idx[0]
                h[j] = np.clip(h[j] + residual, 0.0, 1.0)
    return h

def upsample_to(arr: np.ndarray, target_n: int) -> np.ndarray:
    """Linearly up‑sample a 1‑D array to length target_n on the interval [0,2)."""
    n0 = arr.size
    if n0 == target_n:
        return arr.copy()
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x = np.linspace(0.0, 2.0, target_n, endpoint=False)
    up = np.interp(x, x0, arr)
    np.clip(up, 0.0, 1.0, out=up)
    return up

    cur_h = h.copy()
    cur_c5 = best_c5

    # -------------------- simulated annealing --------------------
    T0 = 5e-2          # initial temperature
    Tf = 1e-6          # final temperature
    step0 = 1e-1       # initial step magnitude
    step_min = 1e-5    # final step magnitude

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        elapsed = now - start
        progress = elapsed / max_time
        T = T0 * (1.0 - progress) + Tf * progress
        step = step0 * (1.0 - progress) + step_min * progress

        move_type = rng.rand()
        if move_type < 0.6:
            # ----- pairwise exchange -----
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            hi, hj = cur_h[i], cur_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = max(low, min(high, delta))
            cand = cur_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
        elif move_type < 0.85:
            # ----- gradient move on worst lag -----
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n - 1)

            g = np.zeros_like(cur_h)
            idx = np.arange(n)

            plus = idx + k_shift
            mask_plus = (plus >= 0) & (plus < n)
            g[mask_plus] += (1.0 - cur_h[plus[mask_plus]])

            minus = idx - k_shift
            mask_minus = (minus >= 0) & (minus < n)
            g[mask_minus] -= cur_h[minus[mask_minus]]

            g -= g.mean()                     # keep total mass unchanged
            cand = cur_h - step * g
            np.clip(cand, 0.0, 1.0, out=cand)
            cand = adjust_sum_to_target(cand, target_sum, rng)
        else:
            # ----- tiny random perturbation of a few entries -----
            cand = cur_h.copy()
            num_change = rng.randint(1, 4)  # 1–3 entries
            indices = rng.choice(n, size=num_change, replace=False)
            for idx_i in indices:

                new_val = cand[idx_i] + delta
                new_val = max(0.0, min(1.0, new_val))
                cand[idx_i] = new_val
            cand = adjust_sum_to_target(cand, target_sum, rng)

        cand_c5 = compute_c5(cand)

        else:

            accept = rng.rand() < prob

            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

    ref_step = step0 * 0.2
    while True:
        if time.time() - start >= max_time:
            break
        corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_idx = int(np.argmax(corr_full))
        k_shift = k_idx - (n - 1)

        g = np.zeros_like(best_h)
        idx = np.arange(n)

        plus = idx + k_shift
        mask_plus = (plus >= 0) & (plus < n)
        g[mask_plus] += (1.0 - best_h[plus[mask_plus]])

        minus = idx - k_shift
        mask_minus = (minus >= 0) & (minus < n)
        g[mask_minus] -= best_h[minus[mask_minus]]

        g -= g.mean()
        cand = best_h - ref_step * g
        np.clip(cand, 0.0, 1.0, out=cand)
        cand = adjust_sum_to_target(cand, target_sum, rng)
        cand_c5 = compute_c5(cand)

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
        else:
            ref_step *= 0.99
            if ref_step < 1e-7:
                break

    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n)
