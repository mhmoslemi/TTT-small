    # When lam -> -inf, v_i - lam -> +inf => clip to upper -> upper => sum = upper*n
    # When lam -> +inf, v_i - lam -> -inf => clip to lower => sum = lower*n
    # So lambda range can be wide. For bisection we can start with lam_low = np.min(v) - upper (maybe less than needed) and lam_high = np.max(v) - lower (maybe larger).
    # Ensure sum at lam_low >= target and sum at lam_high <= target.
    low = np.min(v) - upper
    high = np.max(v) - lower
    # In rare case target is exactly at extremes but not needed.
    for _ in range(max_iter):

        s = x.sum()

        if s > target:
            # need to increase lam to reduce sum
            low = lam
        else:
            high = lam
    # return last x
    return np.clip(v - ((low+high)/2.0), lower, upper)

def worst_shift_and_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    # compute gradient of Q (product sum) for shift:
    g = np.zeros_like(h)
    if shift >= 0:
        a = h[:n - shift]
        b = h[shift:]
        g[:n - shift] += b
        g[shift:] += a
    else:
        s = -shift
        a = h[s:]  # h[i] where i >= s
        b = h[:n - s]
        g[s:] += b
        g[:n - s] += a
    # grad of C5 is -dx * g; we want direction to increase Q, i.e., +g.
    return shift, g, corr[idx] * (2.0 / n)

def worst_shift_and_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)

        b = h[shift:]

        g[shift:] += a
    else:

        a = h[s:]

        g[s:] += b

    # No scaling factor (dx not needed for direction).
    # Return shift, gradient vector g, and current max correlation (not needed)

    return shift, g, max_corr

import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a step function h sampled over [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_onto_box_and_sum(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    """Euclidean projection onto {x: lower≤x≤upper, sum(x)=target}."""
    v = np.asarray(v, dtype=float)
    # Quick feasibility test
    if np.all(v >= lower - tol) and np.all(v <= upper + tol) and abs(v.sum() - target) <= tol:
        return v.copy()
    # Bisection on lambda for shift v - λ
    lam_low = np.min(v) - upper
    lam_high = np.max(v) - lower
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    # Final adjustment
    x = np.clip(v - ((lam_low + lam_high) * 0.5), lower, upper)
    diff = target - x.sum()
    if abs(diff) > 1e-7:
        free = np.where((x > lower + 1e-12) & (x < upper - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, lower, upper)
    return x


def worst_shift_and_gradient(h):
    """Find shift achieving max correlation and return gradient of the overlap term."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    g = np.zeros_like(h)
    if shift >= 0:
        a = h[: n - shift]
        b = h[shift:]
        g[: n - shift] += b
        g[shift:] += a
    else:
        s = -shift
        a = h[s:]
        b = h[: n - s]
        g[s:] += b
        g[: n - s] += a
    return shift, g, corr[idx]


def upsample_h(h, n_target):
    """Linearly interpolate h to a new length and enforce feasibility."""
    n_src = h.shape[0]
    if n_target == n_src:
        return h.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_tgt = np.interp(x_tgt, x_src, h)
    return project_onto_box_and_sum(h_tgt, n_target / 2.0)


def gradient_descent(h0, n, max_iter, start_time, budget_s, rng):
    """Projected sub‑gradient descent targeting the worst shift."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.5
    no_improve = 0
    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.97:
            break
        _, g, _ = worst_shift_and_gradient(h)
        if not np.any(g):
            break
        h_new = h + step * g
        h_new = project_onto_box_and_sum(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
            step = min(step * 1.1, 2.0)
            no_improve = 0
            if best_c5 <= 0.3808:
                # good enough, keep refining a little
                pass
        else:
            step *= 0.5
            no_improve += 1
            if step < 1e-6 or no_improve >= 30:
                # shake to escape stagnation
                noise = (rng.random(n) - 0.5) * 0.01
                h = project_onto_box_and_sum(h + noise, n / 2.0)
                step = 0.5
                no_improve = 0
        if step < 1e-8:
            break
    return best_h, best_c5
