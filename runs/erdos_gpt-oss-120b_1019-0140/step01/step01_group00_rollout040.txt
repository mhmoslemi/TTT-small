def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.time()

    def Omax_of(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return int(corr.max())
    cur_O = Omax_of(cur_h)
    best_h = cur_h.copy()
    best_O = cur_O
    dx = 2.0 / n
    # define temperature schedule
    T0 = 2.0  # initial temperature
    # Simulated annealing loop

        if rng.random() < 0.7:

import time
import numpy as np
from scipy.optimize import minimize

# -------- helper functions ----------
def compute_c5(h):
    """Exact C5 for a (possibly fractional) step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_corr(h, alpha=30.0):
    """Smooth approximation of max correlation using log‑sum‑exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    # stable log‑sum‑exp
    s = np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    return (m + s) * dx

# constraint objects for SLSQP ----------
class SumConstraint:
    def __init__(self, n, target):
        self.n = n
        self.target = target
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapIneq:
    """t - overlap(k)*dx >= 0  (i.e. t >= overlap(k)*dx)."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k > 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:  # k < 0
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# -------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise h (binary) -----
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        # take the largest n/2 entries as a binary start
        idx = np.argsort(-init)[: n // 2]
        h0 = np.zeros(n, dtype=int)
        h0[idx] = 1
    else:
        n = 80
        h0 = np.zeros(n, dtype=int)
        h0[rng.choice(n, n // 2, replace=False)] = 1

    target_sum = n / 2.0
    dx = 2.0 / n

    best_h = h0.astype(float)
    best_c5 = compute_c5(best_h)

    # build constraints for the auxiliary‑variable formulation
    sum_con = SumConstraint(n, target_sum)
    overlap_cons = [OverlapIneq(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]   # last entry is t

    # ----- phase 1: many SLSQP restarts -----
    slsqp_budget = budget_s * 0.75
    while time.time() - start_time < slsqp_budget:

        h_start = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h_start[idx] = 1.0
        t_start = compute_c5(h_start)
        x0 = np.concatenate([h_start, [t_start]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
            )
        except Exception:

        alpha = 30.0

        def penalised_obj(h):
            # smooth max part

            # quadratic penalty for the integral constraint
            p = penalty * (np.sum(h) - target_sum) ** 2
            return m + p

        # a few local restarts around the current best
        while time.time() - start_time < budget_s - 0.5:
            # perturb the current best slightly
            perturb = rng.normal(scale=0.05, size=n)
            h_start = best_h + perturb
            h_start = np.clip(h_start, 0.0, 1.0)
            # re‑normalise the sum (simple affine shift)
            diff = target_sum - np.sum(h_start)
            h_start += diff / n
            h_start = np.clip(h_start, 0.0, 1.0)

            try:
                res = minimize(
                    fun=penalised_obj,
                    x0=h_start,
                    method='L-BFGS-B',
                    bounds=bounds_h,
                    options={'maxiter': 800, 'ftol': 1e-12, 'disp': False}
                )
            except Exception:
                break

            if not res.success:
                continue

            h_ref = np.clip(res.x, 0.0, 1.0)
            # enforce exact sum again
            diff = target_sum - np.sum(h_ref)
            h_ref += diff / n
            h_ref = np.clip(h_ref, 0.0, 1.0)

            c5_ref = compute_c5(h_ref)
            if c5_ref + 1e-12 < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                if best_c5 < 0.3805:
                    break

    # final safety clip and sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-9:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
