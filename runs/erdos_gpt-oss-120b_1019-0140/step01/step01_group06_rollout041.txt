def resample_and_binarize(base_h, n_target):
    # base_h binary
    x0 = np.linspace(0., 2., base_h.shape[0], endpoint=False)
    x = np.linspace(0., 2., n_target, endpoint=False)
    interp = np.interp(x, x0, base_h)
    # choose top half entries to be 1
    idx = np.argpartition(interp, -n_target//2)[-n_target//2:]
    new_h = np.zeros(n_target)
    new_h[idx] = 1.0
    return new_h

def compute_delta(h, p, q):
    n = len(h)
    L = 2*n - 1

def thue_morse_h(n):
    t = np.zeros(n, dtype=int)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    # convert to 0/1.
    # The sequence has about half ones; we adjust: if sum > n/2, flip some 1s to 0; else flip some zeros to 1.
    cur = t.astype(float)
    # adjust: compute diff.
    diff = int(cur.sum() - n/2)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def resample_and_binarize(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        interp = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_target = np.linspace(0.0, 2.0, n_target, endpoint=False)
        interp = np.interp(x_target, x_base, base_h)
    half = n_target // 2
    idx = np.argpartition(interp, -half)[-half:]
    bin_h = np.zeros(n_target, dtype=float)
    bin_h[idx] = 1.0
    return bin_h

def _anneal_h(h_init, max_iters, start_temp, decay, rng, start_time, budget_s):
    h = h_init.copy()
    n = h.shape[0]
    cur_corr = np.correlate(h, 1.0 - h, mode="full")
    cur_val = np.max(cur_corr) * (2.0 / n)
    best_h = h.copy()
    best_val = cur_val

    ks = np.arange(-(n - 1), n, dtype=int)            # length 2n-1
    L = 2 * n - 1
    center = n - 1
    delta = np.empty(L, dtype=float)

    T = start_temp
    for it in range(max_iters):
        if (it & 255) == 0 and (time.time() - start_time > budget_s * 0.98):
            break

        idx_max = np.argmax(cur_corr)
        shift = idx_max - center

        if shift >= 0:
            overlap = np.where((h[: n - shift] == 1) & (h[shift:] == 1))[0]
        else:
            s = -shift
            overlap = np.where((h[s:] == 1) & (h[: n - s] == 1))[0] + s

        if overlap.size and rng.random() < 0.9:
            i_out = rng.choice(overlap)
        else:
            ones = np.where(h == 1)[0]
            i_out = rng.choice(ones)

        zeros = np.where(h == 0)[0]
        i_in = rng.choice(zeros)

        pad = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
        p_plus = pad[i_out + ks + (n - 1)]
        p_minus = pad[i_out - ks + (n - 1)]
        q_plus = pad[i_in + ks + (n - 1)]
        q_minus = pad[i_in - ks + (n - 1)]

        delta[:] = p_plus + p_minus - q_plus - q_minus
        delta[center] -= 2.0
        d = i_in - i_out
        if d != 0:
            delta[center + d] += 1.0
            delta[center - d] += 1.0

        new_corr = cur_corr + delta
        new_val = np.max(new_corr) * (2.0 / n)

        diff = new_val - cur_val
        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            h[i_out] = 0.0
            h[i_in] = 1.0
            cur_corr = new_corr
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        T *= decay
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    base = globals().get("initial_h_values")

    best_h = None
    best_val = np.inf
    best_n = None

    # --- Phase 1: small resolutions ---------------------------------
    for n in (96, 190):
        if time.time() - start_time > budget_s * 0.94:
            break
        half = n // 2
        for restart in range(4):
            if time.time() - start_time > budget_s * 0.94:
                break
            if base is not None:
                h0 = resample_and_binarize(base, n)
                if restart:
                    rng.shuffle(h0)
            else:
                h0 = np.zeros(n, dtype=float)
                h0[:half] = 1.0
                rng.shuffle(h0)

            max_iters = 200000 if n == 96 else 120000
            h_opt, val_opt = _anneal_h(
                h0,
                max_iters=max_iters,
                start_temp=0.025,
                decay=0.9997,
                rng=rng,
                start_time=start_time,
                budget_s=budget_s,
            )
            if val_opt < best_val:
                best_val, best_h, best_n = val_opt, h_opt.copy(), n

    # --- Phase 2: upscale to 380 and refine -------------------------
    target_n = 380
    if best_h is not None and best_n != target_n:
        h_big = resample_and_binarize(best_h, target_n)
    elif base is not None:
        h_big = resample_and_binarize(base, target_n)
    else:
        half_big = target_n // 2
        h_big = np.zeros(target_n, dtype=float)
        h_big[:half_big] = 1.0
        rng.shuffle(h_big)

    max_iters_big = 250000
    h_refined, val_refined = _anneal_h(
        h_big,
        max_iters=max_iters_big,
        start_temp=0.02,
        decay=0.9995,
        rng=rng,
        start_time=start_time,
        budget_s=budget_s,
    )
    if val_refined < best_val:
        best_val, best_h, best_n = val_refined, h_refined.copy(), target_n

    # --- Safety check ------------------------------------------------
    if best_h is None:
        n = 96
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_n = n
        best_val = compute_c5(best_h)

    # ensure exact half ones (tiny rounding fix)
    cur_sum = best_h.sum()
    target = best_n / 2.0
    diff = int(round(target - cur_sum))
    if diff > 0:
        zeros = np.where(best_h < 0.5)[0]
        if zeros.size >= diff:
            best_h[rng.choice(zeros, diff, replace=False)] = 1.0
    elif diff < 0:
        ones = np.where(best_h > 0.5)[0]
        if ones.size >= -diff:
            best_h[rng.choice(ones, -diff, replace=False)] = 0.0

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
