def _greedy_optimize(h, time_limit, rng):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = _c5_value(h_cur)
    N = h_cur.size
    start = time.time()
    max_eps = 0.1  # limit per move
    # loop
    while time.time() - start < time_limit:
        # compute D vector
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        D = corr * (2.0 / N)
        max_idx = np.argmax(D)
        cur_score = D[max_idx]
        shift = max_idx - (N - 1)
        # compute gradient g = 1 - (h[i+shift] + h[i-shift]) (zero if out of range)
        # using vectorized operations
        i = np.arange(N)
        i_plus = i + shift
        i_minus = i - shift
        g = np.ones(N, dtype=np.float64)
        mask_plus = (i_plus >= 0) & (i_plus < N)
        g[mask_plus] -= h_cur[i_plus[mask_plus]]
        mask_minus = (i_minus >= 0) & (i_minus < N)
        g[mask_minus] -= h_cur[i_minus[mask_minus]]
        # pick indices for move
        i_pos = int(np.argmax(g))
        i_neg = int(np.argmin(g))
        # if gradient shows no improvement, we can break or try random
        if g[i_pos] <= 0 and g[i_neg] >= 0:
            # no helpful direction, break out early
            break
        # compute epsilon
        eps = min(h_cur[i_pos], 1.0 - h_cur[i_neg], max_eps)
        if eps <= 0:
            # cannot move; try random move
            i_pos = rng.integers(N)
            i_neg = rng.integers(N)
            if i_pos == i_neg:
                continue
            eps = min(h_cur[i_pos], 1.0 - h_cur[i_neg], max_eps)
            if eps <= 0:
                continue
        # apply move
        h_new = h_cur.copy()
        h_new[i_pos] -= eps
        h_new[i_neg] += eps
        # no need for repair because bounds respected and sum unchanged
        new_score = _c5_value(h_new)
        if new_score < best_score - 1e-12:
            best_h = h_new.copy()
            best_score = new_score
        # accept if improvement, else possibly accept if still reduces current max?
        if new_score <= cur_score:
            h_cur = h_new
            # continue
        else:
            # If not improvement for current max, maybe try a random small move and continue.
            # Might as well accept with low probability to escape small local plateau: simple random.
            if rng.random() < 0.01:
                h_cur = h_new
        # loop
    return best_h, best_score

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0, out=v)  # modifies in place
    target = v.size / 2.0
    s = v.sum()
    if s != target:
        if s == 0:
            # set uniform
            v[:] = target / v.size
        else:
            factor = target / s
            v *= factor
        # clip again
        v = np.clip(v, 0.0, 1.0, out=v)  # again
        s2 = v.sum()
        if abs(s2 - target) > 1e-12:
            # final scaling iteration
            factor2 = target / s2
            v *= factor2
            v = np.clip(v, 0.0, 1.0, out=v)
    return v

def _greedy_optimize(h_start, time_limit, rng):
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_score = _c5_value(h_cur)
    N = h_cur.size
    max_eps = 0.2
    start = time.time()
    no_improve_counter = 0
    while time.time() - start < time_limit:
        # compute D vector and max
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        D = corr * (2.0 / N)
        max_idx = np.argmax(D)
        cur_score = D[max_idx]
        shift = max_idx - (N - 1)

        # compute gradient
        g = np.ones(N, dtype=np.float64)
        if shift >= 0:
            if shift < N:
                g[:N-shift] -= h_cur[shift:]
                g[shift:] -= h_cur[:N-shift]
        else:
            s = -shift
            if s < N:
                g[s:] -= h_cur[:N-s]
                g[:N-s] -= h_cur[s:]

        # pick indices i_pos, i_neg
        i_pos = int(np.argmax(g))
        i_neg = int(np.argmin(g))
        # compute epsilon
        eps = min(h_cur[i_pos], 1.0 - h_cur[i_neg], max_eps)
        if eps <= 0:
            # fallback random
            i_pos = rng.integers(N)
            i_neg = rng.integers(N)
            if i_pos == i_neg:
                continue
            eps = min(h_cur[i_pos], 1.0 - h_cur[i_neg], max_eps)
            if eps <= 0:
                continue

        # perform move
        h_new = h_cur.copy()
        h_new[i_pos] -= eps
        h_new[i_neg] += eps
        # compute new D max
        new_score = _c5_value(h_new)
        if new_score < best_score - 1e-12:
            best_h = h_new.copy()
            best_score = new_score
            no_improve_counter = 0
        else:
            no_improve_counter += 1
        # accept move if new_score <= cur_score (i.e., reduces max D)
        if new_score <= cur_score:
            h_cur = h_new
        else:
            # occasional acceptance for exploration
            if rng.random() < 0.01:
                h_cur = h_new

        # if stuck (no_improve_counter high) try random perturbation
        if no_improve_counter > 200:
            # random tweak: choose two random indices a, b and move small epsilon
            i = rng.integers(N)
            j = rng.integers(N)
            if i != j:
                delta = min(h_cur[i], 1.0 - h_cur[j], max_eps * 0.1)
                if delta > 0:
                    h_cur[i] -= delta
                    h_cur[j] += delta
            no_improve_counter = 0
    return best_h, best_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip values to [0,1] and adjust sum to len(v)/2 exactly."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        return np.clip(v, 0.0, 1.0)
    factor = target / s
    v *= factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simple SA on the feasible set (box + sum constraint)."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.01
    max_eps = 0.2
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
    return best_h, best_score

def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Derivative of D(shift) = sum h_i(1-h_{i+shift}) w.r.t. h."""
    N = h.size
    g = np.ones(N, dtype=np.float64)
    if shift != 0:
        i = np.arange(N)
        ip = i + shift
        mask_p = (ip >= 0) & (ip < N)
        if mask_p.any():
            g[mask_p] -= h[ip[mask_p]]
        im = i - shift
        mask_m = (im >= 0) & (im < N)
        if mask_m.any():
            g[mask_m] -= h[im[mask_m]]
    else:
        # shift zero: subtract twice
        g -= 2.0 * h
    return g

def _greedy_optimize(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Deterministic hill–climbing focusing on the worst shift."""
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_score = _c5_value(h_cur)
    N = h_cur.size
    start = time.time()
    max_eps = 0.1
    stagnant = 0
    while time.time() - start < time_limit:
        # compute all D(k) values
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        D = corr * (2.0 / N)
        max_idx = int(np.argmax(D))
        cur_score = float(D[max_idx])
        shift = max_idx - (N - 1)
        # gradient of the worst D
        g = _gradient_for_shift(h_cur, shift)
        i_pos = int(np.argmax(g))   # most positive -> decreasing h helps
        i_neg = int(np.argmin(g))   # most negative -> increasing h helps
        eps = min(h_cur[i_pos], 1.0 - h_cur[i_neg], max_eps)
        if eps <= 0.0:
            # fallback random tweak
            i_pos = rng.integers(N)
            i_neg = rng.integers(N)
            if i_pos == i_neg:
                continue
            eps = min(h_cur[i_pos], 1.0 - h_cur[i_neg], max_eps)
            if eps <= 0.0:
                continue
        h_new = h_cur.copy()
        h_new[i_pos] -= eps
        h_new[i_neg] += eps
        new_score = _c5_value(h_new)
        if new_score < best_score - 1e-12:
            best_h = h_new.copy()
            best_score = new_score
            stagnant = 0
        else:
            stagnant += 1
        # accept if it does not worsen the current max D
        if new_score <= cur_score:
            h_cur = h_new
        else:
            if rng.random() < 0.01:   # occasional exploration
                h_cur = h_new
        if stagnant > 200:
            # random injection to escape plateaus
            a, b = rng.integers(N, size=2)
            if a != b:
                delta = min(h_cur[a], 1.0 - h_cur[b], max_eps * 0.5)
                if delta > 0.0:
                    h_cur[a] -= delta
                    h_cur[b] += delta
            stagnant = 0
    return best_h, best_score
