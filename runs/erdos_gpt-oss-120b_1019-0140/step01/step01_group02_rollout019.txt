def random_feasible(n, rng):
    # generate random vector with values [0,1] and sum exactly n/2
    # simple: start with zeros, then fill until sum reached
    target = n/2.0
    h = np.zeros(n, dtype=float)
    # set each entry to random value, then adjust.
    h = rng.random(n)  # uniform [0,1)
    # Now need to project onto hyperplane sum = target while staying in [0,1] bounds
    # We'll use iterative clipping: shift all elements by delta = (target - sum)/n, apply clipping,
    # repeat a few times.
    for _ in range(10):
        cur_sum = h.sum()
        diff = target - cur_sum
        # distribute diff uniformly
        delta = diff / n
        h += delta
        # clip to [0,1]
        np.clip(h, 0.0, 1.0, out=h)
    # final repair of residual diff using _repair_sum

def gradient_at_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        # overlapping region i = 0 .. n-1-lag
        # term: h[i]*(1 - h[i+lag])
        # derivative w.r.t h[i] from first factor: (1-h[i+lag])
        # derivative w.r.t h[i+lag] from second factor: -h[i]
        i = np.arange(0, n-lag)
        grad[i] += (1.0 - h[i+lag])
        grad[i+lag] -= h[i]
    else:

        L = -lag
        # overlapping i = 0 .. n-1-L

        # Derivative w.r.t h[i+L] from first: (1 - h[i])

    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine size n
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n_init = init.shape[0]
        # upscale to target size
        n = 200  # can be larger for better ref
        # Interpolate initial to new size
        x_init = np.linspace(0, 2, n_init, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        current = np.interp(x_new, x_init, init)
    else:
        n = 200
        # generate random feasible vector
        current = np.random.rand(n)  # We'll adjust later
    target_sum = n / 2.0
    # Clip and repair to feasible region
    np.clip(current, 0.0, 1.0, out=current)
    current = _repair_sum(current, target_sum)
    # Initialize best
    current_c5 = _compute_c5(current)
    best_h = current.copy()
    best_c5 = current_c5

    # Parameters
    max_step = 0.4
    min_step = 1e-4
    max_temp = 0.02
    min_temp = 1e-6
    # optional: track no improvement epochs
    no_improve_cnt = 0

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s:
            break
        frac = elapsed / budget_s
        step = max_step * (1 - frac) + min_step * frac
        temp = max_temp * (1 - frac) + min_temp * frac

        # worst lag detection
        n = current.shape[0]
        corr = np.correlate(current, 1.0 - current, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        lag = max_idx - (n - 1)
        current_c5 = max_val * (2.0 / n)

        # gradient
        grad = _gradient_at_lag(current, lag)

        # propose move via subgradient
        proposal = current - step * grad
        np.clip(proposal, 0.0, 1.0, out=proposal)
        proposal = _repair_sum(proposal, target_sum)

        # evaluate
        new_c5 = _compute_c5(proposal)

        # acceptance
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < math.exp(-delta / temp):
            current = proposal
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = proposal.copy()
                no_improve_cnt = 0
            else:
                no_improve_cnt += 1
        else:
            no_improve_cnt += 1

        # occasional random perturbation to escape stagnation
        if no_improve_cnt >= 200:
            # random swap move
            i, j = rng.integers(0, n, size=2)
            if i != j:
                hi, hj = current[i], current[j]
                delta_min = max(-hi, -(1.0 - hj))
                delta_max = min(1.0 - hi, hj)
                low = max(delta_min, -step)
                high = min(delta_max, step)
                if low < high:
                    delta_r = rng.random() * (high - low) + low

                    pert[j] = hj - delta_r

            no_improve_cnt = 0

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that sum(h) == target while staying in [0,1]."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase the sum
        idx = np.where(h < 1.0 - 1e-12)[0]
        if idx.size == 0:
            h += diff / h.size
            np.clip(h, 0.0, 1.0, out=h)
            return h
        i = idx[0]
        inc = min(diff, 1.0 - h[i])
        h[i] += inc
        diff -= inc
        if abs(diff) > 1e-12:
            return _repair_sum(h, target)
    else:
        # need to decrease the sum
        idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            h -= diff / h.size
            np.clip(h, 0.0, 1.0, out=h)
            return h
        i = idx[0]
        dec = min(-diff, h[i])
        h[i] -= dec
        diff += dec
        if abs(diff) > 1e-12:
            return _repair_sum(h, target)
    return h


def _gradient_at_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Subgradient of the worst‑lag correlation w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if lag >= 0:
        i = np.arange(0, n - lag)
        grad[i] += 1.0 - h[i + lag]
        grad[i + lag] -= h[i]
    else:
        L = -lag
        i = np.arange(0, n - L)
        grad[i + L] += 1.0 - h[i]
        grad[i] -= h[i + L]
    return grad
