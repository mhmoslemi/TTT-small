def compute_c5_fft(h):
    n = h.shape[0]
    g = 1.0 - h
    g_rev = g[::-1]
    L = 1 << ((2*n - 1).bit_length())  # next power of two
    F = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g_rev, n=L)
    corr = np.fft.irfft(F * G, n=L)[:2*n-1]  # first 2n-1 elements
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def compute_c5(h):
    n = h.shape[0]
    g = 1.0 - h
    # reverse g
    rev = g[::-1]
    L = 1 << ((2*n - 1).bit_length())
    # Real FFT:
    F = np.fft.rfft(h, n=L)
    G = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(F * G, n=L)
    # Take first 2n-1 entries
    corr = conv[:2*n - 1]
    return np.max(corr) * (2.0 / n)

def hill_climb_binary(n, rng, max_iter=2000, max_no_improve=500):
    # start with random binary solution
    half = n // 2
    ones_idxs = rng.choice(n, size=half, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[ones_idxs] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5

    ones_set = set(ones_idxs.tolist())
    zeros_set = set(range(n)) - ones_set

    no_improve_counter = 0
    for it in range(max_iter):
        # choose random swap
        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))
        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)
        if new_c5 < cur_c5 - 1e-12:
            # accept
            cur_c5 = new_c5
            # update sets
            ones_set.remove(i)
            zeros_set.add(i)
            zeros_set.remove(j)
            ones_set.add(j)
            # maybe update best
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                no_improve_counter = 0
            else:
                no_improve_counter += 1
        else:
            # reject, revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_improve_counter += 1

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Fast C5 evaluation using FFT (linear correlation of h with 1-h)."""
    n = h.shape[0]
    g = 1.0 - h
    rev = g[::-1]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    F = np.fft.rfft(h, n=L)
    G = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(F * G, n=L)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=20):
    """Project h onto [0,1]^n while enforcing sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust a single entry with remaining capacity
        idx = np.argmax(1.0 - h) if cur < target_sum else np.argmin(h)
        h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
    return h

def hill_climb_binary(n, rng, max_iter=2000, max_no_improve=500):
    """Local search on binary vectors (half ones) using random swaps."""
    half = n // 2
    ones = rng.choice(n, size=half, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[ones] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)

    cur_h = h.copy()
    cur_c5 = best_c5

    ones_set = set(ones.tolist())
    zeros_set = set(range(n)) - ones_set

    no_improve = 0
    for _ in range(max_iter):
        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))
        # propose swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)
        if new_c5 < cur_c5 - 1e-12:
            # accept improvement
            cur_c5 = new_c5
            ones_set.remove(i)
            zeros_set.add(i)
            zeros_set.remove(j)
            ones_set.add(j)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # reject, revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_improve += 1
        if no_improve >= max_no_improve:
            break
    return best_h, best_c5

def refine_with_slsqp(h, n):
    """Local continuous refinement with SLSQP, respecting box and mass constraints."""
    h = project_to_feasible(h, n / 2.0)
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bnds = [(0.0, 1.0)] * n
    try:

    except Exception:
        return h, compute_c5(h)
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)
