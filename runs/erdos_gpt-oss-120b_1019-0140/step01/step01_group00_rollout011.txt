def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def initial_binary_from_continuous(arr):

    return binary

            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            # record best if improved
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        # occasional local improvement could be added, but not needed
    return best_h, best_c5

import time
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Maximum overlap C5 for a binary (or 0‑1) vector."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def _binary_from_continuous(arr: np.ndarray) -> np.ndarray:
    """Convert any array to a binary vector with exactly half ones."""
    n = arr.shape[0]
    half = n // 2
    # indices of the largest half entries
    idx = np.argpartition(-arr, half)[:half]
    b = np.zeros(n, dtype=int)
    b[idx] = 1
    return b


def _hill_climb_best_swap(h: np.ndarray, start_time: float, budget_s: float) -> tuple:
    """Exhaustive steepest‑descent hill climbing (swap 1↔0)."""
    cur_c5 = _c5_of(h)
    n = h.shape[0]
    while True:
        # stop if time is nearly exhausted
        if time.time() - start_time > budget_s - 0.1:
            break
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        best_swap = None
        best_new_c5 = cur_c5
        # examine all possible swaps
        for i in ones:
            for j in zeros:
                cand = h.copy()
                cand[i], cand[j] = 0, 1
                c5cand = _c5_of(cand)
                if c5cand < best_new_c5 - 1e-12:
                    best_new_c5 = c5cand
                    best_swap = (i, j)
        if best_swap is None:
            break  # local optimum reached
        i, j = best_swap
        h[i], h[j] = 0, 1
        cur_c5 = best_new_c5
    return h, cur_c5

    n = h.shape[0]
    best_h = h.copy()
    best_c5 = cur_c5
    # maintain indices of ones and zeros for fast sampling
    ones = set(np.flatnonzero(h == 1))
    zeros = set(np.flatnonzero(h == 0))
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.1:
            break
        if not ones or not zeros:
            break
        # linear cooling schedule, max temperature 1e-3
        T = max(0.0, 1e-3 * (1.0 - elapsed / budget_s))
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        cand = h.copy()
        cand[i], cand[j] = 0, 1
        cand_c5 = _c5_of(cand)
        delta = cand_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            prob = np.exp(-delta / (T + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            h = cand
            cur_c5 = cand_c5
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5


def _improve(start_vec: np.ndarray, rng: np.random.Generator,
             start_time: float, budget_s: float) -> tuple:
    """Sequence of local descent → SA → final descent."""
    # 1. deterministic steepest descent
    h, c5 = _hill_climb_best_swap(start_vec.copy(), start_time, budget_s)
    # 2. simulated annealing (if time permits)
    if time.time() - start_time < budget_s - 0.5:
        h, c5 = _simulated_annealing(h, c5, rng, start_time, budget_s)
    # 3. final steepest descent
    h, c5 = _hill_climb_best_swap(h, start_time, budget_s)
    return h, c5


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- determine problem size and starting vector ----
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        start_bin = _binary_from_continuous(init_arr)
    else:
        n = 80
        start_bin = np.zeros(n, dtype=int)
        start_bin[rng.choice(n, n // 2, replace=False)] = 1

    # initial best
    best_h = start_bin.copy()
    best_c5 = _c5_of(best_h)

    # improve the deterministic start
    h_opt, c5_opt = _improve(best_h, rng, start, budget_s)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt
    if best_c5 < 0.3808:
        return best_h.astype(float), best_c5, n

    # ---- random restarts -------------------------------------------------
    while time.time() - start < budget_s - 1.0:
        # generate a fresh random half‑ones vector
        cand = np.zeros(n, dtype=int)
        cand[rng.choice(n, n // 2, replace=False)] = 1
        h_opt, c5_opt = _improve(cand, rng, start, budget_s)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break

    # ---- final polishing --------------------------------------------------
    # best_h is binary; convert to float for the required return type
    final_h = best_h.astype(float)
    final_h = np.clip(final_h, 0.0, 1.0)

    # enforce exact integral (tiny numerical corrections)
    cur_sum = final_h.sum()
    target_sum = n / 2.0
    if abs(cur_sum - target_sum) > 1e-9:
        final_h += (target_sum - cur_sum) / n
        final_h = np.clip(final_h, 0.0, 1.0)

    return final_h, best_c5, n
