def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # pick an index where move possible
    if diff > 0:
        # need to increase
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: normalize by average offset
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = np.clip(h[i] + diff, 0.0, 1.0)
    actual = new_val - h[i]
    h[i] = new_val
    # recursion if leftover diff
    if abs(actual - diff) > 1e-12:
        return repair_sum(h, target)
    return h
```

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust one entry (or recursively several) to satisfy the exact sum."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: spread adjustment evenly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    Search for a step function h on [0,2] (discretized) that lowers the
    Erdős overlap constant C5.

        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """

    # Ensure feasibility (tiny numeric drift)
