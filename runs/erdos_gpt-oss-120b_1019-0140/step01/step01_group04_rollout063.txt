import time, numpy as np

def _repair_vector(v, n):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v *= factor
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-9:
            v += (target - s) / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1 - v, mode='full')
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

        fft_g = np.fft.rfft(rev_g, n=L)

    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _optimize_gradient(v0, time_limit):
    """Projected gradient descent that directly reduces the worst overlap."""
    start = time.time()
    n = len(v0)
    L = 1 << ((2 * n - 2).bit_length())  # power of two ≥ 2n‑1
    v = _repair_vector(v0.copy(), n)
    best_v = v.copy()
    best_score = _compute_c5(v)
    it = 0
    while time.time() - start < time_limit:
        rev_g = (1.0 - v)[::-1]
        fft_v = np.fft.rfft(v, n=L)
        fft_g = np.fft.rfft(rev_g, n=L)
        conv = np.fft.irfft(fft_v * fft_g, n=L)
        conv = conv[: 2 * n - 1]
        idx = int(np.argmax(conv))
        lag = idx - (n - 1)

        if lag == 0:
            g = 2.0 * v - 1.0
        else:
            k = abs(lag)
            g = np.zeros_like(v)
            g[: n - k] += v[k:]
            g[k:] += v[: n - k]

        d = g - np.mean(g)                     # zero‑mean direction
        alpha = 0.5 / np.sqrt(it + 1)          # decreasing stepsize
        v_new = v + alpha * d
        v_new = _repair_vector(v_new, n)

        cur_score = conv[idx] * (2.0 / n)       # dx = 2/n
        if cur_score < best_score:
            best_score = cur_score
            best_v = v_new.copy()
        v = v_new
        it += 1
    return best_v, best_score

def _local_search(v0, time_limit):
    """Simple hill‑climbing by transferring mass between two sites."""
    start = time.time()
    n = len(v0)
    v = v0.copy()
    best_v = v.copy()
    best_score = _compute_c5(v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        if delta > 0:
            max_t = min(delta, v[i], 1.0 - v[j])
            if max_t <= 0:
                continue
            v[i] -= max_t
            v[j] += max_t
        else:
            d = -delta
            max_t = min(d, v[j], 1.0 - v[i])
            if max_t <= 0:
                continue
            v[j] -= max_t
            v[i] += max_t
        v = _repair_vector(v, n)
        score = _compute_c5(v)
        if score < best_score:
            best_score = score
            best_v = v.copy()
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # ----- initial construction ------------------------------------------------
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
        parent = _repair_vector(parent, n0)
    except Exception:
        n0 = 200
        parent = np.random.rand(n0)
        parent = _repair_vector(parent, n0)

    current_v = parent.copy()
    current_n = n0
    best_v = current_v.copy()
    best_score = _compute_c5(best_v)
    best_n = current_n

    # ----- multiresolution optimisation ---------------------------------------
    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed - 0.5
        if remaining <= 0:
            break

        # gradient descent on the current grid
        time_opt = min(remaining * 0.75, 300.0)
        v_opt, sc_opt = _optimize_gradient(current_v, time_opt)
        if sc_opt < best_score:
            best_score, best_v, best_n = sc_opt, v_opt.copy(), current_n

        # a quick local‑search polishing step
        elapsed = time.time() - start
        remaining = budget_s - elapsed - 0.5
        if remaining > 0.2:
            t_ls = min(remaining * 0.2, 30.0)
            v_ls, sc_ls = _local_search(v_opt, t_ls)
            if sc_ls < best_score:
                best_score, best_v, best_n = sc_ls, v_ls.copy(), current_n

        # decide whether to go to a finer grid
        if current_n * 2 <= 800:
            next_n = current_n * 2
            current_v = _upsample_vector(v_opt, next_n)
            current_n = next_n
            continue
        break

    # ----- final repair & return ------------------------------------------------
    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)
    return best_v.astype(np.float64), float(best_score), int(best_n)
