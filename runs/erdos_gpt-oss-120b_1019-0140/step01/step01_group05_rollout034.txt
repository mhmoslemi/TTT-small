def _min_auto_corr(h):
    n = h.size
    # Compute convolution of h and reversed h
    # Use FFT with zero padding length Nfft:
    Nfft = 1 << ((2*n-1).bit_length())

def _min_auto_corr_fft(h):
    n = h.size
    Nfft = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, Nfft)
    auto = np.fft.irfft(H * np.conj(H), Nfft)[:2*n - 1]
    # rounding to nearest integer for robustness
    return int(np.rint(auto).min())

def _binary_sa(h_start, time_limit, rng):
    h = h_start.copy()
    n = h.size
    cur_score = _c5_from_h_binary(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.02
    # pre-calc for acceptance
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = 1 - elapsed / time_limit
        T = T0 * max(frac, 0.0)
        # random swap
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or h[i] == h[j]:
            continue
        # swap
        if h[i] == 1:
            hi_idx = i
            lo_idx = j
        else:
            hi_idx = j
            lo_idx = i
        # Perform swap
        h[hi_idx] = 0
        h[lo_idx] = 1
        new_score = _c5_from_h_binary(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[hi_idx] = 1
            h[lo_idx] = 0
    return best_h, best_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value_fft(h: np.ndarray) -> float:
    """Compute C5 = 1 - 2*min_auto/n using FFT based autocorrelation."""
    n = h.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, Nfft)
    auto = np.fft.irfft(H * np.conj(H), Nfft)[: 2 * n - 1]
    min_auto = float(auto.min())
    if min_auto < 0.0:
        min_auto = 0.0
    return 1.0 - (2.0 * min_auto) / n

def _simulated_annealing_v2(h_start: np.ndarray, time_limit: float,
                           rng: np.random.Generator):
    """SA mixing swap moves (preserving sum) and Gaussian perturbations."""
    h_cur = h_start.copy()
    cur_score = _c5_value_fft(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    n = h_cur.size
    start = time.time()
    T0 = 0.02          # initial temperature
    max_eps = 0.5      # max Gaussian std for perturb
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        if rng.random() < 0.5:
            # ---- swap move -------------------------------------------------
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or h_cur[i] == h_cur[j]:

            hi, hj = h_cur[i], h_cur[j]
            h_cur[i], h_cur[j] = hj, hi
            new_score = _c5_value_fft(h_cur)
