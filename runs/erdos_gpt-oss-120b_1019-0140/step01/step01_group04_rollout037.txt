In the context of Erdős minimum overlap problem: Usually for subsets of [0,1] of measure 1/2, C_5 = sup over all shifts t in ℝ of measure of A ∩ (A + t)^c (i.e., portion of A not overlapped). def _simulated_annealing(v_bin, budget):
    """Perform SA on binary vector v_bin, preserving sum."""
    start = time.time()
    n = len(v_bin)
    target = n/2
    # ensure binary: 0 or 1:
    v = v_bin.copy()
    # initial score
    best_v = v.copy()
    best_score = _compute_c5(v)
    curr_score = best_score

    # define temperature schedule: start at T0 = 0.1? or maybe (max_score - min_score) * something
    # We'll set initial temperature to 0.05 (since scores in [0,1]), decays exponentially based on elapsed fraction.
    while time.time() - start < budget:
        # propose swap of one 1 and one 0
        ones = np.where(v > 0.5)[0]
        zeros = np.where(v <= 0.5)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        v[i], v[j] = v[j], v[i]  # swap values (now i becomes 0, j becomes 1)
        new_score = _compute_c5(v)
        dt = new_score - curr_score
        # temperature based on fraction of budget elapsed
        elapsed_frac = (time.time() - start) / budget
        T = (0.1) * (1.0 - elapsed_frac)  # linear cooling from 0.1 to 0
        accept = False
        if dt <= 0:
            accept = True
        else:
            if T > 0:
                if np.random.rand() < np.exp(-dt / T):
                    accept = True
        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score

import time
import numpy as np
from scipy.optimize import minimize

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _optimize_at_resolution(n, init_v, time_limit):
    """Run SLSQP from init_v (and random restarts) within time_limit."""
    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)

    def obj(x):
        return _compute_c5(x)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bounds = [(0.0, 1.0)] * n

    # Primary run
    try:
        res = minimize(
            obj,
            init_v,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 800, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            v = _repair_vector(res.x, n)
            score = _compute_c5(v)
            if score < best_score:
                best_score, best_v = score, v
    except Exception:
        pass

    # Random restarts
    while time.time() - start < time_limit:
        rnd = _random_feasible(n)
        try:
            res = minimize(
                obj,
                rnd,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 400, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n)
                score = _compute_c5(v)
                if score < best_score:
                    best_score, best_v = score, v
        except Exception:
            pass
        if time.time() - start > time_limit * 0.9:
            break
    return best_v, best_score

def _binarize_vector(v):
    """Return a binary (0/1) version of v with the required sum."""
    n = len(v)
    target = n / 2.0
    ones_needed = int(round(target))
    if abs(ones_needed - target) < 1e-9:
        idx = np.argsort(-v)
        v_bin = np.zeros_like(v)
        v_bin[idx[:ones_needed]] = 1.0
        return v_bin
    # odd n case – fall back to threshold+repair
    v_bin = (v >= 0.5).astype(float)
    return _repair_vector(v_bin, n)

def _simulated_annealing(v_bin, budget):
    """SA on binary vector that preserves the number of ones."""
    start = time.time()
    n = len(v_bin)
    v = v_bin.copy()
    best_v = v.copy()
    best_score = _compute_c5(v)
    curr_score = best_score

    # Pre‑compute indices of ones/zeros each iteration (cheap for these sizes)
    while time.time() - start < budget:
        # pick a random 1 and a random 0 to swap
        ones = np.where(v > 0.5)[0]
        zeros = np.where(v <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        v[i], v[j] = v[j], v[i]
        new_score = _compute_c5(v)
        dt = new_score - curr_score
        # linear cooling from 0.1 to 0
        elapsed = time.time() - start
        T = max(0.0, 0.1 * (1.0 - elapsed / budget))
        accept = False
        if dt <= 0:
            accept = True
        elif T > 0.0 and np.random.rand() < np.exp(-dt / T):
            accept = True
        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_v = v.copy()
        else:
            # revert swap
            v[i], v[j] = v[j], v[i]
    return best_v, best_score

        dt = new_score - curr_score
        elapsed = time.time() - start
        T = max(0.0, 0.1 * (1.0 - elapsed / budget))
        accept = False
        if dt <= 0:
            accept = True
        elif T > 0.0 and np.random.rand() < np.exp(-dt / T):
            accept = True
        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_v = v.copy()
        else:
            v[i], v[j] = v[j], v[i]
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function for the Erdős C₅ problem."""
    np.random.seed(seed)
    start = time.time()

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 150
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0 // 2] = 1.0
    parent = _repair_vector(parent, n0)

    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    resolutions = [n0]
    if n0 * 2 <= 400:
        resolutions.append(n0 * 2)

    total_budget = max(0.0, budget_s - 5.0)
    per_res = total_budget / len(resolutions) if resolutions else 0.0

    for n in resolutions:
        elapsed = time.time() - start
        if elapsed >= budget_s - 1.0:
            break
        if best_n != n:
            init_guess = _upsample_vector(best_v, n)
        else:
            init_guess = best_v.copy()

        remaining = budget_s - elapsed - 1.0
        time_allowed = min(per_res, remaining)

        v_opt, sc_opt = _optimize_at_resolution(n, init_guess, time_allowed * 0.6)
        if sc_opt < best_score:
            best_score, best_v, best_n = sc_opt, v_opt, n

        sa_time = max(0.5, time_allowed * 0.4)
        v_bin = _binarize_vector(best_v)
        v_sa, sc_sa = _simulated_annealing(v_bin, sa_time)
        if sc_sa < best_score:
            best_score, best_v, best_n = sc_sa, v_sa, n

    while time.time() - start < budget_s - 0.5:
        rnd = _random_feasible(best_n)
        sc = _compute_c5(rnd)
        if sc < best_score:
            best_score, best_v = sc, rnd

    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(best_n)
