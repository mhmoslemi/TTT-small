```
def project_to_feasible(h, target_sum, tol=1e-8, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - np.sum(h)
        if abs(diff) <= tol:
            break
        # Determine where we can add/subtract
        # For diff>0, we need to increase some interior points (or those at 0) that are not full
        # For diff<0, we need to decrease some interior points (or those at 1) that are not zero
        # We can distribute diff evenly among points that are not at bound
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # all at bounds; adjust some nearest to bound direction
            # find indices with room
            if diff > 0:
                # increase those at 0
                free = np.where(h < 1.0)[0]
            else:
                free = np.where(h > 0.0)[0]
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final clamp to exact sum if needed within tolerance
    cur = np.sum(h)
    if abs(cur - target_sum) > tol:
        # adjust a single entry
        adjustment = target_sum - cur
        # find index with capacity
        if adjustment > 0:
            idx = np.argmax(1.0 - h)  # max room to increase
        else:
            idx = np.argmin(h)  # max room to decrease
        h[idx] = np.clip(h[idx] + adjustment, 0.0, 1.0)
    return h
```

```
def smoothed_obj_and_grad(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    # compute softmax weights
    # To improve numeric stability
    max_val = np.max(alpha * corr)
    exp_term = np.exp(alpha * corr - max_val)
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    # compute weighted shifts
    w_arr = w
    # compute convs
    conv_w = np.convolve(w_arr, h, mode='full')
    conv_wr = np.convolve(w_arr[::-1], h, mode='full')
    offset = n - 1
    weighted_minus = conv_w[offset: offset + n]
    weighted_plus = conv_wr[offset: offset + n]
    grad = 1.0 - (weighted_plus + weighted_minus)
    obj = (np.log(sum_exp) + max_val) / alpha  # log-sum-exp / alpha
    return obj, grad, corr
```
