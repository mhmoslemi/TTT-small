def c5_value_fft(h):
    N = h.size
    # compute convolution of h and (1 - h) via FFT.
    # Use full length = 2N-1; we can pad to next power of 2.
    m = 1 << ((2*N-1).bit_length())
    # real FFT
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * G, n=m)[:2*N-1]
    dx = 2.0 / N
    return float(dx * np.max(corr))

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    g_rev = (1.0 - h)[::-1]
    m = 1 << ((2*N-1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(g_rev, n=m)
    conv = np.fft.irfft(H * G, n=m)[:2*N-1]
    return float(dx * np.max(conv))

Implementation:

def _c5_value(h):
    N = h.size
    dx = 2.0 / N
    g_rev = (1.0 - h)[::-1]
    m = 1 << ((2*N-1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(g_rev, n=m)
    conv = np.fft.irfft(H * G, n=m)[:2*N-1]
    return float(dx * np.max(conv))

This is robust.

def _simulated_annealing_improved(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    # initial temperature based on typical delta maybe
    # we can start with T0 = 0.01 * cur_score
    T0 = max(1e-3, 0.01 * cur_score) # adjust
    # set max epsilon initial large fraction, decays
    max_eps = 0.5
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = 1.0 - elapsed / time_limit
        T = T0 * (frac ** 1.5)  # slower cooling
        eps_limit = max_eps * frac
        # pick two indices i != j
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        # compute allowed epsilon positive and negative as before
        up = min(1.0 - hi, hj)  # amount we can move from j to i: increase hi
        down = min(hi, 1.0 - hj)  # amount we can move from i to j: decrease hi
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down  # uniform [-down, up]
        # restrict magnitude
        if abs(eps) > eps_limit:
            eps = math.copysign(eps_limit, eps)
        # if eps zero skip
        if eps == 0.0:
            continue
        # propose new h
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # evaluate new
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
    return best_h, best_score

def _simulated_annealing_efficient(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    # initial temperature T0 set based on cur_score magnitude
    T0 = max(0.001, cur_score * 0.01)  # e.g., 1% of score
    max_eps0 = 0.5
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        frac = 1.0 - elapsed / time_limit
        # Temperature decays linearly (or exponential)
        T = T0 * (frac ** 1.5)
        max_eps = max_eps0 * frac
        # pick two distinct indices
        i = rng.integers(0, N)
        j = rng.integers(0, N-1)
        if j >= i:
            j += 1
        hi, hj = h_cur[i], h_cur[j]
        # compute up/down as before
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        # Clip to max_eps magnitude
        if max_eps > 0.0 and abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # possible small rounding errors outside bounds: clip
        # but should be within [0,1] due to up/down.
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
    return best_h, best_score

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        # set uniform value target/size
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function using FFT."""
    N = h.size
    dx = 2.0 / N
    g_rev = (1.0 - h)[::-1]
    m = 1 << ((2 * N - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(g_rev, n=m)
    conv = np.fft.irfft(H * G, n=m)[: 2 * N - 1]
    return float(dx * np.max(conv))

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Improved SA moving mass between two coordinates while preserving total sum."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = max(0.001, cur_score * 0.01)  # initial temperature scaled to score
    max_eps0 = 0.5                      # initial move magnitude
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        frac = 1.0 - elapsed / time_limit
        T = T0 * (frac ** 1.5)                # slower cooling
        max_eps = max_eps0 * frac
        # pick distinct indices
        i = rng.integers(0, N)
        j = rng.integers(0, N - 1)
        if j >= i:
            j += 1
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)      # how much we can increase hi
        down = min(hi, 1.0 - hj)    # how much we can decrease hi
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        if max_eps > 0.0 and abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
    return best_h, best_score

def _increase_resolution(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a higher resolution and repair."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _repair_vector(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Obtain starting vector
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 200
        h0 = np.zeros(n0, dtype=np.float64)
        h0[: n0 // 2] = 1.0
        rng.shuffle(h0)
    h0 = _repair_vector(h0)

    start_total = time.time()
    best_h, best_score = h0.copy(), _c5_value(h0)

    # -------- Phase 1: intensive SA at current resolution ----------
    phase1_budget = budget_s * 0.6
    phase1_start = time.time()
    while True:
        elapsed = time.time() - phase1_start
        if elapsed >= phase1_budget:
            break
        remaining = phase1_budget - elapsed
        sub_time = min(20.0, remaining)
        # random restart: half the time start from best with a tiny perturbation,
        # otherwise start from a fresh random binary vector.
        if rng.random() < 0.5:
            h_start = best_h.copy()
            i, j = rng.integers(0, h_start.size, size=2)
            while i == j:
                j = rng.integers(0, h_start.size)
            hi, hj = h_start[i], h_start[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up + down > 0.0:
                eps = (rng.random() * (up + down) - down) * 0.1
                h_start[i] = hi + eps
                h_start[j] = hj - eps
        else:
            N = h0.size
            h_start = np.zeros(N, dtype=np.float64)
            h_start[: N // 2] = 1.0
            rng.shuffle(h_start)
        h_start = _repair_vector(h_start)
        h_candidate, cand_score = _simulated_annealing(h_start, sub_time, rng)
        if cand_score < best_score:
            best_h, best_score = h_candidate, cand_score

    elapsed = time.time() - start_total
    remaining = budget_s - elapsed

    # -------- Phase 2: increase resolution if time permits ----------
    while remaining > 30.0:
        new_n = min(best_h.size * 2, 1000)
        if new_n == best_h.size:
            break
        h_up = _increase_resolution(best_h, new_n)
        # allocate a sizable chunk of the remaining budget
        phase_time = min(remaining * 0.7, 300.0)
        h_candidate, cand_score = _simulated_annealing(h_up, phase_time, rng)
        if cand_score < best_score:
            best_h, best_score = h_candidate, cand_score
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed

    # Final repair and evaluation
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
