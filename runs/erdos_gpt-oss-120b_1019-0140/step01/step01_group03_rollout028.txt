def optimize_candidate(h0, max_iter, time_limit, start_time):
    n = h0.size
    s = n/2
    lr = 0.1
    beta1=0.9; beta2=0.999; eps=1e-8
    m = np.zeros_like(h0)
    v = np.zeros_like(h0)
    h = h0.copy()
    best_local_h = h.copy()
    best_local_score = compute_c5(h)
    for t in range(1, max_iter+1):
        # time check
        if time.time() - start_time > time_limit - 1:
            break
        # compute gradient of worst shift
        # compute correlation; get shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # gradient
        if shift >= 0:
            term1 = np.empty_like(h)
            term2 = np.empty_like(h)

            term2[shift:] = h[:n-shift]
            term2[:shift] = 0.0
        else:

            term1 = np.empty_like(h)
            term2 = np.empty_like(h)

def optimize_adam(h0, max_iter, start_time, budget_s):
    """Projected Adam optimization. Returns best h and its C5."""
    # We use a copy of h0.
    h = h0.copy()
    n = h.size
    s = n / 2.0
    lr = 0.2  # step size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter + 1):
        # Check time
        if time.time() - start_time > budget_s - 1:
            break
        # Compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # Compute gradient of C5 for this shift
        grad = np.ones_like(h)
        if shift >= 0:
            # overlapping region
            k = shift
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            k = -shift
            grad[k:] -= h[:n - k]
            grad[:n - k] -= h[k:]
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        # Project onto box and sum constraint
        h = project_onto_box_sum(h, s)
        # Evaluate
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

import numpy as np
import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    lo_sum = lo * n
    hi_sum = hi * n
    if s <= lo_sum:
        return np.full_like(v, lo)
    if s >= hi_sum:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam_mid, lo, hi)
        x_sum = x.sum()
        if abs(x_sum - s) < 1e-12:
            return x
        if x_sum > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_seq(n):
    """Balanced Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def sine_wave(n, freq=1):
    """Sine-shaped candidate, values in [0,1] and sum exactly n/2."""
    i = np.arange(n, dtype=float)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * i / n)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_box_sum(h, n / 2.0)

def optimize_adam(h0, max_iter, start_time, budget_s):
    """Projected Adam optimizer targeting min max‑overlap (C5)."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    dx = 2.0 / n
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    # initial evaluation
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = np.max(corr) * dx
    best_h = h.copy()
    for t in range(1, max_iter + 1):
        if time.time() - start_time > budget_s - 1:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        score = max_corr * dx
        idx = corr.argmax()
        shift = idx - (n - 1)

        # gradient of the active worst–shift term
        grad = np.ones_like(h)
        if shift >= 0:
            k = shift
            grad[:n - k] -= h[k:]
            grad[k:]     -= h[:n - k]
        else:
            k = -shift
            grad[k:]     -= h[:n - k]
            grad[:n - k] -= h[k:]

        # Adam update with decaying learning rate
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        lr_t = 0.3 / np.sqrt(t)
        step = lr_t * m_hat / (np.sqrt(v_hat) + eps)

        h = h - step
        h = project_onto_box_sum(h, s)

        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score
