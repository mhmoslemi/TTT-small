import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _simulated_annealing(v0, time_limit):
    """Simple SA that conserves the total sum exactly."""
    if time_limit <= 0:
        return v0.copy(), _compute_c5(v0)
    n = len(v0)
    v = _repair_vector(v0, n)
    cur_score = _compute_c5(v)
    best_v, best_score = v.copy(), cur_score
    start = time.time()
    step_max, step_min = 0.3, 0.001
    T0 = max(cur_score * 0.05, 1e-6)
    Tf = 1e-6
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = step_max * (1 - frac) + step_min * frac
        T = T0 * (1 - frac) + Tf * frac
        i, j = np.random.choice(n, 2, replace=False)
        # direction 0: i -> j, direction 1: j -> i
        if np.random.rand() < 0.5:
            max_delta = min(step, v[i], 1.0 - v[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            cand = v.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            max_delta = min(step, v[j], 1.0 - v[i])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            cand = v.copy()
            cand[j] -= delta
            cand[i] += delta
        new_score = _compute_c5(cand)
        dscore = new_score - cur_score
        if dscore <= 0 or np.random.rand() < np.exp(-dscore / max(T, 1e-12)):
            v = cand
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_v = cur_score, v.copy()
    return best_v, best_score

def _targeted_refinement(v0, time_limit):
    """Focus on the shift with maximal overlap and exchange mass to lower it."""
    if time_limit <= 0:
        return v0.copy(), _compute_c5(v0)
    n = len(v0)
    v = _repair_vector(v0, n)
    best_v, best_score = v.copy(), _compute_c5(v)
    start = time.time()
    step = 0.2
    while time.time() - start < time_limit:
        corr = np.correlate(v, 1.0 - v, mode="full")
        s_idx = np.argmax(corr)
        s = s_idx - (n - 1)  # shift value
        if s >= 0:
            i_range = np.arange(0, n - s)
            j_range = i_range + s
        else:
            i_range = np.arange(-s, n)
            j_range = i_range + s
        contrib = v[i_range] * (1.0 - v[j_range])
        if contrib.size == 0:
            break
        i_high = i_range[np.argmax(contrib)]
        # choose a random low-index to receive mass
        low_candidates = np.where(v < 1.0 - 1e-12)[0]
        if low_candidates.size == 0:
            break
        t = np.random.choice(low_candidates)
        max_delta = min(step, v[i_high], 1.0 - v[t])
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        cand = v.copy()
        cand[i_high] -= delta
        cand[t] += delta
        new_score = _compute_c5(cand)
        if new_score < best_score:
            best_score, best_v, v = new_score, cand.copy(), cand
        else:
            # occasional uphill acceptance
            if np.random.rand() < 0.01:
                v = cand
    return best_v, best_score
