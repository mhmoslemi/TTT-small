# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    """Project v onto the hypercube [lo,hi]^n while fixing the sum to target."""
    # Binary search a constant λ so that sum(clip(v+λ, lo, hi)) == target
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Initialise dimensionality and feasible start vector ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---- Prepare constraints (shared across all SLSQP calls) ----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # ---- Phase 1: Repeated SLSQP restarts (≈70% of budget) ----
    phase1_deadline = start_time + budget_s * 0.7
    while time.time() < phase1_deadline:
        # Random feasible start
        h_start = rng.random(n)
        h_start = project_to_box_sum(h_start, target_sum)
        t_start = compute_c5(h_start)

        res = slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800)
        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # Enforce exact sum after optimisation
        cur_sum = h_opt.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            h_opt = project_to_box_sum(h_opt, target_sum)

        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            if best_c5 < 0.3806:
                break

    # ---- Phase 2: Continuous hill‑climbing / simulated annealing ----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    base_step = 0.5
    while time.time() - start_time < budget_s - 0.5:
        # Random zero‑mean direction
        d = rng.normal(size=n)
        d -= d.mean()
        step = base_step * (rng.random() ** 2)
        cand = cur_h + step * d
        cand = project_to_box_sum(cand, target_sum)

        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                if best_c5 < 0.3805:
                    break
        # Occasionally shrink the step size
        if rng.random() < 0.001:
            base_step *= 0.9

    # ---- Final safety projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
