def hill_climb(h, current_c5, budget_remain, rng):
    n = h.shape[0]
    start = time.time()
    while time.time() - start < budget_remain:
        # Choose random indices i, j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        # ensure we can move mass from i to j. Let's move delta from i to j.
        if h[i] <= 1e-12 or h[j] >= 1 - 1e-12:
            continue
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
    return h, current_c5

import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=20):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # indices which can move
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        # apply delta, keep within bounds
        if diff > 0:
            h[free] = np.minimum(h[free] + delta, 1.0)
        else:
            h[free] = np.maximum(h[free] + delta, 0.0)
        h = np.clip(h, 0.0, 1.0)
    # Final fix if still off
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # distribute remaining diff uniformly
        diff = target_sum - cur
        # find any index not at bound
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def hill_climb(h, cur_c5, time_budget, rng):
    n = h.shape[0]
    start = time.time()
    # limit attempts to prevent endless loop
    attempts = 0
    while time.time() - start < time_budget and attempts < 5000:
        attempts += 1
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # move from i to j
        if h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5:
            h = h_new
            cur_c5 = c5_new
    return h, cur_c5

def generate_patterns(n, target_sum):
    patterns = []
    # contiguous block
    h1 = np.zeros(n, dtype=float)
    h1[:n//2] = 1.0
    patterns.append(h1)
    # alternating
    h2 = np.zeros(n, dtype=float)
    h2[::2] = 1.0
    patterns.append(h2)
    # sinusoidal
    x = np.linspace(0.0, 2*np.pi, n, endpoint=False)
    h3 = 0.5 + 0.5*np.sin(x)
    patterns.append(h3)
    # parent interpolation if available
    base_h = globals().get('initial_h_values')
    if base_h is not None:
        base_arr = np.asarray(base_h, dtype=float)
        if base_arr.shape[0] != n:
            h4 = resample_h(base_arr, n)
        else:
            h4 = base_arr.copy()
        patterns.append(h4)
    # project all to feasible
    proj = [project_to_feasible(p, target_sum) for p in patterns]
    return proj

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # candidate sample sizes (balance between speed and resolution)
    candidate_ns = [190, 380]
    for n in candidate_ns:
        target = n / 2.0
        # number of restarts per size
        restarts = 12 if n <= 190 else 8
        # generate deterministic patterns
        patterns = generate_patterns(n, target)
        # loop over restarts
        for r in range(restarts):
            # time check
            if time.time() - start_time > budget_s * 0.96:
                break
            if r < len(patterns):
                h0 = patterns[r].copy()
            else:
                h0 = rng.random(n)
                h0 = project_to_feasible(h0, target)
            # small random perturbation for later restarts
            if r >= len(patterns):
                noise = rng.normal(scale=0.05, size=n)
                h0 = np.clip(h0 + noise, 0.0, 1.0)
                h0 = project_to_feasible(h0, target)

            # SLSQP local refinement
            cons = {'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t}
            bnds = [(0.0, 1.0)] * n
            try:
                res = opt.minimize(
                    fun=lambda x: compute_c5(x),
                    x0=h0,
                    method='SLSQP',
                    bounds=bnds,
                    constraints=cons,

                )
            except Exception:
                continue

            h_opt = project_to_feasible(h_opt, target)
            c5_val = compute_c5(h_opt)

            # local hill climbing improvement if time permits
            remaining = budget_s - (time.time() - start_time)
            if remaining > 0.5:
                # allocate up to 0.3 seconds for hill climb
                hc_time = min(0.3, remaining * 0.3)
                h_opt, c5_val = hill_climb(h_opt, c5_val, hc_time, rng)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

            # early exit if already very good
            if best_c5 <= 0.3807:
                # keep searching a bit more for possible better
                pass

        # If time left, try differential evolution on this n (only once)
        time_left = budget_s - (time.time() - start_time)
        if time_left > 5.0 and n == 190:
            # differential evolution global search
            bounds = [(0.0, 1.0)] * n
            penalty = 200.0
            def de_obj(x, t=target):
                c5 = compute_c5(x)
                diff = np.sum(x) - t
                return c5 + penalty * abs(diff)
            de_seed = int(rng.integers(0, 2**32 - 1))
            try:
                de_res = opt.differential_evolution(
                    de_obj,
                    bounds,
                    seed=de_seed,
                    maxiter=30,
                    popsize=15,
                    polish=False,
                    updating='deferred',
                    atol=0,
                    tol=0,
                )
                h_de = np.clip(de_res.x, 0.0, 1.0)
                h_de = project_to_feasible(h_de, target)
                c5_de = compute_c5(h_de)
                # refine with SLSQP
                cons_de = {'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t}
                try:
                    res_de = opt.minimize(
                        fun=lambda x: compute_c5(x),
                        x0=h_de,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=cons_de,
                        options={'maxiter': 800, 'ftol': 1e-9, 'disp': False},
                    )
                    h_de_opt = np.clip(res_de.x, 0.0, 1.0)
                    h_de_opt = project_to_feasible(h_de_opt, target)
                    c5_de_opt = compute_c5(h_de_opt)
                    if c5_de_opt < best_c5:
                        best_c5 = c5_de_opt
                        best_h = h_de_opt.copy()
                        best_n = n
                except Exception:
                    pass
            except Exception:
                pass

    # Final refinement on best solution with hill climbing if time permits
    if best_h is not None:
        remaining = budget_s - (time.time() - start_time)
        if remaining > 1.0:
            best_h, best_c5 = hill_climb(best_h, best_c5, remaining * 0.5, rng)

    # Fallback if no solution found
    if best_h is None:
        n = 190
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Ensure final feasibility
    best_h = project_to_feasible(best_h, best_n / 2.0)

    # final C5 recompute
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
