    # configuration
    pop_size = 15
    max_gens = 2000  # will break earlier due to time
    # initialize population
    # Use base_h plus random perturbations
    population = np.empty((pop_size, n))
    for i in range(pop_size):
        if i == 0:
            population[i] = base_h
        else:
            # random start: random numbers, project to feasible
            x = rng.random(n)
            population[i] = project_to_feasible(x, n/2.0)
    # evaluate
    scores = np.array([compute_c5(ind) for ind in population])
    # sort
    order = np.argsort(scores)
    population = population[order]
    scores = scores[order]
    best = population[0].copy()
    best_score = scores[0]
    # generation loop
    gen = 0
    while time.time() - start_time < time_limit * 0.95 and gen < max_gens:
        # produce offspring
        offspring = []
        for i in range(pop_size):
            # selection: choose two parents (tournament)
            idx1, idx2 = rng.choice(pop_size, size=2, replace=False)
            p1, p2 = population[idx1], population[idx2]
            # crossover
            alpha = rng.random()
            child = alpha * p1 + (1 - alpha) * p2
            # mutation: small random shift of mass
            if rng.random() < 0.5:
                # pick two indices
                i1, i2 = rng.choice(n, size=2, replace=False)
                delta = rng.normal(scale=0.05)  # adjust scale
                # Transfer delta from i1 to i2
                if delta > 0:
                    # move from i1 to i2
                    move = min(delta, child[i1])
                    child[i1] -= move
                    child[i2] = min(1.0, child[i2] + move)
                else:
                    move = min(-delta, child[i2])
                    child[i2] -= move
                    child[i1] = min(1.0, child[i1] + move)
            # ensure feasible
            child = project_to_feasible(child, n/2.0)
            offspring.append(child)
        offspring = np.stack(offspring)
        offspring_scores = np.array([compute_c5(ind) for ind in offspring])
        # combine and select best pop_size
        combined = np.concatenate([population, offspring], axis=0)
        combined_scores = np.concatenate([scores, offspring_scores], axis=0)
        idx = np.argsort(combined_scores)[:pop_size]
        population = combined[idx]
        scores = combined_scores[idx]
        # update best
        if scores[0] < best_score:
            best = population[0].copy()
            best_score = scores[0]
        gen += 1
    return best, best_score

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # Determine free indices that can be moved (not at bounds)
        free_up = np.where(h < 1.0)[0]
        free_down = np.where(h > 0.0)[0]
        # If diff > 0: need increase sum: can increase free_up entries.
        if diff > 0 and free_up.size > 0:
            delta = diff / free_up.size
            h[free_up] += delta
            # ensure not exceed 1
            h = np.clip(h, 0.0, 1.0)
        elif diff < 0 and free_down.size > 0:
            delta = diff / free_down.size  # diff negative
            h[free_down] += delta
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    # final small correction: adjust a single entry
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust the first non-boundary if possible else any
        idx = 0
        # find index with room
        for i in range(len(h)):
            if (target_sum > cur and h[i] < 1.0) or (target_sum < cur and h[i] > 0.0):
                idx = i
                break
        h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
    return h
```

def resample_h(source_h, n_target):
    source_h = np.asarray(source_h, dtype=float)
    n_src = source_h.shape[0]
    if n_target == n_src:
        return source_h.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_tgt, x_src, source_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new
```

def refine_with_slsqp(h0, n, start_time, budget_s):
    # if not enough time left, skip
    if time.time() - start_time > budget_s * 0.95:
        return h0, compute_c5(h0)
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    # assign maxiter proportional to remaining time
    remaining = budget_s - (time.time() - start_time)
    maxiter = max(100, int(min(500, remaining * 5)))
    try:
        res = opt.minimize(fun=lambda x: compute_c5(x),
                           x0=h0,
                           method="SLSQP",
                           bounds=bounds,
                           constraints=cons,
                           options={"maxiter": maxiter, "ftol": 1e-12, "disp": False})

import time
import numpy as np
import scipy.optimize as opt

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    """Clip to [0,1] and adjust to hit the target sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # entries that can move up or down
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust a single free variable if possible
        idx = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
    return h


def resample_h(source_h, n_target):
    """Linear interpolation of a step function to a new resolution, then project."""
    source_h = np.asarray(source_h, dtype=float)
    n_src = source_h.shape[0]
    if n_target == n_src:
        h_new = source_h.copy()
    else:
        x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
        x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_tgt, x_src, source_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def compute_c5(h):
    """Fast C5 evaluation via FFT‑based convolution (max correlation)."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=size)
    R = np.fft.rfft(rev, n=size)
    conv = np.fft.irfft(H * R, n=size)[: 2 * n - 1]
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)


def evolutionary_optimize(init_h, n, start_time, budget_s, rng):
    """Simple (mu+lambda) evolutionary search with projection."""
    target = n / 2.0
    pop_size = 20
    # initialise population
    pop = np.empty((pop_size, n))
    for i in range(pop_size):
        if i == 0:
            pop[i] = init_h.copy()
        elif i == 1:
            # binary random starter (exactly half ones)
            x = np.zeros(n)
            x[: n // 2] = 1.0
            rng.shuffle(x)
            pop[i] = x.astype(float)
        else:
            x = rng.random(n)
            pop[i] = project_to_feasible(x, target)
    scores = np.array([compute_c5(ind) for ind in pop])
    order = np.argsort(scores)
    pop = pop[order]
    scores = scores[order]
    best = pop[0].copy()
    best_score = scores[0]

    max_gens = 400
    gen = 0
    while gen < max_gens and (time.time() - start_time) < budget_s * 0.95:
        # generate offspring
        off = np.empty_like(pop)
        for i in range(pop_size):
            a, b = rng.choice(pop_size, size=2, replace=False)
            p1, p2 = pop[a], pop[b]
            alpha = rng.random()
            child = alpha * p1 + (1 - alpha) * p2
            child += rng.normal(scale=0.03, size=n)  # mutation noise
            child = project_to_feasible(child, target)
            off[i] = child
        off_scores = np.array([compute_c5(ind) for ind in off])
        # combine and keep the best pop_size individuals
        combo = np.vstack((pop, off))
        combo_scores = np.concatenate((scores, off_scores))
        idx = np.argsort(combo_scores)[:pop_size]
        pop = combo[idx]
        scores = combo_scores[idx]
        if scores[0] < best_score:
            best_score = scores[0]
            best = pop[0].copy()
        gen += 1
    return best, best_score


def refine_with_slsqp(h0, n, start_time, budget_s):
    """Local refinement using SLSQP, respecting the time budget."""
    if (time.time() - start_time) > budget_s * 0.96:
        return h0, compute_c5(h0)
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    remaining = budget_s - (time.time() - start_time)
    maxiter = max(100, int(min(600, remaining * 5)))
    try:
        res = opt.minimize(
            fun=lambda x: compute_c5(x),
            x0=h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-12, "disp": False},
        )
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_to_feasible(h_opt, target)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    parent = globals().get("initial_h_values")

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Determine candidate grid sizes
    if parent is not None:
        base_n = len(parent)
        cand = [base_n, base_n * 2, max(1, base_n // 2)]
    else:
        cand = [190, 380, 760]
    cand = sorted(set(n for n in cand if n > 0 and n <= 1520))

    for n in cand:
        if (time.time() - start) > budget_s * 0.96:
            break

        if parent is not None:
            init_h = resample_h(parent, n)
        else:
            x0 = rng.random(n)
            init_h = project_to_feasible(x0, n / 2.0)

        # Evolutionary global search
        h_evo, c5_evo = evolutionary_optimize(init_h, n, start, budget_s, rng)

        # Local SLSQP refinement
        h_ref, c5_ref = refine_with_slsqp(h_evo, n, start, budget_s)

        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n

        # optional early stop if target achieved
        if best_c5 <= 0.3808:
            # continue searching for possible further improvement
            pass

    if best_h is None:
        # fallback uniform half‑density solution
        best_n = 380
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # final projection and evaluation
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
