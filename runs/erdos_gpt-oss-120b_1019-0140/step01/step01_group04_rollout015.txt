def _repair_vector(v, n, max_iter=15):
    """Project vector onto [0,1]^n with exact sum n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return max overlap integral for a discrete step function."""
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

    v[:half] = 1.0
    np.random.shuffle(v)
    return v

    v = np.random.rand(n)

def _upsample_vector(v, n_target):
    """Linear interpolation to a larger grid followed by repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _hill_climb(v, max_iter, time_limit):
    """Binary hill‑climbing: swap a 1 and a 0 if it improves C5."""
    start = time.time()
    cur = v.copy()
    cur_score = _compute_c5(cur)
    best_v = cur.copy()
    best_score = cur_score
    n = len(cur)
    for _ in range(max_iter):
        if time.time() - start > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if cur[i] == cur[j]:
            continue  # same value, swap would do nothing
        # perform swap
        cur[i], cur[j] = cur[j], cur[i]
        new_score = _compute_c5(cur)
        if new_score < cur_score:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_v = cur.copy()
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]
    return best_v, best_score

def _continuous_local_search(v, max_iter, time_limit):
    """Small fractional transfers between two coordinates while preserving sum."""
    start = time.time()
    cur = v.copy()
    cur_score = _compute_c5(cur)
    n = len(cur)
    for _ in range(max_iter):
        if time.time() - start > time_limit:
            break
        i, j = np.random.choice(n, 2, replace=False)
        if cur[i] <= 0.0 or cur[j] >= 1.0:
            continue
        max_transfer = min(cur[i], 1.0 - cur[j])
        if max_transfer <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta
        new_score = _compute_c5(cur)
        if new_score < cur_score:
            cur_score = new_score
        else:
            cur[i] += delta
            cur[j] -= delta
    return cur, cur_score
