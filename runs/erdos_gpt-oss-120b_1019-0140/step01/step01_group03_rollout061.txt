def adam_optimize(h0, max_iter, alpha, lr):

    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_score = np.inf
    target_n = 256
    # Helper to get sum target
    target_sum = target_n / 2.0
    # Build initial candidates
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != target_n:
            # linear interpolation to target_n
            init = np.interp(np.linspace(0,1,target_n,endpoint=False),
                             np.linspace(0,1,init.size,endpoint=False), init)
        init = project_onto_box_sum(init, target_sum)
        candidates.append(init)
    # Thue-Morse binary
    candidates.append(thue_morse(target_n).astype(float))
    # Random binary balanced
    for _ in range(2):
        candidates.append(random_binary_balanced(target_n, rng))
    # Random fractional
    for _ in range(2):
        v = rng.random(target_n)
        candidates.append(project_onto_box_sum(v, target_sum))
    # Simple half-block pattern
    half_block = np.concatenate([np.ones(target_n//2), np.zeros(target_n//2)], dtype=float)
    candidates.append(project_onto_box_sum(half_block, target_sum))
    # Phase 1: short optimization
    short_iters = 3000
    for h0 in candidates:
        if time.time() - start > budget_s - 2.0:
            break
        h_opt = adam_optimize(h0, max_iter=short_iters, lr=0.2,
                              min_alpha=5.0, max_alpha=15.0,
                              start_time=start, budget_s=budget_s)
        score = compute_c5(h_opt)
        if score < best_score:
            best_score = score
            best_h = h_opt.copy()
    # Phase 2: refinement of best
    if best_h is not None:
        refine_iters = 15000
        h_ref = adam_optimize(best_h, max_iter=refine_iters, lr=0.1,
                              min_alpha=10.0, max_alpha=30.0,
                              start_time=start, budget_s=budget_s)
        score = compute_c5(h_ref)
        if score < best_score:
            best_score = score
            best_h = h_ref.copy()
    # Phase 3: perturbations
    perturb_trials = 8
    perturb_iters = 5000
    for _ in range(perturb_trials):
        if best_h is None:
            break
        if time.time() - start > budget_s - 1.0:
            break
        eps = 0.02
        h_pert = best_h + eps * rng.normal(size=target_n)
        h_pert = project_onto_box_sum(h_pert, target_sum)
        h_opt = adam_optimize(h_pert, max_iter=perturb_iters, lr=0.15,
                              min_alpha=5.0, max_alpha=20.0,
                              start_time=start, budget_s=budget_s)
        score = compute_c5(h_opt)
        if score < best_score:
            best_score = score
            best_h = h_opt.copy()
    # Fallback if no solution found
    if best_h is None:
        best_h = np.full(target_n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, target_sum)
        best_score = compute_c5(best_h)
    else:
        # final projection to ensure constraints exactly
        best_h = project_onto_box_sum(best_h, target_sum)
        best_score = compute_c5(best_h)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo≤x_i≤hi, sum x_i = s}."""
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1 values)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq


def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def smooth_grad(h, alpha):
    """Gradient of the smooth approximation of max correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # softmax over all shifts
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)  # shape (2n-1,)

    grad = np.ones_like(h)
    # compute weighted sum of shifted terms
    for s in range(n):
        total_w = w[n - 1 + s] + w[n - 1 - s]
        if total_w == 0.0:
            continue
        if s == 0:
            grad -= total_w * h
        else:
            grad[:n - s] -= total_w * h[s:]
            grad[s:] -= total_w * h[:n - s]
    return grad


def adam_optimize(h0, max_iter, lr, min_alpha, max_alpha, start_time, budget_s):
    """Adam optimizer with time‑aware early exit and gradually increasing smoothing."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, max_iter + 1):
        if time.time() - start_time > budget_s - 0.5:
            break
        alpha = min_alpha + (max_alpha - min_alpha) * (t / max_iter)
        grad = smooth_grad(h, alpha)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, target_sum)
    return h
