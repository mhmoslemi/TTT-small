def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    a = h.astype(float)
    b = (1.0 - h).astype(float)
    L = int(1 << ((2*n - 1).bit_length()))  # nearest power of two >= 2n
    # pad to L
    A = np.fft.rfft(a, n=L)
    B_rev = np.fft.rfft(b[::-1], n=L)
    conv = np.fft.irfft(A * B_rev, n=L)
    # conv length L, but actual correlation length = 2n-1
    # slice central part: conv[:2*n-1] (since convolution result starts at index 0)
    corr = conv[:2*n-1]
    # multiply by dx and take max
    return np.max(corr) * dx

def crossover(p1, p2, rng):
    n = len(p1)
    mask = rng.integers(0,2,size=n)
    child = np.where(mask, p1, p2)
    # adjust
    target = n // 2
    diff = int(child.sum() - target)
    if diff > 0:
        # too many 1s: turn diff ones to zero
        ones_idx = np.flatnonzero(child)
        # choose diff to set to zero
        turn_off = rng.choice(ones_idx, diff, replace=False)
        child[turn_off] = 0
    elif diff < 0:
        # too few ones: turn -diff zeros to one
        zeros_idx = np.flatnonzero(child == 0)
        turn_on = rng.choice(zeros_idx, -diff, replace=False)
        child[turn_on] = 1
    return child

def mutate_swap(ind, rng):
    n1 = np.flatnonzero(ind == 1)
    n0 = np.flatnonzero(ind == 0)
    if len(n1) == 0 or len(n0) == 0:
        return ind
    i = rng.choice(n1)
    j = rng.choice(n0)
    ind[i] = 0
    ind[j] = 1
    return ind

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    target = n // 2   # sum h = target (since h are binary)

    Hr = np.fft.rfft(h[::-1].astype(np.float64), n=L)
    conv = np.fft.irfft(H * Hr, n=L)
    conv = conv[:2*n-1]
    # ignore central element (index n-1)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    target = n / 2.0  # sum of h values, may not be integer for real h but sum must equal this
    # Use convolution of h with reversed h
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    Hr = np.fft.rfft(h[::-1], n=L)
    conv = np.fft.irfft(H * Hr, n=L)
    conv = conv[:2*n-1]
    # Excluding central element
    mask = np.ones_like(conv, dtype=bool)
    mask[n-1] = False
    min_autocorr = conv[mask].min()
    # Clip negativity
    if min_autocorr < 0:
        min_autocorr = 0.0
    c5 = (target - min_autocorr) * dx
    return c5

def tournament_selection(fitness, tournament_size, rng):
    idx = rng.choice(len(fitness), tournament_size, replace=False)
    best = idx[np.argmin(fitness[idx])]
    return best

def crossover(p1, p2, rng):
    n = len(p1)
    mask = rng.integers(0, 2, size=n, dtype=np.int8)
    child = np.where(mask, p1, p2).astype(np.int8)
    target = n // 2
    diff = int(child.sum() - target)
    if diff > 0:
        # too many ones
        ones_idx = np.flatnonzero(child)
        turn_off = rng.choice(ones_idx, diff, replace=False)
        child[turn_off] = 0
    elif diff < 0:
        zeros_idx = np.flatnonzero(child == 0)
        turn_on = rng.choice(zeros_idx, -diff, replace=False)
        child[turn_on] = 1
    return child

def mutate_swap(ind, rng):
    ones = np.flatnonzero(ind == 1)
    zeros = np.flatnonzero(ind == 0)
    if len(ones) == 0 or len(zeros) == 0:
        return ind
    i = rng.choice(ones)
    j = rng.choice(zeros)
    ind[i] = 0
    ind[j] = 1
    return ind

def hill_climb_continuous(h, cur_c5, start_time, budget, rng):
    best_h = h.copy()
    best_c5 = cur_c5
    n = len(h)
    target = n / 2.0
    eps = 1e-12
    while time.time() - start_time < budget - 0.5:
        # pick i (source) and j (dest)
        # source > 0, dest < 1
        # pick i with probability weighted by amount?
        # just random
        i_candidates = np.where(best_h > eps)[0]
        j_candidates = np.where(best_h < 1 - eps)[0]
        if len(i_candidates) == 0 or len(j_candidates) == 0:
            break
        i = rng.choice(i_candidates)
        j = rng.choice(j_candidates)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        # try move
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best_h = cand
            best_c5 = cand_c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    # Determine n
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = len(init_h)
    else:
        n = 80
        init_h = None

    target_ones = n // 2  # integer for binary

    # Prepare initial binary from init_h if available:
    if init_h is not None:
        # convert to binary by top half selection
        # if init_h contains values not exactly binary, we can use ranking
        idx = np.argsort(-init_h)[:target_ones]
        binary0 = np.zeros(n, dtype=np.int8)
        binary0[idx] = 1
    else:
        binary0 = None

    # GA parameters
    pop_size = 400  # large enough
    elite_size = 5
    tournament_size = 5
    mutation_prob = 0.2
    # allocate time for GA: maybe 80% of budget
    ga_budget = budget_s * 0.85
    # Ensure at least some time for final refinement
    # Create initial population
    pop = np.empty((pop_size, n), dtype=np.int8)
    for i in range(pop_size):
        idx = rng.choice(n, target_ones, replace=False)
        vec = np.zeros(n, dtype=np.int8)
        vec[idx] = 1
        pop[i] = vec
    # insert binary0 if available
    if binary0 is not None:
        pop[0] = binary0

    # Compute fitness of each individual
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(pop[i].astype(float))
    best_idx = np.argmin(fitness)
    best_h_bin = pop[best_idx].copy()
    best_c5 = float(fitness[best_idx])

    # GA main loop
    while time.time() - start_time < ga_budget - 5.0:
        # generate new population
        new_pop = np.empty_like(pop)
        # elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        new_pop[:elite_size] = pop[elite_indices]
        # generate offspring
        for i in range(elite_size, pop_size):
            p1 = tournament_selection(fitness, tournament_size, rng)
            p2 = tournament_selection(fitness, tournament_size, rng)
            child = crossover(pop[p1], pop[p2], rng)
            if rng.random() < mutation_prob:
                child = mutate_swap(child, rng)
            new_pop[i] = child

        # evaluate new pop
        new_fitness = np.empty(pop_size, dtype=float)
        for i in range(pop_size):
            new_fitness[i] = compute_c5(new_pop[i].astype(float))

        # keep best
        min_idx = np.argmin(new_fitness)
        if new_fitness[min_idx] < best_c5 - 1e-12:
            best_c5 = float(new_fitness[min_idx])
            best_h_bin = new_pop[min_idx].copy()

        # replace population
        pop = new_pop
        fitness = new_fitness

        # early stopping
        if best_c5 < 0.3808:
            break

    # Convert best binary to float array for further refinement
    best_h = best_h_bin.astype(float)

    # Continuous refinement (hill climbing) using remaining time
    remaining = budget_s - (time.time() - start_time) - 0.5
    if remaining > 0.5:
        best_h, best_c5 = hill_climb_continuous(best_h, best_c5, start_time, budget_s, rng)

    # Final safety: clip and enforce sum exactly target within small epsilon
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_ones) > 1e-8:
        # adjust uniformly to correct sum
        best_h += (target_ones - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid.
    h is a 1‑D array of length n with sum(h)=n/2."""
    n = h.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    # length for FFT (next power of two >= 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    Hr = np.fft.rfft(h[::-1], n=L)
    conv = np.fft.irfft(H * Hr, n=L)[: 2 * n - 1]
    # exclude the central element (zero shift)
    mask = np.ones(conv.shape, dtype=bool)
    mask[n - 1] = False
    min_autocorr = conv[mask].min()
    if min_autocorr < 0:
        min_autocorr = 0.0
    return (target - min_autocorr) * dx


def _tournament_select(fitness: np.ndarray, k: int, rng: np.random.Generator) -> int:
    idx = rng.choice(len(fitness), k, replace=False)
    return idx[np.argmin(fitness[idx])]


def _crossover(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = p1.shape[0]
    mask = rng.integers(0, 2, size=n, dtype=np.int8)
    child = np.where(mask, p1, p2).astype(np.int8)
    target = n // 2
    diff = int(child.sum() - target)
    if diff > 0:
        ones = np.flatnonzero(child)
        turn_off = rng.choice(ones, diff, replace=False)
        child[turn_off] = 0
    elif diff < 0:
        zeros = np.flatnonzero(child == 0)
        turn_on = rng.choice(zeros, -diff, replace=False)
        child[turn_on] = 1
    return child


def _mutate_swap(ind: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    ones = np.flatnonzero(ind == 1)
    zeros = np.flatnonzero(ind == 0)
    if len(ones) == 0 or len(zeros) == 0:
        return ind
    i = rng.choice(ones)
    j = rng.choice(zeros)
    ind[i] = 0
    ind[j] = 1
    return ind


def _hill_climb_continuous(
    h: np.ndarray,
    cur_val: float,
    start: float,
    budget: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Swap small mass between two coordinates while preserving sum."""
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    eps = 1e-12
    while time.time() - start < budget - 0.5:
        src = np.where(best_h > eps)[0]
        dst = np.where(best_h < 1.0 - eps)[0]
        if src.size == 0 or dst.size == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _compute_c5(cand)
        if val + 1e-12 < best_val:
            best_h = cand
            best_val = val
    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    rng = np.random.default_rng(seed)

    # ---- Determine problem size ----
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.shape[0]
    else:
        n = 80
        init_h = None
    target_ones = n // 2

    # ---- Build initial binary population ----
    pop_size = 400
    elite = 5
    tournament = 5
    mut_prob = 0.2

    pop = np.empty((pop_size, n), dtype=np.int8)
    for i in range(pop_size):
        idx = rng.choice(n, target_ones, replace=False)
        vec = np.zeros(n, dtype=np.int8)
        vec[idx] = 1
        pop[i] = vec

    if init_h is not None:
        # use ranking of the provided vector to obtain a binary start
        idx = np.argsort(-init_h)[:target_ones]
        bin_start = np.zeros(n, dtype=np.int8)
        bin_start[idx] = 1
        pop[0] = bin_start

    # ---- Evaluate initial fitness ----
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = _compute_c5(pop[i].astype(float))

    best_idx = int(np.argmin(fitness))
    best_h_bin = pop[best_idx].copy()
    best_c5 = float(fitness[best_idx])

    # ---- Genetic algorithm phase ----
    ga_budget = budget_s * 0.85
    while time.time() - start_time < ga_budget - 5.0:
        # elitism
        new_pop = np.empty_like(pop)
        elite_idx = np.argsort(fitness)[:elite]
        new_pop[:elite] = pop[elite_idx]

        # generate offspring
        for i in range(elite, pop_size):
            p1 = _tournament_select(fitness, tournament, rng)
            p2 = _tournament_select(fitness, tournament, rng)
            child = _crossover(pop[p1], pop[p2], rng)
            if rng.random() < mut_prob:
                child = _mutate_swap(child, rng)
            new_pop[i] = child

        # evaluate new generation
        new_fit = np.empty(pop_size, dtype=float)
        for i in range(pop_size):
            new_fit[i] = _compute_c5(new_pop[i].astype(float))

        # update best
        cur_best = int(np.argmin(new_fit))
        if new_fit[cur_best] < best_c5 - 1e-12:
            best_c5 = float(new_fit[cur_best])
            best_h_bin = new_pop[cur_best].copy()

        pop = new_pop
        fitness = new_fit

        if best_c5 < 0.3808:
            break

    # ---- Continuous refinement (mass‑swap hill climbing) ----
    remaining = budget_s - (time.time() - start_time) - 0.5
    best_h = best_h_bin.astype(float)
    if remaining > 0.5:
        best_h, best_c5 = _hill_climb_continuous(best_h, best_c5, start_time, budget_s, rng)

    # ---- Final safety projection ----
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_ones) > 1e-8:
        best_h += (target_ones - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, float(best_c5), n
