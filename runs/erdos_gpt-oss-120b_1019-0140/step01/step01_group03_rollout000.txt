def projected_gradient_descent_multi(h0, max_iters, step0=0.5, time_limit=None):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    for it in range(1, max_iters+1):
        if time_limit is not None and time.time() - start > time_limit:
            break
        # compute correlation raw (not scaled)
        corr = np.correlate(h, 1.0 - h, mode='full')
        # get top K indices
        # Since we need largest values; we can use argpartition(-corr, kth) to get top.
        K = 3
        if corr.size <= K:
            top_idxs = np.arange(corr.size)
        else:
            # note: mode='full' yields central index at n-1 shift=0
            # we need shift value indices: shift = idx - (n-1)
            part = np.argpartition(-corr, K-1)[:K]
            top_idxs = part
        # compute gradient sum
        grad_acc = np.zeros_like(h)
        for idx in top_idxs:
            shift = idx - (n - 1)
            grad_acc += compute_gradient(h, shift)
        grad = grad_acc / len(top_idxs)

        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, s)
        # compute objective
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time, math
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of 1‑D array to a new length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo≤x_i≤hi, Σx_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure brackets contain the root
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the correlation term for a given integer shift."""
    n = h.size
    grad = np.ones_like(h)
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad

def projected_gradient_descent_multi(h0, max_iters, step0=0.5, time_limit=None, top_k=3):
    """PGD using an averaged gradient of the top‑k worst shifts."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    for it in range(1, max_iters + 1):
        if time_limit is not None and time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # indices of the top‑k correlations
        if corr.size <= top_k:
            idxs = np.arange(corr.size)
        else:
            # argpartition for speed
            part = np.argpartition(-corr, top_k - 1)[:top_k]
            idxs = part
        grad_sum = np.zeros_like(h)
        for idx in idxs:
            shift = idx - (n - 1)
            grad_sum += compute_gradient(h, shift)
        grad = grad_sum / len(idxs)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def local_search(h_start, cur_score, time_budget, rng):
    """Small random pairwise transfers to improve the best solution."""
    start = time.time()
    best_h = h_start.copy()
    best_score = cur_score
    n = best_h.size
    target_sum = n / 2.0
    # limit to a reasonable number of iterations if time is abundant
    while time.time() - start < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # allowable transfer from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.5  # up to half of the allowed amount
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # re‑project just in case of floating‑point drift
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new
    return best_h, best_score

def thue_morse_seq(n):
    """Balanced Thue‑Morse sequence (0/1) for powers of two."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target_n = 256
    half_sum = target_n / 2.0

    candidates = []

    # 1. Use parent construction if present
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != target_n:
            h0 = upsample(h0, target_n)
        h0 = project_onto_box_sum(h0, half_sum)
        candidates.append(h0)

    # 2. Known deterministic candidates
    candidates.append(thue_morse_seq(target_n))

    # 3. Random balanced binary
    for _ in range(4):
        h = np.zeros(target_n, dtype=float)
        idx = rng.choice(target_n, size=target_n // 2, replace=False)
        h[idx] = 1.0
        candidates.append(h)

    # 4. Random continuous (projected)
    for _ in range(4):
        v = rng.random(target_n)
        v = project_onto_box_sum(v, half_sum)
        candidates.append(v)

    # 5. Coarse‑to‑fine upsampled candidates
    for coarse_n in (64, 128):
        # binary coarse
        h_coarse = np.zeros(coarse_n, dtype=float)
        idx = rng.choice(coarse_n, size=coarse_n // 2, replace=False)
        h_coarse[idx] = 1.0
        h_up = upsample(h_coarse, target_n)
        h_up = project_onto_box_sum(h_up, half_sum)
        candidates.append(h_up)

        # continuous coarse
        v_coarse = rng.random(coarse_n)
        v_coarse = project_onto_box_sum(v_coarse, coarse_n / 2.0)
        v_up = upsample(v_coarse, target_n)
        v_up = project_onto_box_sum(v_up, half_sum)
        candidates.append(v_up)

    best_h = None
    best_score = float('inf')

    # 6. Optimize each candidate with multi‑shift PGD
    for base_h in candidates:
        if time.time() - start_time > budget_s - 5:
            break
        # allocate iterations (more for candidates of size target_n)
        iter_budget = 8000 if base_h.size == target_n else 4000
        # ensure we don't exceed remaining time
        remaining = budget_s - (time.time() - start_time) - 1
        time_limit = max(0.0, min(remaining, iter_budget * 0.001))
        h_opt, sc_opt = projected_gradient_descent_multi(
            base_h,
            max_iters=iter_budget,
            step0=0.5,
            time_limit=time_limit,
            top_k=3,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # 7. Local random search on the best found solution
    time_elapsed = time.time() - start_time
    remaining_time = budget_s - time_elapsed
    if best_h is not None and remaining_time > 2.0:
        # allocate up to 20% of the remaining budget to local search
        local_budget = min(remaining_time * 0.2, remaining_time - 0.5)
        best_h, best_score = local_search(best_h, best_score, local_budget, rng)

    # 8. Final feasibility projection and recomputation
    if best_h is None:
        best_h = np.full(target_n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, half_sum)
    else:
        best_h = project_onto_box_sum(best_h, half_sum)

    best_score = compute_c5(best_h)

    return best_h, float(best_score), target_n
