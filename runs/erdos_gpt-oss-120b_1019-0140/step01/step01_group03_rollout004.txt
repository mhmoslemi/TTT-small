def adam_optimize(h0, max_iter=5000, step0=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter+1):
        # find active shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        # Adam update
        m = beta1*m + (1 - beta1)*grad
        v = beta2*v + (1 - beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 / (np.sqrt(v_hat) + eps)
        h = h - step * m_hat
        # project
        h = project_onto_box_sum(h, s)
        # evaluate
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def compute_gradient(h, shift):
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # contribution from first factor: for i in 0..n-s-1: + (1 - h[i+s])

import time
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Exact gradient of the active correlation term."""
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            grad[:n - s] += 1.0 - h[s:]
            grad[s:] -= h[:n - s]
    else:
        s = -shift
        if s < n:
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] -= h[s:]
    return grad

def adam_optimize(h0, max_iter=5000, step0=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam-like projected optimization on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step0 / (np.sqrt(v_hat) + eps)
        h = h - step * m_hat
        h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def thue_morse_seq(N):
    """Binary Thue–Morse sequence of length N (0/1 values)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(N)),
                       dtype=float, count=N)

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 256  # target resolution
    best_h = None
    best_score = np.inf

    # Assemble initial candidate pool
    candidates = []

    # 1. Parent construction if provided

    candidates.append(thue_morse_seq(n))

    # 3. A few random binary patterns

    # 4. Random continuous starts (projected)
    for _ in range(3):

    # 5. Mixed smoothing of binary bases (alpha in [0.5,0.75,1])
    alphas = (0.5, 0.75, 1.0)
    for base in candidates[:4]:  # use first few binary seeds
        for a in alphas:
            mixed = a * base + (1.0 - a) * 0.5
            mixed = project_onto_box_sum(mixed, n / 2.0)

    # Optimize each candidate with a relatively long run
    for h0 in candidates:
