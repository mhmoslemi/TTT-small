def simulated_annealing(h0, c5_0, start_time, budget_s):
    n = len(h0)
    ones = np.flatnonzero(h0 == 1)
    zeros = np.flatnonzero(h0 == 0)
    best_h = h0.copy()
    best_val = c5_0
    cur_h = h0.copy()
    cur_val = c5_0

    # parameters
    max_iter = 2000000
    # temperature schedule
    T0 = 0.01  # maybe set based on bound magnitude?
    T_end = 1e-5
    # each iteration compute temperature as linear decay
    while time.time() - start_time < budget_s - 0.5:
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # recompute
        new_val = eval_h(cur_h)
        # acceptance
        if new_val < cur_val:
            accept = True
        else:
            # temperature based on elapsed fraction
            elapsed = time.time() - start_time
            frac = (elapsed) / (budget_s - 0.5)
            T = T0 * (1 - frac) + T_end * frac
            accept = rng.random() < math.exp((cur_val - new_val) / T)
        if accept:
            # update lists for ones and zeros
            ones_idx = i; zeros_idx = j
            # after swap, h[i] becomes 0 and h[j] becomes 1
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            # ones/zeros unchanged
            # continue

def run(seed=42, budget_s=1100, **kwargs):
    start_time= time.time()
    rng = np.random.default_rng(seed)
    # set n
    if 'initial_h_values' in globals():
        h_cont = np.array(initial_h_values, dtype=float)
        n = len(h_cont)
        # convert to binary via top half
        order = np.argsort(-h_cont)[:n//2]
        h0 = np.zeros(n, dtype=int)
        h0[order] = 1
    else:
        n = 80
        h0 = np.zeros(n, dtype=int)
        idx = rng.choice(n, n//2, replace=False)
        h0[idx] = 1
    best_h = h0.copy()
    best_val = compute_c5(best_h.astype(float))

    # GA parameters
    pop_size = 30
    elite_frac = 0.2
    mutation_rate = 0.2 # probability of mutation per child
    # Generate initial population
    population = [best_h.copy()]
    # fill rest with random individuals
    while len(population) < pop_size:
        ind = np.zeros(n, dtype=int)
        ind[rng.choice(n, n//2, replace=False)] = 1
        population.append(ind)

    # function to evaluate val for integer vector
    def evaluate(pop):
        vals = np.empty(len(pop))
        for i, ind in enumerate(pop):
            vals[i] = compute_c5(ind.astype(float))
        return vals

    # main loop
    while time.time() - start_time < budget_s - 2.0:
        # evaluate
        fitness = evaluate(population)
        # update best
        idx_best = np.argmin(fitness)
        if fitness[idx_best] < best_val - 1e-12:
            best_val = fitness[idx_best]
            best_h = population[idx_best].copy()
        # elite selection
        elite_count = max(1, int(elite_frac * pop_size))
        elite_idx = np.argsort(fitness)[:elite_count]
        elite = [population[i] for i in elite_idx]

        # generate new population
        new_pop = elite.copy()
        # produce until pop_size
        while len(new_pop) < pop_size:
            # tournament selection
            def tournament():
                cand_idx = rng.choice(pop_size, 5, replace=False)
                cand_fits = fitness[cand_idx]
                return population[cand_idx[np.argmin(cand_fits)]]
            parent1 = tournament()
            parent2 = tournament()
            # crossover
            # We'll take uniform crossover and then repair to half ones.
            mask = rng.random(n) < 0.5
            child = np.where(mask, parent1, parent2).astype(int)
            # repair to exactly n//2 ones
            ones = int(child.sum())
            target = n // 2
            if ones > target:
                # too many ones, turn some ones to zero
                ones_idx = np.flatnonzero(child == 1)
                to_flip = rng.choice(ones_idx, ones - target, replace=False)
                child[to_flip] = 0
            elif ones < target:
                zeros_idx = np.flatnonzero(child == 0)
                to_flip = rng.choice(zeros_idx, target - ones, replace=False)
                child[to_flip] = 1
            # mutation

                for _ in range(rng.integers(1, 4)):

                    zeros_idx = np.flatnonzero(child == 0)
                    if len(ones_idx) == 0 or len(zeros_idx) == 0:

                    i = rng.choice(ones_idx)
                    j = rng.choice(zeros_idx)

            final_h[ones[: -diff]] = 0.0

def local_anneal(h_bin, cur_val, start_time, budget_s):
    n = len(h_bin)
    rng = np.random.default_rng()
    cur_h = h_bin.copy()
    cur_val = cur_val
    best_h = cur_h.copy()
    best_val = cur_val

    # cooling schedule based on fraction of time left
    while time.time() - start_time < budget_s - 1.0:
        # choose a random swap
        ones_idx = np.flatnonzero(cur_h == 1)
        zeros_idx = np.flatnonzero(cur_h == 0)
        if len(ones_idx)==0 or len(zeros_idx)==0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_val = compute_c5(cur_h.astype(float))
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # compute acceptance probability
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            # simple temperature decreasing with time:
            # T = max( (remaining / budget_s) * 0.01, 1e-6)
            T = max(remaining / budget_s * 0.01, 1e-8)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
                # accept even if worse
            else:
                # revert swap
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # maybe early exit if best_val is already below target threshold like 0.3805
        if best_val < 0.3805:
            break
    # return best found
    return best_h, best_val

def random_start_vector(n, rng):
    # generate random uniform, then adjust.
    v = rng.random(n)
    # enforce sum target
    target = n / 2.0
    v = v - v.mean() + target / n
    v = np.clip(v, 0.0, 1.0)
    # now adjust to exact sum by distributing residual
    cur_sum = v.sum()
    diff = target - cur_sum
    if abs(diff) > 1e-12:
        # add diff equally to all entries if possible
        inc = diff / n
        v = np.clip(v + inc, 0.0, 1.0)
        cur_sum = v.sum()
        diff = target - cur_sum
        # Now adjust by adding/subtracting small amounts to random indices
        indices = np.arange(n)
        rng.shuffle(indices)
        for idx in indices:
            if abs(diff) < 1e-12:
                break
            # max we can adjust at idx without violating bounds
            if diff > 0:
                max_inc = 1.0 - v[idx]
                delta = min(diff, max_inc)
                v[idx] += delta
                diff -= delta
            else:
                max_dec = v[idx]
                delta = min(-diff, max_dec)
                v[idx] -= delta
                diff += delta
    return v
```

def continuous_anneal(h_start, best_val, start_time, deadline):
    h = h_start.copy()
    cur_val = best_val
    best_h = h.copy()
    best_val = cur_val
    while time.time() < deadline - 0.1:
        # select i where h_i > 1e-12, j where h_j < 1 - 1e-12
        i_candidates = np.where(h > 1e-12)[0]
        j_candidates = np.where(h < 1 - 1e-12)[0]
        if len(i_candidates)==0 or len(j_candidates)==0:
            break
        i = rng.choice(i_candidates)
        j = rng.choice(j_candidates)
        # max transfer amount is min(h_i, 1 - h_j)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer  # random fraction
        h[i] -= delta
        h[j] += delta
        # compute new C5
        new_val = compute_c5(h)
        # acceptance
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # maybe annealing acceptance
            # compute temperature.
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = max(remaining * 0.001, 1e-8)  # scaling?
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert transfer
                h[i] += delta
                h[j] -= delta
        # early exit
        if best_val < 0.3805:
            break
    return best_h, best_val
```

def continuous_anneal(best_h, best_val, start_time, deadline, rng):
    h = best_h.copy()
    cur_val = best_val
    n = len(h)
    target = n / 2.0
    while time.time() < deadline - 0.1:
        # compute max shift and contributions
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        k = max_idx - (n - 1)
        # contributions vector
        if k >= 0:
            prod = h[:n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[:n - kk])
            i_range = np.arange(kk, n)
        # pick index with largest product
        i_local = np.argmax(prod)
        i = i_range[i_local]  # original index in h
        # choose j random such that can receive mass
        zeros = np.where(h < 1.0 - 1e-12)[0]
        # Ensure j != i and h[i]>0
        if len(zeros) == 0 or h[i] <= 1e-12:
            # fallback random swap
            i_candidates = np.where(h > 1e-12)[0]
            j_candidates = np.where(h < 1.0 - 1e-12)[0]
            if len(i_candidates)==0 or len(j_candidates)==0:
                break
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        else:
            j = rng.choice(zeros)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        # choose delta fraction
        delta = rng.random() * max_transfer
        # apply
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        # acceptance
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # accept with probability based on temperature
            elapsed = time.time() - start_time
            remaining = deadline - elapsed
            T = max(remaining * 0.001, 1e-9)  # scale
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert
                h[i] += delta
                h[j] -= delta
        if best_val < 0.3808:
            break
    return best_h, best_val

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

                    diff -= delta

                    diff += delta
