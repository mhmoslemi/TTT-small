import numpy as np
import time

def _c5_value(h: np.ndarray) -> float:
    """Compute the C5 value for a discrete step function h."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce the exact integral constraint."""
    np.clip(v, 0.0, 1.0, out=v)
    N = v.size
    target = N / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / N
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    # Choose an even number of points to make the cosine sum zero exactly.
    n = 200  # < 1k, fast and sufficiently accurate
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.5 * np.cos(np.pi * x)  # cosine with amplitude 0.5, period 2
    h = _repair_vector(h)
    c5 = _c5_value(h)
    # Ensure we finish well before the budget limit.
    _ = time.time() - start  # (unused, just to consume the timer variable)
    return h, float(c5), n
