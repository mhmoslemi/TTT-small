def _compute_c5_fft(h):
    n = h.shape[0]
    L = 2*n
    a = np.fft.rfft(h, n=L) # rfft returns real FFT half length
    b = np.fft.rfft(1 - h, n=L)
    # correlation: ifft of a * conj(b) => use irfft
    corr = np.fft.irfft(a * np.conj(b), n=L)
    # Take first 2*n-1 entries
    corr = corr[:2*n-1]
    dx = 2.0 / n
    return float(np.max(corr) * dx)

def _compute_c5_fft(h):
    n = h.shape[0]
    # Use convolution via FFT for speed
    L = 2 * n
    # compute correlation via FFT (real)
    a = np.fft.rfft(h, n=L)
    b = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(a * np.conj(b), n=L)
    corr = corr[:2 * n - 1]
    dx = 2.0 / n
    return float(np.max(corr) * dx)

def _compute_c5_fft(h: np.ndarray) -> float:
    """Maximum overlap integral C5 using FFT-based correlation."""
    n = h.shape[0]
    L = 2 * n
    # correlation h * (1-h) (no reversal) via FFT
    a = np.fft.rfft(h, n=L)
    b = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(a * np.conj(b), n=L)
    corr = corr[: 2 * n - 1]
    dx = 2.0 / n
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:

    idx = np.where((h + diff >= 0.0) & (h + diff <= 1.0))[0]

        h[idx[0]] += diff
