def _repair_vector(v):
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    # iterative projection onto hyperplane sum=v_target
    for _ in range(10):
        s = v.sum()
        if abs(s - target) <= 1e-12:
            break
        # compute adjustment delta to distribute equally among non-bound variables
        # if s is zero, just set uniform.
        if s == 0:
            v[:] = target / v.size
            np.clip(v, 0.0, 1.0, out=v)
            continue
        # shift
        factor = target / s
        v *= factor
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_gradient(h):
    N = h.size
    # compute correlation to find shift index of max.
    conv = signal.fftconvolve(h, (1 - h)[::-1], mode='full')
    idx = int(np.argmax(conv))
    shift = idx - (N - 1)  # shift offset (positive = shift right of second argument)
    # Compute gradient g of max correlation (without dx)
    g = np.zeros(N, dtype=np.float64)
    if shift >= 0:
        # i from 0 to N-shift-1
        g[:N-shift] += 1 - h[shift:]
        g[shift:] -= h[:N-shift]
    else:
        t = -shift
        # i from t to N-1
        g[t:] += 1 - h[:N - t]
        g[:N - t] -= h[t:]
    dx = 2.0 / N
    g *= dx  # gradient for C5
    return g

   best_h = h.copy()
   best = cur
   N = h.size
   start = time.time()
   # initial temperature maybe relative to cur magnitude or set manually.
   T0 = 0.1  # or dynamic.
   max_eps = 0.5
   while time.time() - start < time_limit:
        # Compute temperature linearly decreasing
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        # propose move
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi = h[i]; hj = h[j]
        up = min(1.0 - hi, hj)   # can increase i up to 1-hi, decrease j up to hj
        down = min(hi, 1.0 - hj) # can decrease i down to hi, increase j up to 1-hj
        if up == 0 and down == 0:
            continue
        eps = rng.random() * (up + down) - down
        # clip eps size
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        # apply
        h_new = h.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # Should remain within bounds due to eps constraints, but clip anyway.
        if not (0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0):

        # evaluate
        new = _c5_value(h_new)
        delta = new - cur
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h = h_new
            cur = new

def _c5_gradient(h):
    N = h.shape[0]
    corr = fftconvolve(h, (1 - h)[::-1], mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)
    g = np.zeros(N, dtype=np.float64)
    if shift >= 0:
        g[:N - shift] += 1 - h[shift:]
        g[shift:] -= h[:N - shift]
    else:
        t = -shift
        g[t:] += 1 - h[:N - t]
        g[:N - t] -= h[t:]
    return g

def _simulated_annealing_one_run(h_start, time_limit, rng):
    h = h_start.copy()
    cur = _c5_value(h)
    best_h = h.copy()
    best = cur
    N = h.shape[0]
    start = time.time()
    T0 = max(0.05, cur * 0.1)  # maybe adapt
    max_eps = 0.5
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h[i], h[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0 and down == 0:
            continue
        eps = rng.random() * (up + down) - down
        # clamp eps magnitude
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # ensure within bounds (should be by construction)
        if not (0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0):
            continue
        new = _c5_value(h_new)
        delta = new - cur
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h = h_new
            cur = new
            if new < best:
                best = new
                best_h = h_new.copy()
    return best_h, best

def _gradient_descent(h_start, time_limit, rng):
    h = h_start.copy()
    cur = _c5_value(h)
    best_h = h.copy()
    best = cur
    start = time.time()
    step = 0.5  # step magnitude for raw gradient (max change per iteration)
    N = h.shape[0]
    while time.time() - start < time_limit:
        grad = _c5_gradient(h)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            # random jitter
            perturb = rng.uniform(-0.05, 0.05, size=N)
            h_candidate = _repair_vector(h + perturb)
            cand = _c5_value(h_candidate)
            if cand < cur:
                h = h_candidate
                cur = cand
                if cand < best:
                    best = cand
                    best_h = h.copy()
            continue
        improved = False
        # try different step scaling factors
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step_sz = step * factor
            h_candidate = h - step_sz * grad
            h_candidate = _repair_vector(h_candidate)
            cand = _c5_value(h_candidate)
            if cand < cur:
                h = h_candidate
                cur = cand
                step = step_sz * 1.2  # increase step a bit on success

                if cand < best:
                    best = cand
                    best_h = h.copy()
                break
        if not improved:
            step *= 0.5
            if step < 1e-6:
                step = 1e-6
    return best_h, best

        if s == 0.0:
            v[:] = target / v.size
            np.clip(v, 0.0, 1.0, out=v)
            continue
        factor = target / s
        v *= factor
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h):
    """Maximum overlap integral for discretised step function."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = fftconvolve(h, (1.0 - h)[::-1], mode='full')
    return float(np.max(corr) * dx)

def _c5_gradient(h):
    """Sub‑gradient of C5 (without the dx factor)."""
    N = h.shape[0]
    corr = fftconvolve(h, (1.0 - h)[::-1], mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)   # signed shift
    g = np.zeros(N, dtype=np.float64)
    if shift >= 0:
        g[:N - shift] += 1.0 - h[shift:]
        g[shift:] -= h[:N - shift]
    else:
        t = -shift
        g[t:] += 1.0 - h[:N - t]
        g[:N - t] -= h[t:]
    return g

def _interpolate(h, new_n):
    N = h.shape[0]
    if N == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, N, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def _simulated_annealing_one_run(h_start, time_limit, rng):
    """Simple SA that preserves sum and box constraints."""
    h = h_start.copy()
    cur = _c5_value(h)
    best_h = h.copy()
    best = cur
    N = h.shape[0]
    start = time.time()
    T0 = max(0.05, cur * 0.1)   # initial temperature scaled to score
    max_eps = 0.5
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h[i], h[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # bounds already respected
        new = _c5_value(h_new)
        delta = new - cur
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h = h_new
            cur = new
            if new < best:
                best = new
                best_h = h_new.copy()
    return best_h, best

def _gradient_descent(h_start, time_limit, rng):
    """Projected gradient descent on C5 using the sub‑gradient."""
    h = h_start.copy()
    cur = _c5_value(h)
    best_h = h.copy()
    best = cur
    start = time.time()
    step = 0.5  # initial step size (applies to raw gradient)
    N = h.shape[0]
    while time.time() - start < time_limit:
        grad = _c5_gradient(h)
        if np.linalg.norm(grad) < 1e-12:
            # small random perturbation to escape flat region
            pert = rng.uniform(-0.05, 0.05, size=N)
            cand = _repair_vector(h + pert)
            cand_score = _c5_value(cand)
            if cand_score < cur:
                h, cur = cand, cand_score
                if cand_score < best:
                    best, best_h = cand_score, cand.copy()
            continue
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step_sz = step * factor
            h_cand = h - step_sz * grad
            h_cand = _repair_vector(h_cand)
            cand_score = _c5_value(h_cand)
            if cand_score < cur:
                h, cur = h_cand, cand_score
                step = step_sz * 1.2
                improved = True
                if cand_score < best:
                    best, best_h = cand_score, h_cand.copy()
                break
        if not improved:
            step *= 0.5
            if step < 1e-6:
                step = 1e-6
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Initial construction
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 200
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)], dtype=np.float64)
    h0 = _repair_vector(h0)
    best_h = h0.copy()
    best_score = _c5_value(best_h)
    total_start = time.time()
    max_n = 800
    # Build a list of increasing resolutions
    levels = [best_h.size]
    while levels[-1] * 2 <= max_n:
        levels.append(levels[-1] * 2)
    for idx, n in enumerate(levels):
        # Upscale if needed
        if best_h.size != n:
            best_h = _interpolate(best_h, n)
            best_h = _repair_vector(best_h)
            best_score = _c5_value(best_h)
        elapsed = time.time() - total_start
        remaining = budget_s - elapsed
        if remaining <= 0:
            break
        levels_left = len(levels) - idx
        level_time = remaining / levels_left
        # allocate time slices
        sa_time = level_time * 0.4
        gd_time = level_time * 0.6
        # Simulated annealing
        h_sa, s_sa = _simulated_annealing_one_run(best_h, sa_time, rng)
        if s_sa < best_score:
            best_h, best_score = h_sa.copy(), s_sa
        # Gradient descent
        h_gd, s_gd = _gradient_descent(best_h, gd_time, rng)
        if s_gd < best_score:
            best_h, best_score = h_gd.copy(), s_gd
    # Final repair and evaluation
    best_h = _repair_vector(best_h)
    final_score = _c5_value(best_h)
    return best_h, float(final_score), best_h.size

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Project v onto [0,1] box and enforce sum(v) = len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
