def local_search(h0, time_limit, best_c5):
    h = h0.copy()
    n = len(h)
    cur_c5 = compute_c5(h)
    start = time.time()
    # Annealing schedule:
    # temperature decays from T0 to T_end
    T0 = 1.0
    T_end = 1e-4
    while time.time() - start < time_limit:
        # compute fraction of elapsed relative to budget for schedule
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1-frac) + T_end * frac
        # Choose two distinct indices i,j
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # Determine maximum transfer amount
        max_transfer = min(h[i], 1 - h[j])
        if max_transfer <= 0:
            continue
        # Choose epsilon = max_transfer * rng.random() * step_scale
        eps = max_transfer * rng.random() * step_scale
        if eps <= 0:
            continue
        # apply move
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # compute c5
        cand_c5 = compute_c5(h_new)
        # accept if better; else accept with prob exp(-(cand_cur - cur)/T) ?
        # Actually we try to minimize c5, so delta = cand - cur. Accept if delta <0, else maybe accept with probability exp(-delta/T)

                best_h = h.copy()
                best_c5 = cur_c5
        # Optional: adaptive step_scale gradually decreases
        step_scale = max(step_scale * 0.9999, 0.01)   # optional

    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    # schedule parameters
    initial_T = 0.01  # because differences in c5 small (0.001?)?
    final_T = 1e-6
    # step factor reduces linearly
    initial_step = 0.5
    final_step = 0.01
    # We'll compute elapsed fraction each iteration
    it = 0
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= total_time:
            break
        fraction = elapsed / total_time
        T = initial_T * (1 - fraction) + final_T * fraction
        step_factor = initial_step * (1 - fraction) + final_step * fraction
        i, j = rng.integers(0, n, size=2)

            continue
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0.0:

def anneal_optimize(start_h, time_budget, rng):
    h = start_h.copy()
    n = len(h)
    cur_c5 = compute_c5(h)
    best_h = h.copy(); best_c5 = cur_c5
    start = time.time()
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        fraction = elapsed / time_budget
        # Temperature schedule
        T = 1e-3 * (1 - fraction) + 1e-6 * fraction
        # Step size factor
        step_factor = 0.5 * (1 - fraction) + 0.01 * fraction

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # compute max transfer based on boundedness:
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        eps = max_transfer * rng.random() * step_factor
        if eps <= 0:
            continue
        # apply move
        h[i] -= eps
        h[j] += eps
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            # use probability
            prob = np.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
        it += 1
        # periodically for safety: every 1000 iters, maybe check time again explicitly or break if not improving (optional)
    return best_h, best_c5

def get_targeted_move_corr(corr, h, rng):

    max_idx = int(np.argmax(corr))
    L = max_idx - (n - 1)
    # overlapping indices range
    i_start = max(0, -L)
    i_end = min(n, n - L)
    if i_start >= i_end:
        return None, None, None
    idxs = np.arange(i_start, i_end)
    partner = idxs + L
    # compute contributions
    contrib = h[idxs] * (1.0 - h[partner])
    # source candidates: high contrib, with h>0
    src_mask = (h[idxs] > 1e-12) & ((1.0 - h[partner]) > 1e-12)
    if not np.any(src_mask):
        return None, None, None
    src_idx = idxs[src_mask]
    src_contrib = contrib[src_mask]
    # choose source i as one of top contributions: pick random among top ones
    # Determine threshold for top 20%
    if src_idx.size > 5:
        top_k = max(1, src_idx.size // 5)
        # get indices of top contributions
        top_order = np.argsort(src_contrib)[-top_k:]
        i = src_idx[top_order[rng.integers(top_k)]]
    else:
        i = src_idx[rng.integers(src_idx.size)]

    # target candidates: low contribution, and h < 1
    tgt_mask = (h[idxs] < 1.0 - 1e-12)
    if not np.any(tgt_mask):
        return None, None, None
    tgt_idx = idxs[tgt_mask]
    tgt_contrib = contrib[tgt_mask]
    # Choose target with low contribution: pick among lowest 20%
    if tgt_idx.size > 5:
        low_k = max(1, tgt_idx.size // 5)
        low_order = np.argsort(tgt_contrib)[:low_k]
        # ensure not equal to i
        # remove i if present
        filtered = tgt_idx[low_order][tgt_idx[low_order] != i]
        if filtered.size == 0:
            # fallback to any other target
            filtered = np.setdiff1d(tgt_idx, np.array([i]), assume_unique=True)
            if filtered.size == 0:
                return None, None, None
        j = filtered[rng.integers(filtered.size)]
    else:
        # avoid i if same
        possible = tgt_idx[tgt_idx != i]
        if possible.size == 0:
            return None, None, None
        j = possible[rng.integers(possible.size)]

    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        return None, None, None
    # choose epsilon as fraction
    eps = max_transfer * rng.random() * 0.5
    if eps <= 0.0:
        return None, None, None

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    """Clip to [0,1] and adjust to satisfy the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final fix
    cur = h.sum()
    if abs(cur - target_sum) > 1e-8:
        i = np.argmin(h)  # position with most room to increase
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h


def _targeted_move(corr, h, rng):
    """Pick a source and target index based on the current worst lag."""
    n = h.shape[0]
    max_idx = int(np.argmax(corr))
    L = max_idx - (n - 1)                     # lag (negative allowed)
    i_start = max(0, -L)
    i_end = min(n, n - L)
    if i_start >= i_end:
        return None, None, None
    idxs = np.arange(i_start, i_end)
    partner = idxs + L
    contrib = h[idxs] * (1.0 - h[partner])

    # source: large contribution, h > 0
    src_mask = (h[idxs] > 1e-12) & ((1.0 - h[partner]) > 1e-12)
    if not np.any(src_mask):
        return None, None, None
    src_idx = idxs[src_mask]
    src_contrib = contrib[src_mask]
    top_k = max(1, src_idx.size // 5)
    top_order = np.argsort(src_contrib)[-top_k:]
    i = src_idx[top_order[rng.integers(top_k)]]

    # target: low contribution, h < 1
    tgt_mask = h[idxs] < 1.0 - 1e-12
    if not np.any(tgt_mask):
        return None, None, None
    tgt_idx = idxs[tgt_mask]
    tgt_contrib = contrib[tgt_mask]
    low_k = max(1, tgt_idx.size // 5)
    low_order = np.argsort(tgt_contrib)[:low_k]
    cand = tgt_idx[low_order]
    cand = cand[cand != i] if cand.size > 1 else cand
    if cand.size == 0:
        return None, None, None
    j = cand[rng.integers(cand.size)]

    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        return None, None, None
    eps = max_transfer * rng.random() * 0.5
    if eps <= 0.0:
        return None, None, None
    return int(i), int(j), float(eps)


def anneal_optimize(start_h, time_budget, rng):
    """Simulated‑annealing style local search using pairwise mass transfers."""
    h = start_h.copy()
    n = h.shape[0]
    dx = 2.0 / n
    # initial C5
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        fraction = elapsed / time_budget
        T = 0.01 * (1.0 - fraction) + 1e-5 * fraction
        step_factor = 0.5 * (1.0 - fraction) + 0.01 * fraction
        p_target = 0.2 * (1.0 - fraction) + 0.05 * fraction

        if rng.random() < p_target:
            corr = np.correlate(h, 1.0 - h, mode="full")
            cur_c5 = np.max(corr) * dx
            i, j, eps = _targeted_move(corr, h, rng)
            if i is None:
                continue
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0.0:
                continue
            eps = max_transfer * rng.random() * step_factor

        # apply move
        h[i] -= eps
        h[j] += eps

        # evaluate new C5
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_c5 = np.max(new_corr) * dx

        delta = new_c5 - cur_c5
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True

        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps

        it += 1
        if it % 5000 == 0:
            # re‑project to keep the sum exact
            target = n / 2.0
            if abs(h.sum() - target) > 1e-8:
                h = project_to_feasible(h, target)
                corr = np.correlate(h, 1.0 - h, mode="full")
                cur_c5 = np.max(corr) * dx

    return best_h, best_c5


def slsqp_objective(x):
    return compute_c5(x)


def slsqp_eq_constraint(x):
    return np.sum(x) - x.shape[0] / 2.0


def refine_with_slsqp(h_initial, max_iter=400):
    """One SLSQP refinement pass (box bounds + equality)."""
    n = h_initial.shape[0]
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": slsqp_eq_constraint}
    try:
        res = opt.minimize(
            fun=slsqp_objective,
            x0=h_initial,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
    except Exception:
        return h_initial, compute_c5(h_initial)
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the given time budget."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # start from provided construction if available
    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)
        n = base_h.shape[0]
        base_h = project_to_feasible(base_h, n / 2.0)
    else:
        n = 380
        base_h = rng.random(n)
        base_h = project_to_feasible(base_h, n / 2.0)

    best_h = None
    best_c5 = float("inf")
    max_restarts = 25
    restart = 0

    while time.time() - start_time < budget_s * 0.97 and restart < max_restarts:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5.0:
            break
        # allocate up to 30 s per restart
        run_time = min(remaining * 0.5, 30.0)

        if restart == 0:
            h0 = base_h.copy()
        elif best_h is not None and rng.random() < 0.7:
            h0 = best_h.copy()
            noise = rng.normal(scale=0.02, size=n)
            h0 = np.clip(h0 + noise, 0.0, 1.0)
            h0 = project_to_feasible(h0, n / 2.0)
        else:
            h0 = rng.random(n)
            h0 = project_to_feasible(h0, n / 2.0)

        cand_h, cand_c5 = anneal_optimize(h0, run_time, rng)

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h.copy()

        restart += 1

    if best_h is None:
        best_h = base_h.copy()
        best_c5 = compute_c5(best_h)

    # final refinement with SLSQP if time permits
    time_left = budget_s - (time.time() - start_time)
    if time_left > 8.0:
        refined_h, refined_c5 = refine_with_slsqp(best_h, max_iter=600)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # final safety projection
    best_h = project_to_feasible(best_h, best_h.shape[0] / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_h.shape[0]
