def subgradient_descent(h0, max_iters, step0, momentum):
    v = np.zeros_like(h0)
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters+1):
        grad, cur_val = compute_grad_multi(h, eps=1e-12)
        eta = step0 / np.sqrt(it)
        v = momentum * v - eta * grad
        h_new = h + v
        h_new = project(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def binary_hill_climb(h, max_swaps, rng):
    n = h.size
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    best_h = h.copy()
    best_score = compute_c5(h)
    for _ in range(max_swaps):

            # update ones/zeros

def compute_c5(h):

def subgradient_descent(h0, max_iters, step0, momentum, start_time, budget_s):
    h = h0.copy()
    n = h.shape[0]
    sum_target = n / 2.0
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters+1):
        if time.time() - start_time > budget_s - 0.5:
            break
        # correlation and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        thresh = max_val - 1e-12
        idxs = np.where(corr >= thresh)[0]
        shifts = idxs - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        for s in shifts:
            # compute gradient for shift s
            if s >= 0:
                grad[:n - s] -= h[s:]
                grad[s:] -= h[:n - s]
            else:
                s2 = -s
                grad[s2:] -= h[:n - s2]
                grad[:n - s2] -= h[s2:]
        grad += 1.0
        grad /= len(shifts)

        eta = step0 / np.sqrt(it)
        v = momentum * v - eta * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, sum_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

import time
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = int(arr.shape[0])
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) + 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def thue_morse(n):
    """Return Thue–Morse sequence of length n (0/1 as float)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def to_binary_balanced(h):
    """Return a binary vector with the same size as h and exactly half ones."""
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def gradient_multi(h, shifts):
    """Average subgradient of the max‑correlation term for the given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] -= h[s:]
            grad[s:]    -= h[:n - s]
        else:
            s2 = -s
            grad[s2:]   -= h[:n - s2]
            grad[:n - s2] -= h[s2:]
    grad += 1.0
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def subgradient_descent(h0, max_iters, step0, momentum, start_time, budget_s):
    """Projected subgradient descent with momentum."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    dx = 2.0 / n
    v = np.zeros_like(h)
    # initial best
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iters + 1):
        # time guard
        if time.time() - start_time > budget_s - 0.5:
            break

        # correlation and active shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        # tolerance to capture ties
        thresh = max_val - 1e-12
        active_idxs = np.where(corr >= thresh)[0]
        shifts = active_idxs - (n - 1)

        # subgradient
        grad = gradient_multi(h, shifts)

        # step size schedule
        eta = step0 / np.sqrt(it)
        v = momentum * v - eta * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, target_sum)

        # evaluate new point
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        score = corr_new.max() * dx

        if score < best_score:
            best_score = score
            best_h = h_new.copy()

        h = h_new

    return best_h, best_score

def binary_hill_climb(h, max_swaps, rng, start_time, budget_s):
    """Local search on binary vectors by swapping a 1 with a 0."""
    n = h.size
    best = h.copy()
    best_score = compute_c5(best)

    # maintain index lists for fast sampling
    ones = np.flatnonzero(best == 1)
    zeros = np.flatnonzero(best == 0)

    for _ in range(max_swaps):
        if time.time() - start_time > budget_s - 0.5:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        score = compute_c5(cand)
        if score < best_score:
            best_score = score
            best = cand
            ones = np.flatnonzero(best == 1)
            zeros = np.flatnonzero(best == 0)
    return best, best_score
