def simulated_annealing(start_vec, start_f, start_time, budget_s, rng):
    cur = start_vec.copy()
    cur_f = start_f
    best = cur.copy()
    best_f = cur_f
    # define schedule: total steps = a large number, or based on time.
    # We can compute total_iters = int(1e6) or limited by time.
    # Use linear cooling: T = T0 * (1 - it / max_it)
    T0 = 0.001
    # compute max_it based on budget left: approximate 2e5 iterations maybe.
    # For simpler, just compute T = T0 * max(0.0, (budget - elapsed) / total_budget) etc.
    # We'll just use a simple exponential cooling: T = T0 * (0.99 ** it)
    it = 0
    while time.time() - start_time < budget_s - 0.1:
        # propose swap
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        cand = cur.copy()
        cand[i1] = 0
        cand[i0] = 1
        cand_f = compute_c5(cand.astype(float))
        # acceptance
        delta = cand_f - cur_f
        T = T0 * (0.995 ** it)  # cooling
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = cand
            cur_f = cand_f
            if cur_f < best_f:
                best = cur.copy()
                best_f = cur_f
        it += 1
    return best, best_f

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap C5 for step vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def hill_climb_binary(start_vec, start_f, start_time, budget_s, rng):
    """Simple strict-improvement hill climbing on binary vectors."""
    best = start_vec.copy()
    best_f = start_f
    while time.time() - start_time < budget_s - 0.1:
        # pick random 1 and 0 positions
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        cand = best.copy()
        cand[i1] = 0
        cand[i0] = 1
        cand_f = compute_c5(cand.astype(float))
        if cand_f + 1e-12 < best_f:
            best = cand
            best_f = cand_f
    return best, best_f

def simulated_annealing(start_vec, start_f, start_time, budget_s, rng):
    """Simulated annealing on binary vectors preserving Hamming weight."""
    cur = start_vec.copy()
    cur_f = start_f
    best = cur.copy()
    best_f = cur_f
    it = 0
    T0 = 0.001
    # cooling factor per iteration
    cooling = 0.9995
    while time.time() - start_time < budget_s - 0.1:
        # propose a swap
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)

                best_f = cur_f
