def compute_c5(h):

def hill_climb_binary(h_init, budget, start_time):
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)
    n = h_cur.shape[0]
    # keep indices of ones and zeros
    # Can't recompute each iteration; but update after swap.
    ones_idx = np.flatnonzero(h_cur > 0.5)
    zeros_idx = np.flatnonzero(h_cur < 0.5)
    while True:
        if time.time() - start_time > budget:
            break
        # propose swap
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h_new = h_cur.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c5_new = compute_c5(h_new)
        # Accept if improves
        if c5_new < best_c5 - 1e-12: # tolerance
            best_c5 = c5_new
            best_h = h_new
            # update current to new (optional)
            h_cur = h_new
            # update indices
            # Since swapped: i becomes zero, j becomes one
            # Update arrays: remove i from ones_idx, add to zeros_idx; vice versa.
            # Simplify: recompute indices after each accepted swap.
            ones_idx = np.flatnonzero(h_cur > 0.5)
            zeros_idx = np.flatnonzero(h_cur < 0.5)
        else:
            # maybe sometimes accept with probability (annealing)
            # but for simplicity we keep greedy.
            continue
    return best_h, best_c5

def refine_continuous(h_start, n, start_time, time_budget):

    from scipy.optimize import minimize

    bnds = [(0.0, 1.0)] * n

    # We'll need a wrapper for objective: compute_c5
    def obj(x):
        return compute_c5(x)

    # options: maxiter limited by remaining time perhaps 200
    remaining = time_budget - (time.time() - start_time)
    # SLSQP may take unknown time. We'll set maxiter = min(400, int(remaining*10)) or something.
    maxiter = int(min(400, max(20, remaining * 10)))  # heuristics
    
    try:
        res = minimize(obj, h_start.copy(), method='SLSQP', bounds=bnds, constraints=cons,

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(x, target_sum, tol=1e-8, max_iter=50):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(x)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    cur = np.sum(x)
    if abs(cur - target_sum) > 1e-6:
        # adjust smallest or largest
        if cur < target_sum:
            # need increase; pick index with most room
            i = np.argmax(1.0 - x)
            x[i] = np.clip(x[i] + (target_sum - cur), 0.0, 1.0)
        else:
            i = np.argmin(x)
            x[i] = np.clip(x[i] - (cur - target_sum), 0.0, 1.0)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(x, target_sum, tol=1e-10, max_iter=50):
    """
    Project a vector onto the box [0,1]^n and enforce the linear equality sum(x)=target_sum.
    Simple iterative scheme that redistributes the deficit among interior entries.
    """
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(x)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    cur = np.sum(x)
    if abs(cur - target_sum) > 1e-6:
        # Tiny final correction on the most free variable
        if cur < target_sum:
            i = np.argmax(1.0 - x)
            x[i] = np.clip(x[i] + (target_sum - cur), 0.0, 1.0)
        else:
            i = np.argmin(x)
            x[i] = np.clip(x[i] - (cur - target_sum), 0.0, 1.0)
    return x


def _largest_divisor_up_to(limit, n):
    """Largest divisor of n not exceeding limit (minimum 2)."""
    for d in range(limit, 1, -1):
        if n % d == 0:
            return d
    return 2


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Base construction, if supplied.
    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)
        n = base_h.shape[0]
    else:
        n = 380

    # Ensure even number of points (required for the integral constraint to be integral).
    if n % 2 != 0:
        n += 1

    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 1. Initialise best solution (use parent if present, otherwise random)
    # ------------------------------------------------------------------
    if base_h is not None:
        base_h = project_to_feasible(base_h, target_sum)
        best_h = base_h.copy()
        best_c5 = compute_c5(best_h)
    else:
        # random feasible start
        h0 = rng.random(n)
        best_h = project_to_feasible(h0, target_sum)
        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Piecewise‑constant random exploration (large portion of budget)
    # ------------------------------------------------------------------
    # Choose a modest number of equal‑width segments.
    p = _largest_divisor_up_to(20, n)          # e.g. 10 for n=380
    repeats = n // p                           # points per segment
    segment_target = p / 2.0                   # because sum(v_i)=p/2 ensures full integral

    # Allocate ~60 % of total time to this stochastic phase.
    explore_deadline = start_time + budget_s * 0.60
    while time.time() < explore_deadline:
        # Random segment heights
        v = rng.random(p)
        v = project_to_feasible(v, segment_target)
        h_candidate = np.repeat(v, repeats)
        # Small numeric mismatch possible after repeat -> final projection
        h_candidate = project_to_feasible(h_candidate, target_sum)
        c5 = compute_c5(h_candidate)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()

        # Occasionally jitter the current best (help escape flat regions)
        if rng.random() < 0.01:
            # Extract segment averages from best_h, perturb and re‑expand
            seg_means = np.array([best_h[i*repeats:(i+1)*repeats].mean() for i in range(p)])
            seg_means += rng.normal(scale=0.05, size=p)
            seg_means = np.clip(seg_means, 0.0, 1.0)
            seg_means = project_to_feasible(seg_means, segment_target)
            h_alt = np.repeat(seg_means, repeats)
            h_alt = project_to_feasible(h_alt, target_sum)
            c5_alt = compute_c5(h_alt)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()

    # ------------------------------------------------------------------
    # 3. Local refinement with SLSQP (moderate portion of budget)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        # Number of SLSQP iterations loosely tied to remaining wall‑time.
        maxiter = int(min(1200, max(200, remaining * 15)))

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------

    hill_deadline = start_time + budget_s * 0.98  # keep a tiny buffer
    cur_h = best_h.copy()
    cur_c5 = best_c5
    while time.time() < hill_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # Try moving mass from j to i
        delta_max = min(1.0 - cur_h[i], cur_h[j])
        if delta_max > 0.0:
            delta = rng.random() * delta_max
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            c5_new = compute_c5(new_h)
            if c5_new < cur_c5:
                cur_h, cur_c5 = new_h, c5_new
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, new_h.copy()
                continue
        # Try opposite direction
        delta_max = min(1.0 - cur_h[j], cur_h[i])
        if delta_max > 0.0:
            delta = rng.random() * delta_max
            new_h = cur_h.copy()
            new_h[j] += delta
            new_h[i] -= delta
            c5_new = compute_c5(new_h)
            if c5_new < cur_c5:
                cur_h, cur_c5 = new_h, c5_new
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, new_h.copy()
                continue

    # Final safety projection (should already be feasible)
    best_h = project_to_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
