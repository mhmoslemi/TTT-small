def project_box_sum(v, target, tol=1e-9, max_iter=50):
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
        if abs(s - target) < tol:
            break
    return y

def _gradient_min_corr(h):
    n = h.shape[0]
    S_full = np.correlate(h, h, mode='full')
    # find index where S is minimal
    min_idx = int(np.argmin(S_full))
    d = min_idx - (n - 1)  # shift
    g = np.zeros_like(h)
    if d >= 0:
        # overlapping pairs (i,i+d) for i = 0..n-d-1
        # derivative w.r.t h[i] is h[i+d]
        # derivative w.r.t h[i+d] is h[i]
        g[:n-d] += h[d:]
        g[d:] += h[:n-d]
    else:
        # d negative: equivalent to positive shift of -d from second half

import time, math, numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx = 1 - (2/n)*min_{k} autocorr."""
    n = h.shape[0]
    s = np.correlate(h, h, mode="full")
    return 1.0 - (2.0 / n) * float(s.min())

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry (recursively) so that sum(h) == target."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while forcing sum == target."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _gradient_min_corr(h: np.ndarray):
    """Return gradient of the minimal autocorrelation shift."""
    n = h.shape[0]
    full = np.correlate(h, h, mode="full")
    min_idx = int(np.argmin(full))
    d = min_idx - (n - 1)          # shift corresponding to minimum
    g = np.zeros_like(h)
    if d >= 0:
        # pairs (i,i+d) for i=0..n-d-1
        g[: n - d] += h[d:]
        g[d:] += h[: n - d]
    else:
        dpos = -d
        g[dpos:] += h[: n - dpos]
        g[: n - dpos] += h[dpos:]
    return g, d
