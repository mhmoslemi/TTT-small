def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
    else:
        n = 80
        init_arr = None
    target_ones = n // 2
    dx = 2.0 / n

    # Helper to create a binary vector with exactly target_ones ones
    def random_binary():
        h = np.zeros(n, dtype=int)
        idx = rng.choice(n, target_ones, replace=False)
        h[idx] = 1
        return h

    # Create initial population
    pop_size = 30
    population = []
    scores = []
    # If a parent construction is provided, use its top‑half as a seed
    if init_arr is not None:
        top_idx = np.argpartition(-init_arr, target_ones - 1)[:target_ones]
        seed_vec = np.zeros(n, dtype=int)
        seed_vec[top_idx] = 1
        population.append(seed_vec)
        scores.append(compute_c5(seed_vec))
    while len(population) < pop_size:
        vec = random_binary()
        population.append(vec)
        scores.append(compute_c5(vec))

    # Track the best solution found
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_c5 = scores[best_idx]

    # Parameters for the genetic loop
    mutate_prob = 0.2
    tournament_k = 3
    # allocate a small margin for final local optimisation
    end_time = start_time + budget_s - 0.5

    generation = 0
    while time.time() < end_time:
        generation += 1

        # --- tournament selection ---
        def tournament():
            idxs = rng.choice(pop_size, tournament_k, replace=False)
            best = idxs[0]
            best_score = scores[best]
            for ii in idxs[1:]:
                if scores[ii] < best_score:
                    best = ii
                    best_score = scores[ii]
            return best

        p1 = tournament()
        p2 = tournament()
        parent1 = population[p1]
        parent2 = population[p2]

        # --- uniform crossover ---
        mask = rng.random(n) < 0.5
        child = np.where(mask, parent1, parent2).astype(int)

        # --- repair to exact weight ---
        ones = int(child.sum())
        diff = target_ones - ones
        if diff > 0:
            zeros_idx = np.flatnonzero(child == 0)
            if len(zeros_idx) >= diff:
                flip = rng.choice(zeros_idx, diff, replace=False)
                child[flip] = 1
        elif diff < 0:
            ones_idx = np.flatnonzero(child == 1)
            flip = rng.choice(ones_idx, -diff, replace=False)
            child[flip] = 0

        # --- mutation (swap a 1 and a 0) ---
        if rng.random() < mutate_prob:
            ones_idx = np.flatnonzero(child == 1)
            zeros_idx = np.flatnonzero(child == 0)
            if len(ones_idx) and len(zeros_idx):
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                child[i] = 0
                child[j] = 1

        # --- evaluate child ---
        child_score = compute_c5(child)

        # --- replacement (elitist) ---
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score

        # --- update global best ---
        if child_score < best_c5:
            best_c5 = child_score
            best_h = child.copy()
            # early exit if we beat the target
            if best_c5 <= 0.3808:
                break

        # --- occasional local improvement on the current best ---
        if generation % 50 == 0 and time.time() < end_time - 0.2:
            # simple exhaustive swap search (one pass)
            cur_h = best_h.copy()
            cur_score = best_c5
            ones_idx = np.flatnonzero(cur_h == 1)
            zeros_idx = np.flatnonzero(cur_h == 0)
            improved = True
            while improved and time.time() < end_time - 0.1:
                improved = False
                best_delta = 0.0
                best_swap = None
                # explore all swaps (could be limited to a random subset for speed)
                for i in ones_idx:
                    for j in zeros_idx:
                        cur_h[i] = 0
                        cur_h[j] = 1
                        new_score = compute_c5(cur_h)
                        delta = cur_score - new_score
                        if delta > best_delta + 1e-12:
                            best_delta = delta
                            best_swap = (i, j, new_score)
                        # revert
                        cur_h[i] = 1
                        cur_h[j] = 0
                if best_swap is not None:
                    i, j, new_score = best_swap
                    cur_h[i] = 0
                    cur_h[j] = 1
                    cur_score = new_score
                    best_delta = 0.0
                    improved = True
                    # update index lists
                    ones_idx = np.flatnonzero(cur_h == 1)
                    zeros_idx = np.flatnonzero(cur_h == 0)
            # if an improvement was found, adopt it
            if cur_score < best_c5 - 1e-12:
                best_c5 = cur_score
                best_h = cur_h.copy()

    # Final safety: ensure exact half‑ones and bounds
    best_h = np.clip(best_h, 0, 1)
    # (Binary vectors already satisfy the sum exactly)
    return best_h.astype(float), best_c5, n

import time
import numpy as np

def compute_c5(h):
    """Return the overlap bound C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -------------------------------------------------
    # problem size
    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.size
    else:
        init_arr = None
        n = 80
    target_ones = n // 2
    dx = 2.0 / n

    # -------------------------------------------------
    # helpers for binary vectors
    def random_binary():
        """Return a random {0,1} vector with exactly target_ones ones."""
        v = np.zeros(n, dtype=int)
        v[rng.choice(n, target_ones, replace=False)] = 1
        return v

    # -------------------------------------------------
    # initial population for a simple GA
    pop_size = 30
    population = []
    scores = []

    # seed from parent construction if available
    if init_arr is not None:
        top = np.argpartition(-init_arr, target_ones - 1)[:target_ones]
        seed_vec = np.zeros(n, dtype=int)
        seed_vec[top] = 1
        population.append(seed_vec)
        scores.append(compute_c5(seed_vec))

    while len(population) < pop_size:
        vec = random_binary()
        population.append(vec)
        scores.append(compute_c5(vec))

    # best solution seen so far
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_c5 = scores[best_idx]

    # GA parameters
    mutate_prob = 0.2
    tournament_k = 3
    # leave a small margin for a final continuous refinement phase
    phase_end = start_time + budget_s - 0.5

    # -------------------------------------------------
    # Genetic optimisation loop
    generation = 0
    while time.time() < phase_end:
        generation += 1

        # ----- tournament selection -----
        def tournament():
            idxs = rng.choice(pop_size, tournament_k, replace=False)
            best = idxs[0]
            best_score = scores[best]
            for ii in idxs[1:]:
                if scores[ii] < best_score:
                    best = ii
                    best_score = scores[ii]
            return best

        p1 = tournament()
        p2 = tournament()
        parent1 = population[p1]
        parent2 = population[p2]

        # ----- uniform crossover -----
        mask = rng.random(n) < 0.5
        child = np.where(mask, parent1, parent2).astype(int)

        # ----- repair to exact weight -----
        ones = int(child.sum())
        diff = target_ones - ones
        if diff > 0:
            zeros_idx = np.flatnonzero(child == 0)
            if zeros_idx.size >= diff:
                child[rng.choice(zeros_idx, diff, replace=False)] = 1
        elif diff < 0:
            ones_idx = np.flatnonzero(child == 1)
            child[rng.choice(ones_idx, -diff, replace=False)] = 0

        # ----- mutation (swap a 1 and a 0) -----
        if rng.random() < mutate_prob:
            ones_idx = np.flatnonzero(child == 1)
            zeros_idx = np.flatnonzero(child == 0)
            if ones_idx.size and zeros_idx.size:
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                child[i] = 0
                child[j] = 1

        # ----- evaluate -----
        child_score = compute_c5(child)

        # ----- replace worst if better -----
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score

        # ----- update global best -----
        if child_score < best_c5:
            best_c5 = child_score
            best_h = child.copy()
            if best_c5 <= 0.3808:        # reached target
                break

        # ----- occasional local exhaustive swap on the current best -----
        if generation % 50 == 0 and time.time() < phase_end - 0.2:
            cur_h = best_h.copy()
            cur_score = best_c5
            ones_idx = np.flatnonzero(cur_h == 1)
            zeros_idx = np.flatnonzero(cur_h == 0)
            improved = True
            while improved and time.time() < phase_end - 0.1:
                improved = False
                best_delta = 0.0
                best_swap = None
                for i in ones_idx:
                    for j in zeros_idx:
                        cur_h[i] = 0
                        cur_h[j] = 1
                        new_score = compute_c5(cur_h)
                        delta = cur_score - new_score
                        if delta > best_delta + 1e-12:
                            best_delta = delta
                            best_swap = (i, j, new_score)
                        cur_h[i] = 1
                        cur_h[j] = 0
                if best_swap is not None:
                    i, j, new_score = best_swap
                    cur_h[i] = 0
                    cur_h[j] = 1
                    cur_score = new_score
                    improved = True
                    ones_idx = np.flatnonzero(cur_h == 1)
                    zeros_idx = np.flatnonzero(cur_h == 0)
            if cur_score < best_c5 - 1e-12:
                best_c5 = cur_score
                best_h = cur_h.copy()

    # -------------------------------------------------
    # Continuous refinement (fractional moves) if time permits
    refine_end = start_time + budget_s - 0.05
    cur_h = best_h.astype(float)          # start from the best binary vector
    cur_score = best_c5

    while time.time() < refine_end:
        # pick a source with positive mass and a sink with room
        src_idx = rng.choice(np.flatnonzero(cur_h > 1e-12))
        dst_idx = rng.choice(np.flatnonzero(cur_h < 1 - 1e-12))
        max_eps = min(cur_h[src_idx], 1.0 - cur_h[dst_idx])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5   # take at most half of the available mass
        cand = cur_h.copy()
        cand[src_idx] -= eps
        cand[dst_idx] += eps
        cand_score = compute_c5(cand)
        if cand_score < cur_score:
            cur_h = cand
            cur_score = cand_score
            if cand_score < best_c5:
                best_c5 = cand_score
                best_h = cand.copy()
                if best_c5 <= 0.3808:
                    break

    # -------------------------------------------------
    # final safety: ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    # integer solution automatically satisfies the exact sum; fractional
    # version also satisfies it by construction.
    return best_h.astype(float), best_c5, n
