def project_to_bounds_sum(v, target_sum):
    n = len(v)
    # clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # adjust sum
    # Use water-filling algorithm:
    # Compute diff = target_sum - np.sum(v)
    diff = target_sum - np.sum(v)
    # if diff > 0: need to increase some x_i below 1
    # if diff < 0: need to decrease some x_i above 0
    # iterate until diff ~0
    # This simple loop may converge quickly for small n.
    loops = 0
    while abs(diff) > 1e-12 and loops < 1000:
        if diff > 0:
            free = np.where(v < 1.0)[0]
        else:
            free = np.where(v > 0.0)[0]
        if free.size == 0:
            break
        # compute adjustment per free variable
        adj = diff / free.size
        # clip adjustment to avoid exceeding bounds
        if diff > 0:
            # up
            possible = 1.0 - v[free]
            delta = np.minimum(adj, possible)
        else:
            possible = -v[free] # amount we can decrease (negative)
            delta = np.maximum(adj, possible) # adj negative
        v[free] += delta
        diff = target_sum - np.sum(v)
        loops += 1
    return v

def continuous_hill_climb(h, cur_c5, start_time, budget_s):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.default_rng()
    # step size initially maybe 0.2
    step = 0.2
    best_h = h.copy()
    best_val = cur_c5
    cur_h = h.copy()
    cur_val = cur_c5
    no_improve = 0
    while time.time() - start_time < budget_s - 0.5:
        # propose random i, j distinct
        i, j = rng.choice(n, size=2, replace=False)
        # random direction
        if rng.random() < 0.5:
            # move from j to i
            max_delta = min(step, 1.0 - cur_h[i], cur_h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            max_delta = min(step, 1.0 - cur_h[j], cur_h[i])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            new_h = cur_h.copy()
            new_h[j] += delta
            new_h[i] -= delta

        # evaluate
        new_val = compute_c5(new_h)
        if new_val + 1e-12 < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = new_h.copy()
                no_improve = 0
                # optionally reduce step to fine-tune
                step = max(step * 0.99, 1e-4)
            else:
                no_improve = 0
        else:
            no_improve += 1
            # optionally increase step if many failures
            if no_improve > 2000:
                step = min(step * 1.1, 0.5)
                no_improve = 0
    return best_h, best_val

# ---------- Helper constraints ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ---------- Core utilities ----------
def compute_c5(h):
    """Maximum overlap for a vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_bounds(v, target):
    """Project vector onto [0,1] box while fixing the sum."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - np.sum(v)
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(v < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            max_inc = 1.0 - v[free]
            delta = np.minimum(inc, max_inc)
            v[free] += delta
        else:
            free = np.where(v > 0.0)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            max_dec = -v[free]
            delta = np.maximum(dec, max_dec)
            v[free] += delta

        if idx.size:
            i = idx[0]
            v[i] = min(max(v[i] + diff, 0.0), 1.0)
    return v

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise h -----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
    target_sum = n / 2.0
    h0 = project_to_sum_bounds(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----- constraints for SLSQP -----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # ----- Phase 1: many SLSQP restarts -----
    slsqp_budget = budget_s * 0.6
    attempts = 0
    while time.time() - start_time < slsqp_budget:
        attempts += 1
        # random start: sometimes perturb current best, otherwise fresh uniform
        if attempts % 3 == 0 and best_c5 < 0.5:
            start_h = best_h + rng.normal(scale=0.1, size=n)
        else:
            start_h = rng.random(n)
        start_h = project_to_sum_bounds(start_h, target_sum)
        t0 = compute_c5(start_h)
        x0 = np.concatenate([start_h, [t0]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 800, 'ftol': 1e-9, 'disp': False}
            )
        except Exception:
            continue

        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_sum_bounds(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3805:  # early exit if exceptionally good
                break

    # ----- Phase 2: continuous hill climbing -----
    if time.time() - start_time < budget_s - 0.5:
        best_h, best_c5 = continuous_hill_climb(best_h, best_c5, start_time, budget_s)

    # final safety projection
    best_h = project_to_sum_bounds(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
