def gradient_max_shift(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    L = max_idx - (N - 1)  # lag
    g = np.zeros_like(h)
    if L >= 0:
        # term1 for i in [0, N-L-1]: (1 - h[i+L]) 
        g[:N-L] += 1 - h[L:]
        # term2 for i in [L, N-1]: -h[i-L]
        g[L:] -= h[:N-L]
    else:
        L_pos = -L
        # L negative meaning shift left: i+L out for i<L_pos
        # term1 for i where i+L valid: i from L_pos..N-1 (since i+L >=0)
        g[L_pos:] += 1 - h[:N-L_pos]
        # term2: -h[i-L] = -h[i+L_pos] for i in 0..N-L_pos-1 (since i-L = i+L_pos valid)
        g[:N-L_pos] -= h[L_pos:]
    return g, corr[max_idx] * dx, L
```

import numpy as np
import time
import math

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------
def _repair_vector(v: np.ndarray) -> np.ndarray:
    """
    Clip to [0,1] and adjust sum to N/2 exactly (where N = len(v)).
    Returns a new array.
    """
    v = np.clip(v, 0.0, 1.0)
    N = v.size
    target = N / 2.0
    s = v.sum()
    if s == 0.0:
        v = np.full_like(v, target / N)
        v = np.clip(v, 0.0, 1.0)
        return v
    factor = target / s
    v = v * factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v = v * factor2
        v = np.clip(v, 0.0, 1.0)
    return v


def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _gradient_max_shift(h: np.ndarray):
    """
    Compute gradient of the correlation term that attains the current maximum.
    Returns (gradient, current C5 value, lag).
    """
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    lag = idx - (N - 1)                # shift index (can be negative)
    g = np.zeros_like(h)

    if lag >= 0:
        k = lag
        # term from h[i] * (1 - h[i+k])
        g[:N - k] += 1.0 - h[k:]
        # term from -h[i-k] * h[i]
        g[k:] -= h[:N - k]
    else:
        k = -lag
        # i+lag >=0  -> i >= k
        g[k:] += 1.0 - h[:N - k]
        # i-lag < N -> i <= N-k-1
        g[:N - k] -= h[k:]

    c5 = float(max_val * (2.0 / N))
    return g, c5, lag


def _pgd_optimize(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """
    Projected gradient descent on the feasible set.
    Returns best vector found and its C5 value.
    """
    h_cur = _repair_vector(h_start.copy())
    best_h = h_cur.copy()
    best_score = _c5_value(h_cur)

    start = time.time()
    init_step = 0.2
    step = init_step
    stagnation = 0

    while time.time() - start < time_limit:
        # gradient of the active shift
        g, cur_score, _ = _gradient_max_shift(h_cur)
        norm = np.linalg.norm(g)
        if norm == 0.0:
            # random nudge if gradient vanishes
            h_cur = _repair_vector(h_cur + (rng.random(h_cur.size) - 0.5) * 0.1)
            continue

        direction = g / norm
        h_new = _repair_vector(h_cur - step * direction)
        new_score = _c5_value(h_new)

        if new_score < cur_score:
            # successful step
            h_cur = h_new
            step = min(step * 1.1, 1.0)
            stagnation = 0
        else:
            # step not improving
            step *= 0.5
            stagnation += 1
            if step < 1e-6 or stagnation > 15:
                # random restart around the current best
                h_cur = _repair_vector(best_h + (rng.random(best_h.size) - 0.5) * 0.1)
                step = init_step
                stagnation = 0

        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()

    return best_h, best_score


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    """
    rng = np.random.default_rng(seed)

    # ----- initial construction -------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 100
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)], dtype=np.float64)
    h0 = _repair_vector(h0)

    start_total = time.time()
    elapsed = lambda: time.time() - start_total
    remaining = lambda: max(0.0, budget_s - elapsed())

    # ----- Phase 1: optimise at base resolution ---------------------------------
    phase1_time = remaining() * 0.45
    best_h, best_score = _pgd_optimize(h0, phase1_time, rng)

    # ----- Phase 1b: random restarts at same resolution ------------------------
    while remaining() > 12.0:                     # keep a little buffer for later phases
        chunk = min(6.0, remaining() * 0.10)
        h_rand = _repair_vector(rng.random(best_h.size))
        h_opt, sc_opt = _pgd_optimize(h_rand, chunk, rng)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    # ----- Phase 2: increase resolution iteratively -----------------------------
    N_max = 800   # hard cap to keep evaluation cheap
    while remaining() > 30.0:
        N_cur = best_h.size
        if N_cur >= N_max:
            break
        N_new = min(N_cur * 2, N_max)

        # linear interpolation to new grid
        x_old = np.linspace(0.0, 2.0, N_cur, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N_new, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _repair_vector(h_up)

        # allocate a chunk of the remaining budget (roughly half)
        phase_time = remaining() * 0.5
        best_h, best_score = _pgd_optimize(h_up, phase_time, rng)

        # occasional random restarts at the higher resolution
        while remaining() > 12.0 and remaining() * 0.10 > 2.0:
            chunk = min(4.0, remaining() * 0.10)
            h_rand = _repair_vector(rng.random(best_h.size))
            h_opt, sc_opt = _pgd_optimize(h_rand, chunk, rng)
            if sc_opt < best_score:
                best_h, best_score = h_opt, sc_opt

    # ----- final repair & evaluation --------------------------------------------
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
