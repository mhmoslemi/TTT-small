def _anneal(h_start, time_budget, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.01   # initial temperature?
    max_eps_init = 0.5
    max_eps_final = 0.001
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget  # 0..1
        T = T0 * (1 - frac)  # linear decay
        max_eps = max_eps_init * (1 - frac) + max_eps_final * frac
        # choose move type
        if rng.random() < 0.1:  # targeted
            # compute correlation etc
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            s_max = np.argmax(corr)
            shift = s_max - (N - 1)
            # compute contributions
            p = np.zeros(N)
            idx = np.arange(N)
            idx2 = idx + shift
            valid = (idx2 >= 0) & (idx2 < N)
            p[valid] = h_cur[valid] * (1.0 - h_cur[idx2[valid]])
            # if not enough variation, fallback
            if np.all(p == 0):
                i = rng.integers(N)
                j = rng.integers(N)
                while i == j:
                    j = rng.integers(N)
            else:
                # choose high and low indices
                sorted_idx = np.argsort(p)
                low_k = max(1, N // 10)
                high_k = low_k
                low_idxs = sorted_idx[:low_k]
                high_idxs = sorted_idx[-high_k:]
                i = int(rng.choice(high_idxs))
                j = int(rng.choice(low_idxs))
                if i == j:
                    # choose different j
                    alt = rng.integers(N)
                    while alt == i:
                        alt = rng.integers(N)
                    j = alt
        else:
            i = int(rng.integers(N))
            j = int(rng.integers(N))
            while i == j:
                j = int(rng.integers(N))
        hi = h_cur[i]
        hj = h_cur[j]
        # compute feasible eps range
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if lower >= upper:
            continue
        # sample eps
        eps_raw = rng.uniform(-max_eps, max_eps)
        eps = np.clip(eps_raw, lower, upper)
        if eps == 0.0:
            continue
        # apply move
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # ensure clip
        # (should already be within [0,1] thanks to constraints)
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or (T > 0 and rng.random() < math.exp(-delta / (T + 1e-15))):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _greedy_refine(h_start, time_budget, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    N = h_cur.size
    start = time.time()
    max_eps = 0.001  # small steps
    while time.time() - start < time_budget:
        i = rng.integers(N)
        j = rng.integers(N)
        while i == j:
            j = rng.integers(N)
        hi = h_cur[i]; hj = h_cur[j]
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if lower >= upper:
            continue
        eps_raw = rng.uniform(-max_eps, max_eps)
        eps = np.clip(eps_raw, lower, upper)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        if new_score < cur_score:
            h_cur = h_new
            cur_score = new_score
    return h_cur, cur_score

import numpy as np

def _repair_vector(v):
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    v = np.asarray(v, dtype=np.float64)
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    for _ in range(3):
        s = v.sum()
        if abs(s - target) <= 1e-12:
            break
        if s == 0.0:
            v[:] = target / v.size
            np.clip(v, 0.0, 1.0, out=v)
            continue
        factor = target / s
        v *= factor
        np.clip(v, 0.0, 1.0, out=v)
    # final residual correction if needed
    s = v.sum()
    if abs(s - target) > 1e-10:
        diff = target - s
        v += diff / v.size
        np.clip(v, 0.0, 1.0, out=v)
        s = v.sum()
        if abs(s - target) > 1e-10:
            v *= target / s
            np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h):
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _anneal(h_start, time_budget, rng):
    """Simulated annealing on the sum‑preserving simplex."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.02                # initial temperature
    max_eps_init = 0.5       # start step size
    max_eps_final = 0.001    # final step size
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget          # 0 → 1
        T = T0 * (1.0 - frac)                # linear cooling
        max_eps = max_eps_init * (1.0 - frac) + max_eps_final * frac

        if rng.random() < 0.1:                # targeted move ~10%

            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')

            shift = s_max - (N - 1)
            # contribution of each index to this shift
            p = np.zeros(N, dtype=np.float64)
            idx = np.arange(N)

                i = rng.integers(N)
                j = rng.integers(N)

                j = rng.integers(N)
        hi = h_cur[i]
        hj = h_cur[j]
        # feasible epsilon interval while preserving 0‑1 bounds and sum
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if lower >= upper:
            continue
        eps_raw = rng.uniform(-max_eps, max_eps)
        eps = np.clip(eps_raw, lower, upper)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or (T > 0.0 and rng.random() < math.exp(-delta / (T + 1e-15))):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _greedy_refine(h_start, time_budget, rng):
    """Deterministic hill‑climbing (accept only improvements)."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    N = h_cur.size
    start = time.time()
    max_eps = 0.001
    while time.time() - start < time_budget:
        i = rng.integers(N)
        j = rng.integers(N)
        while i == j:
            j = rng.integers(N)
        hi = h_cur[i]
        hj = h_cur[j]
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if lower >= upper:
            continue
        eps_raw = rng.uniform(-max_eps, max_eps)
        eps = np.clip(eps_raw, lower, upper)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        if new_score < cur_score:
            h_cur = h_new
            cur_score = new_score
    return h_cur, cur_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # --- initial construction -------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except Exception:
        n_fallback = 100
        h0 = np.concatenate([np.ones(n_fallback // 2),
                             np.zeros(n_fallback - n_fallback // 2)],
                            dtype=np.float64)
    h0 = _repair_vector(h0)
    n0 = h0.size

    start_total = time.time()
    # Phase 1 – quick optimisation at the original resolution
    phase1_time = max(5.0, budget_s * 0.1)    # at least a few seconds
    if phase1_time > budget_s:
        phase1_time = budget_s
    best_h, best_score = _anneal(h0, phase1_time, rng)

    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining <= 0.5:
        final_h = best_h
        final_score = best_score
        final_h = _repair_vector(final_h)
        final_score = _c5_value(final_h)
        return final_h, float(final_score), final_h.size

    # Phase 2 – upscale and deeper search
    target_n = min(1000, max(200, n0 * 8))   # e.g. 760 for n0≈95
    if target_n != n0:
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _repair_vector(h_up)
    else:
        h_up = best_h.copy()
    # Allocate most of the remaining time to annealing from the current best
    phase2_time = remaining * 0.7
    h2, s2 = _anneal(h_up, phase2_time, rng)

    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    # Optional random‑restart if time permits
    if remaining > 10.0:
        rand_h = rng.random(target_n)
        rand_h = _repair_vector(rand_h)
        h3, s3 = _anneal(rand_h, remaining * 0.5, rng)
        if s3 < s2:
            h2, s2 = h3, s3
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed

    # Final greedy refinement with any leftover budget
    if remaining > 0.5:
        final_h, final_score = _greedy_refine(h2, remaining, rng)
    else:
        final_h, final_score = h2, s2

    final_h = _repair_vector(final_h)
    final_score = _c5_value(final_h)
    return final_h, float(final_score), final_h.size
