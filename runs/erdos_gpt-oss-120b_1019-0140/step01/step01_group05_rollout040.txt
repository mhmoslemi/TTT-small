    N = h.size

    # compute convolution of h and (1 - h) reversed
    # Actually we need correlation: sum_i h[i] * (1 - h[i + k]) with zero padding.
    # This equals convolution of h with rev(1 - h) (where rev means reversed)

    a = h.astype(np.float64)
    b = (1 - h)[::-1].astype(np.float64)

    conv = np.fft.irfft(fa * fb, n=size)[:2*N-1]

    N = h.size

    min_s = auto.min()
    if min_s < 0 and min_s > -1e-12:

import numpy as np
import time
import math

def _repair_vector(v):
    """Clip values to [0,1] and enforce sum(v) == len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size * 0.5
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    v *= target / s
    np.clip(v, 0.0, 1.0, out=v)
    for _ in range(5):
        s = v.sum()
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            v += diff / v.size
            np.clip(v, 0.0, 1.0, out=v)
        else:
            add = diff / free.size
            v[free] += add
            np.clip(v, 0.0, 1.0, out=v)
    s = v.sum()
    if abs(s - target) > 1e-12:
        v *= target / s
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h):
    """Return C5 = 1 - (2/N) * min auto‑correlation of h."""
    N = h.size
    a = h.astype(np.float64)
    size = 1 << ((2 * N - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    auto = np.fft.irfft(fa * np.conj(fa), n=size)[:2 * N - 1]
    min_s = auto.min()
    if min_s < 0 and min_s > -1e-12:
        min_s = 0.0
    return float(1.0 - (2.0 / N) * min_s)

def _anneal(h_start, time_limit, rng,
            init_temp=0.02, final_temp=1e-6,
            max_eps=0.2, target_prob=0.1):
    """Simulated annealing with occasional moves targeting the worst shift."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = init_temp * frac + final_temp * (1.0 - frac)

        if rng.random() < target_prob:
            # identify the shift with minimal auto‑correlation
            size = 1 << ((2 * N - 1).bit_length())
            fa = np.fft.rfft(h_cur, n=size)
            auto = np.fft.irfft(fa * np.conj(fa), n=size)[:2 * N - 1]
            min_idx = int(np.argmin(auto))

                hi, hd = h_cur[target], h_cur[donor]
