def anneal(h, target_sum, max_iter, start_temp, end_temp):
    best = h.copy()
    best_score = compute_c5(best)
    cur = h.copy()
    cur_score = best_score
    for i in range(max_iter):
        T = start_temp * (1 - i / max_iter) + end_temp * (i / max_iter)  # linear cooling
        # propose new candidate: perturb cur by adding gaussian noise * step factor
        step = 0.1  # scale
        cand = cur + np.random.randn(len(cur)) * step
        cand = project_to_constraints(cand, target_sum)
        cand_score = compute_c5(cand)
        if cand_score < cur_score or np.random.rand() < np.exp((cur_score - cand_score)/ (T+1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
    return best, best_score

def project_to_constraints(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return v
    # Need to adjust sum while staying within [0,1].
    # Simple water-filling: If diff > 0, increase low values as much as possible.
    if diff > 0:
        # indices where v < 1
        free = np.where(v < 1)[0]
        # capacities to fill: 1 - v
        caps = 1 - v[free]
        # total capacity
        total_cap = caps.sum()
        if total_cap < diff:  # cannot fill enough; just fill to 1 then scale all?
            # fill all to 1
            v[free] = 1.0
            # scale down others to meet target
            cur = v.sum()
            diff = target_sum - cur
        # distribute diff proportionally maybe equally
        # Here we can fill proportionally to caps
        # use while loop to add to those with capacity
        while diff > 1e-12:
            free = np.where(v < 1 - 1e-12)[0]
            if free.size == 0:
                break
            add = diff / free.size
            # ensure not exceed
            add = np.minimum(add, 1 - v[free])
            v[free] += add
            diff = target_sum - v.sum()
    else:
        # diff < 0, need to reduce sum
        excess = -diff
        free = np.where(v > 0)[0]
        while excess > 1e-12:
            free = np.where(v > 1e-12)[0]
            if free.size == 0:
                break
            sub = excess / free.size
            sub = np.minimum(sub, v[free])
            v[free] -= sub
            excess = v.sum() - target_sum
    # Clip
    v = np.clip(v, 0.0, 1.0)
    # final scaling if small discrepancy
    cur = v.sum()
    if abs(cur - target_sum) > 1e-8:
        # scale up/down (but preserve bounds)
        # Use simple scaling factor
        factor = target_sum / cur
        v = np.clip(v * factor, 0.0, 1.0)
        # after scaling may deviate, fix again with water-filling iteration? But small.
    return v

import numpy as np
import time

def _compute_c5(h):
    """Discrete C5: max_k ∫ h(x)(1-h(x+k)) dx."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust sum(v) to target via water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    eps = 1e-12

    if abs(cur - target) <= eps:
        return v

    # Increase sum if needed
    if cur < target:
        deficit = target - cur
        while deficit > eps:
            free = np.where(v < 1.0 - eps)[0]
            if free.size == 0:
                break
            add = deficit / free.size
            caps = 1.0 - v[free]
            add = np.minimum(add, caps)
            v[free] += add
            deficit = target - v.sum()
    else:
        # Decrease sum
        excess = cur - target
        while excess > eps:
            free = np.where(v > eps)[0]
            if free.size == 0:
                break
            sub = excess / free.size
            caps = v[free]
            sub = np.minimum(sub, caps)
            v[free] -= sub
            excess = v.sum() - target

    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
        factor = target / cur
        v *= factor
        v = np.clip(v, 0.0, 1.0)
    return v

        return budget_s - (time.time() - start_time)
