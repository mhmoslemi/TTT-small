def hill_climb(h, best_c5, start_time, budget_s):
    h_cur = h.copy()
    cur_val = best_c5
    rng = np.random.default_rng()
    attempts_without_improvement = 0
    while time.time() - start_time < budget_s - 0.1:
        # propose swap
        ones = np.flatnonzero(h_cur == 1)
        zeros = np.flatnonzero(h_cur == 0)
        if len(ones) == 0 or len(zeros) == 0: break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_new = h_cur.copy()
        h_new[i] = 0
        h_new[j] = 1
        val = compute_c5(h_new)
        if val < cur_val - 1e-12:
            h_cur, cur_val = h_new, val
            attempts_without_improvement = 0
        else:
            attempts_without_improvement += 1
            if attempts_without_improvement >= 2000:
                # random restart: perform some random swaps to escape
                # maybe random set of 5 swaps
                for _ in range(5):
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    h_cur[i] = 0; h_cur[j] = 1
                cur_val = compute_c5(h_cur)
                attempts_without_improvement = 0
    return h_cur, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = len(init_h)
    else:
        n = 80
        init_h = np.zeros(n)
    # convert to binary: take top half indices
    idx = np.argsort(-init_h)[: n//2]  # descending
    init_binary = np.zeros(n, dtype=np.int8)
    init_binary[idx] = 1

    # Helper function compute_c5
    def compute_c5_vec(h):
        # h is int array of 0/1 length n
        # compute correlation with 1-h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)

    # Setup population
    pop_size = 150
    target_sum = n // 2
    # population as np.ndarray shape (pop_size, n)
    pop = np.empty((pop_size, n), dtype=np.int8)
    # First individual is init_binary
    pop[0] = init_binary
    # Fill rest randomly
    for i in range(1, pop_size):
        # random positions for ones
        ones_idx = rng.choice(n, target_sum, replace=False)
        vec = np.zeros(n, dtype=np.int8)
        vec[ones_idx] = 1
        pop[i] = vec

    # Compute fitness for all individuals
    fitness = np.empty(pop_size)
    for i in range(pop_size):
        fitness[i] = compute_c5_vec(pop[i])

    # Track best
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c5 = float(fitness[best_idx])

    # GA loop
    # parameters
    tournament_size = 5
    crossover_rate = 0.7  # probability to do crossover (else copy)
    mutation_rate = 0.3
    # For speed assign reference to functions
    while time.time() - start_time < budget_s - 2.0:
        # selection
        # tournament: pick tournament_size random indices
        tournament = rng.choice(pop_size, tournament_size, replace=False)
        parent1_idx = tournament[np.argmin(fitness[tournament])]

        tournament = rng.choice(pop_size, tournament_size, replace=False)
        parent2_idx = tournament[np.argmin(fitness[tournament])]

        parent1 = pop[parent1_idx]
        parent2 = pop[parent2_idx]

        # produce child
        child = parent1.copy()
        if rng.random() < crossover_rate:
            # uniform crossover with repair
            diff_positions = np.where(parent1 != parent2)[0]
            # choose proportion of diff positions to adopt from parent2
            # sample half of diff_positions (or up to some ratio)
            if diff_positions.size > 0:

                # ensure at least 1
                if n_swap > 0:
                    chosen = rng.choice(diff_positions, n_swap, replace=False)
                    child[chosen] = parent2[chosen]

            # repair to maintain exact sum
            sum_child = int(np.sum(child))
            diff = sum_child - target_sum
            if diff > 0:
                # too many ones: turn off diff random ones

                # ensure we have enough

                    to_zero = rng.choice(ones_pos, diff, replace=False)
                    child[to_zero] = 0
                else:
                    # fallback: set random positions to zero till sum correct
                    for _ in range(diff):
                        i = rng.choice(np.flatnonzero(child == 1))
                        child[i] = 0
            elif diff < 0:
                # too few ones: turn on -diff random zeros
                zeros_pos = np.flatnonzero(child == 0)
                need = -diff
                if need <= len(zeros_pos):
                    to_one = rng.choice(zeros_pos, need, replace=False)

import time
import numpy as np

def compute_c5_vec(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def generate_random_binary(n, target_sum, rng):
    vec = np.zeros(n, dtype=np.int8)
    vec[rng.choice(n, target_sum, replace=False)] = 1
    return vec

def hill_climb_best(h, cur_c5, start_time, budget_s):
    n = h.shape[0]
    target_sum = n // 2
    rng = np.random.default_rng()
    best_h = h.copy()
    best_val = cur_c5
    while time.time() - start_time < budget_s - 0.2:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        val = compute_c5_vec(cand)
        if val < best_val - 1e-12:
            best_h = cand
            best_val = val
            if best_val < 0.3808:
                break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # initialise problem size and binary start vector
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        half = n // 2
        idx_top = np.argsort(-init_arr)[:half]
        init_binary = np.zeros(n, dtype=np.int8)
        init_binary[idx_top] = 1
    else:
        n = 80
        init_binary = generate_random_binary(n, n // 2, rng)

    target_sum = n // 2

    # GA parameters
    pop_size = 150
    tournament_size = 5
    crossover_rate = 0.7
    mutation_rate = 0.3

    # population allocation
    pop = np.empty((pop_size, n), dtype=np.int8)
    pop[0] = init_binary
    for i in range(1, pop_size):
        pop[i] = generate_random_binary(n, target_sum, rng)

    # fitness evaluation
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5_vec(pop[i])

    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c5 = float(fitness[best_idx])

    # main GA loop
    while time.time() - start_time < budget_s - 5.0:
        # tournament selection
        tour = rng.choice(pop_size, tournament_size, replace=False)
        parent1_idx = tour[np.argmin(fitness[tour])]
        tour = rng.choice(pop_size, tournament_size, replace=False)
        parent2_idx = tour[np.argmin(fitness[tour])]
        parent1 = pop[parent1_idx]
        parent2 = pop[parent2_idx]

        # offspring creation
        child = parent1.copy()
        if rng.random() < crossover_rate:
            diff_pos = np.where(parent1 != parent2)[0]
            if diff_pos.size:
                n_swap = diff_pos.size // 2
                if n_swap:
                    chosen = rng.choice(diff_pos, n_swap, replace=False)
                    child[chosen] = parent2[chosen]
            cur_sum = int(child.sum())
            diff = cur_sum - target_sum
            if diff > 0:
                ones_idx = np.flatnonzero(child == 1)
                child[rng.choice(ones_idx, diff, replace=False)] = 0
            elif diff < 0:
                zeros_idx = np.flatnonzero(child == 0)
                child[rng.choice(zeros_idx, -diff, replace=False)] = 1

        # mutation
        if rng.random() < mutation_rate:
            ones_idx = np.flatnonzero(child == 1)
            zeros_idx = np.flatnonzero(child == 0)
            if ones_idx.size and zeros_idx.size:
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                child[i] = 0
                child[j] = 1

        child_fit = compute_c5_vec(child)

        # replace worst if better
        worst_idx = int(np.argmax(fitness))
        if child_fit < fitness[worst_idx]:
            pop[worst_idx] = child
            fitness[worst_idx] = child_fit
            if child_fit < best_c5:
                best_c5 = float(child_fit)
                best_h = child.copy()
                if best_c5 < 0.3808:
                    break

    # local refinement
    if time.time() - start_time < budget_s - 1.0:
        best_h, best_c5 = hill_climb_best(best_h, best_c5, start_time, budget_s)

    # final sum correction
    cur_sum = int(best_h.sum())
    if cur_sum != target_sum:
        diff = cur_sum - target_sum
        if diff > 0:
            ones_idx = np.flatnonzero(best_h == 1)
            best_h[rng.choice(ones_idx, diff, replace=False)] = 0
        else:
            zeros_idx = np.flatnonzero(best_h == 0)
            best_h[rng.choice(zeros_idx, -diff, replace=False)] = 1

    return best_h.astype(float), best_c5, n
