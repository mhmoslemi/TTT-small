def continuous_sa(h_start, start_c5, start_time, time_limit):
    n = len(h_start)
    target_sum = n / 2.0
    cur_h = h_start.copy()
    cur_c5 = start_c5
    best_h = cur_h.copy()
    best_c5 = cur_c5
    # annealing parameters
    T0 = 0.01
    # exponential cooling factor per iteration maybe something like 0.9999
    iter_count = 0
    while time.time() - start_time < time_limit:
        # compute temperature linearly decreasing
        elapsed = time.time() - start_time
        frac = elapsed / time_limit
        T = T0 * (1 - frac)
        if T < 1e-6:
            T = 1e-6
        # propose move
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)  # ensure distinct
        # determine max transfer
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:

import time
import math
import numpy as np

def compute_c5(h):
    """Maximum overlap integral C5 for a given h (float array)."""
    n = len(h)
    dx = 2.0 / n
    # correlation of h with 1-h (linear, zero-padded)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def adjust_sum(h, target):
    """Project h onto the hyperplane sum(h)=target while keeping 0≤h≤1."""
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # simple redistribution; iterate until residual is negligible
    h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    # fix residual by greedy fill
    for i in range(len(h)):
        if abs(residual) < 1e-12:
            break
        if residual > 0 and h[i] < 1.0:
            inc = min(residual, 1.0 - h[i])
            h[i] += inc
            residual -= inc
        elif residual < 0 and h[i] > 0.0:
            dec = min(-residual, h[i])
            h[i] -= dec
            residual += dec
    return h

def hill_climb_binary(h, start_c5, start_time, deadline, rng):
    """Greedy swap hill‑climbing on a binary vector, strict improvements only."""
    n = len(h)
    best_h = h.copy()
    best_c5 = start_c5
    while time.time() < deadline:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        val = compute_c5(cand)
        if val + 1e-12 < best_c5:
            best_h = cand
            best_c5 = val
            if best_c5 <= 0.3808:  # early exit if we already beat target
                break
    return best_h, best_c5

def continuous_sa(h_start, start_c5, start_time, deadline, rng):
    """Simulated annealing on a continuous vector preserving the sum."""
    n = len(h_start)
    target = n / 2.0
    cur_h = h_start.copy()
    cur_c5 = start_c5
    best_h = cur_h.copy()
    best_c5 = cur_c5

    T0 = 0.02                     # initial temperature
    # linear cooling schedule based on remaining fraction
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = T0 * max(0.0, 1.0 - frac) + 1e-8   # avoid zero division

        # propose a mass transfer between two distinct coordinates
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # direction: try moving from i to j; if impossible, swap roles
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0.0:
            max_transfer = min(cur_h[j], 1.0 - cur_h[i])
            if max_transfer <= 0.0:
                continue
            i, j = j, i    # reverse direction

        # scale the magnitude of the move with temperature (smaller later)
        scale = max(T / T0, 0.01)   # at low T use tiny steps
        delta = rng.random() * max_transfer * scale
        if delta <= 0.0:
            continue

        cand_h = cur_h.copy()
        cand_h[i] -= delta
        cand_h[j] += delta

        # numerical safety
        cand_h[i] = max(0.0, min(1.0, cand_h[i]))
        cand_h[j] = max(0.0, min(1.0, cand_h[j]))

        cand_c5 = compute_c5(cand_h)
        d = cand_c5 - cur_c5
        if d <= 0.0 or rng.random() < math.exp(-d / T):
            cur_h = cand_h
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_h = cand_h.copy()
                best_c5 = cand_c5
                if best_c5 <= 0.3808:   # reached target bound
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ----- initialise ---
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        h0 = np.clip(h0, 0.0, 1.0)

    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    target_sum = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----- Phase 0: binary hill‑climb (quick local improvement) -----
    phase0_deadline = min(deadline, start + budget_s * 0.2)
    if time.time() < phase0_deadline:
        binary_h = (best_h > 0.5).astype(int)
        bin_c5 = compute_c5(binary_h)
        binary_h, bin_c5 = hill_climb_binary(binary_h, bin_c5,
                                             time.time(), phase0_deadline, rng)
        if bin_c5 + 1e-12 < best_c5:
            best_h = binary_h.astype(float)
            best_c5 = bin_c5

    # ----- Phase 1: continuous simulated annealing -----
    if time.time() < deadline:
        sa_start = time.time()
        sa_deadline = deadline
        cont_h, cont_c5 = continuous_sa(best_h, best_c5,
                                        sa_start, sa_deadline, rng)
        if cont_c5 + 1e-12 < best_c5:
            best_h = cont_h
            best_c5 = cont_c5

    # ----- final safety: enforce bounds and exact sum -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
