    start = time.time()
    best = v.copy()
    best_score = _compute_c5(v)
    current = v.copy()
    cur_score = best_score
    n = len(v)
    step = init_step
    # We may adjust step based on remaining fraction: exponential decay based on fraction of time elapsed.

    while time.time() - start < time_limit:
        # compute temperature schedule based on fraction elapsed:
        frac = (time.time() - start) / time_limit
        # temperature floor maybe
        T = max(0.001, (1 - frac))
        # step decay
        step = init_step * (1 - frac) + final_step * frac

        # propose move:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2 * step
        cand = current.copy()
        cand[i] += delta

        cand = _repair_vector(cand, n)

        cand_score = _compute_c5(cand)

def _repair_vector(v, n, max_iter=20):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            # need to reduce sum
            mask = v > 0.0
        else:
            # need to increase sum
            mask = v < 1.0
        n_free = np.count_nonzero(mask)
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final fine correction if needed:
    s = v.sum()
    if abs(s - target) > 1e-9:
        # scaling approach
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        # if still small diff, distribute uniformly
        s = v.sum()
        diff = target - s
        if abs(diff) > 1e-9:
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _upsample_vector(v, n_target):
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

import time
import numpy as np

def _repair_vector(v, n, max_iter=20):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = np.count_nonzero(mask)
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        diff = target - s
        if abs(diff) > 1e-9:
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return C5 for vector v using FFT‑based full correlation."""
    n = v.shape[0]
    dx = 2.0 / n
    rev = (1.0 - v)[::-1]
    L = 1 << (int(np.ceil(np.log2(2 * n - 1))))
    fv = np.fft.rfft(v, L)
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fv * fr, L)[: 2 * n - 1]
    return np.max(conv) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation from len(v) to n_target, then repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Uniform random vector repaired to feasibility."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _anneal(v0, time_limit, init_step=0.3, final_step=0.02, T0=0.1):
    """Simulated‑annealing style pairwise exchange."""
    start = time.time()
    n = len(v0)
    best = v0.copy()
    best_score = _compute_c5(best)
    cur = best.copy()
    cur_score = best_score
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit if time_limit > 0 else 0.0
        T = T0 * (1 - frac) + 1e-6 * frac
        step = init_step * (1 - frac) + final_step * frac
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < np.exp(-(cand_score - cur_score) / T) if T > 0 else False
        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
    return best, best_score
