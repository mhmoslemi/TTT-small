    best_h = None

        # reheating schedule

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper constraints ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ---------- Core utilities ----------
def max_corr(h):
    """Maximum raw overlap count (integer/float)."""
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def c5_value(h):
    """C5 = max_corr * dx."""
    n = len(h)
    return max_corr(h) * (2.0 / n)

def hill_climb_binary(b_start, start_c5, start_time, budget_s):
    """Swap hill‑climbing on a binary vector."""
    n = len(b_start)
    dx = 2.0 / n
    best_h = b_start.copy()
    best_val = start_c5
    rng = np.random.default_rng()
    while time.time() - start_time < budget_s - 0.5:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        val = max_corr(cand) * dx
        if val + 1e-12 < best_val:
            best_h = cand
            best_val = val
    return best_h, best_val

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise ----------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    # enforce exact sum n/2 while staying within [0,1]
    target_sum = n / 2.0
    cur_sum = np.sum(h0)
    if abs(cur_sum - target_sum) > 1e-12:
        h0 = np.clip(h0 + (target_sum - cur_sum) / n, 0.0, 1.0)
        diff = target_sum - np.sum(h0)
        if abs(diff) > 1e-12:
            # simple residual fix
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and h0[i] < 1.0:
                    inc = min(diff, 1.0 - h0[i])
                    h0[i] += inc
                    diff -= inc
                elif diff < 0 and h0[i] > 0.0:
                    dec = min(-diff, h0[i])
                    h0[i] -= dec
                    diff += dec

    best_h = h0.copy()
    best_score = c5_value(best_h)

    # ----- build SLSQP problem -----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # ----- phase 1: many SLSQP restarts -----
    slsqp_deadline = start_time + budget_s * 0.65
    while time.time() < slsqp_deadline:
        # random start, occasionally jitter around best
        if best_h is not None and rng.random() < 0.2:
            h_start = best_h.copy()
            perturb = rng.normal(scale=0.05, size=n)
            h_start = np.clip(h_start + perturb, 0.0, 1.0)
            # re‑project to exact sum
            cur = np.sum(h_start)
            h_start += (target_sum - cur) / n
            h_start = np.clip(h_start, 0.0, 1.0)
            # tiny residual correction
            cur = np.sum(h_start)
            diff = target_sum - cur
            if abs(diff) > 1e-12:
                idx = np.where((diff > 0) & (h_start < 1))[0]
                for i in idx:
                    inc = min(diff, 1 - h_start[i])
                    h_start[i] += inc
                    diff -= inc
                    if diff <= 0:
                        break
                idx = np.where((diff < 0) & (h_start > 0))[0]
                for i in idx:
                    dec = min(-diff, h_start[i])
                    h_start[i] -= dec
                    diff += dec
                    if diff >= 0:
                        break
        else:
            h_start = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h_start[idx] = 1.0

        t_start = max_corr(h_start) * (2.0 / n)
        x0 = np.concatenate([h_start, [t_start]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
            )
        except Exception:
            continue

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # exact sum enforcement
        cur = np.sum(h_opt)
        if abs(cur - target_sum) > 1e-9:
            h_opt += (target_sum - cur) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)

        score_opt = c5_value(h_opt)
        if score_opt + 1e-12 < best_score:
            best_h = h_opt.copy()
            best_score = score_opt
            if best_score < 0.3806:
                break  # early stop

    # ----- phase 2: continuous simulated annealing refinement -----
    refine_deadline = start_time + budget_s * 0.93
    while time.time() < refine_deadline:
        # propose a mass swap between two coordinates
        i, j = rng.choice(n, size=2, replace=False)
        # direction: move mass from i to j or opposite
        if rng.random() < 0.5:
            max_delta = min(best_h[i], 1.0 - best_h[j])
            if max_delta <= 1e-12:
                continue
            delta = rng.random() * max_delta
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            max_delta = min(best_h[j], 1.0 - best_h[i])
            if max_delta <= 1e-12:
                continue
            delta = rng.random() * max_delta
            h_new = best_h.copy()
            h_new[i] += delta
            h_new[j] -= delta

        new_score = c5_value(h_new)
        if new_score + 1e-12 < best_score:
            best_h = h_new
            best_score = new_score

    # ----- phase 3: binary hill climb from best continuous solution -----
    if time.time() < start_time + budget_s - 2.0:
        # create binary vector by picking largest entries
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argsort(-best_h)[: n // 2]
        binary_h[top_idx] = 1
        bin_score = c5_value(binary_h.astype(float))
        binary_h, bin_score = hill_climb_binary(binary_h, bin_score, start_time, budget_s)
        if bin_score + 1e-12 < best_score:
            best_h = binary_h.astype(float)
            best_score = bin_score

    # final safety clip and exact sum
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = np.sum(best_h)
    if abs(cur - target_sum) > 1e-9:
        best_h += (target_sum - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h.astype(float), best_score, n
