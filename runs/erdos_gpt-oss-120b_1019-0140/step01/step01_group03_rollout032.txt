def projected_subgradient_descent(h0, n_iter, rng):
    h = h0.copy()
    n = h.shape[0]; s = n/2.0
    best = h.copy()
    best_score = compute_c5(best)
    v = np.zeros_like(h)
    beta = 0.9
    step0=0.9
    for it in range(1, n_iter+1):
        # compute subgradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find indices close to max
        # tolerance absolute
        tol = 1e-8 * max_corr
        idxs = np.where(corr >= max_corr - tol)[0]
        grad = np.zeros_like(h)
        center = n - 1
        for idx in idxs:
            shift = idx - center
            grad += compute_gradient_for_shift(h, shift)
        grad /= len(idxs)
        # momentum
        v = beta * v + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * v
        # add a small random perturbation occasionally
        if rng.random() < 0.01:  # 1% chance per iteration
            # random pair random mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            # ensure i != j
            if i != j:
                # choose delta
                max_transfer = min(h_new[i], 1 - h_new[j])
                delta = max_transfer * rng.random() * 0.1
                h_new[i] -= delta
                h_new[j] += delta
        # projection
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best = h_new.copy()
        h = h_new
    return best, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate a 1‑D array to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    f_low = f(lam_low)
    f_high = f(lam_high)

    # Ensure brackets: f(lam_low) >= s >= f(lam_high) (f decreasing)
    for _ in range(20):
        if f_low >= s:
            break
        lam_low -= (hi - lo) * 10
        f_low = f(lam_low)
    for _ in range(20):
        if f_high <= s:
            break
        lam_high += (hi - lo) * 10
        f_high = f(lam_high)

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C₅ = max_k ∫ h(x)(1 - h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient(h, shift):
    """Gradient of the overlap term for a given integer shift."""
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2


def compute_subgradient(h):
    """Average gradient over all shifts attaining (or nearly attaining) the maximum."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    tol = 1e-12 + 1e-8 * max_corr
    idxs = np.where(corr >= max_corr - tol)[0]
    grad = np.zeros_like(h)
    center = n - 1
    for idx in idxs:
        shift = idx - center
        grad += compute_gradient(h, shift)
    grad /= len(idxs)
    return grad


def projected_subgradient_descent(h0, n_iter, rng):
    """Subgradient descent with momentum, occasional random perturbations, and feasibility projection."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    v = np.zeros_like(h)          # momentum term
    beta = 0.9                     # momentum decay
    step0 = 0.9                    # initial step size

    for it in range(1, n_iter + 1):
        grad = compute_subgradient(h)

        v = beta * v + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * v

        # small random mass transfer to encourage exploration
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_transfer = min(h_new[i], 1.0 - h_new[j])
                delta = max_transfer * rng.random() * 0.1
                h_new[i] -= delta
                h_new[j] += delta

        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score

    it = int(base_iters * (base_n / n) ** 2)
