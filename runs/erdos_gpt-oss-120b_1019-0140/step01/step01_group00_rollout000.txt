def compute_c5(binary):
    n = len(binary)
    dx = 2.0 / n
    # correlation of binary and 1-binary
    c = np.correlate(binary, 1 - binary, mode='full')
    return np.max(c) * dx

def steepest_descent(h, start_c5):
    n = len(h)
    cur_h = h.copy()
    cur_c5 = start_c5
    while True:
        best_improvement = 0.0
        best_i = -1; best_j = -1
        # list indices
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        # iterate
        for i in ones:
            for j in zeros:
                # swap
                cur_h[i] = 0
                cur_h[j] = 1
                new_c5 = compute_c5(cur_h)
                cur_h[i] = 1
                cur_h[j] = 0
                improvement = cur_c5 - new_c5
                if improvement > best_improvement + 1e-12:
                    best_improvement = improvement
                    best_i = i; best_j = j
        if best_improvement > 1e-12:
            # perform swap
            cur_h[best_i] = 0
            cur_h[best_j] = 1
            cur_c5 -= best_improvement
        else:
            break
    return cur_h, cur_c5
```

import time
import numpy as np

def compute_c5(h):
    """Return max overlap integral C5 for binary or float step function h."""
    n = len(h)
    dx = 2.0 / n
    # correlation h with (1-h); full mode gives all shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def steepest_descent(h, cur_c5):
    """Perform steepest-descent swaps (1↔0) until no improvement."""
    n = len(h)
    # Pre‑compute indices of ones and zeros each iteration
    while True:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        best_impr = 0.0
        best_i = -1
        best_j = -1
        best_val = cur_c5
        # Try every possible swap
        for i in ones:
            for j in zeros:
                # swap temporarily
                h[i] = 0
                h[j] = 1
                new_c5 = compute_c5(h)
                # revert
                h[i] = 1
                h[j] = 0
                impr = cur_c5 - new_c5
                if impr > best_impr + 1e-13:
                    best_impr = impr
                    best_i = i
                    best_j = j
                    best_val = new_c5
        if best_impr > 0.0:
            # apply best swap
            h[best_i] = 0
            h[best_j] = 1
            cur_c5 = best_val
        else:
            break
    return h, cur_c5


def run(seed=42, budget_s=1100, **kwargs):

        n = 80
        h0 = np.zeros(n, dtype=int)
        h0[rng.choice(n, n // 2, replace=False)] = 1

    # Initial local optimisation
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_h, best_c5 = steepest_descent(best_h, best_c5)

    # Main stochastic search with random perturbations and re‑optimisation
    while time.time() - start_time < budget_s - 0.5:
        # Choose a seed vector: either current best (perturb) or a fresh random one
        if rng.random() < 0.7:
            cur = best_h.copy()
        else:
            cur = np.zeros(n, dtype=int)
            cur[rng.choice(n, n // 2, replace=False)] = 1

        # Apply a small random perturbation (several random swaps)
        n_swaps = rng.integers(1, 6)  # 1‑5 random swaps
        for _ in range(n_swaps):
            ones = np.flatnonzero(cur == 1)
            zeros = np.flatnonzero(cur == 0)
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur[i] = 0
            cur[j] = 1

        cur_c5 = compute_c5(cur)
        cur, cur_c5 = steepest_descent(cur, cur_c5)

        if cur_c5 + 1e-13 < best_c5:
            best_h, best_c5 = cur, cur_c5
            if best_c5 < 0.3808:
                break

    # Return float version of the binary solution
    return best_h.astype(float), best_c5, n
