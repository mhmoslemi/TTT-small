def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()

        h_init = np.array(initial_h_values, dtype=float)
        n = h_init.shape[0]
    else:
        n = 100
        h_init = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h_init[idx] = 1.0

    h_init = (h_init >= 0.5).astype(float)
    target_sum = n / 2.0
    cur_sum = h_init.sum()
    if cur_sum != target_sum:
        diff = int(target_sum - cur_sum)
        if diff > 0:
            zeros = np.flatnonzero(h_init == 0)[:diff]
            h_init[zeros] = 1.0
        elif diff < 0:
            ones = np.flatnonzero(h_init == 1)[: -diff]
            h_init[ones] = 0.0

    # correlation / C5 helper
    def compute_c5_and_corr(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = corr.max() * (2.0 / h.shape[0])
        return c5_val, corr

    while time.time() - start < budget_s - 0.1:
