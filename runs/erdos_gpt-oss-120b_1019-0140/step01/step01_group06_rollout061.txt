def hill_climb(h, n, budget_remain):
    h_current = h.copy()
    c5_current = compute_c5(h_current)
    # Precompute maybe? Not needed.
    start = time.time()
    while time.time() - start < budget_remain:
        # choose two indices i,j randomly
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # propose transfer of delta from i to j constrained by bounds
        # compute max delta that can be moved: d_max = min(h_current[i], 1 - h_current[j])
        # choose random delta within [-d_max, d_max] and also ensure we keep sum constant.
        # Actually moving delta from i to j decreases h_i by delta, increase h_j by delta.
        # delta can be positive (move from i to j) or negative (vice versa)
        max_transfer = min(h_current[i], 1 - h_current[j])
        max_transfer_neg = min(h_current[j], 1 - h_current[i])
        # Choose direction randomly
        if rng.random() < 0.5:
            delta = rng.random() * max_transfer
            new_h = h_current.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            delta = rng.random() * max_transfer_neg
            new_h = h_current.copy()
            new_h[i] += delta
            new_h[j] -= delta
        # compute c5
        c5_new = compute_c5(new_h)
        if c5_new < c5_current:
            h_current = new_h
            c5_current = c5_new
        # maybe also accept with probability for simulated annealing.
    return h_current, c5_current

import time, math
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(v, target_sum, tol=1e-9, max_iter=20):
    # Clip to box
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(v)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # Find indices not at bounds
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            # No interior points; distribute onto all and reclip
            delta = diff / v.size
            v += delta
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final correction if still off
    cur = np.sum(v)
    if abs(cur - target_sum) > tol:
        # adjust one element with most slack
        if cur < target_sum:
            i = np.argmax(1.0 - v)  # room to increase
            v[i] = min(1.0, v[i] + (target_sum - cur))
        else:
            i = np.argmax(v)  # room to decrease
            v[i] = max(0.0, v[i] - (cur - target_sum))
    return v

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n/2.0}
    bnds = [(0.0, 1.0)] * n

        res = opt.minimize(fun=lambda x: compute_c5(x),
                           x0=h0,
                           method="SLSQP",
                           bounds=bnds,
                           constraints=cons,
                           options={"maxiter": maxiter, "ftol": 1e-9, "disp": False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n/2.0)
    return h_opt, compute_c5(h_opt)

def de_optimize(n, block_size, budget_seconds, seed):
    start = time.time()
    m = n // block_size
    target_block_sum = n / (2.0 * block_size)
    bounds = [(0.0, 1.0)] * m

    def obj(x):
        xb = project_to_feasible(np.clip(x, 0.0, 1.0), target_block_sum)
        h = np.repeat(xb, block_size)
        h = project_to_feasible(h, n/2.0)
        return compute_c5(h)

    def callback(xk, convergence):
        return (time.time() - start) > budget_seconds

    # Choose modest population and iteration numbers; they can be overridden by time limit.
    result = opt.differential_evolution(
        obj,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=6,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=False,
        init='latinhypercube',
        updating='deferred',
        callback=callback,
    )
    xb_best = project_to_feasible(np.clip(result.x, 0.0, 1.0), target_block_sum)
    h_best = np.repeat(xb_best, block_size)
    h_best = project_to_feasible(h_best, n/2.0)
    return h_best, compute_c5(h_best)

def hill_climb(h, n, rng, max_steps=2000):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(max_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Determine feasible delta range
        max_inc = min(1.0 - h_cur[j], h_cur[i])
        max_dec = min(1.0 - h_cur[i], h_cur[j])
        if rng.random() < 0.5:
            delta = rng.random() * max_inc
            h_new = h_cur.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            delta = rng.random() * max_dec
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
    return h_cur, c5_cur

    best_c5 = float("inf")

        if base_h is not None and base_h.shape[0] == n:

        elif base_h is not None:

    best_h = project_to_feasible(best_h, best_n/2.0)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Overlap bound for a sampled step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v, target_sum, tol=1e-9, max_iter=20):
    """Clip to [0,1] and adjust to hit the required sum."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(v)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            delta = diff / v.size
            v += delta
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target_sum) > tol:
        if cur < target_sum:
            i = np.argmax(1.0 - v)
            v[i] = min(1.0, v[i] + (target_sum - cur))
        else:
            i = np.argmax(v)
            v[i] = max(0.0, v[i] - (cur - target_sum))
    return v


def resample_h(base_h, n_target):
    """Linear interpolation from a base array to a new length, then project."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def local_optimize(h0, n, maxiter=400):
    """Local refinement with SLSQP."""
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            fun=lambda x: compute_c5(x),
            x0=h0,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return h0, compute_c5(h0)
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def hill_climb(h, n, rng, max_steps=2000):
    """Simple pairwise mass‑transfer hill climbing."""

    for _ in range(max_steps):

        c5_new = compute_c5(h_new)
