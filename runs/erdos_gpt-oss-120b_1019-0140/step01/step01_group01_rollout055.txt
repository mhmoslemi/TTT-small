def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.1 # safety

    n = 200
    target_sum = n / 2.0

    # init h
    try:
        parent = np.array(initial_h_values, dtype=float)
        # upsample to n points
        x_parent = np.linspace(0, 2, parent.size, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_parent, parent)
        # adjust sum
        h = adjust_sum_to_target(h, target_sum, rng)
    except Exception:
        # random init
        h = rng.rand(n)
        # scale to sum target
        h = h / h.sum() * target_sum
        h = np.clip(h, 0, 1)
        h = adjust_sum_to_target(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    current_h = h.copy()
    current_c5 = best_c5

    # parameters
    T0 = 1e-2
    step0 = 0.3

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        T = T0 * (1 - progress) + 1e-6 * progress
        step = step0 * (1 - progress) + 1e-4 * progress

        if rng.rand() < 0.3:
            # gradient move
            corr_full = np.correlate(current_h, 1.0 - current_h, mode='full')
            k_idx = int(np.argmax(corr_full))
            shift = k_idx - (n - 1)
            g = np.zeros_like(current_h)
            # i + shift
            idx = np.arange(n)
            plus = idx + shift
            valid = (plus >= 0) & (plus < n)
            g[valid] += 1.0 - current_h[plus[valid]]
            minus = idx - shift
            valid = (minus >= 0) & (minus < n)
            g[valid] -= current_h[minus[valid]]
            # project to zero-sum
            g -= g.mean()
            candidate = current_h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        else:
            # pairwise exchange
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = rng.uniform(low, high)
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        cand_c5 = compute_c5(candidate)
        # acceptance
        if cand_c5 < current_c5 or (T > 0 and rng.rand() < np.exp(-(cand_c5 - current_c5) / T)):
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                if best_c5 <= 0.3808:
                    # early exit if target achieved
                    break

    # final adjust sum and recompute C5
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximation for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Slightly modify h (in‑place) so that its sum equals target while staying in [0,1].
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # greedy random correction
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
    if abs(diff) > 1e-12:
        # distribute remaining tiny error uniformly and clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Simulated‑annealing style optimizer for the Erdős C₅ constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.1  # tiny safety margin

    # ----- problem size -------------------------------------------------
    n = 200                     # number of discretisation points (even)
    target_sum = n / 2.0        # required integral = 1
    dx = 2.0 / n

    # ----- initialise ----------------------------------------------------
    try:
        parent = np.array(initial_h_values, dtype=float)
        # upsample parent to n points via linear interpolation
        x_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_parent, parent)
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum_to_target(h, target_sum, rng)
    except Exception:
        # random start respecting the sum constraint
        h = rng.rand(n)
        h = h / h.sum() * target_sum
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum_to_target(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # ----- annealing parameters -----------------------------------------
    T0 = 1e-2               # initial temperature
    step0 = 0.3             # initial step magnitude for gradient moves

    # ----- optimisation loop --------------------------------------------
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        # linear cooling
        T = T0 * (1.0 - progress) + 1e-6 * progress
        step = step0 * (1.0 - progress) + 1e-4 * progress

        if rng.rand() < 0.3:
            # ----- gradient‑based move -----
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            shift = k_idx - (n - 1)       # integer lag (may be negative)

            g = np.zeros_like(current_h)
            idx = np.arange(n)

            # contribution of (1‑h[i+shift])
            plus = idx + shift
            valid = (plus >= 0) & (plus < n)
            g[valid] += 1.0 - current_h[plus[valid]]

            # contribution of -h[i‑shift]
            minus = idx - shift
            valid = (minus >= 0) & (minus < n)
            g[valid] -= current_h[minus[valid]]

            # keep total mass unchanged (project onto hyperplane sum=0)
            g -= g.mean()

            candidate = current_h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        else:
            # ----- pairwise exchange move -----
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = rng.uniform(low, high)   # amount transferred from i to j
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        cand_c5 = compute_c5(candidate)

        # acceptance test (Metropolis)
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0.0 else 0.0
            accept = rng.rand() < prob

            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                if best_c5 <= 0.3808:
                    # good enough – early termination
                    break

    # final sanity adjustment
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n)
