def greedy_optimize(h, max_iter=5000, step=0.01, tol=1e-6, time_budget=None, start_time=None):
    n = len(h)
    best_val = compute_c5(h)
    best_h = h.copy()
    for iter in range(max_iter):
        if time_budget and time.time() - start_time > time_budget * 0.97:
            break
        # Compute correlation and find shift with max value
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr_full.max() * (2.0 / n)
        # Early exit if no improvement beyond tolerance
        if c5 < best_val - tol:
            best_val = c5
            best_h = h.copy()
        # locate shift
        idx_max = corr_full.argmax()
        s = idx_max - (n - 1)
        # mask overlapping indices
        idx = np.arange(n)
        mask = (idx + s >= 0) & (idx + s < n)
        i_vals = idx[mask]
        j_vals = i_vals + s
        # compute contributions
        contrib = h[i_vals] * (1.0 - h[j_vals])
        # find the index with max contribution
        max_contrib_idx = np.argmax(contrib)
        i = i_vals[max_contrib_idx]
        j = j_vals[max_contrib_idx]
        # determine delta
        max_delta = min(step, h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            # cannot move, break
            break
        # move mass
        h[i] -= max_delta
        h[j] += max_delta
        # optional: occasional random perturbation
        # We'll add random jitter every 100 iterations to escape local minima
        if (iter+1) % 100 == 0:
            # random pair
            i_rand = rng.integers(0, n)
            j_rand = rng.integers(0, n)
            # adjust a small random delta
            delta = (rng.random() - 0.5) * step*5 # maybe plus/minus
            if delta > 0:
                # move from i_rand to j_rand
                delta = min(delta, h[i_rand], 1.0 - h[j_rand])
                h[i_rand] -= delta
                h[j_rand] += delta
            else:
                delta = -delta
                delta = min(delta, h[j_rand], 1.0 - h[i_rand])
                h[j_rand] -= delta
                h[i_rand] += delta

    return best_h, best_val

    step = 0.02 # initial delta for transferring mass
    for iter in range(max_iter):
        # compute correlation and s_max

        c5 = corr.max() * (2.0 / n)
        if c5 + 1e-12 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Determine s_max
        idx_max = corr.argmax()
        s = idx_max - (n - 1)
        # Get mask overlapping i: those i where i+s in [0,n-1]
        i_vals = np.where((np.arange(n) + s >= 0) & (np.arange(n) + s < n))[0]
        j_vals = i_vals + s
        contrib = h[i_vals] * (1.0 - h[j_vals])
        # Find index with highest contribution
        i_max_idx = np.argmax(contrib)
        i = i_vals[i_max_idx]
        j = j_vals[i_max_idx]
        # Transfer from i to j
        delta = min(step, h[i], 1.0 - h[j])
        if delta > 0:
            h[i] -= delta
            h[j] += delta
        else:
            # If cannot transfer, perhaps random perturb
            i_rand = rng.integers(0, n)
            j_rand = rng.integers(0, n)
            # random small delta
            delta_rand = rng.random() * step
            # direction random
            if rng.random() < 0.5:
                delta_eff = min(delta_rand, h[i_rand], 1.0 - h[j_rand])
                h[i_rand] -= delta_eff
                h[j_rand] += delta_eff
            else:
                delta_eff = min(delta_rand, h[j_rand], 1.0 - h[i_rand])
                h[j_rand] -= delta_eff
                h[i_rand] += delta_eff
        # occasionally adapt step size
        if (iter+1) % 500 == 0:
            step = step * 0.5  # reduce step for finer adjustments
            if step < 1e-5:
                step = 1e-5

def greedy_optimize(h0, max_iter, rng, time_budget, start_time):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.03  # initial delta
    for it in range(max_iter):
        if time.time() - start_time > time_budget * 0.95:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * (2.0 / n)
        # track best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        idx_max = corr.argmax()
        s = idx_max - (n - 1)
        # overlapping indices mask
        idx = np.arange(n)
        mask = (idx + s >= 0) & (idx + s < n)
        i_vals = idx[mask]
        j_vals = i_vals + s
        # contributions
        contrib = h[i_vals] * (1.0 - h[j_vals])
        # pick i max
        max_idx = np.argmax(contrib)
        i = i_vals[max_idx]
        j = j_vals[max_idx]
        # compute delta
        delta = min(step, h[i], 1.0 - h[j])
        if delta > 1e-12:
            h[i] -= delta
            h[j] += delta
        else:
            # random small move
            i_rand = rng.integers(0, n)
            j_rand = rng.integers(0, n)
            delta_rand = rng.random() * step
            if rng.random() < 0.5:
                delta2 = min(delta_rand, h[i_rand], 1.0 - h[j_rand])
                if delta2 > 0:
                    h[i_rand] -= delta2
                    h[j_rand] += delta2
            else:
                delta2 = min(delta_rand, h[j_rand], 1.0 - h[i_rand])
                if delta2 > 0:
                    h[j_rand] -= delta2
                    h[i_rand] += delta2

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Overlap constant C5 for sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v, target_sum, tol=1e-9, max_iter=15):
    """
    Orthogonal projection of vector v onto the intersection of
    box [0,1]^n and the affine hyperplane sum(v)=target_sum.
    Simple iterative correction of the sum using free (non‑boundary) entries.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(v)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # tiny final correction if needed
    cur = np.sum(v)
    if abs(cur - target_sum) > 1e-6:
        # adjust the most interior component
        i = np.argmin(v) if cur < target_sum else np.argmax(v)
        v[i] = np.clip(v[i] + (target_sum - cur), 0.0, 1.0)
    return v


def resample_h(base_h, n_target):
    """Linear interpolation of base_h to length n_target with feasibility projection."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

                    h[i] += delta_rev
                else:
                    # fallback random transfer
                    i_r = rng.integers(0, n)
                    j_r = rng.integers(0, n)
                    delta_r = rng.random() * step
                    if rng.random() < 0.5:
                        delta_r = min(delta_r, h[i_r], 1.0 - h[j_r])
                        if delta_r > 0:
                            h[i_r] -= delta_r
                            h[j_r] += delta_r
                    else:
                        delta_r = min(delta_r, h[j_r], 1.0 - h[i_r])
                        if delta_r > 0:
                            h[j_r] -= delta_r
                            h[i_r] += delta_r

        if (it + 1) % 500 == 0:
            step = max(step * 0.5, 1e-5)

    # binary rounding (only meaningful for even n)
    if n % 2 == 0:
        order = np.argsort(-h)
        binary = np.zeros_like(h)
        binary[order[: n // 2]] = 1.0
        c5_bin = compute_c5(binary)
        if c5_bin < best_c5:
            best_h = binary
            best_c5 = c5_bin

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5


def binary_hill_climb(h_bin, max_swaps, rng, time_budget, start_time):
    """
    Simple hill‑climbing on a binary vector (exactly n/2 ones) by swapping
    a 1 with a 0 whenever it reduces C5.
    """
    n = h_bin.shape[0]
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

    ones = np.where(best_h == 1)[0]
    zeros = np.where(best_h == 0)[0]

    for _ in range(max_swaps):
        if time.time() - start_time > time_budget * 0.97:
            break
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best_h = cand
            best_c5 = cand_c5
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    parent = globals().get("initial_h_values")
    if parent is not None:
        parent = np.asarray(parent, dtype=float)

    # candidate resolutions (small -> larger)
    ns = [95, 190, 380]
    best_h = None
    best_c5 = float("inf")
    best_n = None

    for n in ns:
        # decide number of restarts
        if n <= 95:
            max_restarts = 80
            max_iter_greedy = 1200
            max_swaps_hill = 2000
        elif n <= 190:
            max_restarts = 30
            max_iter_greedy = 1800
            max_swaps_hill = 3000
        else:
            max_restarts = 15
            max_iter_greedy = 2500
            max_swaps_hill = 4000

        # base vector for this resolution
        if parent is not None and len(parent) == n:
            base_h = parent.copy()
        elif best_h is not None and best_n is not None:
            base_h = resample_h(best_h, n)
        else:
            # random feasible start
            rnd = rng.random(n)
            base_h = project_to_feasible(rnd, n / 2.0)

        for restart in range(max_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break

            if restart == 0:
                h0 = base_h.copy()
            else:
                noise = rng.normal(scale=0.07, size=n)
                h0 = base_h + noise
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_to_feasible(h0, n / 2.0)

            # Equality constraint for sum(h) = n/2
            cons = {"type": "eq", "fun": lambda h, nd=n: np.sum(h) - nd / 2.0}
            bnds = [(0.0, 1.0)] * n

            try:
                res = opt.minimize(
                    fun=lambda h: compute_c5(h),
                    x0=h0,
                    method="SLSQP",
                    bounds=bnds,
                    constraints=cons,
                    options={"maxiter": 800, "ftol": 1e-9, "disp": False},
                )
            except Exception:
                continue

            if not res.success:
                # still try to use the result
                pass

            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_to_feasible(h_opt, n / 2.0)
            c5_val = compute_c5(h_opt)

            # if this is already promising, launch greedy refinement
            if c5_val < best_c5:
                # greedy local search
                h_ref, c5_ref = greedy_optimize(
                    h_opt,
                    max_iter=max_iter_greedy,
                    rng=rng,
                    time_budget=budget_s,
                    start_time=start_time,
                )
                # choose better of raw and refined
                if c5_ref < c5_val:
                    h_opt, c5_val = h_ref, c5_ref

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

        # optional extra refinement on the best found for this n
        if best_h is not None and best_n == n:
            # a final greedy pass
            h_ref, c5_ref = greedy_optimize(
                best_h,
                max_iter=max_iter_greedy,
                rng=rng,
                time_budget=budget_s,
                start_time=start_time,
            )
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n

            # binary hill‑climbing if n is even
            if n % 2 == 0:
                # produce binary version via rounding
                order = np.argsort(-best_h)
                binary = np.zeros_like(best_h)
                binary[order[: n // 2]] = 1.0
                c5_bin = compute_c5(binary)
                if c5_bin < best_c5:
                    best_h, best_c5 = binary_hill_climb(
                        binary,
                        max_swaps=max_swaps_hill,
                        rng=rng,
                        time_budget=budget_s,
                        start_time=start_time,
                    )
                else:
                    # still try hill climb from current best if it is binary already
                    if np.all((best_h == 0.0) | (best_h == 1.0)):
                        best_h, best_c5 = binary_hill_climb(
                            best_h,
                            max_swaps=max_swaps_hill,
                            rng=rng,
                            time_budget=budget_s,
                            start_time=start_time,
                        )

    # final safety projection and evaluation
    if best_h is not None:
        best_h = project_to_feasible(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        # fallback uniform half
        best_n = 95
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
