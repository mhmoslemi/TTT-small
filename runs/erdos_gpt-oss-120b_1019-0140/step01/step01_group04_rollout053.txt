def _simulated_annealing(v, n, time_limit):
    start = time.time()
    best = v.copy()
    best_score = _compute_c5(best)
    cur = best.copy()
    cur_score = best_score
    # schedule: temperature decays exponentially over time
    while time.time() - start < time_limit:
        # progress fraction
        t = (time.time() - start) / time_limit
        temp = 0.5 * (1 - t) + 0.01 * t  # start high, end low (maybe 0.5 to 0.01)
        # step size related to temperature
        step = 0.2 * temp  # when temp 0.5, step 0.1
        # propose random change to a random index i (or two indices to preserve sum)
        i = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * 2 * step
        cand = cur.copy()
        cand[i] += delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        delta_score = cand_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / max(temp, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()

def _anneal(v0, n, time_limit):
    start = time.time()
    cur = v0.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    # parameters
    T0 = 0.5
    Tf = 1e-4
    # steps: define max_step = 0.2
    max_step = 0.2
    while time.time() - start < time_limit:
        t = (time.time() - start) / time_limit
        temp = T0 * (Tf/T0) ** t  # exponential decay
        step = max_step * (temp / T0)  # step proportionally decays
        # propose
        # choose random indices count 2 or 3
        cnt = 2 if np.random.rand() < 0.7 else 3
        idx = np.random.choice(n, cnt, replace=False)
        delta = (np.random.rand(cnt) - 0.5) * 2 * step
        cand = cur.copy()
        cand[idx] += delta
        # repair
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        diff = cand_score - cur_score
        if diff < 0 or np.random.rand() < math.exp(-diff / (temp + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
    return best, best_score

import time

def _repair_vector(v, n, max_iter=15):
    """Clip to [0,1] and adjust sum to n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return C5 for vector v (dx = 2/n)."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

    best_score = _compute_c5(best_v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_v, best_score = cand, score
    return best_v, best_score

def _anneal(v0, n, time_limit):
    """Simulated annealing with adaptive step size."""
    start = time.time()
    cur = v0.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    T0, Tf = 0.5, 1e-4
    max_step = 0.2
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = T0 * (Tf / T0) ** frac
        step = max_step * (temp / T0)
        cnt = 2 if np.random.rand() < 0.7 else 3
        idx = np.random.choice(n, cnt, replace=False)
        delta = (np.random.rand(cnt) - 0.5) * 2.0 * step
        cand = cur.copy()
        cand[idx] += delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        diff = cand_score - cur_score
        if diff < 0 or np.random.rand() < math.exp(-diff / (temp + 1e-12)):
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function for the Erdős C5 problem."""
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------ #
    # Acquire initial construction if provided.
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 200
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0 // 2] = 1.0
    parent = _repair_vector(parent, n0)

    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    # Determine resolution schedule (doubling while ≤ 800)
    max_n = 800
    resolutions = [n0]
    while resolutions[-1] * 2 <= max_n:
        resolutions.append(resolutions[-1] * 2)

    # Allocate time (reserve a couple seconds for final steps)
    total_budget = max(0.0, budget_s - 2.0)
    per_res_budget = total_budget / len(resolutions) if resolutions else 0.0

    for n in resolutions:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:
            break
        # Prepare initial guess at target resolution
        if best_n != n:
            init_guess = _upsample_vector(best_v, n)
        else:
            init_guess = best_v.copy()
        # Time allowed for this resolution
        remaining = budget_s - elapsed - 0.5
        time_for_res = min(per_res_budget, remaining)
        # Split: 70% for main optimisation, 30% for refinement
        main_time = time_for_res * 0.7
        refine_time = time_for_res * 0.3

        if n <= 400:
            v_opt, sc_opt = _optimize_slsqp(init_guess, n, main_time)
        else:
            v_opt, sc_opt = _anneal(init_guess, n, main_time)

        if refine_time > 0.1:
            v_ref, sc_ref = _exchange_hill_climb(v_opt, n, refine_time)
            if sc_ref < sc_opt:
                v_opt, sc_opt = v_ref, sc_ref

        if sc_opt < best_score:
            best_v, best_score, best_n = v_opt, sc_opt, n

    # Final safety repair and evaluation
    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(best_n)
