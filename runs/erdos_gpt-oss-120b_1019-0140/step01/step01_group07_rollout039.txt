def _smooth_c5(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # For logsumexp, compute stable:
    exp_terms = np.exp((corr - max_corr) / eps)
    sum_exp = np.sum(exp_terms)
    smooth = dx * (max_corr + eps * np.log(sum_exp))
    return smooth

def _smooth_c5_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_terms = np.exp((corr - max_corr) / eps)
    sum_exp = np.sum(exp_terms)
    w = exp_terms / sum_exp  # weights sum to 1

    # compute symmetric weight
    w_rev = w[::-1]
    w_sym = w + w_rev
    # convolution w_sym * h
    conv = np.convolve(w_sym, h, mode='full')
    # extract s: indices from (2*n - 2) to (2*n - 2 + n)
    start = 2 * n - 2
    s = conv[start:start + n]
    grad = dx * (1.0 - s)
    return grad

import numpy as np
import time

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    # if already good, return
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _smooth_c5_grad(h, eps):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # stable softmax
    exp_terms = np.exp((corr - max_corr) / eps)
    w = exp_terms / np.sum(exp_terms)
    w_rev = w[::-1]
    w_sym = w + w_rev
    conv = np.convolve(h, w_sym, mode='full')
    start = n - 1
    s = conv[start:start + n]
    grad = dx * (1.0 - s)
    return grad

def _interpolate_h(old_h, n_new):
    n_old = old_h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, old_h)

    import math

import numpy as np
import time

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _smooth_c5_grad(h, eps):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_terms = np.exp((corr - m) / eps)
    w = exp_terms / np.sum(exp_terms)
    w_sym = w + w[::-1]
    conv = np.convolve(h, w_sym, mode="full")
    start = n - 1
    s = conv[start:start + n]
    return dx * (1.0 - s)

def _interpolate_h(old_h, n_new):
    n_old = old_h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, old_h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    init = globals().get("initial_h_values", None)
    if init is not None:
        h0 = np.array(init, dtype=np.float64).ravel()
    else:
        h0 = np.random.rand(380)

    if h0.size % 2 != 0:
        h0 = np.append(h0, 0.0)
    n0 = h0.size
    target0 = n0 / 2.0
    h0 = _project_to_constraints(h0, target0)

    best_h = h0.copy()
    best_score = _compute_c5(best_h)

    n_vals = [n0]
    if budget_s >= 300:
        n_vals.append(n0 * 2)
    if budget_s >= 600:
        n_vals.append(n0 * 4)

    for n in n_vals:
        if n != best_h.size:
            best_h = _interpolate_h(best_h, n)
        target = n / 2.0
        best_h = _project_to_constraints(best_h, target)

        max_restarts = 5
        max_iters = 1500
        eps_schedule = [5.0, 2.0, 1.0, 0.5]

        for _ in range(max_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
            h = best_h + np.random.randn(n) * 0.05
            h = _project_to_constraints(h, target)
            cur_score = _compute_c5(h)

            for eps in eps_schedule:
                lr = 0.5
                for _ in range(max_iters):
                    if time.time() - start_time > budget_s * 0.95:

                    grad = _smooth_c5_grad(h, eps)

                    step_factor = 1.0
                    accepted = False
                    while step_factor > 1e-4:

                            break
