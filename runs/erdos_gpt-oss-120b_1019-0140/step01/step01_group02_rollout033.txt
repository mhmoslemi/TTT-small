def _project_onto_box_sum(v, target):
    # Project v onto [0,1] hypercube with sum target.
    # Find lambda via binary search.
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # due to numeric error, adjust small diff by tweaking one element
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        # adjust first non-boundary element
        idxs = np.where((proj > 0) & (proj < 1))[0]
        if idxs.size > 0:
            i = idxs[0]
            proj[i] = np.clip(proj[i] + diff, 0.0, 1.0)
        else:
            # adjust any element with tolerance: clip distributed
            proj += diff / len(proj)
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

import time
import numpy as np

def _project_onto_box_sum(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)/2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        # adjust a non-boundary entry if exists, otherwise distribute
        free = np.where((proj > 0) & (proj < 1))[0]
        if free.size:
            i = free[0]
            proj[i] = np.clip(proj[i] + diff, 0.0, 1.0)
        else:
            proj += diff / len(proj)
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def _gradient(h, w, dx):
    n = h.shape[0]
    g = np.zeros_like(h)
    w = w * dx
    # iterate over lag
    for L in range(-(n-1), n):
        wk = w[L + n - 1]
        if wk == 0.0:
            continue
        if L >= 0:
            end = n - L
            g[:end] += wk * (1.0 - h[L:])
            g[L:] -= wk * h[:end]
        else:
            k = -L
            start = k
            end = n - k
            g[start:] += wk * (1.0 - h[:end])
            g[:end] -= wk * h[start:]
    return g
```

import time
import numpy as np

def _project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while enforcing sum(v)=target."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        free = np.where((proj > 0) & (proj < 1))[0]
        if free.size:
            i = free[0]
            proj[i] = np.clip(proj[i] + diff, 0.0, 1.0)
        else:
            proj += diff / proj.size
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def _gradient(h: np.ndarray, w: np.ndarray, dx: float) -> np.ndarray:
    """Gradient of the smoothed max‑C₅ objective with respect to h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    w = w * dx
    for L in range(-(n - 1), n):
        wk = w[L + n - 1]
        if wk == 0.0:
            continue
        if L >= 0:
            end = n - L
            g[:end] += wk * (1.0 - h[L:])
            g[L:] -= wk * h[:end]
        else:
            k = -L
            end = n - k
            g[k:] += wk * (1.0 - h[:end])
            g[:end] -= wk * h[k:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 150                     # discretisation points (even for convenience)
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = None
    best_c5 = float('inf')

    # optional use of provided construction if it matches our n
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.shape[0] == n:
            cand = _project_onto_box_sum(np.clip(init, 0.0, 1.0), target_sum)
            c = float(np.max(np.correlate(cand, 1.0 - cand, mode='full') * dx))
            if c < best_c5:
                best_c5, best_h = c, cand.copy()

    # main optimisation loop with random restarts
    while True:
        now = time.time()
        if now - start_time >= budget_s - 0.2:
            break

        # start from a random feasible point
        h = _project_onto_box_sum(rng.random(n), target_sum)

        # inner projected‑gradient descent
        max_steps = 2000
        for step in range(max_steps):
            now = time.time()
            if now - start_time >= budget_s - 0.2:
                break

            raw = np.correlate(h, 1.0 - h, mode='full')
            C = raw * dx
            cur = float(np.max(C))
            if cur < best_c5:
                best_c5, best_h = cur, h.copy()

            elapsed = now - start_time
            frac = elapsed / budget_s
            eta = 3.0 * (1.0 - frac) + 0.3 * frac          # step size schedule
            alpha = 5.0 * (1.0 - frac) + 150.0 * frac      # smooth‑max temperature

            maxC = np.max(C)
            w = np.exp(alpha * (C - maxC))
            w /= w.sum()

            g = _gradient(h, w, dx)
            h_new = h - eta * g
            h = _project_onto_box_sum(h_new, target_sum)

            # occasional jitter to escape plateaus
            if step % 50 == 0 and step > 0:
                scale = 0.08 * (1.0 - frac) + 0.001 * frac
                h = _project_onto_box_sum(np.clip(h + rng.normal(scale=scale, size=n), 0.0, 1.0), target_sum)

    # fallback if nothing was found
    if best_h is None:
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        if n % 2:
            best_h[n // 2] = 0.5
        best_h = _project_onto_box_sum(best_h, target_sum)

    # final evaluation
    best_c5 = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full') * dx))
    return best_h, best_c5, n
