def hillclimb_binary(h0, n_iter, rng):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    for it in range(n_iter):
        # pick random swap positions
        idx1 = rng.choice(np.where(h==1)[0])
        idx2 = rng.choice(np.where(h==0)[0])
        # swap
        h[idx1] = 0.0
        h[idx2] = 1.0
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[idx1] = 1.0
            h[idx2] = 0.0
    return best_h, best_score

import time
import math
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate a 1‑D array to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    # trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the root is bracketed
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10.0
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10.0

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the active max‑correlation term."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def projected_gradient_descent(h0, n_iter=2000, step0=0.5, rng=None):
    """PGD on the max‑correlation objective with optional noise."""
    if rng is None:
        rng = np.random.default_rng()
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        # active shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        # occasional tiny random jitter to escape plateaus
        if it % 500 == 0:
            h_new += (rng.random(n) - 0.5) * 0.02
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (n must be a power of two)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Random binary vector of length n with exactly n/2 ones (n even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def hillclimb_binary(h0, n_iter=2000, rng=None):
    """Simple greedy hill‑climbing on a balanced binary vector."""
    if rng is None:
        rng = np.random.default_rng()
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    for _ in range(n_iter):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        i_idx = ones[i]
        j_idx = zeros[j]
        # swap
        h[i_idx] = 0.0
        h[j_idx] = 1.0
        new_score = compute_c5(h)
        if new_score <= cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i_idx] = 1.0
            h[j_idx] = 0.0
    return best_h, best_score

    # Random fractional starts
    for _ in range(4):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, half_n))

    # ------------------------------------------------------------------
    # 2. Optimize each candidate (PGD + binary hill‑climb refinement)
    # ------------------------------------------------------------------
    for base_h in candidates:
        if time.time() - start_time > budget_s - 5.0:
            break
        # Projected gradient descent (heavy)
        h_opt, score_opt = projected_gradient_descent(base_h, n_iter=8000, step0=0.8, rng=rng)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

        # Convert to a balanced binary vector (top half values)
        idx = np.argsort(-h_opt)[:half_n]
        binary_h = np.zeros(n, dtype=float)
        binary_h[idx] = 1.0

        # Greedy binary hill‑climbing
        h_bin_opt, score_bin = hillclimb_binary(binary_h, n_iter=3000, rng=rng)

        # Refine the binary result with PGD (allows fractional improvement)
        h_refined, score_refined = projected_gradient_descent(h_bin_opt, n_iter=3000, step0=0.6, rng=rng)
        if score_refined < best_score:
            best_score = score_refined
            best_h = h_refined.copy()

    # ------------------------------------------------------------------
    # 3. Random restarts until budget expires
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 3.0:
        # random start (fractional)
        start_h = rng.random(n)
        start_h = project_onto_box_sum(start_h, half_n)

        h_opt, score_opt = projected_gradient_descent(start_h, n_iter=4000, step0=0.8, rng=rng)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

        # binary round‑up and hill‑climb
        idx = np.argsort(-h_opt)[:half_n]
        binary_h = np.zeros(n, dtype=float)
        binary_h[idx] = 1.0
        h_bin_opt, _ = hillclimb_binary(binary_h, n_iter=1500, rng=rng)
        h_refined, score_refined = projected_gradient_descent(h_bin_opt, n_iter=2000, step0=0.6, rng=rng)
        if score_refined < best_score:
            best_score = score_refined
            best_h = h_refined.copy()

        if time.time() - start_time > budget_s - 3.0:
            break

    # ------------------------------------------------------------------
    # 4. Final projection and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, half_n)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, half_n)
        best_score = compute_c5(best_h)

    return best_h, best_score, n
