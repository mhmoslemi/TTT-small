def _optimize_binary(v, time_limit, rng):
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    start = time.time()
    # We'll do simple hill climbing with occasional random restarts
    while time.time() - start < time_limit:
        # pick random swap
        ones = np.flatnonzero(best_v)
        zeros = np.flatnonzero(best_v == 0)
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Swap
        cand = best_v.copy()
        cand[i], cand[j] = cand[j], cand[i]
        score = _compute_c5(cand)
        # Accept if improvement
        if score < best_score:
            best_v, best_score = cand, score
        else:
            # with small probability accept (annealing)
            # compute temperature schedule
            elapsed = time.time() - start
            t_frac = elapsed / time_limit
            # temperature decays linearly from 0.1 to 0
            temp = max(0.01, 0.1 * (1 - t_frac))
            if rng.random() < np.exp(-(score - best_score)/temp):
                best_v, best_score = cand, score
    return best_v, best_score

def _upsample_binary(v, target_n):
    n = len(v)
    factor = target_n // n
    remainder = target_n - n*factor
    new = np.repeat(v, factor)
    if remainder > 0:
        new = np.concatenate([new, v[:remainder]])
    # Ensure sum = target_n/2. Since original sum = n/2, new sum = (n/2)*factor + sum(v[:remainder]).
    # new sum may be off by some amount; we can repair by flipping bits appropriately.
    target_sum = target_n // 2
    cur_sum = new.sum()
    diff = int(target_sum - cur_sum)
    if diff > 0:
        # need to turn diff zeros into ones
        zeros = np.flatnonzero(new == 0)
        if len(zeros) >= diff:
            indices = np.random.choice(zeros, diff, replace=False)
            new[indices] = 1
        else:
            # fallback: set random positions?
            # Not likely.
            new[:diff] = 1
    elif diff < 0:
        # need to turn -diff ones into zeros
        ones = np.flatnonzero(new == 1)
        indices = np.random.choice(ones, -diff, replace=False)
        new[indices] = 0
    return new.astype(np.int8)

import time
import numpy as np

# ---------------- helper functions ---------------- #

def _compute_c5(v):
    """Return C5 for a (binary or continuous) vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return float(np.max(corr) * dx)


def _binary_from_continuous(arr):
    """Convert a real vector to a binary one with exactly half ones."""
    n = len(arr)
    m = n // 2
    idx = np.argpartition(arr, -m)[-m:]
    b = np.zeros(n, dtype=np.int8)
    b[idx] = 1
    return b


def _upsample_binary(v, target_n, rng):
    """Upsample binary vector v to length target_n by repetition and repair."""
    n = len(v)
    factor = target_n // n
    remainder = target_n - factor * n
    new = np.repeat(v, factor)
    if remainder:
        new = np.concatenate([new, v[:remainder]])
    # repair exact half‑ones constraint
    target_sum = target_n // 2
    cur_sum = int(new.sum())
    diff = target_sum - cur_sum
    if diff > 0:
        zeros = np.flatnonzero(new == 0)
        if len(zeros) >= diff:
            pick = rng.choice(zeros, diff, replace=False)
            new[pick] = 1
    elif diff < 0:
        ones = np.flatnonzero(new == 1)
        pick = rng.choice(ones, -diff, replace=False)
        new[pick] = 0
    return new.astype(np.int8)


def _targeted_hill_climb(v, time_limit, rng):
    """Hill‑climb focusing on the worst shift."""
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    start = time.time()
    while time.time() - start < time_limit:
        # full correlation (integer counts)
        corr = np.correlate(best_v, 1 - best_v, mode="full")
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)          # shift (could be negative)
        # build index sets for this shift
        if s >= 0:
            rng_i = np.arange(n - s)
            vi = best_v[rng_i]
            vj = best_v[rng_i + s]
        else:
            s = -s
            rng_i = np.arange(s, n)
            vi = best_v[rng_i]
            vj = best_v[rng_i - s]
        bad_one = rng_i[(vi == 1) & (vj == 0)]
        bad_zero = rng_i[(vi == 0) & (vj == 1)]
        improved = False
        if bad_one.size and bad_zero.size:
            for _ in range(5):
                i = rng.choice(bad_one)
                j = rng.choice(bad_zero)
                cand = best_v.copy()
                cand[i], cand[j] = cand[j], cand[i]
                sc = _compute_c5(cand)
                if sc < best_score:
                    best_v, best_score = cand, sc
                    improved = True
                    break
        if improved:
            continue
        # generic random swap if targeted step fails
        ones = np.flatnonzero(best_v)
        zeros = np.flatnonzero(best_v == 0)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_v.copy()
        cand[i], cand[j] = cand[j], cand[i]
        sc = _compute_c5(cand)
        if sc < best_score:
            best_v, best_score = cand, sc
    return best_v, best_score


def _optimize_binary(v, time_limit, rng):
    """Repair (if needed) and run targeted hill‑climb."""
    n = len(v)
    target = n // 2
    cur = int(v.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.flatnonzero(v == 0)
            pick = rng.choice(zeros, diff, replace=False)
            v[pick] = 1
        else:
            ones = np.flatnonzero(v == 1)
            pick = rng.choice(ones, -diff, replace=False)
            v[pick] = 0
    return _targeted_hill_climb(v, time_limit, rng)


def _search_binary(n, total_time, rng, init_v=None):
    """Multiple restarts of binary optimisation within total_time."""
    start = time.time()
    best_v = None
    best_score = np.inf

    # Primary run with supplied init (if any)
    if init_v is not None:
        alloc = max(0.2, total_time * 0.6)
        v_opt, sc_opt = _optimize_binary(init_v.copy(), alloc, rng)
        best_v, best_score = v_opt, sc_opt

    # Random restarts
    while time.time() - start < total_time:
        remaining = total_time - (time.time() - start)
        if remaining < 0.1:
            break
        per = min(0.5, remaining)
        rnd = np.zeros(n, dtype=np.int8)

        v_opt, sc_opt = _optimize_binary(rnd, per, rng)
        if sc_opt < best_score:
