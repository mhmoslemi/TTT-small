    fv = np.fft.rfft(v, n=m)

    # correlation (full) = ifft(fft(v) * conj(fft(1 - v))) but with appropriate reversal
    # Actually convolution of v and reverse of (1 - v) equals correlation.
    # To get correlation via FFT, we can reverse one vector.
    # But we can compute convolution directly using np.fft.ifft(np.fft.fft(v, m) * np.fft.fft((1 - v)[::-1], m)).

def repair_vector(v):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    target = n/2.0
    # adjust sum by scaling
    current = v.sum()
    if current == 0:
        v[:] = 0.0
    else:
        # We will adjust uniformly:
        v = v * (target / current)
        v = np.clip(v, 0.0, 1.0)
    # If sum still off due to clipping, we adjust residual
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        # distribute diff uniformly across free entries
        # free entries: those not at bounds (0 or 1)
        free = (v > 0) & (v < 1)
        n_free = free.sum()
        if n_free > 0:
            v[free] += diff / n_free
            v = np.clip(v, 0.0, 1.0)
        # if still diff, adjust all entries slightly
        s = v.sum()
        diff = target - s
        if abs(diff) > 1e-9:
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _repair_vector(v):
    """Clip to [0,1] and enforce sum(v)=len(v)/2."""
    n = v.shape[0]
    target = n * 0.5
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if s == 0.0:
        # nothing to scale, just distribute uniformly
        v[:] = target / n
        return v
    # Scale uniformly
    v = v * (target / s)
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        # Distribute diff over non‑boundary entries if possible
        free = (v > 0.0) & (v < 1.0)
        n_free = free.sum()
        if n_free > 0:
            v[free] += diff / n_free
            v = np.clip(v, 0.0, 1.0)
        else:
            # fallback: spread over all entries
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """C5 for a vector v using FFT‑based linear convolution."""
    n = v.shape[0]
    # Linear convolution length 2n‑1 via zero‑padded FFT of length 2n
    m = 2 * n
    rev = (1.0 - v)[::-1]
    fv = np.fft.fft(v, m)
    fr = np.fft.fft(rev, m)
    conv = np.fft.ifft(fv * fr).real
    corr = conv[: 2 * n - 1]
    dx = 2.0 / n
    return np.max(corr) * dx

            # propose swap
            v_cur[i] = 0.0
            v_cur[j] = 1.0
            new_score = _compute_c5(v_cur)

            T = T0 * (alpha ** it)
            accept = False
            if new_score < cur_score:
                accept = True
            elif T > 1e-12:
                prob = np.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_score = new_score
                # update index sets
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                if cur_score < best_bin_score:
                    best_bin_score = cur_score
                    best_bin = v_cur.copy()
            else:
                # revert
                v_cur[i] = 1.0
                v_cur[j] = 0.0
            it += 1

        # ----- continuous local refinement ----------------------------- #
        v_cont = best_bin.copy()
        cont_score = best_bin_score
        step = 0.12
        local_start = time.time()
        while time.time() - local_start < budget_res * 0.25:
            zeros_idx = np.where(v_cont < 0.5)[0]
            ones_idx = np.where(v_cont > 0.5)[0]
            if zeros_idx.size == 0 or ones_idx.size == 0:
                break
            i = rng.choice(zeros_idx)
            j = rng.choice(ones_idx)
            eps = rng.random() * step
            cand = v_cont.copy()
            cand[i] = eps
            cand[j] = 1.0 - eps
            cand = _repair_vector(cand)
            cand_score = _compute_c5(cand)
            if cand_score < cont_score:
                v_cont = cand
                cont_score = cand_score

        # ----- brief SLSQP polish -------------------------------------- #
        remaining = budget_res - (time.time() - sa_start)
        if remaining > 1.0:
            try:
                cons = {"type": "eq", "fun": lambda x: np.sum(x) - n * 0.5}
                bounds = [(0.0, 1.0)] * n
                res = minimize(
                    lambda x: _compute_c5(x),
                    v_cont,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 120, "ftol": 1e-6, "disp": False},
                )
                if res.success:
                    v_opt = _repair_vector(res.x)
                    opt_score = _compute_c5(v_opt)
                    if opt_score < cont_score:
                        v_cont = v_opt
                        cont_score = opt_score
            except Exception:
                pass

        # ----- update global best -------------------------------------- #
        if cont_score < best_global_score:
            best_global_score = cont_score
            best_global_v = v_cont.copy()
            best_global_n = n

    # ------------------------------------------------------------------ #
    # Final safety repair and scoring
    if best_global_v is None:
        # Should not happen, but provide a simple fallback
        n = n0
        best_global_v = np.zeros(n, dtype=np.float64)
        best_global_v[: n // 2] = 1.0
        best_global_n = n
        best_global_score = _compute_c5(best_global_v)

    best_global_v = _repair_vector(best_global_v)
    best_global_score = _compute_c5(best_global_v)

    return best_global_v.astype(np.float64), float(best_global_score), int(best_global_n)
