def local_refine(h0, n, rng, start_time, budget_s):
    # run SLSQP given time left
    remaining = budget_s - (time.time() - start_time)
    if remaining < 0.1:
        return h0, compute_c5(h0)
    # set maxiter based on remaining: maybe 200
    maxiter = 400
    # constraints
    cons = {"type":"eq","fun":lambda x: np.sum(x) - n/2}
    bnds = [(0.0,1.0)]*n
    try:
        res = opt.minimize(fun=lambda x: compute_c5(x),
                           x0=h0,
                           method='SLSQP',
                           bounds=bnds,
                           constraints=cons,
                           options={"maxiter":maxiter, "ftol":1e-9, "disp":False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n/2)
    return h_opt, compute_c5(h_opt)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # add/subtract to the element with most capacity
        if cur < target_sum:
            idx = np.argmax(1.0 - h)  # most room upward
            h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
        else:
            idx = np.argmin(h)  # most room downward
            h[idx] = np.clip(h[idx] - (cur - target_sum), 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target/2.0)
    return h_new

def sinusoid_search(n, end_time):
    best_h = None
    best_c5 = float('inf')
    # grid
    A_vals = np.linspace(0.05, 0.5, 10)
    phi_vals = np.linspace(0.0, 2*np.pi, 12, endpoint=False)
    i_arr = np.arange(n)
    for A in A_vals:
        for phi in phi_vals:
            if time.time() > end_time:
                return best_h, best_c5
            h = 0.5 + A * np.cos(2*np.pi * i_arr / n + phi)
            h = project_to_feasible(h, n/2.0)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def hill_climb_binary(h_start, end_time, global_start, rng):
    n = h_start.shape[0]
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    # loop until time exhausted
    while time.time() < end_time and (time.time() - global_start) < (end_time - global_start):
        # pick random 1 index
        i = rng.integers(n)
        if best_h[i] < 0.5:
            continue
        j = rng.integers(n)
        if best_h[j] > 0.5:
            continue
        # swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        c5 = compute_c5(cand)
        if c5 < best_c5 - 1e-12:
            best_h = cand
            best_c5 = c5
            # early acceptance for big improvements
    return best_h, best_c5

def binary_hill_climb(n, end_time, rng, global_start):
    best_h = None
    best_c5 = float('inf')
    # number of restarts based on n
    max_restarts = 15 if n <= 380 else 10
    for restart in range(max_restarts):
        if time.time() >= end_time:
            break
        # random binary vector with half ones
        idx = rng.permutation(n)
        h = np.zeros(n)
        h[idx[:n//2]] = 1.0
        # allocate per restart time slice
        remaining = end_time - time.time()
        # allocate at most 2 seconds per restart
        per_restart_end = time.time() + min(remaining, 2.0)
        h_opt, c5_opt = hill_climb_binary(h, per_restart_end, global_start, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        # early stop if very good
        if best_c5 <= 0.3808:
            break
    return best_h, best_c5

def local_refine(h0, n, global_start, budget_s):
    # run SLSQP if time permits
    if time.time() - global_start > budget_s * 0.95:
        return h0, compute_c5(h0)
    # add small jitter
    rng_local = np.random.default_rng()
    h_start = h0 + rng_local.normal(scale=0.02, size=n)
    h_start = np.clip(h_start, 0.0, 1.0)
    h_start = project_to_feasible(h_start, n/2.0)
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n/2.0}
    bnds = [(0.0, 1.0)] * n
    try:

                           x0=h_start,
                           method="SLSQP",
                           bounds=bnds,
                           constraints=cons,
                           options={"maxiter": 400, "ftol": 1e-9, "disp": False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n/2.0)
    return h_opt, compute_c5(h_opt)

        if h_sin is not None and c5_sin < best_c5:
            best_h, best_c5, best_n = h_sin, c5_sin, n

        # 2) Binary hill climbing
        hill_end = time.time() + min(8.0, (budget_s - (time.time() - start_time)) * 0.2)
        h_bin, c5_bin = binary_hill_climb(n, hill_end, rng, start_time)
        if h_bin is not None and c5_bin < best_c5:
            best_h, best_c5, best_n = h_bin, c5_bin, n

        # 3) Local SLSQP refinement of the current best for this n
        if best_h is not None and best_h.shape[0] == n:
            if time.time() - start_time < budget_s * 0.9:
                h_ref, c5_ref = local_refine(best_h, n, start_time, budget_s)
                if c5_ref < best_c5:
                    best_h, best_c5, best_n = h_ref, c5_ref, n

        # 4) Also refine the sinusoidal candidate if distinct
        if h_sin is not None and h_sin.shape[0] == n:
            if time.time() - start_time < budget_s * 0.9:
                h_ref2, c5_ref2 = local_refine(h_sin, n, start_time, budget_s)
                if c5_ref2 < best_c5:
                    best_h, best_c5, best_n = h_ref2, c5_ref2, n

    # Fallback in case nothing succeeded
    if best_h is None:
        n = 380
        h_f = np.full(n, 0.5)
        best_h = project_to_feasible(h_f, n / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        best_h = project_to_feasible(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
