def _repair_vector(v):
    n = v.size
    target = n / 2.0
    # clip
    v = np.clip(v, 0.0, 1.0)
    # iterative projection
    # use water-filling
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        lower = v <= 0.0 + 1e-12
        upper = v >= 1.0 - 1e-12
        free = ~(lower | upper)
        n_free = free.sum()
        if n_free == 0:
            # no free entries, adjust by scaling
            v *= (target / s) if s != 0 else target / n
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / n_free
        v[free] += delta
        # reclip
        v = np.clip(v, 0.0, 1.0)
    # final clip
    v = np.clip(v, 0.0, 1.0)
    # final: ensure sum close; adjust small rounding error by adding to a non-bound element
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        # add diff to first non-bound element if any, else adjust others
        for idx in range(n):
            if 0 < v[idx] < 1:
                v[idx] += diff
                v[idx] = min(max(v[idx], 0.0), 1.0)
                break
    return v

def _repair_vector(v):
    n = v.shape[0]
    target = n / 2.0
    # Clip
    v = np.clip(v, 0.0, 1.0)

    # iterative water-filling
    for _ in range(10):
        s = v.sum()

        if abs(diff) < 1e-12:
            break

                v *= target / s
                v = np.clip(v, 0.0, 1.0)
