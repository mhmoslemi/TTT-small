import time, numpy as np, math, ...

def compute_c5_binary(h):
    # h is binary int or float 0/1
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def hill_climb_best(start_h, start_val, rng, time_limit):
    h = start_h.copy()
    val = start_val
    n = len(h)
    # precompute indices
    while time.time() < time_limit:
        # compute ones and zeros
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        # sample many random swaps and find best improvement
        trials = 200  # can adjust based on time remaining
        best_swap = None
        best_swap_val = val
        for _ in range(trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            v = compute_c5_binary(h_new)
            if v < best_swap_val - 1e-12:
                best_swap_val = v
                best_swap = (i, j)
        if best_swap is not None:
            i,j = best_swap
            h[i] = 0
            h[j] = 1
            val = best_swap_val
        else:
            # no improvement in this batch, break
            break
    return h, val

def solve_milp(n, time_limit):
    # Build variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    Y = [cp.Variable(n - k, boolean=True) for k in range(1, n)]
    constraints = [cp.sum(h) == n/2]
    # linking constraints
    for k in range(1, n):
        y = Y[k-1]
        m = n - k
        for i in range(m):
            constraints.append(y[i] <= h[i])
            constraints.append(y[i] <= h[i + k])
            constraints.append(y[i] >= h[i] + h[i + k] - 1)
        # pos overlap
        constraints.append(cp.sum(h[:m]) - cp.sum(y) <= t)
        # neg overlap
        constraints.append(cp.sum(h[k:]) - cp.sum(y) <= t)
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    # Solve with CBC, pass time limit if supported
    solver_options = {}
    # Some solvers accept "maxSeconds"
    # Attempt to pass known param
    try:
        prob.solve(solver='CBC', **{'maxSeconds': time_limit})
    except TypeError:
        # fallback without time limit
        prob.solve(solver='CBC')
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        # Could be 'TIME_LIMIT' or others, fallback
        return None, None, None
    h_val = np.rint(h.value).astype(int)  # round to nearest integer
    t_val = t.value
    c5 = t_val * (2.0 / n)
    return h_val, t_val, c5

def hill_climb_best(start_h, start_val, rng, deadline):
    h = start_h.copy()
    val = start_val
    n = len(h)
    while time.time() < deadline:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        # try a batch of random swaps
        batch = 100
        best_swap = None
        best_swap_val = val
        for _ in range(batch):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            new_val = compute_c5_binary(h_new)
            if new_val < best_swap_val - 1e-12:
                best_swap_val = new_val
                best_swap = (i, j)
        if best_swap:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            val = best_swap_val
        else:
            # no improvement after batch, break to avoid infinite loop
            break
    return h, val

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def solve_milp(n):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    Y = [cp.Variable(n - k, boolean=True) for k in range(1, n)]
    constraints = [cp.sum(h) == n/2]
    # linking constraints
    for k in range(1, n):
        y = Y[k-1]
        m = n - k
        for i in range(m):
            constraints.append(y[i] <= h[i])
            constraints.append(y[i] <= h[i + k])
            constraints.append(y[i] >= h[i] + h[i + k] - 1)
        # pos overlap
        constraints.append(cp.sum(h[:m]) - cp.sum(y) <= t)
        # neg overlap
        constraints.append(cp.sum(h[k:]) - cp.sum(y) <= t)
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    # try a list of solvers
    for solver in (cp.CBC, cp.ECOS_BB, cp.GLPK_MI):
        try:
            prob.solve(solver=solver, warm_start=True, verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                break
        except Exception:
            continue
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        raise RuntimeError("MILP did not solve to optimality")
    h_val = np.rint(h.value).astype(int)
    t_val = t.value
    c5 = t_val * (2.0 / n)
    return h_val, c5

def hill_climb_best(start_h, start_val, rng, deadline):
    h = start_h.copy()
    val = start_val
    n = len(h)
    while time.time() < deadline:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        # try a batch of swaps
        batch = 200
        best_swap = None
        best_swap_val = val
        for _ in range(batch):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            new_val = compute_c5(h_new)
            if new_val < best_swap_val - 1e-12:
                best_swap_val = new_val
                best_swap = (i, j)
        if best_swap:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            val = best_swap_val
        else:
            break
    return h, val

def heuristic_search(rng, n, start_time, budget_s):
    deadline = start_time + budget_s - 0.2
    best_h = None
    best_val = float('inf')
    # use initial_h if present
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        idx = np.argpartition(-init_arr, n//2 - 1)[:n//2]
        init_bin = np.zeros(n, dtype=int)
        init_bin[idx] = 1
        val = compute_c5(init_bin)
        best_h, best_val = init_bin, val
    # random restarts
    while time.time() < deadline:
        # random binary vector
        start = np.zeros(n, dtype=int)
        idx = rng.choice(n, n//2, replace=False)
        start[idx] = 1
        start_val = compute_c5(start)
        h_opt, v_opt = hill_climb_best(start, start_val, rng, deadline)
        if v_opt < best_val - 1e-12:
            best_h, best_val = h_opt, v_opt
            # early exit if bound sufficiently low
            if best_val < 0.3805:
                break
    if best_h is None:
        # fallback: simple binary half ones at start
        best_h = np.zeros(n, dtype=int)
        best_h[:n//2] = 1
        best_val = compute_c5(best_h)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80
    best_h = None
    best_val = float('inf')
    # attempt MILP
    try:
        h_milp, c5_milp = solve_milp(n)
        best_h, best_val = h_milp, c5_milp
    except Exception:
        # fallback to heuristic
        best_h, best_val = heuristic_search(rng, n, start_time, budget_s)
    # ensure exact sum constraint (numeric tolerance)
    target_sum = n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = best_h.astype(float)
        best_h += (target_sum - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)
    else:
        best_h = best_h.astype(float)
    # final C5 compute for safety
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def solve_milp(n):
    # binary variables for h
    h = cp.Variable(n, boolean=True)
    # auxiliary variable for the maximal (unscaled) overlap
    t = cp.Variable()
    # auxiliary binary variables for products h[i]*h[i+k]
    Y = [cp.Variable(n - k, boolean=True) for k in range(1, n)]
    cons = [cp.sum(h) == n / 2]

    for k in range(1, n):
        y = Y[k - 1]
        m = n - k
        # linearization of y[i] = h[i] * h[i+k]
        for i in range(m):
            cons.append(y[i] <= h[i])
            cons.append(y[i] <= h[i + k])
            cons.append(y[i] >= h[i] + h[i + k] - 1)
        # overlap for positive shift k
        cons.append(cp.sum(h[:m]) - cp.sum(y) <= t)
        # overlap for negative shift -k
        cons.append(cp.sum(h[k:]) - cp.sum(y) <= t)

    cons.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), cons)

    # try a few MILP solvers; most environments have CBC or ECOS_BB
    for solver in (cp.CBC, cp.ECOS_BB, cp.GLPK_MI):
        try:
            prob.solve(solver=solver, warm_start=True, verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                break
        except Exception:
            continue

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        raise RuntimeError("MILP failed to solve optimally")

    h_val = np.rint(h.value).astype(int)
    t_val = float(t.value)
    c5 = t_val * (2.0 / n)
    return h_val, c5

def hill_climb_best(start_h, start_val, rng, deadline):
    """Greedy batch swap hill‑climbing."""
    h = start_h.copy()
    val = start_val
    n = len(h)
    while time.time() < deadline:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        batch = 200          # swaps examined per iteration
        best_swap = None
        best_swap_val = val
        for _ in range(batch):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            v = compute_c5(h_new)
            if v < best_swap_val - 1e-12:
                best_swap_val = v
                best_swap = (i, j)
        if best_swap:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            val = best_swap_val
        else:
            break
    return h, val

def heuristic_search(rng, n, start_time, budget_s):
    deadline = start_time + budget_s - 0.2
    best_h = None
    best_val = float('inf')

    # If a parent construction exists, start from its binary version
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        idx = np.argpartition(-init, n // 2 - 1)[: n // 2]
        bin_init = np.zeros(n, dtype=int)
        bin_init[idx] = 1
        val = compute_c5(bin_init)
        best_h, best_val = bin_init, val

    while time.time() < deadline:
        start = np.zeros(n, dtype=int)
        idx = rng.choice(n, n // 2, replace=False)
        start[idx] = 1
        start_val = compute_c5(start)
        h_opt, v_opt = hill_climb_best(start, start_val, rng, deadline)
        if v_opt < best_val - 1e-12:
            best_h, best_val = h_opt, v_opt
            if best_val < 0.3805:
                break

    if best_h is None:
        # Very simple fallback
        best_h = np.zeros(n, dtype=int)
        best_h[: n // 2] = 1
        best_val = compute_c5(best_h)

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80

        best_h, best_val = h_milp, c5_milp
    except Exception:
        pass

    if best_h is None or time.time() - start_time < budget_s * 0.9:

    else:

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
