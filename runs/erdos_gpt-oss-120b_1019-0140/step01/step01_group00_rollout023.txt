import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    idx = rng.choice(n, target_sum, replace=False)
    v[idx] = 1

        if excess > 0 and len(ones) > 0:

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a binary step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def generate_random_binary(n, target_sum, rng):
    """Uniform random binary vector with exactly target_sum ones."""
    v = np.zeros(n, dtype=np.int8)
    v[rng.choice(n, target_sum, replace=False)] = 1
    return v

def repair_binary(v, target_sum, rng):
    """Adjust v to have exactly target_sum ones (preserves as many bits as possible)."""
    cur = int(v.sum())
    if cur > target_sum:
        excess = cur - target_sum
        ones = np.flatnonzero(v)
        if excess and ones.size:
            v[rng.choice(ones, size=excess, replace=False)] = 0
    elif cur < target_sum:
        deficit = target_sum - cur
        zeros = np.flatnonzero(v == 0)
        if deficit and zeros.size:
            v[rng.choice(zeros, size=deficit, replace=False)] = 1
    return v

def crossover(p1, p2, rng, target_sum):
    """Uniform crossover with repair to preserve the sum constraint."""
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2).astype(np.int8)
    return repair_binary(child, target_sum, rng)

def mutate_swap(v, rng):
    """Swap a random 1 with a random 0."""
    ones = np.flatnonzero(v)
    zeros = np.flatnonzero(v == 0)
    if ones.size == 0 or zeros.size == 0:
        return v
    i = rng.choice(ones)
    j = rng.choice(zeros)
    v[i], v[j] = v[j], v[i]
    return v

def tournament_select(pop, fitness, tournament_size, rng):
    """Select the best individual among a random subset."""
    idx = rng.choice(len(pop), size=tournament_size, replace=False)
    best = idx[0]
    best_f = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_f:
            best = i
            best_f = fitness[i]
    return pop[best]

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and an optional starter vector.
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        target = n // 2
        idx = np.argsort(-h0)[:target]
        binary0 = np.zeros(n, dtype=np.int8)
        binary0[idx] = 1
    else:
        n = 80
        target = n // 2
        binary0 = None

    # Evolutionary algorithm parameters.
    pop_size = 100
    mutation_rate = 0.30
    tournament_size = 5

    # Initial population.
    population = np.empty((pop_size, n), dtype=np.int8)
    for i in range(pop_size):
        population[i] = generate_random_binary(n, target, rng)
    if binary0 is not None:
        population[0] = binary0.copy()

    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(population[i].astype(float))

    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_val = float(fitness[best_idx])

    # Main GA loop.
    while True:
        if time.time() - start > budget_s - 0.5:
            break

        # Produce offspring.
        children = np.empty_like(population)
        child_fitness = np.empty(pop_size, dtype=float)
        for i in range(pop_size):
            p1 = tournament_select(population, fitness, tournament_size, rng)
            p2 = tournament_select(population, fitness, tournament_size, rng)
            child = crossover(p1, p2, rng, target)
            if rng.random() < mutation_rate:
                child = mutate_swap(child, rng)
            children[i] = child
            child_fitness[i] = compute_c5(child.astype(float))

        # Replace worst individuals (elitist selection).
        combined = np.concatenate((population, children), axis=0)
        combined_fit = np.concatenate((fitness, child_fitness), axis=0)
        order = np.argsort(combined_fit)
        population = combined[order[:pop_size]]
        fitness = combined_fit[order[:pop_size]]

        # Update global best.
        if fitness[0] < best_val - 1e-12:
            best_val = fitness[0]
            best_h = population[0].copy()
            if best_val < 0.3808:   # target reached
                break

    # Greedy swap hill‑climbing from the best GA individual.
    while time.time() - start < budget_s - 0.1:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand.astype(float))
        if cand_val + 1e-12 < best_val:
            best_h = cand
            best_val = cand_val

    # Final sanity check (re‑compute objective).
    best_val = compute_c5(best_h.astype(float))
    return best_h.astype(float), best_val, n
