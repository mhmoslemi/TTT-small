def worst_shift_and_gradient(h):
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    k_max_idx = int(np.argmax(corr_full))
    k_shift = k_max_idx - (len(h)-1)  # could be negative
    n = len(h)
    g = np.zeros_like(h)
    # compute derivative for each i
    # contribution from term i to shift k: derivative of h[i]*(1-h[i+k]) wrt h[i] is (1-h[i+k])
    # derivative wrt h[i] from term (i-k) where shift uses h[i]=h[(i-k)+k] = h[i] appears in term h[i-k]*(1-h[i]), derivative = -h[i-k]
    # So:
    for i in range(n):
        if 0 <= i + k_shift < n:
            g[i] += (1.0 - h[i + k_shift])
        if 0 <= i - k_shift < n:
            g[i] -= h[i - k_shift]
    return k_shift, g
```

def upsample_h(old_h, new_n):
    old_n = len(old_h)
    # generate positions
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    # periodic interpolation using np.interp but with wrap-around: we need periodic bc.
    # Append first point + 2 at end.
    extended_x = np.concatenate([old_x, [old_x[0] + 2]])
    extended_h = np.concatenate([old_h, [old_h[0]]])
    new_h = np.interp(new_x, extended_x, extended_h)
    # Clip (should already be within [0,1])
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # Adjust sum to target
    target_sum = new_n / 2.0
    # Scale to target by adjusting overall mass while maintaining bounds:
    # Use simple scaling then clip and adjust leftover.
    new_h = new_h / new_h.sum() * target_sum
    # Now adjust residual
    new_h = adjust_sum_to_target(new_h, target_sum, np.random.RandomState())
    return new_h

    start = time.time()
    max_time = budget_s - 0.5

    # set n_points: try 140 (maybe 200)
    n_points = 140
    target_sum = n_points / 2.0

    # initialize h
    try:
        init_h = np.array(initial_h_values, dtype=float)
        if init_h.size != n_points:
            h = upsample_h(init_h, n_points)
        else:
            h = init_h.copy()
        # ensure sum correct
        h = adjust_sum_to_target(h, target_sum, rng)
    except Exception:
        # no initial values
        h = rng.rand(n_points)
        h = h / h.sum() * target_sum
        # ensure bounds
        h = np.clip(h, 0.0, 1.0, out=h) # may violate sum, adjust
        h = adjust_sum_to_target(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # parameters for annealing
    T0 = 1e-2  # start temperature
    T_end = 1e-6
    step0 = 0.3
    step_end = 1e-4

    # iteration counter
    it = 0
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        it += 1
        progress = (now - start) / max_time
        # linear schedule
        T = T0 * (1 - progress) + T_end * progress
        step = step0 * (1 - progress) + step_end * progress

        # choose move type
        move_type = rng.rand()
        if move_type < 0.5:
            # pairwise random
            i, j = rng.randint(n_points, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            # feasible delta
            low = max(-hi, hj - 1.0)   # lower bound for delta (decrease i)
            high = min(1.0 - hi, hj)   # upper bound for delta (increase i)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            # clip delta to feasible range
            delta = max(low, min(high, delta))
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta
        elif move_type < 0.8:
            # gradient pair move
            k_shift, grad = worst_shift_and_gradient(current_h)
            # Determine indices for positive and negative gradient
            # Choose random among top few to add stochasticity
            # get sorted indices
            pos_idx = np.argmax(grad)  # largest positive
            neg_idx = np.argmin(grad)  # most negative
            # If gradient signs not opposite, skip
            if grad[pos_idx] <= 0 or grad[neg_idx] >= 0:
                # fallback to random pair move
                continue
            # feasible delta from pos (decrease) to neg (increase)
            delta_max = min(current_h[pos_idx], 1.0 - current_h[neg_idx])
            if delta_max <= 0:
                continue
            delta = min(step, delta_max) * rng.rand()  # random proportion of max
            candidate = current_h.copy()
            candidate[pos_idx] -= delta
            candidate[neg_idx] += delta
        else:
            # global gradient step (projected)
            # compute gradient as earlier (for worst shift)
            k_shift, grad = worst_shift_and_gradient(current_h)
            # scale gradient to zero sum: subtract mean
            direction = -grad
            direction -= direction.mean()
            # normalize direction to unit L2 norm (avoid large steps)
            norm = np.linalg.norm(direction)
            if norm == 0:

            direction = direction / norm
            # propose step

            # clip and then adjust sum to target
            np.clip(candidate, 0.0, 1.0, out=candidate)

        # optional periodic restart if no improvement for many iterations

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Adjust h in‑place so that its sum equals target while respecting 0≤h≤1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def worst_shift_and_gradient(h: np.ndarray):
    """Return the integer shift that maximizes overlap and its gradient vector."""
    n = h.shape[0]
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    k_idx = int(np.argmax(corr_full))
    k_shift = k_idx - (n - 1)                     # can be negative
    g = np.zeros_like(h)
    idx = np.arange(n)

    plus = idx + k_shift
    valid_plus = (plus >= 0) & (plus < n)
    g[valid_plus] += (1.0 - h[plus[valid_plus]])

    minus = idx - k_shift
    valid_minus = (minus >= 0) & (minus < n)
    g[valid_minus] -= h[minus[valid_minus]]

    return k_shift, g


def binary_initial(n: int, rng: np.random.RandomState) -> np.ndarray:
    """Create a random binary vector with exactly n/2 ones."""
    b = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    b[ones] = 1.0
    return b


def upsample_initial(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Periodically interpolate old_h onto a finer grid of size new_n."""
    old_n = old_h.size
    if old_n == new_n:
        return old_h.copy()
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    # periodic wrap‑around for interpolation
    ext_x = np.concatenate([old_x, [old_x[0] + 2]])
    ext_h = np.concatenate([old_h, [old_h[0]]])
    up_h = np.interp(new_x, ext_x, ext_h)
    return up_h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5                     # safety margin

    # ---------- problem size ----------
    n_points = 200                                 # even, gives more flexibility
    target_sum = n_points / 2.0                    # exact mass

    # ---------- initial solution ----------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        up_h = upsample_initial(init_h, n_points)
        # binaryize by taking the largest target_sum entries as 1
        order = np.argsort(-up_h)                 # descending
        binary_h = np.zeros(n_points, dtype=float)
        binary_h[order[:int(target_sum)]] = 1.0
        current_h = binary_h
    except Exception:
        # fall back to a random binary start
        current_h = binary_initial(n_points, rng)

    # sanity: exact mass
    if abs(current_h.sum() - target_sum) > 1e-12:
        current_h = adjust_sum_to_target(current_h, target_sum, rng)

    best_h = current_h.copy()
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5

    # ---------- annealing schedule ----------
    T0 = 0.03                                     # initial temperature
    T_end = 1e-6
    it = 0

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        it += 1
        progress = (now - start) / max_time
        T = T0 * (1.0 - progress) + T_end * progress

        move_type = rng.rand()
        # ---- random 1↔0 swap ----
        if move_type < 0.65:
            ones_idx = np.where(current_h > 0.5)[0]
            zeros_idx = np.where(current_h < 0.5)[0]
            if ones_idx.size == 0 or zeros_idx.size == 0:
                continue
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            candidate = current_h.copy()
            candidate[i] = 0.0
            candidate[j] = 1.0
        # ---- gradient‑guided swap ----
        elif move_type < 0.9:
            k_shift, grad = worst_shift_and_gradient(current_h)
            pos_idx = np.where((current_h > 0.5) & (grad > 0))[0]
            neg_idx = np.where((current_h < 0.5) & (grad < 0))[0]
            if pos_idx.size == 0 or neg_idx.size == 0:
                continue
            i = rng.choice(pos_idx)
            j = rng.choice(neg_idx)
            candidate = current_h.copy()
            candidate[i] = 0.0
            candidate[j] = 1.0
        # ---- occasional global perturbation ----
        else:
            # flip a few random bits
            candidate = current_h.copy()
            flips = rng.randint(2, 6)  # 2‑5 flips
            for _ in range(flips):
                ones_idx = np.where(candidate > 0.5)[0]
                zeros_idx = np.where(candidate < 0.5)[0]
                if ones_idx.size == 0 or zeros_idx.size == 0:
                    break
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                candidate[i] = 0.0
                candidate[j] = 1.0

        cand_c5 = compute_c5(candidate)

        # Metropolis acceptance
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()

        # periodic mild diversification
        if it % 50000 == 0:
            # apply a handful of random swaps to current_h
            for _ in range(4):
                ones_idx = np.where(current_h > 0.5)[0]
                zeros_idx = np.where(current_h < 0.5)[0]
                if ones_idx.size == 0 or zeros_idx.size == 0:
                    break
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                current_h[i] = 0.0
                current_h[j] = 1.0
            current_c5 = compute_c5(current_h)

    # final cleanup: ensure exact mass and binary values
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact count of ones
    ones_needed = int(round(target_sum))
    current_ones = int(best_h.sum() + 0.5)
    if current_ones != ones_needed:
        # re‑binaryize: keep the largest entries
        order = np.argsort(-best_h)
        best_h[:] = 0.0
        best_h[order[:ones_needed]] = 1.0
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
