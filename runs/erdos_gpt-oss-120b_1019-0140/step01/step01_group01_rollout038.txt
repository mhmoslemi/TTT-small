def round_to_target(h, target):
    # target is integer (or half-integer)
    # Round down/up to obtain exactly target sum.
    # Compute sorted indices by descending fractional part (value - floor)
    floor_h = np.floor(h).astype(int)
    # number of ones needed
    need = int(round(target - floor_h.sum()))
    # compute fractional parts
    frac = h - floor_h
    # indices that have higher fractional parts
    idx = np.argsort(-frac)  # descending
    add = np.zeros_like(h, dtype=int)
    add[idx[:need]] = 1
    rounded = floor_h + add
    # Clip to range [0,1]
    rounded = np.clip(rounded, 0, 1)
    return rounded
```

import time
import numpy as np
import cvxpy as cp

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of C5 = max_k ∫ h(x)*(1-h(x+k)) dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Tiny iterative fixer: tweak random entries until sum matches target,
    while staying within [0,1]. Works for modest residuals.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    max_iter = 500
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        idx = rng.randint(h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0:
                h[idx] -= dec
                diff += dec
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.3  # safety margin

    # Determine problem size
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.size
    else:
        n = 70  # a reasonable default
    target_sum = n / 2.0

    # ---- LP formulation ----
    h = cp.Variable(n)
    w = cp.Variable((n, n))
    t = cp.Variable()  # upper bound on overlap integrand

    constraints = [
        cp.sum(h) == target_sum,
        h >= 0,
        h <= 1,
        w >= 0,
        w <= cp.reshape(h, (n, 1)),
        w <= cp.reshape(h, (1, n)),
        w >= cp.reshape(h, (n, 1)) + cp.reshape(h, (1, n)) - 1,
        t >= 0
    ]

    # shift constraints: for each lag k, enforce s_k - g_k <= t  (i.e., g_k >= s_k - t)
    for k in range(-n + 1, n):
        if k >= 0:
            idx = range(n - k)
            gk = sum(w[i, i + k] for i in idx)
            sk = sum(h[i] for i in idx)
        else:
            kneg = -k
            idx = range(n - kneg)
            gk = sum(w[i + kneg, i] for i in idx)
            sk = sum(h[i + kneg] for i in idx)
        constraints.append(gk >= sk - t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8, feastol=1e-8, max_iters=5000)
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            raise RuntimeError("LP not optimal")
        h_opt = np.array(h.value).astype(float)
        np.clip(h_opt, 0.0, 1.0, out=h_opt)
        # enforce exact mass (tiny numerical drift)
        h_opt = adjust_sum_to_target(h_opt, target_sum, rng)
        best_h = h_opt.copy()
        best_c5 = compute_c5(best_h)
    except Exception:
        # Fallback: simple random feasible point
        h_opt = rng.rand(n)
        h_opt = adjust_sum_to_target(h_opt, target_sum, rng)
        best_h = h_opt.copy()
        best_c5 = compute_c5(best_h)

    # ---- Quick greedy refinement (swap a 1 and a 0) ----
    # work with binary rounded version of best_h to keep sum exact
    def binary_round(v):
        floor = np.floor(v)
        need = int(round(target_sum - floor.sum()))
        frac = v - floor
        idx = np.argsort(-frac)  # descending fractional part
        add = np.zeros_like(v)
        add[idx[:need]] = 1
        return np.clip(floor + add, 0, 1)

    # attempt a few thousand random swaps within remaining time
    cur_h = binary_round(best_h)
    cur_c5 = compute_c5(cur_h)
    if cur_c5 < best_c5:
        best_c5, best_h = cur_c5, cur_h.copy()

    ones_idx = lambda arr: np.where(arr > 0.5)[0]
    zeros_idx = lambda arr: np.where(arr < 0.5)[0]

    while time.time() - start < max_time:
        o = ones_idx(cur_h)
        z = zeros_idx(cur_h)
        if o.size == 0 or z.size == 0:
            break
        i = rng.choice(o)
        j = rng.choice(z)
        cand = cur_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_c5 = compute_c5(cand)
        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_c5, best_h = cand_c5, cand.copy()

    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n)
