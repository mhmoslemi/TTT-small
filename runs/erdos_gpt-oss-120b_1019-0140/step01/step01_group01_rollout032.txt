    n = h.shape[0]
    dx = 2.0 / n
    size = 2*n
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1-h, size)

def pairwise_move(h, step, rng):
    n = h.shape[0]
    i, j = rng.randint(n, size=2)
    if i == j:
        return h  # no change
    hi, hj = h[i], h[j]

import numpy as np
import time

def adjust_sum_to_target(h, target, rng):
    """Adjust h in-place so that sum(h) == target while keeping 0<=h<=1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated by FFT correlation."""
    n = h.shape[0]
    v = 1.0 - h
    nfft = 2 * n
    H = np.fft.rfft(h, n=nfft)
    Vrev = np.fft.rfft(v[::-1], n=nfft)
    conv = np.fft.irfft(H * Vrev, n=nfft)
    max_val = conv[:2 * n - 1].max()
    return float(max_val * (2.0 / n))

def worst_shift(h):
    """Find integer shift that gives the maximal correlation value."""
    n = h.shape[0]
    v = 1.0 - h
    nfft = 2 * n
    H = np.fft.rfft(h, n=nfft)
    Vrev = np.fft.rfft(v[::-1], n=nfft)
    conv = np.fft.irfft(H * Vrev, n=nfft)
    conv = conv[:2 * n - 1]
    idx = int(np.argmax(conv))
    return idx - (n - 1)

def gradient_move(h, step, rng):
    """Perform a gradient-like move w.r.t. the worst shift."""
    n = h.shape[0]
    shift = worst_shift(h)
    g = np.zeros_like(h)
    idx = np.arange(n)

    plus = idx + shift
    mask_plus = (plus >= 0) & (plus < n)
    g[mask_plus] += (1.0 - h[plus[mask_plus]])

    minus = idx - shift
    mask_minus = (minus >= 0) & (minus < n)
    g[mask_minus] -= h[minus[mask_minus]]

    g -= g.mean()
    cand = h - step * g
    np.clip(cand, 0.0, 1.0, out=cand)
    cand = adjust_sum_to_target(cand, n / 2.0, rng)
    return cand

    elif delta > high:
        delta = high

    cand[j] = hj - delta
    return cand

            current_c5 = compute_c5(current_h)
            last_improve = now

    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
