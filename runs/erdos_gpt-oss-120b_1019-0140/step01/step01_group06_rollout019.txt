def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)  # correct scaling

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = np.sum(h)
        diff = target_sum - cur_sum
        if abs(diff) <= tol:
            break
        # indices that are strictly inside bounds
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # if no free, adjust all values proportionally (but low risk)
            # scaling approach: multiply all by factor to hit target within bounds
            factor = target_sum / cur_sum if cur_sum != 0 else 0.0
            h = np.clip(h * factor, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final minor fix
    cur_sum = np.sum(h)
    if abs(cur_sum - target_sum) > 1e-6:
        # distribute residual to smallest h (to increase) or largest (to decrease)
        # Choose index with largest slack
        slack_up = 1.0 - h
        slack_down = h
        if cur_sum < target_sum:
            i = np.argmax(slack_up)
            h[i] = min(1.0, h[i] + (target_sum - cur_sum))
        else:
            i = np.argmax(slack_down)
            h[i] = max(0.0, h[i] - (cur_sum - target_sum))
    return h

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    # ensure feasibility
    h_new = project_to_feasible(h_new, n_target/2.0)
    return h_new

import time
import numpy as np


def compute_c5(h):
    """Return the overlap constant C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-9, max_iter=10):
    """
    Project vector h onto the box [0,1]^n and the hyperplane sum(h)=target_sum.
    Simple iterative redistribution of the surplus/deficit among interior entries.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            if cur > 0:
                factor = target_sum / cur
                h = np.clip(h * factor, 0.0, 1.0)
            else:
                h = np.full_like(h, target_sum / h.shape[0])
                h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)

    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        diff = target_sum - cur
        if diff > 0:
            i = np.argmax(1.0 - h)
            h[i] = min(1.0, h[i] + diff)
        else:
            i = np.argmax(h)
            h[i] = max(0.0, h[i] + diff)  # diff is negative
    return h


def resample_h(base_h, n_target):
    """Interpolate a base step function to a new resolution and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    return project_to_feasible(h_new, n_target / 2.0)


def gradient_for_shift(h, shift):
    """
    Gradient of the correlation term corresponding to a given integer shift.
    For shift s, gradient_i = (1 - h[i+s]) - h[i-s] with out‑of‑bounds treated as 0.
    """
    n = h.shape[0]
    i = np.arange(n)
    idx1 = i + shift
    idx2 = i - shift
    term1 = np.where((idx1 >= 0) & (idx1 < n), 1.0 - h[idx1], 0.0)
    term2 = np.where((idx2 >= 0) & (idx2 < n), h[idx2], 0.0)
    return term1 - term2
