def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')

            cur = nxt
            cur_c5 = nxt_c5
            if cur_c5 < best_c5 - 1e-12:
                best = cur.copy()
                best_c5 = cur_c5
    return best, best_c5

def hill_climb_binary(start_h, rng, deadline, max_no_improve=5000):
    """Greedy hill climbing: accept only improving swaps."""
    cur = start_h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    no_imp = 0
    while time.time() < deadline and no_imp < max_no_improve:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < cur_c5:
            cur = cand
            cur_c5 = cand_c5
            no_imp = 0
            if cur_c5 < best_c5 - 1e-12:
                best = cur.copy()
                best_c5 = cur_c5
        else:
            no_imp += 1
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # initialise
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        best_h = binary_from_continuous(initial_h_values)
    else:
        n = 80
        best_h = random_binary(n, rng)

    best_c5 = compute_c5(best_h)

    # main simulated‑annealing phase
    sa_deadline = start_time + budget_s * 0.96
    while time.time() < sa_deadline:
        # occasionally start from current best to refine
        if rng.random() < 0.3:
            start_vec = best_h.copy()
        else:
            start_vec = random_binary(n, rng)

        # allocate steps based on remaining time
        rem = sa_deadline - time.time()
        steps = int(min(40000, max(2000, rem * 25000)))
        local_h, local_c5 = simulated_annealing(start_vec, steps, rng, sa_deadline)
        if local_c5 + 1e-12 < best_c5:
            best_h, best_c5 = local_h, local_c5
            if best_c5 < 0.3808:
                break

    # final greedy refinement
    final_deadline = start_time + budget_s * 0.99
    refined_h, refined_c5 = hill_climb_binary(best_h, rng, final_deadline)
    if refined_c5 + 1e-12 < best_c5:
        best_h, best_c5 = refined_h, refined_c5

    # ensure exact sum (should already hold)
    best_h = best_h.astype(float)
    target = n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        diff = target - cur_sum
        best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    return best_h, best_c5, n
