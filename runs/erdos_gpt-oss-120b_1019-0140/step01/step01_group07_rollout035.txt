import numpy as np
import time

def _compute_c5(h):
    """Return max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and enforce sum(v)=target by water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _max_lag(h):
    """Return lag (with sign) that yields the maximal correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return idx - (n - 1)

def _gradient_at_lag(h, lag):
    """Gradient of the correlation at given lag w.r.t h."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        lo = 0
        hi = n - lag
        g[lo:hi] += 1.0 - h[lo + lag:hi + lag]
        g[lo + lag:hi + lag] -= h[lo:hi]
    else:
        lo = -lag
        hi = n
        g[lo:hi] += 1.0 - h[lo + lag:hi + lag]
        g[lo + lag:hi + lag] -= h[lo:hi]
    return g

def _pgd_optimize(start_h, target_sum, max_iter, step_init, start_time, budget_s):
    """Projected gradient descent on the max-correlation objective."""
    h = start_h.copy()
    best_h = h.copy()
    best_score = _compute_c5(h)
    step = step_init
    n = h.size
    no_improve = 0
    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.99:
            break
        lag = _max_lag(h)
        grad = _gradient_at_lag(h, lag)
        new_h = h - step * grad
        new_h = _project_to_constraints(new_h, target_sum)
        new_score = _compute_c5(new_h)
        if new_score < best_score - 1e-12:
            best_h, best_score = new_h.copy(), new_score
            h = new_h
            step = min(step * 1.1, 1.0)
            no_improve = 0
            if best_score < 0.3808:
                break
        else:
            step *= 0.5
            no_improve += 1
            if step < 1e-6 or no_improve >= 200:
                # random perturbation around the best known point
                noise = np.random.randn(n) * 0.02
                h = _project_to_constraints(best_h + noise, target_sum)
                step = step_init
                no_improve = 0
    return best_h, best_score

def _local_random_search(h, target_sum, max_trials, start_time, budget_s):
    """Simple pairwise transfer search to polish the solution."""
    n = h.size
    best_h = h.copy()
    best_score = _compute_c5(best_h)
    for _ in range(max_trials):
        if time.time() - start_time > budget_s * 0.99:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        max_delta = min(hi, 1.0 - hj)
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * 0.5
        new = best_h.copy()
        new[i] -= delta
        new[j] += delta
        new_score = _compute_c5(new)
        if new_score < best_score:
            best_h, best_score = new, new_score
            if best_score < 0.3808:
                break
    return best_h, best_score

def _interpolate_to_n(h, n_new):
    """Linear interpolation of a step vector to a new resolution."""
    n_old = h.size
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
