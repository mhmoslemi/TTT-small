    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    """
    Project a vector onto the intersection of the box [0,1]^n and the hyperplane
    sum(h)=target_sum. Simple iterative scheme that distributes the deficit
    among interior entries.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h)  # most room to increase
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h
