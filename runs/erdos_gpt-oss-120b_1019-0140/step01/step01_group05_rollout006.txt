def _targeted_move(h, rng, max_move):
  N = len(h)
  corr = np.correlate(h, 1-h, mode='full')
  idx = np.argmax(corr)
  k = idx - (N-1)  # shift
  # Determine valid range
  if k >= 0:
    i_vals = np.arange(0, N - k)
  else:
    k_abs = -k
    i_vals = np.arange(k_abs, N)
  # Compute ability to increase
  h_i = h[i_vals]
  h_j = h[i_vals + k] if k >= 0 else h[i_vals - k]  # careful: i_vals + k (for k negative? but we can compute i_vals + k)
  # but simplest: pair_i = i_vals, pair_j = i_vals + k
  # For negative k, i_vals + k is lower index, must be valid.
  # We ensure i_vals + k within [0,N)
  # Compute sum of pairs and capacity.
  capacity = np.minimum(1 - h_i, 1 - h_j) # max epsilon for each pair
  # Mask where capacity > 0
  mask = capacity > 1e-12
  if not np.any(mask):
    return h, None # no possible move
  # For candidates, compute score = (h_i + h_j) (prefer larger)
  scores = (h_i + h_j) * mask
  # Choose i index probabilistically weighted by scores (or simply max)
  idx_pair = np.argmax(scores)  # selects best pair
  i0 = i_vals[idx_pair]
  j0 = i0 + k
  # Choose decrease index jdec:
  # Options: any index where h > 0 and not i0 or j0? Not required but might improve.
  # We'll sample a few random indices until we find one.
  # We'll attempt up to 5 tries.
  for _ in range(5):
    jdec = rng.integers(0, N)
    if jdec not in (i0, j0) and h[jdec] > 0:
        break
  else:
    # fallback to random any jdec with h>0
    # find any with h>0
    j_candidates = np.where(h > 1e-12)[0]
    if len(j_candidates) == 0:
        return h, None
    jdec = rng.choice(j_candidates)
    if jdec in (i0, j0):
        # maybe we can accept
        pass
  # Determine epsilon
  caps = capacity[idx_pair]
  eps_max = min(max_move, caps, h[jdec] / 2.0)
  if eps_max <= 1e-12:
    return h, None
  eps = rng.random() * eps_max
  # Apply move
  h_new = h.copy()
  h_new[i0] += eps
  h_new[j0] += eps
  h_new[jdec] -= 2 * eps
  # Clip just in case due to floating errors
  np.clip(h_new, 0.0, 1.0, out=h_new)
  return h_new, (i0, j0, jdec, eps, k)

def _random_move(h, rng, max_move):
    N = len(h)
    i, j = rng.integers(0,N, size=2)
    while i == j:
        j = rng.integers(0,N)
    hi, hj = h[i], h[j]
    # allowable epsilon as before
    up = min(1.0 - hi, hj)
    down = min(hi, 1.0 - hj)
    if up == 0.0 and down == 0.0:
        return h.copy()  # no move
    eps = rng.random() * (up + down) - down
    eps = max(-max_move, min(max_move, eps))
    if eps == 0.0:
        return h.copy()
    h_new = h.copy()
    h_new[i] = hi + eps
    h_new[j] = hj - eps
    # after just adjusting, clip (should be within)
    np.clip(h_new, 0.0, 1.0, out=h_new)
    return h_new

def _optimize_h(h, time_limit, rng):
    start = time.time()
    h_cur = _repair_vector(h.copy())
    cur_val = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_val = cur_val
    N = h_cur.size
    # parameters
    max_move0 = 0.2
    # temperature scale similar to SA
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        max_move = max_move0 * frac
        # choose move type
        if rng.random() < 0.6:
            # targeted
            h_new = _targeted_move(h_cur, rng, max_move)
        else:
            # random
            h_new = _random_move(h_cur, rng, max_move)
        # repair
        h_new = _repair_vector(h_new)
        new_val = _c5_value(h_new)
        # accept if improvement
        if new_val < cur_val:
            h_cur = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
        else:
            # accept with a small probability decreasing over time
            # temperature for acceptance
            T0 = 0.0005
            T = T0 * frac
            if T > 0 and rng.random() < math.exp(-(new_val - cur_val) / T):
                h_cur = h_new
                cur_val = new_val
        # optional early exit if cur_val <= target (0.3808?) Could break.
        # continue loop
    return best_h, best_val

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v.fill(target / v.size)
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

import numpy as np
import time
import math

# ---------------------------------------------------------
# Helper utilities
# ---------------------------------------------------------

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and scale to satisfy sum(v) == len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v.fill(target / v.size)
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v


def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral (discrete) for a step function h."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _random_move(h: np.ndarray, rng: np.random.Generator, max_move: float) -> np.ndarray:
    """Simple two‑index exchange move preserving the sum."""
    N = h.size
    i, j = rng.integers(0, N, size=2)
    while i == j:
        j = rng.integers(0, N)
    hi, hj = h[i], h[j]
    up = min(1.0 - hi, hj)          # max we can add to i (and remove from j)
    down = min(hi, 1.0 - hj)        # max we can subtract from i (and add to j)
    if up == 0.0 and down == 0.0:
        return h.copy()
    eps = rng.random() * (up + down) - down
    eps = max(-max_move, min(max_move, eps))
    if abs(eps) < 1e-15:
        return h.copy()
    h_new = h.copy()
    h_new[i] = hi + eps
    h_new[j] = hj - eps
    np.clip(h_new, 0.0, 1.0, out=h_new)
    return h_new


def _targeted_move(h: np.ndarray, rng: np.random.Generator, max_move: float) -> np.ndarray:
    """Move aimed at the shift producing the current maximal C5."""
    N = h.size
    # locate worst shift
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    k = idx - (N - 1)                     # shift value

    # valid index pairs for this shift
    if k >= 0:
        i_vals = np.arange(N - k)
    else:
        i_vals = np.arange(-k, N)
    j_vals = i_vals + k
    h_i = h[i_vals]
    h_j = h[j_vals]

    inc_cap = np.minimum(1.0 - h_i, 1.0 - h_j)   # how much both can increase
    inc_mask = inc_cap > 1e-12
    dec_cap = np.minimum(h_i, h_j)               # how much both can decrease
    dec_mask = dec_cap > 1e-12

    best_inc_idx = -1
    if np.any(inc_mask):
        inc_score = (h_i + h_j) * inc_mask
        best_inc_idx = int(np.argmax(inc_score))

    best_dec_idx = -1
    if np.any(dec_mask):
        dec_score = (h_i + h_j) * dec_mask
        best_dec_idx = int(np.argmax(dec_score))

    possible_inc = best_inc_idx >= 0
    possible_dec = best_dec_idx >= 0
    if not possible_inc and not possible_dec:
        return h.copy()

    if possible_inc and possible_dec:
        direction = 'inc' if rng.random() < 0.7 else 'dec'
    elif possible_inc:
        direction = 'inc'
    else:
        direction = 'dec'

    if direction == 'inc':
        pair_idx = best_inc_idx
        i0 = int(i_vals[pair_idx])
        j0 = int(j_vals[pair_idx])
        if i0 == j0:   # shift zero – adjust a single entry
            cap_i = inc_cap[pair_idx]                # = 1 - h[i0]
            donors = np.where(h > 1e-12)[0]
            donors = donors[donors != i0]
            if donors.size == 0:
                donors = np.where(h > 1e-12)[0]
            donor = int(rng.choice(donors))
            max_eps = min(max_move, cap_i, h[donor])
            if max_eps <= 1e-12:
                return h.copy()
            eps = rng.random() * max_eps
            h_new = h.copy()
            h_new[i0] += eps
            h_new[donor] -= eps
            return h_new
        else:
            cap = inc_cap[pair_idx]
            donors = np.where(h > 1e-12)[0]
            donors = donors[(donors != i0) & (donors != j0)]
            if donors.size == 0:
                donors = np.where(h > 1e-12)[0]
            donor = int(rng.choice(donors))
            max_eps = min(max_move, cap, h[donor] / 2.0)
            if max_eps <= 1e-12:
                return h.copy()
            eps = rng.random() * max_eps
            h_new = h.copy()
            h_new[i0] += eps
            h_new[j0] += eps
            h_new[donor] -= 2 * eps
            return h_new
    else:  # direction == 'dec'
        pair_idx = best_dec_idx
        i0 = int(i_vals[pair_idx])
        j0 = int(j_vals[pair_idx])
        if i0 == j0:
            cap_i = dec_cap[pair_idx]                # = h[i0]
            recs = np.where(h < 1.0 - 1e-12)[0]
            recs = recs[recs != i0]

            rec = int(rng.choice(recs))

            if max_eps <= 1e-12:
                return h.copy()
            eps = rng.random() * max_eps
            h_new = h.copy()
            h_new[i0] -= eps
            h_new[rec] += eps
            return h_new
        else:
            cap = dec_cap[pair_idx]
            recs = np.where(h < 1.0 - 1e-12)[0]
            recs = recs[(recs != i0) & (recs != j0)]

            rec = int(rng.choice(recs))
            max_eps = min(max_move, cap, (1.0 - h[rec]) / 2.0)
            if max_eps <= 1e-12:
                return h.copy()
            eps = rng.random() * max_eps
            h_new = h.copy()
            h_new[i0] -= eps
            h_new[j0] -= eps
            h_new[rec] += 2 * eps
            return h_new


def _optimize_h(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Hybrid SA + targeted moves for a fixed resolution."""
    h_cur = _repair_vector(h_start.copy())
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score

    start = time.time()
    T0 = 0.001          # initial temperature for SA acceptance
    max_move0 = 0.2

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        max_move = max_move0 * frac

        # choose move type
        if rng.random() < 0.6:
            h_new = _targeted_move(h_cur, rng, max_move)
        else:
            h_new = _random_move(h_cur, rng, max_move)

        h_new = _repair_vector(h_new)
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h on [0,2] with a low C5 bound."""
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # -----------------------------------------------------
    # Initial construction
    # -----------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        # fallback: generate a random start of moderate size
        N0 = 150
        h0 = rng.random(N0)
    h0 = _repair_vector(h0)

    # -----------------------------------------------------
    # Phase 1 – optimise at the original resolution
    # -----------------------------------------------------
    time_left = budget_s - (time.time() - start_total)
    if time_left <= 0:
        best_h = h0
        best_val = _c5_value(best_h)
        return best_h, float(best_val), best_h.size

    phase1 = time_left * 0.45
    best_h, best_val = _optimize_h(h0, phase1, rng)

    # -----------------------------------------------------
    # Phase 2 – progressive resolution increases
    # -----------------------------------------------------
    max_n = 800                     # keep vectors reasonably small
    while True:
        elapsed = time.time() - start_total
        time_left = budget_s - elapsed
        if time_left < budget_s * 0.08 or best_h.size >= max_n:
            break
        n_cur = best_h.size
        n_next = min(n_cur * 2, max_n)

        # interpolate to higher resolution
        x_cur = np.linspace(0.0, 2.0, n_cur, endpoint=False)
        x_next = np.linspace(0.0, 2.0, n_next, endpoint=False)
        h_up = np.interp(x_next, x_cur, best_h)
        h_up = _repair_vector(h_up)

        # allocate a slice of the remaining budget
        phase_time = min(time_left * 0.40, 120.0)
        h_opt, val_opt = _optimize_h(h_up, phase_time, rng)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

    # -----------------------------------------------------
    # Phase 3 – random restarts at final resolution (if time permits)
    # -----------------------------------------------------
    elapsed = time.time() - start_total
    time_left = budget_s - elapsed
    if time_left > 5.0:
        restarts = min(int(time_left // 10) + 1, 5)
        per_restart = time_left / restarts
        for _ in range(restarts):
            h_rand = rng.random(best_h.size)
            h_rand = _repair_vector(h_rand)
            h_opt, val_opt = _optimize_h(h_rand, per_restart * 0.9, rng)
            if val_opt < best_val:
                best_h, best_val = h_opt, val_opt
            if (budget_s - (time.time() - start_total)) <= 5.0:
                break

    # final repair and evaluation
    best_h = _repair_vector(best_h)
    best_val = _c5_value(best_h)
    return best_h, float(best_val), best_h.size
