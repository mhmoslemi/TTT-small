def build_h_from_blocks(b, n):
    m = len(b)

    h = np.repeat(b, L)
    remainder = n - len(h)
    if remainder > 0:
        # fill remainder with first values of b
        h = np.concatenate([h, b[:remainder]])
    # ensure length exactly n
    return h[:n]

import time
import math
import numpy as np

# Global placeholders for DE objective (set inside run)
_global_de_n = None          # total number of points
_global_de_L = None          # block length
_global_de_target_sum = None # target sum for block variables

def _compute_c5(h):
    """Maximum overlap integral for discretized h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and enforce sum(v)=target via water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def build_h_from_blocks(b, n):
    """Expand block vector b to length n by repetition (exact division assumed)."""
    m = b.size
    L = n // m
    h = np.repeat(b, L)
    rem = n - h.size
    if rem > 0:
        h = np.concatenate([h, b[:rem]])
    return h[:n]

def simulated_annealing(h_initial, target_sum, time_budget, start_time):
    """Simple SA that respects the sum constraint via projection."""
    n = h_initial.size
    cur_h = h_initial.copy()
    cur_score = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    deadline = start_time + time_budget
    # temperature schedule parameters
    T0 = 0.01
    Tmin = 1e-5

    while True:
        now = time.time()
        if now >= deadline:
            break
        # linear cooling
        prop = (now - start_time) / time_budget
        T = T0 * (1.0 - prop) + Tmin
        sigma = 0.2 * (T / T0)  # noise scale shrinks with temperature

        # propose a new candidate
        noise = np.random.randn(n) * sigma
        new_h = cur_h + noise
        new_h = _project_to_constraints(new_h, target_sum)
        new_score = _compute_c5(new_h)

        if new_score < cur_score:
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
                if best_score < 0.3808:
                    break
        else:
            # Metropolis acceptance
            delta = new_score - cur_score
            prob = math.exp(-delta / (T + 1e-12))
            if np.random.rand() < prob:
                cur_h, cur_score = new_h, new_score

    return best_h, best_score
