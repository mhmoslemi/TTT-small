import numpy as np

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if np.abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # order descending capacity (largest first)
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if np.abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _max_corr_and_shift(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    max_corr = corr[idx] * dx
    k = idx - (n - 1)
    return max_corr, k

def _gradient_subgrad(h, k):
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.full(n, dx, dtype=np.float64)
    if k >= 0:
        if k < n:
            # subtract h_{j+k}
            grad[:n - k] -= dx * h[k:]
            # subtract h_{j-k}
            grad[k:] -= dx * h[:n - k]
    else:
        kp = -k
        if kp < n:
            grad[kp:] -= dx * h[:n - kp]
            grad[:n - kp] -= dx * h[kp:]
    return grad

import numpy as np

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _max_corr_and_shift(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    max_corr = corr[idx] * dx
    k = idx - (n - 1)
    return max_corr, k

def _gradient_subgrad(h, k):
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.full(n, dx, dtype=np.float64)
    if k >= 0:
        if k < n:
            grad[: n - k] -= dx * h[k:]
            grad[k:] -= dx * h[: n - k]
    else:
        kp = -k
        if kp < n:
            grad[kp:] -= dx * h[: n - kp]
            grad[: n - kp] -= dx * h[kp:]
    return grad
