import time, numpy as np, math, scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # adjust using any variable with capacity
            # Or break
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final adjust if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-8:
        # adjust smallest index
        i = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def hill_climb(h0, max_iter, budget_s, start_time, target_sum):
   current = h0.copy()
   cur_c5 = compute_c5(current)
   best = current.copy()
   best_c5 = cur_c5
   n = current.shape[0]
   # parameters
   T0 = 0.02
   step_limit = 0.2
   for it in range(max_iter):
       if time.time() - start_time > budget_s * 0.97:
           break
       # temperature schedule
       T = T0 * (1 - it / max_iter)
       i, j = np.random.randint(0, n, size=2)
       while i == j:
           j = np.random.randint(0, n)
       hi = current[i]; hj = current[j]
       # compute allowable delta
       delta_max_pos = min(1 - hi, hj)
       delta_max_neg = -min(hi, 1 - hj)
       lo = max(delta_max_neg, -step_limit)
       hi_delta = min(delta_max_pos, step_limit)
       if lo >= hi_delta:
           continue
       delta = np.random.uniform(lo, hi_delta)
       if delta == 0:

       new = current.copy()
       new[i] += delta
       new[j] -= delta
       # evaluate

def local_refine(h, target_sum, max_iter=200):
    n = h.shape[0]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bnds = [(0.0,1.0)]*n
    try:
        res = opt.minimize(fun=lambda x: compute_c5(x), x0=h, method='SLSQP', bounds=bnds, constraints=[cons], options={'maxiter': max_iter, 'ftol':1e-9, 'disp':False})

        return x, compute_c5(x)
    x = np.clip(res.x, 0, 1)
    x = project_to_feasible(x, target_sum)
    return x, compute_c5(x)

        if cur < target_sum:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
    return h

    cur_c5 = compute_c5(current)

        hi = current[i]
        hj = current[j]
        delta_pos = min(1.0 - hi, hj)   # max positive delta
        delta_neg = -min(hi, 1.0 - hj)   # max negative delta
        lo = max(delta_neg, -step_limit)
        hi_delta = min(delta_pos, step_limit)
        if lo >= hi_delta:
            continue

        if delta == 0.0:

        new[j] = hj - delta

                best_c5 = cur_c5

    except Exception:

    x = np.clip(res.x, 0.0, 1.0)
    x = project_to_feasible(x, target_sum)
    return x, compute_c5(x)

import time
import math
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Overlap bound C5 for a sampled step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-9, max_iter=10):
    """Project onto [0,1]^n ∩ {sum = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-8:
        if cur < target_sum:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
    return h


def hill_climb(h0, max_iter, start_time, budget_s, target_sum, rng):
    """Simulated‑annealing style hill climbing."""
    n = h0.shape[0]
    cur = h0.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    T0 = 0.05            # initial temperature
    step_limit = 0.2    # max magnitude of a single exchange
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.97:
            break
        T = T0 * (1.0 - it / max_iter)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        hi = cur[i]
        hj = cur[j]
        delta_pos = min(1.0 - hi, hj)    # max positive shift i↑, j↓
        delta_neg = -min(hi, 1.0 - hj)   # max negative shift i↓, j↑
        lo = max(delta_neg, -step_limit)
        hi_delta = min(delta_pos, step_limit)
        if lo >= hi_delta:
            continue
        delta = rng.uniform(lo, hi_delta)
        if delta == 0.0:
            continue
        new = cur.copy()
        new[i] = hi + delta
        new[j] = hj - delta
        new_c5 = compute_c5(new)
        dC = new_c5 - cur_c5
        if dC <= 0.0 or (T > 0.0 and math.exp(-dC / T) > rng.random()):
            cur = new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
        if it % 1000 == 0:
            cur = project_to_feasible(cur, target_sum)
            cur_c5 = compute_c5(cur)
    return best, best_c5

    """Callable for the equality sum constraint."""

        self.target = target
    def __call__(self, x):
        return np.sum(x) - self.target


def local_refine(h, target_sum, max_iter=300):
    """One‑shot SLSQP refinement."""
    n = h.shape[0]
    cons = {"type": "eq", "fun": SumConstraint(target_sum)}
    bnds = [(0.0, 1.0)] * n
    try:

            x0=h,

            constraints=[cons],

        )
    except Exception:

    x = np.clip(res.x, 0.0, 1.0)
    x = project_to_feasible(x, target_sum)

def resample_to(h, n_target):
    """Linear interpolation to a new resolution, then project."""
    n_src = h.shape[0]
    if n_target == n_src:

    h_new = np.interp(xt, xs, h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

            150000,
            start,
            budget_s,
            n0 / 2.0,
            rng,
        )
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Local refinement with a gradient‑based optimizer
    if time.time() - start < budget_s * 0.95:
        refined_h, refined_c5 = local_refine(best_h, n0 / 2.0, max_iter=400)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # Optional upscale to twice the resolution if time permits
    elapsed = time.time() - start
    if elapsed < budget_s * 0.85:
        n_up = n0 * 2
        h_up = resample_to(best_h, n_up)
        remaining = budget_s - (time.time() - start)
        # allocate at most 30 % of the original budget for the finer search
        budget2 = min(remaining, budget_s * 0.30)
        max_iter_up = int(200000 * (budget2 / budget_s))
        h_opt2, c5_opt2 = hill_climb(
            h_up,
            max_iter_up,
            start,
            budget_s,
            n_up / 2.0,
            rng,
        )
        if c5_opt2 < best_c5:
            best_h, best_c5 = h_opt2, c5_opt2
        if time.time() - start < budget_s * 0.95:
            refined_h2, refined_c5_2 = local_refine(best_h, best_h.shape[0] / 2.0, max_iter=300)
            if refined_c5_2 < best_c5:
                best_h, best_c5 = refined_h2, refined_c5_2

    best_h = project_to_feasible(best_h, best_h.shape[0] / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_h.shape[0]
