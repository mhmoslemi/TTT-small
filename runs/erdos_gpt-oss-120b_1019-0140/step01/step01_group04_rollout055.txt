def binary_anneal(v, time_limit, initial_temp=0.1, final_temp=1e-4, steps=100000):
    n = len(v)
    ones = np.where(v==1)[0]
    zeros = np.where(v==0)[0]
    cur_score = compute_c5(v)
    best_score = cur_score
    best_v = v.copy()
    start = time.time()
    iter = 0
    while time.time() - start < time_limit and iter < steps:
        T = max(final_temp, initial_temp * (1 - (time.time() - start) / time_limit))
        # pick random swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        v[i], v[j] = 0, 1
        new_score = compute_c5(v)
        dE = new_score - cur_score
        accept = False
        if dE < 0:
            accept = True
        else:
            accept = np.random.rand() < np.exp(-dE / T)
        if accept:
            cur_score = new_score
            # update lists of indices
            # swap indices in ones and zeros arrays
            # We can maintain boolean mask instead of lists for speed
        else:
            # revert swap
            v[i], v[j] = 1, 0
        if cur_score < best_score:
            best_score = cur_score
            best_v = v.copy()
        iter += 1
    return best_v, best_score
```

import time
import numpy as np
from scipy.optimize import minimize

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Maximum overlap integral for discrete vector v."""
    n = v.shape[0]
    dx = 2.0 / n
    nfft = _next_pow2(2 * n - 1)
    b_rev = (1.0 - v)[::-1]
    V = np.fft.rfft(v, nfft)
    B = np.fft.rfft(b_rev, nfft)
    conv = np.fft.irfft(V * B, nfft)[: 2 * n - 1]
    return np.max(conv) * dx

def _random_feasible(n):
    """Random vector satisfying bounds and sum constraint."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _optimize_at_resolution(n, init_v, time_limit):
    """Run SLSQP from init_v (and random restarts) within time_limit."""
    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)

    def obj(x):
        return _compute_c5(x)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bounds = [(0.0, 1.0)] * n

    # primary run
    try:
        res = minimize(
            obj,
            init_v,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 500, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            v = _repair_vector(res.x, n)
            score = _compute_c5(v)
            if score < best_score:
                best_score, best_v = score, v
    except Exception:
        pass

    # random restarts
    while time.time() - start < time_limit:
        rnd = _random_feasible(n)
        try:
            res = minimize(
                obj,
                rnd,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n)
                score = _compute_c5(v)
                if score < best_score:
                    best_score, best_v = score, v
        except Exception:
            pass
        if time.time() - start > time_limit * 0.9:
            break
    return best_v, best_score

def _binary_anneal(v, time_limit, init_temp=0.02, final_temp=1e-6):
    """Simulated annealing on binary 0/1 vector."""
    n = v.shape[0]
    ones = list(np.flatnonzero(v > 0.5))
    zeros = list(np.flatnonzero(v <= 0.5))
    cur_score = _compute_c5(v)
    best_v = v.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = init_temp * (1 - frac) + final_temp * frac
        i_idx = np.random.randint(len(ones))
        j_idx = np.random.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        v[i] = 0.0
        v[j] = 1.0
        new_score = _compute_c5(v)
        dE = new_score - cur_score
        accept = False
        if dE < 0:
            accept = True
        elif T > 0:
            accept = np.random.rand() < np.exp(-dE / T)
        if accept:
            cur_score = new_score
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_score < best_score:
                best_score = cur_score
                best_v = v.copy()
                if best_score <= 0.3808:  # early exit target
                    break
        else:
            v[i] = 1.0
            v[j] = 0.0
    return best_v, best_score

def _upsample_binary(v, n_target):
    """Upsample binary vector to a larger size."""
    n_src = len(v)
    factor = n_target // n_src
    if factor * n_src == n_target:
        return np.repeat(v, factor)
    src_x = np.linspace(0, 2, n_src, endpoint=False)
    tgt_x = np.linspace(0, 2, n_target, endpoint=False)
    v_interp = np.interp(tgt_x, src_x, v)
    idx = np.argsort(v_interp)[::-1]
    v_up = np.zeros(n_target, dtype=np.float64)
    v_up[idx[: n_target // 2]] = 1.0
    return v_up
