    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    # trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # enlarge intervals if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) + 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_gradient(h, shift):
    """Gradient of the active max-correlation term for a given integer shift."""
    n = h.size
    grad = np.empty_like(h)
    if shift >= 0:
        # h[i] interacts with h[i+shift] and h[i-shift] where defined
        grad[:shift] = 1.0 - h[:shift] - 0.0
        grad[shift:n-shift] = 1.0 - h[shift:n-shift] - h[0:n-2*shift]
        grad[n-shift:] = 1.0 - 0.0 - h[n-2*shift:n]
    else:
        s = -shift
        grad[:s] = 1.0 - 0.0 - h[s:n]
        grad[s:n-s] = 1.0 - h[0:n-2*s] - h[s:n-s]
        grad[n-s:] = 1.0 - h[n-2*s:n] - 0.0
    return grad

def compute_c5(h):
    """Maximum overlap integral (discrete) for a given h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def projected_gradient_descent(h0, max_iter=3000, step0=0.5,
                               start_time=None, budget_s=1e9):
    """Projected gradient descent minimizing max-correlation."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        if start_time is not None and time.time() - start_time > budget_s - 0.5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    """Thue–Morse binary (0/1) sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def cosine_h(n):
    """Half‑cosine bump on [0,2] discretised to n points."""
    x = (np.arange(n) + 0.5) * (2.0 / n)          # centre of each interval
    h = 0.5 * (1.0 + np.cos(np.pi * (x - 1.0)))  # 0.5*(1+cos(pi*(x-1)))
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Determine problem size and base candidates
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.asarray(initial_h_values, dtype=float)
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 256
        h0 = None

    candidates = []
    if h0 is not None:
        candidates.append(h0)

    # known deterministic patterns
    if (n & (n - 1)) == 0:          # power of two -> Thue‑Morse exact half‑ones
        candidates.append(thue_morse(n))
    candidates.append(cosine_h(n))
    # a few random balanced binary and fractional starts
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(3):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))

    # ------------------------------------------------------------------
    # Optimize each candidate (moderate iterations)
    # ------------------------------------------------------------------
    for base in candidates:
        if time.time() - start > budget_s - 1.0:
            break
        h_opt, sc = projected_gradient_descent(
            base, max_iter=6000, step0=0.5,
            start_time=start, budget_s=budget_s)
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()

    # ------------------------------------------------------------------
    # Intensify search around the current best: perturb‑refine cycles
    # ------------------------------------------------------------------
    while time.time() - start < budget_s - 0.5:
        # small random perturbation of the best solution
        eps = 0.12
        pert = best_h + eps * (rng.random(n) - 0.5)
        pert = project_onto_box_sum(pert, n / 2.0)
        # short refinement
        h_ref, sc_ref = projected_gradient_descent(
            pert, max_iter=2000, step0=0.5,
            start_time=start, budget_s=budget_s)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()
            continue
        # if not improved, try a pure random restart
        v = rng.random(n)
        v = project_onto_box_sum(v, n / 2.0)
        h_ref, sc_ref = projected_gradient_descent(
            v, max_iter=1500, step0=0.5,
            start_time=start, budget_s=budget_s)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # ------------------------------------------------------------------
    # Final local fine‑tuning via pairwise transfers
    # ------------------------------------------------------------------
    while time.time() - start < budget_s - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move mass from i to j (ensure bounds)
        eps_max = min(best_h[i], 1.0 - best_h[j])
        if eps_max > 1e-12:
            eps = rng.random() * eps_max * 0.6
            h_new = best_h.copy()
            h_new[i] -= eps
            h_new[j] += eps
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
                continue
        # try opposite direction
        eps_max = min(best_h[j], 1.0 - best_h[i])
        if eps_max > 1e-12:
            eps = rng.random() * eps_max * 0.6
            h_new = best_h.copy()
            h_new[j] -= eps
            h_new[i] += eps
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # ------------------------------------------------------------------
    # Guarantee feasibility and recompute final score
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
