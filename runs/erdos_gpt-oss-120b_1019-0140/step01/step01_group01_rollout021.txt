import time
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(h: np.ndarray) -> float:
    """Compute the Erdős C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Scale and clip h so that sum(h) == target while keeping 0<=h_i<=1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # greedy fix using random indices
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # distribute remaining error uniformly, then clip again
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    end_time = start + budget_s
    safety = 0.5  # seconds reserved for final steps

    # ----- problem size -----
    n_points = 200
    target_sum = n_points / 2.0

    # ----- initial solution -----
    try:
        init = np.array(initial_h_values, dtype=float)
        if init.size != n_points:
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Adjust h so that its sum equals target while respecting 0≤h_i≤1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
