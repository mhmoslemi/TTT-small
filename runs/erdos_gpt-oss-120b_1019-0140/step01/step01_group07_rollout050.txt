def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) <= 1e-12:
        return v
    # if need to increase sum
    if cur < target:
        need = target - cur
        # compute capacities which we can add (1 - v[i]) for each i
        caps = 1.0 - v
        # sort caps descending to fill largest capacity first? Actually we can fill any order; but to speed, fill all equally? Might overshoot.
        # Simplify: proportionally distribute need among free positions: v += need * caps / np.sum(caps)
        # But ensuring not exceed 1.
        # We'll iterate.
        free = np.where(caps > 1e-12)[0]
        while need > 1e-12 and free.size > 0:
            caps = 1.0 - v[free]
            total_cap = np.sum(caps)
            if total_cap <= 0:
                break
            inc = np.minimum(caps, need * caps / total_cap)
            v[free] += inc
            need -= np.sum(inc)
            # Update free
            free = np.where(v < 1.0 - 1e-12)[0]

import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Discrete C5: max_k ∫ h(x)(1-h(x+k))dx."""
    n = h.size
    dx = 2.0 / n
    conv = np.convolve(h, (1.0 - h)[::-1], mode="full")
    return np.max(conv) * dx

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust sum to target via water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) <= 1e-12:
        return v
    if cur < target:
        need = target - cur
        while need > 1e-12:
            free = np.where(v < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            caps = 1.0 - v[free]
            total_cap = np.sum(caps)
            if total_cap <= 0:
                break
            inc = np.minimum(caps, need * caps / total_cap)
            v[free] += inc
            need -= np.sum(inc)
    else:
        excess = cur - target
        while excess > 1e-12:
            free = np.where(v > 1e-12)[0]
            if free.size == 0:
                break
            caps = v[free]
            total_cap = np.sum(caps)
            if total_cap <= 0:
                break
            dec = np.minimum(caps, excess * caps / total_cap)
            v[free] -= dec
            excess -= np.sum(dec)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _make_binary(v, target_int):
    """Return binary vector (0/1) with exactly target_int ones, picking top values."""

        target_sum2 = n2 / 2.0
        # duplicate binary pattern
        h_up = np.repeat(best_h, 2)
        h_up = _project_to_constraints(h_up, target_sum2)
        # hill climbing at higher resolution (limited time)
        end_up = start + budget_s * 0.95
        best_h2 = h_up
        best_sc2 = _compute_c5(best_h2)
        while time.time() < end_up:
            ones = np.flatnonzero(best_h2)
            zeros = np.flatnonzero(1.0 - best_h2)
            if ones.size == 0 or zeros.size == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h_new = best_h2.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            sc = _compute_c5(h_new)
            if sc + 1e-12 < best_sc2:
                best_h2 = h_new
                best_sc2 = float(sc)
                if best_sc2 < best_score:
                    best_score = best_sc2
                    best_h = best_h2.copy()
                if best_sc2 < 0.3808:
                    break

    # ------------------------------------------------------------------
    # Final evaluation via provided evaluator if present.
    final_score = best_score
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            pass

    return best_h.astype(np.float64), float(final_score), int(best_h.size)
