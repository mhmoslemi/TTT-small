    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5

    # Determine n_points
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    except Exception:
        n = 200

    target_sum = n / 2.0
    # initialize h:
    if 'init_h' in locals():
        h = adjust_sum_to_target(init_h.copy(), target_sum, rng)
    else:
        h = rng.rand(n)
        h = h / h.sum() * target_sum
        h = adjust_sum_to_target(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # Simulated annealing loop
    # set temperature schedule with parameters:
    T_init = 0.01
    T_final = 1e-5
    step_init = 0.2
    step_final = 1e-4
    iter = 0

    # keep track of stagnation counter
    last_improve_iter = 0

    while time.time() - start < max_time:
        progress = (time.time() - start) / max_time
        T = T_init * (1 - progress) + T_final * progress
        step = step_init * (1 - progress) + step_final * progress

        # propose candidate
        # Choose move type: with p=0.5 gradient, else pairwise
        if rng.rand() < 0.4:
            # gradient move
            # compute correlation (full) to find worst shift index
            corr_full = np.correlate(current_h, 1 - current_h, mode='full')
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n - 1)
            # gradient vector approx same as before:
            g = np.zeros_like(current_h)
            idx = np.arange(n)
            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n)
            g[valid_plus] += (1 - current_h[plus[valid_plus]])
            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n)
            g[valid_minus] -= current_h[minus[valid_minus]]
            # Project g to zero-sum
            g = g - g.mean()
            # apply step
            candidate = current_h - step * g
            # Clip and adjust sum
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        else:
            # pairwise
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            hi = current_h[i]
            hj = current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2 - 1) * step
            delta = np.clip(delta, low, high)
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        cand_c5 = compute_c5(candidate)
        # Acceptance
        if cand_c5 < current_c5 or rng.rand() < np.exp(-(cand_c5 - current_c5) / T):
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                last_improve_iter = iter

            current_h = candidate
            current_c5 = compute_c5(candidate)
            last_improve_iter = iter

        iter += 1

    # final adjust and compute
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n)

def adjust_sum_to_target(h, target, rng):
    total=h.sum()
    diff=target-total
    if abs(diff)<1e-12:
        return h
    # attempt to fix using random indices repeatedly
    attempts=0
    max_attempts=500
    while attempts<max_attempts and abs(diff)>1e-12:
        idx=rng.randint(h.size)
        if diff>0:
            inc=min(diff,1.0-h[idx])
            if inc>0:
                h[idx]+=inc
                diff-=inc
        else:
            dec=min(-diff,h[idx])
            if dec>0:
                h[idx]-=dec
                diff+=dec
        attempts+=1
    if abs(diff)>1e-12:
        # distribute uniformly
        h+=diff/h.size
        np.clip(h,0.0,1.0,out=h)
        # after clipping, may need adjust again
        total=h.sum()
        diff=target-total
        if abs(diff)>1e-12:
            # small scaling
            if total>0:
                h*=target/total
            np.clip(h,0.0,1.0,out=h)
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of C5 = max_k ∫ h(x)(1-h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Project h onto {0≤h_i≤1, sum(h)=target}."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # try to fix the deficit/surplus using random indices
    max_attempts = 500
    attempts = 0
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # distribute the remaining tiny error uniformly, then clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # final scaling if needed
        total = h.sum()
        if total > 0.0:
            h *= target / total
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------------
    # Initialise problem size and starting vector
    # --------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
        h = adjust_sum_to_target(init_h.copy(), n / 2.0, rng)
    except Exception:
        # fallback: use a moderately sized grid (fast enough yet expressive)
        n = 200
        h = rng.rand(n)
        h = h / h.sum() * (n / 2.0)
        h = adjust_sum_to_target(h, n / 2.0, rng)

    target_sum = n / 2.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # --------------------------------------------------------------------
    # Simulated annealing / gradient descent parameters
    # --------------------------------------------------------------------
    T_start = 5e-3      # initial temperature
    T_end = 1e-7        # final temperature
    step_start = 0.25   # initial step magnitude
    step_end = 5e-5     # final step magnitude

    iter_cnt = 0
    last_improve = 0

    # --------------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------------
    while time.time() - start < max_time:
        progress = (time.time() - start) / max_time
        # linear schedules (could be exponential, but linear works)
        T = T_start * (1.0 - progress) + T_end * progress
        step = step_start * (1.0 - progress) + step_end * progress

        # ----------------------------------------------------------------
        # Propose a candidate move
        # ----------------------------------------------------------------
        if rng.rand() < 0.45:
            # ----- Gradient move on the worst shift -----
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n - 1)   # integer lag (can be negative)

            # gradient of the maximal overlap term (projected to zero sum)
            g = np.zeros_like(current_h)
            idx = np.arange(n)

            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n)
            g[valid_plus] += (1.0 - current_h[plus[valid_plus]])

            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n)
            g[valid_minus] -= current_h[minus[valid_minus]]

            # remove the component that changes the total mass
            g -= g.mean()

            candidate = current_h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        else:
            # ----- Pairwise exchange move -----
            i, j = rng.randint(n, size=2)
            if i != j:
                hi, hj = current_h[i], current_h[j]
                low = max(-hi, hj - 1.0)          # smallest delta preserving bounds
                high = min(1.0 - hi, hj)          # largest delta preserving bounds
                if low < high:
                    delta = (rng.rand() * 2.0 - 1.0) * step
                    delta = np.clip(delta, low, high)
                    candidate = current_h.copy()
                    candidate[i] = hi + delta
                    candidate[j] = hj - delta
                else:
                    candidate = current_h.copy()
            else:
                candidate = current_h.copy()

        cand_c5 = compute_c5(candidate)

        # ----------------------------------------------------------------
        # Acceptance test (Metropolis)
        # ----------------------------------------------------------------
        if cand_c5 < current_c5 or rng.rand() < np.exp(-(cand_c5 - current_c5) / max(T, 1e-12)):
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                last_improve = iter_cnt

        # ----------------------------------------------------------------
        # Stagnation handling: occasional shake from the current best
        # ----------------------------------------------------------------
        if iter_cnt - last_improve > 4000:
            # add modest Gaussian noise to best and re‑start from there
            noisy = best_h + rng.normal(scale=0.04, size=n)
            np.clip(noisy, 0.0, 1.0, out=noisy)
            current_h = adjust_sum_to_target(noisy, target_sum, rng)
            current_c5 = compute_c5(current_h)
            last_improve = iter_cnt

        iter_cnt += 1

    # --------------------------------------------------------------------
    # Final refinement: a short deterministic gradient descent from best
    # --------------------------------------------------------------------
    refine_iters = 1500
    for _ in range(refine_iters):
        # compute worst shift
        corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_idx = int(np.argmax(corr_full))
        k_shift = k_idx - (n - 1)

        # gradient
        g = np.zeros_like(best_h)
        idx = np.arange(n)
        plus = idx + k_shift
        valid_plus = (plus >= 0) & (plus < n)
        g[valid_plus] += (1.0 - best_h[plus[valid_plus]])
        minus = idx - k_shift
        valid_minus = (minus >= 0) & (minus < n)
        g[valid_minus] -= best_h[minus[valid_minus]]
        g -= g.mean()

        # small deterministic step
        step = 1e-4
        cand = best_h - step * g
        np.clip(cand, 0.0, 1.0, out=cand)
        cand = adjust_sum_to_target(cand, target_sum, rng)
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
        else:
            break  # no further improvement

    # --------------------------------------------------------------------
    # Return the best solution found
    # --------------------------------------------------------------------
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n)
