def projected_adam(h0, n_iter, lr0=0.5, alpha0=0.0, alpha_end=20.0, beta1=0.9, beta2=0.999, eps=1e-8, sub_iter_check=100):
    h = h0.copy()
    n = h.size
    s = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter+1):
        # step size: maybe lr = lr0 / np.sqrt(t)
        lr = lr0 / np.sqrt(t)
        # alpha schedule
        alpha = alpha0 + (alpha_end - alpha0) * (t-1) / (n_iter-1)
        # compute gradient
        if alpha <= 0:
            # standard subgradient for max shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            # max shift
            idx = np.argmax(corr)
            shift = idx - (n-1)
            grad = compute_gradient(h, shift)
        else:
            # compute gradient for smoothed objective
            corr_raw = np.correlate(h, 1.0 - h, mode='full')
            max_corr = np.max(corr_raw)
            # normalized weights
            # avoid overflow
            ex = np.exp(alpha * (corr_raw - max_corr))
            probs = ex / np.sum(ex)
            # compute weighted term1+term2
            weighted_term = np.zeros_like(h)
            mid = n-1
            for shift_idx, p in enumerate(probs):
                shift = shift_idx - mid
                if shift >= 0:
                    # term1
                    weighted_term[:n-shift] += p * h[shift:]
                    # term2
                    weighted_term[shift:] += p * h[:n-shift]
                else:
                    s = -shift
                    weighted_term[s:] += p * h[:n-s]
                    weighted_term[:n-s] += p * h[s:]
            grad = 1.0 - weighted_term
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # optional time check
        if t % sub_iter_check == 0:
            if time.time() - start_time > budget_s - 0.5:
                break
    return best_h, best_score

import time
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x : lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    # trivial bounds
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral for discretised h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of max‑shift correlation term."""
    n = h.size
    grad = np.full(n, 1.0, dtype=h.dtype)
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad

def compute_smooth_gradient(h, alpha):
    """Gradient of smoothed max (log‑sum‑exp) with parameter alpha>0."""
    n = h.size
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # softmax weights
    ex = np.exp(alpha * (corr - max_corr))
    probs = ex / np.sum(ex)

    weighted = np.zeros_like(h)
    # accumulate contributions for each shift
    for idx, p in enumerate(probs):
        shift = idx - mid
        if shift >= 0:
            weighted[:n - shift] += p * h[shift:]
            weighted[shift:] += p * h[:n - shift]
        else:
            s = -shift
            weighted[s:] += p * h[:n - s]
            weighted[:n - s] += p * h[s:]
    return 1.0 - weighted

def optimizer_adam(h0, max_iter, start_time, budget_s,
                   lr0=0.5, alpha_start=0.0, alpha_end=15.0,
                   alpha_thresh=1e-3):
    """Adam with projection onto box & sum. Returns best (h,score)."""
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_h = h.copy()
    best_score = compute_c5(h)

    for t in range(1, max_iter + 1):
        # time guard (check every 100 steps)
        if (t % 100 == 0) and (time.time() - start_time > budget_s - 0.5):
            break

        # schedule for smoothing
        if max_iter > 1:
            alpha = alpha_start + (alpha_end - alpha_start) * (t - 1) / (max_iter - 1)
        else:
            alpha = alpha_start

        if alpha > alpha_thresh:
            grad = compute_smooth_gradient(h, alpha)
        else:
            # subgradient from the active shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            shift = np.argmax(corr) - (n - 1)
            grad = compute_gradient(h, shift)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step

        # projection onto feasible set
        h = project_onto_box_sum(h, target_sum)

        # evaluate
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()

    return best_h, best_score

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1 values)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h
