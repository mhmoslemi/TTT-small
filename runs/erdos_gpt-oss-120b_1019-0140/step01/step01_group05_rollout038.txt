def _simulated_annealing(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.01  # initial temperature
    max_eps0 = 0.2
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        max_eps = max_eps0 * frac  # shrink eps
        # choose i,j
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        # compute feasible eps
        up = min(1.0 - hi, hj)  # positive move: increase hi, decrease hj
        down = min(hi, 1.0 - hj)  # negative move: decrease hi, increase hj
        # total feasible range = up + down
        total = up + down
        if total <= 0.0:
            continue
        # sample eps uniformly from [-down, up]
        eps = (rng.random() * total) - down
        eps = np.clip(eps, -max_eps, max_eps)  # cap
        if abs(eps) < 1e-12:
            continue
        # apply move
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # clip maybe not needed
        # evaluate
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _gradient_descent(h, time_limit, rng):
    N = h.size
    dx = 2.0 / N
    start = time.time()
    step = 0.02  # base step size
    while time.time() - start < time_limit:
        # compute corr and max shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (N - 1)
        # compute gradient
        grad = np.full(N, dx)  # derivative of sum term
        # handle shift contributions
        if shift >= 0:
            if shift != 0:
                # left side indices where i+shift < N
                grad[:N-shift] -= dx * h[shift:]
                # where i-shift >=0
                grad[shift:] -= dx * h[:N-shift]
            else:
                # shift 0, double count? Actually derivative is dx*(1 - 2*h) because both terms same: code above yields correct if we treat both contributions.
                grad -= dx * h
                grad -= dx * h
        else:
            s = -shift
            grad[s:] -= dx * h[:N-s]
            grad[:N-s] -= dx * h[s:]
        # Determine indices for moving mass
        # Positive gradient: increase would increase max; we want to decrease those positions.
        # So we prefer to move mass from max positive gradient to max negative gradient.
        i_up = np.argmax(grad)  # largest (most positive)
        i_down = np.argmin(grad)  # most negative
        if grad[i_up] <= 0 or grad[i_down] >= 0:
            # No beneficial direction.
            break
        # Determine feasible eps
        up = h[i_up]  # can decrease to 0
        down = 1.0 - h[i_down]  # can increase to 1
        eps = min(up, down) * 0.5  # move half feasible amount
        if eps <= 0:
            break
        h_new = h.copy()
        h_new[i_up] -= eps
        h_new[i_down] += eps
        new_score = _c5_value(h_new)
        if new_score < _c5_value(h):
            h = h_new
            # possibly adjust step
        else:
            # reduce step size if not improving (not used)
            pass
        # Optionally random perturbations to escape local minima
        # maybe also break if time nearly done
        if time.time() - start > time_limit:
            break
    return h

def _repair_vector(v):
    N = v.size
    target = N / 2.0
    # Clip initially
    v = np.clip(v, 0.0, 1.0)
    # Define function to compute sum after shifting by lam
    def sum_with_lambda(lam):
        return np.clip(v + lam, 0.0, 1.0).sum()
    # Find lambda bounds
    lo = -1.0
    hi = 1.0
    sum_lo = sum_with_lambda(lo)
    sum_hi = sum_with_lambda(hi)
    # expand bounds if needed
    while sum_lo > target:
        lo -= 1.0
        sum_lo = sum_with_lambda(lo)
    while sum_hi < target:
        hi += 1.0
        sum_hi = sum_with_lambda(hi)
    # binary search
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = sum_with_lambda(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    v = np.clip(v + lam, 0.0, 1.0)
    # Due to numerical tolerance, adjust to exactly target by scaling
    # Compute remaining diff
    diff = target - v.sum()
    if abs(diff) > 1e-9:
        # Distribute diff uniformly across indices not at bounds
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size > 0:
            v[free] += diff / free.size
            v = np.clip(v, 0.0, 1.0)
        else:
            # fallback uniform adjust
            v += diff / N
            v = np.clip(v, 0.0, 1.0)
    # final clip maybe
    v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    # schedule constants
    T0 = 0.02
    max_eps0 = 0.1
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        max_eps = max_eps0 * frac
        # sample pair
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        hi = h_cur[i]; hj = h_cur[j]
        up = min(1.0 - hi, hj)   # possible positive eps
        down = min(hi, 1.0 - hj) # possible negative eps magnitude
        total = up + down
        if total == 0.0:
            continue
        eps = rng.random() * total - down
        # clip eps
        if eps > max_eps:
            eps = max_eps
        elif eps < -max_eps:
            eps = -max_eps
        if eps == 0.0:
            continue
        # apply
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # ensure within [0,1] (should be)
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

Now gradient descent function:

def _gradient_descent(h, time_limit, rng):
    N = h.size
    dx = 2.0 / N
    start = time.time()
    best_h = h.copy()
    best_score = _c5_value(best_h)
    while time.time() - start < time_limit:
        # compute correlation to find worst shift
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (N - 1)
        if shift == 0:
            grad = dx * (1 - 2 * best_h)
        else:
            s = abs(shift)
            grad = np.full(N, dx)
            # subtract neighbor at +s
            grad[:N-s] -= dx * best_h[s:]
            # subtract neighbor at -s
            grad[s:] -= dx * best_h[:N-s]
        # Identify indices to move mass
        i_up = np.argmax(grad)  # largest positive gradient (bad)
        i_down = np.argmin(grad)  # most negative gradient (good)
        # if no improvement possible
        if grad[i_up] <= 0 or grad[i_down] >= 0:
            break
        # feasible epsilon
        up = best_h[i_up]  # can decrease to 0
        down = 1.0 - best_h[i_down]  # can increase to 1
        eps = min(up, down) * 0.5
        if eps <= 0:
            break
        h_new = best_h.copy()
        h_new[i_up] -= eps
        h_new[i_down] += eps
        new_score = _c5_value(h_new)
        if new_score < best_score - 1e-12:
            best_h = h_new
            best_score = new_score
        else:
            # if no improvement, maybe reduce eps factor
            # try smaller eps
            eps = min(up, down) * 0.1
            if eps <= 0:
                break
            h_new = best_h.copy()
            h_new[i_up] -= eps
            h_new[i_down] += eps
            new_score = _c5_value(h_new)
            if new_score < best_score - 1e-12:
                best_h = h_new
                best_score = new_score
            else:
                # no improvement, break
                break
    return best_h

    sum_hi = sum_lambda(hi)
    while sum_lo > target:
        lo -= 1.0
        sum_lo = sum_lambda(lo)
    while sum_hi < target:
        hi += 1.0
        sum_hi = sum_lambda(hi)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = sum_lambda(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    v = np.clip(v + lam, 0.0, 1.0)
    # small residual adjustment
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size > 0:
            v[free] += diff / free.size
            v = np.clip(v, 0.0, 1.0)
        else:
            v += diff / N
            v = np.clip(v, 0.0, 1.0)
    v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.02
    max_eps0 = 0.1
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        max_eps = max_eps0 * frac
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        hi = h_cur[i]
        hj = h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        total = up + down
        if total == 0.0:
            continue
        eps = rng.random() * total - down
        # cap eps
        if eps > max_eps:
            eps = max_eps
        elif eps < -max_eps:
            eps = -max_eps
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)

            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _gradient_descent(h_init, time_limit, rng):
    h = h_init.copy()
    N = h.size
    dx = 2.0 / N
    best_score = _c5_value(h)
    start = time.time()
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (N - 1)
        if shift == 0:
            grad = dx * (1.0 - 2.0 * h)
        else:
            s = abs(shift)
            grad = np.full(N, dx)
            if s < N:
                grad[:N - s] -= dx * h[s:]
                grad[s:] -= dx * h[:N - s]
        i_up = int(np.argmax(grad))
        i_down = int(np.argmin(grad))
        if grad[i_up] <= 0.0 or grad[i_down] >= 0.0:
            break
        up = h[i_up]          # amount we can decrease
        down = 1.0 - h[i_down]  # amount we can increase
        eps = min(up, down) * 0.5
        if eps <= 0.0:
            break
        h_new = h.copy()
        h_new[i_up] -= eps
        h_new[i_down] += eps
        new_score = _c5_value(h_new)
        if new_score < best_score - 1e-12:
            h = h_new
            best_score = new_score
        else:
            # try a smaller step
            eps = min(up, down) * 0.1
            if eps <= 0.0:
                break
            h_new = h.copy()
            h_new[i_up] -= eps
            h_new[i_down] += eps
            new_score = _c5_value(h_new)
            if new_score < best_score - 1e-12:
                h = h_new
                best_score = new_score
            else:
                break
    return h
