def optimize_h(h0, rng, start_time, budget_s, max_iter=50000):
    n = h0.size
    s = n/2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    momentum_beta = 0.9
    init_lr = 0.5
    # schedule: lr = init_lr / sqrt(iter) perhaps
    stagnation_counter = 0
    for it in range(1, max_iter+1):
        if time.time() - start_time > budget_s - 0.1:
            break
        # compute correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)   # includes 1 term
        # momentum update
        m = momentum_beta * m + (1 - momentum_beta) * grad
        eta = init_lr / math.sqrt(it)
        h_new = h - eta * m
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        # update best
        if score < best_score - 1e-12:   # some tolerance
            best_score = score
            best_h = h_new.copy()
            stagnation_counter = 0
            # reset momentum maybe
            m = np.zeros_like(h)
        else:
            stagnation_counter += 1

        # maybe random perturb if stagnation > some threshold (like 2000)
        if stagnation_counter > 2000:
            # apply random perturb
            # perturb a random fraction of entries
            # choose indices, add noise scaled small ~0.05?
            num_change = max(1, int(0.05 * n))
            idxs = rng.choice(n, size=num_change, replace=False)
            delta = (rng.random(num_change) - 0.5) * 0.2   # [-0.1,0.1]
            h_new[idxs] += delta
            # reproject
            h_new = project_onto_box_sum(h_new, s)
            # reset stagnation
            stagnation_counter = 0
            # reset momentum
            m = np.zeros_like(h)

        h = h_new
    return best_h, best_score

def optimize_with_adam(h0, s, rng, start_time, budget_s, max_iter, lr=0.4):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for it in range(1, max_iter+1):
        if time.time() - start_time > budget_s - 0.1:
            break
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)

            best_h = h_new.copy()

        if it % 1000 == 0:

                # small perturbation
                noise = (rng.random(n) - 0.5) * 0.02
                h_new = project_onto_box_sum(h_new + noise, s)
        h = h_new
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    v = np.asarray(v, dtype=float)
    n = v.size
    if s <= lo * n:
        return np.full(n, lo, dtype=float)
    if s >= hi * n:
        return np.full(n, hi, dtype=float)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if s <= lo * n:
        return np.full(n, lo, dtype=float)
    if s >= hi * n:
        return np.full(n, hi, dtype=float)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def optimize_with_momentum(h0, s, rng, start_time, budget_s,
                          max_iter=20000, lr0=0.5, beta=0.9):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time.time() - start_time > budget_s - 0.1:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        m = beta * m + (1 - beta) * grad
        eta = lr0 / math.sqrt(it)
        h_new = h - eta * m
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            m.fill(0.0)   # reset momentum after improvement
        h = h_new
    return best_h, best_score

def local_search_refine(h, rng, start_time, budget_s, max_trials=20000):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for _ in range(max_trials):
        if time.time() - start_time > budget_s - 0.05:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    final_n = 256
    target_sum = final_n / 2.0
    best_h = None
    best_score = float('inf')

    # ---------- base candidates ----------
    candidates = []
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != final_n:
            h0 = upsample(h0, final_n)
        h0 = project_onto_box_sum(h0, target_sum)
        candidates.append(h0)
    candidates.append(thue_morse(final_n).astype(float))
    candidates.append(random_binary_balanced(final_n, rng))
    fr = project_onto_box_sum(rng.random(final_n), target_sum)
    candidates.append(fr)

    # add upsampled lower‑resolution seeds
    for n_small in (64, 128):
        if time.time() - start_time > budget_s * 0.4:
            break
        h_small = random_binary_balanced(n_small, rng)
        h_up = upsample(h_small, final_n)
        candidates.append(project_onto_box_sum(h_up, target_sum))
        v_small = rng.random(n_small)
        h_up2 = upsample(v_small, final_n)
        candidates.append(project_onto_box_sum(h_up2, target_sum))

    # ---------- intensive optimization ----------
    for h0 in candidates:
        if time.time() - start_time > budget_s * 0.7:
            break
        h_opt, sc_opt = optimize_with_momentum(
            h0, target_sum, rng, start_time, budget_s,
            max_iter=20000, lr0=0.5, beta=0.9)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # ---------- random restarts ----------
    while time.time() - start_time < budget_s - 1.0:
        if rng.random() < 0.3:
            h_start = random_binary_balanced(final_n, rng)
        else:
            h_start = project_onto_box_sum(rng.random(final_n), target_sum)
        h_opt, sc_opt = optimize_with_momentum(
            h_start, target_sum, rng, start_time, budget_s,
            max_iter=8000, lr0=0.5, beta=0.9)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # ---------- final local refinement ----------
    if best_h is not None and time.time() - start_time < budget_s - 0.5:
        h_ref, sc_ref = local_search_refine(best_h, rng, start_time, budget_s, max_trials=15000)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # final guarantee of feasibility
    if best_h is None:
        best_h = np.full(final_n, 0.5, dtype=float)
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)
    return best_h, best_score, final_n
