import time
import math
import numpy as np

def compute_c5_bin(h):
    """Return C5 for a binary vector h (0/1)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all shifts
    corr = np.correlate(h.astype(float), (1 - h).astype(float), mode='full')
    return float(np.max(corr) * dx)

def thue_morse_seq(n):
    """Generate Thue‑Morse parity sequence of length n (0/1)."""
    return np.fromiter((i.bit_count() & 1 for i in range(n)), dtype=int, count=n)

def adjust_to_half(arr, target_ones, rng):
    """Mutate arr (0/1) in place so that sum(arr) == target_ones."""
    cur = int(arr.sum())
    diff = target_ones - cur
    if diff > 0:
        zeros = np.where(arr == 0)[0]
        if len(zeros) >= diff:
            sel = rng.choice(zeros, diff, replace=False)
            arr[sel] = 1
    elif diff < 0:
        ones = np.where(arr == 1)[0]
        sel = rng.choice(ones, -diff, replace=False)
        arr[sel] = 0

def simulated_annealing(start_h, max_time, rng):
    """Simple SA on binary vectors with exact half‑sum constraint."""
    n = start_h.shape[0]
    target = n // 2
    cur = start_h.copy()
    cur_score = compute_c5_bin(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.10  # initial temperature
    while time.time() - start < max_time:
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = T0 * (1.0 - frac)
        if T < 1e-6:
            T = 1e-6
        # pick random 1 and 0 positions
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(1 - cur)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cur[i] = 0
        cur[j] = 1
        new_score = compute_c5_bin(cur)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
                if best_score <= 0.3808:
                    break
        else:

            cur[i] = 1
            cur[j] = 0

def hill_climb(start_h, rng, max_iter=5000, time_limit=None):
    """Deterministic random‑swap hill climbing."""
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_score = compute_c5_bin(cur)
    best = cur.copy()
    best_score = cur_score

            cur[j] = 0

    # Ensure exact half‑sum (numerical safety)
    cur_sum = int(best_h.sum())
    if cur_sum != target_ones:
        # simple repair: flip random bits to fix the count
        diff = target_ones - cur_sum
        if diff > 0:
            zeros = np.where(best_h == 0)[0]
            sel = rng.choice(zeros, diff, replace=False)
            best_h[sel] = 1
        elif diff < 0:
            ones = np.where(best_h == 1)[0]
            sel = rng.choice(ones, -diff, replace=False)
            best_h[sel] = 0

    return best_h.astype(float), best_c5, n
