def _simulated_annealing(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.02
    eps_max0 = 0.3
    eps_min = 0.001
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac + 1e-8
        max_eps = eps_max0 * frac + eps_min
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        # sample eps in range [-down, up]
        eps_range = up + down
        eps = rng.random() * eps_range - down
        # clamp eps to max_eps
        if eps > max_eps:
            eps = max_eps
        elif eps < -max_eps:
            eps = -max_eps
        if eps == 0.0:
            continue
        # apply move
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # clip to [0,1] - but we already ensured within bounds via up/down
        # evaluate
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v)=len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float,
                         rng: np.random.Generator,
                         T0: float = 0.02, eps_max0: float = 0.3) -> tuple:
    """
    Simple SA preserving sum and box constraints.
    Parameters:
        h_start: initial vector (already feasible)
        time_limit: seconds to run
        rng: random generator
        T0: initial temperature
        eps_max0: initial maximal epsilon transfer
    Returns:
        best vector and its score.
    """
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    eps_min = 1e-4
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac + 1e-12
        max_eps = eps_max0 * frac + eps_min
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)      # max increase for i (decrease j)
        down = min(hi, 1.0 - hj)    # max decrease for i (increase j)
        if up == 0.0 and down == 0.0:
            continue
        eps_range = up + down
        eps = rng.random() * eps_range - down
        if eps > max_eps:
            eps = max_eps
        elif eps < -max_eps:
            eps = -max_eps
        if eps == 0.0:
            continue
        # propose move in-place
        h_cur[i] = hi + eps
        h_cur[j] = hj - eps
        new_score = _c5_value(h_cur)
        delta = new_score - cur_score
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            # revert
            h_cur[i] = hi
            h_cur[j] = hj
    return best_h, best_score

                                            T0=0.005, eps_max0=0.05)
        if s_ref < best_score:
            best_h, best_score = h_ref, s_ref

    # final repair and exact evaluation
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
