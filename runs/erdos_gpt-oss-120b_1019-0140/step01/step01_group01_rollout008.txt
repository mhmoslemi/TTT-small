import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Projects h onto the hyperplane sum(h)=target while respecting 0≤h_i≤1.
    Small violations are corrected iteratively.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] minimizing the
    Erdős overlap constant C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start_time = time.time()
    max_time = budget_s - 0.2  # safety margin

    # ----- initialise problem size and starting vector -----
    try:
        init_h = np.array(initial_h_values, dtype=float).ravel()
        n0 = init_h.size
        # upscale to gain flexibility (up to 200 points)
        n_points = min(200, max(70, n0 * 2))
        xs = np.linspace(0, n0 - 1, n_points)
        h0 = np.interp(xs, np.arange(n0), init_h)
    except Exception:
        n_points = 180
        h0 = rng.rand(n_points)

    target_sum = n_points / 2.0
    h0 = h0 / h0.sum() * target_sum
    h0 = adjust_sum_to_target(h0, target_sum, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    current_h = h0.copy()
    current_c5 = best_c5

    # annealing / search parameters
    T0, Tf = 0.05, 1e-5          # temperature start / end
    step0, step_f = 0.30, 1e-4   # step magnitude start / end
    restart_limit = 25000        # iterations without improvement before restart
    no_improve = 0

    while time.time() - start_time < max_time:
        elapsed = time.time() - start_time
        prog = elapsed / max_time
        T = T0 * (1 - prog) + Tf * prog
        step = step0 * (1 - prog) + step_f * prog

        move_type = rng.rand()
        n = n_points

        if move_type < 0.4:
            # ---- gradient (subgradient) move on worst shift ----
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            shift = k_idx - (n - 1)   # signed lag

            # neighbor contribution S_i = h_{i+shift} + h_{i-shift} (zero if out of bounds)
            S = np.zeros_like(current_h)
            if shift >= 0:
                S[:n - shift] += current_h[shift:]
                S[shift:] += current_h[:n - shift]
            else:

                no_improve = 0

    # final projection to ensure exact mass
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
