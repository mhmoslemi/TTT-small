import time
import math
import numpy as np

def _correlation_full(h: np.ndarray) -> np.ndarray:
    """Linear full correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    return corr[: 2 * n - 1]

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = _correlation_full(h)
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0.0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h
