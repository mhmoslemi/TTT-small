def softmin_ascent(h0, max_iters, tau_start, tau_end, step0):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    # precompute exponential schedule factor
    if max_iters>1:
        tau_factor = (tau_end / tau_start) ** (1.0 / (max_iters - 1))
    else:
        tau_factor = 1.0
    tau = tau_start
    for it in range(1, max_iters+1):
        # compute autocorr
        f = np.correlate(h, h, mode='full')
        f_min = f.min()
        # compute weights
        w = np.exp(-(f - f_min) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            # fallback uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad = 2.0 * np.convolve(w, h, mode='same')
        # step size
        eta = step0 / np.sqrt(it)
        h = h + eta * grad
        # projection
        h = project_onto_box_sum(h, s)
        # evaluate
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # update tau
        tau *= tau_factor
    return best_h, best_score

import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # bisection on lambda
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        clipped = np.clip(v - lam_mid, lo, hi)
        total = clipped.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(total - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    """C5 = max_k ∫ h(x)*(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the max‑correlation term for a given integer shift."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def projected_gradient_descent(h0, n_iter=2000, step0=0.5):
    """Projected gradient descent for the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1 values)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def softmin_ascent(h0, max_iters, tau_start, tau_end, step0):
    """
    Gradient ascent of a smooth approximation of min_k sum_i h_i h_{i+k}.
    Returns the best vector seen and its C5 value.
    """
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    # exponential decay factor for tau
    if max_iters > 1:
        tau_factor = (tau_end / tau_start) ** (1.0 / (max_iters - 1))
    else:
        tau_factor = 1.0
    tau = tau_start

    for it in range(1, max_iters + 1):
        # autocorrelation f_k = Σ_i h_i h_{i+k}
        f = np.correlate(h, h, mode='full')
        f_min = f.min()

        # soft‑min weights (larger weight for smaller f_k)
        w = np.exp(-(f - f_min) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * (w * h) convolution
        grad = 2.0 * np.convolve(w, h, mode='same')

        eta = step0 / math.sqrt(it)
        h = h + eta * grad
        h = project_onto_box_sum(h, s)

        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()

        tau *= tau_factor
    return best_h, best_score
