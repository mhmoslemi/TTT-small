        resolutions.append(resolutions[-1] * 2)

    total_budget = max(0.0, budget_s - 2.0)
    weights = [float(r) for r in resolutions]
    weight_sum = sum(weights)

    for n in resolutions:
        elapsed = time.time() - start
        time_left = budget_s - elapsed - 1.0
        if time_left <= 0:
            break
        t_res = total_budget * (float(n) / weight_sum)
        t_res = min(t_res, time_left)
        if len(best_v) != n:
            init_v = _upsample_vector(best_v, n)
        else:
            init_v = best_v.copy()
        t_ga = t_res * 0.6
        if t_ga > 0.1:
            v_ga, sc_ga = _gradient_ascent(init_v, n, t_ga)
        else:
            v_ga, sc_ga = init_v.copy(), _compute_c5(init_v)
        t_opt = t_res - t_ga
        if t_opt > 0.1:
            v_opt, sc_opt = _optimize_slsqp(v_ga, n, t_opt)
        else:
            v_opt, sc_opt = v_ga, sc_ga
        cur_elapsed = time.time() - start
        remaining_budget = budget_s - cur_elapsed - 0.5
        if remaining_budget > 0.2:
            v_ls, sc_ls = _local_search(v_opt, n, min(0.2, remaining_budget))
            if sc_ls < sc_opt:
                v_opt, sc_opt = v_ls, sc_ls
        if sc_opt < best_score:
            best_score, best_v, best_n = sc_opt, v_opt, n

    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)
    return best_v.astype(np.float64), float(best_score), int(best_n)
