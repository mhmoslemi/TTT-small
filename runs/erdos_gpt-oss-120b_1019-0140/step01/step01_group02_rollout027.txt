def mutate_move(h, step, rng):
    # returns a new copy mutated.
    new_h = h.copy()
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return new_h
    hi = new_h[i]
    hj = new_h[j]
    delta_min = max(-hi, -(1.0 - hj))
    delta_max = min(1.0 - hi, hj)
    low = max(delta_min, -step)
    high = min(delta_max, step)
    if low >= high:
        return new_h
    delta = rng.random() * (high - low) + low
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

import time

# ---------- Helper functions ----------
def _compute_c5(h: np.ndarray) -> float:
    """Calculate the discretized C5 value for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float, tol: float = 1e-12) -> np.ndarray:
    """Adjust entries of h so that sum(h) == target while keeping 0≤h≤1."""
    diff = target - h.sum()
    for _ in range(10):
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            # distribute evenly as a fallback
            h += diff / h.size
            np.clip(h, 0.0, 1.0, out=h)
            break
        i = idx[0]
        new_val = np.clip(h[i] + diff, 0.0, 1.0)
        actual = new_val - h[i]
        h[i] = new_val
        diff = target - h.sum()
    return h


def _random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with exact sum equal to target."""
    h = np.empty(n, dtype=float)
    remaining = target
    for i in range(n - 1):
        max_possible = min(1.0, remaining)
        min_possible = max(0.0, remaining - (n - i - 1) * 1.0)
        val = min_possible + rng.random() * (max_possible - min_possible)
        h[i] = val
        remaining -= val
    h[-1] = remaining
    np.clip(h, 0.0, 1.0, out=h)
    return h


def _tournament_select(scores: np.ndarray, k: int, rng: np.random.Generator) -> int:
    """Select an index by tournament (lower score = better)."""
    idx = rng.integers(0, scores.shape[0], size=k)
    best = idx[np.argmin(scores[idx])]
    return int(best)


def _crossover(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """One‑point crossover."""
    n = p1.shape[0]
    cut = rng.integers(1, n)  # ensure both sides non‑empty
    child = np.concatenate((p1[:cut], p2[cut:]))
    return child


def _mutate(h: np.ndarray, step: float, rng: np.random.Generator) -> np.ndarray:
    """Apply a random mutation preserving the sum (or repaired later)."""
    n = h.shape[0]
    new_h = h.copy()
    if rng.random() < 0.5:
        # Transfer between two indices (keeps sum exactly)
        i, j = rng.integers(0, n, size=2)
        if i != j:
            hi, hj = new_h[i], new_h[j]
            delta_min = max(-hi, -(1.0 - hj))
            delta_max = min(1.0 - hi, hj)
            low = max(delta_min, -step)
            high = min(delta_max, step)
            if low < high:
                delta = rng.random() * (high - low) + low
                new_h[i] = hi + delta
                new_h[j] = hj - delta
    else:
        # Perturb a single entry (sum will be repaired)
        i = rng.integers(0, n)
        sigma = step * 0.5
        delta = rng.normal() * sigma
        delta = np.clip(delta, -new_h[i], 1.0 - new_h[i])
        new_h[i] += delta
    return new_h


# ---------- Main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise population ----
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        target_sum = n / 2.0
        init = np.clip(init, 0.0, 1.0)
        init = _repair_sum(init, target_sum)
    except NameError:
        # No supplied construction – choose a moderate size.
        n = 200
        target_sum = n / 2.0
        init = _random_feasible(n, target_sum, rng)

    pop_size = 50
    tournament_k = 5
    p_crossover = 0.7
    max_step = 0.5
    min_step = 0.001
    max_temp = 0.015
    min_temp = 1e-6

    population = [init.copy()]
    while len(population) < pop_size:
        population.append(_random_feasible(n, target_sum, rng))

    scores = np.array([_compute_c5(ind) for ind in population], dtype=float)

    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_c5 = scores[best_idx]

    # ---- Genetic algorithm phase ----
    ga_end = start_time + budget_s * 0.75
    while time.time() < ga_end:
        elapsed = time.time() - start_time
        frac = min(elapsed / (budget_s * 0.75), 1.0)
        step = max_step * (1.0 - frac) + min_step * frac

        # parent selection
        p1_idx = _tournament_select(scores, tournament_k, rng)
        p2_idx = _tournament_select(scores, tournament_k, rng)
        while p2_idx == p1_idx:
            p2_idx = _tournament_select(scores, tournament_k, rng)

        p1 = population[p1_idx]
        p2 = population[p2_idx]

        # offspring creation
        if rng.random() < p_crossover:
            child = _crossover(p1, p2, rng)
        else:
            child = p1.copy()

        child = _mutate(child, step, rng)
        np.clip(child, 0.0, 1.0, out=child)
        child = _repair_sum(child, target_sum)

        child_c5 = _compute_c5(child)

        # replace worst if advantageous
        worst_idx = int(np.argmax(scores))
        if child_c5 < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_c5

        # track global best
        if child_c5 < best_c5:
            best_c5 = child_c5
            best_h = child.copy()

    # ---- Local simulated‑annealing refinement ----
    current = best_h.copy()
    current_c5 = best_c5
    while True:
        now = time.time()
        if now - start_time >= budget_s:
            break

        elapsed = now - start_time
        frac = elapsed / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = current[i], current[j]

        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        low = max(delta_min, -step)
        high = min(delta_max, step)
        if low >= high:
            continue

        delta = rng.random() * (high - low) + low
        current[i] = hi + delta
        current[j] = hj - delta

        new_c5 = _compute_c5(current)
        delta_obj = new_c5 - current_c5

        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / temp):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current.copy()
        else:
            current[i] = hi
            current[j] = hj

    # final repair and evaluation
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
