    max_val = corr.max()
    idxs = np.where(corr >= max_val - tol)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)
        grad += gradient_of_shift(h, shift)
    grad /= len(idxs)
    return grad

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    # goal: find λ such that sum(clip(v - λ, lo, hi)) = s
    # The projection onto a box with sum constraint is given by shifting v by λ and clipping.
    # uses binary search.
    # lambda lower bound: min(v) - hi (since at that shift, all components at hi
    # lambda upper bound: max(v) - lo (since at that shift, all at lo)
    # Evaluate f(λ) = sum(clipped). monotonic decreasing in λ.
    lo_lambda = np.min(v) - hi
    hi_lambda = np.max(v) - lo
    for _ in range(max_iter):
        mid = (lo_lambda + hi_lambda) / 2
        clipped = np.clip(v - mid, lo, hi)
        total = clipped.sum()
        if total > s:
            lo_lambda = mid
        else:
            hi_lambda = mid
        if abs(total - s) < 1e-12:

def upsample(arr, new_n):

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    """Euclidean projection of v onto {x : lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(total - s) < 1e-12:
            break
    return np.clip(v - (lam_low + lam_high) * 0.5, lo, hi)


def compute_c5(h):
    """Discrete approximation of C5 for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def max_corr_shift_and_val(h):
    """Return shift (integer index offset) achieving max correlation and its value."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (h.size - 1)
    return shift, corr[idx]


def gradient_of_shift(h, shift):
    """Subgradient of the correlation term for a given shift."""
    n = h.size
    grad = np.ones_like(h, dtype=float)
    if shift >= 0:
        grad[: n - shift] -= h[shift:]
        grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        grad[s:] -= h[: n - s]
        grad[: n - s] -= h[s:]
    return grad


def adam_optimize(h0, max_iter, start_time, budget_s, alpha=0.05):
    """Adam optimization with projection onto the feasible set."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter + 1):
        if time.time() - start_time > budget_s - 0.5:
            break
        shift, _ = max_corr_shift_and_val(h)
        grad = gradient_of_shift(h, shift)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
