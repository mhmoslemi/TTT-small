def binary_annealing(h_init, target_sum, start_time, budget_s):
    n = len(h_init)
    h_current = h_init.copy()
    best_h = h_current.copy()
    best_c5 = compute_c5(best_h)
    # maintain indices sets
    ones = np.where(h_current == 1)[0]
    zeros = np.where(h_current == 0)[0]
    rng = np.random.default_rng()
    no_improve = 0
    while time.time() - start_time < budget_s * 0.95:
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_current[i] = 0
        h_current[j] = 1
        c5_new = compute_c5(h_current)
        if c5_new < best_c5 - 1e-9:
            best_c5 = c5_new
            best_h = h_current.copy()
            # update indices
            ones = np.where(h_current == 1)[0]

            ones = np.where(h_current == 1)[0]
            zeros = np.where(h_current == 0)[0]

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if np.abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    cur = np.sum(h)
    if np.abs(cur - target_sum) > 1e-6:
        # adjust the element with most capacity
        # find index with max slack
        slack_up = 1.0 - h
        slack_down = h
        if cur < target_sum:
            # need increase sum
            i = np.argmax(slack_up)
            h[i] = min(1.0, h[i] + (target_sum - cur))
        else:
            i = np.argmax(slack_down)
            h[i] = max(0.0, h[i] - (cur - target_sum))
    return h
```

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return C5 = max correlation * dx for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    """Clip to [0,1] and adjust sum to target_sum by distributing the deficit."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        if cur < target_sum:
            i = np.argmax(1.0 - h)  # most room to increase
            h[i] = min(1.0, h[i] + (target_sum - cur))
        else:
            i = np.argmax(h)        # most room to decrease
            h[i] = max(0.0, h[i] - (cur - target_sum))
    return h


def random_binary(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    ones = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[ones] = 1.0
    return h


def binarize_h(h, target_sum):
    """Set the largest target_sum entries of h to 1, rest to 0."""
    idx = np.argsort(-h)
    bin_h = np.zeros_like(h, dtype=np.float64)
    bin_h[idx[:target_sum]] = 1.0
    return bin_h


def binary_annealing(h_init, rng, start_time, budget_s, target_c5=0.3808):
    """Focused search: iteratively swap a 1 with a 0 to reduce the worst lag."""
    n = h_init.shape[0]
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    while time.time() - start_time < budget_s * 0.96:
        # worst lag
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        max_val = corr.max()
        max_idxs = np.where(corr == max_val)[0]
        lag_idx = rng.choice(max_idxs)
        shift = int(lag_idx - (n - 1))

        # positions causing the worst contribution
        if shift >= 0:
            i_vals = np.arange(0, n - shift)
        else:
            i_vals = np.arange(-shift, n)
        j_vals = i_vals + shift
        mask = (h_cur[i_vals] == 1) & (h_cur[j_vals] == 0)
        bad_i = i_vals[mask]

        if bad_i.size:
            i_swap = rng.choice(bad_i)
        else:
            ones = np.flatnonzero(h_cur == 1)
            if ones.size == 0:
                break
            i_swap = rng.choice(ones)

        zeros = np.flatnonzero(h_cur == 0)
        if zeros.size == 0:
            break
        j_swap = rng.choice(zeros)

        # perform swap
        h_cur[i_swap] = 0.0
        h_cur[j_swap] = 1.0
        c5_new = compute_c5(h_cur)

        if c5_new + 1e-12 < best_c5:
            best_c5 = c5_new
            best_h = h_cur.copy()
            no_improve = 0
            if best_c5 <= target_c5:
                break
        else:
            # revert
            h_cur[i_swap] = 1.0
            h_cur[j_swap] = 0.0
            no_improve += 1
            if no_improve >= 3000:
                h_cur = random_binary(n, rng)
                no_improve = 0
    return best_h, best_c5


def continuous_refine(h_start, rng, start_time, budget_s, max_trials=4):
    """Polish a solution with SLSQP (few short runs)."""
    n = h_start.shape[0]
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    for trial in range(max_trials):
        if time.time() - start_time > budget_s * 0.96:
            break
        noise = rng.normal(scale=0.02, size=n)
        h0 = np.clip(best_h + noise, 0.0, 1.0)
        h0 = project_to_feasible(h0, n / 2.0)
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
        bnds = [(0.0, 1.0)] * n
        try:
            res = opt.minimize(
                lambda x: compute_c5(x),
                x0=h0,
                method="SLSQP",
                bounds=bnds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-7, "disp": False},
            )
        except Exception:
            continue
        if not res.success:
            continue
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_to_feasible(h_opt, n / 2.0)
        c5_val = compute_c5(h_opt)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if best_c5 <= 0.3808:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the C5 bound."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # Base construction (if supplied)
    base = globals().get("initial_h_values")
    if base is not None:
        base = np.asarray(base, dtype=np.float64)
    else:
        base = random_binary(380, rng)

    n = base.shape[0]
    target_sum = n // 2

    # Ensure binary start
    if np.all((base == 0) | (base == 1)):
        h_bin = base.copy()
    else:
        h_bin = binarize_h(base, target_sum)

    # Phase 1: binary focused annealing
    best_h, best_c5 = binary_annealing(h_bin, rng, start, budget_s)

    # Phase 2: optional continuous polishing if time permits
    if time.time() - start < budget_s * 0.70 and best_c5 > 0.3808:
        best_h, best_c5 = continuous_refine(best_h, rng, start, budget_s)

    # Phase 3: upscale to double resolution and repeat (if time left)
    elapsed = time.time() - start
    if elapsed < budget_s * 0.85:
        n2 = n * 2
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_feasible(h_interp, n2 / 2.0)
        h_bin2 = binarize_h(h_interp, n2 // 2)
        best_h2, best_c5_2 = binary_annealing(h_bin2, rng, start, budget_s)
        if best_c5_2 < best_c5:
            best_h, best_c5, n = best_h2, best_c5_2, n2
            # one final polish at higher resolution
            if time.time() - start < budget_s * 0.97:
                best_h, best_c5 = continuous_refine(best_h, rng, start, budget_s)

    # Final safety projection and evaluation
    best_h = project_to_feasible(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
