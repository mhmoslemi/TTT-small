def _binary_hill_climb(v, n, time_limit):
    # v binary array with exactly n/2 ones
    best = v.copy()
    best_score = _compute_c5(best)
    cur = v.copy()
    cur_score = best_score
    start = time.time()
    # set temperature schedule
    Tmax = 0.01  # initial temperature (maybe 1e-2)
    Tmin = 1e-6
    while time.time() - start < time_limit:
        # compute temperature as decreasing
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        T = Tmax * (1 - t_frac) + Tmin * t_frac
        # choose random swap
        # indices of ones and zeros
        ones = np.flatnonzero(cur == 1)
        zeros = np.flatnonzero(cur == 0)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cand = cur.copy()
        cand[i] = 0
        cand[j] = 1
        cand_score = _compute_c5(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
    return best, best_score

def _binary_hill_climb_batch(v, n, time_limit, batch_size=50):
    cur_v = v.copy()
    cur_score = _compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < time_limit:
        # generate batch of candidate swaps
        ones = np.flatnonzero(cur_v == 1)
        zeros = np.flatnonzero(cur_v == 0)
        # ensure there are at least 1 of each
        # produce random pairs
        i_idxs = np.random.choice(ones, size=batch_size, replace=True)
        j_idxs = np.random.choice(zeros, size=batch_size, replace=True)
        best_candidate = None
        best_candidate_score = cur_score
        for i, j in zip(i_idxs, j_idxs):
            cand = cur_v.copy()
            cand[i] = 0
            cand[j] = 1
            cand_score = _compute_c5(cand)
            if cand_score < best_candidate_score:
                best_candidate_score = cand_score
                best_candidate = cand
        if best_candidate is not None and best_candidate_score < cur_score:
            cur_v = best_candidate
            cur_score = best_candidate_score
            if cur_score < best_score:
                best_v = cur_v.copy()
                best_score = cur_score
        else:
            # No improvement, break maybe
            # random restart: reinitialize random binary vector
            cur_v = _random_binary_feasible(n)
            cur_score = _compute_c5(cur_v)
    return best_v, best_score

import time

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v) = n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v


def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx


def _random_feasible_continuous(n):
    """Random vector in [0,1]^n with sum = n/2."""
    v = np.random.rand(n)
    return _repair_vector(v, n)


def _random_binary_feasible(n):
    """Random binary vector with exactly n/2 ones (n must be even)."""
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 1.0
    np.random.shuffle(v)
    return v


def _continuous_to_binary(v):
    """Convert a continuous vector to a binary one with exactly floor(n/2) ones."""
    n = len(v)
    half = n // 2
    idx = np.argpartition(-v, half)[:half]

    b[idx] = 1.0
    return b
