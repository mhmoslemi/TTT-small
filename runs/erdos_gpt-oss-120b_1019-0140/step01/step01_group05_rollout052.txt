def simulated_annealing_targeted(h_start, time_limit, rng):
    h = h_start.copy()
    N = h.size
    dx = 2/N
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        # temperature schedule: linear decreasing from T0 to near 0
        # T0 can be maybe 0.5 * (max_score) or some constant.
        T0 = 0.01 # maybe too small? We'll compute scale after first iteration.
        # compute fraction of time left:
        frac = max(0.0, 1.0 - elapsed / time_limit)
        # Temperature may go from initial T_start to small.
        T = T0 * frac

        # compute correlation and max shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (N - 1)

        # attempt a targeted swap
        # generate candidate indices
        if shift >= 0:
            i_candidates = np.where((h[:N-shift]==1) & (h[shift:]==0))[0]
            if i_candidates.size > 0:
                i = rng.choice(i_candidates)
                j = i + shift
                # swap
                h_new = h.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                # compute new score
                new_score = compute_c5(h_new)
                delta = new_score - cur_score
                if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
                    h = h_new
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h_new.copy()
                continue # go next iteration
        else:
            sh = -shift
            i_candidates = np.where((h[sh:] == 1) & (h[:N-sh]==0))[0] + sh  # i index
            if i_candidates.size > 0:
                i = rng.choice(i_candidates)
                j = i - sh
                h_new = h.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                new_score = compute_c5(h_new)
                delta = new_score - cur_score
                if delta < 0 or rng.random() < math.exp(-delta/(T+1e-12)):
                    h = h_new
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h_new.copy()
                continue

        # fallback: random swap of a 1 and a 0
        # find random indices of ones and zeros
        ones_idx = np.where(h==1)[0]
        zeros_idx = np.where(h==0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            # should not happen (balanced)
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - cur_score

            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    
    return best_h, best_score

def binary_repair(v):
    # Clip to [0,1]
    np.clip(v, 0, 1, out=v)
    N = v.size
    target = N // 2  # integer; N even we ensure.
    # Round to 0 or 1 (e.g., >0.5)
    bin_arr = (v >= 0.5).astype(np.float64)  # using float for compatibility
    # Count current ones
    cur = int(bin_arr.sum())
    if cur == target:
        return bin_arr
    # If too many ones: need to flip some ones to zero
    if cur > target:
        # Identify ones indices sorted by value ascending (choose smallest ones to flip)
        ones_idx = np.where(bin_arr == 1)[0]
        # sort by original v (lowest value may be borderline)
        order = np.argsort(v[ones_idx])  # ascending
        to_flip = ones_idx[order[:cur - target]]
        bin_arr[to_flip] = 0.0
    else: # too few ones
        zeros_idx = np.where(bin_arr == 0)[0]
        order = np.argsort(-v[zeros_idx])  # descending by original v
        to_flip = zeros_idx[order[:target - cur]]
        bin_arr[to_flip] = 1.0
    return bin_arr

def _simulated_annealing_targeted(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = len(h_cur)
    start = time.time()
    # Precompute initial temperature based on current_score
    T0 = cur_score * 0.1 + 1e-6 # small baseline
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac

        # compute correlation to find worst shift
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (N - 1)

        # find candidate swap along shift
        if shift >= 0:
            hi = h_cur[:N - shift]
            hj = h_cur[shift:]
            cond = (hi == 1) & (hj == 0)
            cand_i = np.where(cond)[0]
            if cand_i.size > 0:
                i = rng.choice(cand_i)
                j = i + shift
                # swap
                h_new = h_cur.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                new_score = compute_c5(h_new)
                delta = new_score - cur_score
                if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
                    h_cur = h_new
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h_new.copy()
                continue
        else:
            sh = -shift
            hi = h_cur[sh:]
            hj = h_cur[:N - sh]
            cond = (hi == 1) & (hj == 0)
            cand_i = np.where(cond)[0] + sh
            if cand_i.size > 0:
                i = rng.choice(cand_i)
                j = i - sh
                h_new = h_cur.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                new_score = compute_c5(h_new)
                delta = new_score - cur_score
                if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
                    h_cur = h_new
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h_new.copy()
                continue

        # fallback: random swap
        # choose random positions i where h_cur[i] == 1, j where h_cur[j] == 0.
        ones_idx = np.where(h_cur == 1)[0]
        zeros_idx = np.where(h_cur == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h_new = h_cur.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()

    return best_h, best_score

def _epsilon_sa(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = cur_score * 0.1 + 1e-6
    max_eps = 0.2
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        # pick two indices
        i, j = rng.integers(0, N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        # compute allowable epsilon: increase hi up to 1 and decrease hj accordingly to keep sum.
        # Actually we want to transfer epsilon from j to i.
        up = min(1.0 - hi, hj)  # max epsilon we can add to i (increase hi) while decreasing hj (by same amount)
        down = min(hi, 1.0 - hj)  # max epsilon we can subtract from i (decrease hi) while increase hj
        # total range [-down, up]
        if up == 0 and down == 0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import time
import math
import numpy as np

def _repair_vector_float(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and scale to exact sum N/2 (float version)."""
    np.clip(v, 0.0, 1.0, out=v)
    N = v.size
    target = N / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / N
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v


def _binary_repair(v: np.ndarray) -> np.ndarray:
    """Convert to binary 0/1 while keeping exactly N/2 ones (N must be even)."""
    np.clip(v, 0.0, 1.0, out=v)
    N = v.size
    target = N // 2
    bin_arr = (v >= 0.5).astype(np.float64)
    cur = int(bin_arr.sum())
    if cur == target:
        return bin_arr
    if cur > target:
        # too many ones -> flip some to zero (prefer smallest original values)
        excess = cur - target
        ones_idx = np.where(bin_arr == 1)[0]
        order = np.argsort(v[ones_idx])  # ascending
        to_flip = ones_idx[order[:excess]]
        bin_arr[to_flip] = 0.0
    else:
        # too few ones -> flip some zeros to one (prefer largest original values)
        deficit = target - cur
        zeros_idx = np.where(bin_arr == 0)[0]
        order = np.argsort(-v[zeros_idx])  # descending
        to_flip = zeros_idx[order[:deficit]]
        bin_arr[to_flip] = 1.0
    return bin_arr


def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _simulated_annealing_targeted(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Binary swap SA that preferentially reduces the worst overlap shift."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = cur_score * 0.1 + 1e-12  # initial temperature scale
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        # locate worst shift
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (N - 1)

        move_made = False
        if shift >= 0:
            hi = h_cur[: N - shift]
            hj = h_cur[shift:]
            cand = np.where((hi == 1) & (hj == 0))[0]
            if cand.size:
                i = rng.choice(cand)
                j = i + shift
                h_new = h_cur.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                new_score = _c5_value(h_new)
                delta = new_score - cur_score
                if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
                    h_cur, cur_score = h_new, new_score
                    if new_score < best_score:
                        best_h, best_score = h_new.copy(), new_score
                move_made = True
        else:
            sh = -shift
            hi = h_cur[sh:]
            hj = h_cur[: N - sh]
            cand = np.where((hi == 1) & (hj == 0))[0] + sh
            if cand.size:
                i = rng.choice(cand)
                j = i - sh
                h_new = h_cur.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                new_score = _c5_value(h_new)
                delta = new_score - cur_score
                if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
                    h_cur, cur_score = h_new, new_score
                    if new_score < best_score:
                        best_h, best_score = h_new.copy(), new_score
                move_made = True

        if not move_made:
            # fallback random swap of a 1 and a 0
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = h_cur.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            new_score = _c5_value(h_new)
            delta = new_score - cur_score
            if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
                h_cur, cur_score = h_new, new_score
                if new_score < best_score:
                    best_h, best_score = h_new.copy(), new_score
    return best_h, best_score


def _epsilon_sa(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Continuous SA that moves a small epsilon between two positions (preserves sum)."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = cur_score * 0.1 + 1e-12
    max_eps = 0.2
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(0, N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)      # amount we can add to hi (take from hj)
        down = min(hi, 1.0 - hj)    # amount we can remove from hi (give to hj)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur, cur_score = h_new, new_score
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    total_budget = budget_s

    # ---------- Phase A: quick multi-start binary SA (N=200) ----------
    N0 = 200
    phase_a_budget = total_budget * 0.30
    restart_time = 12.0
    best_h = None
    best_score = float('inf')
    while time.time() - start_total < phase_a_budget - restart_time:
        h0 = np.zeros(N0, dtype=np.float64)

        h0[idx] = 1.0
        h_opt, sc_opt = _simulated_annealing_targeted(h0, restart_time, rng)
        if sc_opt < best_score:

    if best_h is None:
        # fallback single start

        idx = rng.choice(N0, size=N0 // 2, replace=False)
        h0[idx] = 1.0

    # ---------- Phase B: longer refinement at N0 ----------
    elapsed = time.time() - start_total
    remaining = total_budget - elapsed
    phase_b_budget = remaining * 0.25
    if phase_b_budget > 5.0:
        best_h, best_score = _simulated_annealing_targeted(best_h, phase_b_budget, rng)

    # ---------- Phase C: upscale to N1=400 and binary SA ----------
    elapsed = time.time() - start_total
    remaining = total_budget - elapsed
    N1 = 400
    x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, N1, endpoint=False)
    h_interp = np.interp(x_new, x_old, best_h)
    h_start_N1 = _binary_repair(h_interp)
    phase_c_budget = remaining * 0.30
    if phase_c_budget > 5.0:
        best_h, best_score = _simulated_annealing_targeted(h_start_N1, phase_c_budget, rng)
        N0 = N1  # update current size

    # ---------- Phase D: upscale to N2=800 (if time) ----------
    elapsed = time.time() - start_total
    remaining = total_budget - elapsed
    if remaining > 30.0:
        N2 = min(800, N0 * 2)
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_start_N2 = _binary_repair(h_interp)
        phase_d_budget = remaining * 0.60
        if phase_d_budget > 5.0:
            best_h, best_score = _simulated_annealing_targeted(h_start_N2, phase_d_budget, rng)

    # ---------- Phase E: fine‑tune with epsilon SA ----------
    elapsed = time.time() - start_total
    remaining = total_budget - elapsed
    if remaining > 5.0:
        best_h, best_score = _epsilon_sa(best_h, remaining, rng)

    # Final repair to enforce constraints precisely
    best_h = _repair_vector_float(best_h)
    final_score = _c5_value(best_h)
    return best_h, float(final_score), best_h.size
