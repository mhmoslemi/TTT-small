def projected_adam(h0, max_iter, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s = h.size/2.0  # target sum
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter+1):
        # compute active shift causing max correlation
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (h.size - 1)
        grad = compute_gradient(h, shift)  # shape n
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    # Clip trivial cases
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Expand bracket if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) + 10 * lo  # hi? Actually we want high bound.
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.size
    grad = np.empty(n, dtype=h.dtype)
    if shift >= 0:
        # term1: h shifted forward
        grad[:] = 1.0
        grad[:n - shift] -= h[shift:]
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[:] = 1.0
        grad[s:] -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad

def thue_morse(n):
    # generate binary Thue-Morse sequence of length n
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def adam_opt(h0, max_iters, deadline, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def pairwise_refine(h, rng, max_trials=2000, timeout=None):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = h.size
    for _ in range(max_trials):
        if timeout is not None and time.time() > timeout:

        i, j = rng.integers(0, n, size=2)
        if i == j:

        # feasible epsilon amount to move from i to j

        h_new[j] += eps
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new
    return best_h, best_score

import time
import numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if abs(total - s) < 1e-12:
            return proj
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.size
    grad = np.empty(n, dtype=h.dtype)
    grad[:] = 1.0
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:]      -= h[:n - shift]
    else:
        s = -shift
        grad[s:]   -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def adam_opt(h0, max_iters, deadline, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def pairwise_refine(h, rng, max_trials=3000, timeout=None):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = h.size
    for _ in range(max_trials):
        if timeout is not None and time.time() > timeout:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5
        if eps <= 0:
            continue
        h_new = best_h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin
    target_n = 256
    best_h = None
    best_score = np.inf
    candidates = []

    # initial construction if provided
    if 'initial_h_values' in globals():
        arr = np.asarray(initial_h_values, dtype=float)
        if arr.size != target_n:
            arr = upsample(arr, target_n)
        arr = project_onto_box_sum(arr, target_n / 2.0)
        candidates.append(arr)
        # complement
        comp = 1.0 - arr
        comp = project_onto_box_sum(comp, target_n / 2.0)
        candidates.append(comp)
        # simple smoothing
        smooth = arr.copy()
        smooth[1:-1] = (arr[:-2] + 2 * arr[1:-1] + arr[2:]) / 4.0
        smooth[0] = (arr[0] + arr[1]) / 2.0
        smooth[-1] = (arr[-2] + arr[-1]) / 2.0
        smooth = project_onto_box_sum(smooth, target_n / 2.0)
        candidates.append(smooth)

    # Thue–Morse binary
    candidates.append(thue_morse(target_n))

    # random binary balanced
    for _ in range(2):
        candidates.append(random_binary_balanced(target_n, rng))

    # random fractional starts
    for _ in range(2):
        v = rng.random(target_n)
        candidates.append(project_onto_box_sum(v, target_n / 2.0))

    # Optimize each candidate with a generous iteration budget
    for h0 in candidates:
        if time.time() > deadline:
            break
        h_opt, score_opt = adam_opt(h0, max_iters=35000, deadline=deadline, lr=0.5)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    # Random restarts until time runs out
    while time.time() < deadline:
        v = rng.random(target_n)
        h_start = project_onto_box_sum(v, target_n / 2.0)
        h_opt, score_opt = adam_opt(h_start, max_iters=12000, deadline=deadline, lr=0.5)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    # Local pairwise refinement if time permits
    if best_h is not None and time.time() < deadline:
        best_h, best_score = pairwise_refine(best_h, rng, max_trials=3000, timeout=deadline)

    # Final feasibility projection and score
    if best_h is None:
        best_h = np.full(target_n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, target_n / 2.0)
    else:
        best_h = project_onto_box_sum(best_h, target_n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
