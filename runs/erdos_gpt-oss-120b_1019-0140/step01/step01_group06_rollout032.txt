def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # Identify indices not at bounds
        mask_free = (h > 0.0) & (h < 1.0)
        n_free = np.count_nonzero(mask_free)
        if n_free == 0:
            # cannot adjust further, clamp diff to bound adjustments
            # distribute to those with room
            # find any with h < 1 and increase
            # break
            break
        delta = diff / n_free
        h[mask_free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust first entry allowable
        i = np.argmin(h)  # choose smallest value
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h
```

import time
import numpy as np
from scipy import optimize as opt
from scipy.special import beta

def compute_c5(h):
    """Compute the Erdős C5 bound for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=50):
    """
    Project h onto the set {0 <= h_i <= 1, sum(h)=target_sum}.
    Simple iterative redistribution among interior entries.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = (h > 0.0) & (h < 1.0)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # distribute remaining diff to any non‑saturated entry
        i = np.argmin(h)  # most room to increase
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def generate_shape_h(p, n):
    """Generate a symmetric hump shape h(x) = c * (x*(2-x))**p on n points."""
    dx = 2.0 / n
    # sample at cell centres for symmetry
    x = (np.arange(n) + 0.5) * dx
    raw = (x * (2.0 - x)) ** p
    # enforce integral by scaling to exact discrete sum
    target = n / 2.0
    scale = target / raw.sum()
    h = raw * scale
    h = np.clip(h, 0.0, 1.0)
    h = project_to_feasible(h, target)
    return h
