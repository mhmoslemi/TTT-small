import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def adjust_sum_to_target(h, target, rng):
    # ensure sum exactly target within epsilon, respecting bounds 0-1
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Clip residual if necessary via simple approach: distribute uniformly then clip.
    # For floats, we can do naive:
    # We'll try random index adjustments approach as earlier.
    attempts = 0
    max_attempts = 1000
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # small uniform adjustment
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h
```

def uniform_crossover(rng, parent1, parent2, target_sum):
    n = parent1.shape[0]
    child = np.where(rng.rand(n) < 0.5, parent1, parent2)
    # ensure sum exactly target_sum
    cur_sum = int(child.sum())
    diff = target_sum - cur_sum
    if diff > 0:
        # need to turn diff zeros to ones
        zero_idx = np.where(child == 0)[0]
        if len(zero_idx) >= diff:
            chosen = rng.choice(zero_idx, diff, replace=False)
            child[chosen] = 1
    elif diff < 0:
        # need to turn -diff ones to zeros
        one_idx = np.where(child == 1)[0]
        if len(one_idx) >= -diff:
            chosen = rng.choice(one_idx, -diff, replace=False)
            child[chosen] = 0
    # diff should now be zero
    return child
```

**mutation swap:**

```
def mutation_swap(rng, child):
    # ensure count preserved: swap one 1 and one 0
    ones = np.where(child == 1)[0]
    zeros = np.where(child == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return child
    i = rng.choice(ones)
    j = rng.choice(zeros)
    child[i], child[j] = 0, 1
    return child
```

def tournament_select(rng, fitness, k):
    idx = rng.choice(len(fitness), k, replace=False)
    best = idx[np.argmin(fitness[idx])]
    return best

    start_time = time.time()
    max_time = budget_s - 0.1  # margin

    # Determine n_points
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n_points = init_h.size
    except Exception:
        n_points = 70
        init_h = None

    # Ensure n_points even and target sum integer
    if n_points % 2 != 0:
        # make it even by adding a point (increase n)
        n_points += 1
        if init_h is not None:
            init_h = np.concatenate([init_h, [0.0]])  # just placeholder
    target_sum = n_points // 2  # integer

    # Helper for converting initial_h to binary
    if init_h is not None:
        # create binary from init_h by threshold
        binary_from_init = (init_h >= 0.5).astype(float)   # ensures sum maybe not target
        # adjust sum to target via flip
        binary_from_init = adjust_sum_to_target(binary_from_init, target_sum, rng)
    else:
        binary_from_init = None

    # GA parameters
    pop_size = 200
    tournament_k = 3
    crossover_rate = 0.8
    mutation_rate = 0.2
    elitism = 5

    # Initialize population
    population = []
    # Insert binary_from_init if available
    if binary_from_init is not None:
        population.append(binary_from_init.copy())
    # Fill rest random
    while len(population) < pop_size:
        vec = random_binary_vector(rng, n_points, target_sum)
        population.append(vec)
    population = np.array(population)
    # Compute fitness
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(population[i])
    # Best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # GA loop
    while time.time() - start_time < max_time * 0.7:  # allocate ~70% of time to GA
        # Selection & breeding
        new_pop = []
        # Elitism: carry over best individuals
        elite_indices = np.argsort(fitness)[:elitism]
        for idx in elite_indices:
            new_pop.append(population[idx].copy())
        # Fill rest
        while len(new_pop) < pop_size:
            # select parents
            p1_idx = tournament_select(rng, fitness, tournament_k)
            p2_idx = tournament_select(rng, fitness, tournament_k)
            p1 = population[p1_idx]
            p2 = population[p2_idx]
            # crossover
            if rng.rand() < crossover_rate:
                child = uniform_crossover(rng, p1, p2, target_sum)
            else:
                child = p1.copy()  # clone
            # mutation
            if rng.rand() < mutation_rate:
                child = mutation_swap(rng, child)
            new_pop.append(child)
        population = np.array(new_pop)
        # compute fitness for new population
        for i in range(pop_size):
            fitness[i] = compute_c5(population[i])
        # update best
        cur_best_idx = np.argmin(fitness)
        cur_best_c5 = fitness[cur_best_idx]
        if cur_best_c5 < best_c5:
            best_c5 = cur_best_c5
            best_h = population[cur_best_idx].copy()
        # potentially break early if near target?
        # if best_c5 <= 0.3808: break  (but need to check time)
        # But we need to respect time
    # After GA, we have best_h (binary). Now optional continuous refinement using simulated annealing
    # Convert to float array
    current_h = best_h.astype(float)
    current_c5 = best_c5

    # Simulated annealing parameters
    # We'll use remaining time
    remaining = max_time - (time.time() - start_time)
    if remaining > 2.0:  # at least some time for refinement
        # Use simple SA with pairwise exchange and gradient move like previous code
        T0 = 1e-3
        step0 = 0.1
        # We'll allocate half of remaining for SA
        sa_time_limit = remaining * 0.9
        while time.time() - start_time < max_time - 0.1:
            elapsed = time.time() - start_time
            progress = elapsed / max_time
            # linear cooling
            T = T0 * (1.0 - progress) + 1e-6 * progress
            step = step0 * (1.0 - progress) + 1e-4 * progress
            # choose move type
            if rng.rand() < 0.5:
                # gradient move
                # compute correlation full
                corr_full = np.correlate(current_h, 1.0 - current_h, mode='full')
                k_idx = int(np.argmax(corr_full))
                k_shift = k_idx - (n_points - 1)
                # gradient computation
                g = np.zeros_like(current_h)
                idx = np.arange(n_points)
                plus = idx + k_shift
                valid_plus = (plus >= 0) & (plus < n_points)
                g[valid_plus] += (1.0 - current_h[plus[valid_plus]])
                minus = idx - k_shift
                valid_minus = (minus >= 0) & (minus < n_points)
                g[valid_minus] -= current_h[minus[valid_minus]]
                # project to sum zero
                g -= g.mean()
                candidate = current_h - step * g
                np.clip(candidate, 0.0, 1.0, out=candidate)
                candidate = adjust_sum_to_target(candidate, target_sum, rng)
            else:
                # pairwise exchange move (similar to earlier)
                i, j = rng.randint(n_points, size=2)
                if i != j:
                    hi, hj = current_h[i], current_h[j]
                    low = max(-hi, hj - 1.0)
                    high = min(1.0 - hi, hj)
                    if low < high:
                        delta = (rng.rand() * 2.0 - 1.0) * step
                        delta = max(min(delta, high), low)
                        candidate = current_h.copy()
                        candidate[i] = hi + delta
                        candidate[j] = hj - delta
                        # adjust sum to exactly target if needed due to rounding errors
                        candidate = adjust_sum_to_target(candidate, target_sum, rng)
                    else:
                        continue
                else:
                    continue
            cand_c5 = compute_c5(candidate)
            if cand_c5 < current_c5:
                accept = True
            else:
                prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
                accept = rng.rand() < prob
            if accept:
                current_h = candidate
                current_c5 = cand_c5
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = candidate.copy()
    # ensure final sum exactly target
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust h in place so that its sum equals target while staying in [0,1].
    Small residuals are distributed uniformly and clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 1000
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def random_binary_vector(rng: np.random.RandomState, n: int, target_sum: int) -> np.ndarray:
    """Binary vector of length n with exactly target_sum ones."""
    vec = np.zeros(n, dtype=float)
    ones = rng.choice(n, target_sum, replace=False)
    vec[ones] = 1.0
    return vec


def uniform_crossover(rng: np.random.RandomState, p1: np.ndarray, p2: np.ndarray, target_sum: int) -> np.ndarray:
    """Uniform crossover preserving exact number of ones."""
    n = p1.shape[0]
    child = np.where(rng.rand(n) < 0.5, p1, p2)
    cur_sum = int(child.sum())
    diff = target_sum - cur_sum
    if diff > 0:
        zeros = np.where(child == 0)[0]
        if zeros.size >= diff:
            chosen = rng.choice(zeros, diff, replace=False)
            child[chosen] = 1.0
    elif diff < 0:
        ones = np.where(child == 1)[0]
        if ones.size >= -diff:
            chosen = rng.choice(ones, -diff, replace=False)
            child[chosen] = 0.0
    return child


def mutation_swap(rng: np.random.RandomState, vec: np.ndarray) -> np.ndarray:
    """Swap a random 1 with a random 0 (preserves count)."""
    ones = np.where(vec == 1)[0]
    zeros = np.where(vec == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return vec
    i = rng.choice(ones)
    j = rng.choice(zeros)
    vec[i], vec[j] = 0.0, 1.0
    return vec


def tournament_select(rng: np.random.RandomState, fitness: np.ndarray, k: int) -> int:
    """Select index of best fitness among k random individuals."""
    idx = rng.choice(len(fitness), k, replace=False)
    return idx[np.argmin(fitness[idx])]


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a low C5 bound."""
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.1  # margin

    # ------------------------------------------------------------
    # Determine problem size and optional parent solution
    # ------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n_points = init_h.size
    except Exception:
        init_h = None
        n_points = 70

    # Ensure even number of points (needed for exact half mass)
    if n_points % 2 != 0:
        n_points += 1
        if init_h is not None:
            init_h = np.concatenate([init_h, np.zeros(1, dtype=float)])

    target_sum = n_points // 2  # exact integer mass needed

    # ------------------------------------------------------------
    # Initialise best solution with parent if available
    # ------------------------------------------------------------
    best_h = None
    best_c5 = np.inf
    if init_h is not None:
        init_h = adjust_sum_to_target(init_h.copy(), target_sum, rng)
        init_c5 = compute_c5(init_h)
        best_h = init_h.copy()
        best_c5 = init_c5

    # Binary version of the parent (if any)
    binary_from_init = None
    if init_h is not None:
        binary_from_init = (init_h >= 0.5).astype(float)
        binary_from_init = adjust_sum_to_target(binary_from_init, target_sum, rng)

    # ------------------------------------------------------------
    # Genetic algorithm on binary representations
    # ------------------------------------------------------------
    pop_size = 200
    tournament_k = 3
    crossover_rate = 0.8
    mutation_rate = 0.2
    elitism = 5
    target_bound = 0.3808

    population = []

    # Add a few deterministic seeds
    # Pattern A: block of ones then zeros
    pat_a = np.zeros(n_points, dtype=float)
    pat_a[:target_sum] = 1.0
    population.append(pat_a)

    # Pattern B: 5 ones, 5 zeros repeated 7 times (works for n=70)
    block = np.concatenate([np.ones(5), np.zeros(5)])
    pat_b = np.tile(block, n_points // block.size)
    population.append(pat_b)

    # Add binary parent if present
    if binary_from_init is not None:
        population.append(binary_from_init.copy())

    # Fill the rest with random binaries
    while len(population) < pop_size:
        population.append(random_binary_vector(rng, n_points, target_sum))

    population = np.array(population)  # shape (pop_size, n_points)

    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(population[i])

    # Update best with GA initial population
    cur_best = np.argmin(fitness)
    if fitness[cur_best] < best_c5:
        best_c5 = fitness[cur_best]
        best_h = population[cur_best].copy()

    # Main GA loop (use ~70% of total time)
    ga_deadline = start + max_time * 0.7
    while time.time() < ga_deadline:
        # Elitism
        elite_idx = np.argsort(fitness)[:elitism]
        new_pop = [population[idx].copy() for idx in elite_idx]

        # Generate offspring
        while len(new_pop) < pop_size:
            p1_idx = tournament_select(rng, fitness, tournament_k)
            p2_idx = tournament_select(rng, fitness, tournament_k)
            p1 = population[p1_idx]
            p2 = population[p2_idx]
            if rng.rand() < crossover_rate:
                child = uniform_crossover(rng, p1, p2, target_sum)
            else:
                child = p1.copy()
            if rng.rand() < mutation_rate:
                child = mutation_swap(rng, child)
            new_pop.append(child)

        population = np.array(new_pop)
        # Re‑evaluate fitness
        for i in range(pop_size):
            fitness[i] = compute_c5(population[i])

        # Track best
        cur_best = np.argmin(fitness)
        if fitness[cur_best] < best_c5:
            best_c5 = fitness[cur_best]
            best_h = population[cur_best].copy()

        if best_c5 <= target_bound:
            # reached desirable bound; stop GA early
            break

    # ------------------------------------------------------------
    # Local continuous refinement (simulated annealing)
    # ------------------------------------------------------------
    current_h = best_h.astype(float)
    current_c5 = best_c5

    # Allocate remaining time (up to ~30% of budget)
    while time.time() - start < max_time - 0.2:
        elapsed = time.time() - start
        progress = elapsed / max_time
        # Linear cooling schedules
        T = (1e-3 * (1.0 - progress)) + (1e-6 * progress)
        step = (0.1 * (1.0 - progress)) + (1e-4 * progress)

        if rng.rand() < 0.5:
            # ----- gradient move -----
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n_points - 1)

            g = np.zeros_like(current_h)
            idx = np.arange(n_points)

            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n_points)
            g[valid_plus] += (1.0 - current_h[plus[valid_plus]])

            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n_points)
            g[valid_minus] -= current_h[minus[valid_minus]]

            # Project to zero‑sum hyperplane
            g -= g.mean()
            candidate = current_h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        else:
            # ----- pairwise exchange move -----
            i, j = rng.randint(n_points, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = max(min(delta, high), low)
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta
            candidate = adjust_sum_to_target(candidate, target_sum, rng)

        cand_c5 = compute_c5(candidate)

        if cand_c5 < current_c5 or rng.rand() < np.exp(-(cand_c5 - current_c5) / T):
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()

        if best_c5 <= target_bound:
            # satisfactory bound achieved; we can exit early
            break

    # Final safety adjustment and evaluation
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
