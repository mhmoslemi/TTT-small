def _project_to_constraints(v, target):
    # v is array of floats
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Compute sum
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    # If cur > target (need to decrease)
    if cur > target:
        # Reduce values starting from smallest?
        # We need to subtract from entries >0.
        # We'll sort by value descending (largest) to decrease.
        excess = cur - target
        order = np.argsort(v)[::-1]  # indices of largest values
        for i in order:
            dec = min(v[i], excess)
            v[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    else:  # cur < target
        need = target - cur
        # Increase values at smallest entries first (to stay <=1)
        order = np.argsort(v)  # indices of smallest first
        for i in order:
            inc = min(1.0 - v[i], need)
            v[i] += inc
            need -= inc
            if need <= 1e-12:
                break
    # final clip
    v = np.clip(v, 0.0, 1.0)
    # Ensure sum close to target; if not, scale proportionally slight
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust its sum to target via a simple water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    if cur > target:
        excess = cur - target
        # decrease from largest entries first
        order = np.argsort(v)[::-1]
        for i in order:
            dec = min(v[i], excess)
            v[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    else:
        need = target - cur
        # increase on smallest entries first
        order = np.argsort(v)
        for i in order:
            inc = min(1.0 - v[i], need)
            v[i] += inc
            need -= inc
            if need <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    # final tiny scaling if necessary
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v
