    corr_wh = np.correlate(w, h, mode='full')

    grad = 1.0 - 2.0 * corr_wh[idx::-1][:n]
    return grad

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def _project(v, target):
    """Project vector v onto [0,1]^n with sum(v)=target via binary search."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        v_new = np.clip(v + lam, 0.0, 1.0)
        s = v_new.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _compute_c5(h):
    """Exact C5 = max_k ∫ h(x)(1‑h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _smooth_max(h, beta):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return logsumexp(beta * corr) / beta

def _smooth_grad(h, beta):
    """Gradient of smoothed max C5 w.r.t. h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    w = np.exp(beta * corr - logsumexp(beta * corr))
    corr_wh = np.correlate(w, h, mode="full")
    idx = n - 1
    # grad_k = 1 - 2 * sum_i w_i * h_{i-k}
    grad = 1.0 - 2.0 * corr_wh[idx::-1][:n]
    return grad
