import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Integral overlap bound C5 for a sampled step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=15):
    """Project onto [0,1]^n ∩ {sum = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if still off
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # distribute remaining error to the most flexible entry
        i = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    """Linear interpolation of a parent step function to a new resolution."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def block_pattern(n, target):
    """Block of ones (with a possible fractional last entry) followed by zeros."""
    h = np.zeros(n)
    k = int(np.floor(target))
    if k > 0:
        h[:k] = 1.0
    if k < n:
        h[k] = target - k
    return h

def alternating_pattern(n):
    """0/1 alternating pattern that exactly satisfies the sum constraint for even n."""
    h = np.zeros(n)
    h[::2] = 1.0
    # if n is odd, adjust the last entry to keep the sum correct
    if n % 2 == 1:
        excess = np.sum(h) - n / 2.0
        if excess > 0:
            h[-1] = max(0.0, h[-1] - excess)
        else:
            h[-1] = min(1.0, h[-1] - excess)
    return h

def symmetrize_and_project(h, target):
    """Enforce reflection symmetry and re‑project onto feasibility."""
    h_sym = (h + h[::-1]) * 0.5
    return project_to_feasible(h_sym, target)

def optimize_one(h0, n, target_sum, maxiter):
    """Run SLSQP from a feasible start and return (h, c5)."""
    cons = {"type": "eq", "fun": lambda x, t=target_sum: np.sum(x) - t}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            fun=lambda x: compute_c5(x),
            x0=h0,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-8, "disp": False},
        )
    except Exception:
        return None, None
    h_opt = project_to_feasible(res.x, target_sum)
    c5_val = compute_c5(h_opt)
    # try a symmetric variant
    h_sym = symmetrize_and_project(h_opt, target_sum)
    c5_sym = compute_c5(h_sym)
    if c5_sym < c5_val:
        return h_sym, c5_sym
    return h_opt, c5_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Retrieve parent if any.
    parent = globals().get("initial_h_values")
    # Candidate resolutions (prefer higher resolution for tighter bounds).
    n_list = [380, 190, 95]
