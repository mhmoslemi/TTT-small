def compute_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n-1)
    # allocate gradient zeros
    grad = np.empty_like(h)
    # compute term1 via slicing
    # i+lag in [0,n)
    if lag >= 0:
        # for i in 0..n-1-lag: term1 = 1 - h[i+lag]
        grad[:n-lag] = 1 - h[lag:]
        grad[n-lag:] = 0
        # term2: i - lag >= 0 => i >= lag
        grad[lag:] -= h[:n-lag]
        # for i < lag: nothing subtracted.
    else:
        # lag negative => i+lag <0 for i < -lag
        # For i >= -lag: i+lag >=0
        lag_abs = -lag
        grad[lag_abs:] = 1 - h[:n - lag_abs]
        grad[:lag_abs] = 0
        # term2: i - lag = i + lag_abs
        # i - lag >=0 always; i + lag_abs < n => i < n - lag_abs
        grad[:n - lag_abs] -= h[lag_abs:]
        # rest unchanged.
    return grad, corr[max_idx] * (2.0 / n)
```

def project_to_feasible(h, target, tol=1e-10, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target - cur
        if abs(diff) <= tol:
            break
        free = (h > 0) & (h < 1)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            # can't adjust within interior; adjust the most free capacity.
            if diff > 0:
                # increase the element with smallest value (most room)
                i = np.argmin(h)
                h[i] = min(1.0, h[i] + diff)
            else:
                i = np.argmax(h)
                h[i] = max(0.0, h[i] + diff)  # diff negative
            break
        delta = diff / nfree
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction if needed
    cur = np.sum(h)
    if abs(cur - target) > 1e-6:
        # adjust a single free variable
        diff = target - cur
        # find variable with capacity in direction
        if diff > 0:
            # increase variable with smallest value
            i = np.argmin(h)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            i = np.argmax(h)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h
```

import time
import numpy as np
from scipy.signal import fftconvolve

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.real(fftconvolve(h, 1.0 - h, mode="full"))
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target, tol=1e-10, max_iter=200):
    """Project onto box [0,1] and hyperplane sum(h)=target."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target - cur
        if abs(diff) <= tol:
            break
        free = (h > 0.0) & (h < 1.0)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            break
        delta = diff / nfree
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    cur = np.sum(h)
    if abs(cur - target) > 1e-6:
        diff = target - cur
        if diff > 0:
            i = np.argmin(h)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            i = np.argmax(h)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def gradient_max_corr(h):
    """Gradient of the max correlation term (unscaled)."""
    n = h.shape[0]
    corr = np.real(fftconvolve(h, 1.0 - h, mode="full"))
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)
    grad = np.empty_like(h)
    if lag >= 0:
        # term1
        grad[: n - lag] = 1.0 - h[lag:]
        grad[n - lag :] = 0.0
        # subtract term2
        grad[lag:] -= h[: n - lag]
    else:
        L = -lag
        grad[L:] = 1.0 - h[: n - L]
        grad[:L] = 0.0
        grad[: n - L] -= h[L:]
    max_corr = corr[max_idx]
    return grad, max_corr


def projected_gradient_descent(h0, target, time_limit, start_time):
    """Simple projected gradient descent on the max-correlation objective."""
    n = h0.shape[0]
    h = np.copy(h0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    iteration = 0
    while time.time() - start_time < time_limit:
        grad, max_corr = gradient_max_corr(h)
        cur_c5 = max_corr * (2.0 / n)
        step = 0.5 / (1.0 + iteration * 0.01)  # decreasing step
        h_candidate = h - step * grad
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_candidate = project_to_feasible(h_candidate, target)
        cand_c5 = compute_c5(h_candidate)
        # accept if we improve the current point
        if cand_c5 < cur_c5:
            h = h_candidate
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = h_candidate.copy()
        iteration += 1
        if best_c5 <= 0.3808:
            # bound already very good; keep searching lightly
            pass
    return best_h, best_c5


def hill_climb_binary(h_bin, time_limit, start_time, rng):
    """Random pairwise swaps on a binary (0/1) vector."""
    n = h_bin.shape[0]
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h_bin.copy()
    cur_c5 = best_c5
    while time.time() - start_time < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if cur_h[i] == cur_h[j]:
            continue
        # ensure i has 1, j has 0
        if cur_h[i] == 0:
            i, j = j, i
        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        if best_c5 <= 0.3808:
            # already under target; continue a bit for possible tiny improvements
            pass
    return best_h, best_c5


def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and re‑project to feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n0 = base_h.shape[0]
    if n_target == n0:
        return base_h.copy()
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x1, x0, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Retrieve parent if available
    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)
    else:
        base_h = None

    # Main loop over resolutions (start with the parent resolution, then try double)
    resolutions = []
    if base_h is not None:
        resolutions.append(base_h.shape[0])
    else:
        resolutions.append(380)   # default resolution if no parent
    # try a finer resolution if time permits
    if budget_s >= 300:
        resolutions.append(resolutions[0] * 2)

    best_h = None
    best_c5 = float("inf")
    best_n = None

    for n in resolutions:
        # Prepare starting vector for this resolution
        if base_h is not None and n == base_h.shape[0]:
            h_start = base_h.copy()
        else:
            # interpolate or random initialise
            if base_h is not None:
                h_start = resample_h(base_h, n)
            else:
                h_start = rng.random(n)
                h_start = project_to_feasible(h_start, n / 2.0)

        # Gradient descent phase
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 0:
            break
        # allocate 60 % of remaining time to gradient descent
        gd_time = remaining * 0.6
        h_opt, c5_opt = projected_gradient_descent(
            h_start, n / 2.0, gd_time, start_time
        )

        # Binary hill‑climbing phase
        remaining = budget_s - (time.time() - start_time)
        if remaining > 0.05:
            # produce a binary version of the current best vector
            idx = np.argsort(-h_opt)[: n // 2]
            h_bin = np.zeros_like(h_opt)
            h_bin[idx] = 1.0
            h_bin, c5_bin = hill_climb_binary(
                h_bin, remaining, start_time, rng
            )
            if c5_bin < c5_opt:
                h_opt, c5_opt = h_bin, c5_bin

        # Record global best
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

        # If sufficient bound achieved, we may stop early
        if best_c5 <= 0.3808:
            # still allow a tiny amount of extra time for possible improvement
            if budget_s - (time.time() - start_time) < 0.2:
                break

    if best_h is None:
        # Fallback: simple uniform function
        n = 380
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Ensure final feasibility and recompute bound
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
