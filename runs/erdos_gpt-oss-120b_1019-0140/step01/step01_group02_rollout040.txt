import numpy as np, time, math

def _compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def _contributions(h, shift):
    n = h.shape[0]
    contrib = np.empty(n, dtype=float)
    if shift >= 0:
        ov = n - shift
        if ov > 0:
            contrib[:ov] = h[:ov] * (1.0 - h[shift:])
        if ov < n:
            contrib[ov:] = h[ov:]  # out-of-range contributions
    else:
        shift_abs = -shift
        ov = n - shift_abs
        if ov > 0:
            contrib[shift_abs:] = h[shift_abs:] * (1.0 - h[:ov])
        if shift_abs > 0:
            contrib[:shift_abs] = h[:shift_abs]
    return contrib

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # choose number of points
    n = 200
    # initial binary configuration
    current = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    current[ones] = 1.0
    current_c5 = _compute_c5(current)
    best = current.copy()
    best_c5 = current_c5
    # annealing parameters
    max_temp = 0.2
    min_temp = 1e-4
    stagnation_time = 30.0   # seconds without improvement before restart
    last_impr = start
    K = 5  # size of high/low sets
    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        # compute correlation for current pattern
        corr = np.correlate(current, 1.0 - current, mode='full')
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)
        # compute per-index contributions for that shift
        contrib = _contributions(current, shift)
        # select high and low indices
        if n > K:
            high_idxs = np.argpartition(contrib, -K)[-K:]
            low_idxs = np.argpartition(contrib, K)[:K]
        else:
            high_idxs = np.arange(n)
            low_idxs = np.arange(n)

        # temperature schedule
        frac = (now - start) / budget_s
        temp = max_temp * (1.0 - frac) + min_temp * frac

        improved = False
        # attempt a targeted swap
        attempts = 0

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for the given discretisation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _contributions(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Contribution of each entry to the overlap for a given shift.
    For a shift s, contribution[i] = h[i] * (1 - h[i+s]) where out‑of‑range
    h[…] is treated as 0.
    """
    n = h.shape[0]
    contrib = np.empty(n, dtype=float)
    if shift >= 0:
        ov = n - shift
        if ov > 0:
            contrib[:ov] = h[:ov] * (1.0 - h[shift:])
        if ov < n:
            contrib[ov:] = h[ov:]          # out‑of‑range part gives h[i]
    else:
        s = -shift
        ov = n - s
        if ov > 0:
            contrib[s:] = h[s:] * (1.0 - h[:ov])
        if s > 0:
            contrib[:s] = h[:s]            # out‑of‑range part gives h[i]
    return contrib


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Length‑n array with values in [0,1].
    c5_bound : float
        C5 value attained by h_values.
    n_points : int
        Number of discretisation points.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Problem size – even so a binary pattern with exact sum is feasible
    # --------------------------------------------------------------
    n = 200                     # moderate size, fast evaluation
    half = n // 2               # number of ones required (exact sum)
    # initialise a random binary feasible vector
    current = np.zeros(n, dtype=float)
    ones = rng.choice(n, half, replace=False)
    current[ones] = 1.0
    current_c5 = _compute_c5(current)

    best = current.copy()
    best_c5 = current_c5

    # annealing / metropolis schedule
    max_temp = 0.2
    min_temp = 1e-4
    stagnation_time = 30.0      # seconds without improvement → re‑initialise
    last_impr = start
    K = 5                       # size of high/low contribution sets

    while True:
        now = time.time()
        if now - start >= budget_s:
            break

        # ----------------------------------------------------------
        # identify worst shift and per‑index contributions
        # ----------------------------------------------------------
        corr = np.correlate(current, 1.0 - current, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)               # shift in index units
        contrib = _contributions(current, shift)

        # high‑contribution (likely a 1 in a bad region) and low‑contribution
        if n > K:
            high_idxs = np.argpartition(contrib, -K)[-K:]
            low_idxs = np.argpartition(contrib, K)[:K]
        else:
            high_idxs = np.arange(n)
            low_idxs = np.arange(n)

        # temperature for this iteration
        frac = (now - start) / budget_s
        temp = max_temp * (1.0 - frac) + min_temp * frac

        improved = False

        # ----------------------------------------------------------
        # try a targeted swap between a high‑ and a low‑contribution index
        # ----------------------------------------------------------
        for _ in range(5):
            i = int(rng.choice(high_idxs))
            j = int(rng.choice(low_idxs))
            if i == j or current[i] == current[j]:
                continue
            cand = current.copy()
            cand[i], cand[j] = current[j], current[i]
            cand_c5 = _compute_c5(cand)
            delta = cand_c5 - current_c5
            if delta < 0.0 or rng.random() < math.exp(-delta / temp):
                current = cand
                current_c5 = cand_c5
                improved = True
                if cand_c5 < best_c5:
                    best = cand.copy()
                    best_c5 = cand_c5
                    last_impr = now
                break

        # ----------------------------------------------------------
        # fallback: a random feasible swap
        # ----------------------------------------------------------
        if not improved:
            i, j = rng.integers(0, n, size=2)
            while i == j or current[i] == current[j]:
                i, j = rng.integers(0, n, size=2)
            cand = current.copy()
            cand[i], cand[j] = current[j], current[i]
            cand_c5 = _compute_c5(cand)
            delta = cand_c5 - current_c5
            if delta < 0.0 or rng.random() < math.exp(-delta / temp):
                current = cand
                current_c5 = cand_c5
                if cand_c5 < best_c5:
                    best = cand.copy()
                    best_c5 = cand_c5
                    last_impr = now

        # ----------------------------------------------------------
        # if no improvement for a while, reshuffle to escape plateaus
        # ----------------------------------------------------------
        if now - last_impr > stagnation_time:

            ones = rng.choice(n, half, replace=False)
            current[ones] = 1.0
            current_c5 = _compute_c5(current)
            last_impr = now

    # best solution already respects the sum constraint exactly (binary)
    return best, best_c5, int(n)
