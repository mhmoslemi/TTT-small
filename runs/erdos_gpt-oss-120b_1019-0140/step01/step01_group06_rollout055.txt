def make_h(n):
    x = np.linspace(0,2,n,endpoint=False)
    h = (1 - np.cos(np.pi * x)) / 2
    # ensure sum exactly
    target = n / 2
    cur = h.sum()
    if cur != target:
        h = h * (target / cur)
    # clip just in case
    h = np.clip(h, 0.0, 1.0)
    # final adjustment to get exact sum: compute diff = target - sum(h)
    # add diff to the element with largest capacity (maybe the one not at 0 or 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # find index with largest capacity (if diff>0, capacity = 1 - h[i]; else capacity = h[i])
        if diff > 0:
            capacities = 1 - h
        else:
            capacities = h
        idx = np.argmax(capacities)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    cur = h.sum()

import time

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C₅ for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_to_feasible(h: np.ndarray, target_sum: float, tol: float = 1e-12) -> np.ndarray:
    """Project h onto [0,1]^n ∩ {sum = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) <= tol:
        return h
    # iterative proportional redistribution
    for _ in range(10):
        if abs(diff) <= tol:
            break
        caps = (1.0 - h) if diff > 0 else h
        total_cap = caps.sum()

            break
        delta = diff * caps / total_cap

        diff = target_sum - h.sum()

    if abs(diff) > tol:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h
