def run(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 256
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            # if different length, we can interpolate
            # Up-sample/dow-sample using linear interpolation.
            init = np.interp(np.linspace(0, 1, n, endpoint=False), np.linspace(0, 1, init.size, endpoint=False), init)
        # convert to binary: top half ones
        sorted_idx = np.argsort(-init) # descending
        h = np.zeros(n, dtype=int)
        h[sorted_idx[:n//2]] = 1
    else:
        # random balanced binary
        h = np.zeros(n, dtype=int)
        ones = rng.choice(n, size=n//2, replace=False)
        h[ones] = 1
    dx = 2.0 / n

        # arr is integer 0/1
        # compute correlation of arr and 1-arr
        corr = np.correlate(arr, 1 - arr, mode='full')
        return corr.max()
    best_h = h.copy()
    best_score = compute_c5_binary(h) * dx
    curr_h = h.copy()
    curr_score = best_score
    # SA parameters
    T0 = 0.5 * dx # about 0.0039?

    decay = 0.99995

    # precompute indices of ones and zeros for quick sampling
    # We'll maintain lists of indices sets.
    ones_idx = np.flatnonzero(curr_h == 1)
    zeros_idx = np.flatnonzero(curr_h == 0)

    iter = 0
    while time.time() - start < budget_s * 0.95:

        # pick random swap
        i = rng.choice(ones_idx)

import time, math
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of arr (length m) to length new_n."""
    m = int(arr.shape[0])
    if m == new_n:
        return arr.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr).astype(float)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx for a float array h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_binary(arr):
    """C5 for binary (0/1) array; returns scaled value."""
    n = arr.size
    dx = 2.0 / n
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * dx

def thue_morse(n):
    """Binary Thue–Morse sequence (exact half ones for powers of two)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                      dtype=int, count=n)
    return seq

def binary_from_continuous(v):
    """Convert a real vector to a balanced binary vector (top half ones)."""
    n = v.shape[0]
    idx = np.argsort(-v)  # descending
    h = np.zeros(n, dtype=int)
    h[idx[:n // 2]] = 1
    return h

def compute_gradient(h, shift):
    """Gradient of the active max-correlation term."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[: n - shift] = h[shift:]
        term2[shift:] = h[: n - shift]
    else:
        s = -shift
        term1[s:] = h[: n - s]
        term2[: n - s] = h[s:]
    return 1.0 - term1 - term2

def projected_gradient_descent(h0, n_iter=2000, step0=0.5):
    """Projected gradient descent on the max-correlation objective."""
    h = h0.astype(float).copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def sa_optimize(start_h, rng, deadline, max_iters=200000):
    """Simple simulated‑annealing on binary vectors."""
    n = start_h.size
    dx = 2.0 / n
    curr_h = start_h.copy()
    curr_score = compute_c5_binary(curr_h)
    best_h = curr_h.copy()
    best_score = curr_score
    T = max(best_score * 0.5, 1e-3)
    decay = 0.99995
    no_improve = 0
    ones_idx = np.flatnonzero(curr_h == 1)
    zeros_idx = np.flatnonzero(curr_h == 0)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        new_h = curr_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_score = compute_c5_binary(new_h)
        delta = new_score - curr_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            curr_h = new_h
            curr_score = new_score
            ones_idx = np.flatnonzero(curr_h == 1)
            zeros_idx = np.flatnonzero(curr_h == 0)
            if curr_score < best_score - 1e-12:
                best_h = curr_h.copy()
                best_score = curr_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= decay
        if no_improve >= 20000:
            # restart from a fresh random balanced vector
            curr_h = np.zeros(n, dtype=int)
            curr_h[rng.choice(n, size=n // 2, replace=False)] = 1
            curr_score = compute_c5_binary(curr_h)
            ones_idx = np.flatnonzero(curr_h == 1)
            zeros_idx = np.flatnonzero(curr_h == 0)
            no_improve = 0
            T = max(best_score * 0.5, 1e-3)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine grid size
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        if n % 2 != 0:
            # enforce even length by upsampling to next even size
            n = n + 1 - (n % 2)
            init_arr = upsample(init_arr, n)
    else:
        n = 256
        init_arr = None

    # Build a list of promising seeds
    seeds = []
    if init_arr is not None:
        seeds.append(binary_from_continuous(init_arr))
    if (n & (n - 1)) == 0:  # power of two
        seeds.append(thue_morse(n))
    for _ in range(3):
        h = np.zeros(n, dtype=int)
        h[rng.choice(n, size=n // 2, replace=False)] = 1
        seeds.append(h)

    # Pick the best initial seed according to the binary C5 value
    best_score = float('inf')
    best_seed = None
    for s in seeds:
        sc = compute_c5_binary(s)
        if sc < best_score:
            best_score = sc
            best_seed = s

    # Simulated annealing phase
    time_limit = start_time + budget_s * 0.85
    best_h_bin, best_c5_bin = sa_optimize(best_seed, rng, time_limit)

    # If time remains, run a short gradient refinement on the best float version
    remaining = start_time + budget_s - time.time()
    if remaining > 5.0:
        h_float = best_h_bin.astype(float)
        # small number of PGD iterations proportional to remaining time
        n_iter = max(500, int(min(4000, remaining * 800)))
        h_opt, c5_opt = projected_gradient_descent(h_float, n_iter=n_iter, step0=0.5)
        if c5_opt < best_c5_bin:
            best_h_bin = h_opt
            best_c5_bin = c5_opt

    # Final projection to guarantee feasibility (tiny numerical drift possible)
    best_h = project_onto_box_sum(best_h_bin, n / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
