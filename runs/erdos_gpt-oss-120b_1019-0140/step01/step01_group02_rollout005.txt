    for idx in idxs[1:]:
        val = fitness[idx]
        if val < best_val:
            best_local = idx
            best_val = val

def _repair_sum(h, target):
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # choose indices that can accommodate diff
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: spread uniformly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        # recursion
        return _repair_sum(h, target)
    return h
```

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry (or distribute uniformly) to hit the exact sum."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: spread diff uniformly and clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h


def _tournament_select(fitness, k, rng):
    """Return index of best (lowest) fitness among k random competitors."""
    idxs = rng.integers(0, len(fitness), size=k)
    best = idxs[0]
    best_val = fitness[best]
    for idx in idxs[1:]:
        val = fitness[idx]
        if val < best_val:
            best = idx
            best_val = val
    return best


def _crossover(p1: np.ndarray, p2: np.ndarray, rng) -> np.ndarray:
    """One‑point crossover."""
    n = p1.shape[0]
    cut = rng.integers(1, n)  # avoid trivial cuts
    return np.concatenate((p1[:cut], p2[cut:]))


def _mutate_swap(child: np.ndarray, step: float, rng) -> np.ndarray:
    """Swap‑delta mutation preserving total sum (up to rounding)."""
    n = child.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return child
    hi = child[i]
    hj = child[j]
    # feasible delta keeping both entries in [0,1]
    delta_min = max(-hi, -(1.0 - hj))
    delta_max = min(1.0 - hi, hj)
    low = max(delta_min, -step)
    high = min(delta_max, step)
    if low >= high:
        return child
    delta = rng.random() * (high - low) + low
    child[i] = hi + delta
    child[j] = hj - delta
    return child

    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values:   1‑D numpy array of length n_points, values in [0,1]
        c5_bound:   the achieved C5 value for h_values
        n_points:   number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- Determine discretisation size and initial vector -----------------
    n_target = 200  # moderate size for speed vs resolution
    try:
        init0 = np.asarray(initial_h_values, dtype=float)
        n0 = init0.shape[0]
        if n0 != n_target:
            # resample by linear interpolation on [0,2)
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n_target, endpoint=False)
            init = np.interp(x_new, x_old, init0)
        else:
            init = init0.copy()
    except NameError:
        # No supplied construction – start from uniform random values
        init = rng.random(n_target)

    np.clip(init, 0.0, 1.0, out=init)
    target_sum = n_target / 2.0
    init = _repair_sum(init, target_sum)

    # ---- Build initial population -----------------------------------------
    pop_size = 50
    population = [init.copy()]
    fitness = [_compute_c5(init)]

    while len(population) < pop_size:

        np.clip(cand, 0.0, 1.0, out=cand)

            fitness[worst_idx] = new_c5
            stagnation = 0

    # ---- Local refinement with simple simulated annealing ------------------
    local_h = best_h.copy()
    local_c5 = best_c5
    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        frac = (now - start) / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        i = rng.integers(0, n_target)
        j = rng.integers(0, n_target)
        if i == j:
            continue
        hi = local_h[i]
        hj = local_h[j]

        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        low = max(delta_min, -step)
        high = min(delta_max, step)
        if low >= high:
            continue
        delta = rng.random() * (high - low) + low

        local_h[i] = hi + delta
        local_h[j] = hj - delta

        delta_obj = new_c5 - local_c5

            local_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = local_h.copy()
        else:
            # revert
            local_h[i] = hi
            local_h[j] = hj

    # final repair and evaluation
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n_target
