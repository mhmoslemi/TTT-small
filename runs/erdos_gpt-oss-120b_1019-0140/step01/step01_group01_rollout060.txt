import time
import numpy as np

def compute_c5_and_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    max_corr = corr[max_idx]
    c5 = float(max_corr * (2.0 / n))
    k = max_idx - (n - 1)
    return c5, k, corr

def adjust_sum_to_target(h, target_sum, rng):
    total = h.sum()
    diff = target_sum - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 1000
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

import time
import numpy as np

def compute_c5_and_shift(h: np.ndarray):
    """Return C5 value, shift where the maximum occurs, and the full correlation array."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    max_corr = corr[max_idx]
    c5 = float(max_corr * (2.0 / n))
    k = max_idx - (n - 1)
    return c5, k, corr


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState):
    """Adjust array h (in‑place) so that its sum equals target while keeping 0 ≤ h ≤ 1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 1000
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h
