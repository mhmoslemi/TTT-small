def hill_climb_binary(h, cur_val, rng, time_limit, start_time):
    best_h = h.copy()
    best_val = cur_val
    current_h = h.copy()
    current_val = cur_val
    # temperature schedule: start temperature = 0.01 maybe? t0 = 0.01
    t0 = 0.01
    # Actually we can start with t = 0.001? We'll see.

    while time.time() - start_time < time_limit:
        # generate neighbor: swap a 1 and a 0
        ones = np.flatnonzero(current_h == 1)
        zeros = np.flatnonzero(current_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # produce candidate
        cand = current_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5(cand)
        # acceptance
        if cand_val < current_val:
            current_h = cand
            current_val = cand_val
            if cand_val < best_val:
                best_h = cand
                best_val = cand_val
        else:
            # compute probability
            # use temperature decreasing linearly with elapsed fraction
            elapsed = time.time() - start_time
            frac = elapsed / time_limit
            temp = t0 * (1 - frac) + 1e-8
            # acceptance probability
            if rng.random() < np.exp(-(cand_val - current_val) / temp):
                current_h = cand
                current_val = cand_val

    return best_h, best_val

import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with 1-h
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _de_objective_factory(n, target_sum):
    """Return a function that maps x (n‑1 free vars) to C5,
    penalising infeasible last coordinate."""
    def obj(x):
        h = np.empty(n, dtype=float)
        h[:-1] = x
        last = target_sum - x.sum()
        # hard penalty if last variable violates bounds
        if (last < 0.0) or (last > 1.0):
            return 10.0
        h[-1] = last
        return _compute_c5(h)
    return obj

def _local_refine(h, best_val, budget, start):
    """Simple continuous hill‑climbing preserving the sum."""
    n = h.shape[0]
    target_sum = n / 2.0
    cur_h = h.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() - start < budget:
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = cur_h[i], cur_h[j]
        # direction: move from i to j or reverse
        if rng.random() < 0.5:
            max_delta = min(hi, 1.0 - hj)
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            max_delta = min(hj, 1.0 - hi)
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            cand = cur_h.copy()
            cand[i] += delta
            cand[j] -= delta
        cand_val = _compute_c5(cand)
        if cand_val < cur_val:
            cur_h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand
        # small random acceptance to escape plateaus
        else:
            # temperature‑free acceptance with tiny probability
            if rng.random() < 1e-4:
                cur_h = cand
                cur_val = cand_val
    return best_h, best_val
