def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # ensure float
    corr = np.correlate(h.astype(np.float64), 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_binary_vector(n, rng):
    # exactly n//2 ones
    h = np.zeros(n, dtype=np.uint8)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1
    return h

def simulated_annealing(start_h, start_cost, rng, start_time, budget_s, target):
    n = len(start_h)
    current = start_h.copy()
    current_cost = start_cost
    best = start_h.copy()
    best_cost = start_cost
    T0 = 0.02  # initial temperature (modify later)
    # maximum iterations based on time
    while time.time() - start_time < budget_s - 0.5:
        elapsed = time.time() - start_time
        fraction = elapsed / budget_s
        temp = T0 * max(0.0, 1 - fraction)  # linear cooling to zero
        # propose swap
        i = rng.integers(n)
        j = rng.integers(n)
        if current[i] == current[j]:
            continue
        # swap
        current[i], current[j] = current[j], current[i]
        new_cost = compute_c5(current)
        delta = new_cost - current_cost
        accept = False
        if delta < 0:
            accept = True
        else:
            if temp > 0 and rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            current_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best = current.copy()
                if best_cost < target:
                    break
        else:
            # revert
            current[i], current[j] = current[j], current[i]
    return best, best_cost

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap for binary (or float) step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # use float conversion for safety
    corr = np.correlate(h.astype(np.float64), 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.uint8)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1
    return h

def hill_climb_exhaustive(h):
    """Deterministic improvement by trying all single swaps."""
    best = h.copy()
    best_cost = compute_c5(best)
    improved = True
    while improved:
        improved = False
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        for i in ones:
            for j in zeros:
                best[i], best[j] = 0, 1
                new_cost = compute_c5(best)
                if new_cost + 1e-12 < best_cost:
                    best_cost = new_cost
                    improved = True
                    break
                best[i], best[j] = 1, 0
            if improved:
                break
    return best, best_cost

def simulated_annealing(start_h, start_cost, rng, start_time, budget_s, target):
    """Simple SA on binary vectors using swap moves."""
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_cost = start_cost
    best = cur.copy()
    best_cost = cur_cost
    T0 = 0.06                       # initial temperature
    while True:
        now = time.time()
        if now - start_time >= budget_s - 0.1:
            break
        # linear cooling
        frac = (now - start_time) / budget_s
        temp = T0 * max(0.0, 1.0 - frac)
        # propose a swap of opposite bits
        i = rng.integers(n)
        j = rng.integers(n)
        if cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_cost = compute_c5(cur)
        delta = new_cost - cur_cost
        accept = False

        elif temp > 0 and rng.random() < np.exp(-delta / temp):

        if accept:

            if new_cost < best_cost:
                best_cost = new_cost
                best = cur.copy()
                if best_cost < target:
                    return best, best_cost
        else:
            cur[i], cur[j] = cur[j], cur[i]   # revert
