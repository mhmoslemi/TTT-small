def _anneal_opt(v, n, time_limit):
    # Simulated annealing / hill climbing
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    cur_v = best_v.copy()
    cur_score = best_score

    # Temperature schedule
    T0 = 0.01 # guess
    # Maybe adjust T0 based on initial best?
    # Actually compute typical improvement magnitude: may be up to 0.1.
    T0 = 0.05
    T = T0
    decay = 0.9999
    # Determine max delta per move: step_max = 0.1 (like previous)
    step_max = 0.1 # can be tuned

    while time.time() - start < time_limit:
        # Random move
        i, j = np.random.choice(n, 2, replace=False)
        # Determine feasible delta
        if np.random.rand() < 0.5:
            # move from j to i
            max_delta = min(step_max, 1.0 - cur_v[i], cur_v[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            cand = cur_v.copy()
            cand[i] += delta
            cand[j] -= delta
        else:
            # move from i to j
            max_delta = min(step_max, cur_v[i], 1.0 - cur_v[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            cand = cur_v.copy()
            cand[i] -= delta
            cand[j] += delta

        # Evaluate
        cand_score = _compute_c5(cand)
        # Accept or not
        if cand_score <= cur_score or np.random.rand() < math.exp(-(cand_score - cur_score)/T):
            cur_v = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_v = cand.copy()
        # Temperature decay
        T = max(T * decay, 1e-6)
    return best_v, best_score

def _anneal_opt(v0, n, time_limit):
    start = time.time()
    cur = v0.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score

    T0 = 0.2
    T_end = 1e-4
    step0 = 0.2
    step_end = 1e-3

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        frac = 1.0 - (elapsed / time_limit)
        T = T_end + (T0 - T_end) * frac
        step_max = step_end + (step0 - step_end) * frac

        # pick distinct i,j
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # decide direction
        # compute feasible delta for moving from j to i
        max_delta_pos = min(step_max, 1.0 - cur[i], cur[j])
        max_delta_neg = min(step_max, cur[i], 1.0 - cur[j])
        # randomly choose direction weighted by feasible max
        if max_delta_pos <= 0 and max_delta_neg <= 0:
            continue
        if max_delta_pos > 0 and max_delta_neg > 0:
            # pick direction with probability proportional to max delta
            if np.random.rand() < max_delta_pos / (max_delta_pos + max_delta_neg):
                direction = 1
                max_delta = max_delta_pos
            else:
                direction = -1
                max_delta = max_delta_neg
        elif max_delta_pos > 0:
            direction = 1
            max_delta = max_delta_pos
        else:
            direction = -1
            max_delta = max_delta_neg

        delta = np.random.rand() * max_delta
        cand = cur.copy()
        if direction == 1:
            cand[i] += delta
            cand[j] -= delta
        else:
            cand[i] -= delta
            cand[j] += delta

        # candidate is feasible by construction
        cand_score = _compute_c5(cand)
        # acceptance
        if cand_score <= cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
    return best, best_score

import time
import math
import numpy as np

def _compute_c5(v):
    """Compute C5 = max_k dx * correlation(v, 1 - v)."""
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _repair_vector(v, n, max_iter=15):
    """Project onto [0,1]^n with exact sum n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _upsample_vector(v, n_target):
    """Linear interpolation to a larger grid, then repair."""
    n_src = v.shape[0]
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _anneal_opt(v0, n, time_limit):
    """Simulated annealing with exchange moves."""
    start = time.time()
    cur = v0.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score

    T0, T_end = 0.2, 1e-4
    step0, step_end = 0.2, 1e-3

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        frac = 1.0 - (elapsed / time_limit)  # 1 -> early, 0 -> final
        T = T_end + (T0 - T_end) * frac
        step_max = step_end + (step0 - step_end) * frac

        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # feasible deltas for the two directions
        max_pos = min(step_max, 1.0 - cur[i], cur[j])   # move from j to i
        max_neg = min(step_max, cur[i], 1.0 - cur[j])   # move from i to j
        if max_pos <= 0 and max_neg <= 0:
            continue
        if max_pos > 0 and max_neg > 0:
            if np.random.rand() < max_pos / (max_pos + max_neg):
                direction = 1
                max_delta = max_pos
            else:
                direction = -1
                max_delta = max_neg
        elif max_pos > 0:
            direction = 1
            max_delta = max_pos
        else:
            direction = -1
            max_delta = max_neg

        delta = np.random.rand() * max_delta
        cand = cur.copy()
        if direction == 1:
            cand[i] += delta
            cand[j] -= delta
        else:
            cand[i] -= delta
            cand[j] += delta

        cand_score = _compute_c5(cand)
        if cand_score <= cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):

    else:

    cur = _repair_vector(cur, target_n)

    best_v = _repair_vector(best_v, target_n)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(target_n)
