Pseudo:

def compute_w(h, lag):
    n = len(h)
    w = np.zeros(n)
    if lag > 0:
        w[lag:] += h[:-lag]
        w[:n-lag] += h[lag:]
    elif lag < 0:
        L = -lag
        w[:n-L] += h[L:]
        w[L:] += h[:n-L]
    else:
        # lag = 0
        w = 2*h
    return w

Note: For lag=0, w_i = h_i + h_i = 2h_i.

def _repair_sum(arr, target):
    cur = arr.sum()
    diff = target - cur
    if abs(diff) < 1e-12: return arr
    # attempt to adjust one entry that can accommodate diff
    if diff > 0:
        idxs = np.where(arr < 1 - 1e-12)[0]
    else:
        idxs = np.where(arr > 1e-12)[0]
    if idxs.size == 0:
        # fallback: distribute uniformly then clip
        arr += diff / arr.size
        np.clip(arr, 0.0, 1.0, out=arr)
        return arr
    i = idxs[0]  # pick first
    arr[i] = np.clip(arr[i] + diff, 0.0, 1.0)
    # if after clipping still diff remains, recurse
    new_diff = target - arr.sum()
    if abs(new_diff) > 1e-12:
        return _repair_sum(arr, target)
    return arr

def _repair_sum(arr, target):
    cur = arr.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return arr
    # Choose an index that can move
    if diff > 0:
        feasible = np.where(arr < 1 - 1e-12)[0]
    else:
        feasible = np.where(arr > 1e-12)[0]
    if feasible.size == 0:
        arr += diff / arr.size
        np.clip(arr, 0.0, 1.0, out=arr)
        return arr
    i = feasible[0]
    arr[i] = np.clip(arr[i] + diff, 0.0, 1.0)
    # Recursively fix any remaining diff
    if abs(arr.sum() - target) > 1e-12:
        return _repair_sum(arr, target)
    return arr

def _compute_c5(arr):
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return float(np.max(corr) * dx)

    return best_h, best_c5, n

import time
import math
import numpy as np

def _repair_sum(arr: np.ndarray, target: float) -> np.ndarray:
    """Adjust a feasible vector so that its sum equals target."""
    cur = arr.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return arr
    # Choose an entry that can move in the needed direction
    if diff > 0:
        idx = np.where(arr < 1.0 - 1e-12)[0]
    else:
        idx = np.where(arr > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute evenly then clip
        arr += diff / arr.size
        np.clip(arr, 0.0, 1.0, out=arr)
        return arr
    i = idx[0]
    arr[i] = np.clip(arr[i] + diff, 0.0, 1.0)
    # If still not exact, recurse (very unlikely)
    if abs(arr.sum() - target) > 1e-12:
        return _repair_sum(arr, target)
    return arr


def _compute_c5(arr: np.ndarray) -> float:
    """Return the C5 value for a discretised step function."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * dx)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimizes C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # Initialise a feasible candidate
    # ----------------------------------------------------------------------
    n_target = 200  # moderate size, good speed / quality trade‑off
    try:
        src = np.asarray(initial_h_values, dtype=float)
        n_src = src.shape[0]
        n = n_target
        # linear interpolation from the supplied vector to the desired length
        src_interp = np.interp(
            np.arange(n), np.linspace(0, n - 1, n_src), src
        )
        src_interp = np.clip(src_interp, 0.0, 1.0)
        target_sum = n / 2.0
        current = _repair_sum(src_interp, target_sum)
    except NameError:
        # No supplied vector – start from a random half‑ones pattern
        n = n_target
        target_sum = n / 2.0
        current = np.zeros(n, dtype=float)
        half = n // 2
        current[:half] = 1.0
        if n % 2:
            current[half] = 0.5
        rng.shuffle(current)
        current = _repair_sum(current, target_sum)

    # Ensure exact feasibility
    target_sum = n / 2.0
    current = _repair_sum(current, target_sum)

    current_c5 = _compute_c5(current)
    best_h = current.copy()
    best_c5 = current_c5

    # ----------------------------------------------------------------------
    # Simulated‑annealing / targeted descent loop
    # ----------------------------------------------------------------------
    max_step = 0.5
    min_step = 0.001
    max_temp = 0.01
    min_temp = 1e-6

    no_improve = 0
    restart_limit = 20000  # iterations without improvement before a restart
    it = 0

    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        it += 1
        frac = (now - start) / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # identify the lag that currently attains the maximum correlation
        corr_full = np.correlate(current, 1.0 - current, mode="full")
        l_idx = int(np.argmax(corr_full))
        lag = l_idx - (n - 1)  # signed shift

        # neighbor‑sum w_i = h_{i+lag} + h_{i-lag}
        w = np.zeros_like(current)
        if lag > 0:
            w[lag:] += current[:-lag]
            w[: n - lag] += current[lag:]
        elif lag < 0:
            L = -lag
            w[: n - L] += current[L:]
            w[L:] += current[: n - L]
        else:
            w = 2.0 * current

        # choose indices i (increase) and j (decrease) using a simple heuristic
        score_inc = w * (1.0 - current)       # high where increasing h helps
        i = int(np.argmax(score_inc))

        score_dec = w * current               # low where decreasing h hurts least
        score_dec_adj = score_dec.copy()
        score_dec_adj[i] = np.inf
        j = int(np.argmin(score_dec_adj))

        hi, hj = current[i], current[j]

        # feasible delta for moving mass from j to i
        low = max(-hi, hj - 1.0)              # lower bound (≤ 0)
        high = min(1.0 - hi, hj)              # upper bound (≥ 0)
        low = max(low, -step)
        high = min(high, step)
        if high <= 0:
            # try the opposite direction
            hi, hj = hj, hi
            i, j = j, i
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            low = max(low, -step)
            high = min(high, step)
            if high <= 0:
                continue

        max_delta = min(high, step)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta   # positive transfer amount

        # Apply move
        current[i] = hi + delta
        current[j] = hj - delta

        new_c5 = _compute_c5(current)
        delta_obj = new_c5 - current_c5

        accept = False
        if delta_obj < 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta_obj / temp):
                accept = True

        if accept:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            current[i] = hi
            current[j] = hj
            no_improve += 1

        # occasional full permutation to escape plateaus
        if it % 10000 == 0:
            rng.shuffle(current)
            current_c5 = _compute_c5(current)

        # occasional repair of tiny drift
        if it % 25000 == 0:
            diff = target_sum - current.sum()
            if abs(diff) > 1e-12:
                if diff > 0:
                    idxs = np.where(current < 1.0 - 1e-12)[0]
                else:
                    idxs = np.where(current > 1e-12)[0]
                if idxs.size:
                    idx = rng.choice(idxs)
                    current[idx] = np.clip(current[idx] + diff, 0.0, 1.0)

        # restart if stagnated
        if no_improve >= restart_limit:
            # fresh random half‑ones start
            current = np.zeros(n, dtype=float)
            half = n // 2
            current[:half] = 1.0
            if n % 2:
                current[half] = 0.5
            rng.shuffle(current)
            current = _repair_sum(current, target_sum)
            current_c5 = _compute_c5(current)
            no_improve = 0

    # ----------------------------------------------------------------------
    # Finalise and return best solution found
    # ----------------------------------------------------------------------
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, n
