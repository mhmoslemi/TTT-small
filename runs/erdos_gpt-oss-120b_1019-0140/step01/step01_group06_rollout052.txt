def binary_anneal(n, time_limit, start_time, seed):
    rng_local = np.random.default_rng(seed)
    # initial binary vector with half ones
    h = np.zeros(n)
    # random shuffle indices
    idx = rng_local.permutation(n)
    h[idx[:n//2]] = 1.0
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    # maintain ones_arr, zeros_arr
    ones_arr = np.where(cur == 1)[0]
    zeros_arr = np.where(cur == 0)[0]
    # annealing parameters
    T = 0.01 # initial temperature
    cooling = 0.9995
    # loop
    iters = 0
    while True:
        # check time occasionally
        if iters % 100 == 0:
            if time.time() - start_time > time_limit:
                break
        # propose swap
        i_idx = rng_local.integers(0, len(ones_arr))
        j_idx = rng_local.integers(0, len(zeros_arr))
        i = ones_arr[i_idx]
        j = zeros_arr[j_idx]
        # new vector via swap
        new = cur.copy()
        new[i] = 0.0
        new[j] = 1.0
        new_c5 = compute_c5(new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng_local.random() < np.exp(-delta / T):
            cur = new
            cur_c5 = new_c5
            # update index arrays
            # swap i and j in arrays
            ones_arr[i_idx] = j
            zeros_arr[j_idx] = i
        if cur_c5 < best_c5:
            best = cur.copy()
            best_c5 = cur_c5
        # temperature update
        T *= cooling
        # Ensure non-zero
        if T < 1e-12:
            T = 1e-12
        iters += 1
    return best, best_c5

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    # Clip; then adjust sum.
    h = np.clip(h, 0.0, 1.0)
    # compute current sum, diff
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # free indices where will not exceed bounds if we adjust
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # adjust the ones that are not at bound in direction of diff
            # Determine direction: if diff >0, need to increase; find index with most margin up
            # else need to decrease; find index with most margin down
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final adjustment: if still off, adjust a single element.
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-8:
        # find index far from bound where we can adjust
        # compute remaining capacity for increase or decrease
        if cur < target_sum:
            # need increase
            caps = 1.0 - h
            i = np.argmax(caps)
            inc = min(caps[i], target_sum - cur)
            h[i] += inc
        else:
            caps = h
            i = np.argmax(caps)
            dec = min(caps[i], cur - target_sum)
            h[i] -= dec
    # final clip to guarantee bounds
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Maximum overlap integral for a discretized step function."""
    n = h.shape[0]
    N = 2 * n - 1
    H = np.fft.fft(h, n=N)
    G = np.fft.fft(1.0 - h, n=N)
    corr = np.fft.ifft(H * np.conj(G)).real
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-12):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        if diff > 0:
            slack = 1.0 - h
            i = np.argmax(slack)
            inc = min(slack[i], diff)
            h[i] += inc
        else:
            slack = h
            i = np.argmax(slack)
            dec = min(slack[i], -diff)
            h[i] -= dec
    return np.clip(h, 0.0, 1.0)


def slsqp_optimize(h0, n, max_iter=400):
    """Local SLSQP refinement under box and sum constraints."""
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            fun=lambda x: compute_c5(x),
            x0=h0,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-8, "disp": False},
        )
    except Exception:
        return None, None
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)


def binary_anneal(n, end_time, seed):
    """Simulated annealing on binary vectors (half ones) respecting sum constraint."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    idx = rng.permutation(n)[: n // 2]
    h[idx] = 1.0
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    T = 0.01
    cooling = 0.9995
    while time.time() < end_time:
        i_idx = rng.integers(0, len(ones))
        j_idx = rng.integers(0, len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        new = cur.copy()
        new[i] = 0.0
        new[j] = 1.0
        new_c5 = compute_c5(new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_c5 = new_c5
            ones[i_idx] = j
            zeros[j_idx] = i
        if cur_c5 < best_c5:
            best = cur.copy()
            best_c5 = cur_c5
        T = max(T * cooling, 1e-12)
    return best, best_c5
