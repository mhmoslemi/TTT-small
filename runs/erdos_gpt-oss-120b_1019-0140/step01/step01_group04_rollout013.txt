    n = len(v)
    dx = 2.0/n
    a = v
    b = np.flip(1.0 - v)
    size = 2**int(np.ceil(np.log2(2*n-1)))  # power of 2 at least 2n-1 for speed.

    Fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(Fa*Fb, n=size)
    corr = conv[:2*n-1]  # full correlation
    return np.max(corr) * dx

def compute_c5(v):
    n = len(v)
    dx = 2.0 / n
    a = v
    b = np.flip(1.0 - v)
    size = 1 << (2*n-1).bit_length()   # smallest power of two >= 2n-1
    # Compute FFT using np.fft.rfft for speed.
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A*B, n=size)
    corr = conv[:2*n-1]  # exactly
    return np.max(corr) * dx

def _anneal(init, time_budget):
    n = len(init)
    v = _repair_vector(init, n)   # ensure feasible
    best_v = v.copy()
    best_score = _compute_c5(v)
    start = time.time()
    # Determine temperature schedule parameters based on initial score
    T0 = max(best_score, 1e-3) * 0.2 + 0.01  # some base
    Tmin = 1e-6
    # compute exponent factor for cooling such that at end we reach Tmin
    # We'll use linear decrease factor for magnitude of moves.
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        # fraction of time passed
        frac = elapsed / time_budget
        # temperature linearly decays
        T = T0 * (1 - frac) + Tmin * frac  # Actually start high then low.
        # define step size scaled with temperature
        step = 0.2 * (T / T0) + 0.001
        # propose move
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] = np.clip(cand[i] + delta, 0.0, 1.0)
        cand[j] = np.clip(cand[j] - delta, 0.0, 1.0)
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        # Accept if better or with Metropolis probability
        if cand_score < best_score:
            best_score = cand_score
            best_v = cand
        else:
            # Accept uphill with prob exp(-(score - best_score)/T)

import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Fast C5 evaluation using FFT convolution."""
    n = v.shape[0]
    dx = 2.0 / n
    a = v
    b = np.flip(1.0 - v)
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    corr = conv[: 2 * n - 1]
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _smooth_candidate(n, amplitude=0.4):
    i = np.arange(n)
    v = 0.5 + amplitude * np.cos(np.pi * i / n)
    return _repair_vector(v, n)

def _smooth_harmonics_candidate(n):
    i = np.arange(n)
    a1 = np.random.uniform(-0.5, 0.5)
    a2 = np.random.uniform(-0.3, 0.3)
    v = 0.5 + a1 * np.cos(np.pi * i / n) + a2 * np.cos(2 * np.pi * i / n)
    return _repair_vector(v, n)

def _local_search(v, n, time_limit):
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.02
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] = np.clip(cand[i] + delta, 0.0, 1.0)
        cand[j] = np.clip(cand[j] - delta, 0.0, 1.0)
        cand = _repair_vector(cand, n)
        sc = _compute_c5(cand)
        if sc < best_score:
            best_score, best_v = sc, cand
    return best_v, best_score

def _anneal(init, time_budget):
    n = len(init)
    cur_v = _repair_vector(init, n)
    cur_score = _compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    start = time.time()
    T0 = max(cur_score, 0.01) * 0.3 + 0.02
    Tmin = 1e-6
    while time.time() - start < time_budget:
        frac = (time.time() - start) / time_budget
        T = T0 * (1.0 - frac) + Tmin * frac
        step = 0.2 * (T / T0) + 0.001
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur_v.copy()
        cand[i] = np.clip(cand[i] + delta, 0.0, 1.0)
        cand[j] = np.clip(cand[j] - delta, 0.0, 1.0)
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_v, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_v = cand_score, cand.copy()
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * step * 0.4
            cand2 = cur_v + noise
            cand2 = _repair_vector(cand2, n)
            cand2_score = _compute_c5(cand2)
            if cand2_score < cur_score or np.random.rand() < np.exp(-(cand2_score - cur_score) / max(T, 1e-12)):
                cur_v, cur_score = cand2, cand2_score
                if cand2_score < best_score:
                    best_score, best_v = cand2_score, cand2.copy()
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    total_budget = max(0.0, budget_s - 1.0)  # reserve a second for final cleanup

    # target resolution (power of two for FFT efficiency)
    target_n = 256
    if target_n > 800:
        target_n = 800

    # Prepare initial candidates
    candidates = []

    # Parent if available
    try:
        parent = np.array(initial_h_values, dtype=np.float64)  # provided by harness
        parent = _repair_vector(parent, len(parent))
        if len(parent) != target_n:
            parent = _upsample_vector(parent, target_n)
        candidates.append(parent)
    except Exception:
        pass

    # Random binary‑like and smooth seeds
    candidates.append(_random_feasible(target_n))
    candidates.append(_random_feasible(target_n))
    candidates.append(_smooth_candidate(target_n, amplitude=0.45))
    candidates.append(_smooth_candidate(target_n, amplitude=0.35))
    candidates.append(_smooth_harmonics_candidate(target_n))

    per_candidate = total_budget / len(candidates) if candidates else 0.0

    best_v = None
    best_score = float("inf")
    # Run annealing for each seed
    for seed_vec in candidates:
        elapsed = time.time() - start
        remaining = total_budget - elapsed
        if remaining <= 0.2:
            break
        time_allowed = min(per_candidate, remaining)
        v_opt, sc_opt = _anneal(seed_vec, time_allowed)
        if sc_opt < best_score:
            best_score, best_v = sc_opt, v_opt

    # Final refinement
    remaining = total_budget - (time.time() - start)
    if remaining > 0.3 and best_v is not None:
        v_ref, sc_ref = _local_search(best_v, target_n, remaining)
        if sc_ref < best_score:
            best_score, best_v = sc_ref, v_ref

    if best_v is None:
        # fallback simple construction
        best_v = np.zeros(target_n, dtype=np.float64)
        best_v[: target_n // 2] = 1.0
        best_v = _repair_vector(best_v, target_n)
        best_score = _compute_c5(best_v)

    best_v = _repair_vector(best_v, target_n)
    final_score = _compute_c5(best_v)
    return best_v.astype(np.float64), float(final_score), int(target_n)
