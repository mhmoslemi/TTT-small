def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    if factor != 1.0:
        v *= factor
        np.clip(v, 0.0, 1.0, out=v)
        s2 = v.sum()
        if abs(s2 - target) > 1e-12:
            factor2 = target / s2
            v *= factor2
            np.clip(v, 0.0, 1.0, out=v)
    return v

    return float(np.max(corr) * dx)

    idx = int(np.argmax(corr))
    shift = idx - (N - 1)
    val = corr[idx]

import numpy as np

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    if factor != 1.0:
        v *= factor
        np.clip(v, 0.0, 1.0, out=v)
        s2 = v.sum()
        if abs(s2 - target) > 1e-12:
            factor2 = target / s2
            v *= factor2
            np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _worst_shift(h: np.ndarray):
    """Return shift (int) achieving maximum overlap and its correlation value."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)
    val = float(corr[idx])
    return shift, val

import numpy as np

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    if factor != 1.0:
        v *= factor
        np.clip(v, 0.0, 1.0, out=v)
        s2 = v.sum()
        if abs(s2 - target) > 1e-12:
            factor2 = target / s2
            v *= factor2
            np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _worst_shift(h: np.ndarray):
    """Return shift (int) attaining the maximum overlap and its correlation value."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)
    val = float(corr[idx])
    return shift, val
