def pgd_with_momentum(h0, max_iter=50000, step0=0.5, momentum=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter+1):
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (h.size - 1)
        grad = compute_gradient(h, shift)
        step = step0 / np.sqrt(it)  # or step0 / it**0.5
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def pgd_adam(h0, max_iter, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):

    m = np.zeros_like(h)
    v = np.zeros_like(h)
    s = h.size/2.0

import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure the root is bracketed
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the correlation term for a given integer shift."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def pgd_subgradient(h0, max_iter=20000, step0=1.0, momentum=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (h.size - 1)
        grad = compute_gradient(h, shift)
        step = step0 / math.sqrt(it)
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climb_binary(b0, max_iter=3000, rng=None):
    """Simple simulated‑annealing hill climbing for binary vectors."""
    if rng is None:
        rng = np.random.default_rng()
    cur = b0.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.size
    # temperature schedule
    T_initial = 0.02
    T_final = 1e-4
    for it in range(1, max_iter + 1):
        # pick a random 1 and a random 0
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # propose swap
        cur[i1] = 0.0
        cur[i0] = 1.0
        new_score = compute_c5(cur)
        delta = new_score - cur_score
        temp = T_initial * (1 - it / max_iter) + T_final
        accept = delta <= 0 or rng.random() < math.exp(-delta / temp)
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i0] = 0.0
            cur[i1] = 1.0
    return best, best_score

def round_to_binary(h):
    """Convert a fractional vector to a balanced binary vector by thresholding."""
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target_score = 0.3808  # we stop early if we beat this

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 256
        h0 = project_onto_box_sum(rng.random(n), n / 2.0)

    # Keep the best solution seen
    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # Primary phase: many PGD restarts
    while time.time() - start_time < budget_s - 1.0:
        # choose a start point
        if rng.random() < 0.5:
            # perturb current best
            perturb = 0.1 * rng.standard_normal(n)
            h_start = best_h + perturb
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = project_onto_box_sum(h_start, n / 2.0)
        else:
            # fresh random start
            h_start = project_onto_box_sum(rng.random(n), n / 2.0)

        # run PGD for a moderate number of iterations
        iters = 25000
        h_opt, score_opt = pgd_subgradient(h_start, max_iter=iters,
                                            step0=1.0, momentum=0.9)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
            if best_score <= target_score:
                break

        # occasional binary refinement
        if rng.random() < 0.2:
            b_start = round_to_binary(best_h)
            b_opt, b_score = hill_climb_binary(b_start, max_iter=4000, rng=rng)
            if b_score < best_score:
                best_score = b_score
                best_h = b_opt.copy()
                if best_score <= target_score:
                    break

    # Final projection to guarantee feasibility
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
