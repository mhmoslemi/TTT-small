    def simulated_annealing(h_start, max_iters, start_temp, end_temp):
        best_h = h_start.copy()
        best_c5 = compute_c5(best_h)
        current_h = best_h.copy()
        current_c5 = best_c5
        T = start_temp
        t_factor = (end_temp/start_temp) ** (1.0 / max_iters)  # geometric cooling

        for it in range(max_iters):
            # random swap
            # select i where h=1, j where h=0
            ones = np.flatnonzero(current_h)
            zeros = np.flatnonzero(1 - current_h)  # if binary
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # swap
            new_h = current_h.copy()
            new_h[i] = 0
            new_h[j] = 1
            new_c5 = compute_c5(new_h)
            delta = new_c5 - current_c5
            if delta <= 0 or rng.random() < math.exp(-delta / T):
                current_h = new_h

import time
import math
import numpy as np

def compute_c5(arr):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse_seq(n):
    """Generate Thue‑Morse binary sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)
    return seq

def rudin_shapiro_seq(n):
    """Generate Rudin‑Shapiro binary sequence (+1/-1 → 1/0) of length n."""
    a = [1]
    while len(a) < n:
        nxt = []
        for k, v in enumerate(a):
            nxt.append(v)                                   # a_{2k}
            nxt.append(v * (1 if (k % 2 == 0) else -1))    # a_{2k+1}=(-1)^k * a_k
        a = nxt
    a = np.array(a[:n], dtype=int)
    # map +1→1, -1→0
    return (a + 1) // 2

def continuous_anneal(start_h, time_end, rng):
    """Simulated annealing on continuous variables preserving the sum."""
    n = len(start_h)
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # annealing schedule
    init_temp = 5e-3
    final_temp = 1e-7
    max_iter = 200000
    it = 0

    while time.time() < time_end and it < max_iter:
        # pick an index to increase and one to decrease
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            it += 1
            continue
        # propose a random delta (scaled with temperature)
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        dE = new_c5 - cur_c5
        # linear temperature decay
        T = init_temp * (1 - it / max_iter) + final_temp * (it / max_iter)
        if dE <= 0 or rng.random() < math.exp(-dE / (T + 1e-18)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        it += 1
    # final small correction to enforce exact sum (tiny numerical drift)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

    """Deterministic hill‑climbing on binary vectors (swap moves)."""
    n = len(start_bin)
    h = start_bin.copy()
    best_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    # attempts without improvement before giving up
    no_improve = 0
    while time.time() < time_end and no_improve < 3000:
        ones = np.flatnonzero(h)

        if ones.size == 0 or zeros.size == 0:

            start_vec = rng.choice(seeds)
        else:
            # perturb current best a little (swap two bits of its binary rounding)
            start_vec = best_h.copy()
            # make a small binary perturbation if best_h is not binary
            bin_start = (start_vec >= 0.5).astype(int)
            for _ in range(rng.integers(1, 4)):
                ones = np.flatnonzero(bin_start)
                zeros = np.flatnonzero(1 - bin_start)
                if ones.size and zeros.size:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    bin_start[i] = 0
                    bin_start[j] = 1
            start_vec = bin_start.astype(float)

        # decide which optimizer to run
        if rng.random() < 0.6:
            # continuous annealing
            cand_h, cand_c5 = continuous_anneal(start_vec, end_time, rng)
        else:
            # binary hill climbing
            start_bin = (start_vec >= 0.5).astype(int)
            cand_h, cand_c5 = binary_hill_climb(start_bin, end_time, rng)

        if cand_c5 < best_c5:
            best_h = cand_h.copy()
            best_c5 = cand_c5

        # early exit if we beat the target
        if best_c5 <= 0.3808:
            break

    # Final refinement: try a pure binary hill climb from the best rounded version
    if time.time() - start_time < budget_s - 0.3:
        bin_start = (best_h >= 0.5).astype(int)
        final_bin, final_c5 = binary_hill_climb(bin_start, start_time + budget_s - 0.1, rng)
        if final_c5 < best_c5:
            best_h = final_bin.astype(float)
            best_c5 = final_c5

    # final safety clip and exact sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        residual = target_sum - np.sum(best_h)
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.clip(residual, -best_h[i], 1 - best_h[i])
            best_h[i] += adj
            residual = target_sum - np.sum(best_h)

    return best_h, best_c5, n
