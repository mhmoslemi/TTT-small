def _c5_value(h):
    N = h.shape[0]
    dx = 2.0 / N
    a = h
    b = 1.0 - h
    # reverse b:
    b_rev = b[::-1]
    M = 1 << ((2*N-1).bit_length())
    # compute convolution via FFT
    A = np.fft.rfft(a, n=M)
    B = np.fft.rfft(b_rev, n=M)
    conv = np.fft.irfft(A*B, n=M)[:2*N-1]
    # correlation full is conv
    # Actually convolution of a and reversed b yields correlation at indices: offset from 0..2*N-2 corresponds to lag - (N-1) ... + (N-1)
    max_corr = conv.max()
    return float(max_corr * dx)

def _repair_vector(v):
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    # If still off, apply a second scaling
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h):
    N = h.shape[0]
    dx = 2.0 / N
    a = h
    b = 1.0 - h
    b_rev = b[::-1]
    # length for linear convolution
    M = 1 << ((2*N-1).bit_length())
    A = np.fft.rfft(a, n=M)
    B = np.fft.rfft(b_rev, n=M)
    conv = np.fft.irfft(A*B, n=M)[:2*N-1]
    return float(conv.max() * dx)
