def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    # compute cross-correlation of h and (1-h)
    a = h
    b = 1.0 - h
    # convolution: a * reversed(b)
    # Use FFT of length L = 2**int(np.ceil(np.log2(2*n-1)))
    L = 1 << (2*n-1).bit_length()
    fa = np.fft.rfft(a, n=L)
    fb_rev = np.fft.rfft(b[::-1], n=L)
    conv = np.fft.irfft(fa * fb_rev, n=L)
    # conv length L, but full convolution length = 2n-1, located at indices 0..2n-2
    corr = conv[:2*n-1]  # shape 2n-1
    return np.max(corr) * dx
```

def anneal_one(h, target_sum, max_iter, start_T, end_T, rng):
    # h is binary array with sum target_sum
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    # store indices of ones and zeros
    ones = np.where(cur_h > 0.5)[0]  # Should reflect positions of 1
    zeros = np.where(cur_h < 0.5)[0]
    for it in range(max_iter):
        T = start_T + (end_T - start_T) * (it / max_iter)
        # random choose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        # compute new score
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # accept with prob exp(-delta/T)
            if T > 1e-12:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            cur_score = new_score
            # update indices
            # after swap: i becomes zero, j becomes one
            # modify arrays efficiently; we can rebuild occasionally or use Python list for indices.
            # Simpler: recompute ones, zeros after each accept? Might be heavy. Instead, maintain as python lists.

            cur_h[i], cur_h[j] = 1.0, 0.0
        # update best

import numpy as np
import time
import math

def _compute_c5_fft(h):
    """Discrete max overlap C5 using FFT convolution."""
    n = h.size
    dx = 2.0 / n
    a = h.astype(np.float64)
    b = 1.0 - a
    # length for linear convolution (next power of two >= 2n)
    L = 1 << ((2 * n).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b[::-1], n=L)
    conv = np.fft.irfft(Fa * Fb, n=L)
    corr = conv[: 2 * n - 1]
    return np.max(corr) * dx


def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust sum(v)=target via water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v


def _binary_from_continuous(vec, target_sum):
    """Create a binary vector with exactly target_sum ones, preferring larger entries."""
    idx = np.argsort(-vec)  # descending
    bin_vec = np.zeros_like(vec, dtype=np.float64)
    bin_vec[idx[: int(target_sum)]] = 1.0
    return bin_vec


def _anneal_one(h, target_sum, max_iter, start_T, end_T, rng, start_time, budget_s):
    """Simple simulated annealing on binary vectors (swap 1↔0)."""
    cur_h = h.copy()
    cur_score = _compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = h.size
    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.97:
            break
        T = start_T + (end_T - start_T) * (it / max_iter)
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = _compute_c5_fft(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            accept = rng.random() < math.exp(-delta / T)
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_score = float("inf")

    # Determine base resolution
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        init_vec = np.array(init_vec, dtype=np.float64).ravel()
        base_n = init_vec.size
    else:
        base_n = 380

    # Build a modest resolution schedule, doubling while budget permits
    resolutions = [base_n]
    while True:
        if len(resolutions) >= 4:
            break
        next_n = resolutions[-1] * 2
        # reserve at least 60 s for each further resolution
        if time.time() - start_time + 60 > budget_s * 0.97:
            break
        resolutions.append(next_n)

    # Main multi‑resolution loop
    for n in resolutions:
        target_sum = n / 2.0
        # Determine remaining time and decide number of restarts
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        # Rough estimate: ~5 s per restart (including SA + optional SLSQP)
        est_per_restart = 5.0
        max_restarts = max(1, int(remaining // est_per_restart))
        max_restarts = min(max_restarts, 20)

        # Helper to create a binary start vector
        def make_start():
            if best_h is not None:
                # interpolate previous best
                x_src = np.linspace(0, 2, best_h.size, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                interp = np.interp(x_new, x_src, best_h)
                return _binary_from_continuous(interp, target_sum)
            if init_vec is not None and init_vec.size == n:
                return _binary_from_continuous(init_vec, target_sum)
            # random binary
            perm = rng.permutation(n)
            b = np.zeros(n, dtype=np.float64)
            b[perm[: int(target_sum)]] = 1.0
            return b

        for _ in range(max_restarts):
            # time guard
            if time.time() - start_time > budget_s * 0.97:
                break
            h0 = make_start()
            # Simulated annealing parameters
            max_iter = 2500 if n <= 760 else 1500
            start_T, end_T = 0.02, 0.0005
            h_opt, sc_opt = _anneal_one(
                h0,
                target_sum,
                max_iter,
                start_T,
                end_T,
                rng,
                start_time,
                budget_s,
            )
            # Optional local refinement with SLSQP (if time permits)
            if time.time() - start_time < budget_s * 0.90:
                try:
                    from scipy.optimize import minimize

                    cons = {"type": "eq", "fun": lambda x, tgt=target_sum: np.sum(x) - tgt}
                    bounds = [(0.0, 1.0)] * n
                    res = minimize(
                        _compute_c5_fft,
                        h_opt,
                        method="SLSQP",
                        bounds=bounds,
                        constraints=[cons],
                        options={"maxiter": 80, "ftol": 1e-9, "disp": False},
                    )
                    h_ref = _project_to_constraints(res.x, target_sum)
                    sc_ref = _compute_c5_fft(h_ref)
                except Exception:
                    h_ref = h_opt
                    sc_ref = sc_opt
            else:
                h_ref = h_opt
                sc_ref = sc_opt

            if sc_ref < best_score:
                best_score = float(sc_ref)
                best_h = h_ref.copy()
                # Early success
                if best_score < 0.3808:
                    break
        if best_score < 0.3808:
            break

    # Fallback if nothing found
    if best_h is None:
        n = base_n
        b = np.zeros(n, dtype=np.float64)
        b[: int(n / 2)] = 1.0
        best_h = b
        best_score = _compute_c5_fft(best_h)

    # Final evaluation using provided evaluator if available
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_h.size)
