def gradient_for_k(h, k):

def propose_move_gradient(h, max_delta, rng):
    k, _ = get_worst_k(h)
    grad = gradient_for_k(h, k)
    i_max = int(np.argmax(grad))
    i_min = int(np.argmin(grad))
    if i_max == i_min:
        # fallback to random move
        i, j = rng.integers(0, len(h), size=2)
        return i, j, max_delta * (2 * rng.random() - 1)
    # feasible delta
    delta_max = min(1.0 - h[i_max], h[i_min])
    if delta_max <= 0:
        # fallback random
        i, j = rng.integers(0, len(h), size=2)

    # limit to step magnitude
    delta_bound = min(delta_max, max_delta)
    delta = rng.random() * delta_bound  # positive delta only (move from min to max)
    return i_max, i_min, delta

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0.0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute uniformly and clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _worst_shift(h: np.ndarray):
    """Return lag (relative shift) giving minimal self‑correlation."""
    corr = np.correlate(h, h, mode="full")
    min_idx = int(np.argmin(corr))
    lag = min_idx - (h.shape[0] - 1)
    return lag, corr

def _gradient_for_lag(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the product sum for a given lag."""
    n = h.shape[0]
    if k == 0:
        return 2.0 * h
    kk = abs(k)
    left = np.concatenate((np.zeros(kk), h[: n - kk]))
    right = np.concatenate((h[kk:], np.zeros(kk)))
    return left + right

def _propose_gradient_move(h: np.ndarray, max_delta: float, rng: np.random.Generator):
    """Select a pair of indices guided by the gradient of the worst lag."""
    k, _ = _worst_shift(h)
    grad = _gradient_for_lag(h, k)
    i_max = int(np.argmax(grad))
    i_min = int(np.argmin(grad))
    if i_max == i_min:
        # fallback to a random pair
        i, j = rng.integers(0, h.size, size=2)
        low = max(-h[i], -(1.0 - h[j]))
        high = min(1.0 - h[i], h[j])
        if low >= high:
            return i, j, 0.0
        delta = rng.random() * (high - low) + low
        return i, j, delta
    # feasible increase at i_max and decrease at i_min
    delta_max = min(1.0 - h[i_max], h[i_min])
    if delta_max <= 0.0:
        # no feasible move, fall back
        i, j = rng.integers(0, h.size, size=2)
        low = max(-h[i], -(1.0 - h[j]))
        high = min(1.0 - h[i], h[j])
        if low >= high:
            return i, j, 0.0
        delta = rng.random() * (high - low) + low
        return i, j, delta
    delta = rng.random() * min(delta_max, max_delta)
    return i_max, i_min, delta

    """
