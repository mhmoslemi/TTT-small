def _compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    m = 1 << int(np.ceil(np.log2(2 * n)))
    H = np.fft.fft(h, m)
    G = np.fft.fft(1.0 - h, m)
    corr = np.fft.ifft(H * np.conj(G)).real
    corr = corr[:2 * n - 1]
    return np.max(corr) * dx

def _initialize_population(n, pop_size, target_sum, init_vec=None):
    pop = []
    if init_vec is not None:
        # Include the provided init vector
        v = _project_to_constraints(np.array(init_vec, dtype=np.float64).ravel(), target_sum)
        pop.append(v)
    # generate remaining random
    while len(pop) < pop_size:
        v = np.random.rand(n)
        v = _project_to_constraints(v, target_sum)
        pop.append(v)
    return pop

def _tournament(pop_fitness, tournament_size=3):
    # returns index
    competitors = np.random.choice(len(pop_fitness), tournament_size, replace=False)
    best = competitors[0]
    best_fit = pop_fitness[best]
    for idx in competitors[1:]:
        if pop_fitness[idx] < best_fit:
            best_fit = pop_fitness[idx]
            best = idx
    return best

def _crossover(parent1, parent2):
    n = parent1.size
    i, j = sorted(np.random.choice(n, 2, replace=False))
    child = np.concatenate([parent1[:i], parent2[i:j], parent1[j:]])
    return child

def _mutate(child, sigma):
    child = child + sigma * np.random.randn(child.size)
    return child

import time
import numpy as np
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust its sum to `target` via water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-9 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5_fft(h):
    """Discrete C5 using FFT: max_k sum h_i*(1-h_{i+k}) * dx."""
    n = h.shape[0]
    dx = 2.0 / n
    m = 1 << int(np.ceil(np.log2(2 * n)))  # next power of two ≥2n
    H = np.fft.fft(h, m)
    G = np.fft.fft(1.0 - h, m)
    corr = np.fft.ifft(H * np.conj(G)).real
    corr = corr[: 2 * n - 1]
    return np.max(corr) * dx

def _initialize_population(n, pop_size, target_sum, init_vec=None):
    pop = []
    if init_vec is not None:
        vec = np.array(init_vec, dtype=np.float64).ravel()
        pop.append(_project_to_constraints(vec, target_sum))
    while len(pop) < pop_size:
        vec = np.random.rand(n)
        pop.append(_project_to_constraints(vec, target_sum))
    return pop

def _tournament_select(fitness, k=3):
    n = len(fitness)
    kk = min(k, n)
    idx = np.random.choice(n, kk, replace=False)
    best = idx[0]
    best_val = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_val:
            best = i
            best_val = fitness[i]
    return best

def _crossover_two_point(p1, p2):
    n = p1.size
    i, j = np.sort(np.random.choice(n, 2, replace=False))
    return np.concatenate([p1[:i], p2[i:j], p1[j:]])

def _mutate(vec, sigma):
    return vec + sigma * np.random.randn(vec.size)

def _local_hill_climb(h, target_sum, max_iters, sigma, time_left):
    best = h.copy()
    best_score = _compute_c5_fft(best)
    for _ in range(max_iters):
        if time_left() < 0.005:
            break
        cand = best.copy()
        idx = np.random.randint(cand.size)
        cand[idx] += sigma * np.random.randn()
        cand = _project_to_constraints(cand, target_sum)
        sc = _compute_c5_fft(cand)
        if sc < best_score:
            best_score = sc
            best = cand
    return best, best_score
