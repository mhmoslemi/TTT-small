class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        h = x[:self.n]
        return np.sum(h) - self.n/2.0

class OverlapConstraint:
    def __init__(self, shift, n):
        self.shift = shift
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        shift = self.shift
        if shift >= 0:
            overl = np.dot(h[:self.n - shift], 1.0 - h[shift:])
        else:
            kk = -shift
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        t = x[-1]
        return t - overl * self.dx

```
def hill_climb(h, init_c5, start_time, budget_s):
    n = len(h)
    best_h = h.copy()
    best_val = init_c5
    rng = np.random.default_rng()
    while time.time() - start_time < budget_s - 0.1:
        # pick random one and zero to swap
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]  # swap
        val = compute_c5_binary(cand)
        if val + 1e-12 < best_val:
            best_h = cand
            best_val = val
    return best_h, best_val
```

```python
def repair_balance(ind, target_sum, rng):
    diff = int(ind.sum() - target_sum)
    if diff > 0:
        # too many ones: turn diff ones to zero
        one_idxs = np.flatnonzero(ind == 1)
        to_zero = rng.choice(one_idxs, diff, replace=False)
        ind[to_zero] = 0
    elif diff < 0:
        # too few ones: turn -diff zeros to one
        zero_idxs = np.flatnonzero(ind == 0)
        to_one = rng.choice(zero_idxs, -diff, replace=False)
        ind[to_one] = 1
    return ind
```

def compute_c5(h):
    """C5 overlap bound for a step function h (binary or continuous)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def repair_balance(ind, target_sum, rng):
    """Adjust binary vector to have exactly target_sum ones."""
    diff = int(ind.sum() - target_sum)
    if diff > 0:
        one_idx = np.flatnonzero(ind == 1)
        to_zero = rng.choice(one_idx, diff, replace=False)
        ind[to_zero] = 0
    elif diff < 0:
        zero_idx = np.flatnonzero(ind == 0)
        to_one = rng.choice(zero_idx, -diff, replace=False)
        ind[to_one] = 1
    return ind

def crossover(p1, p2, target_sum, rng):
    """Uniform crossover followed by repair to maintain exact sum."""
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2).astype(int)
    child = repair_balance(child, target_sum, rng)
    return child

def mutate_swap(ind, rng):
    """Swap a random 1 with a random 0 (preserves sum)."""
    ones = np.flatnonzero(ind == 1)
    zeros = np.flatnonzero(ind == 0)
    if ones.size == 0 or zeros.size == 0:
        return ind
    i = rng.choice(ones)
    j = rng.choice(zeros)
    ind[i], ind[j] = ind[j], ind[i]
    return ind

                if abs(s - n / 2.0) > 1e-9:
                    h_cont += (n / 2.0 - s) / n
                    h_cont = np.clip(h_cont, 0.0, 1.0)
                c5_cont = compute_c5(h_cont)
                if c5_cont + 1e-12 < best_c5:
                    best_h = h_cont
                    best_c5 = c5_cont
        except Exception:
            pass

    # ----- final sanity adjustments -----
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    if abs(s - n / 2.0) > 1e-9:
        best_h += (n / 2.0 - s) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
