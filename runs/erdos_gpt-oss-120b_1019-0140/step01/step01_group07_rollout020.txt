    if L >= 0:

def gradient_of_corr(h, L):

    np.random.seed(seed)
    start = time.time()
    # initial vector
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        h0 = np.array(init_vec, dtype=np.float64).ravel()
        n0 = h0.size
    else:
        n0 = 200
        h0 = np.random.rand(n0)

    # maybe apply product of randomness to get starting candidate: project
    target_sum = n0 / 2.0
    h0 = _project_to_constraints(h0, target_sum)

    # multi-res: list of n's: [n0, n0*2] depending on budget
    n_vals = [n0]
    if budget_s >= 400:
        n_vals.append(n0*2)
    if budget_s >= 800:
        n_vals.append(n0*4)

    best_h = None
    best_score = float('inf')
    # store best across all res
    for n in n_vals:
        # Get start vector scaled to n
        if n == n0:
            h_start = h0.copy()
        else:
            # interpolate from best at previous (if exists) else from h0
            src = best_h if best_h is not None else h0
            src_n = src.size
            x_src = np.linspace(0, 2, src_n, endpoint=False)  # not include endpoint
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_start = np.interp(x_new, x_src, src)
        target_sum = n / 2.0
        h_start = _project_to_constraints(h_start, target_sum)

        # optimization parameters
        max_iters = 5000
        # adapt steps based on n
        step = 0.1  # base step (on gradient)
        # Reduce step magnitude as n grows
        step = step * 2.0 / n  # scaling: ensure similar absolute change? Actually gradient already has dx factor.
        # maybe step = 0.3

        # track time for this resolution
        while time.time() - start < budget_s * 0.97:
            # Compute C5 and corr
            c5, corr = _compute_c5_and_corr(h_start)  # separate function
            if c5 < best_score:
                best_score = float(c5)
                best_h = h_start.copy()
                # early exit threshold
                if best_score < 0.3808:
                    break
            # Identify worst shift
            idx = np.argmax(corr)
            L = idx - (n - 1)
            grad = _gradient_corr(h_start, L)  # compute gradient per dx
            # Gradient descent step
            h_new = h_start - step * grad
            # project
            h_new = _project_to_constraints(h_new, target_sum)
            # optional acceptance: if c5 not improved, reduce step
            # compute new c5 quickly? compute only new c5:
            new_c5 = _compute_c5(h_new)
            if new_c5 <= c5:
                # accept and maybe increase step
                h_start = h_new
                step = min(step * 1.05, 1.0)  # cap step
            else:
                # reject, reduce step
                step = step * 0.5
                # optionally apply small random perturb
                # Keep h_start unchanged
            # early break if step too small
            if step < 1e-8:
                break
            # also break if iteration limit
            max_iters -= 1
            if max_iters <= 0:
                break
        if best_score < 0.3808:
            break

    # After loops, maybe evaluate with evaluate_erdos_solution if available
    final_score = best_score
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            pass

    return best_h.astype(np.float64), final_score, int(best_h.size)

import numpy as np
import time
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    """Project vector v onto [0,1] box and enforce sum(v)=target via water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # fill the largest capacities first
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-14:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-14:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _gradient_corr(h, L):
    """Gradient of raw correlation (without dx) for shift L."""
    n = h.size
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if L >= 0:
        # forward part
        grad[:n - L] += (1.0 - h[L:])
        # backward part
        grad[L:] += -h[:n - L]
    else:
        k = -L
        # forward part (L negative)
        grad[k:] += (1.0 - h[:n - k])
        # backward part
        grad[:n - k] += -h[k:]
    return grad * dx

def _pgd_optimize(h0, target, start_time, budget):
    """Projected gradient descent on C5 using the worst shift gradient."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = _compute_c5(h)
    step = 0.2  # initial step magnitude
    max_iter = 5000
    for it in range(max_iter):
        if time.time() - start_time > budget * 0.96:
            break
        # identify worst shift
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr_raw)
        L = idx - (n - 1)
        grad = _gradient_corr(h, L)
        # descent step
        h_new = h - step * grad
        h_new = _project_to_constraints(h_new, target)
        new_score = _compute_c5(h_new)
        # accept if improvement, else shrink step
        if new_score <= best_score:
            best_score = new_score
            best_h = h_new.copy()
        if new_score <= _compute_c5(h):
            h = h_new
            step = min(step * 1.02, 1.0)
        else:
            step *= 0.5
        if step < 1e-9:
            break
    return best_h, best_score

            pass

    # Final evaluation using provided evaluator if available
    final_score = best_score
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            pass

    return best_h.astype(np.float64), final_score, int(best_h.size)
