import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _binary_from_continuous(h, target):
    k = int(round(target))
    idx = np.argpartition(-h, k)[:k]
    b = np.zeros_like(h)
    b[idx] = 1.0
    return b

def _anneal_binary(h_bin, target_sum, time_end):
    cur = h_bin.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    while time.time() < time_end:
        ones = np.flatnonzero(cur > 0.5)
        zeros = np.flatnonzero(cur < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cur[i], cur[j] = cur[j], cur[i]
        cand_score = _compute_c5(cur)
        progress = (time.time() - start) / (time_end - start)
        T = max(1e-8, 0.02 * (1 - progress))
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_score

def _slsqp_restarts(h0, target_sum, time_limit, max_restarts=40, maxiter=200):
    n = h0.size
    best = h0.copy()
    best_score = _compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    A = np.ones((1, n))
    lin_con = LinearConstraint(A, np.array([target_sum]), np.array([target_sum]))
    while time.time() < time_limit and max_restarts > 0:
        max_restarts -= 1
        x0 = best + np.random.randn(n) * 0.02
        x0 = _project_to_constraints(x0, target_sum)
        try:
            res = minimize(_compute_c5, x0, method="SLSQP", bounds=bounds,
                           constraints=[lin_con],
                           options={"maxiter": maxiter, "ftol": 1e-9, "disp": False})
        except Exception:
            continue
        h_opt = _project_to_constraints(res.x, target_sum)
        score = _compute_c5(h_opt)
        if score < best_score:
            best_score = score
            best = h_opt.copy()
            if best_score < 0.3808:
                break
    return best, best_score

def _optimize_n(n, init_h, start_time, budget_s):
    target = n / 2.0
    h_start = _project_to_constraints(init_h.copy(), target)
    bin_h = _binary_from_continuous(h_start, target)

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining <= 0:
        return h_start, _compute_c5(h_start)

    anneal_time = min(200.0, remaining * 0.5)
    anneal_end = time.time() + anneal_time
    best_ann, score_ann = _anneal_binary(bin_h, target, anneal_end)

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining <= 0:
        candidates = [(h_start, _compute_c5(h_start)), (best_ann, score_ann)]
        return min(candidates, key=lambda x: x[1])

    slsqp_end = time.time() + remaining * 0.5
    best_loc, score_loc = _slsqp_restarts(best_ann, target, slsqp_end)

    candidates = [(h_start, _compute_c5(h_start)),
                  (best_ann, score_ann),
                  (best_loc, score_loc)]
    return min(candidates, key=lambda x: x[1])

            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_h.size)
