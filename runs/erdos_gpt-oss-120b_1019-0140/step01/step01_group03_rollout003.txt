import time
import numpy as np

def upsample(arr, new_n):
    """Linearly upsample a 1‑D array to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto { lo <= x_i <= hi, sum x_i = s }."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen brackets if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Exact C5 = max_k ∫ h(x)(1‑h(x+k)) dx (discrete approximation)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient(h, shift):
    """Gradient of the correlation term for a given integer shift."""
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[: n - shift] = 1.0 - h[shift:]
        grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        grad[s:] = 1.0 - h[: n - s]
        grad[: n - s] -= h[s:]
    return grad


def smooth_c5_and_grad(h, alpha):
    """Smooth approximation of C5 and its gradient using log‑sum‑exp."""
    n = h.size
    dx = 2.0 / n
    a = 1.0 - h
    corr = np.convolve(h, a, mode='full')           # length 2n‑1
    max_corr = np.max(corr)
    # stable exponentials
    w = np.exp(alpha * (corr - max_corr))
    w_sum = w.sum()
    if w_sum == 0:
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum
    # forward term: sum_k w_k * (1‑h shifted forward)
    conv_f = np.convolve(w_norm[::-1], a, mode='full')
    g_f = conv_f[n - 1: n - 1 + n]
    # backward term: sum_k w_k * h shifted backward
    conv_b = np.convolve(w_norm, h, mode='full')
    g_b = conv_b[n - 1: n - 1 + n]
    grad = (g_f - g_b) * dx
    smooth_val = np.dot(w_norm, corr) * dx
    return smooth_val, grad


def smooth_pgd(h0, max_iter, step0, alpha_start, alpha_end, end_time):
    """Projected gradient descent on smooth C5."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        if time.time() > end_time - 0.05:
            break
        # linear schedule for alpha
        if max_iter > 1:
            alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (max_iter - 1)
        else:
            alpha = alpha_end
        _, grad = smooth_c5_and_grad(h, alpha)
        eta = step0 / np.sqrt(it)
        h = project_onto_box_sum(h - eta * grad, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score


def final_pgd_max(h0, n_iter, step0, end_time):
    """Fine‑tune using the true max‑correlation gradient."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        if time.time() > end_time - 0.05:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad_corr = compute_gradient(h, shift)
        grad = grad_corr * dx
        eta = step0 / np.sqrt(it)
        h = project_onto_box_sum(h - eta * grad, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score


def thue_morse(n):
    """Binary Thue–Morse sequence of length n (values 0/1)."""
    bits = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return bits


def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h
