def gradient_move(h, target, rng, step):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    g = np.zeros_like(h)
    idx_arr = np.arange(n)
    plus = idx_arr + shift
    valid_plus = (plus >= 0) & (plus < n)
    g[valid_plus] += (1.0 - h[plus[valid_plus]])
    minus = idx_arr - shift
    valid_minus = (minus >= 0) & (minus < n)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Force sum(h) == target while keeping 0 <= h_i <= 1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # Uniform correction loop
    for _ in range(5):
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            return h
    # Tiny residual: adjust a random non‑boundary entry
    if abs(diff) > 1e-12:
        idxs = np.where((h > 0.0) & (h < 1.0))[0]
        if idxs.size == 0:
            # all at bounds – just spread uniformly one last time
            h += diff / n

            max_inc = 1.0 - h[i] if diff > 0 else h[i]
            delta = np.clip(diff, -max_inc, max_inc)
            h[i] += delta
    return h

def random_h(n: int, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Generate a random feasible h with given sum."""
    h = rng.rand(n)
    h = h / h.sum() * target

def gradient_move(h: np.ndarray, target: float, rng: np.random.RandomState, step: float) -> np.ndarray:
    """Perform a projected gradient descent move on the worst overlap shift."""
    n = h.shape[0]
    # locate the shift that gives the current maximal C5
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # integer lag (can be negative)

    # gradient of the worst term w.r.t. h
    g = np.zeros_like(h)
    i = np.arange(n)
    plus = i + shift
    valid_plus = (plus >= 0) & (plus < n)
    g[valid_plus] += (1.0 - h[plus[valid_plus]])
    minus = i - shift
    valid_minus = (minus >= 0) & (minus < n)
    g[valid_minus] -= h[minus[valid_minus]]

    # project gradient to keep total mass unchanged
    g -= g.mean()

    cand = h - step * g
    np.clip(cand, 0.0, 1.0, out=cand)
    # fix any slight sum deviation
    cand = adjust_sum_to_target(cand, target, rng)
    return cand

    """Exchange a small amount of mass between two entries."""
    n = h.shape[0]
    i, j = rng.randint(n, size=2)
    if i == j:
        return h.copy()
    hi, hj = h[i], h[j]

    # feasible delta so that both stay in [0,1]
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if low >= high:
        return h.copy()

    delta = (rng.rand() * 2.0 - 1.0) * step
    delta = np.clip(delta, low, high)

    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    return cand

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # ----- choose discretisation size -----
    n_points = 500                      # higher resolution gives more flexibility
    target_sum = n_points / 2.0

    # initialise from parent if possible, otherwise random
    try:
        init = np.array(initial_h_values, dtype=float)
        if init.size != n_points:
            old_n = init.size
            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n_points, endpoint=False)
            init = np.interp(new_x, old_x, init)
        h = adjust_sum_to_target(init.copy(), target_sum, rng)
    except Exception:
        h = random_h(n_points, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    # annealing / step schedule
    T0, T_min = 5e-2, 1e-6
    step0, step_min = 5e-1, 1e-5
    prob_grad = 0.6
    no_improve = 0

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        T = T0 * (1.0 - progress) + T_min * progress
        step = step0 * (1.0 - progress) + step_min * progress

        # occasional random restart to escape stagnation
        if no_improve > 3000 and rng.rand() < 0.5:
            cur_h = random_h(n_points, target_sum, rng)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0
            continue

        if rng.rand() < prob_grad:
            cand = gradient_move(cur_h, target_sum, rng, step)
        else:
            cand = pairwise_move(cur_h, target_sum, rng, step)

        cand_c5 = compute_c5(cand)

        # simulated‑annealing acceptance
        if cand_c5 < cur_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - cur_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            cur_h = cand
            cur_c5 = cand_c5
            no_improve = 0
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
        else:
            no_improve += 1

    # final sanity correction
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
