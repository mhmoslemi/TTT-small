import numpy as np, time, math
from scipy.optimize import minimize

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-10:
        return v
    # water-filling algorithm: adjust values while respecting bounds
    # Use iterative filling
    # We'll perform a simple approach: distribute diff equally among free vars
    # and then if bounds violated, repeat.
    diff = target - cur
    # Repeat few times to converge
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            # all at bound, distribute proportionally to all
            free = np.arange(v.size)
        add = diff / free.size
        v[free] += add
        # Clip and recompute diff
        v = np.clip(v, 0.0, 1.0)
        cur = np.sum(v)
        diff = target - cur
    # final scaling if still off
    if abs(np.sum(v) - target) > 1e-8:
        v = v * (target / np.sum(v))
        v = np.clip(v, 0.0, 1.0)
    return v

def _refine_with_slsqp(start, target_sum, maxiter=200):
    # Local refinement via SLSQP
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * start.size

        res = minimize(_compute_c5, start, method="SLSQP", bounds=bounds,
                       constraints=[cons],

    except Exception:
        return start

        return start
    # Project again to ensure exact sum
    refined = _project_to_constraints(res.x, target_sum)
    return refined

def _compute_c5(h):
    """Discrete C5: max_k ∫ h(x)(1-h(x+k)) dx approximated."""

    corr = np.correlate(h, 1.0 - h, mode="full")

    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    diff = target - cur
    # Distribute diff among non‑bound elements iteratively
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]

        add = diff / free.size
