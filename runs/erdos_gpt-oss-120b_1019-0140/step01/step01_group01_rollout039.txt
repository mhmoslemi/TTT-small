import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of C5 = max_k ∫ h(x)*(1-h(x+k)) dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Project h onto the simplex {0 <= h_i <= 1, sum h_i = target}.
    The operation is done in-place and returns h.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h

    if diff > 0.0:
        # need to increase the total mass
        inc_mask = h < 1.0
        capacity = (1.0 - h[inc_mask]).sum()
        if capacity > 0.0:
            proportion = (1.0 - h[inc_mask]) / capacity
            h[inc_mask] += diff * proportion
    else:
        # need to decrease the total mass
        dec_mask = h > 0.0
        capacity = h[dec_mask].sum()
        if capacity > 0.0:
            proportion = h[dec_mask] / capacity
            h[dec_mask] += diff * proportion  # diff is negative

    # Clip to bounds and fix any residual error
    np.clip(h, 0.0, 1.0, out=h)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        h += residual / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # final scaling if still off
        scale = target / h.sum()
        h *= scale
        np.clip(h, 0.0, 1.0, out=h)
    return h

def local_search(init_h: np.ndarray, target: float, rng: np.random.RandomState, end_time: float):
    """
    Simulated annealing / stochastic descent on a single start point.
    Returns the best (h, C5) encountered before end_time.
    """
    n = init_h.size
    current_h = init_h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # annealing schedule parameters
    T0 = 2e-2          # initial temperature
    Tmin = 1e-6
    step0 = 2.5e-1     # initial step magnitude
    step_min = 1e-4

    no_improve = 0
    max_no_improve = 2000
    start = time.time()

    while time.time() < end_time:
        elapsed = time.time() - start
        total = end_time - start
        prog = elapsed / total if total > 0 else 0.0

        T = T0 * (1.0 - prog) + Tmin * prog
        step = step0 * (1.0 - prog) + step_min * prog

        move = rng.rand()
        cand = None

        if move < 0.35:
            # pairwise exchange with a bounded delta
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = max(low, min(high, delta))
            cand = current_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta

        elif move < 0.65:
            # gradient step against the worst shift
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n - 1)
            idx = np.arange(n)
            plus = idx + k_shift
            minus = idx - k_shift
            g = np.zeros_like(current_h)
            vp = (plus >= 0) & (plus < n)
            g[vp] += (1.0 - current_h[plus[vp]])
            vm = (minus >= 0) & (minus < n)
            g[vm] -= current_h[minus[vm]]
            cand = current_h - step * g
            np.clip(cand, 0.0, 1.0, out=cand)
            cand = adjust_sum_to_target(cand, target, rng)

        elif move < 0.85:
            # pure swap of two entries
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            cand = current_h.copy()
            cand[i], cand[j] = cand[j], cand[i]

        elif move < 0.95:
            # rotate a random block (preserves sum)
            L = rng.randint(2, n // 2 + 1)
            start_idx = rng.randint(n)
            inds = (np.arange(L) + start_idx) % n
            block = current_h[inds]
            cand = current_h.copy()
            cand[inds] = np.roll(block, 1)

        else:
            # small random noise with projection
            noise = (rng.rand(n) - 0.5) * step * 0.5
            cand = current_h + noise
            np.clip(cand, 0.0, 1.0, out=cand)
            cand = adjust_sum_to_target(cand, target, rng)

        if cand is None:
            continue

        cand_c5 = compute_c5(cand)

        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = cand
            current_c5 = cand_c5
            if cand_c5 < best_c5 - 1e-12:
                best_c5 = cand_c5
                best_h = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # restart from the best found with a tiny perturbation
            current_h = best_h.copy()
            perturb = (rng.rand(n) - 0.5) * step * 0.1
            current_h += perturb
            np.clip(current_h, 0.0, 1.0, out=current_h)
            current_h = adjust_sum_to_target(current_h, target, rng)
            current_c5 = compute_c5(current_h)
            no_improve = 0

    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function h on [0,2] with ∫h = 1 to minimise the Erdős
    overlap constant C5. Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start_time = time.time()
    max_time = budget_s - 0.5  # safety margin
    deadline = start_time + max_time

    # initialise h
    try:
        init_vals = np.array(initial_h_values, dtype=float)
        n = init_vals.size
        target = n / 2.0
        h0 = np.clip(init_vals, 0.0, 1.0)
        h0 = adjust_sum_to_target(h0, target, rng)
    except Exception:
        # fallback to random start (n=70 matches the common size of the problem)
        n = 70
        target = n / 2.0
        h0 = rng.rand(n)
        h0 = adjust_sum_to_target(h0, target, rng)

    # perform a single, time‑bounded stochastic search
    best_h, best_c5 = local_search(h0, target, rng, deadline)

    # final projection to guarantee feasibility
    best_h = adjust_sum_to_target(best_h, best_h.size / 2.0, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), best_h.size)
