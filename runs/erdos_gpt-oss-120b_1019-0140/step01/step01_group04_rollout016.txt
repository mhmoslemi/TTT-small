def compute_corr_fft(v):
    n = len(v)
    f = 1 - v
    size = 1 << (2 * n - 1).bit_length()  # pad to next power of 2
    V = np.fft.rfft(v, n=size)
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(V * F, n=size)
    # cross-correlation = conv[n-1: n-1 + 2n-1]
    corr = conv[n-1: n-1 + 2*n-1]  # length 2n-1
    return corr

    start = time.time()
    n = len(v)
    cur_v = v.copy()
    cur_score = _compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score

    init_temp = 0.02
    final_temp = 1e-5

        return best_v, best_score

    while True:

        if elapsed >= time_budget:
            break
        progress = elapsed / time_budget

import time
import math
import numpy as np

def _compute_c5(v):
    """Maximum overlap integral for binary or float vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return float(np.max(corr) * dx)

def _binary_from_continuous(v):
    """Make a binary vector with exactly floor(len(v)/2) ones, choosing the largest entries."""
    n = len(v)
    k = n // 2
    if k == 0:
        return np.zeros(n, dtype=np.int8)
    idx = np.argpartition(v, -k)[-k:]
    bin_vec = np.zeros(n, dtype=np.int8)
    bin_vec[idx] = 1
    return bin_vec

def _random_binary(n):
    """Return a random binary vector with exactly n//2 ones."""
    v = np.zeros(n, dtype=np.int8)
    ones = np.random.choice(n, n // 2, replace=False)
    v[ones] = 1
    return v

def _random_shuffle(v, swaps):
    """Perform a number of random 1↔0 swaps."""
    n = len(v)
    v = v.copy()
    for _ in range(swaps):
        i = np.random.randint(n)
        while v[i] == 0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while v[j] == 1:
            j = np.random.randint(n)
        v[i], v[j] = 0, 1
    return v

def _simulated_annealing(v0, time_budget):
    """Simple SA on a binary vector under the sum‑half constraint."""
    start = time.time()
    v = v0.copy()
    n = len(v)
    cur_score = _compute_c5(v)
    best_v = v.copy()
    best_score = cur_score

    init_temp = 0.05      # start temperature
    final_temp = 1e-5     # end temperature

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # geometric temperature schedule
        progress = elapsed / time_budget
        T = init_temp * (final_temp / init_temp) ** progress

        # propose a single 1↔0 swap
        i = np.random.randint(n)
        while v[i] == 0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while v[j] == 1:
            j = np.random.randint(n)

        v[i], v[j] = 0, 1
        new_score = _compute_c5(v)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and np.random.rand() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_v = v.copy()
        else:
            v[i], v[j] = 1, 0  # revert

        # occasional multi‑swap kick to escape plateaus
        if np.random.rand() < 0.01:

                i = np.random.randint(n)
                while v[i] == 0:
                    i = np.random.randint(n)
                j = np.random.randint(n)
                while v[j] == 1:
                    j = np.random.randint(n)
                v[i], v[j] = 0, 1
                swapped.append((i, j))
            new_score2 = _compute_c5(v)
            delta2 = new_score2 - cur_score

            if delta2 < 0:
                accept2 = True
            else:

                    best_v = v.copy()

    # Main SA run
