def subgradient_opt(v, n, time_budget):
    best_v = v.copy()
    best_score = compute_c5(v)
    start = time.time()
    step0 = 0.5 # initial step size may need adjustment
    iteration = 0
    while time.time() - start < time_budget:
        # compute correlation
        corr = np.correlate(v, 1 - v, mode='full')
        # get index of max
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # compute gradient g
        g = np.ones(n)
        if k >= 0:
            if k < n:
                g[:n - k] -= v[k:]
                g[k:] -= v[:n - k]
        else:
            kk = -k
            if kk < n:
                g[kk:] -= v[:n - kk]
                g[:n - kk] -= v[kk:]
        # step size schedule
        step = step0 / np.sqrt(iteration + 1)
        # update
        v_new = v - step * g
        # repair
        v_new = _repair_vector(v_new, n)
        # compute score
        score = compute_c5(v_new)
        if score < best_score:
            best_score = score
            best_v = v_new.copy()
        # Might occasionally add random noise to escape plateau
        # but keep simple.
        v = v_new
        iteration += 1
    return best_v, best_score

def anneal(v, n, time_limit):
    cur = v.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    T = 0.01  # initial temperature
    eps = 0.05
    decay = 0.995
    iterations = 0
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2 * eps
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = compute_c5(cand)
        diff = cand_score - cur_score
        if diff < 0 or np.random.rand() < np.exp(-diff / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
        # schedule
        if iterations % 100 == 0:
            T *= decay
            eps *= decay
        iterations += 1
    return best, best_score
```

def _subgradient_opt(v_init, n, time_budget):
    v = v_init.copy()
    best_v = v.copy()
    best_score = _compute_c5(v)
    start = time.time()
    iteration = 0
    # pick initial step size relative to typical gradient magnitude; we can set step0=0.5
    step0 = 0.4
    while time.time() - start < time_budget:
        corr = np.correlate(v, 1 - v, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        g = np.ones(n)
        if k >= 0:
            if k < n:
                g[:n - k] -= v[k:]
                g[k:] -= v[:n - k]
        else:
            kk = -k
            if kk < n:
                g[kk:] -= v[:n - kk]

        score = _compute_c5(v_new)

import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = int(mask.sum())
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _subgradient_opt(v_init, n, time_budget):
    """Projected sub‑gradient descent on the max‑correlation objective."""
    v = v_init.copy()
    best_v = v.copy()
    best_score = _compute_c5(v)
    start = time.time()
    step0 = 0.4
    it = 0
    while time.time() - start < time_budget:
        corr = np.correlate(v, 1 - v, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)          # shift attaining the maximum
        g = np.ones(n)
        if k >= 0:
            if k < n:
                g[: n - k] -= v[k:]
                g[k:]      -= v[: n - k]
        else:
            kk = -k
            if kk < n:
                g[kk:]     -= v[: n - kk]
                g[: n - kk] -= v[kk:]
        step = step0 / np.sqrt(it + 1)
        v_new = v - step * g
        v_new = _repair_vector(v_new, n)
        score = _compute_c5(v_new)
        if score < best_score:
            best_score, best_v = score, v_new.copy()
        v = v_new
        it += 1
        # occasional random perturbation to avoid stagnation
        if it % 200 == 0 and np.random.rand() < 0.3:
            v = _repair_vector(v + 0.02 * (np.random.rand(n) - 0.5), n)
    return best_v, best_score

def _anneal(v_init, n, time_budget):
    """Simulated annealing with swap‑exchange moves."""
    cur = v_init.copy()
    cur_score = _compute_c5(cur)
    best_v = cur.copy()
    best_score = cur_score
    start = time.time()
