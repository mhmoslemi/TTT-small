import time, numpy as np, math

def _repair_vector(v, n, max_iter=20):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # adjust free variables that can move in direction of diff
        if diff > 0:
            # need to reduce sum: can reduce values >0
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final scale if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        # scale uniformly
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        # small residual distribution
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    # correlation of v and (1-v) via np.correlate
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _anneal(v_start, time_budget):
    # simulated annealing on vector v_start
    v = v_start.copy()
    cur_score = _compute_c5(v)
    best_v = v.copy()
    best_score = cur_score
    n = len(v)
    start = time.time()
    # temperature schedule
    start_temp = 0.05
    end_temp = 0.0005
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break
        progress = elapsed / time_budget
        temp = start_temp * (1 - progress) + end_temp * progress
        # propose move: choose i,j
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # direction: from i to j
        max_delta = min(v[i], 1.0 - v[j])
        # optionally try opposite direction
        if max_delta <= 0.0:
            max_delta = min(v[j], 1.0 - v[i])
            if max_delta <= 0.0:
                continue
            i, j = j, i
        delta = np.random.rand() * max_delta
        # apply
        cand = v.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _compute_c5(cand)
        if cand_score < cur_score:
            v = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_v = cand.copy()
        else:
            # uphill acceptance
            prob = math.exp(-(cand_score - cur_score) / temp) if temp > 0 else 0.0
            if np.random.rand() < prob:
                v = cand
                cur_score = cand_score
    return best_v, best_score

def _local_refine(v_start, time_budget):
    v = v_start.copy()
    cur_score = _compute_c5(v)
    best_v = v.copy()
    best_score = cur_score
    n = len(v)
    start = time.time()
    step = 0.1  # max delta
    while True:
        now = time.time()
        if now - start >= time_budget:
            break
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # limit delta based on bounds
        max_delta = min(v[i], 1.0 - v[j])
        if max_delta <= 0.0:
            # try opposite direction
            max_delta = min(v[j], 1.0 - v[i])
            if max_delta <= 0.0:
                continue
            i, j = j, i
        # choose delta uniformly up to step*max_delta (or step)
        delta = (np.random.rand() * 2 - 1) * step * max_delta
        # ensure not exceed bounds
        delta = max(-v[i], min(delta, 1.0 - v[j]))
        cand = v.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _compute_c5(cand)
        if cand_score < cur_score:
            v = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_v = cand.copy()
    return best_v, best_score

def _anneal(v_start, time_budget):
    v = v_start.copy()
    cur_corr = np.correlate(v, 1.0 - v, mode='full')
    n = len(v)
    dx = 2.0 / n
    cur_score = np.max(cur_corr) * dx
    idx = np.argmax(cur_corr)
    worst_k = idx - (n - 1)  # shift where correlation max
    best_v = v.copy()
    best_score = cur_score

    start = time.time()
    start_temp = 0.05
    end_temp = 0.0005
    bias_prob = 0.7

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break
        progress = elapsed / time_budget
        temp = start_temp * (1 - progress) + end_temp * progress

        # propose move
        if worst_k != 0 and np.random.rand() < bias_prob:
            # biased move towards equalizing pair for worst shift
            if worst_k > 0:
                i1 = np.random.randint(0, n - worst_k)
                i2 = i1 + worst_k
            else:
                i2 = np.random.randint(-worst_k, n)
                i1 = i2 + worst_k
            # decide direction to reduce difference
            if v[i1] > v[i2]:
                src, tgt = i1, i2
            else:
                src, tgt = i2, i1
            max_delta = min(v[src], 1.0 - v[tgt])
            if max_delta <= 0:
                # fallback to random
                i = np.random.randint(n)
                j = np.random.randint(n-1)
                if j >= i:
                    j += 1
                src, tgt = i, j
                max_delta = min(v[src], 1.0 - v[tgt])
                if max_delta <= 0:
                    continue
            delta = np.random.rand() * max_delta * 0.5  # move half at most
        else:
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            src, tgt = i, j
            max_delta = min(v[src], 1.0 - v[tgt])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta

        cand = v.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_corr = np.correlate(cand, 1.0 - cand, mode='full')
        cand_score = np.max(cand_corr) * dx
        # acceptance
        if cand_score < cur_score:
            v = cand
            cur_corr = cand_corr
            cur_score = cand_score
            idx = np.argmax(cur_corr)
            worst_k = idx - (n - 1)
            if cand_score < best_score:
                best_score = cand_score
                best_v = cand.copy()

                cur_corr = cand_corr
                cur_score = cand_score
                idx = np.argmax(cur_corr)
                worst_k = idx - (n - 1)
    return best_v, best_score

import time, math, numpy as np

def _repair_vector(v, n, max_iter=20):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """C5 = max correlation * dx."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation + repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

        if now - start >= time_budget:
            break
        prog = (now - start) / time_budget
        temp = start_temp * (1 - prog) + end_temp * prog

        # biased move towards equalising the worst shift
        if worst_k != 0 and np.random.rand() < bias_prob:
            if worst_k > 0:
                i1 = np.random.randint(0, n - worst_k)
                i2 = i1 + worst_k
            else:
                i2 = np.random.randint(-worst_k, n)
                i1 = i2 + worst_k
            if v[i1] > v[i2]:
                src, tgt = i1, i2
            else:
                src, tgt = i2, i1
            max_delta = min(v[src], 1.0 - v[tgt])
            if max_delta <= 0:
                # fall back to random swap
                i = np.random.randint(n)
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
                src, tgt = i, j
                max_delta = min(v[src], 1.0 - v[tgt])
                if max_delta <= 0:
                    continue
            delta = np.random.rand() * max_delta * 0.5
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            src, tgt = i, j
            max_delta = min(v[src], 1.0 - v[tgt])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta

        cand = v.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_corr = np.correlate(cand, 1.0 - cand, mode="full")
        cand_score = np.max(cand_corr) * dx

        if cand_score < cur_score:
            v = cand
            cur_corr = cand_corr
            cur_score = cand_score
            worst_k = np.argmax(cur_corr) - (n - 1)
            if cand_score < best_score:
                best_score, best_v = cand_score, cand.copy()
        else:
            prob = math.exp(-(cand_score - cur_score) / temp) if temp > 0 else 0.0
            if np.random.rand() < prob:
                v = cand
                cur_corr = cand_corr
                cur_score = cand_score
                worst_k = np.argmax(cur_corr) - (n - 1)

    return best_v, best_score

def _local_refine(v_start, time_budget):
    """Deterministic hill‑climbing with decreasing step."""
    v = v_start.copy()
    n = len(v)
    cur_score = _compute_c5(v)
    best_v, best_score = v.copy(), cur_score
    start = time.time()
    step = 0.1

    while time.time() - start < time_budget:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        src, tgt = i, j
        max_delta = min(v[src], 1.0 - v[tgt])
        if max_delta <= 0:
            continue
        delta = (np.random.rand() * 2 - 1) * step * max_delta
        delta = max(-v[src], min(delta, 1.0 - v[tgt]))
        cand = v.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand_score = _compute_c5(cand)
        if cand_score < cur_score:
            v, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_v = cand_score, cand.copy()
        else:
            step *= 0.999  # slight decay
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function for the Erdős C5 problem."""
    np.random.seed(seed)
    start_all = time.time()

    # initial construction
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 200
        parent = np.random.rand(n0)
    parent = _repair_vector(parent, n0)

    # target resolution (up to 800 points)
    target_n = min(800, max(200, n0))
    if n0 != target_n:
        cur_v = _upsample_vector(parent, target_n)
    else:
        cur_v = parent.copy()
    cur_v = _repair_vector(cur_v, target_n)

    # overall time planning
    elapsed = time.time() - start_all
    total_budget = max(0.0, budget_s - elapsed - 1.0)  # keep a safety second
    if total_budget <= 0.0:
        best_v = cur_v
        best_score = _compute_c5(best_v)
        return best_v.astype(np.float64), float(best_score), int(target_n)

    # allocate portions of the budget
    first_part = total_budget * 0.65
    second_part = total_budget * 0.25
    refine_part = total_budget - first_part - second_part

    # first annealing from the parent‑derived guess
    best_v, best_score = _anneal(cur_v, first_part)

    # second annealing from a fresh random start (if time permits)
    if second_part > 5.0:
        rand_start = _random_feasible(target_n)
        v2, s2 = _anneal(rand_start, second_part)
        if s2 < best_score:
            best_v, best_score = v2, s2

    # final deterministic refinement
    if refine_part > 0.5:
        v_ref, s_ref = _local_refine(best_v, refine_part)
        if s_ref < best_score:
            best_v, best_score = v_ref, s_ref

    # final projection to satisfy constraints exactly
    best_v = _repair_vector(best_v, len(best_v))
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(len(best_v))
