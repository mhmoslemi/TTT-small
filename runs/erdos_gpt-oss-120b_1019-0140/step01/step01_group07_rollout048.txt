def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) <= 1e-12:
        return v
    # want to adjust to make sum=target while staying in [0,1]
    # Use distribution proportionally to remaining capacity.
    # If cur < target: need to add, distribute to entries with v<1.
    # If cur > target: need to subtract, distribute from entries with v>0.
    # We'll use simple iterative approach: compute free amount per element, allocate uniformly.
    # But to guarantee termination, we can compute the needed adjustments exactly in one pass using water-filling.
    # Let's implement general water-filling (as in prior code) for both cases.
    if cur < target:
        deficit = target - cur
        # free capacity per element:
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # sort caps descending to fill largest first (greedy fill)
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(deficit, caps[idx])
            v[i] += add
            deficit -= add
            if deficit <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    # final clip & renormalize if tiny error remains
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v *= target / cur
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

import numpy as np
import time

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) <= 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # sort descending to fill largest capacity first
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v *= target / cur
        v = np.clip(v, 0.0, 1.0)
    return v

def _tournament(pop, fitness, k=3):
    size = pop.shape[0]
    ks = min(k, size)
    idx = np.random.choice(size, ks, replace=False)
    best_idx = idx[np.argmin(fitness[idx])]
    return pop[best_idx]

def _crossover(p1, p2):
    n = p1.size
    if n < 2:
        return p1.copy()
    cuts = np.random.choice(n-1, 2, replace=False) + 1
    c1, c2 = sorted(cuts)
    child = np.empty_like(p1)
    child[:c1] = p1[:c1]
    child[c1:c2] = p2[c1:c2]
    child[c2:] = p1[c2:]
    return child

    else:

    for n in resolutions:
        # Time check before starting new resolution

            break
        target_sum = n / 2.0
        # Build starting vector for this resolution
        if best_h is not None:

            x_old = np.arange(best_h.size) * (2.0 / best_h.size)
            x_new = np.arange(n) * (2.0 / n)
            h_start = np.interp(x_new, x_old, best_h)

            score = _compute_c5(cand)
            if score < best_local_score:
                best_local_score = float(score)
                best_local_h = cand.copy()
        # Record global best
        if best_local_score < best_score:

            best_h = best_local_h.copy()
            best_n = n

            final_score = float(best_score)
