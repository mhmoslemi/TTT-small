def _smooth_obj(v, alpha=50.0):
    # compute correlation * dx inside this function, then compute smooth max
    # to ensure gradient approximates max.
    corr = np.correlate(v, 1.0 - v, mode='full')
    # Multiply by dx = 2/n for consistent scaling before max.
    # Actually scaling factor constant, we can maybe ignore for optimization.
    # Keep dp.
    max_corr = np.max(corr)  # maybe not needed.
    # Use log-sum-exp
    # To avoid overflow, subtract max before exponent.
    max_val = corr.max()
    if np.isinf(max_val) or np.isnan(max_val):
        return np.inf
    exp_sum = np.exp(alpha * (corr - max_val)).sum()
    smooth = max_val + np.log(exp_sum) / alpha
    # Multiply by dx to match objective scale for comparison.
    # Use dx
    dx = 2.0 / len(v)
    return smooth * dx

def _simulated_annealing(v0, n, time_limit, start_temp=0.1, end_temp=1e-4, step=0.1):
   # Simulated annealing.
   best_v = v0.copy()
   best_score = _compute_c5(best_v)
   cur_v = best_v.copy()
   cur_score = best_score
   start = time.time()
   # Determine number of total iterations based on time.
   # We will loop until time ran out.
   while time.time() - start < time_limit:
        # compute temperature based on time progress.
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = start_temp * ( (end_temp/start_temp) ** progress )
        i, j = np.random.randint(n), np.random.randint(n)
        while j == i:
            j = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * 2 * step
        cand = cur_v.copy()
        cand[i] = cand[i] + delta
        cand[j] = cand[j] - delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        delta_score = cand_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / max(T, 1e-12)):
            cur_v, cur_score = cand, cand_score
            if cur_score < best_score:
                best_v, best_score = cur_v.copy(), cur_score
   return best_v, best_score

def _optimize_at_resolution(n, init_v, time_limit):
    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)

    def obj(x):
        return _smooth_obj(x)  # smooth.

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n/2.0}
    bounds = [(0.0, 1.0)] * n

    # Try from init.
    try:
        res = minimize(obj, init_v, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 800, 'ftol': 1e-9, 'disp': False})
        if res.success:
            v = _repair_vector(res.x, n)
            sc = _compute_c5(v)
            if sc < best_score:
                best_v, best_score = v, sc
    except Exception:
        pass

    # Random restarts
    while time.time() - start < time_limit * 0.9:
        rnd = _random_feasible(n)
        try:
            res = minimize(obj, rnd, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
            if res.success:
                v = _repair_vector(res.x, n)
                sc = _compute_c5(v)

                    best_v, best_score = v, sc
        except Exception:
            pass
        if time.time() - start > time_limit * 0.85:
            break
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, random
    np.random.seed(seed)
    start = time.time()
    # get parent or random.
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 200
        parent = _random_feasible(n0)
    parent = _repair_vector(parent, n0)

    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    resolutions = [n0]
    # add upsampling limits.
    if n0 * 2 <= 800:
        resolutions.append(n0 * 2)
    if n0 * 4 <= 800:
        resolutions.append(n0 * 4)

    total_budget = max(0.0, budget_s - 1.0)  # keep 1 sec for final steps.

    # allocate time proportionally to size: larger maybe need more time.
    # We'll allocate equal time per resolution.
    per_res_time = total_budget / len(resolutions) if resolutions else 0

    for n in resolutions:
        if time.time() - start >= budget_s - 0.5:
            break
        # Up-sample if needed.
        if len(best_v) != n:
            init_guess = _upsample_vector(best_v, n)
        else:
            init_guess = best_v.copy()
        # time for this resolution
        elapsed = time.time() - start
        remaining_for_this = budget_s - elapsed - 0.5
        time_allowed = min(per_res_time, remaining_for_this)
        if time_allowed <= 0:
            break
        # Run optimization (SLSQP + restarts)
        opt_time = time_allowed * 0.75
        v_opt, sc_opt = _optimize_at_resolution(n, init_guess, opt_time)
        if sc_opt < best_score:
            best_v, best_score, best_n = v_opt, sc_opt, n
        # Simulated annealing on remaining time
        sa_time = time_allowed - opt_time
        if sa_time > 0.2:
            v_sa, sc_sa = _simulated_annealing(best_v, best_n, sa_time)
            if sc_sa < best_score:
                best_v, best_score = v_sa, sc_sa
        # Ensure best_v is feasible
        best_v = _repair_vector(best_v, best_n)
        best_score = _compute_c5(best_v)

    # final repair
    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(best_n)

import time
import numpy as np
from scipy.optimize import minimize

# --- helper functions ---------------------------------------------------------

def _repair_vector(v, n, max_iter=20):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        free = mask.sum()
        if free == 0:
            break
        adj = diff / free
        v[mask] -= adj
        v = np.clip(v, 0.0, 1.0)
    # final tiny fix
    s = v.sum()
    if abs(s - target) > 1e-9:
        v *= target / s if s != 0 else 0.0
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Exact C5 value for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _random_feasible(n):
    """Return a random feasible vector."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _upsample_vector(v, n_target):
    """Linear interpolation to a new resolution, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

# smooth approximation of the max for SLSQP
_ALPHA = 30.0

def _smooth_obj(v):
    """Smooth max of the correlation using log-sum-exp."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode="full")
    m = corr.max()
    # log-sum-exp stabilized
    exp_sum = np.exp(_ALPHA * (corr - m)).sum()
    smooth = m + np.log(exp_sum) / _ALPHA
    return smooth * (2.0 / n)

def _optimize_at_resolution(n, init_v, time_limit):
    """SLSQP with smooth objective and random restarts within time_limit."""
    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _smooth_obj(x)

    # primary run
    try:
        res = minimize(
            obj,
            init_v,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 800, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            v = _repair_vector(res.x, n)
            sc = _compute_c5(v)
            if sc < best_score:
                best_v, best_score = v, sc
    except Exception:
        pass

    # random restarts
    while time.time() - start < time_limit * 0.9:
        rnd = _random_feasible(n)
        try:
            res = minimize(
                obj,
                rnd,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n)
                sc = _compute_c5(v)
                if sc < best_score:
                    best_v, best_score = v, sc
        except Exception:
            pass
        if time.time() - start > time_limit * 0.85:
            break
    return best_v, best_score

def _simulated_annealing(v0, n, time_limit, start_temp=0.08, end_temp=5e-4, step=0.12):
    """Simple SA on the exact max‐overlap objective."""
    best_v = v0.copy()
    best_score = _compute_c5(best_v)
    cur_v = best_v.copy()
    cur_score = best_score
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        prog = elapsed / time_limit
        T = start_temp * ( (end_temp / start_temp) ** prog )
        i, j = np.random.randint(n), np.random.randint(n)
        while j == i:
            j = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        dscore = cand_score - cur_score
        if dscore < 0 or np.random.rand() < np.exp(-dscore / max(T, 1e-12)):
            cur_v, cur_score = cand, cand_score
            if cur_score < best_score:
                best_v, best_score = cur_v.copy(), cur_score
    return best_v, best_score
