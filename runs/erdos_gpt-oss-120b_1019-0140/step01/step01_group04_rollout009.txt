            frac = elapsed / time_limit
            T = max(0.02 * (1.0 - frac), 1e-6)
            if np.random.rand() < math.exp(-(cand_score - cur_score) / T):
                cur_v, cur_score = cand, cand_score
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function that yields a low Erdős C₅ bound."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initial construction (may be supplied as `initial_h_values`).
    try:
        parent = np.array(initial_h_values, dtype=np.float64)  # pre‑imported if present
        n0 = len(parent)
    except Exception:
        n0 = 190
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0 // 2] = 1.0

    # Convert to a binary vector with exactly half ones.
    half = n0 // 2
    idx = np.argsort(parent)[-half:]                     # top half entries become 1
    bin_parent = np.zeros(n0, dtype=np.float64)
    bin_parent[idx] = 1.0

    # ------------------------------------------------------------------
    # Time budgeting.
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed - 0.2)      # keep a tiny safety margin
    binary_budget = remaining * 0.8
    slsqp_budget = remaining - binary_budget

    # ------------------------------------------------------------------
    # Binary hill‑climbing with several restarts.
    best_v = bin_parent
    best_score = _compute_c5(best_v)
    n_restarts = 5
    per_restart = binary_budget / max(1, n_restarts)

    for r in range(n_restarts):
        if r == 0:
            init = bin_parent.copy()
        else:
            init = np.zeros(n0, dtype=np.float64)
            init[:half] = 1.0
            np.random.shuffle(init)
        v, sc = _binary_hill_climb(init, per_restart)
        if sc < best_score:
            best_v, best_score = v, sc

    # ------------------------------------------------------------------
    # Optional continuous refinement (SLSQP).
    if slsqp_budget > 0.1:
        def obj(x):
            return _compute_c5(x)
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - n0 / 2.0}
        bounds = [(0.0, 1.0)] * n0
        try:
            res = minimize(
                obj,
                best_v.astype(np.float64),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 600, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n0)
                sc = _compute_c5(v)
                if sc < best_score:
                    best_v, best_score = v, sc
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Try a higher‑resolution refinement if time permits.
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed - 0.1)
    if remaining > 0.5 and n0 * 2 <= 800:
        n2 = n0 * 2
        up_v = np.repeat(best_v, 2)               # binary up‑sampling keeps the sum exact
        binary_up_budget = remaining * 0.7
        slsqp_up_budget = remaining - binary_up_budget

        best_up_v = up_v
        best_up_score = _compute_c5(best_up_v)
        up_restarts = 4
        per_up = binary_up_budget / max(1, up_restarts)

        for r in range(up_restarts):
            if r == 0:
                init_up = best_up_v.copy()
            else:
                init_up = np.zeros(n2, dtype=np.float64)
                init_up[: n2 // 2] = 1.0
                np.random.shuffle(init_up)
            v2, sc2 = _binary_hill_climb(init_up, per_up)
            if sc2 < best_up_score:
                best_up_v, best_up_score = v2, sc2

        if slsqp_up_budget > 0.2:
            def obj2(x):
                return _compute_c5(x)
            cons2 = {"type": "eq", "fun": lambda x: np.sum(x) - n2 / 2.0}
            bounds2 = [(0.0, 1.0)] * n2
            try:
                res2 = minimize(
                    obj2,
                    best_up_v.astype(np.float64),
                    method="SLSQP",
                    bounds=bounds2,
                    constraints=cons2,
                    options={"maxiter": 600, "ftol": 1e-9, "disp": False},
                )
                if res2.success:
                    v2 = _repair_vector(res2.x, n2)
                    sc2 = _compute_c5(v2)
                    if sc2 < best_up_score:
                        best_up_v, best_up_score = v2, sc2
            except Exception:
                pass

        if best_up_score < best_score:
            return best_up_v.astype(np.float64), float(best_up_score), int(n2)

    # ------------------------------------------------------------------
    # Return the best solution found at the original resolution.
    return best_v.astype(np.float64), float(best_score), int(n0)
