def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    # ensure the sum is feasible:
    n = v.size
    lo_sum = lo * n
    hi_sum = hi * n
    if s <= lo_sum:
        return np.full_like(v, lo)
    if s >= hi_sum:
        return np.full_like(v, hi)

    # Bisection on lambda such that sum_i clip(v_i - lambda, lo, hi) = s.
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    # ensure brackets
    # we can compute f(lam) = sum clip(v - lam, lo, hi)
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(v - lam_mid, lo, hi)
        f_mid = proj.sum()
        if abs(f_mid - s) < 1e-12:
            return proj
        if f_mid > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    lo_sum = lo * n
    hi_sum = hi * n
    if s <= lo_sum:
        return np.full_like(v, lo)
    if s >= hi_sum:
        return np.full_like(v, hi)

    # bisection on lambda such that sum clip(v - lam, lo, hi) = s
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        fm = proj.sum()
        if abs(fm - s) < 1e-12:
            return proj
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Exact C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def thue_morse_array(n):
    """Binary Thue–Morse sequence of length n (requires n a power of two)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    # the Thue–Morse sequence has exactly n/2 ones for powers of two
    return seq


def random_feasible(n, rng):
    """Random vector in [0,1]^n with sum == n/2."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def compute_weights(inner, beta):
    """Soft-min weights for the inner-product vector (stable)."""
    max_inner = inner.max()
    # avoid overflow/underflow
    exp_vals = np.exp(-beta * (inner - max_inner))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        p = np.zeros_like(inner)
        p[inner.argmax()] = 1.0
    else:
        p = exp_vals / sum_exp
    # enforce symmetry (correlation is symmetric)
    p = (p + p[::-1]) * 0.5
    return p


def smoothed_gradient(h, beta):
    """Gradient of (1/beta) log-sum-exp(-beta * inner) w.r.t. h."""
    inner = np.correlate(h, h, mode='full')
    p = compute_weights(inner, beta)
    # convolution of p with h
    conv = np.convolve(p, h, mode='full')
    n = h.shape[0]
    # gradient = -2 * conv slice (see derivation)
    return -2.0 * conv[n - 1:n - 1 + n]

        it = 0
        while True:
            now = time.time()
            if now - start_time > budget_s - safe_margin:
                break

            # gradient of smoothed objective
            grad = smoothed_gradient(h_cur, beta)

            # Adam moment updates
            it += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            # projected update
            h_next = h_cur - step
            h_next = project_onto_box_sum(h_next, n / 2.0)

            # evaluate true C5
            score = compute_c5(h_next)

            # keep global best
            if score < best_score - 1e-9:
                best_score = float(score)
                best_h = h_next.copy()
                stagn = 0
            else:
                stagn += 1

            # prepare next iteration
            h_cur = h_next

            # update smoothing parameter
            if beta < beta_max:
                beta = min(beta_max, beta * beta_growth)

            # restart if stuck for too long
            if stagn >= stagn_limit:
                break

            # limit very long runs per seed to keep responsiveness
            if it >= 8000:
                break

        # after a seed finishes, loop continues with next seed or random restart
        # and the global best is preserved.

    # ------------------------------------------------------------------
    # 3. Final safeguard and return
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5, dtype=float), n / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, float(best_score), int(n)
