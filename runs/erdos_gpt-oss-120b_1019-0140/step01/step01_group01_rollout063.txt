import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), (1.0 - h).astype(float), mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Scale / clip a vector so that its sum equals target while respecting 0 ≤ h_i ≤ 1.
    Works with floating point vectors.
    """
    # Simple scaling if the vector is already within bounds
    total = h.sum()
    if abs(total - target) < 1e-12:
        return h

    # Clip extremes first
    np.clip(h, 0.0, 1.0, out=h)
    total = h.sum()
    diff = target - total

    # Distribute the remaining difference by iterating over random indices
    attempts = 0
    max_attempts = 2000
    n = h.size
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(n)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1

    if abs(diff) > 1e-12:
        # Tiny residual – spread uniformly and re‑clip
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        # Final proportional correction if needed
        total = h.sum()
        if abs(total - target) > 1e-12:
            h *= target / total
            np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Simulated‑annealing search for a step function h on [0,2] that
    minimizes the Erdős overlap constant C5.

    Returns (h_values, c5_bound, n_points)
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.3  # safety margin

    # ---------- initialise ----------
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        target_sum = n / 2.0
        np.clip(init, 0.0, 1.0, out=init)
        h = adjust_sum_to_target(init.copy(), target_sum, rng)
    except Exception:
        # Fallback: random binary vector of length 70
        n = 70
        target_sum = n / 2.0
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, int(target_sum), replace=False)
        h[ones] = 1.0

    h = adjust_sum_to_target(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    # ----- simulated annealing parameters -----
    T_start = 0.02          # initial temperature
    T_end   = 1e-6          # final temperature
    step_start = 0.5        # initial step magnitude
    step_end   = 1e-4       # final step magnitude
    last_improve = start

    # ----- main optimisation loop -----
    while True:
        now = time.time()
        if now - start >= max_time:
            break

        progress = (now - start) / max_time
        # exponential cooling for temperature and step
        T = T_start * (T_end / T_start) ** progress
        step = step_start * (step_end / step_start) ** progress

        # choose move type
        if rng.rand() < 0.6:
            # ---- gradient move on the worst lag ----
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n - 1)

            g = np.zeros_like(cur_h)
            idx = np.arange(n)

            plus = idx + k_shift
            mask_plus = (plus >= 0) & (plus < n)
            g[mask_plus] += (1.0 - cur_h[plus[mask_plus]])

            minus = idx - k_shift
            mask_minus = (minus >= 0) & (minus < n)

            # keep the sum invariant
            g -= g.mean()

            cand[i] = hi + delta

    return (best_h, final_c5, n)
