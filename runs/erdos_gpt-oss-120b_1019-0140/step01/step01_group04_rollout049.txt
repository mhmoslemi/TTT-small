def compute_c5(v):
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def generate_binary_random(n):
    half = n // 2
    # generate random permutation of zeros and ones
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 1.0
    np.random.shuffle(v)
    return v

def upsample_vector(v, n_target):
    n_src = len(v)
    src_x = np.linspace(0, 2, n_src, endpoint=False)  # coordinates
    tgt_x = np.linspace(0, 2, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return repair_vector(new_v, n_target)

def binary_simulated_annealing(v_init, n, time_budget):
    start = time.time()
    cur_v = v_init.copy()
    cur_score = compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    T0 = 0.01
    Tf = 1e-5
    # Determine total iterations based on time budget and some per iteration estimate.
    # We'll loop until time runs out.
    while time.time() - start < time_budget:
        # Estimate progress
        elapsed = time.time() - start
        progress = elapsed / time_budget
        T = T0 * (1 - progress) + Tf * progress
        # propose swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if cur_v[i] == cur_v[j]:
            continue
        # swap
        cand = cur_v.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = compute_c5(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            cur_v = cand
            cur_score = cand_score
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_v = cur_v.copy()
    return best_v, best_score

def repair_vector(v, n, max_iter=30):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # adjust values that are not at bounds opposite direction of diff
        if diff > 0:
            mask = v > 0.0  # can decrease
        else:
            mask = v < 1.0  # can increase
        n_free = mask.sum()
        if n_free == 0:
            break
        # distribute diff uniformly among free entries
        adj = diff / n_free
        v[mask] -= adj
        v = np.clip(v, 0.0, 1.0)
    # final small correction if still off
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def binary_simulated_annealing(v_init, n, time_budget):
    start = time.time()
    cur = v_init.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    # track iteration count
    it = 0
    # track last improvement iteration
    last_improve_it = 0
    # Set some max iterations fudge
    while time.time() - start < time_budget:
        # choose random i, j with different values
        i = np.random.randint(n)
        j = np.random.randint(n)
        if cur[i] == cur[j]:
            continue
        # swap
        cand = cur.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = compute_c5(cand)
        # progress fraction for temperature
        elapsed = time.time() - start
        prog = elapsed / time_budget
        T = 0.01 * (1 - prog) + 1e-6 * prog
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                last_improve_it = it
        it += 1
        # optional restart
        if it % 2000 == 0 and it - last_improve_it > 5000:
            # reinitialize random binary
            cur = generate_binary_random(n)
            cur_score = compute_c5(cur)
            # will continue
    return best, best_score

def slsqp_refine(v0, n, time_budget):

    # we may want to limit iterations based on time_budget
    # compute max_iter as min(200, int(time_budget * 10))
    max_iter = max(10, min(200, int(time_budget * 10)))  # each iteration ~some time
    def obj(x):
        return compute_c5(x)
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n/2}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    except Exception:
        return v0, compute_c5(v0)
    # repair result
    v_opt = repair_vector(res.x, n)
    score_opt = compute_c5(v_opt)
    return v_opt, score_opt

def h_from_a(a, n):
    # a between 0 and 1
    b = 2.0 - a
    dx = 2.0 / n
    x = np.arange(n) * dx
    # compute piecewise
    h = np.empty(n, dtype=np.float64)

    mask2 = (x >= a) & (x < b)

def golden_section_search(func, a_low, a_high, n_iter, n, time_limit):
    # typical golden ratio search to minimize func(a)
    phi = (1 + np.sqrt(5)) / 2
    resphi = 2 - phi  # = 1/phi^2
    # initialize points
    x1 = a_low + resphi * (a_high - a_low)
    x2 = a_high - resphi * (a_high - a_low)
    f1 = func(x1, n)
    f2 = func(x2, n)
    start = time.time()
    for _ in range(n_iter):
        if time.time() - start > time_limit:
            break
        if f1 < f2:
            a_high = x2
            x2 = x1
            f2 = f1
            x1 = a_low + resphi * (a_high - a_low)
            f1 = func(x1, n)
        else:
            a_low = x1
            x1 = x2
            f1 = f2
            x2 = a_high - resphi * (a_high - a_low)
            f2 = func(x2, n)
    # Return best
    if f1 < f2:
        return x1, f1
    else:
        return x2, f2

import time
import numpy as np
from scipy.optimize import minimize

def _repair_vector(v, n, max_iter=20):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def _h_from_a(a, n):
    # a in [0,1]; b = 2 - a
    b = 2.0 - a
    dx = 2.0 / n
    x = np.arange(n) * dx
    h = np.empty(n, dtype=np.float64)
    mask1 = x < a
    mask2 = (x >= a) & (x < b)
    h[mask1] = 1.0
    # avoid division by zero when a==b (which only occurs at a=1 -> b=1)
    denom = b - a
    if denom == 0:
        h[mask2] = 0.0
    else:
        h[mask2] = (b - x[mask2]) / denom
    h[~mask1 & ~mask2] = 0.0
    return _repair_vector(h, n)

def _c5_for_a(a, n):
    h = _h_from_a(a, n)
    return _compute_c5(h)

def _golden_section_search(n, time_limit):
    # search a in [0,1] to minimize C5
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    resphi = 2.0 - phi
    lo, hi = 0.0, 1.0
    x1 = lo + resphi * (hi - lo)
    x2 = hi - resphi * (hi - lo)
    f1 = _c5_for_a(x1, n)
    f2 = _c5_for_a(x2, n)
    start = time.time()
    while time.time() - start < time_limit:
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = lo + resphi * (hi - lo)
            f1 = _c5_for_a(x1, n)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = hi - resphi * (hi - lo)
            f2 = _c5_for_a(x2, n)
        # break if interval is very small
        if hi - lo < 1e-5:
            break
    if f1 < f2:
        best_a, best_score = x1, f1
    else:
        best_a, best_score = x2, f2
    best_h = _h_from_a(best_a, n)
    return best_h, best_score, best_a

def _slsqp_refine(v0, n, time_limit):
    # Allocate maxiter based on time
    maxiter = max(20, int(time_limit * 5))  # rough scaling
    def obj(x):
        return _compute_c5(x)
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, v0, method="SLSQP", bounds=bounds,
                       constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-9, "disp": False})
        if not res.success:
            # fallback to original
            v_opt = v0
        else:
            v_opt = res.x
    except Exception:
        v_opt = v0
    v_opt = _repair_vector(v_opt, n)
    return v_opt, _compute_c5(v_opt)

    best_h, best_score, best_a = _golden_section_search(n, time_golden)
