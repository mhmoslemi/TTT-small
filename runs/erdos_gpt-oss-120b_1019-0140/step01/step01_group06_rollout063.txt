def project_to_feasible(x, target_sum, tol=1e-9, max_iter=100):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(x)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # find indices not at bounds
        if diff > 0:
            free = np.where(x < 1.0)[0]
        else:
            free = np.where(x > 0.0)[0]
        if free.size == 0:
            break
        # compute delta distribution
        delta = diff / free.size
        # Apply but clip further
        if diff > 0:
            # add up to 1
            x[free] = np.minimum(1.0, x[free] + delta)
        else:
            x[free] = np.maximum(0.0, x[free] + delta)  # delta negative
        # Clip again
        x = np.clip(x, 0.0, 1.0)
    # final correction for numerical sum errors:
    cur = np.sum(x)
    if abs(cur - target_sum) > 1e-6:
        # adjust a single index with capacity
        if cur < target_sum:
            # need increase
            capacity = 1.0 - x
            idx = np.argmax(capacity)
            inc = min(target_sum - cur, capacity[idx])
            x[idx] += inc
        else:
            # need decrease
            capacity = x
            idx = np.argmax(capacity)
            dec = min(cur - target_sum, capacity[idx])
            x[idx] -= dec
    return x

import time, numpy as np, scipy.optimize as opt

_current_n = None

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(x, target_sum, tol=1e-9, max_iter=100):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(x)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(x < 1.0)[0]
        else:
            free = np.where(x > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        if diff > 0:
            x[free] = np.minimum(1.0, x[free] + delta)
        else:
            x[free] = np.maximum(0.0, x[free] + delta)
        x = np.clip(x, 0.0, 1.0)
    # final tiny fix
    cur = np.sum(x)
    if abs(cur - target_sum) > 1e-6:
        # adjust one index with capacity
        if cur < target_sum:
            cap = 1.0 - x
            idx = np.argmax(cap)
            inc = min(cap[idx], target_sum - cur)
            x[idx] += inc
        else:
            cap = x
            idx = np.argmax(cap)
            dec = min(cap[idx], cur - target_sum)
            x[idx] -= dec
    return x

def sum_constraint(x):
    global _current_n
    return np.sum(x) - _current_n / 2.0

def objective_fun(x):
    return compute_c5(x)

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    target_sum = n_target / 2.0
    h_new = project_to_feasible(h_new, target_sum)
    return h_new

    # small random perturbation frequency

    while True:

            break

        j = rng.integers(0, n-1)

            src, dst = i, j
        else:
            src, dst = j, i

        if delta_max <= 0:
            continue
        delta = step_frac * delta_max * rng.random()

        h_new[src] -= delta
        h_new[dst] += delta
        # compute if improved
        val_new = compute_c5(h_new)
        if val_new + 1e-12 < best_val:
            best_h = h_new
            best_val = val_new
            # optionally increase step_frac if progress

                best_val = val_trial
        # possibly break if we hit very low bound?

import time
import numpy as np
import scipy.optimize as opt

_current_n = None  # used by sum_constraint


def compute_c5(h):
    """Overlap constant C5 for a discrete step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(x, target_sum, tol=1e-9, max_iter=100):
    """Clip to [0,1] and adjust to meet the linear sum constraint."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(x)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(x < 1.0)[0]
        else:
            free = np.where(x > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        if diff > 0:
            x[free] = np.minimum(1.0, x[free] + delta)
        else:
            x[free] = np.maximum(0.0, x[free] + delta)
        x = np.clip(x, 0.0, 1.0)
    cur = np.sum(x)
    if abs(cur - target_sum) > 1e-6:
        if cur < target_sum:
            cap = 1.0 - x
            idx = np.argmax(cap)
            inc = min(cap[idx], target_sum - cur)
            x[idx] += inc
        else:
            cap = x
            idx = np.argmax(cap)
            dec = min(cap[idx], cur - target_sum)
            x[idx] -= dec
    return x


def sum_constraint(x):
    """Equality constraint: sum(x) = n/2, where n is stored globally."""
    global _current_n
    return np.sum(x) - _current_n / 2.0


def objective_fun(x):
    """Objective for SLSQP: C5."""
    return compute_c5(x)


def resample_h(base_h, n_target):
    """Linear interpolation of a parent vector to a new length, with feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    return project_to_feasible(h_new, n_target / 2.0)


def hill_climb(h_start, target_sum, n, deadline, rng):
    """Simple local search: exchange moves, occasional global noise, and gradient steps."""
    best_h = h_start.copy()
    best_val = compute_c5(best_h)
    step_frac = 0.5          # fraction of the maximum transferable mass
    perturb_prob = 0.02      # chance of global Gaussian perturbation
    grad_step_prob = 0.05    # chance of a gradient-oriented move
    while time.time() < deadline:
        # ---- exchange move ----
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        delta_max = min(best_h[src], 1.0 - best_h[dst])
        if delta_max > 0:
            delta = step_frac * delta_max * rng.random()
            h_new = best_h.copy()
            h_new[src] -= delta
            h_new[dst] += delta
            val_new = compute_c5(h_new)
            if val_new + 1e-12 < best_val:
                best_h, best_val = h_new, val_new

        # ---- occasional global perturbation ----
        if rng.random() < perturb_prob:
            noise = rng.normal(scale=0.02, size=n)
            trial = best_h + noise
            trial = project_to_feasible(trial, target_sum)
            val_trial = compute_c5(trial)
            if val_trial + 1e-12 < best_val:
                best_h, best_val = trial, val_trial

        # ---- occasional gradient step toward reducing the worst lag ----
        if rng.random() < grad_step_prob:
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            max_idx = np.argmax(corr)
            lag = max_idx - (n - 1)                # signed shift
            if lag != 0:
                grad = -(2.0 / n) * (np.roll(best_h, lag) + np.roll(best_h, -lag))
                alpha = 0.1
                h_candidate = best_h - alpha * grad
                h_candidate = project_to_feasible(h_candidate, target_sum)
                val_cand = compute_c5(h_candidate)
                if val_cand + 1e-12 < best_val:
                    best_h, best_val = h_candidate, val_cand

        # early exit if we have already beaten the target bound
        if best_val <= 0.3805:
            break
    return best_h, best_val


def optimize_n(n, base_h, total_budget, start_time, rng):
    """Search for a good h at a given resolution n."""
    target_sum = n / 2.0
    # ---- initial vector ----
    if base_h is not None:

    best_val = np.inf

        # initial guess for this restart
        if restart == 0:
            guess = h0.copy()
        else:
            if rng.random() < 0.6:

        global _current_n

        try:
            res = opt.minimize(
                fun=objective_fun,
                x0=guess,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 400, "ftol": 1e-8, "disp": False},

        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_to_feasible(h_opt, target_sum)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()
        if best_val <= 0.3805:
            break

    if best_h is None:
        best_h = h0.copy()

    # ---- local refinement via hill climbing ----
    deadline = start_time + total_budget * 0.97

    # final safety projection
    best_h = project_to_feasible(best_h, target_sum)
