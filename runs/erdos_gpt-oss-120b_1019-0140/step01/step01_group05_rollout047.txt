def _simulated_annealing(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = len(h_cur)
    start = time.time()
    T0 = 0.02
    max_eps = 0.5   # allow large moves
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        # fraction of time remaining
        elapsed = now - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi, hj = h_cur[i], h_cur[j]
        # feasible epsilon: cannot exceed box constraints
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up + down == 0.0:
            continue
        # random epsilon
        eps = (rng.random() * (up + down) - down)
        # Clip epsilon magnitude to max_eps
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:

        new_h[j] = hj - eps

            cur_score = new_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and rescale to satisfy sum(v) = len(v)/2."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx for discretised h."""
    N = h.size
    dx = 2.0 / N
    if N <= 200:
        corr = np.correlate(h, 1.0 - h, mode='full')
    else:
        # FFT based convolution (full correlation)
        m = 1 << ((2 * N - 1).bit_length())  # next power of two
        a = np.fft.rfft(h, n=m)
        b = np.fft.rfft((1.0 - h)[::-1], n=m)
        conv = np.fft.irfft(a * b, n=m)
        corr = conv[: 2 * N - 1]
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simple SA that keeps the sum constraint exactly."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.02          # initial temperature
    max_eps = 1.0      # allow full feasible moves
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)      # max we can add to i
        down = min(hi, 1.0 - hj)    # max we can subtract from i
        if up + down == 0.0:
            continue
        eps = rng.random() * (up + down) - down  # uniform in [-down, up]
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        new_h = h_cur.copy()
        new_h[i] = hi + eps
        new_h[j] = hj - eps
        # feasibility holds by construction
        new_score = _c5_value(new_h)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
    return best_h, best_score

    # resolution ladder (doubling, up to 1000 points)
    levels = [best_h.size]
    while levels[-1] < 1000:
        nxt = min(levels[-1] * 2, 1000)
        if nxt == levels[-1]:
            break
        levels.append(nxt)

    for lvl_idx, n in enumerate(levels):
        now = time.time()
        elapsed = now - start_time
        remaining = budget_s - elapsed
        if remaining <= 0:
            break
        levels_left = len(levels) - lvl_idx
        # allocate a share of the remaining time to this resolution
        if lvl_idx == len(levels) - 1:
            time_for_level = remaining * 0.95
        else:
            time_for_level = remaining * 0.9 / levels_left

        # upsample / adapt current best to the new resolution
        if n != best_h.size:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_cur = np.interp(x_new, x_old, best_h)
        else:
            h_cur = best_h.copy()
        h_cur = _repair_vector(h_cur)

        # split time between two SA runs (upscaled start + random start)
        run_time = max(0.1, time_for_level / 2.0)

        # SA from upscaled current best
        cand_h, cand_score = _simulated_annealing(h_cur, run_time, rng)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score

        # SA from a fresh random start (binary + tiny noise)
        half = n // 2
        rand_vec = np.zeros(n, dtype=np.float64)
        rand_vec[:half] = 1.0
        rng.shuffle(rand_vec)
        # add a tiny perturbation to allow non‑binary moves
        rand_vec += (rng.random(n) - 0.5) * 0.01
        rand_vec = _repair_vector(rand_vec)
        cand_h2, cand_score2 = _simulated_annealing(rand_vec, run_time, rng)
        if cand_score2 < best_score:
            best_h, best_score = cand_h2, cand_score2

        # small deterministic hill‑climbing phase
        local_iters = 2000
        cur_h = best_h.copy()
        cur_score = best_score
        N = cur_h.size
        for _ in range(local_iters):
            if time.time() - start_time > budget_s * 0.97:
                break
            i = rng.integers(N)
            j = rng.integers(N - 1)
            if j >= i:
                j += 1
            hi, hj = cur_h[i], cur_h[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up + down == 0.0:
                continue
            # try the two boundary moves which often give improvement
            for eps in (up, -down):
                new_h = cur_h.copy()
                new_h[i] = hi + eps
                new_h[j] = hj - eps
                new_score = _c5_value(new_h)
                if new_score < cur_score - 1e-12:
                    cur_h, cur_score = new_h, new_score
                    break
        if cur_score < best_score:
            best_h, best_score = cur_h, cur_score

    # final fine‑tuning if any time left
    leftover = budget_s - (time.time() - start_time)
    if leftover > 5.0:
        final_h, final_score = _simulated_annealing(best_h, leftover, rng)
        if final_score < best_score:
            best_h, best_score = final_h, final_score

    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
