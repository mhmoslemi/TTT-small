    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def propose_move(h, rng):
    n = len(h)
    # pick two distinct indices i, j
    i, j = rng.choice(n, size=2, replace=False)
    # propose delta move
    # allowable delta range limited by bounds of h[i] and h[j]
    max_inc_i = 1.0 - h[i]
    max_dec_i = h[i]
    max_inc_j = 1.0 - h[j]
    max_dec_j = h[j]
    # We will move delta from i to j: h[i] -= delta, h[j] += delta to keep sum constant (or opposite)
    # We'll randomly decide direction
    direction = rng.integers(2)  # 0: i -> j, 1: j -> i
    if direction == 0:
        # transfer from i to j
        delta_max = min(max_dec_i, max_inc_j)
        delta_min = -min(max_inc_i, max_dec_j) # negative? Actually if we allow reverse direction we would adjust sign
        # Actually we only allow positive delta. For i->j, delta>=0 and <= delta_max
        if delta_max <= 0:
            return None, None
        delta = rng.uniform(0, delta_max)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_and_fix(h, target_sum):
    """One Gaussian‑type smoothing step with sum correction."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(h, kernel, mode="same")
    sm = np.clip(sm, 0.0, 1.0)
    cur = np.sum(sm)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((sm > 0) & (sm < 1))[0]
        if free.size:
            inc = diff / free.size
            sm[free] += inc
            sm = np.clip(sm, 0.0, 1.0)
        # simple residual fix
        diff = target_sum - np.sum(sm)
        for i in range(len(sm)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and sm[i] < 1.0:
                add = min(diff, 1.0 - sm[i])
                sm[i] += add
                diff -= add
            elif diff < 0 and sm[i] > 0.0:
                sub = min(-diff, sm[i])
                sm[i] -= sub
                diff += sub
        sm = np.clip(sm, 0.0, 1.0)
    return sm

            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 400, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                # enforce exact sum once more
                cur = np.sum(h_opt)
                if abs(cur - target_sum) > 1e-9:
                    h_opt += (target_sum - cur) / n
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                val_opt = compute_c5(h_opt)
                if val_opt + 1e-12 < best_val:
                    best_val = val_opt
                    best_h = h_opt
        except Exception:
            pass

    # final safety correction
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = np.sum(best_h)
    if abs(cur - target_sum) > 1e-9:
        best_h += (target_sum - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val, n
