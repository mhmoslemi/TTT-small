import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------
def _compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    n = h.size
    if n == 0:
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_to_constraints(v, target):
    """Clip to [0,1] and enforce sum(v)=target by water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    tol = 1e-9
    if abs(cur - target) < tol:
        return v
    # Simple scaling fallback when water-filling is expensive
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v


def _binary_hill_climb(start_h, deadline, target_score):
    """Greedy hill‑climb on a binary vector (0/1) respecting the sum constraint."""
    cur_h = start_h.copy()
    cur_score = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.size
    no_improve = 0
    max_no_improve = 5000  # stop after many futile swaps
    while time.time() < deadline and no_improve < max_no_improve:
        # choose random 1↔0 swap
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = _compute_c5(cur_h)
        if new_score + 1e-12 < cur_score:  # improvement
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                if best_score < target_score:
                    break
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_improve += 1
    return best_h, best_score


def _local_continuous_refine(h_start, target_sum, deadline):
    """One brief SLSQP refinement (optional, may be skipped if time low)."""
    if time.time() + 1.0 > deadline:
        return h_start, _compute_c5(h_start)
    x0 = h_start + np.random.randn(h_start.size) * 0.02
    x0 = _project_to_constraints(x0, target_sum)
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * h_start.size
    try:
        res = minimize(
            _compute_c5,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": 80, "ftol": 1e-7, "disp": False},
        )
        h_opt = _project_to_constraints(res.x, target_sum)
        return h_opt, _compute_c5(h_opt)
    except Exception:
        return h_start, _compute_c5(h_start)


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    target_bound = 0.3808

    # ---- Determine base dimension ----
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        init_vec = np.asarray(init_vec, dtype=np.float64).ravel()
        n0 = init_vec.size
    else:
        n0 = 380
        init_vec = np.random.rand(n0)

    # ---- Resolution schedule (doubling while time permits) ----
    seq = [n0]
    # cautiously extend; cap at 2000 for speed
    while seq[-1] * 2 <= 2000:
        seq.append(seq[-1] * 2)

    best_h_global = None
    best_score_global = np.inf
    best_n_global = None

    for n in seq:
        # time bookkeeping
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        target_sum = n / 2.0
        # ---- Build initial binary vector for this resolution ----
        if n == n0:
            # use top‑half of the provided (or random) init_vec
            idx = np.argpartition(init_vec, -int(target_sum))[-int(target_sum) :]
            h_bin = np.zeros(n, dtype=np.float64)
            h_bin[idx] = 1.0
        else:
            # upscale previous best if available, else random binary
            if best_h_global is not None:
                factor = n // best_h_global.size
                h_bin = np.repeat(best_h_global, factor)
                h_bin = np.where(h_bin >= 0.5, 1.0, 0.0)
            else:
                h_bin = np.zeros(n, dtype=np.float64)
                ones_idx = np.random.choice(n, int(target_sum), replace=False)
                h_bin[ones_idx] = 1.0

        # ensure exact sum
        cur_sum = h_bin.sum()
        diff = int(target_sum - cur_sum)
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            add_idx = np.random.choice(zeros, diff, replace=False)
            h_bin[add_idx] = 1.0
        elif diff < 0:
            ones = np.where(h_bin == 1)[0]
            rm_idx = np.random.choice(ones, -diff, replace=False)
            h_bin[rm_idx] = 0.0

        # ---- Allocation of time for this resolution ----
        # use up to 60 % of remaining time (leaving room for later levels)
        res_time = min(remaining * 0.6, 300.0)
        res_deadline = time.time() + res_time

        # ---- Multiple random restarts with hill‑climbing ----
        local_best_h = None
        local_best_score = np.inf
        restart = 0
        while time.time() < res_deadline:
            restart += 1
            if restart == 1:
                start_h = h_bin.copy()
            else:
                # fresh random binary start
                start_h = np.zeros(n, dtype=np.float64)
                start_h[np.random.choice(n, int(target_sum), replace=False)] = 1.0

            # hill‑climb
            cand_h, cand_score = _binary_hill_climb(start_h, res_deadline, target_bound)

            # optional brief continuous refinement (if time remains)
            if time.time() < res_deadline - 0.5:
                cand_h, cand_score = _local_continuous_refine(cand_h, target_sum, res_deadline)

            if cand_score < local_best_score:
                local_best_score = cand_score
                local_best_h = cand_h.copy()
                if local_best_score < target_bound:
                    break

            # early exit if overall time is low
            if time.time() - start_time > budget_s * 0.95:
                break

        # ---- Record global best if improved ----
        if local_best_score < best_score_global:
            best_score_global = float(local_best_score)
            best_h_global = local_best_h.copy()
            best_n_global = n

        # ---- Stop early if target met ----
        if best_score_global < target_bound:
            break

    # ---- Final safety projection and scoring ----
    if best_h_global is None:
        # Fallback: simple uniform half‑filled vector
        best_n_global = n0
        best_h_global = np.zeros(best_n_global, dtype=np.float64)
        best_h_global[: int(best_n_global / 2)] = 1.0
        np.random.shuffle(best_h_global)

    best_h_global = _project_to_constraints(best_h_global, best_n_global / 2.0)

    # Evaluate with the official evaluator if available
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h_global))
        except Exception:
            final_score = float(best_score_global)
    else:
        final_score = float(best_score_global)

    return best_h_global.astype(np.float64), final_score, int(best_n_global)
