import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def hill_climb_binary(start_h, start_val, start_time, budget_s):
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_val = start_val
    rng = np.random.default_rng()
    while True:
        if time.time() - start_time >= budget_s - 0.01:
            break
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        val = compute_c5(cand)
        if val < best_val - 1e-15:
            best_h = cand
            best_val = val

    else:
        n = 80
        h0 = np.zeros(n, dtype=np.uint8)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1

    target_ones = n // 2

    # GA hyper‑parameters
    pop_size = 80
    tournament = 5
    cx_prob = 0.9
    mut_prob = 0.2

    population = [h0.copy()]
    for _ in range(pop_size - 1):
