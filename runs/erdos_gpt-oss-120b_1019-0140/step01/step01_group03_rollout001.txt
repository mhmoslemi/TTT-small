def projected_gradient_descent_adam(h0, max_iter, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    n=h0.size; s=n/2.
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = gradient_for_shift(h, shift)
        m = beta1 * m + (1 - beta1) * grad

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n: return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import math
import numpy as np

def _resample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation resampling to a new length."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def _project_onto_box_sum(v: np.ndarray, s: float, lo: float = 0.0, hi: float = 1.0, max_iter: int = 50) -> np.ndarray:
    """Euclidean projection of v onto {x: lo≤x_i≤hi, sum x_i = s}."""
    n = v.size
    # Trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    # Search interval for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure proper bracketing
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of correlation term for a given integer shift."""
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        # forward contribution (1 - h[i+shift]) for i = 0..n-shift-1
        grad[: n - shift] += 1.0 - h[shift:]
        # backward contribution -h[i] for i = 0..n-shift-1, placed at i+shift
        grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        grad[s:] += 1.0 - h[: n - s]
        grad[: n - s] -= h[s:]
    return grad

def _adam_optimize(h0: np.ndarray,
                   max_iter: int,
                   lr: float,
                   end_time: float) -> tuple[np.ndarray, float]:
    """Projected Adam optimizer on the max‑correlation objective."""
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_h = h.copy()
    best_score = _compute_c5(h)

    for t in range(1, max_iter + 1):
        if time.time() > end_time:
            break
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = _gradient_for_shift(h, shift)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h_new = h - step
        h_new = _project_onto_box_sum(h_new, target_sum)

        score = _compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step‑function approximation improving the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    # reserve a tiny safety margin
    end_time = start + budget_s - 2.0

    N = 256                     # fixed resolution for the final answer
    target_sum = N / 2.0

    def make_feasible(arr):
        arr = np.clip(arr, 0.0, 1.0)
        return _project_onto_box_sum(arr, target_sum)

    # ------------------------------------------------------------------

    candidates = []

        # b) Short refinement with a smaller step size

        best_h = make_feasible(best_h)
        best_score = _compute_c5(best_h)

    return best_h, best_score, best_h.size
