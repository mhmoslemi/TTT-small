def _simulated_annealing(...):
   start = time.time()
   cur_h = h_start.copy()
   cur_score = _c5_value(cur_h)
   best_h, best_score = cur_h.copy(), cur_score

   N = cur_h.size
   # compute total allowed time internal1 for this run.

   while True:
       now = time.time()
       elapsed = now - start
       if elapsed >= time_limit:
           break
       # fraction of remaining time: 1 - elapsed/time_limit
       frac = 1 - elapsed / time_limit
       # temperature: linearly reduce
       T = max(1e-6, T0 * frac)
       # max eps decreasing linearly? maybe max_eps = max_eps_initial * frac + min_eps
       max_eps = max_eps_initial * frac + 1e-4

       # pick two random distinct indices
       i, j = rng.integers(N, size=2)
       while i == j:
           j = rng.integers(N)
       hi, hj = cur_h[i], cur_h[j]
       # compute epsilon bounds
       up = min(1.0 - hi, hj)
      down = min(hi, 1.0 - hj)
       if up + down == 0: continue
       # sample epsilon within [-down, up] uniformly, with magnitude limited by max_eps:
       eps = (rng.random() * (up + down) - down)
       # cut to max_eps
       if eps > max_eps: eps = max_eps
       elif eps < -max_eps: eps = -max_eps
       if eps == 0: continue

       new_h = cur_h.copy()
       new_h[i] = hi + eps
       new_h[j] = hj - eps
       new_score = _c5_value(new_h)
       delta = new_score - cur_score
       if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
           cur_h = new_h
           cur_score = new_score
           if new_score < best_score:
               best_h, best_score = new_h.copy(), new_score

   return best_h, best_score

def _c5_value(h):
   N = h.size
   # Compute convolution via FFT: full correlation of h and (1-h) = conv(h, reversed(1-h))? Actually np.correlate with mode='full' computes sum_i h[i] * (1-h[i+k]) for each k (like cross-correlation with offset). For FFT convolution we can compute:
   # Use np.fft.ifft(np.fft.fft(h, n=2*N-1) * np.fft.fft(1-h, n=2*N-1).conj())? Wait convolution vs correlation.
   # The correlation of a and b (full) is conv(a, flip(b)). Let's compute conv = np.fft.ifft(np.fft.fft(a, n) * np.fft.fft(np.flip(b), n)). So better:
   # r = np.fft.ifft(np.fft.fft(h, n) * np.fft.fft(np.flip(1-h), n)).real
   # Then r is of length n (2*N-1). Then max(r) * dx.

   # Implementation:

   M = 2 * N - 1
   # Compute FFT of h padded to M
   H = np.fft.rfft(h, n=M)
   B = np.fft.rfft(np.flip(1.0 - h), n=M)
   conv = np.fft.irfft(H * B, n=M)

def _c5_value(h):
    N = h.shape[0]
    dx = 2.0 / N
    M = 2 * N - 1
    # Use rfft for speed
    H = np.fft.rfft(h, n=M)
    rev = np.flip(1.0 - h)
    R = np.fft.rfft(rev, n=M)
    conv = np.fft.irfft(H * R, n=M)
    max_val = conv.max()
    return float(max_val * dx)

        # Use simple linear

        while i == j:

        # bound by max_eps
        if eps > max_eps:

        elif eps < -max_eps:
            eps = -max_eps

                best_h = new_h.copy()
                best_score = new_score

    """Clip entries to [0,1] and adjust sum to len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function using FFT."""
    N = h.shape[0]
    dx = 2.0 / N
    M = 2 * N - 1
    # convolution of h with reversed (1-h) gives the linear correlation needed
    H = np.fft.rfft(h, n=M)
    R = np.fft.rfft(np.flip(1.0 - h), n=M)
    conv = np.fft.irfft(H * R, n=M)
    max_val = conv.max()
    return float(max_val * dx)

def _simulated_annealing(start_h: np.ndarray, time_limit: float,
                         rng: np.random.Generator,
                         init_temp: float = 0.5,
                         max_eps_initial: float = 0.5) -> tuple:
    """Pairwise‑exchange simulated annealing that preserves the sum constraint."""
    N = start_h.size
    cur_h = start_h.copy()
    cur_score = _c5_value(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = init_temp * frac
        max_eps = max_eps_initial * frac + 1e-8
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi, hj = cur_h[i], cur_h[j]
        up = min(1.0 - hi, hj)       # positive epsilon bound
        down = min(hi, 1.0 - hj)     # negative epsilon magnitude bound
        if up + down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        # enforce max_eps
        if eps > max_eps:
            eps = max_eps
        elif eps < -max_eps:
            eps = -max_eps
        if eps == 0.0:
            continue
        # apply move
        cur_h[i] = hi + eps
        cur_h[j] = hj - eps
        new_score = _c5_value(cur_h)
        delta = new_score - cur_score
        accept = (delta < 0.0) or (rng.random() < math.exp(-delta / (T + 1e-15)))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_h = cur_h.copy()
                best_score = new_score
        else:
            # revert
            cur_h[i] = hi
            cur_h[j] = hj
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Obtain initial construction
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 100
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)], dtype=np.float64)
        rng.shuffle(h0)

    h0 = _repair_vector(h0)
    best_h = h0
    best_score = _c5_value(best_h)

    start_time = time.time()
    total_budget = float(budget_s)

    # Phase 1 – low‑resolution optimisation
    phase1_budget = min(total_budget * 0.3, 300.0)
    if phase1_budget > 5.0:
        starts = [best_h]

        # random uniform start
        rand_vec = rng.random(best_h.size)
        starts.append(_repair_vector(rand_vec))

        # random binary start (half ones)
        half = best_h.size // 2
        bin_vec = np.concatenate([np.ones(half), np.zeros(best_h.size - half)], dtype=float)
        rng.shuffle(bin_vec)
        starts.append(_repair_vector(bin_vec))

        # circularly shifted version of current best
        shift = rng.integers(best_h.size)
        starts.append(np.roll(best_h, shift))

        per = phase1_budget / len(starts)
        for vec in starts:
            elapsed = time.time() - start_time
            remaining = total_budget - elapsed
            if remaining < per + 5.0:
                break
            cand_h, cand_score = _simulated_annealing(vec, per, rng,
                                                      init_temp=0.5,
                                                      max_eps_initial=0.5)
            if cand_score < best_score:
                best_h, best_score = cand_h, cand_score

    # Phase 2 – higher resolution
    elapsed = time.time() - start_time
    remaining = total_budget - elapsed
    if remaining > 20.0:
        # increase resolution (up to 800 points)
        target_n = min(800, max(best_h.size * 2, 200))
        target_n = int(target_n)

        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _repair_vector(h_up)

        # random uniform high‑res start
        rand_up = _repair_vector(rng.random(target_n))

        # random binary high‑res start
        half = target_n // 2
        bin_up = np.concatenate([np.ones(half), np.zeros(target_n - half)], dtype=float)
        rng.shuffle(bin_up)
        bin_up = _repair_vector(bin_up)

        high_starts = [h_up, rand_up, bin_up]
        per_high = remaining * 0.85 / len(high_starts)

        for vec in high_starts:
            elapsed = time.time() - start_time
            remaining = total_budget - elapsed
            if remaining < per_high + 5.0:
                break
            cand_h, cand_score = _simulated_annealing(vec, per_high, rng,
                                                      init_temp=0.5,
                                                      max_eps_initial=0.5)
            if cand_score < best_score:
                best_h, best_score = cand_h, cand_score

    # final repair and exact evaluation
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), int(best_h.size)
