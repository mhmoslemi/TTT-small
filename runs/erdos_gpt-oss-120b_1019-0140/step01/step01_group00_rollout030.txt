def solve_milp(n, time_limit):
    dx = 2.0 / n
    target = n // 2  # assume n even
    # variables
    h = cp.Variable(n, boolean=True)
    # m variable for min sum w
    m = cp.Variable(integer=True)   # integer bound: between 0 and target
    # constraints list
    constraints = [cp.sum(h) == target]
    # list of w vars can be stored in list of cp.Variable for each k and index.
    # could use dictionary.
    w_vars = []   # list of cp.Variable for all pair
    # also we will keep sum_w_k per k
    sum_w_per_k = []
    for k in range(1, n):
        # number of pairs = n - k
        # create w for each i
        w_k = cp.Variable(n - k, boolean=True)
        w_vars.append(w_k)
        # constraints linking w_k to h[i] and h[i+k]
        # w_k[i] <= h[i]
        # w_k[i] <= h[i+k]
        # w_k[i] >= h[i] + h[i+k] - 1
        # we can vectorize:
        # h_i = h[:n-k]; h_i_k = h[k:]; constraints:
        constraints += [
            w_k <= h[:n-k],
            w_k <= h[k:],
            w_k >= h[:n-k] + h[k:] - 1
        ]
        # sum constraint for min:
        sum_w = cp.sum(w_k)
        sum_w_per_k.append(sum_w)
    # now m <= sum_w_per_k for all k? Actually we want lower bound: sum_w_per_k >= m
    for sum_w in sum_w_per_k:
        constraints.append(sum_w >= m)
    # m bounded:
    constraints.append(m >= 0)
    constraints.append(m <= target)
    # objective maximize m
    objective = cp.Maximize(m)
    prob = cp.Problem(objective, constraints)
    # solve with time limit using solver parameter features:
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit*1000))

import time, numpy as np, cvxpy as cp, math

def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def solve_binary_milp(n, time_limit):
    # Build MILP and solve using cp.CBC with time limit if possible
    dx = 2.0/n
    target = n//2
    h = cp.Variable(n, boolean=True)
    constraints = [cp.sum(h) == target]
    # we'll introduce t as continuous variable
    t = cp.Variable()
    constraints.append(t >= 0)
    # For each shift k in range(-n+1, n) except 0
    for k in range(-n+1, n):
        if k == 0:
            continue
        if k > 0:
            h_prefix = h[:n-k]
            h_shift = h[k:]
        else:
            # k < 0: shift left
            kk = -k
            h_prefix = h[kk:]
            h_shift = h[:n-kk]
        # w binary product
        w = cp.Variable(len(h_prefix), boolean=True)
        constraints += [
            w <= h_prefix,
            w <= h_shift,
            w >= h_prefix + h_shift - 1
        ]
        overlap = cp.sum(h_prefix) - cp.sum(w)  # sum of h[i] * (1-h[i+k])
        constraints.append(overlap * dx <= t + 1e-9)  # t >= overlap*dx
    # objective minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    # solver options:
    try:
        prob.solve(solver=cp.CBC, maximumSeconds=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit*1000))
        except Exception:
            prob.solve(solver=cp.ECOS_BB, max_iters=200000)
    if prob.status not in ["optimal", "optimal_inaccurate", "feasible"]:
        return None, None
    h_sol = np.array(h.value).astype(float)
    c5 = compute_c5(h_sol)
    return h_sol, c5
```

def hill_climb_binary(n, budget, seed, start_h=None):
    rng = np.random.default_rng(seed)
    if start_h is not None:
        h = start_h.copy()
    else:
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
    best_h = h.copy()
    best_val = compute_c5(best_h)
    start = time.time()
    temperature = 0.1
    # schedule: linear decay to 0
    while time.time() - start < budget:
        # choose indices
        ones = np.flatnonzero(best_h == 1.0)
        zeros = np.flatnonzero(best_h == 0.0)
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        val = compute_c5(h)
        delta = val - best_val
        if delta < -1e-12 or rng.random() < math.exp(-delta / max(temperature, 1e-9)):
            # accept new h as current (but keep best if improved)
            if val < best_val - 1e-12:
                best_val = val
                best_h = h.copy()
        else:
            # reject, revert
            h[i] = 1.0
            h[j] = 0.0
        # temperature schedule: exponential decay based on elapsed
        elapsed = time.time() - start
        frac = elapsed / budget
        temperature = max(1e-4, 0.5 * (1 - frac))
        # restart occasionally if no improvement for many steps
        # maybe maintain counter...
    return best_h, best_val

def hill_climb_binary(n, budget_s, seed, start_h=None):
    rng = np.random.default_rng(seed)
    if start_h is None:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n//2, replace=False)] = 1.0
    else:
        h = start_h.astype(float).copy()
        # ensure integer?
        # maybe also random rebalancing if not exact half ones
        # ensure sum = n/2
        current_sum = int(round(np.sum(h)))
        target = n//2
        if current_sum != target:
            # adjust by flipping bits
            diff = target - current_sum
            # if diff >0: set diff zeros to 1
            # else set -diff ones to 0
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            if diff > 0:
                idx = rng.choice(zeros, diff, replace=False)
                h[idx] = 1.0
            elif diff < 0:
                idx = rng.choice(ones, -diff, replace=False)
                h[idx] = 0.0
    # compute current val
    current_val = compute_c5(h)
    best_h = h.copy()
    best_val = current_val
    start = time.time()
    # temperature schedule parameters
    T_initial = 0.5
    T_final = 1e-4
    # counters for improvement
    no_improve = 0
    while time.time() - start < budget_s:
        # random swap
        ones_idx = np.flatnonzero(h == 1.0)
        zeros_idx = np.flatnonzero(h == 0.0)
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        val = compute_c5(h)
        # acceptance logic
        delta = val - current_val
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = max(T_final, T_initial * (1 - frac))
        accept = False
        if delta < -1e-12:
            accept = True
        else:
            # accept with prob exp(-delta/T)
            prob = math.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            current_val = val
            no_improve = 0
            if val < best_val - 1e-12:
                best_val = val
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        # restart if stuck
        if no_improve >= 2000:
            # reinitialize
            h = np.zeros(n, dtype=float)
            h[rng.choice(n, n//2, replace=False)] = 1.0
            current_val = compute_c5(h)
            no_improve = 0
    return best_h, best_val

def solve_milp_binary(n):
    import cvxpy as cp
    dx = 2.0 / n
    target = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == target, t >= 0]
    # loop over shifts
    for k in range(1, n):
        # prefix h before shift
        h_prefix = h[:n - k]
        h_shift = h[k:]
        # product var w
        w = cp.Variable(n - k, boolean=True)
        constraints += [
            w <= h_prefix,
            w <= h_shift,
            w >= h_prefix + h_shift - 1
        ]
        overlap_pos = cp.sum(h_prefix) - cp.sum(w)
        overlap_neg = cp.sum(h_shift) - cp.sum(w)
        constraints += [overlap_pos * dx <= t, overlap_neg * dx <= t]
    prob = cp.Problem(cp.Minimize(t), constraints)
    # try CBC then GLPK_MI then ECOS_BB
    try:
        prob.solve(solver=cp.CBC, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
        except Exception:
            prob.solve(solver=cp.ECOS_BB, verbose=False)
    if prob.status not in ["optimal", "optimal_inaccurate", "feasible"]:
        return None, None
    h_sol = np.array(h.value).astype(float).flatten()
    c5 = compute_c5(h_sol)
    return h_sol, c5
```

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def solve_milp_binary(n: int):
    """Solve the binary Erdős C5 problem for a given n via MILP.
    Returns (h_opt, c5_opt) or (None, None) on failure."""
    if cp is None:
        return None, None

    dx = 2.0 / n
    target = n // 2

    # binary variables for the set
    h = cp.Variable(n, boolean=True)

    # auxiliary variable for the maximal overlap (continuous)
    t = cp.Variable()

    constraints = [cp.sum(h) == target, t >= 0]

    # For each shift magnitude generate product variables w_k
    for k in range(1, n):
        # overlapping parts
        h_pref = h[: n - k]          # length n-k
        h_suf = h[k:]                # length n-k
        w = cp.Variable(n - k, boolean=True)  # w_i = h_i * h_{i+k}

        # linearization of the product
        constraints += [
            w <= h_pref,
            w <= h_suf,
            w >= h_pref + h_suf - 1,
        ]

        # overlap for shift +k
        overlap_pos = cp.sum(h_pref) - cp.sum(w)
        constraints.append(overlap_pos * dx <= t)

        # overlap for shift -k (same w, different prefix sum)
        overlap_neg = cp.sum(h_suf) - cp.sum(w)
        constraints.append(overlap_neg * dx <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try a sequence of MILP solvers
    solved = False
    for solver in (cp.CBC, cp.GLPK_MI, cp.ECOS_BB):
        try:
            prob.solve(solver=solver, verbose=False)
            solved = True
            break
        except Exception:
            continue
    if not solved or prob.status not in ("optimal", "optimal_inaccurate", "feasible"):
        return None, None

    h_opt = np.rint(np.array(h.value)).astype(float)  # force binary 0/1
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt


def hill_climb_binary(n: int, budget_s: float, seed: int, start_h: np.ndarray = None):
    """Simple simulated‑annealing hill climb on binary vectors of length n."""
    rng = np.random.default_rng(seed)
    half = n // 2

    if start_h is not None:
        h = start_h.astype(float).copy()
        # enforce exact half‑ones
        cur = int(round(h.sum()))
        diff = half - cur
        if diff > 0:
            zeros = np.flatnonzero(h == 0)
            idx = rng.choice(zeros, diff, replace=False)
            h[idx] = 1.0
        elif diff < 0:
            ones = np.flatnonzero(h == 1)
            idx = rng.choice(ones, -diff, replace=False)
            h[idx] = 0.0
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, half, replace=False)] = 1.0

    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    no_improve = 0
    T0, T1 = 0.5, 1e-4  # temperature schedule bounds

    while time.time() - start < budget_s:
        # pick a random 1 and a random 0
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_val = compute_c5(h)

        # acceptance
        delta = new_val - cur_val
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = max(T1, T0 * (1.0 - frac))
        accept = False
        if delta < -1e-12:
            accept = True
        else:
            prob = math.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

        # occasional random restart to avoid stagnation
        if no_improve >= 2000:
            h = np.zeros(n, dtype=float)
            h[rng.choice(n, half, replace=False)] = 1.0
            cur_val = compute_c5(h)
            no_improve = 0

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Small problem (use initial_h_values if present)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init_small = np.array(initial_h_values, dtype=float)
        n_small = init_small.shape[0]
    else:
        n_small = 80
        init_small = None

    half_small = n_small // 2
    if init_small is not None:
        # binary by taking the largest half entries
        binary_small = np.zeros(n_small, dtype=float)
        top_idx = np.argpartition(-init_small, half_small - 1)[:half_small]
        binary_small[top_idx] = 1.0
    else:
        binary_small = np.zeros(n_small, dtype=float)
        binary_small[rng.choice(n_small, half_small, replace=False)] = 1.0

    best_h = binary_small.copy()
    best_c5 = compute_c5(best_h)

    # try exact MILP for the small size (quick)
    elapsed = time.time() - start_time
    if cp is not None and (budget_s - elapsed) > 30:
        h_opt, c5_opt = solve_milp_binary(n_small)
        if h_opt is not None and c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # ------------------------------------------------------------------
    # Large problem – hill‑climb on a larger discretisation
    # ------------------------------------------------------------------
    n_large = 200                       # final resolution (even)
    half_large = n_large // 2

    # build a deterministic upscale of the current best small solution
    repeats = n_large // n_small
    remainder = n_large % n_small
    large_h = np.tile(best_h, repeats)
    if remainder:
        large_h = np.concatenate([large_h, best_h[:remainder]])
    # ensure exact half‑ones
    cur_sum = int(round(large_h.sum()))
    diff = half_large - cur_sum
    if diff > 0:
        zeros = np.flatnonzero(large_h == 0)
        if len(zeros) >= diff:
            large_h[rng.choice(zeros, diff, replace=False)] = 1.0
    elif diff < 0:
        ones = np.flatnonzero(large_h == 1)
        if len(ones) >= -diff:
            large_h[rng.choice(ones, -diff, replace=False)] = 0.0

    # hill‑climbing budget: remaining time minus a small safety margin
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed - 5.0)
    if remaining > 5.0:
        h_large, c5_large = hill_climb_binary(
            n_large, remaining, seed + 1, start_h=large_h
        )
        if c5_large < best_c5:
            best_h = h_large
            best_c5 = c5_large

    # final safety – enforce exact binary form and correct sum
    best_h = np.rint(np.clip(best_h, 0.0, 1.0)).astype(float)
    cur = int(round(best_h.sum()))
    if cur != best_h.shape[0] // 2:
        # simple repair
        diff = (best_h.shape[0] // 2) - cur
        if diff > 0:
            zeros = np.flatnonzero(best_h == 0)
            if len(zeros) >= diff:
                best_h[rng.choice(zeros, diff, replace=False)] = 1.0
        elif diff < 0:
            ones = np.flatnonzero(best_h == 1)
            if len(ones) >= -diff:
                best_h[rng.choice(ones, -diff, replace=False)] = 0.0

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_h.shape[0]
