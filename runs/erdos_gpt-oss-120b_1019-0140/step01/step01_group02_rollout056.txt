import time
import math
import numpy as np
import scipy.optimize as opt

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust one entry (or a few) so that sum(h) == target while staying in [0,1]."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Choose direction
    if diff > 0:
        # need to increase some entry not at 1
        idx = np.where(h < 1.0 - 1e-12)[0]
        if idx.size == 0:
            # all at 1, distribute evenly (should not happen)
            h += diff / h.size
            np.clip(h, 0.0, 1.0, out=h)
            return h
        i = idx[0]
        delta = min(diff, 1.0 - h[i])
    else:
        # need to decrease some entry not at 0
        idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            h += diff / h.size
            np.clip(h, 0.0, 1.0, out=h)
            return h
        i = idx[0]
        delta = max(diff, -h[i])
    h[i] += delta
    # recurse if not fully fixed
    return _repair_sum(h, target)

def _simulate_annealing(init_h: np.ndarray, budget_s: float, start_time: float) -> tuple:
    """Simple simulated‑annealing optimiser for C5."""
    rng = np.random.default_rng()
    h = init_h.copy()
    n = h.size
    target_sum = n / 2.0
    # Annealing schedule parameters (tuned for quick convergence)
    max_step = 0.4
    min_step = 1e-4
    max_temp = 5e-3
    min_temp = 1e-8

    best_h = h.copy()
    best_c5 = _compute_c5(h)

    while True:
        now = time.time()
        if now - start_time >= budget_s:
            break
        frac = (now - start_time) / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        # feasible delta so both stay inside [0,1]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        low = max(delta_min, -step)
        high = min(delta_max, step)
        if low >= high:
            continue
        delta = rng.random() * (high - low) + low

        h[i] = hi + delta
        h[j] = hj - delta
        new_c5 = _compute_c5(h)
        delta_obj = new_c5 - best_c5  # use best_c5 as reference for Metropolis

        if new_c5 < best_c5 or rng.random() < math.exp(-delta_obj / temp):
            # accept move
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # reject
            h[i] = hi
            h[j] = hj

    # final repair (tiny drift)
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5

def _refine_with_slsqp(h_start: np.ndarray, budget_s: float, start_time: float) -> tuple:
    """Run a few SLSQP refinements starting from h_start."""
    n = h_start.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}

    def obj(h):
        # objective = max correlation (unnormalised)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr)

    best_h = h_start.copy()
    best_c5 = _compute_c5(best_h)

    # try a couple of restarts within remaining time
    attempts = 3
    for _ in range(attempts):
        # time guard
        if time.time() - start_time >= budget_s * 0.98:
            break
        # random perturbation for diversity
        perturb = np.random.default_rng().normal(scale=0.05, size=n)
        h0 = np.clip(best_h + perturb, 0.0, 1.0)
        h0 = _repair_sum(h0, target_sum)

        res = opt.minimize(
            obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": 400, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            continue
        h_opt = _repair_sum(np.clip(res.x, 0.0, 1.0), target_sum)
        c5_opt = _compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose a reasonably fine discretisation
    n = 200  # even, gives integer half‑sum
    target_sum = n / 2.0

    # Initialise from provided construction if present, otherwise random half‑ones
    try:
        init = np.asarray(initial_h_values, dtype=float)
        m = init.shape[0]
        # Upsample to n points via linear interpolation
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
    except NameError:
        # Simple half‑ones configuration shuffled randomly
        h0 = np.zeros(n, dtype=float)
        half = n // 2
        h0[:half] = 1.0
        if n % 2:
            h0[half] = 0.5
        rng.shuffle(h0)

    # Ensure feasibility exactly
    h0 = _repair_sum(h0, target_sum)

    # ----- Phase 1: Simulated annealing ------------------------------------
    remaining = budget_s * 0.7  # allocate ~70% time to SA
    best_h, best_c5 = _simulate_annealing(h0, remaining, start_time)

    # ----- Phase 2: Local refinement with SLSQP ---------------------------
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5.0:
        # give SLSQP at most the rest of the budget
        best_h, best_c5 = _refine_with_slsqp(best_h, remaining_time, start_time)

    # Final sanity check (clip, repair, recompute)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
