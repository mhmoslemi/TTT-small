import time

import numpy as np

def compute_c5_int(h):
    """Return integer max overlap for binary vector h (0/1)."""
    # Using np.correlate: full mode gives correlation values for all shifts.
    # Correlation of h with 1-h; both are integer arrays.
    corr = np.correlate(h, 1 - h, mode='full')
    return int(corr.max())

def random_binary(n, rng):
    """Return a binary vector of length n with exactly n//2 ones."""
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1
    return h

        order = np.argsort(-init_vals, kind='stable')
        h0 = np.zeros(n, dtype=int)
        h0[order[: n // 2]] = 1
    else:
        n = 80
        h0 = random_binary(n, rng)

    dx = 2.0 / n
    target_int = int(math.floor(0.3808 / dx + 1e-12))  # 15 for n=80

    best_int = compute_c5_int(best_h)

    # early exit if already good enough
    if best_int <= target_int:
        return best_h.astype(float), best_int * dx, n

    final_limit = start_time + budget_s * 0.99
    cur_h = best_h.copy()
    cur_int = best_int
    no_improve = 0
    while time.time() < final_limit and no_improve < 5000:
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        cand = cur_h.copy()
        cand[i1] = 0
        cand[i0] = 1
        cand_int = compute_c5_int(cand)
        if cand_int < cur_int:
            cur_h = cand
            cur_int = cand_int
            no_improve = 0
            if cur_int < best_int:
                best_h = cur_h.copy()
                best_int = cur_int
                if best_int <= target_int:
                    break
        else:
            no_improve += 1

    return best_h.astype(float), best_int * dx, n
