def compute_corr_vector(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_gradient(h, tau):
    corr = compute_corr_vector(h)
    # prevent overflow: subtract max
    max_c = np.max(corr)
    expo = np.exp((corr - max_c) / tau)  # shift to avoid overflow
    w = expo / expo.sum()
    # compute weighted forward/backward using convolution
    n = h.size
    # w is length 2n-1, order corresponds to shift index from -(n-1) to (n-1)
    # Weighted forward: sum_k w_k * h_{i+k}
    conv_forward = np.convolve(w[::-1], h, mode='full')
    weighted_forward = conv_forward[n-1:2*n-1]  # length n
    conv_backward = np.convolve(w, h, mode='full')
    weighted_backward = conv_backward[n-1:2*n-1]
    grad = 1.0 - weighted_forward - weighted_backward
    return grad

    weighted_b = conv_b[n-1:2*n-1]

def compute_gradient_shift(h, shift):
    n = h.size
    grad = np.ones_like(h)
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad

import time, math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def upsample(arr, new_n):
    """Linearly interpolate a 1-D array to length new_n."""
    arr = np.asarray(arr, dtype=float)
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    # trivial cases
    if s <= lo * n:
        return np.full(n, lo, dtype=float)
    if s >= hi * n:
        return np.full(n, hi, dtype=float)

    # binary search on lambda for clipping
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure proper bracketing
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Discrete C5 value for a candidate h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient_shift(h, shift):
    """Gradient of the correlation term for a specific integer shift."""
    n = h.size
    g = np.ones_like(h)
    if shift >= 0:
        g[: n - shift] -= h[shift:]
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        g[s:] -= h[: n - s]
        g[: n - s] -= h[s:]
    return g


def smooth_gradient(h, corr, tau):
    """Smoothed gradient of max_corr using temperature tau."""
    n = h.size
    max_c = np.max(corr)
    # weights proportional to exp((corr-max)/tau)
    w = np.exp((corr - max_c) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # weighted forward/backward sums via convolution
    conv_f = np.convolve(w[::-1], h, mode='full')
    conv_b = np.convolve(w, h, mode='full')
    # slice central part (length n)
    start = n - 1
    weighted_f = conv_f[start:start + n]
    weighted_b = conv_b[start:start + n]
    return 1.0 - weighted_f - weighted_b

# Core optimization routine
# ----------------------------------------------------------------------
def refine_candidate(h0, deadline, rng,
                    max_iters=4000,
                    step0=0.8,
                    tau_start=5.0,
                    tau_end=0.01):
    """Projected (smoothed) gradient descent + final subgradient polish."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(h)

    # schedule for temperature (exponential decay)
    if max_iters > 0:
        tau_factor = (tau_end / max(tau_start, 1e-12)) ** (1.0 / max_iters)
    else:
        tau_factor = 1.0
    tau = tau_start

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        grad = smooth_gradient(h, corr, tau)
        step = step0 / math.sqrt(it)
        h = project_onto_box_sum(h - step * grad, target_sum)
        score = np.max(corr) * dx
        if score < best_score:
            best_score = score
            best_h = h.copy()
        tau *= tau_factor

    # final refinement using the active shift (subgradient)
    extra = max_iters // 4
    for it in range(1, extra + 1):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient_shift(h, shift)
        step = step0 / math.sqrt(it + max_iters)
        h = project_onto_box_sum(h - step * grad, target_sum)
        score = np.max(corr) * dx
        if score < best_score:
            best_score = score
            best_h = h.copy()

    # small local exchange polishing
    best_h, best_score = exchange_polish(best_h, best_score, deadline, rng)

    return best_h, best_score


def exchange_polish(h, cur_score, deadline, rng,
                    rounds=800):
    """Attempt random pairwise mass exchanges to lower C5."""
    n = h.size
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best = cur_score
    for _ in range(rounds):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        # allowable delta keeping values in [0,1]
        delta_max = min(best_h[i], 1.0 - best_h[j])
        delta_min = -min(best_h[j], 1.0 - best_h[i])
        if delta_max <= delta_min:
            continue
        # random delta scaled to a modest size
        delta = (rng.random() - 0.5) * 2.0 * (delta_max - delta_min) * 0.2
        delta = np.clip(delta, delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best:
            best = score
            best_h = h_new
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin
    target_n = 256  # resolution for primary search
    candidates = []

    # 1) Use parent construction if available

        return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),

    candidates.append(thue_morse(target_n))

    # 3) Random balanced binary vectors
    for _ in range(2):
        vec = np.zeros(target_n, dtype=float)
        idx = rng.choice(target_n, size=target_n // 2, replace=False)
        vec[idx] = 1.0
        candidates.append(vec)

    # 4) Random fractional starts
    for _ in range(2):
        vec = rng.random(target_n)
        candidates.append(project_onto_box_sum(vec, target_n / 2.0))

    best_h = None
    best_score = float('inf')

    # Process each base candidate with a relatively long refinement
    for base in candidates:
        if time.time() > deadline:
            break
        h_opt, score_opt = refine_candidate(base, deadline, rng,

                                            tau_start=5.0,
                                            tau_end=0.01)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    # Additional random restarts while time permits
    while time.time() < deadline:
        vec = rng.random(target_n)
        start_vec = project_onto_box_sum(vec, target_n / 2.0)
        h_opt, score_opt = refine_candidate(start_vec, deadline, rng,

    # Final sanity projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
