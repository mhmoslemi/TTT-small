import time
import math
import numpy as np

def _compute_O_vals(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1-h); returns the raw overlap sums (without dx)."""
    return np.correlate(h, 1.0 - h, mode="full")

def _targeted_swap_positions(h: np.ndarray, lag: int, rng: np.random.Generator):
    """Find a pair (i_bad, i_good) where swapping a 1 at i_bad with a 0 at i_good
    strictly reduces the overlap for the given lag. Returns (i_bad, i_good) or (None, None)."""
    n = h.shape[0]
    if lag >= 0:
        i = np.arange(0, n - lag)
        j = i + lag
    else:
        i = np.arange(-lag, n)
        j = i + lag
    hi = h[i]
    hj = h[j]
    bad_mask = (hi == 1.0) & (hj == 0.0)   # 1 aligns with 0 -> contributes to O_lag
    good_mask = (hi == 0.0) & (hj == 1.0)  # 0 aligns with 1 -> safe to turn into 1
    bad_idxs = i[bad_mask]
    good_idxs = i[good_mask]
    if bad_idxs.size == 0 or good_idxs.size == 0:
        return None, None
    i_bad = rng.choice(bad_idxs)
    i_good = rng.choice(good_idxs)
    return i_bad, i_good

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a binary step function h on [0,2] that minimizes C5.
    Returns:
        h_values: numpy array with values in [0,1] (binary)
        c5_bound: achieved C5 value (float)
        n_points: length of the discretisation (int)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose discretisation size.
    try:
        init_len = len(initial_h_values)  # may be undefined
    except NameError:
        init_len = None

    if init_len and init_len >= 20:
        n = init_len if (init_len % 2 == 0) else init_len + 1
    else:
        n = 100                     # moderate size, easy to evaluate
    if n % 2 != 0:
        n += 1                     # ensure even for exact half weight

    half = n // 2
    target_Omax = int(np.floor(0.3808 * half))   # Omax that guarantees C5 <= 0.3808

    max_temp = 0.5
    min_temp = 1e-6

    best_h = None
    best_Omax = np.inf

    # Main multi‑restart loop
    while True:
        if time.time() - start_time >= budget_s:
            break

        # Random binary start with exact half weight
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=half, replace=False)
        h[ones] = 1.0

        O_vals = _compute_O_vals(h)
        cur_Omax = int(O_vals.max())
        stagnation = 0

        # Local improvement loop
        while True:
            now = time.time()
            if now - start_time >= budget_s:
                break

            # Record global best
            if cur_Omax < best_Omax:
                best_Omax = cur_Omax
                best_h = h.copy()
                if best_Omax <= target_Omax:
                    break

            # Identify one of the worst lags
            max_idxs = np.where(O_vals == O_vals.max())[0]
            idx_max = rng.choice(max_idxs)
            lag = int(idx_max - (n - 1))

            # Try a targeted swap that improves the worst lag
            i_bad, i_good = _targeted_swap_positions(h, lag, rng)
            if i_bad is not None:
                h[i_bad] = 0.0
                h[i_good] = 1.0
                new_O_vals = _compute_O_vals(h)
                new_Omax = int(new_O_vals.max())
                if new_Omax <= cur_Omax:
                    cur_Omax = new_Omax
                    O_vals = new_O_vals
                    stagnation = 0
                    continue
                # revert if not beneficial
                h[i_bad] = 1.0
                h[i_good] = 0.0

            # Fallback: random swap (maintain weight)
            pos_one = rng.choice(np.where(h == 1.0)[0])
            pos_zero = rng.choice(np.where(h == 0.0)[0])
            h[pos_one] = 0.0
            h[pos_zero] = 1.0
            new_O_vals = _compute_O_vals(h)
            new_Omax = int(new_O_vals.max())

            # Simulated‑annealing acceptance
            elapsed = now - start_time
            frac = elapsed / budget_s
            temp = max_temp * (1.0 - frac) + min_temp
            delta = new_Omax - cur_Omax
            accept = False
            if delta <= 0:
                accept = True
            else:
                if rng.random() < math.exp(-delta / temp):
                    accept = True

            if accept:
                cur_Omax = new_Omax
                O_vals = new_O_vals
                stagnation = 0 if delta <= 0 else stagnation + 1
            else:
                # revert the random swap
                h[pos_one] = 1.0
                h[pos_zero] = 0.0
                stagnation += 1

            # Restart if stuck too long
            if stagnation > 5000:
                break

        if best_Omax <= target_Omax:
            break

    if best_h is None:
        # Fallback deterministic construction
        best_h = np.zeros(n, dtype=float)
        best_h[:half] = 1.0
        best_Omax = int(_compute_O_vals(best_h).max())

    c5_bound = best_Omax * (2.0 / n)   # convert raw overlap to integral value
    return best_h, c5_bound, n
