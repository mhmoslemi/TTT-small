def annealing_one_restart(h_start, cur_c5, time_limit, start_time):
    # returns local best h and c5
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    cur = cur_c5
    best_local_h = h.copy()
    best_local_c5 = cur

    # define total steps based on time left
    steps = 0
    # temperature schedule param
    T0 = 0.005
    # Determine max steps per restart to allow some progress
    max_steps = 200000  # may be adjusted
    while time.time() - start_time < time_limit and steps < max_steps:
        steps += 1
        # compute temperature linearly decreasing
        t_frac = steps / max_steps
        T = T0 * (1 - t_frac)

        # select random swap
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h.astype(float))
        delta = new_c5 - cur
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            # accept
            cur = new_c5
            # update best local if improved
            if cur < best_local_c5 - 1e-12:
                best_local_c5 = cur
                best_local_h = h.copy()
        else:
            # reject: revert swap
            h[i] = 1
            h[j] = 0
        # optional early break if T small and no improvement for certain iterations
        # Not needed
    return best_local_h, best_local_c5

    improved = True
    while improved:
        improved = False
        # get indices
        ones = np.flatnonzero(h == 1)

    mask = rng.integers(0, 2, size=n, dtype=bool)  # random boolean mask
    child = np.where(mask, p1, p2)   # take bits from p1 where mask True, else p2

        ones_idx = np.flatnonzero(child == 1)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # h must be float
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def to_binary(vals):
    n = vals.shape[0]
    half = n // 2
    idx = np.argsort(-vals)[:half]
    h = np.zeros(n, dtype=int)
    h[idx] = 1
    return h

def tournament_select(pop, scores, rng, size=3):
    idx = rng.choice(pop.shape[0], size, replace=False)
    best = idx[np.argmin(scores[idx])]
    return pop[best]

def crossover(p1, p2, target_sum, rng):
    # random mask
    mask = rng.integers(0, 2, size=p1.shape[0], dtype=bool)
    child = np.where(mask, p1, p2).astype(int)
    cur = int(child.sum())
    if cur > target_sum:
        # too many ones: turn off some
        excess = cur - target_sum
        ones_idx = np.flatnonzero(child == 1)
        to_off = rng.choice(ones_idx, excess, replace=False)
        child[to_off] = 0
    elif cur < target_sum:
        deficit = target_sum - cur
        zeros_idx = np.flatnonzero(child == 0)
        to_on = rng.choice(zeros_idx, deficit, replace=False)
        child[to_on] = 1
    return child

def mutate(ind, target_sum, rng, p_mut=0.2):
    if rng.random() < p_mut:
        # perform a single swap
        ones = np.flatnonzero(ind == 1)
        zeros = np.flatnonzero(ind == 0)
        if len(ones) == 0 or len(zeros) == 0:
            return ind
        i = rng.choice(ones)
        j = rng.choice(zeros)
        ind[i] = 0
        ind[j] = 1
    return ind

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size

        target_sum = n // 2
        h0 = np.zeros(n, dtype=int)
        h0[rng.choice(n, target_sum, replace=False)] = 1

    target_sum = n // 2  # exact half

    # Population initialization
    pop_size = 50
    # Create list of individuals
    population = np.empty((pop_size, n), dtype=int)
    # Insert initial solution as first individual
    population[0] = h0
    # Fill rest with random individuals
    for i in range(1, pop_size):
        ind = np.zeros(n, dtype=int)
        ind[rng.choice(n, target_sum, replace=False)] = 1
        population[i] = ind

    for i in range(pop_size):

    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # Genetic algorithm loop
    # Parameters

        n = init_vals.shape[0]
        h0 = to_binary(init_vals)
    else:
        n = 80
        target_sum = n // 2
        h0 = np.zeros(n, dtype=int)
        h0[rng.choice(n, target_sum, replace=False)] = 1

    target_sum = n // 2

    # Population initialization
    pop_size = 50
    population = np.empty((pop_size, n), dtype=int)
    population[0] = h0
    for i in range(1, pop_size):
        ind = np.zeros(n, dtype=int)
        ind[rng.choice(n, target_sum, replace=False)] = 1
        population[i] = ind

    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        scores[i] = compute_c5(population[i].astype(float))

    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # Genetic algorithm main loop
    tournament_k = 3
    p_mut = 0.3
    while True:
        if time.time() - start >= budget_s - 2.0:
            break

        parent1 = tournament_select(population, scores, rng, size=tournament_k)
        parent2 = tournament_select(population, scores, rng, size=tournament_k)

        child = crossover(parent1, parent2, target_sum, rng)
        child = mutate(child, target_sum, rng, p_mut=p_mut)

        child_score = compute_c5(child.astype(float))

        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()
                if best_score < 0.3808:
                    break

    # Final hill‑climbing refinement (strict improvement)
    while True:
        if time.time() - start >= budget_s - 0.5:
            break
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best_h[i] = 0
        best_h[j] = 1
        new_score = compute_c5(best_h.astype(float))
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best_h[i] = 1
            best_h[j] = 0

    best_h_float = np.clip(best_h.astype(float), 0.0, 1.0)
    return best_h_float, best_score, n
