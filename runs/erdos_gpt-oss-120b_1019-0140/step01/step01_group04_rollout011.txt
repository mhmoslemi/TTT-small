def compute_c5(v):
    n = len(v)
    dx = 2.0 / n
    a = v
    b = 1.0 - v
    b_rev = b[::-1]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b_rev, size)
    conv = np.fft.irfft(fa * fb, size)[:2*n-1]
    return np.max(conv) * dx

def _repair_vector(v, n, max_iter=25):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # If sum too large, reduce some positive entries; if too small, increase zeros.
        if diff > 0:
            # need to reduce sum
            mask = v > 0.0
        else:

        n_free = mask.sum()
        if n_free == 0:

        v = np.clip(v, 0.0, 1.0)

    s = v.sum()
    if abs(s - target) > 1e-9:

        v = np.clip(v, 0.0, 1.0)

        if abs(s - target) > 1e-8:
