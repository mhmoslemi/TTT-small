    h = h_start.copy()
    best_h = h.copy()
    best_score = _c5_value(h)
    N = len(h)
    start = time.time()
    while time.time() - start < time_limit:
        # compute correlation and worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        d = max_idx - (N - 1)
        if d < 0:
            d = -d
        # compute diff array
        if d == 0:
            # skip zero shift because product sum is zero on average? Actually we don't care, but we cannot improve g0 via pairwise moving because diff = zeros; skip random move.
            i = rng.integers(0, N)
            j = rng.integers(0, N)
            if i == j:
                continue
            # random epsilon
            hi, hj = h[i], h[j]
            up = min(1 - hi, hj)
            down = min(hi, 1 - hj)
            if up + down == 0:
                continue

                best_score = new_score
                best_h = h_new.copy()
            continue

        pos = np.where(diff > 1e-12)[0]
        if pos.size == 0:

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip v to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    n = v.size
    target = n * 0.5
    s = v.sum()
    if s == 0.0:
        v[:] = target / n
        np.clip(v, 0.0, 1.0, out=v)
        return v
    # simple scaling + re‑clip; iterate once for safety
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        # distribute residual uniformly over non‑saturated entries
        diff = target - s2
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size:
            add = diff / free.size
            v[free] = np.clip(v[free] + add, 0.0, 1.0)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simple SA preserving sum and box constraints."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.01            # initial temperature
    max_eps = 0.2        # absolute move cap
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)      # possible positive transfer (i -> j)
        down = min(hi, 1.0 - hj)    # possible negative transfer (j -> i)
        if up + down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _targeted_descent(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Greedy descent focused on the worst shift."""
    h = h_start.copy()
    best_h = h.copy()
    best_score = _c5_value(h)
    N = h.size
    start = time.time()
    while time.time() - start < time_limit:
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        d = max_idx - (N - 1)
        d = abs(d)                     # symmetry, treat non‑negative
        cur_score = best_score        # current best for this iteration
        if d == 0:
            # fallback random transfer
            i, j = rng.integers(0, N, size=2)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up + down == 0.0:
                continue
            eps = rng.random() * (up + down) - down
            h_new = h.copy()
            h_new[i] = hi + eps
            h_new[j] = hj - eps
            new_score = _c5_value(h_new)
            if new_score < best_score:
                h = h_new
                best_score = new_score
                best_h = h_new.copy()
            continue

        # diff = h[i] - h[i+d] over overlapping region
        diff = h[:N - d] - h[d:]
        pos = np.where(diff > 1e-12)[0]
        if pos.size == 0:
            # try opposite direction (move from larger at i+d to i)
            neg = np.where(diff < -1e-12)[0]
            if neg.size == 0:
                # all equal – random perturbation
                i, j = rng.integers(0, N, size=2)
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                up = min(1.0 - hi, hj)
                down = min(hi, 1.0 - hj)
                if up + down == 0.0:
                    continue
                eps = rng.random() * (up + down) - down
                h_new = h.copy()
                h_new[i] = hi + eps
                h_new[j] = hj - eps
                new_score = _c5_value(h_new)
                if new_score < best_score:
                    h = h_new
                    best_score = new_score
                    best_h = h_new.copy()
                continue
            idx = neg[0]
            i_big = idx + d
            j_small = idx
            max_eps = min(h[i_big], 1.0 - h[j_small])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps
            h_new = h.copy()
            h_new[i_big] -= eps
            h_new[j_small] += eps
            new_score = _c5_value(h_new)
            if new_score < best_score:
                h = h_new
                best_score = new_score
                best_h = h_new.copy()
            continue

        # choose the pair with largest positive diff
        idx = pos[np.argmax(diff[pos])]
        i = idx
        j = idx + d
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        # limit step to half the excess to avoid overshoot
        step = min(max_eps, diff[idx] * 0.5)
        if step <= 0.0:
            step = max_eps * rng.random() * 0.1
        h_new = h.copy()
        h_new[i] -= step
        h_new[j] += step
        new_score = _c5_value(h_new)
        if new_score < best_score:
            h = h_new
            best_score = new_score
            best_h = h_new.copy()
        # if not improving, reject (keep current h)
    return best_h, best_score

def _resample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to change number of points."""
    old_n = h.size
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new
