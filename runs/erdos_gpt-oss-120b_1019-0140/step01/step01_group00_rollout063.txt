    # uniform: for each index, choose from p1 or p2 with prob 0.5
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2).astype(np.int8)
    # adjust number of ones to target (n/2)
    current = child.sum()
    if current > target_sum:
        # need to turn some ones to zero
        ones_idx = np.flatnonzero(child)
        # randomly select (current - target) to zero
        to_zero = rng.choice(ones_idx, int(current - target_sum), replace=False)

        zeros_idx = np.flatnonzero(1 - child)

    return child

def crossover(p1, p2, rng, target):
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2).astype(np.int8)
    # adjust count to target
    diff = int(child.sum() - target)
    if diff > 0:
        # too many ones, turn diff ones to zero
        ones_idx = np.flatnonzero(child)
        to_zero = rng.choice(ones_idx, diff, replace=False)
        child[to_zero] = 0
    elif diff < 0:
        # too few ones
        zeros_idx = np.flatnonzero(1 - child)
        to_one = rng.choice(zeros_idx, -diff, replace=False)
        child[to_one] = 1
    return child

def mutate(h, rng):

        return h
    i = rng.choice(ones_idx)
    j = rng.choice(zeros_idx)
    h[i], h[j] = h[j], h[i]
    return h

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def crossover(p1, p2, rng, target_sum):
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2).astype(np.int8)
    diff = int(child.sum() - target_sum)
    if diff > 0:
        # too many ones
        ones_idx = np.flatnonzero(child)
        to_zero = rng.choice(ones_idx, diff, replace=False)
        child[to_zero] = 0
    elif diff < 0:
        # too few ones
        zeros_idx = np.flatnonzero(1 - child)
        to_one = rng.choice(zeros_idx, -diff, replace=False)
        child[to_one] = 1
    return child

def mutate(h, rng):
    ones_idx = np.flatnonzero(h)
    zeros_idx = np.flatnonzero(1 - h)
    if len(ones_idx) == 0 or len(zeros_idx) == 0:
        return h
    i = rng.choice(ones_idx)
    j = rng.choice(zeros_idx)
    h[i], h[j] = h[j], h[i]
    return h

def hill_climb_binary(h, score, rng, deadline):
    # try swaps until time runs out
    best_h = h.copy()
    best_score = score
    while time.time() < deadline:
        # attempt random swap
        ones_idx = np.flatnonzero(best_h)
        zeros_idx = np.flatnonzero(1 - best_h)
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = compute_c5(cand)
        if cand_score + 1e-12 < best_score:
            best_h = cand
            best_score = cand_score
    return best_h, best_score

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap C5 for binary step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def crossover(p1, p2, rng, target):
    """Uniform crossover keeping exactly target ones."""
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2).astype(np.int8)
    diff = int(child.sum() - target)
    if diff > 0:
        ones = np.flatnonzero(child)
        child[rng.choice(ones, diff, replace=False)] = 0
    elif diff < 0:
        zeros = np.flatnonzero(1 - child)
        child[rng.choice(zeros, -diff, replace=False)] = 1
    return child

def mutate(h, rng):
    """Swap a 1 and a 0 (preserves count)."""
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    if len(ones) == 0 or len(zeros) == 0:
        return h
    i = rng.choice(ones)
    j = rng.choice(zeros)
    h[i], h[j] = h[j], h[i]
    return h

def improve_local(h, score, rng, attempts=2):
    """A few greedy swaps to possibly lower the score."""
    best, best_score = h, score
    for _ in range(attempts):
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = compute_c5(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
    return best, best_score

def hill_climb_binary(h, score, rng, deadline):
    """Simple hill‑climbing by swaps until time runs out."""
    best, best_score = h.copy(), score
    while time.time() < deadline:
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(1 - best)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = compute_c5(cand)
        if cand_score + 1e-12 < best_score:
            best, best_score = cand, cand_score
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- problem size ---
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        base = np.array(initial_h_values, dtype=float)
        idx = np.argsort(-base)[: n // 2]
        init_bin = np.zeros(n, dtype=np.int8)
        init_bin[idx] = 1
        have_init = True
    else:
        n = 80
        have_init = False
    target = n // 2

    # --- initial population ---
    pop_size = 80
    population = []
    scores = []
    if have_init:
        population.append(init_bin)
        scores.append(compute_c5(init_bin))
    while len(population) < pop_size:
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, target, replace=False)] = 1
        population.append(h)
        scores.append(compute_c5(h))
    population = np.array(population, dtype=np.int8)
    scores = np.array(scores, dtype=float)

    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = float(scores[best_idx])

    tournament = 3
    mut_prob = 0.2

    # --- genetic optimisation ---
    while time.time() - start < budget_s * 0.9:
        children = []
        child_scores = []
        for _ in range(pop_size):
            # tournament selection
            p_idx = rng.choice(pop_size, tournament, replace=False)
            parent1 = population[p_idx[np.argmin(scores[p_idx])]]
            p_idx2 = rng.choice(pop_size, tournament, replace=False)
            parent2 = population[p_idx2[np.argmin(scores[p_idx2])]]
            child = crossover(parent1, parent2, rng, target)
            if rng.random() < mut_prob:
                child = mutate(child, rng)
            c0 = compute_c5(child)
            child, c0 = improve_local(child, c0, rng, attempts=2)
            children.append(child)
            child_scores.append(c0)

        combined_pop = np.concatenate([population, np.array(children, dtype=np.int8)], axis=0)
        combined_scores = np.concatenate([scores, np.array(child_scores, dtype=float)], axis=0)

        # keep best individuals
        keep = np.argpartition(combined_scores, pop_size)[:pop_size]
        population = combined_pop[keep]
        scores = combined_scores[keep]

        cur_best_idx = int(np.argmin(scores))
        cur_best_score = float(scores[cur_best_idx])
        if cur_best_score < best_score - 1e-12:
            best_score = cur_best_score
            best_h = population[cur_best_idx].copy()

        if best_score < 0.3808:
            break

    # --- final greedy refinement ---
    deadline = start + budget_s * 0.99
    best_h, best_score = hill_climb_binary(best_h, best_score, rng, deadline)

    # return as floats
    h_values = best_h.astype(float)
    return h_values, best_score, n
