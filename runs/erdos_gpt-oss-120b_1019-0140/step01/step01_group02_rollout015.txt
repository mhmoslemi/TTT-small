import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise candidate -------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = int(init.shape[0])
        h = np.clip(init, 0.0, 1.0)
    except NameError:
        n = 200
        h = np.zeros(n, dtype=float)
        half = n // 2
        h[:half] = 1.0
        if n % 2:
            h[half] = 0.5
        rng.shuffle(h)

    target_sum = n / 2.0
    h = _repair_sum(h, target_sum)

    current_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    # ----- optimisation parameters ---------------------------------------------
    max_step = 0.5
    min_step = 0.001
    max_temp = 0.01
    min_temp = 1e-6
    targeted_prob = 0.4
    no_improve_limit = 20000  # perturb after this many iterations without improvement
    iters_since_improve = 0

    # ----- main loop -----------------------------------------------------------
    while True:
        now = time.time()
        if now - start >= budget_s:
            break

        frac = (now - start) / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # --- targeted gradient step -------------------------------------------
        if rng.random() < targeted_prob:
            corr_full = np.correlate(h, 1.0 - h, mode="full")
            max_idx = int(np.argmax(corr_full))
            lag = max_idx - (n - 1)

            # gradient w.r.t. the worst lag
            w = np.zeros(n)
            if lag >= 0:
                lim = n - lag
                w[:lim] += h[lag:]
                w[lag:] += h[:lim]
            else:
                l = -lag
                lim = n - l
                w[l:] += h[:lim]
                w[:lim] += h[l:]

            inc_cands = np.where(h < 1.0 - 1e-12)[0]
            dec_cands = np.where(h > 1e-12)[0]

            if inc_cands.size and dec_cands.size:
                inc_idx = inc_cands[np.argmax(w[inc_cands])]
                dec_idx = dec_cands[np.argmin(w[dec_cands])]
                if inc_idx != dec_idx:
                    max_delta = min(step, 1.0 - h[inc_idx], h[dec_idx])
                    if max_delta > 1e-12:
                        delta = rng.random() * max_delta
                        h[inc_idx] += delta
                        h[dec_idx] -= delta
                        new_c5 = _compute_c5(h)
                        if new_c5 < current_c5 or rng.random() < math.exp(-(new_c5 - current_c5) / temp):
                            current_c5 = new_c5
                            if new_c5 < best_c5:
                                best_c5 = new_c5
                                best_h = h.copy()
                                iters_since_improve = 0
                        else:
                            h[inc_idx] -= delta
                            h[dec_idx] += delta
                        continue  # skip to next iteration

        # --- random pair move ---------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)

        low = max(delta_min, -step)
        high = min(delta_max, step)
        if low >= high:
            continue

        delta = rng.random() * (high - low) + low
        h[i] = hi + delta
        h[j] = hj - delta

        new_c5 = _compute_c5(h)
        if new_c5 < current_c5 or rng.random() < math.exp(-(new_c5 - current_c5) / temp):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                iters_since_improve = 0
        else:
            h[i] = hi
            h[j] = hj

        iters_since_improve += 1

        # --- perturb if stuck ---------------------------------------------------
        if iters_since_improve >= no_improve_limit:
            num_perturb = max(1, n // 20)  # ~5% of entries
            idxs = rng.choice(n, size=num_perturb, replace=False)
            h[idxs] = rng.random(num_perturb)
            h = _repair_sum(h, target_sum)
            current_c5 = _compute_c5(h)
            iters_since_improve = 0

    # ----- final repair and return ---------------------------------------------
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, n
