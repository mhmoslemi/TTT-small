def _project_to_feasible(v, target, tol=1e-12):
    lo = -np.max(v) - 1.0  # ensure v+lo <= -epsilon
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # final adjust for any residual sum due to rounding:
    diff = target - x.sum()
    if abs(diff) > tol:
        # adjust first entry not at bounds
        idx = np.where((x > 0) & (x < 1))[0]
        if idx.size == 0:
            # adjust any entry
            i = np.argmax(x) if diff < 0 else np.argmin(x)
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            i = idx[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    # final clip again just in case
    np.clip(x, 0.0, 1.0, out=x)
    return x

    start = time.time()
    rng = np.random.default_rng(seed)
    # Determine n
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        h = np.clip(init, 0.0, 1.0)
        h = _project_to_feasible(h, n/2.0)
    except NameError:
        n = 200
        target = n/2.0
        # binary random
        h = np.zeros(n, dtype=float)
        half = int(target)
        h[:half] = 1.0
        if n % 2:
            h[half] = 0.5
        rng.shuffle(h)
        h = _project_to_feasible(h, target)
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    target_sum = n / 2.0
    # main loop
    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        # compute gradients and update
        O = np.correlate(h, h, mode='full')
        w = target_sum - O
        # smoothing parameter
        frac = (now - start) / budget_s
        alpha = 0.1 + frac * (50.0 - 0.1)  # linear schedule
        # compute softmax weights
        w_max = np.max(w)
        exp_w = np.exp(alpha * (w - w_max))
        p = exp_w / np.sum(exp_w)
        # grad
        corr = np.correlate(p, h, mode='full')
        grad = -2.0 * corr[n-1 : n-1 + n]
        # learning rate schedule
        lr = 0.5 * (1 - frac) + 0.001 * frac
        h_new = h - lr * grad
        # project back onto feasible region
        h_new = _project_to_feasible(h_new, target_sum)
        # evaluate
        c5 = _compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        # occasional random perturbation
        if int(frac * 1000) % 50 == 0:  # roughly every 2% of time
            # Do a random swap move like in annealing
            i, j = rng.integers(0, n, size=2)
            if i != j:
                hi, hj = h[i], h[j]
                # feasible delta
                delta_min = max(-hi, -(1.0 - hj))
                delta_max = min(1.0 - hi, hj)
                if delta_min < delta_max:
                    delta = rng.random() * (delta_max - delta_min) + delta_min
                    h_tmp = h.copy()
                    h_tmp[i] = hi + delta
                    h_tmp[j] = hj - delta
                    # no projection needed due to delta bounds preserving constraints
                    c5_tmp = _compute_c5(h_tmp)
                    # accept if better
                    if c5_tmp < c5:
                        h = h_tmp
                        if c5_tmp < best_c5:
                            best_c5 = c5_tmp
                            best_h = h_tmp.copy()
    # final repair
    best_h = _project_to_feasible(best_h, target_sum)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, n

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the max overlap integral C5 for a discretised step function."""
    n = h.shape[0]
    target = n / 2.0
    # autocorrelation of h with itself
    O = np.correlate(h, h, mode="full")
    # C5 = (target - min autocorrelation) * dx ,  dx = 2 / n
    return float((target - O.min()) * (2.0 / n))

def _project_to_feasible(v: np.ndarray, target: float, tol: float = 1e-12) -> np.ndarray:
    """Project vector onto the intersect of box [0,1]^n and hyperplane sum(v)=target."""
    # Binary search on the shift lambda so that sum(clamp(v+lambda,0,1)) = target
    lo = -np.max(v) - 1.0    # ensures v+lo <= 0  -> clamp gives all zeros
    hi = 1.0 - np.min(v) + 1.0  # ensures v+hi >= 1  -> clamp gives all ones
    for _ in range(70):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix any tiny residual (due to floating point)
    diff = target - x.sum()
    if abs(diff) > tol:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size == 0:
            # adjust any non‑saturated entry
            cand = np.where(x < 1.0)[0] if diff > 0 else np.where(x > 0.0)[0]
            if cand.size:
                i = cand[0]
                x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            i = free[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        np.clip(x, 0.0, 1.0, out=x)
    return x

                    c5_tmp = (target_sum - O_tmp.min()) * (2.0 / n)
                    if c5_tmp < best_c5 - 1e-12:
                        best_c5 = c5_tmp
                        best_h = h_tmp.copy()
                        last_improve = now
                        h = h_tmp

        # restart if stuck for a while
        if now - last_improve > 12.0 and budget_s - elapsed > 5.0:
            # re‑initialise from a fresh random binary pattern
            h = np.zeros(n, dtype=float)
            half = int(target_sum)
            h[:half] = 1.0
            if n % 2:
                h[half] = 0.5
            rng.shuffle(h)
            h = _project_to_feasible(h, target_sum)
            last_improve = now

    # final projection and evaluation
    best_h = _project_to_feasible(best_h, target_sum)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, n
