def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if np.isclose(cur, target_sum):
        return h
    # if need to increase sum:
    if cur < target_sum:
        # capacity to increase:
        inc = 1.0 - h
        total_inc = inc.sum()
        if total_inc == 0:
            # cannot increase
            return h
        # scale inc to needed amount:
        scale = (target_sum - cur) / total_inc
        h += inc * scale
    else:
        # need to decrease
        dec = h
        total_dec = dec.sum()
        scale = (cur - target_sum) / total_dec
        h -= dec * scale
    # after scaling, might have numerical rounding errors; clip again and adjust a final small correction
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-8:
        # adjust smallest or largest element accordingly
        if diff > 0:
            # add to element with most capacity
            i = np.argmax(1.0 - h)
            h[i] = min(1.0, h[i] + diff)
        else:
            i = np.argmax(h)  # largest value
            h[i] = max(0.0, h[i] + diff)
    return h
```

import time
import math
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target):
    """Clip to [0,1] and adjust to hit the required sum exactly."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if np.isclose(cur, target, atol=1e-12):
        return h
    if cur < target:
        inc = 1.0 - h
        total_inc = inc.sum()
        if total_inc > 0:
            scale = (target - cur) / total_inc
            h += inc * scale
    else:
        dec = h
        total_dec = dec.sum()
        if total_dec > 0:
            scale = (cur - target) / total_dec
            h -= dec * scale
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-9:
        if diff > 0:
            i = np.argmax(1.0 - h)
            h[i] = min(1.0, h[i] + diff)
        else:
            i = np.argmax(h)
            h[i] = max(0.0, h[i] + diff)
    return h

def random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    return project_to_feasible(h, n / 2.0)

def upsample_h(h, n_target):
    """Linear interpolation of h to a different length, then project."""
    n_old = h.shape[0]
    if n_target == n_old:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_to_feasible(h_new, n_target / 2.0)

def local_optimize(h0, target, max_iter=300):
    """Run a few SLSQP steps starting from h0."""
    n = h0.shape[0]
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            compute_c5,
            h0,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
        )
        h_opt = project_to_feasible(np.clip(res.x, 0.0, 1.0), target)
        return h_opt
    except Exception:
        return h0

def optimize_at_n(h_start, n, time_budget, rng):
    """Combined local search + simulated annealing for a fixed n."""
    target = n / 2.0
    best = project_to_feasible(h_start, target)
    best_c5 = compute_c5(best)
    cur = best.copy()
    cur_c5 = best_c5

    # A few quick local refinements
    for _ in range(3):
        if time.time() >= end_time:
            break
        cur = local_optimize(cur, target, max_iter=300)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best, best_c5 = cur.copy(), cur_c5

    # Simulated annealing parameters
    temp = 0.02
    decay = 0.9995
    min_temp = 1e-6
    step_scale = 0.05
    start = time.time()
    deadline = start + time_budget

    while time.time() < deadline:
        # propose a new candidate
        new = cur + rng.normal(scale=step_scale, size=n)
        new = np.clip(new, 0.0, 1.0)
        new = project_to_feasible(new, target)
        new_c5 = compute_c5(new)
