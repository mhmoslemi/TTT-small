Implementation of correlation via FFT:

```
def max_corr(h):
    # compute linear convolution of h with (1-h)
    # Using FFT: pad to length 2*n - 1 for linear conv.
    n = h.size
    size = 1
    while size < 2*n:
        size <<= 1
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1 - h, size)
    conv = np.fft.irfft(H * G, size)[:2*n-1]
    dx = 2.0 / n
    return np.max(conv) * dx
```

```
def gradient_worst_shift(h):
    n = h.shape[0]
    dx = 2.0 / n
    size = 1 << (2*n-1).bit_length()
    H = np.fft.rfft(h, size)
```

```
def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    # adjust using water-filling
    # diff = target - cur (positive if need to add)
    diff = target - cur
    if diff > 0:
        # need to add mass to entries < 1
        free = np.where(v < 1.0)[0]
        if free.size == 0:
            # scaling
            v = v * (target / cur)
            return v
        # capacity for adding
        caps = 1.0 - v[free]
        # total capacity
        cap_sum = caps.sum()
        if cap_sum < diff - 1e-12:
            # fill all to 1, then scale
            v[free] = 1.0
            v = v * (target / v.sum())
            return v
        # distribute proportionally to caps
        add_frac = diff / cap_sum
        v[free] += caps * add_frac
    else:
        excess = -diff
        free = np.where(v > 0.0)[0]
        if free.size == 0:
            v = v * (target / cur)
            return v
        caps = v[free]
        cap_sum = caps.sum()
        if cap_sum < excess - 1e-12:
            v[free] = 0.0
            v = v * (target / v.sum())
            return v
        sub_frac = excess / cap_sum
        v[free] -= caps * sub_frac
    v = np.clip(v, 0.0, 1.0)
    # final scaling if needed
    cur = v.sum()
    if abs(cur - target) > 1e-8:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v
```

def _project_to_constraints(v, target):
    # clip
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    diff = target - cur
    if diff > 0:
        # need to add
        free = np.where(v < 1.0)[0]
        if free.size == 0:
            # all at 1, scale down
            v = v * (target / cur)
            return np.clip(v, 0.0, 1.0)
        caps = 1.0 - v[free]
        cap_sum = caps.sum()
        if cap_sum <= diff:
            # fill to ones
            v[free] = 1.0
            v = v * (target / v.sum())
            return np.clip(v, 0.0, 1.0)
        # allocate proportionally
        v[free] += caps * (diff / cap_sum)
    else:
        excess = -diff
        free = np.where(v > 0.0)[0]
        if free.size == 0:
            v = v * (target / cur)
            return np.clip(v, 0.0, 1.0)
        caps = v[free]
        cap_sum = caps.sum()
        if cap_sum <= excess:
            v[free] = 0.0
            v = v * (target / v.sum())
            return np.clip(v, 0.0, 1.0)
        v[free] -= caps * (excess / cap_sum)
    v = np.clip(v, 0.0, 1.0)
    # final scaling if residual diff
    cur = v.sum()
    if abs(cur - target) > 1e-9:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

    """Linear correlation of h with (1-h) using FFT (zero‑padded)."""
    n = h.shape[0]
    size = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, size)
    rev = (1.0 - h)[::-1]
    R = np.fft.rfft(rev, size)
    conv = np.fft.irfft(H * R, size)[: 2 * n - 1]
    return conv * (2.0 / n)

def _max_corr(h):
    """Maximum correlation value (the C5 bound for this h)."""
    return _corr_linear(h).max()

def _max_corr_and_shift(h):
    """Return full correlation, shift attaining the maximum, and its index."""
    corr = _corr_linear(h)
    idx = int(np.argmax(corr))
    shift = idx - (h.size - 1)
    return corr, shift, idx

def _gradient_worst_shift(h):
    """Gradient of the worst (max) correlation term w.r.t. h."""
    n = h.size
    _, shift, _ = _max_corr_and_shift(h)
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        g[k:] += 1 - h[: n - k]
        g[: n - k] -= h[k:]
    else:
        m = -shift
        g[: n - m] += 1 - h[m:]
        g[m:] -= h[: n - m]
    g *= 2.0 / n
    return g

def _optimize_h(h0, time_limit):
    """Stochastic gradient+annealing refinement of h within given time."""
    n = h0.size
    target = n / 2.0
    h = _project_to_constraints(h0.copy(), target)
    best_h = h.copy()
    best_score = _max_corr(h)
    cur_score = best_score
    start = time.time()
    it = 0
    step0 = 0.5
    no_improve = 0
    while time.time() - start < time_limit:
        it += 1
        T = 0.01 * np.exp(-it / 5000.0)
        if np.random.rand() < 0.3:
            # random mass transfer
            a = np.random.randint(0, n)
            b = np.random.randint(0, n)
            if a == b:
                continue
            delta_max = min(h[a], 1.0 - h[b])
            if delta_max <= 1e-12:
                continue
            delta = delta_max * np.random.rand() * 0.5
            h_new = h.copy()
            h_new[a] -= delta
            h_new[b] += delta
        else:
            # gradient descent on worst shift
            g = _gradient_worst_shift(h)
            step = step0 * np.exp(-it / 2000.0)
            h_new = h - step * g
            h_new = _project_to_constraints(h_new, target)
        new_score = _max_corr(h_new)

        # acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional jitter from best if stuck
        if no_improve >= 2000:
            h = _project_to_constraints(best_h + np.random.randn(n) * 0.02, target)
            cur_score = _max_corr(h)
            no_improve = 0

        if best_score < 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    ini = globals().get("initial_h_values")
    if ini is not None:
        h0 = np.array(ini, dtype=np.float64).ravel()
        n0 = h0.size
    else:
        n0 = 380
        h0 = np.random.rand(n0)
    target0 = n0 / 2.0
    h0 = _project_to_constraints(h0, target0)
    best_h = h0.copy()
    best_score = _max_corr(best_h)

    # resolution schedule
    scales = [n0]
    if budget_s >= 300:
        scales.append(n0 * 2)
    if budget_s >= 600:
        scales.append(n0 * 4)

    for i, n in enumerate(scales):
        if best_h.size != n:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            best_h = _project_to_constraints(h_interp, n / 2.0)

        elapsed = time.time() - start
        remain = budget_s - elapsed
        if remain < 5.0:
            break
        # keep a fraction for later scales
        if i < len(scales) - 1:
            t_opt = remain * 0.45
        else:
            t_opt = remain * 0.95
        h_opt, sc_opt = _optimize_h(best_h, t_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        if best_score < 0.3808:
            break

    # final reporting
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_h.size)
