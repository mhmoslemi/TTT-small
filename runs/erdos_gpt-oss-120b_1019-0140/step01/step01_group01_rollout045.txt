import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Rescale h in-place so that sum(h) == target while keeping 0<=h<=1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Use a simple balanced redistribution
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # distribute the remaining tiny error uniformly (clip afterwards)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # final correction to hit the target exactly (may violate bounds very slightly)
        excess = h.sum() - target
        if abs(excess) > 1e-12:
            # adjust by a small amount on a random non‑boundary index
            idxs = np.where((h > 0.0) & (h < 1.0))[0]
            if idxs.size:
                i = rng.choice(idxs)
                h[i] -= excess
                h[i] = np.clip(h[i], 0.0, 1.0)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize the step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.2  # safety margin

    # ---- initialise ----
    try:
        init_h = np.array(initial_h_values, dtype=float)  # global var if present
        n_points = init_h.size
        target_sum = n_points / 2.0
        h0 = init_h.copy()
        np.clip(h0, 0.0, 1.0, out=h0)
        h0 = adjust_sum_to_target(h0, target_sum, rng)
    except Exception:
        # fallback: random start
        n_points = 200
        target_sum = n_points / 2.0
        h0 = rng.rand(n_points)
        h0 = h0 / h0.sum() * target_sum
        h0 = adjust_sum_to_target(h0, target_sum, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    current_h = h0.copy()
    current_c5 = best_c5

    # annealing parameters
    T0, Tmin = 0.03, 1e-6          # temperature range
    step0, step_min = 0.25, 1e-5   # step magnitude range
    no_improve = 0
    restart_limit = 8000

    while True:
        if time.time() - start >= max_time:
            break

        # progress based schedules
        elapsed = time.time() - start
        progress = elapsed / max_time
        T = T0 * (1.0 - progress) + Tmin * progress
        step = step0 * (1.0 - progress) + step_min * progress

        # choose move type
        r = rng.rand()
        if r < 0.45:
            # ----- pairwise exchange (mass swap) -----
            i, j = rng.randint(n_points, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = np.clip(delta, low, high)
            cand = current_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
        elif r < 0.75:
            # ----- gradient move on the worst shift -----
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n_points - 1)

            g = np.zeros_like(current_h)
            idx = np.arange(n_points)

            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n_points)
            g[valid_plus] += (1.0 - current_h[plus[valid_plus]])

            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n_points)
            g[valid_minus] -= current_h[minus[valid_minus]]

            # project to hyperplane sum(g)=0
            g -= g.mean()

            cand = current_h - step * g
            np.clip(cand, 0.0, 1.0, out=cand)
            cand = adjust_sum_to_target(cand, target_sum, rng)
        else:
            # ----- multi‑point random redistribution -----
            m = rng.randint(3, 7)  # number of affected indices
            idxs = rng.choice(n_points, size=m, replace=False)
            perturb = rng.randn(m)
            perturb -= perturb.mean()               # zero‑sum
            # scale to the current step size
            if np.max(np.abs(perturb)) > 0:
                perturb = perturb / np.max(np.abs(perturb)) * step
            cand = current_h.copy()
            cand[idxs] += perturb
            np.clip(cand, 0.0, 1.0, out=cand)
            cand = adjust_sum_to_target(cand, target_sum, rng)

        cand_c5 = compute_c5(cand)

        # ----- acceptance test -----
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = cand
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- occasional restart from best with small jitter -----
        if no_improve >= restart_limit:
            jitter = rng.randn(n_points) * step * 0.5
            cand = best_h + jitter
            np.clip(cand, 0.0, 1.0, out=cand)
            cand = adjust_sum_to_target(cand, target_sum, rng)
            current_h = cand
            current_c5 = compute_c5(cand)
            no_improve = 0

        # early exit if we already beat the target bound
        if best_c5 <= 0.3808:
            break

    # final sanity adjustment
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
