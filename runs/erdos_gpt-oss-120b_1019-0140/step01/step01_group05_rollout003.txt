def sa_optimize(h_start, time_limit, rng, T0=0.1, Tf=1e-6, eps0=0.4, epsf=1e-4):
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    cur_score = evaluate_c5(h_cur)
    best_score = cur_score
    N = len(h_cur)
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # temperature schedule, linear or exponential
        t_frac = 1 - elapsed / time_limit
        T = Tf + (T0 - Tf) * t_frac  # linear decreasing
        eps = epsf + (eps0 - epsf) * t_frac # linear decreasing max move
        
        # pick two distinct indices i,j
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi = h_cur[i]
        hj = h_cur[j]
        # allowable move to keep within [0,1]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0 and down == 0:
            continue
        # sample epsilon uniformly in [-down, up], but limit magnitude by eps
        eps_move = rng.random() * (up + down) - down
        if abs(eps_move) > eps:
            eps_move = math.copysign(eps, eps_move)
        if eps_move == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps_move
        h_new[j] = hj - eps_move
        # evaluate
        new_score = evaluate_c5(h_new)
        delta = new_score - cur_score
        # accept if improvement or with probability exp(-delta/T)
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score
```

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0)
    n = v.size
    target = n / 2.0
    s = v.sum()
    if s == 0.0:
        # fill equally
        v[:] = target / n
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    # after maybe slight error, we can do final small adjustment:
    # compute deficit = target - sum(v)
    # adjust any element not at bound by adding deficit (or subtract)
    # but we can ignore minor error.
    return v

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    n = v.size
    target = n / 2.0
    np.clip(v, 0.0, 1.0, out=v)
    s = v.sum()
    if s == 0.0:
        v[:] = target / n
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(
    h_start: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
    T0: float = 0.5,
    Tf: float = 1e-6,
    eps0: float = 0.5,
    epsf: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """SA preserving sum exactly via two‑point exchange moves."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = 1.0 - elapsed / time_limit
        T = Tf + (T0 - Tf) * frac
        eps_max = epsf + (eps0 - epsf) * frac

        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi, hj = h_cur[i], h_cur[j]

        up = min(1.0 - hi, hj)      # max we can add to i (decrease j)
        down = min(hi, 1.0 - hj)    # max we can subtract from i (increase j)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down  # uniform in [-down, up]
        if abs(eps) > eps_max:
            eps = math.copysign(eps_max, eps)
        if eps == 0.0:
            continue

        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # Bounds already respected by construction
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score
