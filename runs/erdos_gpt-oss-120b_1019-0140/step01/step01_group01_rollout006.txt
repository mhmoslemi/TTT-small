def compute_c5(h):
    n = h.shape[0]
    m = 1 << (2 * n - 1).bit_length()   # Next power of 2 >= 2n-1

import time, numpy as np

def compute_c5_corr(h):
    n = h.shape[0]
    nfft = 1
    while nfft < 2 * n:
        nfft <<= 1
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)
    corr = corr[:2*n-1]
    return corr

def compute_c5(h):
    corr = compute_c5_corr(h)
    dx = 2.0 / h.shape[0]
    return corr.max() * dx

def adjust_sum_to_target(h, target, rng):
    # as before
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # iterate adjusting random entries
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        idx = rng.randint(h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0:
                h[idx] -= dec
                diff += dec
    if abs(diff) > 1e-12:
        # distribute residual uniformly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # re-adjust if still off
        # small residual can be left
    return h
```

import time
import numpy as np

def _compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    nfft = 1
    while nfft < 2 * n:
        nfft <<= 1
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)
    return corr[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by discrete correlation."""
    dx = 2.0 / h.shape[0]
    return float(_compute_corr_fft(h).max() * dx)


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Adjust h (in‑place) so that its sum equals target while keeping 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # greedy fix using random indices
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
    if abs(diff) > 1e-12:
        # tiny uniform correction, then clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h
