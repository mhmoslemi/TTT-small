    grad = np.empty(n, dtype=float)
    # compute contributions
    # for i: (1 - h[i+shift]) - h[i-shift]

    # Use np.take with mode 'clip'? but need out-of-range treat as 0 for h[i+shift] when out of range.
    # For (1 - h[i+shift]): if i+shift out of bounds, term = 1
    # For -h[i-shift]: if i-shift out of bounds, term = 0.
    # So:
    # define array h_padded of length n+2*abs(shift) perhaps.
    # Simpler: compute arrays with shift using np.where.
    idx_plus = np.arange(n) + shift
    idx_minus = np.arange(n) - shift

    term1 = np.where((idx_plus >= 0) & (idx_plus < n), 1.0 - h[idx_plus], 1.0)

    grad = term1 - term2

def _repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # indices eligible to adjust
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute on all entries
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    # distribute diff equally among eligible entries
    # amount = diff / idx.size
    # but ensure we don't exceed bounds
    while True:
        idx = np.where((diff > 0) & (h < 1.0 - 1e-12) | (diff < 0) & (h > 1e-12))[0]
        if idx.size == 0:
            break
        amt = diff / idx.size
        # clip per element
        add = np.clip(h[idx] + amt, 0.0, 1.0) - h[idx]
        h[idx] += add
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h
```

    cur = float(h.sum())
    diff = target - cur

import time
import math
import numpy as np

def _full_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (full mode)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    a = np.fft.rfft(h, n=L)
    b = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(a * b, n=L)
    return conv[: 2 * n - 1]

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.shape[0]
    corr = _full_corr_fft(h)
    return float(np.max(corr) * (2.0 / n))

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust entries to satisfy sum(h) == target while staying in [0,1]."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # choose a direction
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _gradient_from_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Subgradient of the current maximal correlation term w.r.t. h."""
    n = h.shape[0]
    idx = np.arange(n)
    plus = idx + shift
    minus = idx - shift
    term1 = np.where((plus >= 0) & (plus < n), 1.0 - h[plus], 1.0)
    term2 = np.where((minus >= 0) & (minus < n), h[minus], 0.0)
    return term1 - term2

def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (sum = n/2, entries in [0,1])."""
    target = n / 2.0
    h = rng.random(n)
    np.clip(h, 0.0, 1.0, out=h)
    return _repair_sum(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h minimizing the Erdős C₅ overlap constant."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -----
    try:
        init = np.asarray(initial_h_values, dtype=float)  # provided by harness
        n_init = init.shape[0]
        # Upsample to a moderate size for richer search space
        n = max(200, n_init)
        old_x = np.linspace(0.0, 2.0, n_init, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n, endpoint=False)
        init_large = np.interp(new_x, old_x, init)
        target_sum = n / 2.0
        np.clip(init_large, 0.0, 1.0, out=init_large)
        init_large = _repair_sum(init_large, target_sum)
        current = init_large.copy()
    except NameError:
        n = 200
        target_sum = n / 2.0
        current = _random_feasible(n, rng)

    current = _repair_sum(current, target_sum)

    # initial evaluation
    corr = _full_corr_fft(current)
    current_c5 = float(np.max(corr) * (2.0 / n))
    best_h = current.copy()
    best_c5 = current_c5

    # schedule parameters
    max_step = 0.5
    min_step = 0.001
    max_temp = 0.01
    min_temp = 1e-6

    no_improve = 0  # counter for stagnation

    # ----- main optimisation loop -----
    while True:
        now = time.time()
        if now - start >= budget_s:
            break

        # compute shift that gives the current maximal overlap
        corr = _full_corr_fft(current)
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)

        # gradient tells where decreasing / increasing h helps
        grad = _gradient_from_shift(current, shift)
        pos_idx = np.where(grad > 1e-9)[0]
        neg_idx = np.where(grad < -1e-9)[0]

        # choose indices i (to decrease) and j (to increase)
        if pos_idx.size and neg_idx.size:
            i = int(rng.choice(pos_idx))
            j = int(rng.choice(neg_idx))
        else:
            # fallback random feasible pair
            feasible_i = np.where(current > 1e-12)[0]
            feasible_j = np.where(current < 1.0 - 1e-12)[0]
            if feasible_i.size == 0 or feasible_j.size == 0:
                break
            i = int(rng.choice(feasible_i))
            j = int(rng.choice(feasible_j))

        old_i = float(current[i])
        old_j = float(current[j])
        max_delta = min(old_i, 1.0 - old_j)
        if max_delta <= 0:
            continue

        # time‑dependent step and temperature
        elapsed = now - start
        frac = elapsed / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac
        delta = rng.random() * min(max_delta, step)
        if delta <= 0:
            continue

        # apply move (decrease i, increase j)
        current[i] = old_i - delta
        current[j] = old_j + delta

        # evaluate new objective
        new_corr = _full_corr_fft(current)
        new_c5 = float(np.max(new_corr) * (2.0 / n))
        delta_obj = new_c5 - current_c5

        # Metropolis acceptance
        if delta_obj < 0 or rng.random() < math.exp(-delta_obj / temp):
            current_c5 = new_c5
            # update best if improved
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = current.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert move
            current[i] = old_i
            current[j] = old_j
            no_improve += 1

        # occasional restart on stagnation
        if no_improve >= 5000:
            # perturb the best solution slightly and continue
            current = best_h.copy()
            noise = rng.normal(scale=0.02, size=n)
            current += noise
            np.clip(current, 0.0, 1.0, out=current)
            current = _repair_sum(current, target_sum)
            corr = _full_corr_fft(current)
            current_c5 = float(np.max(corr) * (2.0 / n))
            no_improve = 0

    # final repair and evaluation
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
