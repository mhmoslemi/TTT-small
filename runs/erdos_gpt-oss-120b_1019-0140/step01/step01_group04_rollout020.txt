def _repair_vector(v, n, max_iter=15):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # Determine free indices that can be adjusted without hitting bounds
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if still off
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

import time

# --------------------------------------------------------------
# Helper utilities

def _repair_vector(v, n, max_iter=20):
    """Project v onto [0,1]^n while enforcing sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx  (discrete version)."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _random_feasible(n):
    """Random feasible vector (sum = n/2, 0≤v≤1)."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

# Local refinement (exchange moves)
# --------------------------------------------------------------
def _local_search(v, n, time_limit):
    """Simple stochastic hill climbing using pairwise exchanges."""
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        # maximal admissible delta respecting box bounds
        max_inc = min(step, 1.0 - best_v[i])
        max_dec = min(step, best_v[j])
        if max_inc <= 0 or max_dec <= 0:
            continue
        delta = np.random.rand() * min(max_inc, max_dec)
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score, best_v = score, cand
            step = min(step * 1.05, 0.12)
        else:
            step = max(step * 0.995, 0.008)
    return best_v, best_score

# Gradient‑based refinement focused on the worst shift

def _gradient_refine(v, n, time_limit):
    """Increase the minimal self‑correlation by moving mass along the worst shift."""
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    while time.time() - start < time_limit:
        # self‑correlation (no scaling needed for argmin)
        corr = np.correlate(best_v, best_v, mode="full")
        min_idx = np.argmin(corr)
        shift = min_idx - (n - 1)   # signed shift
        # gradient of the correlation at this shift
        g = np.zeros_like(best_v)
        if shift >= 0:
            g[: n - shift] += best_v[shift:]
            g[shift:] += best_v[: n - shift]
        else:
            s = -shift
            g[: n - s] += best_v[s:]
            g[s:] += best_v[: n - s]
        # choose where to add / subtract mass
        inc_score = g * (1.0 - best_v)   # benefit of increasing this component
        dec_score = g * best_v           # benefit of decreasing this component
        i = np.argmax(inc_score)
        j = np.argmin(dec_score)
        max_inc = 1.0 - best_v[i]
        max_dec = best_v[j]
        if max_inc <= 0 or max_dec <= 0:

        sc = _compute_c5(cand)
        if sc < best_score:

        # else keep trying with new random picks in next loop
    return best_v, best_score

def _make_logistic_basis(N, L):
    """Cosine basis for N points, L frequencies."""
    i = np.arange(N)
    basis = np.empty((N, L), dtype=np.float64)
    for l in range(1, L + 1):
        basis[:, l - 1] = np.cos(2.0 * np.pi * l * i / N)
    return basis

def _v_from_params(params, basis, N, target):
    """Map low‑dimensional params to a feasible h‑vector via a logistic transform."""
    raw = basis @ params
    bias = 0.0
    # Newton iteration to hit the required sum
    for _ in range(10):
        u = raw + bias
        # stable logistic
        exp_u = np.exp(-np.clip(u, -700, 700))
        v = 1.0 / (1.0 + exp_u)
        s = v.sum()
        if abs(s - target) < 1e-12:
            break
        deriv = (v * (1.0 - v)).sum()  # d sum / d bias
        if deriv == 0:
            break
        bias += (target - s) / deriv
    # final exact repair (tiny numerical drift)
    v = _repair_vector(v, N)
    return v

    start_time = time.time()

    best_v = _repair_vector(best_v, N)
