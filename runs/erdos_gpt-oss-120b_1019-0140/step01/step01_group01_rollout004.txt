import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximation for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Shift h so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute uniformly, clip, repeat a few times
    for _ in range(10):
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    # if still off, fix by greedy single-index adjustments
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 500:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    return h

def gradient_refine(h: np.ndarray, target_sum: float, rng: np.random.RandomState,
                    time_limit: float) -> tuple:
    """Projected gradient descent on the worst lag with simulated‐annealing acceptance."""
    n = h.shape[0]
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    cur_c5 = best_c5

    T0, Tmin = 1e-3, 1e-6
    step0, step_end = 0.2, 1e-4
    max_iters = 20000

    for it in range(max_iters):
        if time.time() >= time_limit:
            break
        progress = it / max_iters
        T = T0 * (1.0 - progress) + Tmin * progress
        step = step0 * (1.0 - progress) + step_end * progress

        # gradient of the lag giving the current maximal correlation
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        lag = np.argmax(corr) - (n - 1)

        g = np.ones_like(cur)
        idx = np.arange(n)
        pos = idx + lag
        mask = (pos >= 0) & (pos < n)
        g[idx[mask]] -= cur[pos[mask]]
        neg = idx - lag
        mask2 = (neg >= 0) & (neg < n)
        g[idx[mask2]] -= cur[neg[mask2]]
        g -= g.mean()                     # keep sum unchanged

        cand = cur - step * g
        np.clip(cand, 0.0, 1.0, out=cand)
        cand = adjust_sum_to_target(cand, target_sum, rng)

        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5
        if delta < 0.0 or rng.rand() < np.exp(-delta / T):
            cur = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best = cand.copy()
    return best, best_c5
