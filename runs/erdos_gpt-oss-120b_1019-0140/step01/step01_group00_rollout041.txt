class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx  (i.e., t - overlap*dx >= 0)."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

    """Return max overlap integral C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(vec, target_sum):
    """Adjust vec to satisfy sum(vec) == target_sum while staying in [0,1]."""
    vec = np.clip(vec, 0.0, 1.0)
    cur_sum = np.sum(vec)
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return vec
    # add uniform correction
    vec += diff / len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    n = len(vec)
    for i in range(n):
        if abs(diff) < 1e-12:
            break
        if diff > 0 and vec[i] < 1.0:
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
        elif diff < 0 and vec[i] > 0.0:
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
    return vec


def generate_random_feasible(n, target_sum, rng):
    """Create a random feasible continuous vector (values in [0,1] and exact sum)."""
    x = rng.random(n)
    s = np.sum(x)
    if s == 0:
        x = np.full(n, target_sum / n)
    else:
        x = x * (target_sum / s)
    x = np.clip(x, 0.0, 1.0)
    # fix residual sum difference
    diff = target_sum - np.sum(x)
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - x[i])
            x[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    elif diff < 0:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            dec = min(-diff, x[i])
            x[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return x


def refine_continuous(h, dx, target_sum, time_limit, start_time, rng):
    """Local coordinate descent transferring mass between entries."""
    cur_h = h.copy()
    cur_val = compute_c5_vec(cur_h, dx)
    step = 0.05
    n = len(h)
    while time.time() < time_limit:
        donors = np.where(cur_h > 1e-12)[0]
        recipients = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recipients.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(recipients)
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            step *= 0.5
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_vec(cand, dx)
        if cand_val < cur_val - 1e-12:
            cur_h = cand
            cur_val = cand_val
            step = min(step * 1.1, 0.5)
            if cur_val < 0.3808:
                break
        else:
            step *= 0.99
    cur_h = project_to_sum(cur_h, target_sum)
    cur_val = compute_c5_vec(cur_h, dx)
    return cur_h, cur_val


def binary_hill_climb(h_bin, cur_val, dx, budget_s, start_time, rng):
    """Simple swap hill‑climbing on a binary vector."""
    best_h = h_bin.copy()
    best_val = cur_val
    n = len(h_bin)
    while time.time() - start_time < budget_s - 2:
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_vec(cand.astype(float), dx)
        if cand_val < best_val - 1e-12:
            best_h = cand
            best_val = cand_val
            if best_val < 0.3808:
                break
    return best_h.astype(float), best_val


# ---------- Constraint classes ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        return np.sum(x[: self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx  (i.e. t - overlap*dx >= 0)."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overl * self.dx


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- determine problem size ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # ---- initialise best solution ----
    if h0 is not None:
        best_h = project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        best_h = generate_random_feasible(n, target_sum, rng)
    best_val = compute_c5_vec(best_h, dx)

    # ---- set up SLSQP constraints ----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{"type": "eq", "fun": sum_con}]
    constraints += [{"type": "ineq", "fun": oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # ---- main optimisation loop (continuous SLSQP + refinement) ----
    while time.time() - start_time < budget_s * 0.75:
        h_start = generate_random_feasible(n, target_sum, rng)
        t_start = compute_c5_vec(h_start, dx)
        x0 = np.concatenate([h_start, [t_start]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 800, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_sum(h_opt, target_sum)

        # local refinement
        time_limit = start_time + budget_s - 5.0
        h_ref, val_ref = refine_continuous(
            h_opt, dx, target_sum, time_limit, start_time, rng
        )
        if val_ref < best_val - 1e-12:
            best_h, best_val = h_ref, val_ref
            if best_val < 0.3808:
                break

    # ---- optional binary hill‑climbing phase ----
    time_left = budget_s - (time.time() - start_time)
    if time_left > 10.0:
        # round to binary by taking largest entries
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argsort(-best_h)[: int(target_sum)]
        binary_h[top_idx] = 1
        bin_val = compute_c5_vec(binary_h.astype(float), dx)
        binary_h, bin_val = binary_hill_climb(
            binary_h, bin_val, dx, budget_s, start_time, rng
        )
        if bin_val < best_val - 1e-12:
            best_h, best_val = binary_h, bin_val

    # ---- final safety enforcement ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_val = compute_c5_vec(best_h, dx)

    return best_h, best_val, n
