    v2[b] += delta

def _repair_vector(v, n):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    # iterative distribution
    for _ in range(10):
        s = v.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        # Identify free indices
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        # scale
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        # final fix
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v
```

def _repair_vector(v, n):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(15):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return float(np.max(corr) * dx)

def _upsample_vector(v, n_target):

    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])

def _anneal_optimize(v0, time_limit):
    start = time.time()
    n = len(v0)
    best_v = v0.copy()
    best_score = _compute_c5(best_v)
    cur_v = best_v.copy()
    cur_score = best_score
    T0 = 0.008
    reset_interval = 30.0
    last_reset = start
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = max(T0 * (1.0 - frac), 1e-8)
        # decide move
        if np.random.rand() < 0.65:
            # need correlation to identify worst shift
            corr = np.correlate(cur_v, 1.0 - cur_v, mode='full')
            j_star = int(np.argmax(corr))
            cand = _targeted_move(cur_v, j_star)
        else:
            cand = _random_move(cur_v)
        if cand is None:
            continue
        cand_score = _compute_c5(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur_v, cur_score = cand, cand_score
            if cur_score < best_score:

            cur_v = best_v + jitter
            cur_v = _repair_vector(cur_v, n)
            cur_score = _compute_c5(cur_v)
            last_reset = time.time()
    return best_v, best_score

            return None

def _random_move(v):
    n = len(v)
    a, b = np.random.choice(n, 2, replace=False)
    if np.random.rand() < 0.5:
        max_delta = min(v[a], 1.0 - v[b])
        if max_delta <= 0:
            return None
        delta = np.random.rand() * max_delta
        v2 = v.copy()
        v2[a] -= delta
        v2[b] += delta
        return v2
    else:
        max_delta = min(v[b], 1.0 - v[a])
        if max_delta <= 0:
            return None
        delta = np.random.rand() * max_delta
        v2 = v.copy()
        v2[b] -= delta
        v2[a] += delta
        return v2

def _anneal_optimize(v0, time_limit):
    start = time.time()
    n = len(v0)
    best_v = v0.copy()
    best_score = _compute_c5(best_v)
    cur_v = best_v.copy()
    cur_score = best_score
    T0 = 0.008
    reset_interval = 30.0
    last_reset = start
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = max(T0 * (1.0 - frac), 1e-8)
        if np.random.rand() < 0.65:
            corr = np.correlate(cur_v, 1.0 - cur_v, mode='full')
            j_star = int(np.argmax(corr))
            cand = _targeted_move(cur_v, j_star)
        else:
            cand = _random_move(cur_v)
        if cand is None:
            continue
        cand_score = _compute_c5(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur_v, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score, best_v = cur_score, cur_v.copy()
        if np.random.rand() < 0.001:
            cur_v = _repair_vector(cur_v, n)
            cur_score = _compute_c5(cur_v)
        if time.time() - last_reset > reset_interval:
            jitter = np.random.normal(scale=0.02, size=n)
            cur_v = best_v + jitter
            cur_v = _repair_vector(cur_v, n)
            cur_score = _compute_c5(cur_v)
            last_reset = time.time()
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n_parent = len(parent)
    except Exception:
        parent = None
        n_parent = None
    max_n = 800
    target_n = 2 ** int(np.floor(np.log2(max_n)))
    candidates = []
    if parent is not None:
        if n_parent != target_n:
            parent_up = _upsample_vector(parent, target_n)
        else:
            parent_up = _repair_vector(parent, target_n)
        candidates.append(parent_up)
    candidates.append(_thue_morse_vector(target_n))
    for _ in range(3):
        candidates.append(_random_feasible(target_n))
    best_v = None
    best_score = None
    for v in candidates:
        sc = _compute_c5(v)
        if best_score is None or sc < best_score:
            best_score, best_v = sc, v.copy()
    elapsed = time.time() - start
    time_left = budget_s - elapsed - 2.0
    if time_left > 0.5 and best_v is not None:
        best_v, best_score = _anneal_optimize(best_v, time_left)
    best_v = _repair_vector(best_v, target_n)
    best_score = _compute_c5(best_v)
    return best_v.astype(np.float64), float(best_score), int(target_n)
