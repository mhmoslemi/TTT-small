def adjust_sum_to_target(h, target, rng):
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Try to distribute diff proportionally to free capacity
    # max increase per element: 1 - h[i]; max decrease: h[i]
    # We'll use simple loop limited.
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # Choose a random index that can accommodate diff sign
        if diff > 0:
            # need increase
            inc_cap = 1.0 - h
            # select those with capacity > 0
            idxs = np.where(inc_cap > 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            inc = min(diff, inc_cap[i])
            h[i] += inc
            diff -= inc
        else:
            dec_cap = h
            idxs = np.where(dec_cap > 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            dec = min(-diff, dec_cap[i])
            h[i] -= dec
            diff += dec
    # Minor residual
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h
```

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Adjust entries of h (in-place) so that sum(h) == target while staying in [0,1]."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Distribute the deficit/surplus iteratively
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            inc_cap = 1.0 - h
            idxs = np.where(inc_cap > 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            inc = min(diff, inc_cap[i])
            h[i] += inc
            diff -= inc
        else:
            dec_cap = h
            idxs = np.where(dec_cap > 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            dec = min(-diff, dec_cap[i])
            h[i] -= dec
            diff += dec
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def greedy_binary(n: int, target: int, rng: np.random.RandomState, time_limit: float):
    """Construct a binary vector by greedy addition of 1s, trying to keep C5 low."""
    start = time.time()
    best_h = None
    best_c5 = float('inf')
    attempts = 0
    while attempts < 10 and time.time() - start < time_limit:
        h = np.zeros(n, dtype=np.int8)
        for _ in range(target):
            zeros = np.where(h == 0)[0]
            best_local = float('inf')
            candidates = []
            for idx in zeros:
                h[idx] = 1
                c5 = compute_c5(h.astype(float))
                h[idx] = 0
                if c5 < best_local - 1e-12:
                    best_local = c5
                    candidates = [idx]
                elif abs(c5 - best_local) < 1e-12:
                    candidates.append(idx)
            # break ties randomly
            chosen = rng.choice(candidates)
            h[chosen] = 1
        final_c5 = compute_c5(h.astype(float))
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h.copy()
        attempts += 1
    if best_h is None:
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, target, replace=False)] = 1
        best_h = h
        best_c5 = compute_c5(best_h.astype(float))
    return best_h, best_c5

def binary_sa(initial_h: np.ndarray, target: int, rng: np.random.RandomState, time_limit: float):
    """Simulated annealing on binary vectors (swap a 1 with a 0)."""
    h = initial_h.copy()
    cur_c5 = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5
    n = h.size
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = 1.0 * (1.0 - progress) + 1e-4 * progress
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand.astype(float))
        if cand_c5 < cur_c5 or rng.rand() < np.exp(-(cand_c5 - cur_c5) / T):
            h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
    return best_h, best_c5

def continuous_sa(initial_h: np.ndarray, target: float, rng: np.random.RandomState, time_limit: float):
    """Simulated annealing on continuous values while preserving mass."""
    h = initial_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n = h.size
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = 0.5 * (1.0 - progress) + 1e-6 * progress
        step = 0.2 * (1.0 - progress) + 1e-4 * progress
        i, j = rng.randint(0, n, size=2)
        while i == j:
            j = rng.randint(0, n)
        delta = (rng.rand() * 2.0 - 1.0) * step
        cand = h.copy()
        cand[i] += delta
        cand[j] -= delta
        np.clip(cand, 0.0, 1.0, out=cand)
        cand = adjust_sum_to_target(cand, target, rng)
        cand_c5 = compute_c5(cand)
        if cand_c5 < cur_c5 or rng.rand() < np.exp(-(cand_c5 - cur_c5) / T):
            h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Optimize a step function h on [0,2] to obtain a low C5 bound."""
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # Determine problem size
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n_points = init_h.size
    except Exception:
        init_h = None
        n_points = 70

    target_sum = int(round(n_points / 2))

    # Initialise binary vector
    if init_h is not None:
        # Choose the largest entries to become 1
        order = np.argsort(-init_h)
        h_bin = np.zeros(n_points, dtype=np.int8)
        h_bin[order[:target_sum]] = 1
    else:
        h_bin = np.zeros(n_points, dtype=np.int8)
        h_bin[rng.choice(n_points, target_sum, replace=False)] = 1

    # Phase 1: greedy construction (optional, time‑controlled)
    time_left = max_time - (time.time() - start)
    if time_left > 10.0:
        greedy_time = min(0.15 * max_time, time_left * 0.5)
        h_bin, _ = greedy_binary(n_points, target_sum, rng, greedy_time)

    # Phase 2: binary simulated annealing
    time_left = max_time - (time.time() - start)
    if time_left > 5.0:
        binary_time = 0.6 * time_left
        h_bin, _ = binary_sa(h_bin, target_sum, rng, binary_time)

    # Phase 3: continuous refinement
    time_left = max_time - (time.time() - start)
    if time_left > 2.0:
        h_cont = h_bin.astype(float)
        h_cont, _ = continuous_sa(h_cont, target_sum, rng, time_left)
    else:
        h_cont = h_bin.astype(float)

    # Final adjustment to exact mass and compute final bound
    h_cont = adjust_sum_to_target(h_cont, target_sum, rng)
    final_c5 = compute_c5(h_cont)

    return (h_cont, float(final_c5), n_points)
