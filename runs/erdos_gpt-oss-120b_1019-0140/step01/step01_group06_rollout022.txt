def adjust_to_target_binary(h_frac, target):
    n = len(h_frac)
    h = np.rint(h_frac).astype(int)
    cur = h.sum()
    diff = int(target - cur)
    if diff > 0:
        # need to set diff zeros to 1
        # choose zeros with largest fractional part (since rounding down)
        zero_inds = np.where(h == 0)[0]
        # sort by descending h_frac[zero_inds]
        order = zero_inds[np.argsort(-h_frac[zero_inds])]
        h[order[:diff]] = 1
    elif diff < 0:
        # need to set -diff ones to 0
        one_inds = np.where(h == 1)[0]
        order = one_inds[np.argsort(h_frac[one_inds])]  # smallest fractional part
        h[order[: -diff]] = 0
    return h.astype(float)

def project_to_feasible(v, target_sum, tol=1e-8, max_iter=10):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(v)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final small adjustment
    cur = np.sum(v)
    if abs(cur - target_sum) > 1e-6:
        # adjust a random index with capacity
        i = np.argmin(v) if cur < target_sum else np.argmax(v)
        v[i] = np.clip(v[i] + (target_sum - cur), 0.0, 1.0)
    return v
```

def adjust_to_target_binary(frac, target):
    # frac: 1D array of floats (nonnegative)
    n = len(frac)
    h = np.rint(frac).astype(int)
    cur = h.sum()
    diff = int(target - cur)
    if diff > 0:
        zero_inds = np.where(h == 0)[0]
        # order by descending frac (closer to 1)
        order = zero_inds[np.argsort(-frac[zero_inds])]
        h[order[:diff]] = 1
    elif diff < 0:
        one_inds = np.where(h == 1)[0]
        order = one_inds[np.argsort(frac[one_inds])]  # smallest fractional part
        h[order[: -diff]] = 0
    return h.astype(float)
```

import time
import numpy as np
import cvxpy as cp
import scipy.optimize as opt


def compute_c5(h):
    """Overlap bound C5 for a sampled step function."""
    n = int(h.shape[0])
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(v, target_sum, tol=1e-8, max_iter=10):
    """Project onto [0,1]^n ∩ {sum = target_sum}."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(v)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target_sum) > 1e-6:
        # adjust a single entry with remaining capacity
        if cur < target_sum:
            i = np.argmin(v)
        else:
            i = np.argmax(v)
        v[i] = np.clip(v[i] + (target_sum - cur), 0.0, 1.0)
    return v


def adjust_to_target_binary(frac, target):
    """Round a fractional vector to {0,1} while fixing the sum to target."""
    h = np.rint(frac).astype(int)
    cur = int(h.sum())
    diff = int(target - cur)
    if diff > 0:
        zeros = np.where(h == 0)[0]
        order = zeros[np.argsort(-frac[zeros])]
        h[order[:diff]] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        order = ones[np.argsort(frac[ones])]
        h[order[: -diff]] = 0
    return h.astype(float)


def hill_climb_binary(h, time_limit, rng):
    """Swap 0/1 entries preserving the total sum to lower C5."""
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best_h = cand
            best_val = val
    return best_h, best_val


def local_optimize_slsqp(h0, n, max_iter=500):
    """Refine a feasible h using SLSQP."""
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bnds = [(0.0, 1.0)] * n
    res = opt.minimize(
        fun=lambda x: compute_c5(x),
        x0=h0,
        method="SLSQP",
        bounds=bnds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_to_feasible(x, n / 2.0)
    return x
