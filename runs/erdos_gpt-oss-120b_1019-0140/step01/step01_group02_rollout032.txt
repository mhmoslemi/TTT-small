def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    L = 2 * n
    # Use real FFT for speed
    # compute linear correlation via FFT
    # for correlation: conv = ifft(fft(h, L) * conj(fft(1-h, L))) real
    # This gives circular correlation; but with zero padding to length L, it's linear full.
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    # Keep only first 2n-1 entries (since last result includes last zero padding)
    corr = corr[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * dx

def repair_sum(h, target):

    if abs(diff) < 1e-12:
        return h
    # Add diff to first element that can accommodate it.
    # If diff > 0, increase an element < 1; else decrease > 0.
    if diff > 0:
        idx = np.where(h < 1 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # redistribute diff evenly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = h[i] + diff

    h[i] = new_val
    # If still diff left due to clipping, recursively call
    if abs(h.sum() - target) > 1e-12:

    return h

def _compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    L = 2 * n
    Ha = np.fft.rfft(h, n=L)
    Hb = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(Ha * np.conj(Hb), n=L)
    # Keep only first 2n-1 entries where correlation potentially non-zero
    max_corr = np.max(corr[:2*n-1])
    return max_corr * dx

def _repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # choose index that can absorb diff
    if diff > 0:
        idx = np.where(h < 1 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback evenly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = h[i] + diff
    new_val = max(0.0, min(1.0, new_val))
    h[i] = new_val
    # if still not exact, recurse (should converge)
    if abs(h.sum() - target) > 1e-12:
        return _repair_sum(h, target)
    return h

def _compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    L = 2 * n
    Ha = np.fft.rfft(h, n=L)
    Hb = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(Ha * np.conj(Hb), n=L)
    max_corr = np.max(corr[:2*n-1])
    return max_corr * dx

def _repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = h[i] + diff
    new_val = max(0.0, min(1.0, new_val))
    h[i] = new_val
    # if not fixed fully, recurse
    if abs(h.sum() - target) > 1e-12:
        return _repair_sum(h, target)
    return h

import time, math
import numpy as np

def _compute_c5_fft(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretised step function using FFT."""
    n = h.shape[0]
    dx = 2.0 / n
    L = 2 * n                     # zero‑pad length for linear correlation
    Ha = np.fft.rfft(h, n=L)
    Hb = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(Ha * np.conj(Hb), n=L)
    max_corr = np.max(corr[:2 * n - 1])
    return float(max_corr * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that sum(h) equals target while respecting bounds."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = h[i] + diff
    new_val = max(0.0, min(1.0, new_val))
    h[i] = new_val
    if abs(h.sum() - target) > 1e-12:
        return _repair_sum(h, target)
    return h
