import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral approximated by discrete full correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Project vector v onto [0,1] box and enforce sum(v)=target (water-filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # fill the largest capacities first
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _optimize_local_sa(h_start, time_budget):
    """Simulated‑annealing style local search on a fixed resolution."""
    n = h_start.size
    target = n / 2.0
    cur_h = h_start.copy()
    cur_score = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    temp_start = 0.5
    temp_end = 1e-4
    sigma_max = 0.2

        elapsed = time.time() - t0
        frac = elapsed / time_budget
        temp = temp_start * (1.0 - frac) + temp_end * frac
        sigma = sigma_max * (1.0 - frac) ** 2 + 1e-6

        delta = np.random.randn(n) * sigma
        cand = cur_h + delta
        cand = _project_to_constraints(cand, target)
        cand_score = _compute_c5(cand)

            cur_h = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                if best_score < 0.3808:
                    break
    return best_h, best_score

def _refine_by_gradient(h, max_iters=500, step=0.05):
    """Deterministic refinement: descend the gradient of the worst lag."""
    n = h.size
    target = n / 2.0
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)

        grad = np.zeros_like(h)
        if lag >= 0:
            i_end = n - lag
            if i_end > 0:
                idx_i = np.arange(i_end)
                grad[idx_i] += (1.0 - h[idx_i + lag])
            if lag < n:
                idx_j = np.arange(lag, n)
                grad[idx_j] -= h[idx_j - lag]
        else:
            lag = -lag
            if lag < n:
                idx_i = np.arange(lag, n)
                grad[idx_i] += (1.0 - h[idx_i - lag])
            i_end = n - lag
            if i_end > 0:
                idx_j = np.arange(i_end)
                grad[idx_j] -= h[idx_j + lag]

        new = h - step * grad
        new = _project_to_constraints(new, target)

        new_max = np.max(np.correlate(new, 1.0 - new, mode="full"))
        old_max = np.max(corr)
        if new_max < old_max - 1e-9:
            h = new
        else:
            step *= 0.5
            if step < 1e-6:
                break
    return h

    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            pass

    return best_h.astype(np.float64), float(final_score), int(best_h.size)
