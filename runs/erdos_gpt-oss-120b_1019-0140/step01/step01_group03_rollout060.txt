def compute_smooth_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)  # for stability
    # compute weights with softmax
    shift = corr - max_corr  # maybe subtract to avoid overflow
    w = np.exp(beta * shift)
    w = w / w.sum()
    # compute weighted sums
    conv_fwd = np.convolve(h, w, mode='full')
    fwd = conv_fwd[(n-1):(n-1)+n]
    conv_rev = np.convolve(h, w[::-1], mode='full')
    rev = conv_rev[(n-1):(n-1)+n]
    grad = 1.0 - fwd - rev
    return grad

def optimize_h(...):
    n = h0.size
    s = n * 0.5
    h = h0.copy()
    m_vec = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters+1):
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        # gradient
        if it <= warmup_iters:
            max_corr = np.max(corr)
            # stable softmax
            w = np.exp(beta_start + (beta_end-beta_start)*(it-1)/max(warmup_iters-1,1) * (corr - max_corr))
            w = w / w.sum()
            # compute weighted sums via convolution
            conv_fwd = np.convolve(h, w, mode='full')
            fwd = conv_fwd[(n-1):(n-1)+n]
            conv_rev = np.convolve(h, w[::-1], mode='full')
            rev = conv_rev[(n-1):(n-1)+n]
            grad = 1.0 - fwd - rev
        else:
            # identify top_m shifts
            idxs = np.argpartition(-corr, top_m-1)[:top_m]
            # compute gradients sum
            grad = np.zeros_like(h)
            for idx in idxs:
                shift = idx - (n-1)
                grad += compute_gradient(h, shift)
            grad = grad / top_m
        # momentum update
        m_vec = momentum * m_vec + (1 - momentum) * grad
        step = step0 / np.sqrt(it)
        # update and project
        h_new = h - step * m_vec
        h_new = project_onto_box_sum(h_new, s)
        # evaluate score
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # optional early exit if step too small and little improvement (ignore)
    return best_h, best_score

import time
import math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto the simplex with box constraints."""
    v = np.asarray(v, dtype=float)
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) + 10 * lo

    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1‑h(x+k)) dx."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of a single correlation term (shift can be negative)."""
    n = h.shape[0]
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[: n - shift] = h[shift:]
        term2[shift:] = h[: n - shift]
    else:
        s = -shift
        term1[s:] = h[: n - s]
        term2[: n - s] = h[s:]
    return 1.0 - term1 - term2

def optimize_h(
    h0,
    max_iters,
    warmup_iters,
    top_m,
    step0,
    momentum,
    beta_start,
    beta_end,
    start_time,
    budget_s,
):
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    m_vec = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iters + 1):
        if time.time() - start_time > budget_s - 0.5:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")

        if it <= warmup_iters and warmup_iters > 0:
            # smooth max via softmax weights
            beta = beta_start + (beta_end - beta_start) * (it - 1) / max(warmup_iters - 1, 1)
            max_corr = np.max(corr)
            w = np.exp(beta * (corr - max_corr))
            w_sum = w.sum()

            else:
                w /= w_sum
            # weighted forward and backward sums via convolution
            conv_fwd = np.convolve(h, w, mode="full")

            conv_rev = np.convolve(h, w[::-1], mode="full")

        else:
            # use top‑m shifts (average their gradients)
            idxs = np.argpartition(-corr, top_m - 1)[:top_m]
            grad = np.zeros_like(h)
            for idx in idxs:
                shift = idx - (n - 1)
                grad += compute_gradient(h, shift)
            grad /= float(top_m)

        # momentum update
        m_vec = momentum * m_vec + (1.0 - momentum) * grad
        step = step0 / math.sqrt(it)
        h_new = h - step * m_vec
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_score = float("inf")
    sizes = [256, 180, 128, 64]

    def is_power_of_two(x):
        return x > 0 and (x & (x - 1)) == 0

    # ----- deterministic candidates per size -----
    for n in sizes:
        if time.time() - start_time > budget_s - 5:
            break
        candidates = []

        if "initial_h_values" in globals():
            init_arr = np.asarray(initial_h_values, dtype=float)
            if init_arr.size != n:
                arr_res = upsample(init_arr, n)
            else:
                arr_res = init_arr.copy()
            arr_res = project_onto_box_sum(arr_res, n / 2.0)
            candidates.append(arr_res)

        if is_power_of_two(n):
            tm = thue_morse(n)
            tm = project_onto_box_sum(tm, n / 2.0)
            candidates.append(tm)

        for _ in range(2):
            candidates.append(random_binary_balanced(n, rng))
        for _ in range(2):
            cand = rng.random(n)
            cand = project_onto_box_sum(cand, n / 2.0)
            candidates.append(cand)

        uniform = np.full(n, 0.5, dtype=float)
        uniform = project_onto_box_sum(uniform, n / 2.0)
        candidates.append(uniform)

        if best_h is not None:
            cand = upsample(best_h, n)
            cand = project_onto_box_sum(cand, n / 2.0)
            candidates.append(cand)

        # ----- optimize each candidate -----
        for h0 in candidates:
            if time.time() - start_time > budget_s - 5:
                break
            max_iters = 12000 if n >= 180 else 8000
            warmup = int(0.15 * max_iters)
            h_opt, score_opt = optimize_h(
                h0,
                max_iters=max_iters,
                warmup_iters=warmup,
                top_m=3,
                step0=0.5,
                momentum=0.9,
                beta_start=0.5,
                beta_end=8.0,
                start_time=start_time,
                budget_s=budget_s,
            )
            if score_opt < best_score:
                best_score = score_opt
                best_h = h_opt.copy()

    # ----- random restarts until budget expires -----
    prob = np.array([0.5, 0.2, 0.2, 0.1])
    prob /= prob.sum()
    while time.time() - start_time < budget_s - 3:
        n = int(rng.choice(sizes, p=prob))
        h0 = rng.random(n)
        h0 = project_onto_box_sum(h0, n / 2.0)
        max_iters = 3000
        warmup = int(0.2 * max_iters)
        h_opt, score_opt = optimize_h(
            h0,
            max_iters=max_iters,
            warmup_iters=warmup,
            top_m=3,
            step0=0.5,
            momentum=0.9,
            beta_start=0.5,
            beta_end=5.0,
            start_time=start_time,
            budget_s=budget_s,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    # ----- final projection and return -----
    if best_h is None:
        n = 256
        best_h = np.full(n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, n / 2.0)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)

    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
