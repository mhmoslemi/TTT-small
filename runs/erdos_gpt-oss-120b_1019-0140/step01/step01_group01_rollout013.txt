def gradient_move(h, step, target_sum, rng):
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr_full))
    lag = idx - (len(h) - 1)
    n = len(h)
    g = np.zeros_like(h)
    # plus indices: i where i+lag in range
    if lag >= 0:
        plus_i = np.arange(n - lag)
        g[plus_i] += (1.0 - h[plus_i + lag])
        minus_i = np.arange(lag, n)
        g[minus_i] -= h[minus_i - lag]
    else:
        lag_neg = -lag
        plus_i = np.arange(lag_neg, n)

import time
import numpy as np

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Adjust h in place so that sum(h) == target while keeping 0 <= h_i <= 1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    n = h.size
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(n)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_c5_fft(h: np.ndarray) -> float:
    """Fast evaluation of C5 using FFT based linear correlation."""
    n = h.shape[0]
    size = 2 * n
    H = np.fft.fft(h, size)
    G = np.fft.fft(1.0 - h, size)
    corr = np.fft.ifft(H * np.conj(G)).real
    corr = corr[: 2 * n - 1]                      # linear (full) correlation
    dx = 2.0 / n
    return float(np.max(corr) * dx)


def gradient_move(h: np.ndarray, step: float, target_sum: float,
                  rng: np.random.RandomState) -> np.ndarray:
    """Move h in the direction that reduces the worst overlap (gradient of max term)."""
    n = h.shape[0]
    # linear full correlation to locate worst lag
    full_corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(full_corr.argmax())
    lag = idx - (n - 1)

    idxs = np.arange(n)
    plus = idxs + lag
    valid_plus = (plus >= 0) & (plus < n)
    minus = idxs - lag
    valid_minus = (minus >= 0) & (minus < n)

    g = np.zeros_like(h)
    g[valid_plus] += (1.0 - h[plus[valid_plus]])
    g[valid_minus] -= h[minus[valid_minus]]

    # project to hyperplane sum(g)=0
    g -= g.mean()
    cand = h - step * g
    np.clip(cand, 0.0, 1.0, out=cand)
    cand = adjust_sum_to_target(cand, target_sum, rng)
    return cand


def pairwise_move(h: np.ndarray, step: float, target_sum: float,
                  rng: np.random.RandomState) -> np.ndarray | None:
    """Exchange mass between two positions while respecting bounds."""
    n = h.shape[0]
    i, j = rng.randint(n, size=2)
    if i == j:
        return None
    hi, hj = h[i], h[j]
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if low >= high:
        return None
    delta = (rng.rand() * 2.0 - 1.0) * step
    delta = np.clip(delta, low, high)
    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    return cand


def noise_move(h: np.ndarray, step: float, target_sum: float,
               rng: np.random.RandomState) -> np.ndarray:

    np.clip(cand, 0.0, 1.0, out=cand)
    cand = adjust_sum_to_target(cand, target_sum, rng)
    return cand

                np.clip(cand, 0.0, 1.0, out=cand)
                cand = adjust_sum_to_target(cand, target_sum, rng)
            else:
                cand = rng.rand(n_points)
                cand = adjust_sum_to_target(cand, target_sum, rng)
            current_h = cand
            current_c5 = compute_c5_fft(current_h)
            stagnation = 0

        # Choose move type
        move_type = rng.rand()
        if move_type < 0.2:
            candidate = gradient_move(current_h, step, target_sum, rng)
        elif move_type < 0.7:
            candidate = pairwise_move(current_h, step, target_sum, rng)
            if candidate is None:
                continue
        else:
            candidate = noise_move(current_h, step, target_sum, rng)

        cand_c5 = compute_c5_fft(candidate)

        # Acceptance test
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            stagnation = 0
            # occasional symmetry enforcement (helps reduce peaks)
            if rng.rand() < 0.1:
                sym = (current_h + current_h[::-1]) * 0.5
                current_h = adjust_sum_to_target(sym, target_sum, rng)
                current_c5 = compute_c5_fft(current_h)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
        else:
            stagnation += 1

        total_iters += 1

    # final sanity: ensure constraints are met exactly
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5_fft(best_h)

    return (best_h, float(final_c5), n_points)
