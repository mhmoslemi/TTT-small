    best_val = compute_c5(best_h)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Compute the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-8, max_iter=15):
    """
    Project a vector onto the box [0,1]^n and the hyperplane sum(h)=target_sum.
    Simple iterative scheme that adjusts interior entries.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # all at bounds, distribute to the most flexible entry
            i = np.argmin(h) if diff > 0 else np.argmax(h)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Linearly interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new
