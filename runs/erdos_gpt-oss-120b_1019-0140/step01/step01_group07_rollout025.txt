def _compute_c5_fft(h):
    n = h.size
    dx = 2.0 / n
    # compute convolution via FFT with zero padding
    L = int(2 ** np.ceil(np.log2(2*n)))
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * G, n=L)[:2*n-1]   # real part

import numpy as np, time
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    # same as earlier
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # sort descending caps to fill
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5_fft(h):
    n = h.size
    dx = 2.0 / n
    L = int(2 ** np.ceil(np.log2(2*n)))
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * np.conj(G), n=L)[:2*n-1]
    return float(np.max(conv) * dx)

def tournament_select(pop, fitness, k=3):
    idx = np.random.choice(len(pop), size=k, replace=False)
    best_idx = idx[np.argmin(fitness[idx])]
    return pop[best_idx].copy()

def crossover(parent1, parent2):
    n = parent1.size
    if n < 2:
        return parent1.copy()
    p1 = np.random.randint(1, n-1)
    p2 = np.random.randint(p1, n-1)
    child = np.empty_like(parent1)
    child[:p1] = parent1[:p1]
    child[p1:p2] = parent2[p1:p2]
    child[p2:] = parent1[p2:]
    return child

def mutate(individual, sigma):
    n = individual.size
    pert = individual + np.random.randn(n) * sigma
    target = n / 2.0
    return _project_to_constraints(pert, target)

def local_optimize(h, target_sum, maxiter=200):
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x, tgt=target_sum: np.sum(x) - tgt}
    try:
        res = minimize(_compute_c5_fft, h, method="SLSQP", bounds=bounds,
                       constraints=[cons],
                       options={"maxiter": maxiter, "ftol": 1e-9, "disp": False})
        h_opt = _project_to_constraints(res.x, target_sum)
        score = _compute_c5_fft(h_opt)
        return h_opt, score
    except Exception:
        return h, _compute_c5_fft(h)

import numpy as np
import time
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust sum(v) to target by water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5_fft(h):
    """Return max_k ∫ h(x)(1-h(x+k))dx using FFT based correlation."""
    n = h.size
    dx = 2.0 / n
    L = int(2 ** np.ceil(np.log2(2 * n)))
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * np.conj(G), n=L)[: 2 * n - 1]
    return float(np.max(conv) * dx)

def tournament_select(pop, fitness, k=3):
    idx = np.random.choice(len(pop), size=k, replace=False)
    best_idx = idx[np.argmin(fitness[idx])]
    return pop[best_idx].copy()

def crossover(p1, p2):
    n = p1.size
    if n < 2:
        return p1.copy()
    a = np.random.randint(1, n - 1)
    b = np.random.randint(a, n - 1)
    child = np.empty_like(p1)
    child[:a] = p1[:a]
    child[a:b] = p2[a:b]
    child[b:] = p1[b:]
    return child

def mutate(ind, sigma):
    pert = ind + np.random.randn(ind.size) * sigma
    return _project_to_constraints(pert, ind.size / 2.0)

def local_optimize(h, target_sum, maxiter=200):
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x, tgt=target_sum: np.sum(x) - tgt}
    try:
        res = minimize(
            _compute_c5_fft,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
        )
        h_opt = _project_to_constraints(res.x, target_sum)
        score = _compute_c5_fft(h_opt)
        return h_opt, score
    except Exception:
        return h, _compute_c5_fft(h)
