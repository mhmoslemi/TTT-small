def compute_corr_fft(h):
    n = h.size
    # length for linear correlation via FFT: N = 2*n-1 maybe next pow2.
    N = 1 << ((2*n - 2).bit_length())
    # Compute FFT of h and reversed (1-h)
    h_fft = np.fft.rfft(h, n=N)

    corr = np.fft.irfft(h_fft * g_rev_fft, n=N)
    # The result length N, but linear correlation length = 2*n -1 in first part.
    corr = corr[:2*n-1]

def compute_gradient(h, shift):
    n = h.shape[0]
    grad = np.ones_like(h)
    if shift >= 0:
        # shift forward
        grad[:n-shift] -= h[shift:]
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        grad[s:] -= h[:n-s]
        grad[:n-s] -= h[s:]
    return grad
```

import time
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    # Quick bounds
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # Find lambda such that sum clip(v-lam) = s
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure proper bracketing
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10.0
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10.0

    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def linear_correlation_fft(h):
    """Linear correlation of h with (1-h) via FFT (full mode)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 2).bit_length())  # next power of two >= 2n-1
    h_fft = np.fft.rfft(h, n=N)
    g_rev_fft = np.fft.rfft((1.0 - h)[::-1], n=N)
    corr = np.fft.irfft(h_fft * g_rev_fft, n=N)[:2 * n - 1]
    return corr

def compute_c5_fast(h):
    """Maximum overlap integral for discretized h."""
    n = h.size
    corr = linear_correlation_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_gradient(h, shift):
    """Gradient of the active max‑overlap term."""
    n = h.size
    grad = np.ones_like(h)
    if shift >= 0:
        if shift != 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:]   -= h[:n - shift]
    else:
        s = -shift
        grad[s:]       -= h[:n - s]
        grad[:n - s]   -= h[s:]
    return grad

def thue_morse(n):
    """Generate Thue–Morse binary sequence of length n (0/1)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def optimize_candidate(h0, max_iters, rng):
    """Projected subgradient descent with momentum using FFT correlation."""
    n = h0.size
    s = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    v = np.zeros_like(h)
    step0 = 1.0
    beta = 0.9
    N_fft = 1 << ((2 * n - 2).bit_length())
    for it in range(1, max_iters + 1):
        # correlation to find worst shift
        h_fft = np.fft.rfft(h, n=N_fft)
        g_rev_fft = np.fft.rfft((1.0 - h)[::-1], n=N_fft)
        corr = np.fft.irfft(h_fft * g_rev_fft, n=N_fft)[:2 * n - 1]
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # gradient of the active term
        grad = compute_gradient(h, shift)
        eta = step0 / np.sqrt(it)
        v = beta * v - eta * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
        # evaluate new C5
        h_fft = np.fft.rfft(h, n=N_fft)
        g_rev_fft = np.fft.rfft((1.0 - h)[::-1], n=N_fft)
        corr = np.fft.irfft(h_fft * g_rev_fft, n=N_fft)[:2 * n - 1]
        score = np.max(corr) * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # occasional small perturbation to escape shallow minima
        if it % 5000 == 0:
            perturb = (rng.random(n) - 0.5) * 0.005
            h = project_onto_box_sum(h + perturb, s)
    return best_h, best_score
