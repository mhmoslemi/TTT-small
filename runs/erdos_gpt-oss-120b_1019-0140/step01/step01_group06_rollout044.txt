def hill_climb(h, n_steps, rng):
    h = h.copy()
    c5 = compute_c5(h)
    for step in range(n_steps):
        # Choose two indices
        i, j = rng.integers(0, len(h), size=2)
        if i == j:
            continue
        # Determine max delta allowed
        max_up = 1.0 - h[i]
        max_down = h[j]
        # we want increase i and decrease j, so delta ≤ min(max_up, max_down)
        delta_max = min(max_up, max_down)
        if delta_max <= 0:
            continue
        delta = rng.random() * min(0.05, delta_max)
        # apply
        h[i] += delta
        h[j] -= delta
        new_c5 = compute_c5(h)
        if new_c5 < c5:
            c5 = new_c5
        else:
            # revert
            h[i] -= delta
            h[j] += delta
    return h, c5

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    """Project onto box [0,1]^n and hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # distribute remaining error greedily
        err = target_sum - cur
        idx = np.argmin(h) if err > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + err, 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

    """

    # Load possible parent construction.

    if base_h is not None:

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Resolution schedule (increasing degrees of freedom).
    resolutions = [120, 240, 480, 760]
    if base_h is not None:
        n_base = base_h.shape[0]

            resolutions = [n_base] + resolutions

    for n in resolutions:
        target_sum = n / 2.0

        # Determine restart and optimizer settings based on n.
        if n <= 120:

            maxiter = 1500
        elif n <= 240:
            max_restarts = 60
