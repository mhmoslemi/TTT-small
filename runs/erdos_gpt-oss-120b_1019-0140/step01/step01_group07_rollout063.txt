        elapsed = time.time() - start

        else:
            h_new[j] -= a; h_new[i] += a

def simulated_annealing(x, score, timelimit):
    start = time.time()
    best_x = x.copy()
    best_score = score
    cur_x = x.copy()
    cur_score = score
    n = len(x)
    while time.time() - start < timelimit:
        elapsed = time.time() - start
        frac = elapsed / timelimit
        T = max(0.001, 1.0 - frac)  # linear schedule from 1 to 0.001
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # decide direction: transfer from i to j
        if np.random.rand() < 0.5:
            max_transfer = min(T * 0.5, cur_x[i], 1.0 - cur_x[j])
            if max_transfer <= 0:
                continue
            a = np.random.rand() * max_transfer
            new_x = cur_x.copy()
            new_x[i] -= a
            new_x[j] += a
        else:
            max_transfer = min(T * 0.5, cur_x[j], 1.0 - cur_x[i])
            if max_transfer <= 0:
                continue
            a = np.random.rand() * max_transfer
            new_x = cur_x.copy()
            new_x[j] -= a
            new_x[i] += a
        # compute score
        new_score = _compute_c5(new_x)
        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T+1e-12)):
            cur_x = new_x
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_x = new_x.copy()
                # early exit if target achieved
                if best_score < 0.3808:
                    break
    return best_x, best_score

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-14:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-14:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

    n = h_start.size
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda h, tgt=target: np.sum(h) - tgt}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _compute_c5,
            h_start,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": 300, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            h_opt = _project_to_constraints(res.x, target)
            return h_opt, _compute_c5(h_opt)
    except Exception:
        pass
    return h_start, _compute_c5(h_start)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        init_vec = np.array(init_vec, dtype=np.float64).ravel()
        n0 = init_vec.size
    else:
        n0 = 380
        init_vec = np.random.rand(n0)

    resolutions = [n0]
    if budget_s >= 300:
        resolutions.append(n0 * 2)
    if budget_s >= 600:
        resolutions.append(n0 * 4)

    best_h = None
    best_score = float("inf")

    for n in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
        target_sum = n / 2.0

        if n == init_vec.size:
            base_h = init_vec.copy()
        else:
            src = best_h if best_h is not None else init_vec
            src_n = src.size
            x_src = np.linspace(0, 2, src_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            base_h = np.interp(x_new, x_src, src)
        base_h = _project_to_constraints(base_h, target_sum)

        start_points = [base_h]
        for _ in range(5):
            bh = np.zeros(n, dtype=np.float64)
            bh[np.random.choice(n, n // 2, replace=False)] = 1.0
            start_points.append(bh)
        for _ in range(5):
            rh = np.random.rand(n)
            rh = _project_to_constraints(rh, target_sum)
            start_points.append(rh)

        time_remaining = budget_s - (time.time() - start_time)
        if time_remaining <= 0:
            break
        per_start = max(5.0, time_remaining / (len(start_points) + 1))

        for h0 in start_points:
            if time.time() - start_time > budget_s * 0.95:
                break
            s0 = _compute_c5(h0)
            # quick refinement
            if per_start * 0.2 > 1.0:
                h_ref, s_ref = _refine_slsqp(h0)
                if s_ref < s0:
                    h0, s0 = h_ref, s_ref
            # simulated annealing
            h_sa, s_sa = _simulated_annealing(h0, s0, per_start * 0.8)
            # final local refinement
            h_fin, s_fin = _refine_slsqp(h_sa)
            if s_fin < best_score:
                best_score = s_fin
                best_h = h_fin.copy()
                if best_score < 0.3808:
                    break
        if best_score < 0.3808:
            break

    if best_h is None:
        best_h = _project_to_constraints(np.random.rand(n0), n0 / 2.0)
        best_score = _compute_c5(best_h)

    final_score = best_score
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            pass

    return best_h.astype(np.float64), final_score, best_h.size
