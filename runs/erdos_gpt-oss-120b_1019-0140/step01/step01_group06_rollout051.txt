def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    # iterative scaling
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # compute mask of variables not at bounds
        interior = (h > 0.0) & (h < 1.0)
        num_interior = np.count_nonzero(interior)
        if num_interior == 0:
            # no interior, distribute diff proportionally to capacity
            # we will just adjust the variable farthest from bound
            idx = np.argmin(h) if diff > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            break
        # distribute diff equally among interior variables
        delta = diff / num_interior
        h[interior] += delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
    # final adjustment if necessary
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust a random interior or any variable
        idx = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
    return h
```

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    rng = np.random.default_rng(seed)
    base = globals().get('initial_h_values')
    if base is not None:
        base = np.asarray(base, dtype=float)
        n = base.shape[0]

    target_sum = n/2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step = 0.05
    iter_cnt = 0
    # main

    while time.time() - start < budget_s * 0.96:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        c5 = corr[max_idx] * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # early exit possible but continue
        # choose top lags
        top_k = 3
        # indices of top k correlations
        if top_k < len(corr):
            top_idxs = np.argpartition(-corr, top_k-1)[:top_k]
        else:
            top_idxs = np.arange(len(corr))
        grad_total = np.zeros(n)
        for idx in top_idxs:
            lag = idx - (n - 1)
            if lag >= 0:
                shift_plus = np.concatenate([h[lag:], np.zeros(lag)])
                shift_minus = np.concatenate([np.zeros(lag), h[:n - lag]])
            else:
                L = -lag
                shift_plus = np.concatenate([np.zeros(L), h[:n - L]])
                shift_minus = np.concatenate([h[L:], np.zeros(L)])
            grad_total += (1.0 - (shift_plus + shift_minus))
        grad = grad_total / max(1, len(top_idxs))
        # update
        h = h - step * grad
        # project to feasible
        h = project_to_feasible(h, target_sum)
        iter_cnt += 1
        if iter_cnt % 500 == 0:
            step *= 0.9
        # occasional random perturb
        if iter_cnt % 2000 == 0:
            # add small noise to escape
            h = np.clip(h + rng.normal(scale=0.02, size=n), 0.0, 1.0)
            h = project_to_feasible(h, target_sum)
    # final local random perturb search
    for _ in range(200):
        if time.time() - start > budget_s * 0.96:

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h: np.ndarray, target_sum: float, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """
    Project vector h onto the intersection of the box [0,1]^n and the hyperplane sum(h)=target_sum.
    Simple iterative redistribution of the surplus/deficit among interior entries.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        interior = (h > 0.0) & (h < 1.0)
        free = np.count_nonzero(interior)
        if free == 0:
            # no interior variables; adjust the element with most slack
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            break
        delta = diff / free
        h[interior] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        idx = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Retrieve parent construction if present
    base_h = globals().get("initial_h_values")
    if base_h is not None:
        h = np.asarray(base_h, dtype=float).copy()
        n = h.shape[0]
        h = project_to_feasible(h, n / 2.0)
    else:
        n = 380
        h = rng.random(n)
        h = project_to_feasible(h, n / 2.0)

    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    step = 0.08               # initial gradient step size
    iter_cnt = 0
    top_k = 3                 # number of worst lags to blend in the gradient

    # Main projected‑gradient descent loop
    while time.time() - start_time < budget_s * 0.96:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        c5 = corr[max_idx] * (2.0 / n)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # select the top‑k lags (worst overlaps)
        if top_k < corr.size:
            idxs = np.argpartition(-corr, top_k - 1)[:top_k]
        else:
            idxs = np.arange(corr.size)

        grad = np.zeros_like(h)
        for idx in idxs:
            lag = idx - (n - 1)
            if lag >= 0:
                shift_plus = np.concatenate([h[lag:], np.zeros(lag)])
                shift_minus = np.concatenate([np.zeros(lag), h[: n - lag]])
            else:
                L = -lag
                shift_plus = np.concatenate([np.zeros(L), h[: n - L]])
                shift_minus = np.concatenate([h[L:], np.zeros(L)])
            grad += 1.0 - (shift_plus + shift_minus)

        grad /= max(1, len(idxs))

        # gradient step and projection back to the feasible set
        h = h - step * grad
        h = project_to_feasible(h, target_sum)

        iter_cnt += 1
        # gradually shrink step size
        if iter_cnt % 800 == 0:
            step *= 0.85
        # occasional random jitter to escape shallow plateaus
        if iter_cnt % 2400 == 0:
            h = np.clip(h + rng.normal(scale=0.015, size=n), 0.0, 1.0)
            h = project_to_feasible(h, target_sum)

    # Post‑processing: small random local search around the best found point
    for _ in range(300):
        if time.time() - start_time > budget_s * 0.97:
            break
        cand = best_h + rng.normal(scale=0.012, size=n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_to_feasible(cand, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()

    # final safety projection and bound recomputation
    best_h = project_to_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
