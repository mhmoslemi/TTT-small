    N = h_cur.size
    idx = rng.choice(N, size=m, replace=False)
    s = float(h_cur[idx].sum())

def _repair_vector(v):
    v = np.clip(v, 0, 1)
    target = v.size / 2.0
    s = v.sum()
    if s == 0:
        # fill uniformly
        v = np.full_like(v, target / v.size)
        v = np.clip(v, 0, 1)
        return v
    # scaling factor to achieve target sum before clipping, but after scaling may exceed 1.
    scale = target / s
    v = v * scale
    # Now clamp
    v = np.clip(v, 0, 1)
    # recompute sum
    s2 = v.sum()
    # If not exact, adjust again
    if abs(s2 - target) > 1e-12:
        scale2 = target / s2
        v = v * scale2
        v = np.clip(v, 0, 1)
    return v

    cur_score = float(np.max(corr_cur) * dx)

    best_h = h_cur.copy()
    best_score = cur_score

        if elapsed >= time_limit:
            break

        T = T0 * frac

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and scale to make sum == len(v)/2 exactly."""
    v = np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _perturb_solution(h: np.ndarray, rng: np.random.Generator, sigma: float = 0.05) -> np.ndarray:
    """Add Gaussian noise and repair."""
    h_new = h + sigma * rng.standard_normal(h.shape)
    return _repair_vector(h_new)

            x_new = np.linspace(0.0, 2.0, n1, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = _repair_vector(h_up)
            time_stage2 = remaining * 0.6
            best_h, best_score = _run_stage(h_up, time_stage2, rng)
            elapsed = time.time() - start_total
            remaining = total_budget - elapsed

    # ----- stage 3: quadruple resolution if still time -----
    if remaining > total_budget * 0.1:
        n2 = min(n0 * 4, 500)
        if n2 != best_h.size:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
            h_up2 = np.interp(x_new, x_old, best_h)
            h_up2 = _repair_vector(h_up2)
            best_h, best_score = _run_stage(h_up2, remaining, rng)

    # ----- final local refinement with any slack time -----
    elapsed = time.time() - start_total
    slack = total_budget - elapsed
    if slack > 5.0:
        best_h, best_score = _local_refinement(best_h, slack, rng)

    # ----- final repair and exact evaluation -----
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
