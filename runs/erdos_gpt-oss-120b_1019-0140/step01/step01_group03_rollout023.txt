import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = len(h)
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n-shift] = h[shift:]
        term2[shift:] = h[:n-shift]
    else:
        s = -shift
        term1[s:] = h[:n-s]
        term2[:n-s] = h[s:]
    return 1.0 - term1 - term2

def projected_gradient_descent(h0, n_iter=5000, step0=0.5):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    # Optionally momentum
    vel = np.zeros_like(h)
    beta = 0.9
    for it in range(1, n_iter+1):
        # correlation and shift of worst term
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        # step size schedule
        eta = step0 / math.sqrt(it)
        # momentum update
        vel = beta * vel + eta * grad
        h_new = h - vel
        # projection
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    # produce binary 0/1 sequence with balanced ones for power of two
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):

    k = int(round(target_sum))
    idx = np.argpartition(v, -k)[-k:]  # top k
    h = np.zeros_like(v)
    h[idx] = 1.0
    return h

def projected_gradient_descent_time(h0, step0, time_budget, max_iter=50000):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()

    vel = np.zeros_like(h)
    beta = 0.9
    start = time.time()
    it = 0
    while it < max_iter and (time.time() - start) < time_budget:
        it += 1
        # same as before but compute...
        corr = np.correlate(h, 1.0 - h, mode='full')

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand bounds if needed
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of the active correlation term for a given integer shift."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def projected_gradient_descent(h0, max_iter=5000, step0=0.5):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    vel = np.zeros_like(h)
    beta = 0.9
    for it in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        shift = np.argmax(corr) - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        vel = beta * vel + eta * grad
        h_new = project_onto_box_sum(h - vel, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binary_from_continuous(v, target_sum):
    """Convert a vector to a binary one with exactly target_sum ones."""
    k = int(round(target_sum))
    if k <= 0:
        return np.zeros_like(v)
    idx = np.argpartition(v, -k)[-k:]
    h = np.zeros_like(v)
    h[idx] = 1.0
    return h

def simulated_annealing_binary(start_h, rng, time_limit):
    """Random‑swap simulated annealing on binary vectors within a time budget."""
    n = start_h.size
    best_h = start_h.copy()
    best_score = compute_c5(best_h)
    cur_h = start_h.copy()
    cur_score = best_score
    start_time = time.time()
    # schedule parameters
    T0, T_end = 0.02, 0.0001
    # cap total steps to keep overhead manageable
    est_step = 0.0003
    max_steps = int(min(time_limit / est_step, 300000))
    if max_steps < 1000:
        max_steps = int(min(time_limit / est_step, 2000))
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        t = step / max_steps
        T = T0 * (1 - t) + T_end * t
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1.0 - cur_h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = compute_c5(new_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
    return best_h, best_score
