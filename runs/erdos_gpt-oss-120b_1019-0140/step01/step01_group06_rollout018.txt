def anneal_h(h_init, time_budget, start_time):
    h = h_init.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    n = h.shape[0]
    target_sum = n/2.0
    rng = np.random.default_rng()
    # schedule: temperature decays linearly from T0 to 0 across allocated moves.
    T0 = 0.01
    moves = 0
    max_moves = 50000  # limit or time.
    while True:
        if time.time() - start_time > time_budget * 0.97:
            break
        if moves >= max_moves:
            break
        moves += 1
        # compute temperature schedule
        T = T0 * (1 - moves / max_moves)
        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Determine direction: move from i to j if possible
        if h[i] > 0 and h[j] < 1:
            max_delta = min(h[i], 1.0 - h[j])
        else:
            continue
        delta = max_delta * rng.random()
        # apply move
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h)
        # acceptance
        if new_c5 < current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / max(T, 1e-12)):
            current_c5 = new_c5
            # update best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # reject: revert move
            h[i] += delta
            h[j] -= delta
    return best_h, best_c5

def generate_random_harmonic(n, rng, num_modes=8):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for mode in range(1, num_modes+1):
        amp = rng.uniform(-0.5, 0.5)
        phase = rng.uniform(0, 2*np.pi)
        h += amp * np.cos(2*np.pi * mode * x / 2 + phase)
    # shift to nonnegative
    h -= h.min()
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # project sum
    target_sum = n / 2.0
    h = project_to_feasible(h, target_sum)
    return h

def anneal_h(h_init, budget_s, start_time, max_moves=5000):
    h = h_init.copy()
    n = h.shape[0]
    target_sum = n/2.0
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng_local = np.random.default_rng()
    # Temperature schedule
    T0 = 0.01
    for move in range(max_moves):
        # check time
        if time.time() - start_time > budget_s * 0.96:
            break
        # linear temperature decay
        T = T0 * (1 - move / max_moves)
        # propose move
        i = rng_local.integers(n)
        j = rng_local.integers(n)
        if i == j:
            continue
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng_local.random()
        # apply
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h)
        # decide acceptance
        if new_c5 < current_c5 or rng_local.random() < np.exp(-(new_c5 - current_c5)/max(T, 1e-12)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final fix if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-8:
        # increase or decrease at most permissive index
        if cur < target_sum:
            idx = np.argmin(h)  # smallest value, room to increase
            h[idx] = min(1.0, h[idx] + (target_sum - cur))
        else:
            idx = np.argmax(h)  # largest value, room to decrease
            h[idx] = max(0.0, h[idx] - (cur - target_sum))
    return h

def generate_random_harmonic(n, rng, num_modes=8):

    if n_target == n_base:
        h_new = h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_target = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_target, x_base, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

        h[j] += delta
        new_c5 = compute_c5(h)
        if new_c5 < current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / max(T, 1e-12)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
    # ensure feasibility (sum unchanged)
    best_h = project_to_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discrete step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-9, max_iter=10):
    """Clip to [0,1] and adjust to satisfy the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-8:
        if cur < target_sum:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + (target_sum - cur))
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] - (cur - target_sum))
    return h
