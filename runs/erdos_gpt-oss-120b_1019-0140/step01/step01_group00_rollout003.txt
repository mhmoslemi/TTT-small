def build_milp(n):
    dx = 2.0 / n
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n // 2, t >= 0]

    w_by_shift = {}
    w_vars = [] # just to have list

    for k in range(1, n):

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, cvxpy as cp
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80

    dx = 2.0 / n

    # Build MILP
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(nonneg=True)
    constraints = [cp.sum(h) == n // 2, t >= 0]

    # We'll store w variables per shift
    w_by_shift = {}
    for k in range(1, n):
        wk = []
        for i in range(n - k):
            w = cp.Variable(boolean=True)
            wk.append(w)
            constraints.append(w <= h[i])
            constraints.append(w <= h[i + k])
            constraints.append(w >= h[i] + h[i + k] - 1)
        w_by_shift[k] = wk
        region_sum = cp.sum(h[:n - k])
        overlap = region_sum - sum(wk)  # Python sum yields cp expression
        constraints.append(overlap * dx <= t)

    # Solve
    try:
        prob = cp.Problem(cp.Minimize(t), constraints)
        # compute remaining budget
        rem = budget_s - (time.time() - start_time)
        # add time limit if positive
        # using solver options
        prob.solve(solver=cp.CBC, maxSeconds=rem)
    except Exception as e:
        # solver failed
        prob = None

    # Prepare best solution
    best_h = None
    best_c5 = np.inf

    if prob is not None and prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        h_vals = np.array(h.value).flatten()
        # convert to binary using top n/2 indices to ensure exact sum
        top_idx = np.argsort(-h_vals)[: n // 2]
        binary_h = np.zeros(n, dtype=float)
        binary_h[top_idx] = 1.0
        best_h = binary_h
        best_c5 = compute_c5(best_h)
    else:
        # fallback: construct random binary vector with half ones
        binary_h = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        binary_h[idx] = 1.0
        best_h = binary_h
        best_c5 = compute_c5(best_h)

    # Additional hill climb improvements if time left
    while time.time() - start_time < budget_s - 0.5:
        # simple random swap hill climb with occasional acceptance
        # find positions of 1s and 0s
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_c5 = compute_c5(cand)
        # Accept if improvement, else maybe accept with probability? Use strict improvement
        if cand_c5 + 1e-12 < best_c5:
            best_h = cand
            best_c5 = cand_c5
        # else continue

    # final clip
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure exact sum (if slight numerical issues)
    cur_sum = np.sum(best_h)
    target_sum = n / 2.0
    if abs(cur_sum - target_sum) > 1e-9:
        # fix: adjust using top values
        adjusted = np.zeros(n, dtype=float)
        idx = np.argsort(-best_h)[:int(target_sum)]
        adjusted[idx] = 1.0
        best_h = adjusted

    return best_h, best_c5, n

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def hill_climb_binary(start_h, start_c5, start_time, budget_s, rng):
    """Simple random swap hill‑climbing on a binary vector."""
    n = len(start_h)
    best_h = start_h.copy()
    best_val = start_c5
    while time.time() - start_time < budget_s - 0.1:
        ones = np.flatnonzero(best_h == 1.0)
        zeros = np.flatnonzero(best_h == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = compute_c5(cand)
        if val + 1e-12 < best_val:
            best_h = cand
            best_val = val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80
    target_ones = n // 2
    dx = 2.0 / n

    # ---------- MILP formulation ----------
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(nonneg=True)
    constraints = [cp.sum(h) == target_ones]

    # auxiliary binary variables for products h_i * h_{i+k}
    for k in range(1, n):
        wk = []
        for i in range(n - k):
            w = cp.Variable(boolean=True)
            wk.append(w)
            constraints.append(w <= h[i])
            constraints.append(w <= h[i + k])
            constraints.append(w >= h[i] + h[i + k] - 1)
        # overlap constraint for this shift
        region_sum = cp.sum(h[: n - k])
        overlap = region_sum - sum(wk)   # linear expression
        constraints.append(overlap * dx <= t)

    # ---------- Solve ----------
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, warm_start=True)
    except Exception:
        prob = None

    # ---------- Extract solution ----------
    if prob is not None and prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        h_vals = np.asarray(h.value).flatten()
        # Enforce exact half‑ones by taking the largest entries
        idx = np.argsort(-h_vals)[:target_ones]
        best_h = np.zeros(n, dtype=float)
        best_h[idx] = 1.0
        best_c5 = compute_c5(best_h)
    else:
        # fall back to a random feasible binary vector
        best_h = np.zeros(n, dtype=float)
        idx = rng.choice(n, target_ones, replace=False)
        best_h[idx] = 1.0
        best_c5 = compute_c5(best_h)

    # ---------- Optional local improvement ----------
    # Use simple hill climbing while time permits
    while time.time() - start_time < budget_s - 0.5:
        best_h, best_c5 = hill_climb_binary(best_h, best_c5, start_time, budget_s, rng)

    # final safety: ensure exact sum and range
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = np.sum(best_h)
    if abs(cur_sum - target_ones) > 1e-9:
        # re‑select top entries to guarantee the sum
        idx = np.argsort(-best_h)[:target_ones]
        best_h[:] = 0.0
        best_h[idx] = 1.0

    return best_h, best_c5, n
