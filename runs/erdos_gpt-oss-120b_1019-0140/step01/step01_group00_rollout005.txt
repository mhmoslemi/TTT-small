def simulated_annealing(h, start_c5, start_time, budget_s):
    n = len(h)
    current_h = h.copy()
    current_c5 = start_c5
    best_h = h.copy()
    best_c5 = start_c5
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:
            break
        # Temperature schedule: linear decay from T0 to T_end
        T0 = 1e-2
        T_end = 1e-5
        T = T0 * (1 - elapsed / (budget_s - 0.5)) + T_end * (elapsed / (budget_s - 0.5))
        # pick random swap
        ones = np.flatnonzero(current_h == 1)
        zeros = np.flatnonzero(current_h == 0)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        current_h[i], current_h[j] = 0, 1
        new_c5 = compute_c5(current_h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            # revert

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step function sampled on n points."""
    h = np.asarray(h, dtype=float)
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def random_binary(n, ones, rng):
    """Return a binary int array of length n with exactly `ones` ones."""
    arr = np.zeros(n, dtype=int)
    arr[rng.choice(n, ones, replace=False)] = 1
    return arr

def to_binary_vector(h):
    """Convert arbitrary float vector to a binary vector with exactly half ones."""
    n = len(h)
    target = n // 2
    # already binary with correct sum?
    if np.all((h == 0) | (h == 1)) and int(np.round(np.sum(h))) == target:
        return h.astype(float)
    idx = np.argpartition(-h, target - 1)[:target]
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise vector length and starting point -----
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = random_binary(n, n // 2, rng).astype(float)

    # enforce binary with exact half ones
    best_h = to_binary_vector(h0)
    best_c5 = compute_c5(best_h)

    # ----- Phase 1: quick random sampling -----
    total_time = budget_s
    random_phase = min(30.0, total_time * 0.1)
    while time.time() - start_time < random_phase:
        cand = random_binary(n, n // 2, rng)
        c5 = compute_c5(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.astype(float)

    # ----- Phase 2: Simulated annealing from best solution -----
    current_h = best_h.astype(int)
    current_c5 = best_c5
    T0 = 2e-2            # initial temperature
    T_end = 5e-6         # final temperature
    # main SA loop
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= total_time - 0.2:
            break
        # linear temperature schedule
        t_frac = elapsed / (total_time - 0.2)
        T = T0 * (1 - t_frac) + T_end * t_frac

        # choose a random 1↔0 swap
        ones_idx = np.flatnonzero(current_h == 1)

        if ones_idx.size == 0 or zeros_idx.size == 0:
