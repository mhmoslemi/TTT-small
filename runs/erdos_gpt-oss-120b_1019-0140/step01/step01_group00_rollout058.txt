def enforce_sum_and_bounds(h, target):
    g = np.clip(h,0,1)
    diff = target - g.sum()
    # iterative adjust
    while abs(diff) > 1e-12:
        free = (g > 0) & (g < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        # apply delta
        g[free] += delta
        # clip again
        np.clip(g, 0, 1, out=g)
        diff = target - g.sum()
        # If adjustments stall due to rounding, break
        if abs(delta) < 1e-12:
            break
    return g

def simulated_annealing(h_init, budget, start_time):

    T0 = 0.1 # or best_val? Actually difference scales.

    iter = 0

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def enforce_sum_and_bounds(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to achieve exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # iterative water‑filling on the free (non‑boundary) entries
    while abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        # apply delta but keep within bounds
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        # safeguard against stagnation
        if abs(delta) < 1e-12:
            break
    return h


class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n: int):
        self.n = n
        self.target = n / 2.0

    def __call__(self, x):
        return np.sum(x[: self.n]) - self.target


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k: int, n: int):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overl * self.dx


def hill_climb_binary(b_start: np.ndarray, start_c5: float,
                      start_time: float, budget_s: float):
    """Swap‑based hill climbing on a binary vector."""
    n = b_start.size
    dx = 2.0 / n
    best_h = b_start.copy()
    best_val = start_c5
    rng = np.random.default_rng()
    while time.time() - start_time < budget_s - 0.1:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        val = np.max(np.correlate(cand, 1 - cand, mode="full")) * dx
        if val + 1e-12 < best_val:
            best_h = cand
            best_val = val
    return best_h, best_val


def simulated_annealing(h_start: np.ndarray, time_budget: float,
                        start_time: float, rng: np.random.Generator):
    """Simple SA on the continuous variables (preserves sum)."""
    n = h_start.size
    target = n / 2.0
    h = h_start.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    T0 = 0.1  # initial temperature
    T = T0
    decay = 0.9995

    while time.time() - start_time < time_budget:
        i, j = rng.choice(n, size=2, replace=False)
        # decide direction of mass transfer
        if h[i] > 0.0 and h[j] < 1.0:
            max_delta = min(h[i], 1.0 - h[j])
            delta = rng.random() * max_delta
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        elif h[j] > 0.0 and h[i] < 1.0:
            max_delta = min(h[j], 1.0 - h[i])
            delta = rng.random() * max_delta
            new_h = h.copy()
            new_h[j] -= delta
            new_h[i] += delta
        else:
            continue

        new_val = compute_c5(new_h)
        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val

        T = max(T * decay, 1e-6)

    return best_h, best_val


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- Determine problem size and initial h -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = enforce_sum_and_bounds(h0, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        # symmetric triangular shape (integrates to 1)
        x = np.arange(n)
        h0 = 1.0 - np.abs(x - (n - 1) / 2.0) * 2.0 / (n - 1)
        h0 = enforce_sum_and_bounds(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----- Build constraints for SLSQP -----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{"type": "eq", "fun": sum_con}]
    constraints += [{"type": "ineq", "fun": oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # ----- Phase 1: many SLSQP runs (random starts) -----
    slsqp_budget = budget_s * 0.4
    while time.time() - start_time < slsqp_budget:
        # random feasible start
        rand = rng.random(n)
        rand = enforce_sum_and_bounds(rand, target_sum)

        t0 = compute_c5(rand)
        x0 = np.concatenate([rand, [t0]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 1500, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            break

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = enforce_sum_and_bounds(h_opt, target_sum)
        val_opt = compute_c5(h_opt)

        if val_opt + 1e-12 < best_c5:
            best_h = h_opt.copy()
            best_c5 = val_opt

        if best_c5 < 0.3808:
            break

    # ----- Phase 2: simulated annealing from current best -----
    remaining = budget_s - (time.time() - start_time) - 1.0
    if remaining > 10.0:
        h_sa, val_sa = simulated_annealing(best_h, remaining, start_time, rng)
        if val_sa + 1e-12 < best_c5:
            best_h = h_sa
            best_c5 = val_sa

    # ----- Phase 3: binary hill‑climb fallback (optional) -----
    elapsed = time.time() - start_time
    if elapsed < budget_s - 1.0:
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argsort(-best_h)[: n // 2]
        binary_h[top_idx] = 1
        bin_c5 = compute_c5(binary_h.astype(float))
        binary_h, bin_c5 = hill_climb_binary(binary_h, bin_c5, start_time, budget_s)
        if bin_c5 + 1e-12 < best_c5:
            best_h = binary_h.astype(float)
            best_c5 = bin_c5

    # final sanity clip & sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_and_bounds(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
