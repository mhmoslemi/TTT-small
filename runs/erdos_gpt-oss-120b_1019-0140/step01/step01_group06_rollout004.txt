def hill_climb(h0, n_iters, rng, start_time, budget_s, T0=0.02):
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for it in range(n_iters):
        if time.time() - start_time > budget_s * 0.985:
            break
        # compute temperature: linear decay
        t = T0 * (1 - it / n_iters)
        # move
        i, j = rng.integers(0, len(h), size=2, endpoint=False)
        while i == j:
            j = rng.integers(0, len(h))
        # decide direction:
        if rng.random() < 0.5:
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / max(t, 1e-12)):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    # compute diff
    diff = target_sum - cur
    # if diff approx zero, return
    if abs(diff) < 1e-12:
        return h
    # allocate diff uniformly to free entries
    free = np.where((h > 0) & (h < 1))[0]
    if free.size > 0:
        delta = diff / free.size
        h[free] += delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
        # recompute diff and correct by adjusting first element if needed
        cur = np.sum(h)
        diff2 = target_sum - cur
        if abs(diff2) > 1e-9:
            # adjust element with most slack
            if diff2 > 0:
                slack = 1.0 - h
                i = np.argmax(slack)
                inc = min(diff2, slack[i])
                h[i] += inc
            else:
                slack = h
                i = np.argmax(slack)
                dec = min(-diff2, slack[i])
                h[i] -= dec
    else:
        # no free entries (unlikely): adjust all entries equally within bounds
        # fallback: distribute diff across all entries, then clip
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # final adjustment
        cur = np.sum(h)
        diff2 = target_sum - cur
        if abs(diff2) > 1e-9:
            i = 0
            h[i] = np.clip(h[i] + diff2, 0.0, 1.0)
    return h

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Overlap bound C5 for sampled function h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-12, max_iter=10):
    """Project onto {0<=h<=1, sum(h)=target_sum} by simple iterative clipping."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-9:
        if diff > 0:
            slack = 1.0 - h
            i = np.argmax(slack)
            inc = min(diff, slack[i])
            h[i] += inc
        else:
            slack = h
            i = np.argmax(slack)
            dec = min(-diff, slack[i])
            h[i] -= dec
    return h

def resample_h(base_h, n_target):
    """Linear interpolation of base_h to new length and projection to feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h = np.interp(x_new, x_base, base_h)
    h = project_to_feasible(h, n_target / 2.0)
    return h

def random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    return project_to_feasible(h, n / 2.0)

def init_fourier(base_h, n_target, rng, scale=0.12):
    """Fourier‑based perturbation of a parent function."""
    h = resample_h(base_h, n_target)
    H = np.fft.rfft(h)
    mask = rng.random(H.shape) < 0.35
    noise = (rng.normal(scale=scale, size=H.shape) +
             1j * rng.normal(scale=scale, size=H.shape))
    H_noisy = H.copy()
    H_noisy[mask] += noise[mask]
    h_noisy = np.fft.irfft(H_noisy, n=n_target)
    h_noisy = np.clip(h_noisy, 0.0, 1.0)
    return project_to_feasible(h_noisy, n_target / 2.0)

def hill_climb(h0, n_iters, rng, start_time, budget_s, T0=0.018):
    """Simulated‑annealing hill climbing using pairwise swaps preserving sum."""
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    n = len(h)
    for it in range(n_iters):
        if time.time() - start_time > budget_s * 0.985:
            break
        # linear temperature schedule
        t = T0 * (1.0 - it / n_iters)
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # decide direction of transfer
        if rng.random() < 0.5:
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / max(t, 1e-12)):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)

    # candidates – small to moderate size for speed vs flexibility
    candidates = [95, 190, 380, 760]
    # per‑size configuration
    config = {
        95:  {"restarts": 12, "iters": 2500},
        190: {"restarts": 10, "iters": 3000},
        380: {"restarts":  8, "iters": 3500},
        760: {"restarts":  5, "iters": 3000},
    }

    best_h = None
    best_c5 = float("inf")
    best_n = None

    for n in candidates:
        if time.time() - start_time > budget_s * 0.975:
            break
        target = n / 2.0
        cfg = config.get(n, {"restarts": 5, "iters": 2000})
        restarts = cfg["restarts"]
        iters = cfg["iters"]

        for r in range(restarts):
            if time.time() - start_time > budget_s * 0.975:
                break

            # Choose initialization method
            if r == 0 and base_h is not None:
                h0 = resample_h(base_h, n)
            elif r == 1 and base_h is not None:
                h0 = init_fourier(base_h, n, rng)
            else:
                h0 = random_feasible(n, rng)

            # Local stochastic search
            h_loc, c5_loc = hill_climb(h0, iters, rng, start_time, budget_s)

            # Fine‑tune with a short SLSQP run
            try:
                cons = {"type": "eq", "fun": lambda h, tar=target: np.sum(h) - tar}
                bnds = [(0.0, 1.0)] * n
                res = opt.minimize(
                    lambda h: compute_c5(h),
                    h_loc,
                    method="SLSQP",
                    bounds=bnds,
                    constraints=cons,
                    options={"maxiter": 400, "ftol": 1e-10, "disp": False},
                )
                h_opt = project_to_feasible(res.x, target)
                c5_opt = compute_c5(h_opt)
                if c5_opt < c5_loc:
                    h_loc, c5_loc = h_opt, c5_opt
            except Exception:
                pass

            # Update global best
            if c5_loc < best_c5:
                best_c5, best_h, best_n = c5_loc, h_loc.copy(), n

            # Early termination if we already beat the target
            if best_c5 <= 0.3805:
                break
        if best_c5 <= 0.3805:
            break

    # Fallback if nothing succeeded
    if best_h is None:
        n = 190
        best_n = n
        best_h = np.full(n, 0.5)
        best_h = project_to_feasible(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)

    # Ensure feasibility before returning
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
