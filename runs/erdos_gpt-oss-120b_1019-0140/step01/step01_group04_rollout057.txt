# ---------- helper utilities ----------
def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with exact sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Exact overlap bound for discrete h."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _max_corr_shift(v):
    """Return integer shift achieving the maximal overlap."""
    corr = np.correlate(v, 1.0 - v, mode="full")
    idx = np.argmax(corr)
    return idx - (len(v) - 1)

def _gradient_for_shift(v, shift):
    """Gradient of the max-overlap functional w.r.t. v."""
    n = len(v)
    grad = np.zeros(n, dtype=np.float64)
    if shift >= 0:
        # +1 contributions where the linear term is present
        grad[: n - shift] += 1.0
        # subtract v[i+shift]
        grad[: n - shift] -= v[shift:]
        # subtract v[i-shift]
        grad[shift:] -= v[: n - shift]
    else:
        s = -shift
        grad[s:] += 1.0
        grad[s:] -= v[: n - s]        # subtract v[i+shift] = v[i-s]
        grad[: n - s] -= v[s:]        # subtract v[i-shift] = v[i+s]
    return grad

def _gradient_step(v, step_factor):
    """Take a projected gradient descent step using the active shift."""
    n = len(v)
    shift = _max_corr_shift(v)
    grad = _gradient_for_shift(v, shift)
    dx = 2.0 / n
    step = step_factor * dx
    v_new = v - step * grad
    return _repair_vector(v_new, n)

    v = np.random.rand(n)

def _random_perturb(v, scale):

    n = len(v)

def _upsample_vector(v, n_target):
    """Linear interpolation to a finer grid followed by repair."""
    n_src = len(v)

    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)
