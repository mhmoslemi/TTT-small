def simulated_annealing(h_start, cmax_start, start_time, budget_s, rng, n):
    m = n // 2
    h = h_start.copy()
    auto = np.correlate(h, h, mode='full')
    cmax = cmax_start
    best_h = h.copy()
    best_cmax = cmax
    len_corr = 2 * n - 1
    # maintain ones list and zeros list
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    while True:
        # check time
        if time.time() - start_time > budget_s * 0.9:
            break
        # temperature schedule
        frac = (time.time() - start_time) / (budget_s * 0.9)
        T = max(1e-5, 1.0 * (1 - frac))
        # random swap
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # compute delta_i:
        ones_arr = np.array(ones, dtype=np.int64)
        diffs_i = np.concatenate([ones_arr - i, i - ones_arr])
        delta_i = np.bincount(diffs_i + (n-1), minlength=len_corr)
        delta_i[n-1] -= 1  # remove double count self
        # compute delta_j
        mask = ones_arr != i
        S_wo_i_arr = ones_arr[mask]
        if S_wo_i_arr.size == 0:
            # No other ones, skip? But m>1 so ignore.
            continue
        diffs_j = np.concatenate([S_wo_i_arr - j, j - S_wo_i_arr])
        delta_j = np.bincount(diffs_j + (n-1), minlength=len_corr)
        delta_j[n-1] += 1  # add self contribution
        # new auto
        new_auto = auto - delta_i + delta_j
        new_cmax = int(np.max(m - new_auto))
        # acceptance
        if new_cmax < cmax or rng.random() < math.exp(-(new_cmax - cmax) / T):
            # accept
            # update h
            h[i] = 0
            h[j] = 1
            # update ones and zeros
            ones[i_idx] = j
            zeros[j_idx] = i
            # update auto and cmax
            auto = new_auto
            cmax = new_cmax
            if cmax < best_cmax:
                best_cmax = cmax
                best_h = h.copy()
        # else reject, do nothing
    return best_h, best_cmax

def _binary_from_initial(vals):
    """Convert a (possibly continuous) initial vector to a binary step function."""
    n = len(vals)
    arr = np.asarray(vals, dtype=float)

    idx = np.argpartition(-arr, n // 2 - 1)[: n // 2]

    return bin_vec

def _cmax_of_h(h):
    """Return the integer max overlap count c_max = max_k sum h[i]*(1-h[i+k]) for binary h."""
    n = len(h)
    m = n // 2
    auto = np.correlate(h, h, mode='full')
    return int(np.max(m - auto))

def _simulated_annealing(h_start, cmax_start, start_time, budget_s, rng, n):
    """Simulated annealing on binary vectors, preserving exactly n/2 ones."""
    m = n // 2
    h = h_start.copy()
    auto = np.correlate(h, h, mode='full')
    cmax = cmax_start
    best_h = h.copy()
    best_cmax = cmax
    len_corr = 2 * n - 1
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))

    # anneal until ~85% of total budget is used (reserve time for polishing)
    while True:
        if time.time() - start_time > budget_s * 0.85:
            break
        # temperature linearly decays from 1.0 to 1e-4
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.85)
        T = max(1e-4, 1.0 * (1.0 - frac))

        # propose a swap between a 1 and a 0
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # contributions of the removed position i
        ones_arr = np.array(ones, dtype=np.int64)
        diffs_i = np.concatenate([ones_arr - i, i - ones_arr])
        delta_i = np.bincount(diffs_i + (n - 1), minlength=len_corr)
        delta_i[n - 1] -= 1                     # self counted twice

        # contributions of the added position j (with i already removed)
        mask = ones_arr != i                     # exclude i
        s_wo_i = ones_arr[mask]
        diffs_j = np.concatenate([s_wo_i - j, j - s_wo_i])
        delta_j = np.bincount(diffs_j + (n - 1), minlength=len_corr)
        delta_j[n - 1] += 1                      # add the new self term

        new_cmax = int(np.max(m - new_auto))

        if new_cmax < cmax or rng.random() < math.exp(-(new_cmax - cmax) / T):

            h[j] = 1
            ones[i_idx] = j
            zeros[j_idx] = i
            auto = new_auto
            cmax = new_cmax
            if cmax < best_cmax:
                best_cmax = cmax
                best_h = h.copy()
    return best_h, best_cmax

            # accept improving swap
            h[i] = 0
            h[j] = 1
            auto = new_auto
            cmax = new_cmax
            if new_cmax < best_cmax:
                best_cmax = new_cmax
                best_h = h.copy()
    return best_h, best_cmax

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise problem size and a starting binary vector -----
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = _binary_from_initial(initial_h_values)
    else:
        n = 80
        h0 = _random_binary(n, rng)

    m = n // 2
    # ----- quick random sampling to get a decent starting point -----
    best_h = h0.copy()
    best_cmax = _cmax_of_h(best_h)
    # try a few random candidates (up to 15% of time budget)
    while time.time() - start_time < budget_s * 0.15:
        cand = _random_binary(n, rng)
        cand_cmax = _cmax_of_h(cand)
        if cand_cmax < best_cmax:
            best_cmax = cand_cmax
            best_h = cand
        if best_cmax <= 15:          # target reached (bound 0.375)
            break

    # ----- simulated annealing phase -----
    best_h, best_cmax = _simulated_annealing(best_h, best_cmax,
                                            start_time, budget_s, rng, n)

    # ----- final deterministic hill‑climbing refinement -----
    best_h, best_cmax = _hill_climb_swaps(best_h, best_cmax,
                                          start_time, budget_s, rng, n)

    # Ensure exact half‑ones (should already hold)
    # Convert to float array for the required return format
    h_float = best_h.astype(float)
    c5_bound = best_cmax * (2.0 / n)

    return h_float, c5_bound, n
