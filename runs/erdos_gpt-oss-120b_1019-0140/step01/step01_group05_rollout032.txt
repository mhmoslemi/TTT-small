def project_to_box_and_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    # water-filling
    diff = target - v.sum()
    # tolerance for early exit
    if abs(diff) < 1e-10:
        return v
    # Initialize free mask
    free = ( (v > 0) & (v < 1) )
    # If all entries are saturated but diff not zero, we can't satisfy; but target = N/2 ensures feasible as long as N/2 <= N (Yes).
    # We need to add diff distributed equally among free entries; but some may go over/under bound; handle iteratively.
    while abs(diff) > 1e-12:
        free_indices = np.where(free)[0]
        if free_indices.size == 0:
            # No free entries, can't adjust further: clip again maybe
            break
        add = diff / free_indices.size
        # update
        v[free_indices] += add
        # clip again to bounds, and recompute diff
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
        # recompute free mask

    return v

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    # Compute diff
    diff = target - v.sum()
    # If diff approx zero, return
    if abs(diff) < 1e-12:
        return v
    # We'll do iterative approach:
    # while not within tolerance...
    # we maintain a mask of free entries
    free = (v > 0) & (v < 1)
    # If diff > 0, need to increase values; if diff < 0, need to decrease.
    # We'll add diff equally among free entries (stepwise)
    # But if there are no free entries, break.
    # Implementation loop:
    max_iter = 10 * len(v) # safe
    for _ in range(max_iter):
        free = (v > 0) & (v < 1)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        v[free] += delta
        # Clip again
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
    return v

def _gradient_descent(h_start, time_limit, rng):
    h = h_start.copy()
    N = h.size
    target = N / 2.0
    best_h = h.copy()
    best_val = _c5_value(h)
    start = time.time()
    iteration = 0
    # set some hyperparameters: initial step alpha
    alpha0 = 0.2  # maybe bigger step
    # maybe dynamic: alpha = alpha0 / (1 + iteration*0.001)
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        # compute current C5 and shift
        cur_val, shift = _c5_shift_and_val(h)
        # compute gradient
        grad = np.ones(N)
        s = shift
        if s >= 0:
            if s < N:
                grad[:N-s] -= h[s:]
                grad[s:] -= h[:N-s]
        else:
            s_abs = -s
            # note: s is negative
            # forward: indices s_abs..N-1 subtract h[:N-s_abs]
            grad[s_abs:] -= h[:N - s_abs]
            # backward: indices 0..N-s_abs-1 subtract h[s_abs:]
            grad[:N - s_abs] -= h[s_abs:]
        # step size decreasing
        alpha = alpha0 / (1.0 + iteration * 0.001)
        # take step
        h_new = h - alpha * grad
        # project
        h_new = _project(h_new, target)
        # evaluate new
        new_val = _c5_value(h_new)
        # accept if improve or probabilistically
        if new_val < cur_val:
            h = h_new
            cur_val = new_val
        else:
            # acceptance probability with temperature schedule
            # temperature decreases linearly to zero
            frac = max(0.0, 1.0 - (now - start) / time_limit)
            T = 0.01 * frac  # small temperature
            if T > 0 and rng.random() < math.exp(-(new_val - cur_val) / T):
                h = h_new
                cur_val = new_val
        # update best
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # occasional random perturbations
        if iteration % 500 == 0 and iteration > 0:
            # perturb a few entries
            noise = (rng.random(N) - 0.5) * 0.02

                h = h_pert
                cur_val = val_pert
                if val_pert < best_val:
                    best_val = val_pert
                    best_h = h.copy()
        iteration += 1
    return best_h, best_val

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce exact sum = len(v)/2."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        return np.clip(v, 0.0, 1.0)
    factor = target / s
    v *= factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        v = np.clip(v, 0.0, 1.0)
    return v

def _project(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto box [0,1] and linear equality sum(v)=target_sum."""
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - v.sum()
    if abs(diff) < 1e-12:
        return v
    # iterative water-filling
    for _ in range(10):
        free = (v > 0.0) & (v < 1.0)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
        diff = target_sum - v.sum()
        if abs(diff) < 1e-12:
            break
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_shift_and_val(h: np.ndarray):
    """Return (C5 value, shift) for current vector."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx] * dx), idx - (N - 1)

def _gradient_descent(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Gradient descent on max overlap with simple simulated‑annealing acceptance."""
    h = h_start.copy()
    N = h.size
    target = N / 2.0
    best_h = h.copy()
    best_val = _c5_value(h)
    start = time.time()
    it = 0
    alpha0 = 0.2
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        cur_val, shift = _c5_shift_and_val(h)
        # gradient of the max‑shift term (dx factor omitted)
        grad = np.ones(N)
        s = shift
        if s >= 0:
            if s < N:
                grad[:N - s] -= h[s:]
                grad[s:] -= h[:N - s]
        else:
            s_abs = -s
            grad[s_abs:] -= h[:N - s_abs]
            grad[:N - s_abs] -= h[s_abs:]
        # step size schedule
        alpha = alpha0 / (1.0 + it * 0.001)
        h_new = h - alpha * grad
        h_new = _project(h_new, target)
        new_val = _c5_value(h_new)
        # acceptance
        if new_val < cur_val:
            h = h_new
            cur_val = new_val
        else:
            frac = max(0.0, 1.0 - (now - start) / time_limit)
            T = 0.01 * frac
            if T > 0.0 and rng.random() < math.exp(-(new_val - cur_val) / T):
                h = h_new
                cur_val = new_val
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # occasional random perturbation to escape plateaus
        if it % 500 == 0 and it > 0:
            noise = (rng.random(N) - 0.5) * 0.02
            h_pert = _project(h + noise, target)
            pert_val = _c5_value(h_pert)
            if pert_val < cur_val:
                h = h_pert
                cur_val = pert_val
                if pert_val < best_val:
                    best_val = pert_val
                    best_h = h.copy()
        it += 1
    return best_h, best_val
