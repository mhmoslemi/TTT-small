import time
import numpy as np

# ---------- helper functions ----------
def compute_corr_full(h: np.ndarray) -> np.ndarray:
    """
    Full linear correlation of h with (1-h), multiplied by dx.
    Returns an array of length 2*len(h)-1 containing the correlation values.
    """
    n = h.shape[0]
    dx = 2.0 / n
    a = h.astype(np.float64)
    b = 1.0 - a
    m = 1 << ((2 * n - 1).bit_length())  # next power of two for zero‑padding
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    corr = np.fft.irfft(fa * np.conj(fb), m)[: 2 * n - 1]
    return corr * dx


def conflict_positions(h: np.ndarray, k: int) -> np.ndarray:
    """
    Return indices i such that h[i] == 1 and h[i+k] == 1 (both within bounds).
    The returned i correspond to the smaller index of the pair when k < 0.
    """
    n = h.shape[0]
    if k >= 0:
        left = h[: n - k]
        right = h[k:]
        pos = np.where((left == 1) & (right == 1))[0]
    else:
        kk = -k
        left = h[kk:]
        right = h[: n - kk]
        pos = np.where((left == 1) & (right == 1))[0] + kk
    return pos


def init_from_parent(rng: np.random.RandomState, n: int) -> np.ndarray | None:
    """
    If the global variable `initial_h_values` exists, create a binary seed
    of length n by up‑sampling and adjusting the number of ones to n//2.
    Returns None if the variable is unavailable.
    """
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        return None

    # binary version of the parent (threshold at 0.5)
    parent_bin = (parent >= 0.5).astype(np.float64)

    # repeat to reach length n (simple tiling)
    repeat = max(1, n // parent_bin.size)
    h = np.tile(parent_bin, repeat)[:n]

    # enforce exactly half ones
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[: cur - target]] = 0.0
    elif cur < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[: target - cur]] = 1.0
    return h


# ---------- main optimisation ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Simulated‑annealing / hill‑climbing search for a step function h on [0,2]
    that minimises the Erdős overlap constant C₅.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 1.0  # safety margin

    # problem size – larger values give more flexibility (still cheap)
    n = 500  # must be even so that sum(h)=n/2 gives integral 1
    target_ones = n // 2

    # initialise a binary vector, preferably using the parent solution
    h0 = init_from_parent(rng, n)
    if h0 is None:
        h0 = np.zeros(n, dtype=np.float64)
        h0[:target_ones] = 1.0
        rng.shuffle(h0)

    best_h = h0.copy()
    best_c5 = compute_corr_full(best_h).max()
    current_h = h0.copy()
    current_c5 = best_c5

    # annealing schedule parameters
    T0 = 5e-3
    Tmin = 1e-6

    while True:
        now = time.time()
        if now - start >= max_time:
            break

        # linear temperature decay
        elapsed = now - start
        progress = elapsed / max_time
        T = T0 * (1.0 - progress) + Tmin * progress

        # decide move type
        if rng.rand() < 0.7:
            # targeted move: address the worst lag
            corr = compute_corr_full(current_h)
            k_idx = int(np.argmax(corr))
            k_shift = k_idx - (n - 1)  # lag value (can be negative)

            conflict = conflict_positions(current_h, k_shift)
            if conflict.size == 0:
                # fall back to a random 1↔0 swap
                ones = np.where(current_h == 1)[0]
                zeros = np.where(current_h == 0)[0]
                if ones.size == 0 or zeros.size == 0:
                    continue
                i = rng.choice(ones)
                j = rng.choice(zeros)
                candidate = current_h.copy()
                candidate[i] = 0.0
                candidate[j] = 1.0
            else:
                src_i = rng.choice(conflict)
                # pick either of the two colliding positions
                src = src_i if rng.rand() < 0.5 else src_i + k_shift
                zeros = np.where(current_h == 0)[0]
                if zeros.size == 0:
                    continue
                dst = rng.choice(zeros)
                candidate = current_h.copy()
                candidate[src] = 0.0
                candidate[dst] = 1.0
        else:
            # simple random 1↔0 swap
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            candidate = current_h.copy()
            candidate[i] = 0.0
            candidate[j] = 1.0

        cand_c5 = compute_corr_full(candidate).max()
        delta = cand_c5 - current_c5

        if delta < 0.0 or rng.rand() < np.exp(-delta / (T + 1e-12)):
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()

    final_c5 = compute_corr_full(best_h).max()
    return (best_h, float(final_c5), n)
