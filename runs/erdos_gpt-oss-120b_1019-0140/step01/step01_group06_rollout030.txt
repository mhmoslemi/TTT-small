            # can increase values where x < 1

        else:
            mask = x > 1e-12
        free = mask.sum()

            break
        delta = diff / free

        # clip again

        g[Lneg:] = 1 - h[:n - Lneg]

        g[:n - Lneg] -= h[Lneg:]
    return g, max_idx

import time
import numpy as np

def compute_c5(h):
    """Compute the Erdős C5 value for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(x, target):
    """Project x onto the set {0≤x_i≤1, sum(x)=target} via bisection on a shift."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    if abs(cur - target) <= 1e-12:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        xp = np.clip(x + lam, 0.0, 1.0)
        sp = xp.sum()
        if sp > target:
            hi = lam
        else:
            lo = lam
        if abs(sp - target) < 1e-12:
            break
    return xp


def gradient_subgrad(h):
    """
    Subgradient of the max‑lag correlation at the current maximizing lag.
    Returns (g, max_idx) where g has the same shape as h.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    L = max_idx - (n - 1)          # shift corresponding to max correlation
    g = np.zeros_like(h)

    if L >= 0:
        # (1 - h[i+L]) part
        g[: n - L] = 1.0 - h[L:]
        # - h[i-L] part
        g[L:] -= h[: n - L]
    else:
        Lneg = -L
        # (1 - h[i+L]) = (1 - h[i-Lneg])
        g[Lneg:] = 1.0 - h[: n - Lneg]
        # - h[i-L] = -h[i+Lneg]
        g[: n - Lneg] -= h[Lneg:]
    return g, max_idx


def random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_to_feasible(v, n / 2.0)

                    if cur_val < best_val:
                        best_val = cur_val
                        best_h = cur_h.copy()
                        best_n = n

    # Fallback in case nothing was found
    if best_h is None:
        best_n = n0
        best_h = cosine_initial(best_n)
        best_val = compute_c5(best_h)
    else:
        best_h = project_to_feasible(best_h, best_n / 2.0)
        best_val = compute_c5(best_h)

    return best_h, best_val, best_n
