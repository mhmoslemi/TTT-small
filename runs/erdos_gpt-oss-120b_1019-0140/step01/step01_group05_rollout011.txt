import numpy as np, time, math
from scipy.optimize import differential_evolution

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        return np.clip(v, 0.0, 1.0)
    factor = target / s
    v *= factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _de_optimize(N, time_limit, rng, h_start=None):
    start = time.time()

    # Objective that repairs before evaluation
    def obj(v):
        h = _repair_vector(np.asarray(v, dtype=np.float64))
        return _c5_value(h)
    # Callback to respect time limit
    def cb(xk, convergence):
        if time.time() - start >= time_limit:
            return True
    # If we have a starting point, we can embed it into initial population via 'init' argument.
    # SciPy accepts 'init' being an array of shape (popsize, N). We'll just use default.
    # Use seed as int for reproducibility
    de_res = differential_evolution(
        obj,

        disp=False,

    cur_score = _c5_value(h_cur)

    T0 = 0.2                # initial temperature

        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = 1.0 - elapsed / time_limit
        T = T0 * frac
        max_eps = max_eps0 * frac + 0.001
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]

        if eps == 0.0:

import numpy as np
import time
import math
from scipy.optimize import differential_evolution

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        return np.clip(v, 0.0, 1.0)
    factor = target / s
    v *= factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discretised h."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _de_optimize(N: int, time_limit: float, rng: np.random.Generator, h_start: np.ndarray = None):
    """Differential evolution with time‑bounded callback."""
    start = time.time()
    bounds = [(0.0, 1.0)] * N

    def obj(v):
        h = _repair_vector(np.asarray(v, dtype=np.float64))
        return _c5_value(h)

    def cb(xk, convergence):
        if time.time() - start >= time_limit:
            return True

    # initialise population with h_start if supplied
    if h_start is not None:
        popsize = 15
        pop = rng.random((popsize, N))
        pop[0] = np.clip(h_start, 0.0, 1.0)
        init = pop
    else:
        init = "random"

    res = differential_evolution(
        obj,
        bounds,
        seed=int(rng.integers(0, 2**32 - 1)),
        maxiter=2000,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        polish=False,
        callback=cb,
        disp=False,
        init=init,
    )
    h_best = _repair_vector(res.x.copy())
    best_score = _c5_value(h_best)
    return h_best, best_score
