import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to satisfy the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute uniformly and clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _gradient_step(h: np.ndarray, target: float, alpha: float, lr: float, rng: np.random.Generator) -> np.ndarray:
    """One projected gradient descent step on a smooth approximation of max C5."""
    n = h.shape[0]
    # g_k = sum_i h_i * (1 - h_{i+k})
    g = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights for a smooth max
    max_g = np.max(g)
    exp_vals = np.exp(alpha * (g - max_g))
    w = exp_vals / np.sum(exp_vals)

    # Gradient of weighted sum of g_k
    grad = np.full(n, 1.0)                     # derivative of sum_i h_i part
    shifts = np.arange(-(n - 1), n)            # corresponding lags
    for idx, shift in enumerate(shifts):
        weight = w[idx]
        if shift == 0:
            grad -= 2.0 * weight * h
        elif shift > 0:
            # forward contribution
            grad[: n - shift] -= weight * h[shift:]
            # backward contribution
            grad[shift:] -= weight * h[: n - shift]
        else:
            s = -shift
            grad[s:] -= weight * h[: n - s]
            grad[: n - s] -= weight * h[s:]

    # Gradient descent step
    new_h = h - lr * grad
    np.clip(new_h, 0.0, 1.0, out=new_h)
    new_h = _repair_sum(new_h, target)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise feasible candidate ---------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float)  # provided construction
        n = int(h0.shape[0])
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _repair_sum(h0, target)
        current = h0.copy()
    except NameError:
        # No pre‑supplied construction; start with a random feasible vector.
        n = 200
        target = n / 2.0
        current = rng.random(n)
        current = _repair_sum(current, target)

    # Ensure feasibility
    current = _repair_sum(current, target)

    best_h = current.copy()
    best_c5 = _compute_c5(best_h)

    total_budget = float(budget_s)
    # Phase allocation
    phase1_budget = total_budget * 0.6   # gradient descent
    phase2_budget = total_budget - phase1_budget

    # ----- Phase 1: smooth gradient descent -------------------------------
    alpha = 1.0          # smoothness parameter
    lr = 0.5             # learning rate
    iter_cnt = 0
    while time.time() - start < phase1_budget:
        current = _gradient_step(current, target, alpha, lr, rng)
        if iter_cnt % 10 == 0:
            cur_c5 = _compute_c5(current)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = current.copy()
        iter_cnt += 1
        if iter_cnt % 100 == 0:
            alpha = min(alpha * 1.1, 200.0)   # increase focus on the max
            lr = max(lr * 0.9, 1e-4)          # shrink step size

    # ----- Phase 2: Simulated annealing ----------------------------------
    current = best_h.copy()
    current_c5 = best_c5

    max_step = 1.0
    min_step = 0.001
    max_temp = 0.1
    min_temp = 1e-6

    while time.time() - start < total_budget:
        now = time.time()
        elapsed = now - start
        frac = elapsed / total_budget
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        hi, hj = current[i], current[j]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        if delta_min >= delta_max:
            continue

        # Occasionally attempt a full swap (large delta)
        if rng.random() < 0.1:
            delta = delta_max
        else:
            low = max(delta_min, -step)
            high = min(delta_max, step)
            if low >= high:
                continue
            delta = rng.random() * (high - low) + low

        # apply move
        current[i] = hi + delta
        current[j] = hj - delta

        new_c5 = _compute_c5(current)
        delta_obj = new_c5 - current_c5

        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / temp):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current.copy()
        else:
            # revert
            current[i] = hi
            current[j] = hj

    # ----- Final random local refinements --------------------------------
    for _ in range(200):
        cand = best_h.copy()
        i, j = rng.integers(0, n, size=2)
        if i != j:
            cand[i], cand[j] = cand[j], cand[i]          # simple swap
        # small jitter
        cand += (rng.random(n) - 0.5) * 0.01
        np.clip(cand, 0.0, 1.0, out=cand)
        cand = _repair_sum(cand, target)
        c5cand = _compute_c5(cand)
        if c5cand < best_c5:
            best_c5 = c5cand
            best_h = cand.copy()

    # ----- final repair and evaluation -----------------------------------
    best_h = _repair_sum(best_h, target)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
