def hill_climb_exhaustive(h_init, start_time, budget_s):
    h = h_init.copy()
    cur_val = compute_c5(h)
    n = len(h)
    improved = True
    while improved:
        if time.time() - start_time > budget_s - 0.1:
            break
        improved = False
        # list of ones, zeros
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        best_val = cur_val
        best_i = None
        best_j = None
        # Evaluate each possible swap
        for i in ones:
            for j in zeros:
                # swap
                h[i] = 0
                h[j] = 1
                cand_val = compute_c5(h)
                # revert
                h[i] = 1
                h[j] = 0
                if cand_val + 1e-12 < best_val:
                    best_val = cand_val
                    best_i = i
                    best_j = j
        if best_i is not None:
            # Apply best swap
            h[best_i] = 0
            h[best_j] = 1
            cur_val = best_val
            improved = True
    return h, cur_val

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate does the required zero‑padding automatically.
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def hill_climb_exhaustive(h_init: np.ndarray,
                          start_time: float,
                          budget_s: float) -> tuple[np.ndarray, float]:
    """
    Exhaustive best‑swap hill climbing for a binary vector.
    Returns the locally optimal vector and its C5 value.
    """
    h = h_init.copy()
    cur_val = compute_c5(h)
    n = h.shape[0]

    # main loop – stop when no single swap improves the objective
    while True:
        if time.time() - start_time > budget_s - 0.2:
            break

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)

        best_val = cur_val
        best_i = -1
        best_j = -1

        # try every possible 1↔0 exchange
        for i in ones:
            for j in zeros:
                # perform provisional swap
                h[i] = 0
                h[j] = 1
                new_val = compute_c5(h)
                # revert immediately
                h[i] = 1
                h[j] = 0

                if new_val + 1e-12 < best_val:
                    best_val = new_val
                    best_i, best_j = i, j

        if best_i == -1:               # no improving swap found
            break

        # apply the best swap discovered
        h[best_i] = 0
        h[best_j] = 1
        cur_val = best_val

    return h.copy(), cur_val

    # ------------------------------------------------------------------

    if 'initial_h_values' in globals():

        # take the half with largest values

        base_h = np.zeros(n, dtype=int)
        base_h[idx] = 1

        n = 80
        half = n // 2

        base_h[idx] = 1

    # ------------------------------------------------------------------
    # First local optimisation from the (possibly) informed start
    # ------------------------------------------------------------------
    best_h, best_c5 = hill_climb_exhaustive(base_h, start_time, budget_s)

    target = 0.3808
    if best_c5 <= target:
