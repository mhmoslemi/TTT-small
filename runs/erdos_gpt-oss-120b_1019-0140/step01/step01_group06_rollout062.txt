def project_feasible(h, target_sum, tol=1e-12):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # compute sum_diff
    diff = target_sum - np.sum(h)
    # While diff not close to zero:
    # If diff > 0: need to increase some values below 1.
    # If diff < 0: need to decrease some values above 0.
    # We can compute adjustable amount for each index: for diff>0, max increase = 1 - h_i; for diff<0, max decrease = h_i.
    # Use water-filling: sort by amount of slack, then allocate equally among those with remaining slack.
    while abs(diff) > tol:
        if diff > 0:
            # indices that can increase
            slack = 1.0 - h
            mask = slack > 1e-12
        else:
            slack = -h  # negative values represent amount we can decrease (since diff negative)
            mask = h > 1e-12
        if not np.any(mask):
            break
        # total slack available
        total_slack = np.sum(np.abs(slack[mask]))

def project_feasible(h, target_sum, tol=1e-12, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # Identify indices not at bounds:
        free = ((h > 0.0) & (h < 1.0))
        if not np.any(free):
            # if at boundaries cannot adjust further.
            break
        # capacity available for each free index: if diff>0: can increase up to 1-h[i]; if diff<0 can decrease down to -h[i].
        if diff > 0:
            cap = 1.0 - h[free]
        else:
            cap = -h[free]  # negative
        # sum of capacities
        total_cap = np.sum(cap)
        if diff > 0 and total_cap < diff - 1e-12:
            # Not enough capacity, set all free to 1.
            h[free] = 1.0
            continue
        if diff < 0 and -total_cap < diff + 1e-12:  # diff negative, need -total_cap >= -diff?
            h[free] = 0.0
            continue
        # allocate proportionally: delta_i = cap_i * diff/total_cap
        if diff > 0:
            delta = cap * (diff / total_cap)
            h[free] += delta
        else:
            # diff negative, cap negative values -h. Use same formula.
            delta = cap * (diff / total_cap)  # both negative
            h[free] += delta
        # clip to [0,1] because due to floating errors may overshoot
        h = np.clip(h, 0.0, 1.0)
    # final small leftover adjustment: if diff still remains within tolerance but might be due to rounding, adjust one index
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-9:
        # find index with max slack in direction
        if diff > 0:
            # increase where not at 1
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h)  # smallest value, can reduce
            h[idx] = max(0.0, h[idx] + diff)
    # final clip and return
    return np.clip(h, 0.0, 1.0)

    return np.max(corr) * (2.0 / n)


def project_feasible(h, target_sum, tol=1e-12, max_iter=200):
    """Project onto [0,1]^n ∩ {sum = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
