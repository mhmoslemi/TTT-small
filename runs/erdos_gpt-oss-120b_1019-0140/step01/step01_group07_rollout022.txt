def compute_c5(h):
    n = h.shape[0]
    dx = 2.0/n
    # compute convolution of h and (1-h)
    # use FFT: conv = ifft(fft(h,pad) * fft(1-h,pad)).real
    # pad length = next power of two >= 2*n
    N = 1
    while N < 2*n:
        N <<= 1
    # forward FFT
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(H * G, n=N)[:2*n-1]
    max_val = np.max(conv) * dx
    return max_val

import numpy as np
import time
import math
import random

def compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    N = 1 << ((2*n - 1).bit_length())
    # compute rfft
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(H * G, n=N)[:2*n - 1]
    return np.max(conv) * dx

def _project_binary_to_target(h, target):
    # h is binary int 0/1, but we maybe need to adjust to target sum.
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(h == 0)
        if zeros.size > 0:
            to_flip = np.random.choice(zeros, size=diff, replace=False)
            h[to_flip] = 1
    elif diff < 0:
        ones = np.flatnonzero(h == 1)
        if ones.size > 0:
            to_flip = np.random.choice(ones, size=-diff, replace=False)
            h[to_flip] = 0
    return h

import numpy as np
import time
import math
import random

def _compute_c5_fft(h):
    """Maximum overlap integral for binary (or real) h using FFT."""
    n = h.shape[0]
    dx = 2.0 / n
    # pad to next power of two for speed
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(H * G, n=N)[: 2 * n - 1]
    return np.max(conv) * dx


def _project_binary_to_target(h, target):
    """Adjust a binary vector to have sum == target (exactly)."""
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(h == 0)
        if zeros.size:
            to_flip = np.random.choice(zeros, size=diff, replace=False)
            h[to_flip] = 1
    elif diff < 0:
        ones = np.flatnonzero(h == 1)
        if ones.size:
            to_flip = np.random.choice(ones, size=-diff, replace=False)
            h[to_flip] = 0
    return h


def _binary_simulated_annealing(h_init, time_limit):
    """Simple SA on a binary vector with fixed number of ones."""
    start = time.time()
    h = h_init.copy()
    n = h.size
    target = n // 2
    if int(h.sum()) != target:
        h = _project_binary_to_target(h, target)

    cur_score = _compute_c5_fft(h.astype(np.float64))
    best_h = h.copy()
    best_score = cur_score

    ones_list = list(np.flatnonzero(h == 1))
    zeros_list = list(np.flatnonzero(h == 0))

    T0 = 0.02  # initial temperature

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        t = elapsed / time_limit
        T = T0 * (1 - t)  # linear cooling

        i1 = random.choice(ones_list)   # will become 0
        i0 = random.choice(zeros_list)  # will become 1

        # perform swap
        h[i1] = 0
        h[i0] = 1

        new_score = _compute_c5_fft(h.astype(np.float64))
        delta = new_score - cur_score

        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 1e-12:
                prob = math.exp(-delta / T)
                if random.random() < prob:
                    accept = True

        if accept:
            cur_score = new_score
            # move indices between lists
            ones_list.remove(i1)
            zeros_list.append(i1)

            zeros_list.remove(i0)
            ones_list.append(i0)

            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            if best_score < 0.3808:
                break
        else:
            # revert swap
            h[i1] = 1
            h[i0] = 0

    return best_h.astype(np.float64), best_score


def _upsample_repeat(h, factor):
    """Upsample by repeating each entry `factor` times."""
    return np.repeat(h, factor).astype(np.int8)
