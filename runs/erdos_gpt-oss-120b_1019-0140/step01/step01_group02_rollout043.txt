def _gradient_move(h, step, max_shift):
    n = h.shape[0]
    idxs = np.arange(n)
    # compute s
    plus = (idxs + max_shift) % n
    minus = (idxs - max_shift) % n
    s = h[plus] + h[minus]
    # mask up and down
    mask_up = h < 1.0 - 1e-12
    mask_down = h > 1e-12
    # if no up or down, return None
    if not mask_up.any() or not mask_down.any():
        return None
    # compute indices with max s among mask_up, min s among mask_down
    i_max = np.argmax(np.where(mask_up, s, -np.inf))
    i_min = np.argmin(np.where(mask_down, s, np.inf))
    if i_max == i_min:
        # need distinct; find second best maybe
        # try to find next best up
        up_candidates = np.where(mask_up)[0]
        # remove i_max
        up_candidates = up_candidates[up_candidates != i_max]
        if up_candidates.size == 0:
            return None
        # pick one with max s from these
        i_max = up_candidates[np.argmax(s[up_candidates])]
        # similarly for down
        down_candidates = np.where(mask_down)[0]
        down_candidates = down_candidates[down_candidates != i_min]
        if down_candidates.size == 0:
            return None
        i_min = down_candidates[np.argmin(s[down_candidates])]
    # compute delta capacity
    delta_up = min(step, 1.0 - h[i_max])
    delta_down = min(step, h[i_min])
    delta = min(delta_up, delta_down)
    if delta <= 1e-12:
        return None
    return i_max, i_min, delta

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _rand_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a feasible vector with sum n/2 (binary except one 0.5 if n odd)."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    ones = rng.choice(n, size=half, replace=False)
    h[ones] = 1.0
    if n % 2:
        remain = np.setdiff1d(np.arange(n), ones)
        h[rng.choice(remain)] = 0.5
    return h

def _select_move_random(h: np.ndarray, step: float, rng: np.random.Generator):
    """Propose a random feasible pair move."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None
    hi, hj = h[i], h[j]
    delta_min = max(-hi, -(1.0 - hj))
    delta_max = min(1.0 - hi, hj)
    low = max(delta_min, -step)
    high = min(delta_max, step)
    if low >= high:
        return None
    delta = rng.random() * (high - low) + low
    return i, j, delta

def _select_move_gradient(h: np.ndarray, shift: int, step: float):
    """Propose a gradient‐driven move for the current worst shift."""
    n = h.shape[0]
    idxs = np.arange(n)
    s = np.zeros(n, dtype=h.dtype)

    # contribution from forward direction (i+shift)
    fwd_mask = (idxs + shift >= 0) & (idxs + shift < n)
    s[fwd_mask] += h[idxs[fwd_mask] + shift]

    # contribution from backward direction (i-shift)
    bwd_mask = (idxs - shift >= 0) & (idxs - shift < n)
    s[bwd_mask] += h[idxs[bwd_mask] - shift]

    mask_up = h < (1.0 - 1e-12)
    mask_down = h > 1e-12
    if not mask_up.any() or not mask_down.any():
        return None

    i_max = np.argmax(np.where(mask_up, s, -np.inf))
    i_min = np.argmin(np.where(mask_down, s, np.inf))

    if i_max == i_min:
        up_cands = np.where(mask_up)[0]
        up_cands = up_cands[up_cands != i_max]
        if up_cands.size == 0:
            return None
        i_max = up_cands[np.argmax(s[up_cands])]
        down_cands = np.where(mask_down)[0]
        down_cands = down_cands[down_cands != i_min]
        if down_cands.size == 0:
            return None
        i_min = down_cands[np.argmin(s[down_cands])]

    delta_up = min(step, 1.0 - h[i_max])
    delta_down = min(step, h[i_min])
    delta = min(delta_up, delta_down)
    if delta <= 1e-12:
        return None
    return i_max, i_min, delta

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that minimizes the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise a feasible candidate ---------------------------------
    try:
        base = np.asarray(initial_h_values, dtype=float)
        base_n = base.shape[0]
        n = 300                                     # target resolution
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        x_old = np.linspace(0.0, 2.0, base_n, endpoint=False)
        h0 = np.interp(x_new, x_old, base)
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        current = _repair_sum(h0, target)
    except NameError:
        n = 300
        target = n / 2.0
        current = _rand_binary_vector(n, rng)

    # Ensure feasibility
    current = _repair_sum(current, target)

    dx = 2.0 / n
    cur_c5 = _compute_c5(current)
    best_h = current.copy()
    best_c5 = cur_c5

    # Annealing / schedule parameters
    max_step = 0.5
    min_step = 0.001
    max_temp = 0.01
    min_temp = 1e-6
    max_rand = 0.2
    min_rand = 0.01

    iteration = 0
    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        iteration += 1
        frac = (now - start) / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac
        p_rand = max_rand * (1.0 - frac) + min_rand * frac

        # correlation and worst shift
        raw_corr = np.correlate(current, 1.0 - current, mode="full")
        max_idx = int(np.argmax(raw_corr))
        max_shift = max_idx - (n - 1)
        cur_c5 = raw_corr[max_idx] * dx

        # decide move type
        move = None
        if max_shift == 0 or rng.random() < p_rand:
            move = _select_move_random(current, step, rng)
        else:
            move = _select_move_gradient(current, max_shift, step)

        if move is None:
            continue

        i, j, delta = move
        old_i, old_j = current[i], current[j]
        current[i] = old_i + delta
        current[j] = old_j - delta

        # evaluate new candidate
        new_raw = np.correlate(current, 1.0 - current, mode="full")
        new_c5 = np.max(new_raw) * dx
        delta_obj = new_c5 - cur_c5

        # Metropolis acceptance
        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / temp):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current.copy()
        else:
            # revert
            current[i] = old_i
            current[j] = old_j

        # occasional numeric repair
        if iteration % 1000 == 0:
            total = current.sum()
            if abs(total - target) > 1e-12:
                current = _repair_sum(current, target)

    # final repair and evaluation
    best_h = _repair_sum(best_h, target)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
