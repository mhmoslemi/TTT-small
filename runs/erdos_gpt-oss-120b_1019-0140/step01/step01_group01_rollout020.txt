def compute_c5(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fx = np.fft.rfft(h, n=L)
    fy = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:2*n-1]
    return float(conv.max() * 2.0 / n)
```

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def adjust_sum_to_target(h, target, rng):
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # distribute diff across random indices while staying within bounds
    # We'll attempt up to max_attempts adjusting single indices.
    max_attempts = 500
    attempts = 0
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # re-balance using simple method: compute new diff and correct
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            # distribute remaining diff2 using a few random indices
            for _ in range(10):
                if abs(diff2) < 1e-12:
                    break
                idx = rng.randint(h.size)
                if diff2 > 0:
                    inc = min(diff2, 1.0 - h[idx])
                    h[idx] += inc
                    diff2 -= inc
                else:
                    dec = min(-diff2, h[idx])
                    h[idx] -= dec
                    diff2 += dec
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # decide number of points
    base_n = 200
    # use initial_h_values if exists and upsample if smaller
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n0 = init_h.size
        n_points = max(base_n, n0*3)  # upsample to at least 3x size maybe
        if n_points > 1000:
            n_points = 1000
        # upsample via linear interpolation
        if n0 != n_points:
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n_points, endpoint=False)
            h = np.interp(x_new, x_old, init_h)
        else:
            h = init_h.copy()
    except Exception:
        n_points = base_n
        target_sum = n_points / 2.0
        h = rng.rand(n_points)
        h = h / h.sum() * target_sum
        h = adjust_sum_to_target(h, target_sum, rng)

    # ensure constraints
    target_sum = n_points / 2.0
    h = np.clip(h, 0.0, 1.0, out=h)
    h = adjust_sum_to_target(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # annealing parameters
    T0 = 0.01  # initial temperature
    Tf = 1e-6
    step0 = 0.3
    step_end = 1e-4

    # timers for restart
    last_improve_time = start

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        T = T0 * (1.0 - progress) + Tf * progress
        step = step0 * (1.0 - progress) + step_end * progress

        # Move selection
        move_type = rng.rand()
        if move_type < 0.4:
            # pairwise exchange (as before)
            i, j = rng.randint(n_points, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            if delta < low:
                delta = low
            elif delta > high:
                delta = high
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        elif move_type < 0.7:
            # random multi-index perturbation
            k = rng.randint(3, 7)  # number of indices to perturb
            idxs = rng.choice(n_points, size=k, replace=False)
            candidate = current_h.copy()
            # add Gaussian perturbation scaled by step
            perturb = rng.randn(k) * step
            candidate[idxs] += perturb
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)

        else:
            # gradient move
            corr_full = np.correlate(current_h, 1.0 - current_h, mode='full')
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n_points - 1)

            # compute subgradient
            g = np.zeros_like(current_h)
            idx = np.arange(n_points)
            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n_points)
            g[valid_plus] += (1.0 - current_h[plus[valid_plus]])

            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n_points)
            g[valid_minus] -= current_h[minus[valid_minus]]

            # project to sum-preserving direction (subtract mean)
            g -= g.mean()
            candidate = current_h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)

        cand_c5 = compute_c5(candidate)

        # acceptance
        if cand_c5 < current_c5:
            accept = True
        else:
            # Metropolis
            if T > 0:
                prob = np.exp(-(cand_c5 - current_c5) / T)
            else:
                prob = 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                last_improve_time = now

        # restart if stagnation > 30 seconds
        if now - last_improve_time > 30.0:
            # reheat: perturb best solution
            current_h = best_h.copy()
            # add small noise
            noise = rng.randn(n_points) * step * 5.0
            current_h += noise
            np.clip(current_h, 0.0, 1.0, out=current_h)
            current_h = adjust_sum_to_target(current_h, target_sum, rng)
            current_c5 = compute_c5(current_h)
            last_improve_time = now

    # final best
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Compute the Erdős C5 value for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust `h` in-place so that its sum equals `target` while keeping 0 ≤ h_i ≤ 1.
    Small residual errors are distributed uniformly and clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h

    # Try to fix the deficit/surplus using random single-index updates
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1

    if abs(diff) > 1e-12:
        # Distribute remaining error uniformly and fix any bound violations
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # If still off, make a few targeted corrections
        diff2 = target - h.sum()
        for _ in range(10):
            if abs(diff2) < 1e-12:
                break
            idx = rng.randint(h.size)
            if diff2 > 0.0:
                inc = min(diff2, 1.0 - h[idx])
                h[idx] += inc
                diff2 -= inc
            else:
                dec = min(-diff2, h[idx])
                h[idx] -= dec
                diff2 += dec
    return h

    """

    max_time = budget_s - 0.5  # safety margin

    # Annealing / step schedules

    last_improve = start

    while True:
        now = time.time()
        if now - start >= max_time:
            break

        progress = (now - start) / max_time
        T = T0 * (1.0 - progress) + Tf * progress
        step = step0 * (1.0 - progress) + step_end * progress

        move_type = rng.rand()

        # ----- pairwise exchange -----
        if move_type < 0.4:
            i, j = rng.randint(n_points, size=2)
            if i == j:

            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)

            current_h = adjust_sum_to_target(current_h, target_sum, rng)
            current_c5 = compute_c5(current_h)
            last_improve = now

    # ----- final polishing (gradient only) -----
    while time.time() - start < max_time:
        corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_idx = int(np.argmax(corr_full))
        k_shift = k_idx - (n_points - 1)

        g = np.zeros_like(best_h)
        idx = np.arange(n_points)
        plus = idx + k_shift
        valid_plus = (plus >= 0) & (plus < n_points)
        g[valid_plus] += (1.0 - best_h[plus[valid_plus]])
        minus = idx - k_shift
        valid_minus = (minus >= 0) & (minus < n_points)
        g[valid_minus] -= best_h[minus[valid_minus]]
        g -= g.mean()

        step_local = step_end * 5.0
        cand = best_h - step_local * g
        np.clip(cand, 0.0, 1.0, out=cand)
        cand = adjust_sum_to_target(cand, target_sum, rng)
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()

    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
