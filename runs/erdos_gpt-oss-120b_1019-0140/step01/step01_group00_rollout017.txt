    return b

    return b

    # local hill climbing with targeted swaps
    cur_h = start_h.copy()

import time
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated by samples."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def binary_from_float(arr):
    """Convert a float array to a binary vector with exactly half ones."""
    n = arr.shape[0]
    half = n // 2
    idx = np.argpartition(-arr, half)[:half]
    b = np.zeros(n, dtype=np.int8)
    b[idx] = 1
    return b

def random_binary(n, half, rng):
    """Generate a random binary vector with exactly `half` ones."""
    b = np.zeros(n, dtype=np.int8)
    b[rng.choice(n, half, replace=False)] = 1
    return b

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    # keep a small safety margin so the harness can still retrieve the result
    deadline = start_time + budget_s - 0.5
    rng = np.random.default_rng(seed)

    # ----- initialise problem size and starting vector -----
    if "initial_h_values" in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        half = n // 2
        cur_h = binary_from_float(init_arr)
    else:
        n = 80
        half = n // 2
        cur_h = random_binary(n, half, rng)

    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ----- simulated annealing on the binary space -----
    T0 = 1e-3  # initial temperature
    while time.time() < deadline:
        # linear cooling schedule
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 0.5)
        T = max(0.0, T0 * (1.0 - progress))

        # propose a swap of a 1 with a 0
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # execute swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5

        # acceptance test
        if delta < -1e-12 or (T > 0.0 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 + 1e-12 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                if best_c5 < 0.3808:
                    break
        else:
            # revert the move
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        # occasional full random restart to diversify search
        if rng.random() < 5e-4:
            cur_h = random_binary(n, half, rng)
            cur_c5 = compute_c5(cur_h)

    # ----- final greedy refinement starting from the best found -----
    while time.time() < deadline:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best_h[i], best_h[j] = best_h[j], best_h[i]
        cand_c5 = compute_c5(best_h)
        if cand_c5 + 1e-12 < best_c5:
            best_c5 = cand_c5
            # keep the improvement
            continue
        else:
            # revert
            best_h[i], best_h[j] = best_h[j], best_h[i]

    diff = best_h_float.sum() - half

        best_h_float = np.clip(best_h_float, 0.0, 1.0)

    return best_h_float, best_c5, n
