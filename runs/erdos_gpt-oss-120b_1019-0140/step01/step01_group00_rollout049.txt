def compute_c5(h):

    corr = np.correlate(h, 1 - h, mode='full')

import time, math
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """Return a binary step function h on [0,2] minimizing the overlap bound C5.

    Returns
    -------
    h_values : np.ndarray
        Array of length n with entries in [0,1] and sum(h)*dx == 1.
    c5_bound : float
        The computed max overlap integral for h.
    n_points : int
        Length of the discretisation (len(h_values)).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- problem size and initial binary vector -----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        # convert to a binary vector by taking the largest n/2 entries
        idx = np.argsort(-h0)[: n // 2]
        cur_best = np.zeros(n, dtype=float)
        cur_best[idx] = 1.0
    else:
        n = 80
        cur_best = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        cur_best[idx] = 1.0

    dx = 2.0 / n
    target_sum = n / 2.0

    # ----- helper to evaluate C5 -----
    def compute_c5(vec):
        """Maximum overlap integral for a binary (or [0,1]) vector."""
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        return np.max(corr) * dx

    best_h = cur_best.copy()
    best_c5 = compute_c5(best_h)

    # ----- simulated annealing on the binary space -----
    sa_time = budget_s * 0.78          # allocate most of the budget to SA
    T0 = 0.05                           # initial temperature
    max_iter_per_run = 20000

    while time.time() - start_time < sa_time:
        # occasional random restart to diversify search
        if rng.random() < 0.3:
            h = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1.0
            cur_c5 = compute_c5(h)
        else:
            h = best_h.copy()
            cur_c5 = best_c5

        for it in range(max_iter_per_run):
            if time.time() - start_time >= sa_time:
                break
            # linear cooling schedule
            T = T0 * (1.0 - it / max_iter_per_run) + 1e-9
            # pick a random 1 and a random 0
            ones = np.flatnonzero(h == 1.0)
            zeros = np.flatnonzero(h == 0.0)
            if ones.size == 0 or zeros.size == 0:

            h[j] = 1.0

                h[i] = 1.0
                h[j] = 0.0
