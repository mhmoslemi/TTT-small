    seq = np.zeros(p, dtype=float)
    for i in range(p):
        # compute Legendre symbol (i/p)

    current_c5 = _compute_c5(current)
    best_h = current.copy()
    best_c5 = current_c5

        elapsed = now - start
        if elapsed >= budget_s:
            break

        frac = elapsed / budget_s
        step = max_step * (1 - frac) + min_step * frac
        temp = max_temp * (1 - frac) + min_temp * frac

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-15)[0]
    else:
        idx = np.where(h > 1e-15)[0]
    if idx.size == 0:
        # distribute evenly as a fallback
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-15:
        return _repair_sum(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- choose problem size ------------------------------------------------
    n = 400                     # even size, < 1000 samples
    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- initialise a feasible candidate ------------------------------------
    try:
        # use the supplied construction as a seed pattern
        init0 = np.asarray(initial_h_values, dtype=float)
        repeats = (n + init0.size - 1) // init0.size
        h = np.tile(init0, repeats)[:n]
        h = np.clip(h, 0.0, 1.0)
        h = _repair_sum(h, target_sum)
    except NameError:
        # no parent construction – start from a random half‑ones pattern
        h = np.zeros(n, dtype=float)
        h[:n // 2] = 1.0
        rng.shuffle(h)

    # Ensure exact feasibility (tiny numeric drift)
    h = _repair_sum(h, target_sum)

    current = h.copy()
    current_c5 = _compute_c5(current)

    best_h = current.copy()
    best_c5 = current_c5

    # ----- annealing schedule -------------------------------------------------
    max_step = 1.0     # allow full swaps early on
    min_step = 0.001
    max_temp = 0.05
    min_temp = 1e-6

    # ----- main optimisation loop --------------------------------------------
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s:
            break

        # progress fraction for schedules
        frac = elapsed / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # compute correlation to locate the worst shift
        corr_full = np.correlate(current, 1.0 - current, mode="full")
        max_idx = int(np.argmax(corr_full))
        max_corr = float(corr_full[max_idx])
        current_c5 = max_corr * dx
        shift = max_idx - (n - 1)

        # build list of indices that actually contribute to the worst shift
        if shift >= 0:
            i_start = 0
            i_end = n - shift
        else:
            i_start = -shift
            i_end = n
        i_range = np.arange(i_start, i_end, dtype=int)
        j_range = i_range + shift
        mask = (current[i_range] > 0.5) & (current[j_range] < 0.5)
        cand_i = i_range[mask]

        if cand_i.size > 0:
            # choose a contributing 1 and swap it with a random 0 elsewhere
            i = int(rng.choice(cand_i))
            zero_idxs = np.where(current < 0.5)[0]
            zero_idxs = zero_idxs[zero_idxs != i]
            if zero_idxs.size == 0:
                continue
            j = int(rng.choice(zero_idxs))
            # perform the swap (full exchange)
            current[i] = 0.0
            current[j] = 1.0
            new_c5 = _compute_c5(current)
            delta_obj = new_c5 - current_c5
            if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / temp):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current.copy()
            else:
                # revert
                current[i] = 1.0
                current[j] = 0.0
        else:
            # fallback: random two–index delta move (as in the original scheme)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            hi = current[i]
            hj = current[j]
            if hi == hj:
                continue
            # feasible delta interval keeping both in [0,1]
            delta_min = max(-hi, -(1.0 - hj))
            delta_max = min(1.0 - hi, hj)
            low = max(delta_min, -step)
            high = min(delta_max, step)
            if low >= high:
                continue
            delta = rng.random() * (high - low) + low
            current[i] = hi + delta
            current[j] = hj - delta
            new_c5 = _compute_c5(current)
            delta_obj = new_c5 - current_c5
            if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / temp):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current.copy()
            else:
                # revert move
                current[i] = hi
                current[j] = hj

    # ----- final repair & report ----------------------------------------------
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
