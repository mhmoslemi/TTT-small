import time
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """
    Euclidean projection of vector v onto the set
    {x : lo <= x_i <= hi, sum x_i = s}.
    """
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_gradient(h, shift):
    """Gradient of correlation term at a specific integer shift."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def compute_c5(h):
    """Exact C5 value on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Random 0/1 vector with exactly n/2 ones (n even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_fractional(n, rng):
    """Random vector projected onto the box-sum constraints."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def optimize_h(h, n_iter, step0=0.6):
    """Projected gradient descent with a warm‑up soft‑max phase."""
    n = h.shape[0]
    target_sum = n / 2.0
    warm_iters = max(1, int(n_iter * 0.15))  # 15 % warm‑up
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        if it <= warm_iters:
            # soft‑max smoothing of the max
            half = warm_iters // 2
            tau = 0.1 if it <= half else 0.02
            maxc = np.max(corr)
            # scale to avoid overflow
            exps = np.exp((corr - maxc) / tau)
            weights = exps / np.sum(exps)

            grad = np.zeros_like(h)
            # accumulate weighted gradients
            for idx, shift in enumerate(range(-n + 1, n)):
                w = weights[idx]
                if w == 0.0:
                    continue
                grad += w * compute_gradient(h, shift)
        else:
            # sub‑gradient of the active (max) shift
            idx = np.argmax(corr)
            shift = idx - (n - 1)
            grad = compute_gradient(h, shift)

        eta = step0 / np.sqrt(it)
        h = h - eta * grad
        h = project_onto_box_sum(h, target_sum)
    return h, compute_c5(h)
