    if abs(diff) < 1e-12:

        if diff > 0:

            h[idx] += inc
            diff -= inc

            h[idx] -= dec

        attempts += 1

        h += diff / h.size

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust array h so that its sum equals target while keeping 0 <= h_i <= 1.
    Small residuals are distributed uniformly and clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 1000
    size = h.size
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # distribute remaining error uniformly (possible slight bound violation)
        h += diff / size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):

    # ----- initialise -----
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    except Exception:
        n = 200
        init_h = None

    target_sum = n / 2.0

    if init_h is not None and init_h.size == n:
        h_current = adjust_sum_to_target(init_h.copy(), target_sum, rng)
    else:
        h_current = rng.rand(n)
        h_current = h_current / h_current.sum() * target_sum
        h_current = np.clip(h_current, 0.0, 1.0)
        h_current = adjust_sum_to_target(h_current, target_sum, rng)

    best_h = h_current.copy()
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5

    # annealing schedule parameters
    T0, T_end = 0.02, 1e-6
    step0, step_end = 0.35, 1e-4

    # stagnation handling
    it_since_improve = 0
    max_stagnation = 2500

    while True:
        now = time.time()
        if now - start >= max_time:
            break

        progress = (now - start) / max_time
        T = T0 * (1.0 - progress) + T_end * progress
        step = step0 * (1.0 - progress) + step_end * progress

        move_type = rng.rand()

        # -------- gradient move on worst shift --------
        if move_type < 0.4:
            corr_full = np.correlate(h_current, 1.0 - h_current, mode="full")
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n - 1)

            g = np.zeros_like(h_current)
            idx = np.arange(n)

            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n)
            if np.any(valid_plus):
                g[valid_plus] += (1.0 - h_current[plus[valid_plus]])

            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n)
            if np.any(valid_minus):
                g[valid_minus] -= h_current[minus[valid_minus]]

            # Zero‑mean adjustment to keep sum unchanged
            g -= g.mean()

            candidate = h_current - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)

        # -------- pairwise exchange move --------
        elif move_type < 0.8:
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            hi, hj = h_current[i], h_current[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = np.clip(delta, low, high)
            candidate = h_current.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        # -------- random multi‑index perturbation --------
        else:
            candidate = h_current.copy()
            idxs = rng.choice(n, size=3, replace=False)
            deltas = (rng.rand(3) * 2.0 - 1.0) * step
            candidate[idxs] += deltas
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)

        cand_c5 = compute_c5(candidate)

        # ----- acceptance logic -----
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0.0 else 0.0
            accept = rng.rand() < prob

        if accept:
            h_current = candidate
            current_c5 = cand_c5
            it_since_improve = 0
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
        else:
            it_since_improve += 1

        # ----- restart on stagnation -----
        if it_since_improve >= max_stagnation:
            # re‑initialise either from best or fresh random
            if rng.rand() < 0.5:
                h_current = best_h.copy()
            else:
                h_current = rng.rand(n)
                h_current = h_current / h_current.sum() * target_sum
                h_current = np.clip(h_current, 0.0, 1.0)
                h_current = adjust_sum_to_target(h_current, target_sum, rng)
            current_c5 = compute_c5(h_current)
            it_since_improve = 0

    # final sanity projection
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n)
