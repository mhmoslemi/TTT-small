import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

            best_h = h.copy()
        eps = max(eps_final, eps_init * (0.9999 ** it))
        grad = grad_smooth_max(h, corr, eps)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        h = project_onto_box_sum(h, n / 2.0)
    return best_h, best_score


def random_start(n, rng):
    """Random feasible start."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def local_refine(h, rng, budget_s, start_time, max_iters=3000):
    """Simple pairwise swap refinement keeping sum fixed."""
    n = h.size
    best_h = h.copy()
    best_score = c5_from_corr(compute_corr(best_h), n)
    for _ in range(max_iters):
        if time.time() - start_time > budget_s - 0.1:
            break
        i, j = rng.choice(n, size=2, replace=False)
        delta = (rng.random() - 0.5) * 0.08  # up to ±0.04
        hi = best_h[i] + delta
        hj = best_h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h_new = best_h.copy()
        h_new[i] = hi
        h_new[j] = hj
        new_score = c5_from_corr(compute_corr(h_new), n)
        if new_score < best_score:
            best_h = h_new
            best_score = new_score
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine base size
    if "initial_h_values" in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
    else:
        base_h = random_start(256, rng)

    n = base_h.size
    base_h = project_onto_box_sum(base_h, n / 2.0)

    best_h = None
    best_score = float("inf")

    # ---------- Phase 1: several short Adam runs ----------
    for _ in range(4):
        if time.time() - start > budget_s - 0.7:
            break
        h0 = base_h.copy() if _ == 0 else random_start(n, rng)
        h_opt, sc_opt = adam_optimize(h0, budget_s, start)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # ---------- Phase 2: longer refinement ----------
    if best_h is None:
        best_h = base_h.copy()
    if time.time() - start < budget_s - 0.4:
        h_opt, sc_opt = adam_optimize(best_h, budget_s, start)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # ---------- Phase 3: local swap refinement ----------
    if time.time() - start < budget_s - 0.2:
        h_ref, sc_ref = local_refine(best_h, rng, budget_s, start)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # final projection and score
    best_h = project_onto_box_sum(best_h, n / 2.0)
    final_score = c5_from_corr(compute_corr(best_h), n)

    return best_h, final_score, n
