import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust array h (in-place) so that its sum equals target while respecting 0<=h_i<=1.
    Small residuals are distributed uniformly and clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 1000
    size = h.size
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # distribute any remaining tiny error uniformly (may slightly violate bounds)
        h += diff / size
        np.clip(h, 0.0, 1.0, out=h)
        # final tiny correction
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            for _ in range(100):
                idx = rng.randint(size)
                if diff2 > 0 and h[idx] < 1.0:
                    delta = min(diff2, 1.0 - h[idx])
                elif diff2 < 0 and h[idx] > 0.0:
                    delta = max(diff2, -h[idx])
                else:
                    continue
                h[idx] += delta
                diff2 -= delta
                if abs(diff2) < 1e-12:
                    break
    return h

    best_h = h_current.copy()
    best_score = compute_c5(best_h)

    cur_h = h_current.copy()
    cur_score = best_score

    # Annealing parameters
    T0 = 0.02                 # initial temperature
    Tf = 1e-6                 # final temperature
    step0 = 0.2               # initial step magnitude
    step_min = 1e-4           # final step magnitude

    no_improve = 0
    RESTART_AFTER = 5000      # iterations without improvement before a soft restart

    # ----- Main optimisation loop -----
    while True:
        now = time.time()
        if now - start_time >= max_time:
            break

        elapsed = now - start_time
        progress = elapsed / max_time
        # temperature and step size (linear schedules)
        T = T0 * (1.0 - progress) + Tf * progress
        step = step0 * (1.0 - progress) + step_min * progress

        move_type = rng.rand()
        n = n_points

        if move_type < 0.3:
            # ---- gradient move on the worst shift ----
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n - 1)  # integer lag, can be negative

            # gradient of the maximal overlap term
            g = np.zeros_like(cur_h)
            idx = np.arange(n)

            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n)
            if np.any(valid_plus):
                g[valid_plus] += (1.0 - cur_h[plus[valid_plus]])

            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n)
            if np.any(valid_minus):
                g[valid_minus] -= cur_h[minus[valid_minus]]

            # keep total mass unchanged
            g -= g.mean()

            cand = cur_h - step * g
            np.clip(cand, 0.0, 1.0, out=cand)
            cand = adjust_sum_to_target(cand, target_sum, rng)

        elif move_type < 0.7:
            # ---- pairwise exchange move ----
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            hi, hj = cur_h[i], cur_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = np.clip(delta, low, high)
            cand = cur_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
            # sum already preserved; just clip (should stay within bounds)
            np.clip(cand, 0.0, 1.0, out=cand)

        else:
            # ---- small random perturbation of a few entries ----
            cand = cur_h.copy()
            num = rng.randint(1, 5)  # 1 to 4 entries
            idxs = rng.choice(n, size=num, replace=False)
            cand[idxs] = rng.rand(num)
            cand = adjust_sum_to_target(cand, target_sum, rng)

        cand_score = compute_c5(cand)

        # acceptance test
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = rng.rand() < prob

        if accept:
            cur_h = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
        else:
            no_improve += 1

        # occasional soft restart if stuck
        if no_improve >= RESTART_AFTER:
            # start from best solution with a tiny random jitter
            cur_h = best_h.copy()
            jitter = rng.normal(scale=0.02, size=n)
            cur_h += jitter
            np.clip(cur_h, 0.0, 1.0, out=cur_h)
            cur_h = adjust_sum_to_target(cur_h, target_sum, rng)
            cur_score = compute_c5(cur_h)
            no_improve = 0

    # ----- Final sanitisation -----
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
