def compute_c5(h):
    n = h.shape[0]
    N = 1 << (2*n - 1).bit_length()  # next power of two >= 2n-1
    # Actually need length at least 2n-1 for full correlation.
    # Setup FFT of h and rev of (1-h)
    h_fft = np.fft.rfft(h, n=N)
    rev_b_fft = np.fft.rfft((1 - h)[::-1], n=N)
    corr = np.fft.irfft(h_fft * rev_b_fft, n=N)
    # The result length N, but meaningful part is from 0 to 2n-2 (or something)
    # After conv of length n + n -1 = 2n-1. So we take first 2n-1 entries.
    corr = corr[:2*n-1]
    # np.correlate returns length 2n-1, with lags from -(n-1) to n-1.
    # This is fine: max is same.
    return np.max(corr) * (2.0 / n)

def hill_climb(h, rng, start_elapsed, budget_limit, compute_c5, max_iter=2000):
   n = h.shape[0]; best_h = h.copy(); best_c5 = compute_c5(best_h)
   start = time.time()
   iter = 0
   while iter < max_iter and time.time() - start_elapsed < budget_limit:
        # choose random i, j
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # compute allowable delta
        delta_max = min(1 - best_h[i], best_h[j])
        if delta_max <= 0: continue
        # choose delta randomly within [0, delta_max]
        delta = rng.random() * delta_max
        # apply
        best_h[i] += delta
        best_h[j] -= delta
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
           best_c5 = c5_new
        else:
           # revert
           best_h[i] -= delta
           best_h[j] += delta
        iter += 1
   return best_h, best_c5

def compute_c5(h):
    n = h.shape[0]
    N = 1 << ((2*n-1).bit_length())  # next power of two >= 2n-1
    # FFT via rfft for real input; rfft returns N//2+1 components.
    H = np.fft.rfft(h, n=N)
    rev = (1 - h)[::-1]
    R = np.fft.rfft(rev, n=N)
    corr = np.fft.irfft(H * R, n=N)
    # Trim to length 2n-1
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Fast C5 using FFT."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=N)
    rev = (1.0 - h)[::-1]
    R = np.fft.rfft(rev, n=N)
    corr = np.fft.irfft(H * R, n=N)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=15):
    """Clamp to [0,1] and enforce sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny fix if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h)  # most room to increase
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def hill_climb(h_start, rng, time_start, time_limit, max_iter=2000):
    """Simple mass‑swap hill climbing with sum constraint."""
    h = h_start.copy()
    best_c5 = compute_c5(h)
    n = h.shape[0]
    it = 0
    while it < max_iter and time.time() - time_start < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        up = 1.0 - h[i]
        down = h[j]
        if up <= 0.0 or down <= 0.0:
            continue
        delta = rng.random() * min(up, down)
        if delta <= 0.0:
            continue
        h[i] += delta
        h[j] -= delta
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # revert
            h[i] -= delta
            h[j] += delta
        it += 1
    return h, best_c5
