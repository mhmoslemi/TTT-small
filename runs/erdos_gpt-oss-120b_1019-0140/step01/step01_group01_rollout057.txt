def smooth_c5_and_grad(h, alpha):
    n = h.shape[0]
    # compute f = correlation values (full mode)
    f_full = np.correlate(h, 1.0 - h, mode='full')
    # weights
    max_f = f_full.max()
    # scaling to avoid overflow: subtract max before exp
    exp_terms = np.exp(alpha * (f_full - max_f))
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp
    # Compute gradient
    grad = np.zeros_like(h)
    # shift index mapping: f_full index correspond to shift = idx - (n-1)
    for shift_idx, wk in enumerate(w):

def smooth_c5_and_grad(h, alpha):
    n = h.shape[0]
    f_full = np.correlate(h, 1.0 - h, mode='full')
    max_f = f_full.max()
    exp_terms = np.exp(alpha * (f_full - max_f))
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp  # shape (2*n-1,)
    # compute smooth value (log-sum-exp scaled)
    smooth_val = max_f + np.log(sum_exp) / alpha * 1.0  # Actually log-sum-exp = max_f + log(sum_exp)/alpha
    # but we don't need smooth_val maybe
    grad = np.zeros_like(h)
    n_shift = n - 1
    for shift_idx, wk in enumerate(w):
        k = shift_idx - n_shift
        # term for i where i+k in range
        if k >= 0:
            i = np.arange(0, n - k)
            j = i + k
        else:
            i = np.arange(-k, n)
            j = i + k
        # gradient plus term
        grad[i] += wk * (1.0 - h[j])
        # gradient minus term: for i where i - k in range
        # This is similar to using shift -k?
        # We can compute i2 range similarly:
        if k <= 0:
            i2 = np.arange(0, n + k)  # because k negative => n + k < n
            j2 = i2 - k
        else:
            i2 = np.arange(k, n)
            j2 = i2 - k
        grad[i2] -= wk * h[j2]
    # after loop, need to subtract mean to satisfy sum constraint? Not needed if we later project.
    return smooth_val, grad

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of C5."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Scale/shift h in-place so that sum(h) == target while keeping 0<=h_i<=1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # try to correct by adjusting random entries directly
    max_trials = 500
    for _ in range(max_trials):
        if abs(diff) < 1e-12:
            break
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
    if abs(diff) > 1e-12:
        # distribute tiny residual uniformly and clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_smooth(h: np.ndarray, alpha: float, shift_data):
    """Gradient of the smoothed max (log-sum-exp) of the overlap terms."""
    n = h.shape[0]
    f = np.correlate(h, 1.0 - h, mode="full")
    max_f = f.max()
    exp_term = np.exp(alpha * (f - max_f))
    w = exp_term / exp_term.sum()  # weights for each shift (2n-1 entries)

    grad = np.zeros_like(h)
    for (i_plus, j_plus, i_minus, j_minus), wk in zip(shift_data, w):
        grad[i_plus] += wk * (1.0 - h[j_plus])
        grad[i_minus] -= wk * h[j_minus]
    # Project to hyperplane sum(grad)=0 (helps keep total mass unchanged)
    grad -= grad.mean()
    return grad

def build_shift_data(n):
    """Pre‑compute index arrays for each lag in [-n+1, n-1]."""
    data = []
    for k in range(-n + 1, n):
        if k >= 0:
            i_plus = np.arange(0, n - k, dtype=int)
            j_plus = i_plus + k
        else:
            i_plus = np.arange(-k, n, dtype=int)
            j_plus = i_plus + k
        if k <= 0:
            i_minus = np.arange(0, n + k, dtype=int)  # k negative
            j_minus = i_minus - k
        else:
            i_minus = np.arange(k, n, dtype=int)
            j_minus = i_minus - k
        data.append((i_plus, j_plus, i_minus, j_minus))
    return data

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # ----- problem size -----
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.size
    except Exception:
        n = 200  # fallback grid size
        init_arr = None
    target_sum = n / 2.0
    dx = 2.0 / n

    shift_data = build_shift_data(n)

    # ----- keep best solution -----
    best_h = None
    best_c5 = float("inf")
    if init_arr is not None:
        h0 = init_arr.copy()
        h0 = adjust_sum_to_target(h0, target_sum, rng)
        c0 = compute_c5(h0)
        best_h, best_c5 = h0, c0

    # ----- main optimisation loop -----
    # parameters for schedules
    step0, step_end = 0.12, 1e-5
    T0, T_end = 0.015, 1e-6
    alpha_start, alpha_end = 5.0, 60.0

    # loop over restarts
    while time.time() - start < max_time:
        # random start respecting the mass constraint
        h = rng.rand(n)
        h = h / h.sum() * target_sum
        np.clip(h, 0.0, 1.0, out=h)
        h = adjust_sum_to_target(h, target_sum, rng)
        cur_c5 = compute_c5(h)

        # inner iterations for this restart
        max_iter = 3000
        for it in range(max_iter):
            now = time.time()
            if now - start >= max_time:
                break

            # global progress for schedules
            prog = (now - start) / max_time
            step = step0 * (1.0 - prog) + step_end * prog
            T = T0 * (1.0 - prog) + T_end * prog
            alpha = alpha_start * (1.0 - prog) + alpha_end * prog

            if rng.rand() < 0.65:
                # ----- gradient move -----
                grad = gradient_smooth(h, alpha, shift_data)
                cand = h - step * grad
                np.clip(cand, 0.0, 1.0, out=cand)
                cand = adjust_sum_to_target(cand, target_sum, rng)
            else:
                # ----- pairwise exchange move -----
                i, j = rng.randint(n, size=2)
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                low = max(-hi, hj - 1.0)
                high = min(1.0 - hi, hj)
                if low >= high:
                    continue
                delta = low + rng.rand() * (high - low) * 0.5
                cand = h.copy()
                cand[i] = hi + delta
                cand[j] = hj - delta
                # sum unchanged; just clip to bounds (already respected)

            cand_c5 = compute_c5(cand)

            # acceptance (simulated annealing)
            if cand_c5 < cur_c5:
                accept = True
            else:
                prob = np.exp(-(cand_c5 - cur_c5) / max(T, 1e-12))
                accept = rng.rand() < prob

            if accept:
                h = cand
                cur_c5 = cand_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    # early exit if target achieved
                    if best_c5 <= 0.3808:
                        break
            # else keep current h and cur_c5 unchanged
        # break outer loop if target already met
        if best_c5 <= 0.3808:
            break

    # final sanity correction
    if best_h is None:
        # fallback to a simple uniform construction
        best_h = np.full(n, 0.0)
        best_h[:int(target_sum)] = 1.0
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n)
