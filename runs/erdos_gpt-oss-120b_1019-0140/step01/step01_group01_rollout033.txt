def compute_c5(h):
    n = h.shape[0]; dx = 2.0/n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def generate_random_binary(N, target_sum, rng):
    arr = np.zeros(N, dtype=np.uint8)
    ones_idx = rng.choice(N, size=target_sum, replace=False)
    arr[ones_idx] = 1
    return arr

def fix_sum_binary(arr, target_sum, rng):
    current = arr.sum()
    diff = target_sum - current
    if diff == 0: return arr
    # diff positive: need to turn some zeros to ones
    if diff > 0:
        zero_idx = np.flatnonzero(arr == 0)
        # ensure enough zeros
        choose = rng.choice(zero_idx, size=diff, replace=False)
        arr[choose] = 1
    else:
        # diff negative: turn some ones to zeros
        one_idx = np.flatnonzero(arr == 1)
        choose = rng.choice(one_idx, size=-diff, replace=False)
        arr[choose] = 0
    return arr

def crossover_two_point(parent1, parent2, rng):
    N = parent1.shape[0]
    # random cut points
    c1 = rng.randint(0, N)
    c2 = rng.randint(0, N)
    if c1 > c2:
        c1, c2 = c2, c1
    # create child
    child = np.empty(N, dtype=np.uint8)
    child[:c1] = parent1[:c1]
    child[c1:c2] = parent2[c1:c2]
    child[c2:] = parent1[c2:]
    return child

def mutate_swap(arr, rng):
    # swap one randomly chosen 1 with a random 0
    ones = np.flatnonzero(arr)
    zeros = np.flatnonzero(arr == 0)
    if ones.size == 0 or zeros.size == 0:
        return arr
    i = rng.choice(ones)
    j = rng.choice(zeros)
    arr[i], arr[j] = arr[j], arr[i]
    return arr

def select_parent(pop, fitness, rng, tournament_size=3):
    idx = rng.randint(0, len(pop), size=tournament_size)
    best_idx = idx[0]
    best_fit = fitness[best_idx]
    for i in idx[1:]:
        if fitness[i] < best_fit:
            best_fit = fitness[i]
            best_idx = i
    return pop[best_idx]

def solve_mip(N, target_sum, time_limit, rng):
    import cvxpy as cp
    x = cp.Variable(N, boolean=True)
    p = cp.Variable((N,N), boolean=True)
    # constraints list
    constraints = []
    constraints.append(cp.sum(x) == target_sum)
    # product constraints
    for i in range(N):
        for j in range(N):
            constraints.append(p[i,j] <= x[i])
            constraints.append(p[i,j] <= x[j])
            constraints.append(p[i,j] >= x[i] + x[j] - 1)
    # L variable
    L = cp.Variable(integer=True)
    constraints.append(L >= 0)
    # shift constraints
    for k in range(-(N-1), N):
        # compute expression sum_i (x[i] - p[i,i+k]) for valid i
        expr = 0
        for i in range(N):
            j = i + k
            if 0 <= j < N:
                expr += x[i] - p[i,j]
        constraints.append(expr <= L)
    # objective
    obj = cp.Minimize(L)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    return prob, x, L
```

    x = cp.Variable(N, boolean=True)
    p = cp.Variable((N,N), boolean=True)
    L = cp.Variable(integer=True)
    constraints = [cp.sum(x) == target_sum, L >= 0]
    for i in range(N):
        for j in range(N):
            constraints.append(p[i,j] <= x[i])
            constraints.append(p[i,j] <= x[j])
            constraints.append(p[i,j] >= x[i] + x[j] - 1)

        constraints.append(expr <= L)
    prob = cp.Problem(cp.Minimize(L), constraints)
    prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of C5 = max_k ∫ h(x)*(1-h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    hs = h.astype(float)
    corr = np.correlate(hs, 1.0 - hs, mode="full")
    return float(np.max(corr) * dx)


def generate_random_binary(n: int, target: int, rng: np.random.RandomState) -> np.ndarray:
    """Random binary vector with exactly `target` ones."""
    arr = np.zeros(n, dtype=np.uint8)
    ones = rng.choice(n, size=target, replace=False)
    arr[ones] = 1
    return arr


def fix_sum_binary(arr: np.ndarray, target: int, rng: np.random.RandomState) -> np.ndarray:
    """Adjust a binary array to have exactly `target` ones (preserving 0/1)."""
    cur = int(arr.sum())
    diff = target - cur
    if diff == 0:
        return arr
    if diff > 0:
        zero_idx = np.flatnonzero(arr == 0)
        choose = rng.choice(zero_idx, size=diff, replace=False)
        arr[choose] = 1
    else:
        one_idx = np.flatnonzero(arr == 1)
        choose = rng.choice(one_idx, size=-diff, replace=False)
        arr[choose] = 0
    return arr


def solve_mip(n: int, target: int, time_limit: float, rng: np.random.RandomState):
    """Mixed‑integer formulation: minimize maximal overlap L."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None, None  # cvxpy not available

    # variables
    x = cp.Variable(n, boolean=True)                     # h_i
    p = cp.Variable((n, n), boolean=True)               # x_i * x_j
    L = cp.Variable(integer=True)

    constraints = [cp.sum(x) == target, L >= 0]

    # linearization of product p_ij = x_i * x_j
    for i in range(n):
        for j in range(n):
            constraints.append(p[i, j] <= x[i])
            constraints.append(p[i, j] <= x[j])
            constraints.append(p[i, j] >= x[i] + x[j] - 1)

    # bound each shift's overlap by L
    for k in range(-(n - 1), n):
        expr = 0
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                expr += x[i] - p[i, j]   # contribution h_i*(1-h_j)
        constraints.append(expr <= L)

    prob = cp.Problem(cp.Minimize(L), constraints)
    # CBC understands the parameter `maxSeconds`
    prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None, None

    # round solution to binary
    hx = np.rint(x.value).astype(np.uint8)
    hx = fix_sum_binary(hx, target, rng)  # safety
    Lval = int(round(L.value)) if L.value is not None else None
    return hx, Lval, prob


def select_parent(pop, fitness, rng, tourn_size=3):
    """Tournament selection (lower fitness is better)."""
    idx = rng.randint(0, len(pop), size=tourn_size)
    best = idx[0]
    best_fit = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_fit:
            best = i
            best_fit = fitness[i]
    return pop[best]


def crossover_two_point(p1, p2, rng):
    """Two‑point crossover preserving length."""
    n = p1.shape[0]
    c1, c2 = rng.randint(0, n, size=2)
    if c1 > c2:
        c1, c2 = c2, c1
    child = np.empty(n, dtype=np.uint8)
    child[:c1] = p1[:c1]
    child[c1:c2] = p2[c1:c2]
    child[c2:] = p1[c2:]
    return child


def mutate_swap(arr, rng):
    """Swap a 1 with a 0 (preserves total sum)."""
    ones = np.flatnonzero(arr)
    zeros = np.flatnonzero(arr == 0)
    if len(ones) == 0 or len(zeros) == 0:
        return arr
    i = rng.choice(ones)
    j = rng.choice(zeros)
    arr[i], arr[j] = arr[j], arr[i]
    return arr

    Optimize a step function h on [0,2] (discretized) to minimise C5.
    Returns (h_values, c5_bound, n_points).

            # if we already achieve a very good bound, we may skip GA
            # but we still use remaining time for possible improvement
    # ==============================================================
    # 2) Genetic algorithm (always executed, uses remaining time)
    # ==============================================================
    remaining = max_time - (time.time() - start)

        mutation_rate = 0.3

        while True:

            if now - start >= max_time:

                if fitness[i] < best_c5:
                    best_c5 = fitness[i]
                    best_h = ind.astype(float).copy()

        # ==========================================================

            cur = best_h.astype(np.uint8)
            ones_idx = np.flatnonzero(cur == 1)
            zeros_idx = np.flatnonzero(cur == 0)
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            cur[i], cur[j] = cur[j], cur[i]  # swap
            new_c5 = compute_c5(cur.astype(float))
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur.astype(float).copy()
            else:
                # revert
                cur[i], cur[j] = cur[j], cur[i]

    # Ensure final vector satisfies constraints exactly
    if best_h is None:
        # extreme fallback: simple alternating half‑ones (guaranteed feasible)
        best_h = generate_random_binary(n_points, target_sum, rng).astype(float)
        best_c5 = compute_c5(best_h)

    # final sanity check
    best_h = fix_sum_binary(best_h.astype(np.uint8), target_sum, rng).astype(float)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
