def _simulated_annealing_custom(start_h, time_limit, rng):
    h_cur = start_h.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    # Set parameters
    T0 = 0.02 # initial temperature
    T_min = 0.0001
    max_eps_start = 0.3
    max_eps_end = 0.02

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # progress = elapsed / time_limit
        frac = elapsed / time_limit
        # temperature linearly decreasing
        T = max(T_min, T0 * (1 - frac))
        max_eps = max_eps_start * (1 - frac) + max_eps_end * frac
        # choose distinct i,j
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi = h_cur[i]
        hj = h_cur[j]
        # compute allowable eps
        up = min(1.0 - hi, hj)  # positive epsilon bound (increase hi)
        down = min(hi, 1.0 - hj)  # negative epsilon bound (decrease hi)
        if up + down == 0:
            continue
        # sample eps uniform in [-down, up]
        eps = rng.random() * (up + down) - down
        # limit magnitude
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        # apply move
        h_new = h_cur
        # we will modify in place for speed but copy after acceptance?
        # Probably more safe to compute in new array.
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # compute new score
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        # acceptance test
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        # optionally occasional random re-anneal? Not needed
    return best_h, best_score

def _repair_vector(v):

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        return np.clip(v, 0.0, 1.0)
    factor = target / s
    v = v * factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v = v * factor2
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(start_h: np.ndarray, time_limit: float, rng: np.random.Generator) -> tuple:
    """SA that preserves the sum constraint by moving mass between two entries."""
    h_cur = start_h.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.025          # initial temperature
    T_min = 1e-4
    eps_start = 0.3     # max epsilon at start
    eps_end = 0.02      # at end
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        T = max(T_min, T0 * (1.0 - frac))
        max_eps = eps_start * (1.0 - frac) + eps_end * frac
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)          # positive epsilon bound
        down = min(hi, 1.0 - hj)        # negative epsilon bound
        if up + down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _refine_gradient(h: np.ndarray, time_limit: float, rng: np.random.Generator) -> np.ndarray:
    """Directionally reduce the maximal correlation using a simple gradient step."""
    start = time.time()
    N = h.size
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (N - 1)  # signed shift
        # compute gradient of the maximal correlation entry
        d = np.zeros(N)
        if shift >= 0:
            i_range = np.arange(N - shift)
            d[i_range] += 1.0 - h[i_range + shift]
            d[i_range + shift] -= h[i_range]
        else:
            s = -shift
            i_range = np.arange(s, N)
            d[i_range] += 1.0 - h[i_range + shift]
            d[i_range + shift] -= h[i_range]
        # positions where increasing reduces the max correlation
        pos = np.where((d > 1e-12) & (h > 1e-12))[0]
        neg = np.where((d < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps  # move as much as allowed
        h[i] -= eps
        h[j] += eps
        # check improvement; if not better, undo and stop
        new_max = np.max(np.correlate(h, 1.0 - h, mode='full'))
        if new_max * (2.0 / N) >= np.max(corr) * (2.0 / N):
            h[i] += eps
            h[j] -= eps
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # initial construction (may come from the harness)
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 100
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)], dtype=np.float64)
    h0 = _repair_vector(h0)

    start_total = time.time()
    best_h = h0
    best_score = _c5_value(best_h)

    # ------------------------------------------------------------------
    # Phase 1: repeated short SA runs on the current resolution
    phase1_time = budget_s * 0.6
    while True:
        elapsed = time.time() - start_total
        if elapsed >= phase1_time:
            break
        remaining = phase1_time - elapsed
        sub_time = min(12.0, remaining)          # each SA ≤12 s
        # start from a perturbed version of the current best
        start_vec = best_h + rng.uniform(-0.2, 0.2, size=best_h.shape)
        start_vec = _repair_vector(start_vec)
        cand_h, cand_score = _simulated_annealing(start_vec, sub_time, rng)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score

    # ------------------------------------------------------------------
    # Phase 2: increase resolution iteratively while time permits
    while True:
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 25 or best_h.size >= 800:
            break
        n_new = min(best_h.size * 2, 800)
        if n_new == best_h.size:
            break
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _repair_vector(h_up)
        # allocate half of remaining time to this level
        sub_time = remaining * 0.5
        cand_h, cand_score = _simulated_annealing(h_up, sub_time, rng)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score

    # ------------------------------------------------------------------
    # Phase 3: gradient‑based refinement on the final resolution
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining > 5:
        best_h = _refine_gradient(best_h, remaining, rng)

    # final repair and evaluation
    best_h = _repair_vector(best_h)
    final_score = _c5_value(best_h)
    return best_h, float(final_score), best_h.size
