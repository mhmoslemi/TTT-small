def compute_smooth_gradient(h, beta):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # Use shift normalization: smaller shift with less weight.
    w = np.exp(beta * (corr - max_corr))
    w_sum = np.sum(w)
    w = w / w_sum
    # gradient via convolutions
    # forward contribution
    conv_fwd = np.convolve(w[::-1], h, mode='full')

    term_fwd = conv_fwd[start:start + n]
    term_rev = conv_rev[start:start + n]
    grad = 1.0 - (term_fwd + term_rev)
    return grad

def smooth_gradient_descent(h0, max_iters, beta_start, beta_end, step0=0.5):
    h = h0.copy()
    target_sum = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    for it in range(1, max_iters+1):
        # compute correlation and max
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # compute weight vector
        beta = beta_start + (beta_end - beta_start) * (it - 1) / (max_iters - 1)
        w = np.exp(beta * (corr - max_corr))
        w_sum = np.sum(w)
        w /= w_sum
        # compute gradient via convolution
        conv_fwd = np.convolve(w[::-1], h, mode='full')
        conv_rev = np.convolve(w, h, mode='full')
        start = n - 1
        grad = 1.0 - (conv_fwd[start:start + n] + conv_rev[start:start + n])
        # step size
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        # compute score new
        score = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time, math
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # enlarge interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def projected_gradient_descent(h0, n_iter=2000, step0=0.5):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_fractional(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def smooth_gradient_descent(h0, max_iters=3000, beta_start=0.1, beta_end=5.0,
                           step0=0.5, deadline=None):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters + 1):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        beta = beta_start + (beta_end - beta_start) * (it - 1) / (max_iters - 1)
        w = np.exp(beta * (corr - max_corr))
        w /= w.sum()
        # forward and reverse contributions via convolution
        conv_fwd = np.convolve(w[::-1], h, mode='full')
        conv_rev = np.convolve(w, h, mode='full')
        start = n - 1
        grad = 1.0 - (conv_fwd[start:start + n] + conv_rev[start:start + n])
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.3  # spare margin
    target_n = 256
    best_h = None
    best_score = float('inf')
    candidates = []

    # 1. base candidates
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != target_n:
            h0 = upsample(h0, target_n)
        h0 = project_onto_box_sum(h0, target_n / 2.0)
        candidates.append(h0)

    # uniform half‑density
    uniform = np.full(target_n, 0.5, dtype=float)
    uniform = project_onto_box_sum(uniform, target_n / 2.0)
    candidates.append(uniform)

    # Thue–Morse binary patterns (upsampled when needed)
    for n in (64, 128, 256):
        if n % 2 == 0:
            tm = thue_morse(n)
            if n != target_n:
                tm = upsample(tm, target_n)
            tm = project_onto_box_sum(tm, target_n / 2.0)
            candidates.append(tm)

    # a few random fractional starts
    for _ in range(6):
        candidates.append(random_fractional(target_n, rng))

    # 2. intensive optimization of each candidate
    for base_h in candidates:
        if time.time() > deadline:
            break
        # allocate more iterations for the primary size
        iters = 12000 if base_h.size == target_n else 6000
        # smooth descent (global exploration)
        h_opt, score_opt = smooth_gradient_descent(
            base_h, max_iters=iters,
            beta_start=0.1, beta_end=6.0,
            step0=0.5, deadline=deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
        # final refinement with classic subgradient descent
        if time.time() < deadline - 1.0:
            h_ref, scr_ref = projected_gradient_descent(best_h, n_iter=3000, step0=0.5)
            if scr_ref < best_score:
                best_score = scr_ref
                best_h = h_ref.copy()

    # 3. random restarts under remaining time
    while time.time() < deadline - 0.5:
        start_h = random_fractional(target_n, rng)
        h_opt, score_opt = smooth_gradient_descent(
            start_h, max_iters=4000,
            beta_start=0.1, beta_end=6.0,
            step0=0.5, deadline=deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    # 4. final feasibility projection and return
    if best_h is None:
        best_h = np.full(target_n, 0.5, dtype=float)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
