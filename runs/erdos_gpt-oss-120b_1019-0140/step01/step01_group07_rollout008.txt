def _optimize_h(h0, target_sum, start_time, budget_s):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = _compute_c5(h)
    # parameters
    max_iters = 10000
    lr = 0.05
    beta = 2.0
    beta_max = 40.0
    beta_inc = (beta_max - beta) / (max_iters // 2) # gradually increase
    patience = 0
    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.95:
            break
        # compute corr
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n) # dx
        # compute softmax weights stable
        maxc = np.max(corr)
        w = np.exp(beta * (corr - maxc))
        w_norm = w / w.sum()
        # conv backward
        conv_b = np.convolve(h, w_norm, mode='same')
        conv_f = np.convolve(h, w_norm[::-1], mode='same')
        grad = 1.0 - (conv_b + conv_f)
        # gradient descent step
        h_new = h - lr * grad
        # projection
        h_new = _project_to_constraints(h_new, target_sum)
        # evaluate actual C5
        score_new = _compute_c5(h_new)
        # Accept if better
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        # Possibly adapt lr based on improvement
        if patience > 50:
            lr *= 0.5
            patience = 0
        # Update h
        h = h_new
        # increase beta gradually
        if it < max_iters // 2:
            beta = min(beta_max, beta + (beta_max - beta) / (max_iters // 2))
    return best_h, best_score

import numpy as np
import time

def _compute_c5(h):
    """Exact C5 value for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust to satisfy sum(v)=target exactly (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        while need > 1e-12:
            free = np.where(v < 1.0)[0]
            if free.size == 0:
                break
            add = need / free.size
            caps = 1.0 - v[free]
            add = np.minimum(add, caps)
            v[free] += add
            need -= add.sum()
    else:
        excess = cur - target
        while excess > 1e-12:
            free = np.where(v > 0.0)[0]
            if free.size == 0:
                break
            sub = excess / free.size
            caps = v[free]
            sub = np.minimum(sub, caps)
            v[free] -= sub
            excess -= sub.sum()
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0.0:
        v *= target / cur
        v = np.clip(v, 0.0, 1.0)
    return v

def _optimize_h(h0, target_sum, start_time, budget_s, max_iters=4000):
    """Projected gradient descent on a soft‑max approximation of C5."""
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_score = _compute_c5(h)
    lr = 0.5                     # base step size (will be scaled by dx)
    beta = 2.0                   # initial smoothing parameter
    beta_max = 30.0
    patience = 0
    it = 0
    while it < max_iters:
        if time.time() - start_time > budget_s * 0.95:
            break
        # correlation (with dx)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        # stable soft‑max weights
        m = np.max(corr)
        w = np.exp(beta * (corr - m))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.ones_like(corr) / corr.size
        else:
            w /= w_sum
        # convolutions for gradient
        conv_b = np.convolve(h, w, mode="same")
        conv_f = np.convolve(h, w[::-1], mode="same")
        grad = dx * (1.0 - (conv_b + conv_f))
        # gradient step and projection
        h_new = _project_to_constraints(h - lr * grad, target_sum)
        score = _compute_c5(h_new)
        # keep best
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        # adapt learning rate if stalled
        if patience > 200:
            lr *= 0.5
            patience = 0
        # update beta slowly
        if beta < beta_max:
            beta = min(beta_max, beta + (beta_max - beta) / 2000.0)
        h = h_new
        it += 1
        if best_score < 0.3808:
            break
    return best_h, best_score

def _local_search(h0, target_sum, start_time, budget_s, max_moves=1500):
    """Simple pairwise mass exchange hill‑climbing."""
    h = h0.copy()
    best_h = h.copy()
    best_score = _compute_c5(best_h)
    n = h.size
    moves = 0
    while moves < max_moves:
        if time.time() - start_time > budget_s * 0.95:
            break
        i, j = np.random.choice(n, 2, replace=False)
        # maximum transferable amount without violating bounds
        max_transfer = min(0.05, best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0.0:
            moves += 1
            continue
        delta = np.random.rand() * max_transfer
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # sum unchanged automatically
        score = _compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new
        moves += 1
        if best_score < 0.3808:
            break
    return best_h, best_score
