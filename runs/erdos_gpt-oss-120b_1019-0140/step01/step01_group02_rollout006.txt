def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback uniform adjustment
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

            new_h = current_h + alpha * d
            new_c5 = _compute_c5(new_h)
            if new_c5 < current_c5 - 1e-12:
                improved = True
                break
            alpha *= 0.5

        if improved:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            # fallback: random pairwise exchange (similar to simulated annealing)
            i, j = rng.integers(0, n, size=2)
            if i != j:
                hi, hj = current_h[i], current_h[j]
                delta_min = max(-hi, -(1.0 - hj))
                delta_max = min(1.0 - hi, hj)
                if delta_min < delta_max:
                    delta = rng.uniform(delta_min, delta_max)
                    current_h[i] = hi + delta
                    current_h[j] = hj - delta
                    new_c5 = _compute_c5(current_h)
                    if new_c5 < current_c5:
                        current_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = current_h.copy()
                            stagnation = 0
                        else:
                            stagnation += 1
                    else:
                        # revert
                        current_h[i] = hi
                        current_h[j] = hj
                        stagnation += 1
                else:
                    stagnation += 1
            else:
                stagnation += 1

        # occasional reset if stuck for many iterations
        if stagnation > 5000:
            # re‑shuffle current around best to escape flat region
            current_h = best_h + rng.normal(scale=0.02, size=n)
            np.clip(current_h, 0.0, 1.0, out=current_h)
            current_h = _repair_sum(current_h, target_sum)
            current_c5 = _compute_c5(current_h)
            stagnation = 0

        it += 1

    # final repair to guarantee exact sum (tiny drift may exist)
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
