def hill_climb_opt(v_init, time_limit):
    v = v_init.copy()
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    best_score = corr.max() * dx
    best_v = v.copy()
    start = time.time()
    # define shift array for delta calculations
    shifts = np.arange(-n+1, n, dtype=int)  # length 2n-1
    # parameters
    step = 0.2 # initial max step
    min_step = 1e-5
    # temperature schedule
    T0 = 0.001 # initial temperature
    # optionally anneal over iterations based on time
    # keep track of iteration count
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # maybe reduce step over time
        # Let's compute fraction of time passed
        elapsed = time.time() - start
        frac = elapsed / time_limit
        # step decays
        cur_step = max(min_step, step * (0.98 ** (it // 1000)))  # reduce every 1000 iterations
        # compute temperature decays
        T = T0 * (1 - frac)  # linear cooling
        # compute worst shift index
        s_star_idx = np.argmax(corr)
        s_star = s_star_idx - (n - 1)
        # compute contributions for that shift
        i_valid = np.arange(n)
        i_plus = i_valid + s_star
        mask_i = (i_plus >= 0) & (i_plus < n)
        contrib = np.zeros(n)
        contrib[mask_i] = v[mask_i] * (1.0 - v[i_plus[mask_i]])
        # select i with probability proportional to contrib+epsilon
        prob = contrib + 1e-12
        prob_sum = prob.sum()
        if prob_sum > 0:
            prob /= prob_sum
            i = np.random.choice(n, p=prob)
        else:
            i = np.random.randint(n)
        # choose j distinct, maybe with low contrib
        # we can choose uniformly from indices not equal i
        j = i
        while j == i:
            j = np.random.randint(n)

        # determine max delta
        d_max = min(cur_step, 1.0 - v[i], v[j])
        if d_max <= 0:
            continue
        d = np.random.rand() * d_max
        # compute delta correlation for i (+d)
        delta_i = np.zeros_like(corr)
        # mask1
        mask1 = (i + shifts >= 0) & (i + shifts < n)
        idx1 = i + shifts[mask1]
        delta_i[mask1] += d * (1.0 - v[idx1])
        # mask2
        mask2 = (i - shifts >= 0) & (i - shifts < n)
        idx2 = i - shifts[mask2]
        delta_i[mask2] -= d * v[idx2]

        # compute delta correlation for j (-d)
        delta_j = np.zeros_like(corr)
        # using d2 = -d
        d2 = -d
        mask1j = (j + shifts >= 0) & (j + shifts < n)
        idx1j = j + shifts[mask1j]
        delta_j[mask1j] += d2 * (1.0 - v[idx1j])
        mask2j = (j - shifts >= 0) & (j - shifts < n)
        idx2j = j - shifts[mask2j]
        delta_j[mask2j] -= d2 * v[idx2j]

        new_corr = corr + delta_i + delta_j
        new_score = new_corr.max() * dx
        accept = False
        if new_score < best_score:
            accept = True
        else:
            if T > 0:
                prob_accept = np.exp(-(new_score - best_score) / T)
                if np.random.rand() < prob_accept:
                    accept = True
        if accept:
            # update v and corr
            v[i] += d
            v[j] -= d
            corr = new_corr
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_v = v.copy()
        # possibly occasional random restart: after many iterations without improvement, maybe perturb?
        # Not needed maybe.

    return best_v, best_score

def hill_climb_opt(v_init, time_limit):

    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    best_score = corr.max() * dx
    best_v = v.copy()
    start = time.time()
    shifts = np.arange(-n+1, n, dtype=int)
    # step schedule
    step = 0.2
    min_step = 1e-5
    iter_no_improve = 0
    while time.time() - start < time_limit:
        # choose i based on worst shift contributions
        s_star_idx = np.argmax(corr)
        s_star = s_star_idx - (n - 1)
        # compute contributions for i
        i_arr = np.arange(n)
        i_plus = i_arr + s_star
        mask_plus = (i_plus >= 0) & (i_plus < n)
        contrib = np.zeros(n)
        contrib[mask_plus] = v[mask_plus] * (1.0 - v[i_plus[mask_plus]])
        # choose i with probability prop to contrib
        prob = contrib + 1e-12
        prob_sum = prob.sum()
        if prob_sum > 0:
            prob = prob / prob_sum
            i = np.random.choice(n, p=prob)
        else:
            i = np.random.randint(n)
        # choose j different uniformly
        j = i

            j = np.random.randint(n)

import time
import numpy as np
from scipy.optimize import minimize

def _repair_vector(v, n, max_iter=15):
    """Project vector onto the feasible set {0≤v≤1, sum(v)=n/2}."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final cosmetic correction
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Exact C5 value for a discrete step function."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation followed by repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

    return best_x, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function for the Erdős C₅ problem."""
    np.random.seed(seed)
    start_all = time.time()
    # ---- initialise -------------------------------------------------
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 190
        parent = _random_feasible(n0)
    parent = _repair_vector(parent, n0)

    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    # ---- stage 0 : base resolution ----------------------------------
    remaining = budget_s - (time.time() - start_all)
    if remaining > 5.0:
        t0 = remaining * 0.5
        v0, s0 = _hill_climb_opt(best_v, t0)
        if s0 < best_score:
            best_v, best_score = v0, s0
        # small SLSQP refinement
        t0_sl = remaining * 0.1
        if time.time() - start_all < budget_s - 1.0:
            v0_sl, s0_sl = _slsqp_optimize(best_v, t0_sl)
            if s0_sl < best_score:
                best_v, best_score = v0_sl, s0_sl

    # ---- successive up‑sampling stages -------------------------------
    for factor in (2, 2):  # 190→380→760 (stop if time runs out)
        elapsed = time.time() - start_all
        if elapsed >= budget_s - 1.0:
            break
        n_new = best_n * factor
        if n_new > 2000:   # safety guard
            break
        v_up = _upsample_vector(best_v, n_new)
        # local hill climbing
        t_rem = budget_s - (time.time() - start_all) - 0.5
        if t_rem <= 0:
            break
        t_hc = t_rem * 0.6
        v_opt, s_opt = _hill_climb_opt(v_up, t_hc)
        if s_opt < best_score:
            best_v, best_score = v_opt, s_opt
            best_n = n_new
        # brief SLSQP refinement
        t_rem2 = budget_s - (time.time() - start_all) - 0.5
        if t_rem2 > 0:
            t_sl = t_rem2 * 0.3
            v_sl, s_sl = _slsqp_optimize(best_v, t_sl)
            if s_sl < best_score:
                best_v, best_score = v_sl, s_sl
                best_n = len(best_v)

    # final safety repair
    best_v = _repair_vector(best_v, best_n)
    final_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(final_score), int(best_n)
