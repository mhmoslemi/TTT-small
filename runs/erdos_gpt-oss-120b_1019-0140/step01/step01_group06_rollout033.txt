def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    base_h = globals().get('initial_h_values')
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)
        n = base_h.shape[0]
    else:
        n = 380

    if n % 2 != 0:
        n = n + 1  # ensure even, though may break alignment; but ensure target_sum integer

    target_sum = n // 2
    dx = 2.0 / n
    const = target_sum

    # conversion to binary start
    def init_binary():
        if base_h is not None:
            h_bin = (base_h >= 0.5).astype(float)
            cur = int(h_bin.sum())
            if cur > target_sum:
                # turn some ones to zero
                ones_idx = np.where(h_bin == 1)[0]
                to_zero = rng.choice(ones_idx, size=cur - target_sum, replace=False)
                h_bin[to_zero] = 0.0
            elif cur < target_sum:
                zeros_idx = np.where(h_bin == 0)[0]
                to_one = rng.choice(zeros_idx, size=target_sum - cur, replace=False)
                h_bin[to_one] = 1.0
            return h_bin
        else:
            h = np.zeros(n, dtype=float)
            idx = rng.choice(n, size=target_sum, replace=False)
            h[idx] = 1.0
            return h

    # compute initial crosscorr for binary vector
    def compute_crosscorr(h):
        # using np.correlate
        corr = np.correlate(h, h, mode='full')
        return corr[n-1:]  # length n

    # function to compute c5 from crosscorr
    def c5_from_cross(cross):
        return dx * (const - np.min(cross))

    # Update crosscorr for a position
    def update_crosscorr(cross, pos, delta, h):
        # cross and h are numpy arrays
        # self term
        cross[0] += delta
        # positive lag contributions where pos is first index
        maxk1 = n - 1 - pos
        if maxk1 >= 1:
            # k from 1..maxk1
            cross[1:maxk1+1] += delta * h[pos+1:pos+maxk1+1]
        # contributions where pos is second index (k from 1..pos)
        if pos >= 1:
            # reversed slice of length pos
            cross[1:pos+1] += delta * h[pos-1::-1]

    # Simulate swap (returns new cross not modify original)
    def simulate_swap(h, cross, i, j):
        # copy arrays
        h_new = h.copy()
        cross_new = cross.copy()
        # remove i
        update_crosscorr(cross_new, i, -1, h_new)
        h_new[i] = 0.0
        # add j
        update_crosscorr(cross_new, j, +1, h_new)
        h_new[j] = 1.0
        return h_new, cross_new

    # Local search function
    def local_search(initial_h):
        h_cur = initial_h.copy()
        cross_cur = compute_crosscorr(h_cur)
        cur_c5 = c5_from_cross(cross_cur)
        best_local_h = h_cur.copy()
        best_local_cross = cross_cur.copy()
        best_local_c5 = cur_c5

        T = 0.005
        cooling = 0.9997
        it = 0
        # track number of iterations without improvement for early break?
        no_improve = 0
        while time.time() - start < budget_s * 0.90:
            # random swap
            ones_idx = np.where(h_cur == 1)[0]
            zeros_idx = np.where(h_cur == 0)[0]
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            # simulate swap
            h_new, cross_new = simulate_swap(h_cur, cross_cur, i, j)
            new_c5 = c5_from_cross(cross_new)
            # acceptance
            accept = False
            if new_c5 < cur_c5 - 1e-12:
                accept = True
            else:
                # probability
                prob = np.exp(-(new_c5 - cur_c5) / T) if T > 1e-12 else 0.0
                if rng.random() < prob:
                    accept = True
            if accept:
                h_cur = h_new
                cross_cur = cross_new
                cur_c5 = new_c5
                if cur_c5 < best_local_c5:
                    best_local_c5 = cur_c5
                    best_local_h = h_cur.copy()
                    best_local_cross = cross_cur.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
            # temperature cooling
            T *= cooling
            it += 1

            if no_improve >= 2000:

                if rng.random() < 0.5:
                    # restart from global best (if allowed)
                    h_cur = best_local_h.copy()
                    cross_cur = best_local_cross.copy()
                    cur_c5 = best_local_c5
                else:
                    # random new binary
                    h_cur = random_binary()
                    cross_cur = compute_crosscorr(h_cur)
                    cur_c5 = c5_from_cross(cross_cur)
                no_improve = 0
            # early exit if reach target
            if best_local_c5 <= 0.3808:
                break
        return best_local_h, best_local_cross, best_local_c5

    # random binary generator
    def random_binary():

        h[idx] = 1.0
        return h

    best_cross = None

    # Add a few random starts

import time
import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (sampled on n points) to minimize
    the Erdős overlap constant C₅ = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)
        n = int(base_h.shape[0])
    else:
        n = 380                     # default resolution when no parent is given
    if n % 2 != 0:
        n += 1                      # ensure n/2 is integer

    target_sum = n // 2            # sum_i h_i must equal this (∫h = 1)
    dx = 2.0 / n
    const = target_sum

    # ----- helper utilities ---------------------------------------------
    def compute_c5(h):
        """C₅ for a sampled function h."""
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * (2.0 / n)

    def project_to_feasible(h, target, tol=1e-8, max_iter=20):
        """Project onto [0,1]^n ∩ {sum = target}."""
        h = np.clip(h, 0.0, 1.0)
        for _ in range(max_iter):
            cur = np.sum(h)
            diff = target - cur
            if abs(diff) <= tol:
                break
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size == 0:
                break
            delta = diff / free.size
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        if abs(cur - target) > 1e-6:
            if cur < target:
                room = np.where(h < 1.0)[0]
            else:
                room = np.where(h > 0.0)[0]
            if room.size:
                i = rng.choice(room)
                h[i] = np.clip(h[i] + (target - cur), 0.0, 1.0)
        return h

    def compute_crosscorr(h):
        """Aperiodic autocorrelation of h with itself for non‑negative lags."""
        corr = np.correlate(h, h, mode="full")
        return corr[n - 1 :]  # length n, lag 0..n‑1

    def c5_from_cross(cross):
        """C₅ expressed via autocorrelation."""
        return dx * (const - np.min(cross))

    def update_crosscorr(cross, pos, delta, h):
        """Incrementally update autocorrelation when h[pos] changes by delta (+‑1)."""
        # self‑term
        cross[0] += delta
        # pos as first index (k >= 1)
        maxk1 = n - 1 - pos
        if maxk1 >= 1:
            cross[1 : maxk1 + 1] += delta * h[pos + 1 : pos + maxk1 + 1]
        # pos as second index (k >= 1)
        if pos >= 1:
            cross[1 : pos + 1] += delta * h[pos - 1 :: -1]

    def simulate_swap(h, cross, i, j):
        """Result of swapping a 1 at i with a 0 at j (copies, no side‑effects)."""
        h_new = h.copy()
        cross_new = cross.copy()
        # remove the 1 at i
        update_crosscorr(cross_new, i, -1, h_new)
        h_new[i] = 0.0
        # add a 1 at j
        update_crosscorr(cross_new, j, +1, h_new)
        h_new[j] = 1.0
        return h_new, cross_new

    # ----- construction of binary starting vectors ----------------------
    def binary_from_parent():
        """Turn a provided parent into a binary vector with correct mass."""
        h_bin = (base_h >= 0.5).astype(float)
        cur = int(h_bin.sum())
        if cur > target_sum:
            excess = cur - target_sum
            idx = np.where(h_bin == 1)[0]
            to_zero = rng.choice(idx, size=excess, replace=False)
            h_bin[to_zero] = 0.0
        elif cur < target_sum:
            deficit = target_sum - cur
            idx = np.where(h_bin == 0)[0]
            to_one = rng.choice(idx, size=deficit, replace=False)
            h_bin[to_one] = 1.0
        return h_bin

    def random_binary():
        """Random binary vector with exactly target_sum ones."""
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=target_sum, replace=False)
        h[idx] = 1.0
        return h

    # ----- local stochastic search (hill‑climbing + SA) -----------------
    def local_search(start_h):
        h_cur = start_h.copy()
        cross_cur = compute_crosscorr(h_cur)
        cur_c5 = c5_from_cross(cross_cur)

        best_loc_h = h_cur.copy()
        best_loc_cross = cross_cur.copy()
        best_loc_c5 = cur_c5

        T = 0.005
        cooling = 0.9997
        no_improve = 0

        while time.time() - start < budget_s * 0.90:
            ones = np.flatnonzero(h_cur == 1)
            zeros = np.flatnonzero(h_cur == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

            h_candidate, cross_candidate = simulate_swap(h_cur, cross_cur, i, j)
            cand_c5 = c5_from_cross(cross_candidate)

            if cand_c5 < cur_c5 - 1e-12:
                accept = True
            else:
                prob = np.exp(-(cand_c5 - cur_c5) / T) if T > 1e-12 else 0.0
                accept = rng.random() < prob

            if accept:
                h_cur = h_candidate
                cross_cur = cross_candidate
                cur_c5 = cand_c5
                if cur_c5 < best_loc_c5:
                    best_loc_c5 = cur_c5
                    best_loc_h = h_cur.copy()
                    best_loc_cross = cross_cur.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            T *= cooling

            if no_improve >= 2000:
                # restart: with 50 % probability from current best, otherwise random
                if rng.random() < 0.5:
                    h_cur = best_loc_h.copy()
                    cross_cur = best_loc_cross.copy()
                    cur_c5 = best_loc_c5
                else:
                    h_cur = random_binary()
                    cross_cur = compute_crosscorr(h_cur)
                    cur_c5 = c5_from_cross(cross_cur)
                no_improve = 0

            if best_loc_c5 <= 0.3808:
                break

        return best_loc_h, best_loc_cross, best_loc_c5

    # ----- orchestrate multiple starts ----------------------------------
    candidates = [binary_from_parent() if base_h is not None else random_binary()]
    for _ in range(3):
        candidates.append(random_binary())

    best_h = None
    best_c5 = float("inf")
    best_n = n

    for cand in candidates:
        if time.time() - start > budget_s * 0.90:
            break
        h_sol, cross_sol, c5_val = local_search(cand)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_sol.copy()
        if best_c5 <= 0.3808:
            break

    # ----- fallback if nothing was found --------------------------------
    if best_h is None:
        best_h = random_binary()
        best_c5 = compute_c5(best_h)

    # ----- optional continuous refinement (SLSQP) -----------------------
    if time.time() - start < budget_s * 0.85:
        cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
        bnds = [(0.0, 1.0)] * n
        try:
            res = opt.minimize(
                fun=lambda h: compute_c5(h),
                x0=best_h,
                method="SLSQP",
                bounds=bnds,
                constraints=cons,
                options={"maxiter": 600, "ftol": 1e-9, "disp": False},
            )
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_to_feasible(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # ----- final safety projection ---------------------------------------
    best_h = project_to_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
