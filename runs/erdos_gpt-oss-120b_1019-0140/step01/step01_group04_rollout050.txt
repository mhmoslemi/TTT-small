def anneal(v, time_limit, seed=0):
    np.random.seed(seed)
    start = time.time()
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(v)
    cur_v = v.copy()
    cur_score = best_score
    # pre-heat schedule
    T0 = 0.005 * (2.0 / n)  # scale maybe?
    Tmin = 1e-6
    steps = 0
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        # compute temperature as exponential decay
        T = max(T0 * (0.99 ** steps), Tmin)
        # propose move: random pair
        i, j = np.random.choice(n, 2, replace=False)
        # propose delta from uniform -step to step
        delta = (np.random.rand() - 0.5) * 2 * 0.05  # adjust later
        # try to adjust:
        cand = cur_v.copy()
        cand[i] += delta
        cand[j] -= delta

        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)

def _anneal(v, time_limit, step=0.05):
    n = len(v)
    cur_v = v.copy()
    cur_score = _compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    start = time.time()
    steps = 0
    # temperature schedule: start T proportional to dx maybe
    T0 = 0.01 * (2.0 / n)  # about 0.0001 for n=200
    Tmin = 1e-6
    while time.time() - start < time_limit:
        # temperature decays
        T = max(T0 * (0.995 ** steps), Tmin)
        # Random pair move
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        diff = cand_score - cur_score
        if diff < 0 or np.random.rand() < np.exp(-diff / max(T, 1e-12)):

            if cur_score < best_score:
                best_v, best_score = cur_v.copy(), cur_score

import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project vector v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        free = mask.sum()
        if free == 0:
            break
        adj = diff / free
        v[mask] -= adj
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        # final scaling fallback
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Exact C5 for a discrete step function."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return float(np.max(corr) * dx)

def _upsample_vector(v, n_target):
    """Linear interpolation to a finer grid followed by feasibility repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _improve_worst_shift(v, shift, step, cur_score):
    """Single targeted exchange aimed at increasing the smallest autocorrelation."""
    n = len(v)
    if shift >= 0:
        i_vals = np.arange(0, n - shift)
        partner = i_vals + shift
    else:
        i_vals = np.arange(-shift, n)
        partner = i_vals + shift
    partner_vals = v[partner]

    # choose index to increase
    inc_mask = v[i_vals] < 1.0
    i_best = None
    if np.any(inc_mask):
        inc_idx = i_vals[inc_mask]
        inc_partners = partner_vals[inc_mask]
        i_best = inc_idx[np.argmax(inc_partners)]

    # choose index to decrease
    dec_mask = v > 0.0
    j_best = None
    if np.any(dec_mask):
        dec_candidates = np.where(dec_mask)[0]
        if shift >= 0:
            in_region = (dec_candidates >= 0) & (dec_candidates < n - shift)
            partner_idx = dec_candidates[in_region] + shift
        else:
            in_region = (dec_candidates >= -shift) & (dec_candidates < n)
            partner_idx = dec_candidates[in_region] + shift
        dec_partner_vals = np.empty_like(dec_candidates, dtype=float)
        dec_partner_vals.fill(np.inf)
        dec_partner_vals[in_region] = v[partner_idx]
        dec_partner_vals[~in_region] = 0.0
        j_best = dec_candidates[np.argmin(dec_partner_vals)]

    if i_best is None or j_best is None or i_best == j_best:
        return v, cur_score

    delta = step
    delta = min(delta, 1.0 - v[i_best])
    delta = min(delta, v[j_best])
    if delta <= 0:
        return v, cur_score

    new_v = v.copy()
    new_v[i_best] += delta
    new_v[j_best] -= delta
    new_v = _repair_vector(new_v, n)
    new_score = _compute_c5(new_v)
    if new_score < cur_score:
        return new_v, new_score
    return v, cur_score

def _anneal(v, time_limit, step=0.05):
    """Simulated‑annealing style search with occasional targeted improvement."""
    n = len(v)
    cur_v = v.copy()
    cur_score = _compute_c5(cur_v)
    best_v, best_score = cur_v.copy(), cur_score
    start = time.time()
    steps = 0
    T0 = 0.001
    Tmin = 1e-6

    while time.time() - start < time_limit:
        T = max(T0 * (0.995 ** steps), Tmin)

        # random exchange move
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        diff = cand_score - cur_score
        if diff < 0 or np.random.rand() < np.exp(-diff / max(T, 1e-12)):
            cur_v, cur_score = cand, cand_score
            if cur_score < best_score:
                best_v, best_score = cur_v.copy(), cur_score

        # every few steps try a deterministic worst‑shift improvement
        if steps % 50 == 0:
            corr = np.correlate(cur_v, cur_v, mode='full')
            min_idx = np.argmin(corr)
            shift = min_idx - (n - 1)
            cur_v, cur_score = _improve_worst_shift(cur_v, shift, step, cur_score)
            if cur_score < best_score:
                best_v, best_score = cur_v.copy(), cur_score

        steps += 1

    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------ #
    # Acquire initial parent construction if supplied.
    try:
        parent = np.array(initial_h_values, dtype=np.float64)  # pre‑imported
        n0 = len(parent)
    except Exception:
        n0 = 190
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0 // 2] = 1.0
    parent = _repair_vector(parent, n0)

    best_v = parent.copy()
    best_score = _compute_c5(best_v)

    # resolution schedule (keep under 800 points)
    sched = [n0]
    if n0 * 2 <= 800:
        sched.append(n0 * 2)
    if n0 * 4 <= 800:
        sched.append(n0 * 4)

    # reserve a little buffer for final repairs/returns
    reserve = min(2.0, 0.02 * budget_s)

    for idx, n in enumerate(sched):
        elapsed = time.time() - start
        if elapsed >= budget_s - reserve:
            break
        # time slice for this resolution
        remaining = budget_s - elapsed - reserve
        slice_time = remaining / (len(sched) - idx)

        # initial guess at this resolution
        if len(best_v) != n:
            init = _upsample_vector(best_v, n)
        else:
            init = best_v.copy()

        # main annealing stage
        v_opt, sc_opt = _anneal(init, time_limit=slice_time * 0.85, step=0.05)
        if sc_opt < best_score:
            best_v, best_score = v_opt, sc_opt

        # a few quick random restarts within the leftover slice
        left = budget_s - (time.time() - start) - reserve
        if left > 0.4:
            restarts = int(left // 0.4)
            for _ in range(restarts):
                rnd = _random_feasible(n)
                v_r, sc_r = _anneal(rnd, time_limit=0.35, step=0.05)
                if sc_r < best_score:
                    best_v, best_score = v_r, sc_r

    # final safety repair and evaluation
    best_v = _repair_vector(best_v, len(best_v))
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(len(best_v))
