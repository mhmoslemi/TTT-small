import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Shift values (in-place) to hit an exact sum while staying in [0,1]."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    max_iter = 500
    it = 0
    while it < max_iter and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0:
                h[idx] -= dec
                diff += dec
        it += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

    if n % 2 != 0:

    target_sum = n // 2          # exact number of "ones"

    milp_ok = False
    try:

        t = cp.Variable()                     # objective: max overlap
