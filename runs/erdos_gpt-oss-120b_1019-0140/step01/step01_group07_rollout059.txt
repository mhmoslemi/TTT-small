def compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0/n
    a = h
    b = 1.0 - h
    b_rev = b[::-1]
    L = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b_rev, L)
    conv = np.fft.irfft(fa * fb, L)
    # conv length L, we need first 2*n-1 values
    conv = conv[:2*n-1]
    corr = conv[::-1]  # correlation as per np.correlate
    # The maximum correlation
    max_corr = np.max(corr)
    return max_corr * dx

def ga_optimize(n, init_vec, target_sum, time_remaining, start_time):
    pop_size = 30
    max_gens = 200 (but early stop)
    sigma = 0.05
    crossover_prob = 0.7
    mutation_prob = 0.5  # each child mutated maybe
    tournament_size = 3

    # Initialize population
    pop = []
    # include init_vec (projected correct)
    if init_vec is not None:
        base_vec = _project_to_constraints(init_vec, target_sum)
        pop.append({'vec': base_vec, 'score': compute_c5_fft(base_vec)})
    while len(pop) < pop_size:
        v = np.random.rand(n)
        v = _project_to_constraints(v, target_sum)
        pop.append({'vec': v, 'score': compute_c5_fft(v)})

    # Keep best
    best = min(pop, key=lambda x: x['score'])
    for gen in range(max_gens):
        # Check time budget
        if time.time() - start_time > time_remaining * 0.95:
            break
        new_pop = []
        while len(new_pop) < pop_size:
            # tournament selection
            idxs = np.random.choice(len(pop), tournament_size, replace=False)
            p1 = pop[idxs[np.argmin([pop[i]['score'] for i in idxs])]]['vec']
            idxs = np.random.choice(len(pop), tournament_size, replace=False)
            p2 = pop[idxs[np.argmin([pop[i]['score'] for i in idxs])]]['vec']
            child = p1.copy()
            if np.random.rand() < crossover_prob:
                # two-point crossover
                pt1 = np.random.randint(1, n-1)
                pt2 = np.random.randint(pt1, n-1)
                child = np.concatenate([p1[:pt1], p2[pt1:pt2], p1[pt2:]])
            # mutation
            if np.random.rand() < mutation_prob:
                child += np.random.randn(n) * sigma
            child = _project_to_constraints(child, target_sum)
            new_pop.append({'vec': child, 'score': compute_c5_fft(child)})
        # combine and select best
        combined = pop + new_pop
        combined.sort(key=lambda x: x['score'])
        pop = combined[:pop_size]
        # update best
        if pop[0]['score'] < best['score']:
            best = pop[0].copy()
            # optional early termination if meets target
            if best['score'] < 0.3808:
                break
    return best

def ga_optimize(n, init_vec, target_sum, start_time, budget_s):
    # allocate time: let's compute time left
    # we will stop when time left <= 5 sec
    # Determine pop_size, max_gens based on n (like smaller n larger gen).
    pop_size = min(40, max(10, n // 10))   # for n=380 => pop=38; ok.
    max_gens = 200
    sigma = 0.05
    crossover_prob = 0.7
    mutation_prob = 0.8
    tournament_size = 3

    # Initialize population
    pop = []
    base = _project_to_constraints(init_vec, target_sum)
    pop.append({'vec': base, 'score': compute_c5_fft(base)})

    # Fill remaining population with random individuals
    while len(pop) < pop_size:
        v = np.random.rand(n)
        v = _project_to_constraints(v, target_sum)
        pop.append({'vec': v, 'score': compute_c5_fft(v)})

    # best record
    best = min(pop, key=lambda x: x['score'])
    # main loop
    for gen in range(max_gens):
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break
        # generate children
        children = []
        for _ in range(pop_size):
            # tournament selection
            idxs = np.random.choice(len(pop), tournament_size, replace=False)
            p1 = pop[idxs[np.argmin([pop[i]['score'] for i in idxs])]]['vec']
            idxs = np.random.choice(len(pop), tournament_size, replace=False)
            p2 = pop[idxs[np.argmin([pop[i]['score'] for i in idxs])]]['vec']
            child = p1.copy()
            if np.random.rand() < crossover_prob:
                # two-point crossover
                pt1 = np.random.randint(1, n-1)
                pt2 = np.random.randint(pt1, n-1)
                child = np.concatenate([p1[:pt1], p2[pt1:pt2], p1[pt2:]])
            if np.random.rand() < mutation_prob:
                child += np.random.randn(n) * sigma
            child = _project_to_constraints(child, target_sum)
            children.append({'vec': child, 'score': compute_c5_fft(child)})
        # combine and select top pop_size
        combined = pop + children
        combined.sort(key=lambda x: x['score'])
        pop = combined[:pop_size]
        # update best
        if pop[0]['score'] < best['score']:
            best = {'vec': pop[0]['vec'].copy(), 'score': pop[0]['score']}
            if best['score'] < 0.3808:
                break

        # optionally adapt sigma (reduce gradually)
        if gen % 20 == 0 and sigma > 0.01:
            sigma *= 0.9

    return best

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    # water fill
    if cur < target:
        need = target - cur
        # increase values up to 1.0, prioritize those with most capacity
        free = np.where(v < 1.0)[0]
        if free.size > 0:
            caps = 1.0 - v[free]
            # sort descending by capacity
            order = np.argsort(caps)[::-1]
            for idx in order:
                i = free[idx]
                add = min(need, caps[idx])
                v[i] += add
                need -= add
                if need <= 1e-12:
                    break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        if free.size > 0:
            caps = v[free]
            order = np.argsort(caps)[::-1]
            for idx in order:
                i = free[idx]
                sub = min(excess, caps[idx])
                v[i] -= sub
                excess -= sub
                if excess <= 1e-12:
                    break
    v = np.clip(v, 0.0, 1.0)
    # If still off by small amount, scale uniformly
    cur = np.sum(v)
    if abs(cur - target) > 1e-9 and cur > 0.0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    a = h
    b = 1.0 - h
    b_rev = b[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b_rev, L)
    conv = np.fft.irfft(fa * fb, L)
    conv = conv[:2 * n - 1]
    corr = conv[::-1]
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust to satisfy sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:  # need to add mass
        need = target - cur
        free = np.where(v < 1.0)[0]
        if free.size:
            caps = 1.0 - v[free]
            order = np.argsort(caps)[::-1]
            for idx in order:
                i = free[idx]
                add = min(need, caps[idx])
                v[i] += add
                need -= add
                if need <= 1e-12:
                    break
    else:  # excess mass, need to subtract
        excess = cur - target
        free = np.where(v > 0.0)[0]
        if free.size:
            caps = v[free]
            order = np.argsort(caps)[::-1]
            for idx in order:
                i = free[idx]
                sub = min(excess, caps[idx])
                v[i] -= sub
                excess -= sub
                if excess <= 1e-12:
                    break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-9 and cur > 0.0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

        v = _project_to_constraints(v, target_sum)
        pop.append({"vec": v, "score": _compute_c5(v)})

    best = min(pop, key=lambda x: x["score"])
    for gen in range(max_gens):
        if time.time() - start_time > budget_s * 0.95:
            break
        children = []
        for _ in range(pop_size):
            p1 = _select_parent(pop, tournament_size)
            p2 = _select_parent(pop, tournament_size)
            child = p1.copy()
            if np.random.rand() < cx_pb:
                # two‑point crossover
                c1 = np.random.randint(1, n - 2)
                c2 = np.random.randint(c1 + 1, n - 1)
                child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
            if np.random.rand() < mut_pb:
                child += np.random.randn(n) * sigma
            child = _project_to_constraints(child, target_sum)
            children.append({"vec": child, "score": _compute_c5(child)})

        combined = pop + children
        combined.sort(key=lambda x: x["score"])
        pop = combined[:pop_size]

        if pop[0]["score"] < best["score"]:
            best = {"vec": pop[0]["vec"].copy(), "score": pop[0]["score"]}
            if best["score"] < 0.3808:
                break

        # slowly reduce mutation strength
        if gen % 20 == 0 and sigma > 0.02:
            sigma *= 0.9

    return best


def _local_refine(vec, target_sum, start_time, budget_s):
    """SLSQP refinement starting from vec."""
    if time.time() - start_time > budget_s * 0.95:
        return vec
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * vec.size
    try:
        res = minimize(_compute_c5, vec, method="SLSQP",
                       bounds=bounds, constraints=[cons],
                       options={"maxiter": 300, "ftol": 1e-9, "disp": False})
    except Exception:
        return vec
    return _project_to_constraints(res.x, target_sum)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    init = globals().get("initial_h_values", None)
    if init is not None:
        init = np.asarray(init, dtype=np.float64).ravel()
        n0 = init.size
    else:
        n0 = 380
        init = np.random.rand(n0)

    # start with base resolution
    resolutions = [n0]
    if budget_s > 650:
        resolutions.append(n0 * 2)
    if budget_s > 950:
        resolutions.append(n0 * 4)

    best_vec = None
    best_score = float("inf")
    best_n = None

    for n in resolutions:
        # time check before new resolution
        if time.time() - start_time > budget_s * 0.95:
            break
        target = n / 2.0

        if best_vec is not None:
            # upscale previous best
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            start_vec = np.interp(x_new, x_old, best_vec)
        elif n == n0:
            start_vec = init.copy()
        else:
            # pure interpolation from initial if no previous best
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            start_vec = np.interp(x_new, x0, init)

        start_vec = _project_to_constraints(start_vec, target)

        # GA phase (size and generations modestly tuned to n)
        pop_sz = max(20, min(50, n // 10))
        max_gen = 200 if n == n0 else 120
        best_local = _ga_optimize(
            n, start_vec, target, start_time, budget_s,
            max_gens=max_gen, pop_size=pop_sz,
        )
        vec = best_local["vec"]
        score = best_local["score"]

        # Local SLSQP refinement
        vec_ref = _local_refine(vec, target, start_time, budget_s)
        score_ref = _compute_c5(vec_ref)
        if score_ref < score:
            vec, score = vec_ref, score_ref

        # record global best
        if score < best_score:
            best_score = float(score)
            best_vec = vec.copy()
            best_n = n

        if best_score < 0.3808:
            break

    # fallback
    if best_vec is None:
        best_vec = _project_to_constraints(init, n0 / 2.0)
        best_score = _compute_c5(best_vec)
        best_n = n0

    # final evaluation (use provided evaluator if available)
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_vec))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_vec.astype(np.float64), final_score, int(best_n)
