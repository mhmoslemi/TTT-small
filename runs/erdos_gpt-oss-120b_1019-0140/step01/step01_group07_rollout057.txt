def interpolate_solution(old_h, new_n):
    old_n = old_h.size
    x_old = np.linspace(0, 2, old_n, endpoint=False)  # positions for samples
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, old_h)
```

Note: The [0,2) domain with step dx. Use endpoint=False to avoid duplicate at 2.

    target_bound = 0.3808

    init_h = globals().get('initial_h_values')

            pop.append(cand)
            fitness.append(_compute_c5(cand))
        # Keep best
        best_idx = np.argmin(fitness)
        best_h_local = pop[best_idx].copy()
        best_score_local = fitness[best_idx]
        # DE loop
        max_generations = 5000  # but limited by time
        F = 0.8
        CR = 0.9
        generation = 0
        while True:
            # time check: if less than 0.3s left, break

            if elapsed > budget_s * 0.95:
                break
            if best_score_local < target_bound:
                break
            if generation >= max_generations:
                break

            for i in range(pop_size):

import numpy as np
import time

def _compute_c5(h):
    """Maximum overlap integral for discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust sum(v) to target using water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 0:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 0:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-9 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _interp_to_n(old_h, n_new):
    """Linear interpolation of old_h to length n_new on [0,2)."""
    n_old = old_h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, old_h)

    init_vec = globals().get("initial_h_values")

    else:
        n0 = 380
        init_vec = np.random.rand(n0)
    target_sum0 = n0 / 2.0
    init_vec = _project_to_constraints(init_vec, target_sum0)

    # resolutions to try (doubling while time permits)
    resolutions = [n0]
    if budget_s > 600:
        resolutions.append(n0 * 2)

        resolutions.append(n0 * 4)

    best_h = None
    best_score = float("inf")

    for n in resolutions:
        # prepare starting vector for this resolution
        if best_h is None:
            h0 = init_vec
        else:
            h0 = _interp_to_n(best_h, n)
        target_sum = n / 2.0
        h0 = _project_to_constraints(h0, target_sum)

        pop_size = max(15, min(30, n // 28))
        population = []

        half = pop_size // 2

        best_idx = int(np.argmin(fitness))

            except Exception:

            best_score = best_local_score
            best_h = best_local.copy()
        if best_score < target_bound:

        best_h = init_vec

        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_h.size)
