def gradient_move(h, step, target, rng):
    n = h.size
    # compute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_idx = int(np.argmax(corr))
    k_shift = k_idx - (n - 1)
    g = np.zeros_like(h)
    if k_shift == 0:
        # derivative of sum_i h_i (1 - h_i) = sum h_i - sum h_i^2
        g = 1.0 - 2.0 * h

        g[i_range] += 1.0 - h[i_range + k_shift]
        g[i_range + k_shift] -= h[i_range]

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k))dx approximated discretely."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with 1 - h
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Adjust array h in-place so its sum equals target while staying within [0,1]."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    max_attempts = 2000
    attempts = 0
    while abs(diff) > 1e-12 and attempts < max_attempts:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # distribute remaining tiny error uniformly, then clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_move(h: np.ndarray, step: float, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Perform a gradient descent step aimed at the worst shift."""
    n = h.size
    # correlation to locate worst shift (max correlation)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_idx = int(np.argmax(corr))
    k_shift = k_idx - (n - 1)

    g = np.zeros_like(h)
    if k_shift == 0:
        # derivative of sum_i h_i*(1-h_i)
        g = 1.0 - 2.0 * h
    elif k_shift > 0:
        i_range = np.arange(0, n - k_shift)
        g[i_range] += 1.0 - h[i_range + k_shift]
        g[i_range + k_shift] -= h[i_range]
    else:  # k_shift < 0
        shift = -k_shift
        i_range = np.arange(shift, n)
        g[i_range] += 1.0 - h[i_range - shift]
        g[i_range - shift] -= h[i_range]

    # project gradient to hyperplane sum(g)=0
    g -= g.mean()
    cand = h - step * g
    np.clip(cand, 0.0, 1.0, out=cand)
    cand = adjust_sum_to_target(cand, target, rng)
    return cand

def pairwise_move(h: np.ndarray, step: float, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Swap a small amount of mass between two positions while preserving the total sum."""
    n = h.size
    i, j = rng.randint(n, size=2)
    if i == j:
        # try a few times to get distinct indices
        for _ in range(5):
            i, j = rng.randint(n, size=2)
            if i != j:
                break
        if i == j:
            return h.copy()
    hi, hj = h[i], h[j]
    # feasible delta so that both stay in [0,1]
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if low >= high:
        return h.copy()
    delta = (rng.rand() * 2.0 - 1.0) * step
    delta = np.clip(delta, low, high)
    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    # Sum unchanged (up to rounding); enforce exactness
    cand = adjust_sum_to_target(cand, target, rng)
    return cand

    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # Determine problem size
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n_points = init.size
        target_sum = n_points / 2.0
        h0 = adjust_sum_to_target(init.copy(), target_sum, rng)
    else:
        n_points = 70
        target_sum = n_points / 2.0
        h0 = rng.rand(n_points)
        # scale to meet integral constraint
        h0 = h0 / h0.sum() * target_sum
        h0 = adjust_sum_to_target(h0, target_sum, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    current_h = best_h.copy()
    current_c5 = best_c5

    # Annealing schedule parameters
    T_start = 5e-2
    T_end = 1e-5
    step_start = 5e-1
    step_end = 1e-4
    grad_prob = 0.7

    no_improve = 0
    no_improve_limit = 30000

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        elapsed = now - start
        frac = elapsed / max_time  # 0 -> 1

        # linearly interpolated schedule
        T = T_start * (1.0 - frac) + T_end * frac
        step = step_start * (1.0 - frac) + step_end * frac

        # generate candidate
        if rng.rand() < grad_prob:
            candidate = gradient_move(current_h, step, target_sum, rng)
        else:
            candidate = pairwise_move(current_h, step, target_sum, rng)

        cand_c5 = compute_c5(candidate)

        # Acceptance test
        if cand_c5 < current_c5:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_c5 - current_c5) / max(T, 1e-12))
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                no_improve = 0
                # early exit if target achieved
                if best_c5 <= 0.3808:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional restart to avoid stagnation
        if no_improve >= no_improve_limit:
            # restart from best with a small perturbation
            current_h = best_h.copy()
            # add gaussian noise scaled by step
            noise = rng.normal(scale=step, size=n_points)
            current_h = np.clip(current_h + noise, 0.0, 1.0)
            current_h = adjust_sum_to_target(current_h, target_sum, rng)
            current_c5 = compute_c5(current_h)
            no_improve = 0

    # final safety adjustment
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
