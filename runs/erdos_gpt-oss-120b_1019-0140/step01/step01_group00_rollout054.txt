def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

import time
import math
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 for a given (binary or continuous) h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _random_binary(n, ones, rng):
    """Generate a binary vector of length n with exactly `ones` ones."""
    vec = np.zeros(n, dtype=int)
    if ones > 0:
        idx = rng.choice(n, ones, replace=False)
        vec[idx] = 1
    return vec

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality constraint: t >= overlap(k) * dx."""
    def __init__(self, shift, n):
        self.shift = shift
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        t = x[-1]
        k = self.shift
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return t - overl * self.dx

def _project_bounds_sum(v, target):
    """Project vector v onto [0,1] while fixing sum(v)=target (approx)."""
    v = np.clip(v, 0.0, 1.0)
    # simple water‑filling: add uniform offset, clip, repeat a few times
    for _ in range(5):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise problem size and a starting solution ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        # compute its C5 directly (may be fractional)
        best_score = compute_c5(h0)
        best_h = h0.copy()
        # also keep a binary version for the GA
        binary_start = np.zeros(n, dtype=int)
        if n // 2 > 0:
            top_idx = np.argpartition(-h0, n // 2 - 1)[: n // 2]
            binary_start[top_idx] = 1
    else:
        n = 80
        target_sum = n / 2.0
        binary_start = _random_binary(n, n // 2, rng)
        best_h = binary_start.astype(float)
        best_score = compute_c5(best_h)

    # ---- Genetic algorithm on binary vectors ----
    popsize = 120
    tournament = 3
    mutate_prob = 0.2
    half_ones = n // 2

    # initialise population (list of int arrays) and fitness
    population = [binary_start.copy()]
    fitness = [compute_c5(binary_start)]
    # fill the rest randomly
    while len(population) < popsize:
        ind = _random_binary(n, half_ones, rng)
        val = compute_c5(ind)
        population.append(ind)
        fitness.append(val)
        if val < best_score:
            best_score = val
            best_h = ind.astype(float)

    # GA main loop – use ~60 % of total budget
    ga_deadline = start_time + budget_s * 0.6
    while time.time() < ga_deadline:
        children = []
        children_fitness = []
        for _ in range(popsize):
            # tournament selection for two parents
            idxs = rng.choice(popsize, tournament, replace=False)
            p1 = population[int(idxs[np.argmin(np.take(fitness, idxs))])]
            idxs = rng.choice(popsize, tournament, replace=False)
            p2 = population[int(idxs[np.argmin(np.take(fitness, idxs))])]

            # uniform crossover
            mask = rng.random(n) < 0.5
            child = np.where(mask, p1, p2).astype(int)

            # repair to exact number of ones
            cur = int(child.sum())
            if cur != half_ones:
                if cur > half_ones:

                    ones = np.flatnonzero(child)
                    drop = rng.choice(ones, excess, replace=False)
                    child[drop] = 0
                else:
                    deficit = half_ones - cur
                    zeros = np.flatnonzero(child == 0)
                    inc = rng.choice(zeros, deficit, replace=False)
                    child[inc] = 1

                    zeros = np.flatnonzero(child == 0)
                    if len(ones) == 0 or len(zeros) == 0:

                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    child[i] = 0
                    child[j] = 1

            val = compute_c5(child)
            children.append(child)
            children_fitness.append(val)

                best_score = val
                best_h = child.astype(float)

        combined = population + children
        combined_fit = fitness + children_fitness
        order = np.argsort(combined_fit)[:popsize]
        population = [combined[i] for i in order]

    # ---- Continuous refinement with SLSQP (if time permits) ----
    now = time.time()

        constraints = [{'type': 'eq', 'fun': sum_con}]

                x0=x0,
                method='SLSQP',

                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}

            if res.success:

                # enforce exact sum (tiny uniform adjustment)
                cur_sum = h_opt.sum()
                if abs(cur_sum - target_sum) > 1e-9:
                    h_opt = _project_bounds_sum(h_opt, target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt + 1e-12 < best_score:
                    best_score = c5_opt
                    best_h = h_opt
        except Exception:
            pass  # ignore any optimizer failure

    # ---- Final binary hill‑climb / simulated annealing ----
    now = time.time()
    while now < start_time + budget_s - 0.3:
        # work with a binary copy for easy swaps
        cur = (best_h > 0.5).astype(int)
        cur_score = best_score
        # temperature schedule (simple exponential cooling)
        T = 0.02
        # number of inner attempts limited by remaining time
        inner_deadline = start_time + budget_s - 0.15
        while time.time() < inner_deadline:
            ones_idx = np.flatnonzero(cur)
            zeros_idx = np.flatnonzero(cur == 0)
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            cur[i] = 0
            cur[j] = 1
            new_val = compute_c5(cur)
            delta = new_val - cur_score
            if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
                cur_score = new_val
                if new_val + 1e-12 < best_score:
                    best_score = new_val
                    best_h = cur.astype(float)
            else:
                # revert
                cur[i] = 1
                cur[j] = 0
            T *= 0.9995
            if T < 1e-6:
                T = 0.0
        now = time.time()

    # ---- final safety projection ----
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = _project_bounds_sum(best_h, target_sum)

    return best_h, best_score, n
