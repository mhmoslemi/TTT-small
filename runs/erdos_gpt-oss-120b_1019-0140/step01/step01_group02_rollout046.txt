   def compute_c5(h):
       # use FFT to compute convolution h * (1 - h) full.
       # Let's compute length = 2*n-1.
       size = 1
       while size < 2*n:
           size <<= 1
       # pad h and (1-h) to size.
       H = np.fft.rfft(h, n=size)
       G = np.fft.rfft(1 - h, n=size)
       conv = np.fft.irfft(H * G, n=size)
       # conv contains linear convolution length size; the full conv of length 2*n -1 is at indices [:2*n-1].
       corr = conv[:2*n-1]  # shape (2*n-1,)
       # Multiply by dx and return max.
       return np.max(corr) * dx, corr

def gradient_at_lag(h, lag):
    # lag index in range 0..2*n-2 (where lag_center = n-1). Convert to shift ℓ = lag - (n-1)
    n = len(h)
    shift = lag - (n - 1)
    # Compute g_i = (1 - h[i + shift]) - (h[i - shift])
    # For i+shift out-of-range, term is 0.
    # For i-shift out-of-range, term is 0.
    g = np.zeros_like(h)
    if shift >= 0:
        # i+shift within range for i < n - shift
        # i-shift within range for i >= shift
        # For i from 0 to n - shift - 1, have (1 - h[i+shift])
        g[:n - shift] += (1 - h[shift:])  # indexing: h[i+shift] for i from 0 -> n-shift-1 corresponds to h[shift:].
        # For i from shift to n-1, have - h[i-shift]
        g[shift:] -= h[:n - shift]
    else:
        # shift negative, s = -shift > 0
        s = -shift
        # i+shift = i - s within range for i >= s
        # i-shift = i + s within range for i < n - s
        # For i from s to n-1: (1 - h[i-s])
        g[s:] += (1 - h[:n - s])
        # For i from 0 to n - s -1: - h[i + s]
        g[:n - s] -= h[s:]
    return g

import time
import math
import numpy as np

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust entries of h to satisfy sum constraint exactly."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _next_pow2(x: int) -> int:
    """Small utility for FFT length."""
    p = 1
    while p < x:
        p <<= 1
    return p

def _full_corr_fft(h: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear correlation of h with (1-h) using FFT."""
    H = np.fft.rfft(h, n=fft_len)
    G = np.fft.rfft(1.0 - h, n=fft_len)
    conv = np.fft.irfft(H * G, n=fft_len)
    n = h.shape[0]
    return conv[: 2 * n - 1]

def _c5_from_corr(corr: np.ndarray, dx: float) -> float:
    return float(corr.max() * dx)

def _gradient_at_lag(h: np.ndarray, lag_idx: int) -> np.ndarray:
    """
    Gradient of the correlation value at a given lag (full mode index)
    with respect to the entries of h.
    """
    n = h.shape[0]
    shift = lag_idx - (n - 1)  # shift in index units
    g = np.zeros_like(h)
    if shift >= 0:
        # (1 - h[i+shift]) term
        if shift < n:
            g[: n - shift] += 1.0 - h[shift:]
        # -h[i-shift] term
        if shift > 0:
            g[shift:] -= h[: n - shift]
    else:
        s = -shift
        # (1 - h[i-s]) term
        if s < n:
            g[s:] += 1.0 - h[: n - s]
        # -h[i+s] term
        if s > 0:
            g[: n - s] -= h[s:]
    return g

def _compute_c5(h: np.ndarray, dx: float, fft_len: int) -> float:
    corr = _full_corr_fft(h, fft_len)
    return _c5_from_corr(corr, dx)
