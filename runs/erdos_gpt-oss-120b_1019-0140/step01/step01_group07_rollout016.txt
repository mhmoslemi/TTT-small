import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _compute_c5(h):
    """Maximum of ∫ h(x)*(1-h(x+k)) dx over integer shifts k,
    using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _overlaps(h):
    """Return the full overlap array ∫ h(x)*h(x+k) dx for all integer k."""
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, h, mode="full") * dx


def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust to satisfy sum(v)=target (water-filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        # need to add mass, fill up from the most empty entries
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # fill largest capacities first
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 0:
                break
    else:
        # need to remove excess mass, drain from the fullest entries
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 0:
                break
    # final safety scaling (rarely needed)
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-9 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v


def _shift(arr, lag):
    """Shift array by integer lag with zero padding (no wrap)."""
    n = arr.size
    if lag == 0:
        return arr.copy()
    if lag > 0:
        out = np.empty_like(arr)
        out[:lag] = 0.0
        out[lag:] = arr[: n - lag]
        return out
    # lag < 0
    L = -lag
    out = np.empty_like(arr)
    out[-L:] = 0.0
    out[: n - L] = arr[L:]
    return out
