def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # get parent vector
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 150
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0//2] = 1.0
    parent = _repair_vector(parent, n0)
    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    # Determine resolution schedule
    resolutions = [n0]
    # escalate up to 760 at most
    if n0 * 2 <= 800:
        resolutions.append(n0 * 2)
    if n0 * 4 <= 800:
        resolutions.append(n0 * 4)

    total_time = budget_s
    # allocate time for each resolution equally
    for idx, n in enumerate(resolutions):
        elapsed = time.time() - start
        remaining_total = total_time - elapsed
        if remaining_total <= 0.5:
            break
        # allocate portion of remaining time to this resolution
        # we can allocate proportionally: remaining_total / (len(resolutions)-idx)
        time_for_res = remaining_total / (len(resolutions) - idx)
        # upsamples if n differs
        if len(best_v) != n:
            init_guess = _upsample_vector(best_v, n)
        else:
            init_guess = best_v.copy()
        # run gradient descent with most of time
        gd_time = time_for_res * 0.6
        if gd_time > 0:
            v_opt, sc_opt = _gradient_descent(init_guess, gd_time, init_step=0.5)
            if sc_opt < best_score:
                best_score, best_v, best_n = sc_opt, v_opt, n
        # remaining time for local search or random restarts
        rem = time_for_res - gd_time
        if rem > 0:
            # local search as before
            v_ls, sc_ls = _local_search(best_v, best_n, rem)
            if sc_ls < best_score:
                best_score, best_v = sc_ls, v_ls
        # also maybe perform a few random restarts of gradient descent if time left
        # loop while we have leftover > some min
        while True:
            elapsed = time.time() - start
            remaining = total_time - elapsed
            if remaining < 5.0:
                break
            # allocate small chunk maybe 30 seconds
            chunk = min(30.0, remaining)
            rnd = _random_feasible(best_n)
            v_opt2, sc_opt2 = _gradient_descent(rnd, chunk, init_step=0.5)
            if sc_opt2 < best_score:
                best_score, best_v = sc_opt2, v_opt2
    # final repair and evaluate
    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)
    return best_v.astype(np.float64), float(best_score), int(best_n)

import time, numpy as np

def _repair_vector(v, n, max_iter=15):
    # clip then adjust sum
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def _max_corr_and_shifts(v):
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    max_c = corr.max()
    eps = 1e-12
    idx = np.where(np.abs(corr - max_c) <= eps)[0]
    shifts = idx - (n - 1)
    return max_c, shifts

def _grad_for_shifts(v, shifts):
    n = len(v)
    if len(shifts) == 0:
        return np.zeros_like(v)
    # start with constant term: +1 per shift for each component
    g = np.full(n, float(len(shifts)), dtype=np.float64)
    for k in shifts:
        if k >= 0:
            if k < n:
                g[: n - k] -= v[k:]
                g[k:] -= v[: n - k]
        else:
            k2 = -k
            if k2 < n:
                g[k2:] -= v[: n - k2]
                g[: n - k2] -= v[k2:]
    return g

def _projected_step(v, direction, step):
    u = v - step * direction
    u = np.clip(u, 0.0, 1.0)
    return _repair_vector(u, len(v))

def _gradient_descent(v, time_limit, init_step=0.5, min_step=1e-6):
    start = time.time()
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(v)
    cur_score = best_score
    step = init_step
    while time.time() - start < time_limit:
        # compute max shift(s)
        max_c, shifts = _max_corr_and_shifts(v)
        # gradient of correlation (unscaled)
        g = _grad_for_shifts(v, shifts)
        norm = np.linalg.norm(g)
        if norm == 0:
            # add small random move
            v = _repair_vector(v + 0.01 * (np.random.rand(n) - 0.5), n)
            cur_score = _compute_c5(v)
            continue
        direction = g / norm
        accepted = False
        # try decreasing step if needed
        for _ in range(8):
            v_new = _projected_step(v, direction, step)
            new_score = _compute_c5(v_new)
            if new_score <= cur_score + 1e-12:
                v = v_new
                cur_score = new_score
                if new_score < best_score:
                    best_score, best_v = new_score, v_new.copy()
                accepted = True
                break
            else:
                step *= 0.5
        if not accepted:
            # small random perturbation to escape
            v = _repair_vector(v + 0.01 * (np.random.rand(n) - 0.5), n)
            cur_score = _compute_c5(v)
        # gradually reduce step
        step = max(step * 0.97, min_step)
    return best_v, best_score

def _upsample_vector(v, n_target):
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _local_search(v, n, time_limit):
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score, best_v = score, cand
    return best_v, best_score

import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n and enforce sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """C5 value for a discrete step function."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def _max_corr_and_shifts(v):
    """Return the maximal unscaled correlation and the shifts attaining it."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    max_c = corr.max()
    eps = 1e-12
    idx = np.where(np.abs(corr - max_c) <= eps)[0]
    shifts = idx - (n - 1)
    return max_c, shifts

def _grad_for_shifts(v, shifts):
    """Gradient of the (unscaled) max correlation w.r.t. v."""
    n = len(v)
    if not shifts:
        return np.zeros_like(v)
    g = np.full(n, float(len(shifts)), dtype=np.float64)  # constant term +1 per shift
    for k in shifts:
        if k >= 0:
            if k < n:
                g[: n - k] -= v[k:]
                g[k:] -= v[: n - k]
        else:
            k2 = -k
            if k2 < n:
                g[k2:] -= v[: n - k2]
                g[: n - k2] -= v[k2:]
    return g

def _projected_step(v, direction, step):
    u = v - step * direction
    u = np.clip(u, 0.0, 1.0)
    return _repair_vector(u, len(v))

def _gradient_descent(v, time_limit, init_step=0.5, min_step=1e-6):
    """Projected gradient descent on the max overlap objective."""
    start = time.time()
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(v)
    cur_score = best_score
    step = init_step
    while time.time() - start < time_limit:
        max_c, shifts = _max_corr_and_shifts(v)
        g = _grad_for_shifts(v, shifts)
        norm = np.linalg.norm(g)
        if norm == 0.0:
            v = _repair_vector(v + 0.01 * (np.random.rand(n) - 0.5), n)
            cur_score = _compute_c5(v)
            continue
        direction = g / norm
        accepted = False
        for _ in range(8):
            v_new = _projected_step(v, direction, step)
            new_score = _compute_c5(v_new)
            if new_score <= cur_score + 1e-12:
                v = v_new
                cur_score = new_score
                if new_score < best_score:
                    best_score, best_v = new_score, v_new.copy()
                accepted = True
                break
            step *= 0.5
        if not accepted:
            v = _repair_vector(v + 0.01 * (np.random.rand(n) - 0.5), n)
            cur_score = _compute_c5(v)
        step = max(step * 0.97, min_step)
    return best_v, best_score

def _upsample_vector(v, n_target):
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _local_search(v, n, time_limit):
    """Simple hill‑climbing exchanging small amounts between two coordinates."""
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score, best_v = score, cand
    return best_v, best_score
