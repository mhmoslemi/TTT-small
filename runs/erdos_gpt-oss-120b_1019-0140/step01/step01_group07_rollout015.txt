import numpy as np, time, math, sys
from scipy.optimize import minimize

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # max over all k
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    # water-filling adjust
    if cur < target:
        need = target - cur
        # increase values where v < 1
        # order: choose those with largest capacity (1-v)
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def gradient_refine(h, time_budget, start_time):
    # Use SLSQP with random perturbations to attempt improvement
    n = h.size
    target_sum = n/2.0
    best_h = h.copy()
    best_score = _compute_c5(best_h)
    # limit number of restarts based on time left

        now = time.time()
        if now - start_time >= time_budget:

        # perturb

        x0 = best_h + perturb
        x0 = _project_to_constraints(x0, target_sum)
        # define constraint
        cons = {"type": "eq", "fun": lambda hh, tgt=target_sum: np.sum(hh) - tgt}
        bounds = [(0.0, 1.0)] * n
        try:

        h_opt = _project_to_constraints(res.x, target_sum)

def gradient_refine(h_start, time_budget):
    phase_start = time.time()
    n = h_start.size
    target_sum = n/2.0
    best_h = h_start.copy()
    best_score = _compute_c5(best_h)
    max_restarts = 15
    for _ in range(max_restarts):
        if time.time() - phase_start >= time_budget:
            break
        # perturb
        perturb = np.random.randn(n) * 0.02
        x0 = best_h + perturb
        x0 = _project_to_constraints(x0, target_sum)
        cons = {"type":"eq","fun":lambda hh, tgt=target_sum: np.sum(hh) - tgt}
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_compute_c5, x0, method="SLSQP", bounds=bounds,
                           constraints=[cons], options={"maxiter":200,"ftol":1e-9,"disp":False})
        except Exception:
            continue
        h_opt = _project_to_constraints(res.x, target_sum)
        score = _compute_c5(h_opt)
        if score < best_score:
            best_h, best_score = h_opt, score
            if best_score < 0.3808:

import numpy as np
import time
from scipy.optimize import minimize

TARGET_BOUND = 0.3808  # early exit threshold


def _compute_c5(h):
    """Maximum overlap integral C5 for discrete step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_to_constraints(v, target):
    """Project vector onto [0,1] box and enforce sum(v)=target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-14:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-14:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v


def _binary_projection(v, target):
    """Return binary (0/1) vector with exactly `target` ones (top values kept)."""
    idx = np.argsort(v)[::-1]
    h = np.zeros_like(v, dtype=np.float64)
    h[idx[:target]] = 1.0
    return h


def _targeted_swap(h):
    """Propose a swap that directly attacks the current worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_idx = np.argmax(corr)
    shift = k_idx - (n - 1)  # positive shift means i+shift
    # candidate 1‑positions contributing to the worst overlap
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if shift >= 0:
        mask_i = (ones + shift < n) & (h[ones + shift] < 0.5)
        i_cand = ones[mask_i]
        mask_j = (zeros + shift >= n) | (h[zeros + shift] > 0.5)
        j_cand = zeros[mask_j]
    else:
        s = -shift
        mask_i = (ones >= s) & (h[ones - s] < 0.5)
        i_cand = ones[mask_i]
        mask_j = (zeros >= s) & (h[zeros - s] > 0.5)
        out_of_range = zeros < s
        j_cand = zeros[out_of_range | mask_j]
    if i_cand.size == 0 or j_cand.size == 0:
        return None
    i = np.random.choice(i_cand)
    j = np.random.choice(j_cand)
    new_h = h.copy()
    new_h[i] = 0.0
    new_h[j] = 1.0
    return new_h


def binary_local_search(h_start, time_budget):
    """Greedy (with targeted swaps) search on binary vectors within time_budget."""
    start = time.time()
    n = h_start.size
    target = int(round(np.sum(h_start)))
    best_h = h_start.copy()
    best_score = _compute_c5(best_h)
    no_improve = 0
    while True:
        if time.time() - start >= time_budget:
            break
        # decide between targeted and random swap
        if np.random.rand() < 0.8:
            new_h = _targeted_swap(best_h)
        else:
            new_h = None
        if new_h is None:
            # fallback random swap
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h <= 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            new_h = best_h.copy()
            new_h[i] = 0.0
            new_h[j] = 1.0
        new_score = _compute_c5(new_h)
        if new_score < best_score:
            best_h, best_score = new_h, new_score
            no_improve = 0
            if best_score < TARGET_BOUND:
                break
        else:
            no_improve += 1
        if no_improve >= 2000:
            # random restart
            rand = np.random.rand(n)
            best_h = _binary_projection(rand, target)
            best_score = _compute_c5(best_h)
            no_improve = 0
    return best_h, best_score


def gradient_refine(h_start, time_budget):
    """Local continuous refinement using SLSQP within time_budget."""
    start = time.time()
    n = h_start.size
    target = n / 2.0
    best_h = h_start.copy()
    best_score = _compute_c5(best_h)
    max_restarts = 12
    for _ in range(max_restarts):
        if time.time() - start >= time_budget:
            break
        pert = np.random.randn(n) * 0.02
        x0 = best_h + pert
        x0 = _project_to_constraints(x0, target)
        cons = {"type": "eq", "fun": lambda hh, tgt=target: np.sum(hh) - tgt}
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                _compute_c5,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[cons],
                options={"maxiter": 200, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue
        h_opt = _project_to_constraints(res.x, target)
        sc = _compute_c5(h_opt)
        if sc < best_score:
            best_h, best_score = h_opt, sc
            if best_score < TARGET_BOUND:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    overall_start = time.time()

    # ------------------------------------------------------------------
    # Initial vector (may be provided)
    init = globals().get("initial_h_values", None)
    if init is not None:
        vec = np.asarray(init, dtype=np.float64).ravel()
        n0 = vec.size
    else:
        n0 = 200
        vec = np.random.rand(n0)

    vec = _project_to_constraints(vec, n0 / 2.0)
    # start from a binary version (exactly half ones)
    cur_h = _binary_projection(vec, n0 // 2)
    best_h = cur_h.copy()
    best_score = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Multi‑scale refinement: 1×, 2×, 4× the base resolution
    prev_factor = 1
    for factor in (1, 2, 4):
        # time guard
        if time.time() - overall_start > budget_s * 0.95:
            break
        n = n0 * factor
        # upscale previous best by repeating each entry equally
        if factor == prev_factor:
            h_cur = best_h.copy()
        else:
            repeat = factor // prev_factor
            h_cur = np.repeat(best_h, repeat)
            if h_cur.size != n:
                h_cur = h_cur[:n]
        # enforce binary constraint at the new resolution
        h_cur = _binary_projection(h_cur, n // 2)

        # time budget for this scale
        rem = budget_s - (time.time() - overall_start)
        if rem <= 0:
            break
        local_time = rem * 0.65
        if local_time < 1.0:
            local_time = max(0.5, rem * 0.6)

        # binary greedy search
        h_bin, sc_bin = binary_local_search(h_cur, local_time)

        # gradient refinement with whatever time is left
        after_local = time.time()
        rem2 = budget_s - (after_local - overall_start)
        grad_time = min(rem2 * 0.30, rem2)
        if grad_time > 0.5:
            h_ref, sc_ref = gradient_refine(h_bin, grad_time)
        else:
            h_ref, sc_ref = h_bin, sc_bin

        # keep global best
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref
        if best_score < TARGET_BOUND:
            break
        prev_factor = factor

    # ------------------------------------------------------------------
    # Final evaluation (use provided evaluator if present)
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_h.size)
