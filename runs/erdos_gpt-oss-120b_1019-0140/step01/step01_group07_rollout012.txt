def anneal_binary(h, max_iter=1000000, time_limit=10.0):
    n = h.shape[0]
    k = n // 2  # number of ones
    # compute initial score
    current_score = _compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    start = time.time()
    # Determine initial temperature
    T = current_score * 0.3 + 1e-4
    decay = 0.9995
    iteration = 0
    while iteration < max_iter:
        if time.time() - start > time_limit:
            break
        # Random swap
        # For speed, we can pick random index and check its value; but need ensure we swap 0/1
        i1 = np.random.randint(n)
        if h[i1] == 0:
            # find a 1 index to swap
            i0 = np.random.randint(n)
            while h[i0] == 0:
                i0 = np.random.randint(n)
            # swap i0 (1) with i1 (0)
            i_one = i0
            i_zero = i1
        else:
            # h[i1]==1, find zero index
            i0 = np.random.randint(n)
            while h[i0] == 1:
                i0 = np.random.randint(n)
            i_one = i1
            i_zero = i0
        # Perform swap
        h[i_one] = 0.0
        h[i_zero] = 1.0
        new_score = _compute_c5(h)
        delta = new_score - current_score
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            # accept
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score < 0.3808:
                    # early exit
                    break
        else:
            # reject - revert swap
            h[i_one] = 1.0
            h[i_zero] = 0.0
        T *= decay
        iteration += 1
    return best_h, best_score
```

def refine_continuous(h_bin, max_iter=200):
    n = h_bin.size
    target_sum = n/2
    # Set bounds (0,1). Equality constraint sum.
    cons = {"type":"eq", "fun": lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)]*n
    try:
        res = minimize(_compute_c5, h_bin, method='SLSQP', bounds=bounds, constraints=[cons], options={'maxiter': max_iter, 'ftol':1e-9})
    except Exception:
        return h_bin, _compute_c5(h_bin)
    h_opt = res.x
    h_opt = np.clip(h_opt, 0.0, 1.0)
    # re-project sum if needed
    cur = np.sum(h_opt)
    if abs(cur - target_sum) > 1e-8:
        h_opt = h_opt * (target_sum / cur)
        h_opt = np.clip(h_opt, 0.0, 1.0)
    score = _compute_c5(h_opt)
    return h_opt, score

import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral C5 for discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_sum(v, target):
    """Scale v to satisfy sum(v)=target while staying in [0,1]."""
    cur = np.sum(v)
    if cur == 0:
        return np.clip(v, 0.0, 1.0)
    v = v * (target / cur)
    v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment if needed
    cur = np.sum(v)
    if abs(cur - target) > 1e-8:
        delta = target - cur
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size > 0:
            inc = delta / free.size
            v[free] += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _binary_from_continuous(vec):
    """Convert a continuous vector to a binary (or half‑binary) vector with exact sum."""
    n = vec.size
    k = n // 2
    idx = np.argsort(vec)[::-1]
    h = np.zeros(n, dtype=np.float64)
    h[idx[:k]] = 1.0
    if n % 2 == 1:
        h[idx[k]] = 0.5
    return h

def anneal_binary(h, max_iter=500000, time_limit=10.0, target_score=None):
    """Simulated annealing on a binary (or half‑binary) vector."""
    n = h.size
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    T = cur_score * 0.3 + 1e-4
    decay = 0.9995
    it = 0
    while it < max_iter:
        if time.time() - start > time_limit:
            break
        # pick a random 1 and a random 0 (or 0.5) to swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        half = np.where(h == 0.5)[0]
        if ones.size == 0 or (zeros.size + half.size) == 0:
            break
        i_one = np.random.choice(ones)
        if zeros.size > 0 and (half.size == 0 or np.random.rand() < 0.8):
            i_zero = np.random.choice(zeros)
        else:
            i_zero = np.random.choice(half)
        # perform swap
        h[i_one], h[i_zero] = 0.0, 1.0
        new_score = _compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = np.exp(-delta / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if target_score is not None and best_score < target_score:
                    break
        else:
            # revert swap
            h[i_one], h[i_zero] = 1.0, 0.0
        T *= decay
        it += 1
    return best_h, best_score

def search_binary(n, time_budget, init_h=None, target_score=None):
    """Repeated annealing runs within a given time budget."""
    best_h = None
    best_score = float("inf")
    start = time.time()
    while time.time() - start < time_budget:
        remaining = time_budget - (time.time() - start)
        if remaining < 0.5:
            break
        sub_time = min(remaining, 5.0)
        # initialise
        if init_h is not None and np.random.rand() < 0.25:
            h = init_h.copy()
            # occasional random perturbation: flip a few bits
            flips = max(1, int(0.02 * n))
            for _ in range(flips):
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i1 = np.random.choice(ones)
                i0 = np.random.choice(zeros)
                h[i1], h[i0] = 0.0, 1.0
        else:
            h = np.zeros(n, dtype=np.float64)
            half = n // 2
            h[:half] = 1.0
            if n % 2 == 1:
                h[half] = 0.5
            np.random.shuffle(h)
        h_opt, sc = anneal_binary(h, max_iter=2000000, time_limit=sub_time, target_score=target_score)
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()
            if target_score is not None and best_score < target_score:
                break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    total_budget = budget_s * 0.95
    end_time = start_time + total_budget

    # Try a single upscale if time permits
    while time.time() < end_time:
        remaining = end_time - time.time()
        if remaining < 5.0:

        # upscale by factor 2 (keep below 1000 samples)

        if best_score_up < best_score:
            best_h, best_score = best_h_up, best_score_up
        if best_score < 0.3808:

    # Final projection to satisfy sum precisely
