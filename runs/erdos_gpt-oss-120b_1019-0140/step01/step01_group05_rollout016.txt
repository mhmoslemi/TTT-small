import numpy as np, time, math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for discretised h."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _sa_optimize(h_start: np.ndarray, time_budget: float,
                 rng: np.random.Generator,
                 T0: float = 0.02,
                 max_eps_start: float = 0.5,
                 max_eps_end: float = 0.01) -> tuple[np.ndarray, float]:
    """Simulated annealing that preserves sum and box constraints."""
    h_cur = _repair_vector(h_start.copy())
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = max(0.0, 1.0 - elapsed / time_budget)
        T = T0 * frac
        max_eps = max_eps_end + (max_eps_start - max_eps_end) * frac
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _local_optimize(h_start: np.ndarray, time_budget: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Gradient descent on the worst shift, with projection onto feasible set."""
    h = _repair_vector(h_start.copy())
    N = h.size
    best_h = h.copy()
    best_score = _c5_value(h)
    cur_score = best_score
    start = time.time()
    while True:
        now = time.time()
        if now - start >= time_budget:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        Lmax = int(np.argmax(corr))
        k = Lmax - (N - 1)
        # mask of indices participating in this shift
        mask = np.zeros(N)
        if k >= 0:
            mask[:N - k] = 1.0
        else:
            mask[-k:] = 1.0
        # forward term h[i+k]
        forward = np.zeros(N)
        if k >= 0:
            forward[:N - k] = h[k:]
        else:
            forward[-k:] = h[:N + k]
        # backward term h[i-k]
        backward = np.zeros(N)
        if k >= 0:
            backward[k:] = h[:N - k]
        else:
            backward[:N + k] = h[-k:]
        grad = mask * (1.0 - (forward + backward))
        if np.max(np.abs(grad)) < 1e-12:
            break
        elapsed = now - start
        frac = max(0.0, 1.0 - elapsed / time_budget)
        step = 0.05 * frac * (0.5 + 0.5 * rng.random())
        if step <= 0.0:
            break
        h_new = _repair_vector(h - step * grad)
        new_score = _c5_value(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # reject non‑improving move
            continue
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    # Initialise from parent if available
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 120
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)], dtype=np.float64)
    h0 = _repair_vector(h0)
    # Phase 1 – low‑resolution SA
    phase1_budget = budget_s * 0.12
    best_h, best_score = _sa_optimize(h0, phase1_budget, rng)
    # Determine higher resolution (keep ≤ 800 points)
    n_target = min(800, max(200, h0.size * 4))
    if n_target != best_h.size:
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
    else:
        h_up = best_h
    h_up = _repair_vector(h_up)
    # Phase 2 – higher‑resolution SA with restarts
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining > 5.0:
        # allocate ~85 % of remaining time to SA restarts
        sa_budget = remaining * 0.85
        # split into several independent runs (≈30 s each, up to 8 runs)
        n_runs = max(1, int(sa_budget // 30))
        n_runs = min(n_runs, 8)
        per_run = sa_budget / n_runs if n_runs else 0.0
        best_hr = h_up
        best_hr_score = _c5_value(best_hr)
        for _ in range(n_runs):
            # small random perturbation of the current best
            start_vec = best_hr.copy()
            for __ in range(5):
                i, j = rng.integers(start_vec.size, size=2)
                if i == j:
                    continue
                hi, hj = start_vec[i], start_vec[j]
                up = min(1.0 - hi, hj)
                down = min(hi, 1.0 - hj)
                if up == 0.0 and down == 0.0:
                    continue
                eps = (rng.random() * (up + down) - down) * 0.2
                start_vec[i] = hi + eps
                start_vec[j] = hj - eps
            start_vec = _repair_vector(start_vec)
            h_opt, s_opt = _sa_optimize(start_vec, per_run, rng)
            if s_opt < best_hr_score:
                best_hr_score = s_opt
                best_hr = h_opt.copy()
            if time.time() - start_total >= budget_s:
                break
        # Final local gradient refinement with any leftover time
        elapsed2 = time.time() - start_total
        leftover = budget_s - elapsed2
        if leftover > 5.0:
            best_hr, best_hr_score = _local_optimize(best_hr, leftover, rng)
        best_h = best_hr
        best_score = best_hr_score
    # Final repair and exact evaluation
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
