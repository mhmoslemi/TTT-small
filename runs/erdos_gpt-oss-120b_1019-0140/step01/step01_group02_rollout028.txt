    if abs(diff) < 1e-12:

        # ----- evaluate new C5 -------------------------------------------
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(corr_new) * (2.0 / n_target))

        # ----- Metropolis acceptance --------------------------------------
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                if best_c5 <= 0.3808:          # early exit if target reached
                    break
        else:
            # revert
            h[i] = hi
            h[j] = hj

    # ----- finalise -------------------------------------------------------
    best_h = _repair_sum(best_h, n_target / 2.0)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, n_target
