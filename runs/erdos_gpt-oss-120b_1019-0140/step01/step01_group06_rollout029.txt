def compute_c5_fft(h):
    n = len(h)
    # pad length = 2*n -1
    m = 2*n -1
    # Next power of 2 for speed? Use np.fft.rfft? Simpler: use np.fft.fft of length m.
    fh = np.fft.fft(h, n=m)
    fg = np.fft.fft(1.0 - h, n=m)
    corr = np.real(np.fft.ifft(fh * np.conj(fg)))
    # corr length m, matches what np.correlate(..., mode='full') returns.
    # Return max * (2.0 / n)
    return np.max(corr) * (2.0 / n)

def random_move(h, rng):

        if max_eps <= 0:

    eps = max_eps * rng.random()
    if eps <= 0:

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h)  # simplest
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

    H = np.fft.fft(h, n=m)
    G = np.fft.fft(1.0 - h, n=m)
    corr = np.fft.ifft(H * np.conj(G)).real
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / n), idx

def _project_to_feasible(h, target, tol=1e-8, max_iter=10):
    """Clip to [0,1] and adjust to meet the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target) > 1e-6:
        i = np.argmin(h)
        h[i] = np.clip(h[i] + (target - cur), 0.0, 1.0)
    return h

def _random_feasible(n, rng):
    """Generate a random feasible vector of length n."""
    x = rng.random(n)
    return _project_to_feasible(x, n / 2.0)

def _random_move(h, rng):
    """Transfer a random amount from one entry to another, preserving the sum."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return None
        i, j = j, i
    eps = max_eps * rng.random()
    if eps <= 0:
        return None
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def _targeted_move(h, idx, rng):
    """Move mass from a high‑contribution index to its partner for the current worst lag."""
    n = h.shape[0]
    shift = idx - (n - 1)
    if shift == 0:
        return None
    if shift > 0:
        end = n - shift
        hi = h[:end]
        hj = h[shift:]
        contrib = hi * (1.0 - hj)
        loc = int(np.argmax(contrib))
        i = loc
        j = i + shift
    else:
        d = -shift
        start = d
        hi = h[start:]
        hj = h[:n - d]
        contrib = hi * (1.0 - hj)
        loc = int(np.argmax(contrib))
        i = start + loc
        j = i - d
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return None
    # use full possible transfer for maximal effect
    eps = max_eps
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    base = globals().get("initial_h_values")
    if base is not None:
        h0 = np.asarray(base, dtype=float).copy()
    else:
        n = 380
        h0 = _random_feasible(n, rng)

    n = h0.shape[0]
    target_sum = n / 2.0
    h0 = _project_to_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5, best_idx = _c5_fft_and_idx(best_h)

    cur_h = best_h.copy()
    cur_c5, cur_idx = best_c5, best_idx

    no_improve = 0
    max_no_improve = 5000
    # temperature schedule parameters
    T0 = 5e-3
    Tmin = 1e-7

    while time.time() - start < budget_s * 0.96:
        # decide move type
        if rng.random() < 0.7:
            cand = _targeted_move(cur_h, cur_idx, rng)
            if cand is None:
                cand = _random_move(cur_h, rng)
        else:
            cand = _random_move(cur_h, rng)

        if cand is None:
            continue

        cand_c5, cand_idx = _c5_fft_and_idx(cand)

        if cand_c5 < cur_c5:
            cur_h, cur_c5, cur_idx = cand, cand_c5, cand_idx
            no_improve = 0
            if cand_c5 < best_c5:
                best_h, best_c5, best_idx = cand.copy(), cand_c5, cand_idx
                # early stop if we already beat the target bound
                if best_c5 <= 0.3808:
                    break
        else:
            elapsed = time.time() - start
            frac = elapsed / (budget_s * 0.96)
            T = T0 * (1.0 - frac) + Tmin
            if rng.random() < np.exp(-(cand_c5 - cur_c5) / T):
                cur_h, cur_c5, cur_idx = cand, cand_c5, cand_idx
            no_improve += 1

        if no_improve > max_no_improve:
            # restart from best with a few random perturbations
            cur_h = best_h.copy()
            for _ in range(12):
                tmp = _random_move(cur_h, rng)
                if tmp is not None:
                    cur_h = tmp
            cur_c5, cur_idx = _c5_fft_and_idx(cur_h)
            no_improve = 0

    # final projection and evaluation
    best_h = _project_to_feasible(best_h, target_sum)
    best_c5 = _c5_fft_and_idx(best_h)[0]

    return best_h, float(best_c5), int(n)
