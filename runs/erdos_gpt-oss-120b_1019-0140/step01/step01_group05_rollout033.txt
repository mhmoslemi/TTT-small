def _simulated_annealing_upgraded(h_start, time_limit, rng, T0=0.1, max_eps_initial=0.5):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    # Use temperature that linearly decays to near 0
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * (frac)  # linear cooling
        # shrink move size with temperature
        max_eps = max_eps_initial * (frac + 0.01)  # ensure min size
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)

        hi, hj = h_cur[i], h_cur[j]
        # Determine max eps to stay in bounds:
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        # allowed total range
        total_range = up + down
        if total_range <= 0:
            continue
        # pick eps uniformly from allowed range, scale by factor to max_eps if needed
        eps = rng.random() * total_range - down
        # cap eps to max_eps
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        # Apply move
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # No need to check bounds as eps ensures within [0,1]
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def _gradient_descent_step(h, step_size):
    N = h.shape[0]
    dx = 2.0 / N
    # compute correlation vector
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (N - 1)  # can be negative
    # Compute gradient
    grad = np.full(N, dx)
    # subtract contributions
    if shift >= 0:
        # first term: h_{i+shift}
        grad[:N - shift] -= dx * h[shift:]
        # second term: h_{i-shift}
        grad[shift:] -= dx * h[:N - shift]
    else:
        s = -shift
        grad[s:] -= dx * h[:N - s]
        grad[:N - s] -= dx * h[s:]
    # Take step opposite to gradient (descending)
    h_new = h - step_size * grad
    # Clip & repair
    h_new = _repair_vector(h_new)
    return h_new

def _repair_vector(v):
    n = v.shape[0]
    target = n / 2.0
    np.clip(v, 0.0, 1.0, out=v)
    total = v.sum()
    if total == 0.0:
        v[:] = target / n
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / total
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    # Resolve residual
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size > 0:
            inc = residual / free.size
            v[free] += inc
            np.clip(v, 0.0, 1.0, out=v)
        else:
            # distribute residual to the element with most room
            if residual > 0:
                i = np.argmin(v)
                v[i] = min(1.0, v[i] + residual)
            else:
                i = np.argmax(v)
                v[i] = max(0.0, v[i] + residual)
    # After this, may still have small diff due to clipping; repeat maybe
    final_residual = target - v.sum()
    if abs(final_residual) > 1e-10:
        # small correction by scaling again
        v *= target / v.sum()
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _simulated_annealing_upgraded(h_start, time_limit, rng):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()

    start = time.time()
    # Temperature schedule parameters
    T0 = 0.1
    max_eps_start = 0.5
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        max_eps = max_eps_start * (frac + 0.01)  # ensure not zero
        # propose move
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        total_range = up + down
        if total_range <= 0.0:

        eps = rng.random() * total_range - down
        if abs(eps) > max_eps:

    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = float(budget_s)

    # initial vector
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
        n0 = h0.shape[0]
        if n0 < 200:
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, 200, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
            n0 = 200
    except NameError:
        n0 = 200
        h0 = rng.random(n0)
    h0 = _repair_vector(h0)

    best_h = h0
    best_score = _c5_value(best_h)

    # Phase 1: many short SA runs at base resolution
    phase1_end = start_time + total_budget * 0.4
    while time.time() < phase1_end:
        remaining = phase1_end - time.time()
        if remaining < 2.0:
            break
        chunk = min(5.0, remaining)
        h_start = rng.random(best_h.size)
        h_start = _repair_vector(h_start)
        h_opt, s_opt = _simulated_annealing_upgraded(h_start, chunk, rng)
        h_opt, s_opt = _local_optimize(h_opt, max_iters=15, rng=rng)
        if s_opt < best_score:
            best_score, best_h = s_opt, h_opt

    # Phase 2: progressive upsampling and refinement
    cur_n = best_h.size
    max_n = min(800, cur_n * 4)
    while time.time() < start_time + total_budget * 0.95:
        remaining = start_time + total_budget - time.time()
        if remaining < 5.0:
            break
        # Determine remaining doublings
        temp = cur_n
        levels_left = 0
        while temp < max_n:
            temp = min(temp * 2, max_n)
            levels_left += 1
        if levels_left == 0:
            break
        time_for_level = max(5.0, remaining * 0.7 / levels_left)
        n_new = min(cur_n * 2, max_n)
        x_old = np.linspace(0.0, 2.0, cur_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _repair_vector(h_up)
        h_opt, s_opt = _simulated_annealing_upgraded(h_up, time_for_level, rng)
        h_opt, s_opt = _local_optimize(h_opt, max_iters=20, rng=rng)
        if s_opt < best_score:
            best_score, best_h = s_opt, h_opt
            cur_n = n_new
        else:
            cur_n = n_new
            best_h = h_up

    best_h = _repair_vector(best_h)
    final_score = _c5_value(best_h)
    return best_h, float(final_score), best_h.size
