def refine_h(h, max_iter, step0, beta0, beta_rate, max_time, start):
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    v = np.zeros_like(h)
    for it in range(1, max_iter+1):
        # time check every some steps
        if time.time() > max_time:
            break
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # compute soft weights: use top K (maybe all)
        beta = beta0 * (beta_rate ** (it-1))
        # compute softmax weights over corr values
        # shift indices from -(n-1) to (n-1)
        # to avoid overflow, subtract max before exponent
        exp_vals = np.exp(beta * (corr - max_corr))  # max_corr get 1 at max
        # weights sum
        w = exp_vals / exp_vals.sum()
        # compute gradient as sum over shifts weighted by w
        # we need to compute gradient for each shift: but naive loop over all n shifts O(n^2) => heavy (n * n)
        # Instead approximate by using top few shifts only allows O(K*n)
        # So we will get top K indices:
        K = 5
        # get indices of top K corr
        top_idx = np.argpartition(corr, -K)[-K:]
        # compute gradient sum weighted
        grad = np.zeros_like(h)
        # compute weights for each of top indices:
        # recompute weights only for those:
        top_corr = corr[top_idx]
        w_top = np.exp(beta * (top_corr - max_corr))
        w_top = w_top / w_top.sum()
        for idx, w_i in zip(top_idx, w_top):
            shift = idx - (n - 1)
            grad_i = compute_gradient(h, shift)
            grad += w_i * grad_i
        # add momentum
        mu = 0.9
        v = mu * v - step0 / np.sqrt(it) * grad
        h_new = h + v
        # project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time, math
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_gradient(h, shift):
    """Gradient of ∑ h[i]*(1-h[i+shift]) w.r.t. h."""
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)*(1-h(x+k)) dx."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_projection_descent(h0, max_iter=1000, step0=0.5, K=5,
                              beta0=0.5, beta_rate=1.02,
                              momentum=0.9, time_deadline=None):
    """Gradient descent on a softmax approximation of the max-correlation."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iter + 1):
        if time_deadline is not None and time.time() > time_deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)

        # select top‑K shifts
        if K >= corr.size:
            top_idx = np.arange(corr.size)
        else:
            top_idx = np.argpartition(corr, -K)[-K:]

        beta = beta0 * (beta_rate ** (it - 1))
        exp_vals = np.exp(beta * (corr[top_idx] - max_corr))
        w = exp_vals / exp_vals.sum()

        grad = np.zeros_like(h)
        for idx, wi in zip(top_idx, w):
            shift = idx - (n - 1)
            grad += wi * compute_gradient(h, shift)

        eta = step0 / math.sqrt(it)
        v = momentum * v - eta * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, s)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

                               time_deadline=None):

    h = h0.copy()
    n = h.size
    s = n / 2.0
    v = np.zeros_like(h)
    mu = 0.9
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iter + 1):
