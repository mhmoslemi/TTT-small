import time
import numpy as np
import scipy.optimize as opt

_current_target_sum = 0.0

def sum_eq_target(x):
    return np.sum(x) - _current_target_sum

def compute_c5(h):
    """Overlap bound C5 for sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=100):
    """Clip to [0,1] and adjust to meet sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            inc = np.minimum(delta, 1.0 - h[free])
            h[free] += inc
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            dec = np.maximum(delta, -h[free])
            h[free] += dec
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        diff = target_sum - cur
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def resample_h(base_h, n_target):
    """Interpolate `base_h` to `n_target` samples and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def hill_climb(h0, n_iters, step, rng):
    """Simple hill climbing: transfer up to `step` between two entries."""
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    n = h.shape[0]
    for _ in range(n_iters):
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_give = min(step, h[i])
        max_take = min(step, 1.0 - h[j])
        max_delta = min(max_give, max_take)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_val = compute_c5(h_new)
        if new_val < cur_val:
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_h = h_new.copy()
                best_val = new_val
    return best_h, best_val

def local_refine(h0, n, maxiter=200):
    """Refine via SLSQP with equality sum constraint."""
    global _current_target_sum
    _current_target_sum = n / 2.0

            constraints=cons,

        )
    except Exception:
        return h0
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt

            break

    # Fallback (unlikely)
    if best_h is None:
        n = 190
        best_h = np.concatenate([np.ones(n // 2), np.zeros(n // 2)])
        best_h = project_to_feasible(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final projection and recompute guarantee
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
