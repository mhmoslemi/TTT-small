def crossover_and_repair(p1, p2, S, rng):
    n = len(p1)
    mask = rng.random(n) < 0.5
    child = np.where(mask, p1, p2).astype(np.uint8)
    ones = child.sum()
    if ones > S:
        # need to turn off some 1s
        idx = np.flatnonzero(child)
        to_off = rng.choice(idx, ones - S, replace=False)
        child[to_off] = 0
    elif ones < S:
        idx = np.flatnonzero(1 - child)  # zeros
        to_on = rng.choice(idx, S - ones, replace=False)
        child[to_on] = 1
    return child
```

def mutate_swap(vec, S, rng):
    # pick random 1 and 0 positions and swap
    ones_idx = np.flatnonzero(vec)
    zeros_idx = np.flatnonzero(1 - vec)
    if len(ones_idx) == 0 or len(zeros_idx) == 0:
        return vec
    i = rng.choice(ones_idx)
    j = rng.choice(zeros_idx)
    vec[i], vec[j] = vec[j], vec[i]
    return vec

import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # ensure h is float for (1 - h)
    h_float = h.astype(float)
    corr = np.correlate(h_float, 1.0 - h_float, mode='full')
    return float(np.max(corr) * dx)

def random_binary_vector(n, S, rng):
    vec = np.zeros(n, dtype=np.uint8)
    vec[rng.choice(n, S, replace=False)] = 1
    return vec

def tournament_select(pop, fitness, tourn_size, rng):
    idx = rng.choice(len(pop), tourn_size, replace=False)
    best = idx[np.argmin(fitness[idx])]
    return pop[best]

def crossover_and_repair(p1, p2, S, rng):
    n = len(p1)
    mask = rng.random(n) < 0.5
    child = np.where(mask, p1, p2).astype(np.uint8)
    ones = int(child.sum())
    if ones > S:
        # turn off excess ones
        excess = ones - S
        idx = np.flatnonzero(child)
        turn_off = rng.choice(idx, excess, replace=False)
        child[turn_off] = 0
    elif ones < S:
        deficit = S - ones
        idx = np.flatnonzero(1 - child)
        turn_on = rng.choice(idx, deficit, replace=False)
        child[turn_on] = 1
    return child

def mutate_swap(vec, rng):
    # swap a random 1 and a random 0
    ones_idx = np.flatnonzero(vec)
    zeros_idx = np.flatnonzero(1 - vec)

    j = rng.choice(zeros_idx)

import time
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for a discrete vector h."""
    n = len(h)
    dx = 2.0 / n
    # correlation with (1-h)
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(np.max(corr) * dx)

def random_binary_vector(n, s, rng):
    """Binary vector of length n with exactly s ones."""
    vec = np.zeros(n, dtype=np.uint8)
    vec[rng.choice(n, s, replace=False)] = 1
    return vec

def tournament_select(pop, fit, size, rng):
    """Select one individual by tournament."""
    idx = rng.choice(len(pop), size, replace=False)
    return pop[idx[np.argmin(fit[idx])]]

def crossover_and_repair(p1, p2, s, rng):
    """Uniform crossover then repair to keep exactly s ones."""
    n = len(p1)
    mask = rng.random(n) < 0.5
    child = np.where(mask, p1, p2).astype(np.uint8)
    ones = int(child.sum())
    if ones > s:
        excess = ones - s
        idx = np.flatnonzero(child)
        turn_off = rng.choice(idx, excess, replace=False)
        child[turn_off] = 0
    elif ones < s:
        deficit = s - ones
        idx = np.flatnonzero(1 - child)
        turn_on = rng.choice(idx, deficit, replace=False)
        child[turn_on] = 1
    return child

def mutate_swap(vec, rng):
    """Swap a random 1 with a random 0 (preserves sum)."""
    ones = np.flatnonzero(vec)
    zeros = np.flatnonzero(1 - vec)
    if len(ones) == 0 or len(zeros) == 0:
        return vec
    i = rng.choice(ones)
    j = rng.choice(zeros)
    vec[i], vec[j] = vec[j], vec[i]
    return vec

def hill_climb(vec, cur_f, start, budget, rng):
    """Simple 1‑swap hill climbing within remaining time."""
    n = len(vec)
    while time.time() - start < budget - 0.05:
        ones = np.flatnonzero(vec)
        zeros = np.flatnonzero(1 - vec)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        vec[i] = 0
        vec[j] = 1
        new_f = compute_c5(vec)
        if new_f + 1e-12 < cur_f:
            cur_f = new_f
        else:
            vec[i] = 1
            vec[j] = 0
    return vec, cur_f
