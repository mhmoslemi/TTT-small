import time
import numpy as np

def compute_c5(arr, dx):
    """Return C5 = max correlation of arr with (1-arr) times dx."""
    # correlation using linear (full) mode, zero-padded outside range
    return np.max(np.correlate(arr, 1 - arr, mode='full')) * dx

def try_targeted_swap(h, dx, rng):
    """Attempt to decrease the current maximal overlap by swapping a mismatched pair.
    Returns (improved, new_c5). The array h is modified in‑place when a swap is kept.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    # indices where the maximum occurs
    max_idxs = np.flatnonzero(corr == max_val)
    idx = rng.choice(max_idxs)
    k = idx - (n - 1)                 # shift corresponding to the max value

    # overlapping region for this shift
    if k >= 0:
        i_start = 0
        i_end = n - k
    else:
        i_start = -k
        i_end = n
    i_range = np.arange(i_start, i_end)

    # positions where h[i]=1 but h[i+k]=0 (a mismatch for this shift)
    mask = (h[i_range] == 1) & (h[i_range + k] == 0)
    mism = i_range[mask]
    if mism.size == 0:
        # nothing we can do for this shift
        return False, max_val * dx

    i = rng.choice(mism)
    j = i + k                         # the partner position (currently 0)

    # perform the swap
    h[i] = 0
    h[j] = 1
    new_c5 = compute_c5(h, dx)

    old_c5 = max_val * dx
    if new_c5 <= old_c5:
        # keep the swap
        return True, new_c5
    # revert – not an improvement
    h[i] = 1
    h[j] = 0
    return False, old_c5

    start_time = time.time()

    # construction is available we turn it into a binary vector by taking the
    # top half of its entries.
    # ----------------------------------------------------------------------
    if init_arr is not None:

        # select the half entries with largest values
        top = np.argpartition(-init_arr, half)[:half]
        binary[top] = 1
        cur_h = binary
    else:
        cur_h = np.zeros(n, dtype=int)
        cur_h[rng.choice(n, half, replace=False)] = 1

    cur_c5 = compute_c5(cur_h, dx)
    best_h = cur_h.astype(float).copy()
    best_c5 = cur_c5

    # ----------------------------------------------------------------------
    # Main stochastic search.
    # ----------------------------------------------------------------------
    stagnation = 0
    iter_cnt = 0
    max_iter = 5_000_000          # generous upper bound; actual run ends by budget

    while time.time() - start_time < budget_s * 0.99 and iter_cnt < max_iter:
        # First try a directed move that specifically attacks the current worst shift.
        improved, new_c5 = try_targeted_swap(cur_h, dx, rng)
        if improved:

            stagnation = 0
