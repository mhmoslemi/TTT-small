def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)
    else:
        # default initial: uniform half
        base_h = np.full(380, 0.5)  # can be any, will be replaced
    best_h = None
    best_c5 = float('inf')
    best_n = None
    # Determine list of n's
    candidate_ns = [380, 760]  # maybe add 1520 if time
    for n in candidate_ns:
        target_sum = n/2.0
        # Get initial h via resampling base_h if base exists, else random uniform.
        if base_h.shape[0] == n:
            base_resampled = base_h.copy()
        else:
            base_resampled = resample_h(base_h, n)
        # evaluate base resampled
        c5_base = compute_c5(base_resampled)
        if c5_base < best_c5:
            best_h, best_c5, best_n = base_resampled, c5_base, n
        # Determine number of restarts
        # For n=380: restarts ~ 60; for n=760: ~30; adjust if time limited.
        max_restarts = {380:60, 760:30}[n]
        for r in range(max_restarts):
            if time.time() - start > budget_s * 0.96: # leave margin
                break
            if r == 0:
                h0 = base_resampled.copy()
            else:
                # random perturbation around base or best
                h0 = generate_random_h(n, target_sum, rng)
                # occasional perturb around best solution for fine search
                if best_h is not None and rng.random() < 0.3:
                    # use best as base and add small noise
                    h0 = best_h.copy()
                    noise = rng.normal(scale=0.05, size=n)
                    h0 = np.clip(h0 + noise, 0.0, 1.0)
                    h0 = project_to_feasible(h0, target_sum)
            # local optimization
            # constraints
            cons = {"type":"eq", "fun": lambda x, nd=n: np.sum(x) - nd/2}
            bnds = [(0.0,1.0)]*n
            try:
                res = opt.minimize(fun=lambda x: compute_c5(x), x0=h0,
                                   method="SLSQP", bounds=bnds, constraints=cons,
                                   options={"maxiter":1500, "ftol":1e-9, "disp":False})
            except Exception:
                continue
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_to_feasible(h_opt, target_sum)
            c5_val = compute_c5(h_opt)
            if c5_val < best_c5:
                best_h, best_c5, best_n = h_opt, c5_val, n
                # if good enough early exit?
                if best_c5 <= 0.3808:
                    # maybe break early if time short
                    pass
    # Ensure final projection
    if best_h is not None:
        best_h = project_to_feasible(best_h, best_n/2.0)
        best_c5 = compute_c5(best_h)
    else:
        # fallback: uniform 0.5 with n=380
        best_n = 380
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n

import time
import numpy as np
import scipy.optimize as opt

# Global placeholder for the equality constraint target sum
CURRENT_TARGET_SUM = None


def compute_c5(h):
    """Compute the discretized Erdős overlap constant C5 for a step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    """Project a vector onto [0,1]^n and the hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)

    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        if cur < target_sum:
            idx = np.argmax(1.0 - h)  # most room to increase
            h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
        else:
            idx = np.argmax(h)  # largest to decrease
            h[idx] = np.clip(h[idx] - (cur - target_sum), 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Interpolate a base step function to a new resolution and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def random_feasible_h(n, target_sum, rng):
    """Generate a random feasible vector (sum = target_sum, entries in [0,1])."""
    h = rng.random(n)
    h = project_to_feasible(h, target_sum)
    return h


def sum_constraint(x):
    """Equality constraint function for SLSQP: sum(x) = CURRENT_TARGET_SUM."""
    return np.sum(x) - CURRENT_TARGET_SUM


def objective_for_opt(h):
    """Objective wrapper for scipy.optimize."""
    return compute_c5(h)


def local_refine(h_start, n):
    """Run a short SLSQP refinement starting from h_start."""
    cons = [{"type": "eq", "fun": sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(fun=objective_for_opt,
                           x0=h_start,
                           method="SLSQP",
                           bounds=bnds,
                           constraints=cons,
                           options={"maxiter": 400, "ftol": 1e-9, "disp": False})
    except Exception:
        return h_start
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt


def hill_climb(h, n, rng, max_iters, step_max, start_time, budget_s):
    """Simple greedy pairwise move hill‑climbing to reduce C5."""
    c5 = compute_c5(h)
    no_improve = 0
    for _ in range(max_iters):
        if time.time() - start_time > budget_s * 0.96:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        max_inc = min(1.0 - hi, hj)   # how much we can increase hi (decrease hj)
        max_dec = min(hi, 1.0 - hj)   # how much we can decrease hi (increase hj)
        if max_inc == 0 and max_dec == 0:
            continue
        delta = rng.uniform(-max_dec, max_inc)
        if abs(delta) > step_max:
            delta = step_max * np.sign(delta)
        h[i] = hi + delta
        h[j] = hj - delta
        new_c5 = compute_c5(h)
        if new_c5 < c5:
            c5 = new_c5
            no_improve = 0
        else:
            h[i] = hi
            h[j] = hj
            no_improve += 1
            if no_improve >= 200:
                break
    return h, c5

    Optimize a step function h to minimize the Erdős overlap constant C5.

        else:
            if base_h is not None:
                base_n_h = resample_h(base_h, n)
            else:
                base_n_h = np.full(n, 0.5)
                base_n_h = project_to_feasible(base_n_h, target_sum)

        # Evaluate the base candidate
        c5_base = compute_c5(base_n_h)
        if c5_base < best_c5:

        # Configure search effort per resolution
        if n <= 380:

            hill_iters = 5000
        else:
            restarts = 30
            hill_iters = 3000
        step_max = 0.2  # maximum pairwise transfer per move

            else:
                h0 = random_feasible_h(n, target_sum, rng)
                if best_h is not None and rng.random() < 0.15:
                    h0 = best_h.copy()
                    noise = rng.normal(scale=0.03, size=n)
                    h0 = np.clip(h0 + noise, 0.0, 1.0)
                    h0 = project_to_feasible(h0, target_sum)

            # Perform greedy hill‑climbing
            h_hc, c5_hc = hill_climb(h0, n, rng, hill_iters, step_max, start_time, budget_s)
