import numpy as np

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) == len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _binary_sa(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simulated annealing on binary vectors with exact half‑ones."""
    N = h_start.size
    # force binary
    h = np.rint(h_start).astype(np.float64)
    target_ones = N // 2
    cur_ones = int(h.sum())
    if cur_ones > target_ones:
        idx = rng.choice(np.where(h == 1)[0], size=cur_ones - target_ones, replace=False)
        h[idx] = 0.0
    elif cur_ones < target_ones:
        idx = rng.choice(np.where(h == 0)[0], size=target_ones - cur_ones, replace=False)
        h[idx] = 1.0
    cur_score = _c5_value(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.02          # initial temperature
    last_improve = 0
    it = 0
    while True:
        if time.time() - start >= time_limit:
            break

        frac = 1.0 - elapsed / time_limit

def _continuous_sa(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simulated annealing on continuous values, preserving sum and box constraints."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.005
    max_eps = 0.15
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)      # max epsilon (increase i, decrease j)

        h_cur[j] = hj - eps
        new_score = _c5_value(h_cur)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            h_cur[i] = hi
            h_cur[j] = hj
