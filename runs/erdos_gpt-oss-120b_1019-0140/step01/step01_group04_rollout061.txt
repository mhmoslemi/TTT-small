def propose(v, step):
   i, j = np.random.choice(n, 2, replace=False)
   max_decrease_i = v[i]   # can decrease at most v[i]
   max_increase_i = 1 - v[i]   # can increase at most
   # Actually we will move from i to j: delta positive means moving from i to j (decrease i, increase j)
   delta_max = min(step, max_decrease_i, 1 - v[j])
   delta_min = -min(step, max_increase_i, v[j])   # delta negative: move from j to i
   if delta_max <= delta_min: return v unchanged
   delta = np.random.uniform(delta_min, delta_max)
   v_new = v.copy()
   v_new[i] -= delta
   v_new[j] += delta
   return v_new

import time
import numpy as np

def _compute_c5(v):
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return float(np.max(corr) * dx)

def _repair_vector(v, n, max_iter=20):
    """Project v onto [0,1]^n with exact sum = n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final fine‑tune if still off
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v *= factor
        v = np.clip(v, 0.0, 1.0)
    return v

def _random_binary_feasible(n):
    """Binary vector with exactly n/2 ones, shuffled."""
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 1.0
    np.random.shuffle(v)
    return v

def _upsample_vector(v, n_target):
    """Linear interpolation to a larger grid, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _simulated_annealing(v_init, time_limit):
    """SA on a feasible vector. Keeps sum constant by exchanging mass."""
    v = v_init.copy()
    n = len(v)
    best = v.copy()
    best_score = _compute_c5(best)
    cur_score = best_score
    start = time.time()
    # schedule parameters
    start_temp, end_temp = 0.1, 0.001
    step_start, step_end = 0.2, 0.005
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = start_temp * (1 - frac) + end_temp * frac
        step = step_start * (1 - frac) + step_end * frac
        i, j = np.random.choice(n, 2, replace=False)
        vi, vj = v[i], v[j]
        # move amount from i to j (delta positive means i->j)
        delta_max = min(step, vi, 1.0 - vj)
        delta_min = -min(step, vj, 1.0 - vi)
        if delta_max <= delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        cand = v.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _compute_c5(cand)
        diff = cand_score - cur_score
        if diff < 0 or np.random.rand() < np.exp(-diff / (temp + 1e-12)):
            v = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = v.copy()
                best_score = cur_score
    return best, best_score

def _local_search_swap(v, time_limit):
    """Hill‑climbing swapping mass between a 1‑site and a 0‑site."""
    best = v.copy()
    best_score = _compute_c5(best)
    n = len(v)
    start = time.time()
    eps = 1e-12
    while time.time() - start < time_limit:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:

        if best[i] <= eps or best[j] >= 1.0 - eps:

        max_delta = min(best[i], 1.0 - best[j])
        delta = max_delta * np.random.rand()
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _compute_c5(cand)
        if cand_score < best_score:
