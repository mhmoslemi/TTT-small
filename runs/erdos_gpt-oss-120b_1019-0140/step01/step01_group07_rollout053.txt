import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral C5 for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and enforce sum(v)=target by a water-filling step."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    # Increase or decrease total mass while staying in [0,1]
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

    target0 = n0 / 2.0
    init_vec = _project_to_constraints(init_vec, target0)

    # ----------------------------------------------------------------------
    # Resolution schedule (doubling while time permits)
    resolutions = [n0]
    if budget_s >= 400:
        resolutions.append(n0 * 2)
    if budget_s >= 800:
        resolutions.append(n0 * 4)
    # limit size for safety
    resolutions = [r for r in resolutions if r <= 2000]

    best_h = None
    best_score = float("inf")

    # ----------------------------------------------------------------------
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
        target = n / 2.0

        # Interpolate previous best solution or the given initial vector
        if best_h is not None:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_start = np.interp(x_new, x_old, best_h)
        else:
            if n == n0:
                h_start = init_vec.copy()
            else:
                x0 = np.linspace(0.0, 2.0, init_vec.size, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h_start = np.interp(x_new, x0, init_vec)

        h_start = _project_to_constraints(h_start, target)

        # ------------------------------------------------------------------
        # Genetic algorithm core
        remaining = max(0.0, budget_s - (time.time() - start_time))
        pop_size = max(10, min(30, int(remaining // 15)))  # modest size
        max_gen = 100

        # Initialise population
        pop = np.empty((pop_size, n), dtype=np.float64)
        for i in range(pop_size):
            if i == 0:
                pop[i] = h_start.copy()
            else:
                rnd = np.random.rand(n)
                pop[i] = _project_to_constraints(rnd, target)

        fitness = np.empty(pop_size, dtype=np.float64)
        for i in range(pop_size):
            fitness[i] = _compute_c5(pop[i])

        elite_cnt = 2
        best_local_h = pop[np.argmin(fitness)].copy()
        best_local_score = float(fitness.min())

        gen = 0
        while gen < max_gen and time.time() - start_time < budget_s * 0.9:
            # Produce next generation
            new_pop = np.empty_like(pop)
            # Elitism
            elite_idx = fitness.argsort()[:elite_cnt]
            for ei, idx in enumerate(elite_idx):
                new_pop[ei] = pop[idx]
            # Offspring
            for i in range(elite_cnt, pop_size):
                # tournament selection for parent 1
                t1 = np.random.choice(pop_size, size=3, replace=False)
                p1 = pop[t1[np.argmin(fitness[t1])]]
                # tournament selection for parent 2
                t2 = np.random.choice(pop_size, size=3, replace=False)
                p2 = pop[t2[np.argmin(fitness[t2])]]
                # blend crossover
                alpha = np.random.rand()
                child = alpha * p1 + (1.0 - alpha) * p2
                # mutation
                if np.random.rand() < 0.3:
                    child += np.random.randn(n) * 0.05
                child = _project_to_constraints(child, target)
                new_pop[i] = child

            pop = new_pop
            # evaluate new population
            for i in range(pop_size):
                fitness[i] = _compute_c5(pop[i])
            # update best of this resolution
            cur_best_idx = int(np.argmin(fitness))
            cur_best_score = float(fitness[cur_best_idx])
            if cur_best_score < best_local_score:
                best_local_score = cur_best_score
                best_local_h = pop[cur_best_idx].copy()
                if best_local_score < 0.3808:
                    break
            gen += 1

        # ------------------------------------------------------------------
        # Local refinement with SLSQP (few restarts)
        for _ in range(5):
            if time.time() - start_time > budget_s * 0.97:
                break
            perturb = np.random.randn(n) * 0.02
            x0 = best_local_h + perturb
            x0 = _project_to_constraints(x0, target)
            cons = {"type": "eq", "fun": lambda h, tgt=target: np.sum(h) - tgt}
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(
                    _compute_c5,
                    x0,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=[cons],
                    options={"maxiter": 300, "ftol": 1e-9, "disp": False},
                )
                h_opt = _project_to_constraints(res.x, target)
                s_opt = _compute_c5(h_opt)
                if s_opt < best_local_score:
                    best_local_score = s_opt
                    best_local_h = h_opt.copy()
                    if best_local_score < 0.3808:
                        break
            except Exception:
                continue

        # ------------------------------------------------------------------
        # Update global best solution
        if best_local_score < best_score:
            best_score = best_local_score
            best_h = best_local_h.copy()
            if best_score < 0.3808:
                break

    # ----------------------------------------------------------------------
    # Fallback safety
    if best_h is None:
        best_h = _project_to_constraints(np.random.rand(200), 100.0)
        best_score = _compute_c5(best_h)

    # ----------------------------------------------------------------------
    # Final evaluation via provided helper if available
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(best_h.size)
