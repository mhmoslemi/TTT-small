import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10.0
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10.0

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap C5 for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient(h, shift):
    """Gradient of C(k) for a given integer shift (zero padding)."""
    n = h.size
    grad = np.full(n, 1.0, dtype=h.dtype)
    if shift >= 0:
        if shift == 0:
            grad -= 2.0 * h
        else:
            grad[:n - shift] -= h[shift:]
            grad[shift:]   -= h[:n - shift]
    else:
        s = -shift
        grad[s:]   -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad


def weighted_gradient_and_max_corr(h, alpha):
    """Gradient of the soft‑max of the correlations and the raw max correlation."""
    n = h.size
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    max_raw = np.max(corr_raw)
    # stable soft‑max weights
    exp_vals = np.exp(alpha * (corr_raw - max_raw))
    w = exp_vals / exp_vals.sum()

    grad = np.full(n, 1.0, dtype=h.dtype)
    # shifts run from -(n-1) .. (n-1)
    idx = 0
    for shift in range(-(n - 1), n):
        weight = w[idx]
        idx += 1
        if weight == 0.0:
            continue
        if shift >= 0:
            if shift == 0:
                grad -= 2.0 * weight * h
            else:
                grad[:n - shift] -= weight * h[shift:]
                grad[shift:]   -= weight * h[:n - shift]
        else:
            s = -shift
            grad[s:]   -= weight * h[:n - s]
            grad[:n - s] -= weight * h[s:]
    return grad, max_raw


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
        h0 = project_onto_box_sum(h0, n / 2.0)
    else:
        n = 256
        h0 = project_onto_box_sum(rng.random(n), n / 2.0)

    # best solution tracking
    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # parameters for annealed soft‑max descent
    step0 = 0.5
    alpha_start = 0.5
    alpha_end = 8.0
    max_iter = 50000  # safe upper bound, actual stop by time
    no_improve = 0
    improve_limit = 2000

    h = h0.copy()
    for it in range(1, max_iter + 1):
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break

        # schedule
        eta = step0 / np.sqrt(it)
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)

        grad, _ = weighted_gradient_and_max_corr(h, alpha)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, n / 2.0)

        cur_score = compute_c5(h_new)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # occasional perturbation if stagnating
        if no_improve >= improve_limit:
            h_new = best_h + rng.normal(scale=0.01, size=n)
            h_new = project_onto_box_sum(h_new, n / 2.0)
            no_improve = 0

        h = h_new

    # final refinement using classic projected gradient on the worst shift
    refined = best_h.copy()
    s_target = n / 2.0
    extra_iters = 3000
    for i in range(1, extra_iters + 1):
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break
        corr = np.correlate(refined, 1.0 - refined, mode='full')
        shift = int(np.argmax(corr) - (n - 1))
        grad = compute_gradient(refined, shift)
        eta = 0.3 / np.sqrt(i)
        refined = refined - eta * grad
        refined = project_onto_box_sum(refined, s_target)
        cur = compute_c5(refined)
        if cur < best_score:
            best_score = cur
            best_h = refined.copy()

    best_h = project_onto_box_sum(best_h, n / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
