def pgd_momentum(h0, n_iter=5000, step0=0.5, momentum=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    s = h.size / 2
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter+1):
        # correlation to find shift
        corr = compute_full_corr(h) # maybe using FFT
        shift_idx = np.argmax(corr)
        shift = shift_idx - (h.size - 1)  # shift in range
        grad = compute_gradient(h, shift)
        # step size
        eta = step0 / (np.sqrt(it))
        v = momentum * v - eta * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def compute_full_corr(h):
    n = h.size
    L = 2 * n
    fa = np.fft.rfft(h, n=L)
    fb = np.fft.rfft(1.0 - h[::-1], n=L)
    conv = np.fft.irfft(fa * fb, n=L)

def compute_c5(h):
    n = h.size
    corr = compute_full_corr(h)
    return np.max(corr) * (2.0 / n)

def compute_shift(h):

    corr = compute_full_corr(h)
    idx = np.argmax(corr)
    return idx - (n - 1)

def compute_gradient(h, shift):
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def projected_gradient_descent_momentum(h0, n_iter=5000, step0=0.5, momentum=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    s = h.size / 2.0

    best_score = compute_c5(h)
    for it in range(1, n_iter+1):
        shift = compute_shift(h)
        grad = compute_gradient(h, shift)
        eta = step0 / np.sqrt(it)

        score = compute_c5(h_new)
        if score < best_score:

import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # extend interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_full_corr(h):
    """Linear correlation of h with (1-h) using FFT; returns length 2n-1 array."""
    n = h.shape[0]
    L = 2 * n
    fa = np.fft.rfft(h, n=L)
    fb = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[:2 * n - 1]

def compute_c5(h):
    n = h.shape[0]
    corr = compute_full_corr(h)
    return np.max(corr) * (2.0 / n)

def compute_shift(h):
    """Return integer shift k such that correlation at k is maximal."""
    n = h.shape[0]
    corr = compute_full_corr(h)
    idx = np.argmax(corr)
    # mapping: k = (n-1) - idx
    return (n - 1) - idx

def compute_gradient(h, shift):
    """Gradient of max-correlation term for a given integer shift."""
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def projected_gradient_descent_momentum(h0, n_iter=5000, step0=0.5, momentum=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        shift = compute_shift(h)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        v = momentum * v - eta * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def resample_h(arr, target_n):
    """Linear interpolation from arr (any length) to target_n samples."""
    m = arr.shape[0]
    if m == target_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, target_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Choose resolution
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 256
    s_target = n / 2.0

    # Population for PSO
    pop_size = 30
    pop = np.empty((pop_size, n), dtype=float)
    idx = 0
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.shape[0] != n:
            init = resample_h(init, n)
        pop[idx] = project_onto_box_sum(init, s_target)
        idx += 1
    while idx < pop_size:
        pop[idx] = project_onto_box_sum(rng.random(n), s_target)
        idx += 1

    vel = np.zeros_like(pop)
    pbest = pop.copy()
    pbest_score = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        pbest_score[i] = compute_c5(pbest[i])
    best_idx = int(np.argmin(pbest_score))
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_score[best_idx]

    # PSO parameters
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    iteration = 0
    max_iters = 20000  # safety cap

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 2.0 or iteration >= max_iters:
            break
        w = w_start - (iteration / max_iters) * (w_start - w_end)
        r1 = rng.random((pop_size, n))
        r2 = rng.random((pop_size, n))
        vel = w * vel + c1 * r1 * (pbest - pop) + c2 * r2 * (gbest - pop)
        pop = pop + vel
        for i in range(pop_size):
            pop[i] = project_onto_box_sum(pop[i], s_target)
        for i in range(pop_size):
            score = compute_c5(pop[i])
            if score < pbest_score[i]:
                pbest_score[i] = score
                pbest[i] = pop[i].copy()
                if score < gbest_score:
                    gbest_score = score
                    gbest = pop[i].copy()
        iteration += 1

    # Final refinement with momentum PGD
    remaining = budget_s - (time.time() - start)
    if remaining > 0.5:
        n_iter = int(min(20000, max(2000, remaining * 30)))
        gbest, gbest_score = projected_gradient_descent_momentum(gbest, n_iter=n_iter,
                                                                step0=0.5, momentum=0.9)
    # Ensure feasibility one last time
    gbest = project_onto_box_sum(gbest, s_target)
    final_score = compute_c5(gbest)
    return gbest, final_score, n
