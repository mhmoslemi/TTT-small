def project_to_feasible(h, target_sum, tol=1e-12, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = np.sum(h)
        diff = target_sum - cur_sum
        if abs(diff) <= tol:
            break
        # Determine indices that are not at bound
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # All at bounds, cannot adjust; break
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final adjustment if needed due to floating error
    cur_sum = np.sum(h)
    diff = target_sum - cur_sum
    if abs(diff) > 1e-9:
        # adjust one element not at bound
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h
```

import time
import numpy as np
import scipy.optimize as opt


def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-12, max_iter=100):
    """Iteratively project onto box [0,1]^n and the hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-9:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def resample_h(h, n_target):
    """Linear interpolation of h to a different number of points and enforce feasibility."""
    h = np.asarray(h, dtype=float)
    n_base = h.shape[0]
    if n_target == n_base:
        h_new = h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def sum_eq_constraint(x, target):
    """Equality constraint for sum(x) = target."""
    return np.sum(x) - target


def local_optimize_slsqp(h, n):
    """Run SLSQP from a feasible start."""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint, "args": (target,)}
    try:
        res = opt.minimize(
            compute_c5,
            x0=h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 500, "ftol": 1e-12, "disp": False},
        )
    except Exception:
        return h
    if not res.success:
        return h
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, target)
    return h_opt


def random_harmonic_init(n, rng):
    """Create a smooth random initialization using a few sinusoidal components."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    num_terms = rng.integers(2, 5)  # 2‑4 sinusoidal terms
    for _ in range(num_terms):
        freq = rng.integers(1, max(2, n // 4))
        amp = rng.uniform(-0.5, 0.5)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.sin(np.pi * freq * x + phase)
    h_min, h_max = h.min(), h.max()
    if h_max - h_min > 1e-12:
        h = (h - h_min) / (h_max - h_min)
    else:
        h = np.full(n, 0.5)
    h = project_to_feasible(h, n / 2.0)
    return h


def hill_climb(initial_h, n, rng, time_limit, max_iter):
    """Simple greedy projected hill‑climbing."""
    h = initial_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    max_step = 0.2
    min_step = 0.005
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        step = max_step - (max_step - min_step) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
