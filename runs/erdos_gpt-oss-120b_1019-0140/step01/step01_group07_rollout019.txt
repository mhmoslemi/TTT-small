import numpy as np, time, math

def _c5_corr_fft(h):
    n = h.size
    dx = 2.0 / n
    cp = 1.0 - h
    # length for linear correlation = 2*n
    H = np.fft.rfft(h, n=2*n)
    C = np.fft.rfft(cp, n=2*n)
    corr = np.fft.irfft(H * np.conj(C), n=2*n)
    max_corr = np.max(corr[:2*n-1])
    return max_corr * dx

def _binary_from_continuous(vals):
    n = vals.size
    # ensure n is even
    half = n // 2
    idx = np.argpartition(vals, -half)[-half:]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

import time
import math
import numpy as np
from scipy.optimize import minimize

def _c5_corr_fft(h):
    """Maximum overlap C5 for a step function h (values in [0,1])."""
    n = h.size
    dx = 2.0 / n
    cp = 1.0 - h
    H = np.fft.rfft(h, n=2 * n)
    C = np.fft.rfft(cp, n=2 * n)
    corr = np.fft.irfft(H * np.conj(C), n=2 * n)
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * dx

def _binary_from_continuous(vals):
    """Convert continuous vector to binary with exactly half ones."""
    n = vals.size
    half = n // 2
    idx = np.argpartition(vals, -half)[-half:]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def _sa_optimize(n, init_h, time_limit, global_best):
    """Simulated annealing on binary vectors of size n."""
    start = time.time()
    h_cur = init_h.copy()
    ones = list(np.where(h_cur == 1)[0])
    zeros = list(np.where(h_cur == 0)[0])
    cur_score = _c5_corr_fft(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    T0 = 0.05
    no_improve = 0
    max_no_improve = 5000
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / time_limit)
        if T < 1e-6:
            T = 1e-6
        i = np.random.randint(len(ones))
        j = np.random.randint(len(zeros))
        idx_one = ones[i]
        idx_zero = zeros[j]
        h_cur[idx_one] = 0.0
        h_cur[idx_zero] = 1.0
        new_score = _c5_corr_fft(h_cur)
        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / T):
            cur_score = new_score
            ones[i] = idx_zero
            zeros[j] = idx_one
            no_improve = 0
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
                if best_score < 0.3808:
                    break
        else:
            h_cur[idx_one] = 1.0
            h_cur[idx_zero] = 0.0
            no_improve += 1
        if no_improve >= max_no_improve:
            rand_h = np.zeros(n, dtype=np.float64)
            idx = np.random.choice(n, n // 2, replace=False)
            rand_h[idx] = 1.0
            h_cur = rand_h
            ones = list(np.where(h_cur == 1)[0])
            zeros = list(np.where(h_cur == 0)[0])
            cur_score = _c5_corr_fft(h_cur)
            no_improve = 0
        if best_score < 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    total_budget = float(budget_s)

    parent = globals().get('initial_h_values', None)

    base_n = 380
    if base_n % 2:

    if parent is not None:

        if parent_arr.size != base_n:

            interp = np.interp(x_new, x_parent, parent_arr)
            h0 = _binary_from_continuous(interp)
        else:
            h0 = _binary_from_continuous(parent_arr)
    else:
        h0 = np.zeros(base_n, dtype=np.float64)
        idx = np.random.choice(base_n, base_n // 2, replace=False)
        h0[idx] = 1.0

    best_h = h0.copy()
    best_score = _c5_corr_fft(best_h)

    n = base_n
    while True:
        elapsed = time.time() - start_time
        time_left = total_budget - elapsed
        if time_left < 30.0:

        interp_vals = np.interp(x_new, x_old, best_h)
        best_h = _binary_from_continuous(interp_vals)

    # optional continuous refinement with SLSQP if time permits
    remaining = total_budget - (time.time() - start_time)
    if remaining > 20.0:

        cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
        bounds = [(0.0, 1.0)] * n
        try:

                _c5_corr_fft,
                best_h.copy(),
                method="SLSQP",
                bounds=bounds,
                constraints=[cons],
                options={"maxiter": 200, "ftol": 1e-9, "disp": False},
            )
            if res.success:

                # enforce exact sum if numeric drift
                cur = np.sum(h_cont)
                if abs(cur - target_sum) > 1e-8 and cur > 0:
                    h_cont *= target_sum / cur
                sc_cont = _c5_corr_fft(h_cont)
                if sc_cont < best_score:
                    best_score = float(sc_cont)
                    best_h = h_cont.astype(np.float64)
        except Exception:
            pass

    if "evaluate_erdos_solution" in globals():

            final_score = float(best_score)
    else:
        final_score = float(best_score)
