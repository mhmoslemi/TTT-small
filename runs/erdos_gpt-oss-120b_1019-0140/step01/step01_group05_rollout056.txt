import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simple SA preserving sum and box constraints."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.02          # initial temperature
    max_eps0 = 0.5    # initial move limit
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        max_eps = max_eps0 * frac
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)      # positive epsilon bound (transfer from j to i)
        down = min(hi, 1.0 - hj)    # negative epsilon bound (transfer from i to j)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        if not (0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0):
            continue
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Obtain an initial construction
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 120
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)], dtype=np.float64)
    h0 = _repair_vector(h0)
    best_h = h0
    best_score = _c5_value(best_h)
    start_total = time.time()
    # Build a list of resolutions (doubling until a cap)
    max_n = 800
    res_levels = [best_h.size]
    while res_levels[-1] < max_n:
        nxt = min(res_levels[-1] * 2, max_n)
        if nxt == res_levels[-1]:
            break
        res_levels.append(nxt)
    # Main multi‑resolution optimisation
    for lvl_idx, N in enumerate(res_levels):
        elapsed_total = time.time() - start_total
        remaining_total = budget_s - elapsed_total
        if remaining_total < 5.0:
            break
        # Interpolate best solution to current resolution if needed
        if best_h.size != N:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_cur = _repair_vector(h_interp)
        else:
            h_cur = best_h.copy()
        # Allocate time for this level (share remaining equally)
        levels_left = len(res_levels) - lvl_idx
        level_time = remaining_total * (1.0 / levels_left) * 0.9
        level_start = time.time()
        # Run repeated SA chunks within the allocated time
        while True:
            now = time.time()
            elapsed_level = now - level_start
            time_left = level_time - elapsed_level
            if time_left < 5.0:
                break
            chunk = min(30.0, time_left)
            # Produce a start vector: perturb best if not first chunk
            if elapsed_level == 0.0:
                start_vec = h_cur
            else:
                pert = best_h + rng.normal(scale=0.1, size=N)
                start_vec = _repair_vector(pert)
            cand_h, cand_score = _simulated_annealing(start_vec, chunk, rng)
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_h = cand_h
                # Update current vector for next chunk
                h_cur = best_h.copy()
        # Ensure best_h is repaired before moving to finer level
        best_h = _repair_vector(best_h)
        best_score = _c5_value(best_h)
    # Final repair and evaluation
    best_h = _repair_vector(best_h)
    final_score = _c5_value(best_h)
    return best_h, float(final_score), best_h.size
