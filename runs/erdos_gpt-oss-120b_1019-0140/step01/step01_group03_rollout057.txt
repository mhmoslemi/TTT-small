def adam_optimize(h0, max_iter, step0=0.5):
    h = h0.astype(float).copy()
    s = h.size / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter+1):
        # compute correlation and active shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (h.size - 1)
        grad = compute_gradient(h, shift)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # project onto box sum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def upsample_to_n(arr, n):
    m = arr.size
    if m == n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

import time
import numpy as np

# ---------- utility functions ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    # monotone decreasing f(lambda) = sum clip(v - lambda, lo, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets contain the root
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient(h, shift):
    """Gradient (subgradient) of the active correlation term."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[: n - shift] = h[shift:]
        term2[shift:] = h[: n - shift]
    else:
        s = -shift
        term1[s:] = h[: n - s]
        term2[: n - s] = h[s:]
    return 1.0 - term1 - term2


def momentum_optimize(h0, n_iter=2000, step0=0.5, mu=0.9,
                      start_time=None, budget_s=None):
    """Projected gradient descent with momentum."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    mom = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, n_iter + 1):
        if start_time is not None and budget_s is not None:
            if time.time() - start_time > budget_s - 0.5:
                break
        corr = np.correlate(h, 1.0 - h, mode='full')
        shift = np.argmax(corr) - (n - 1)
        grad = compute_gradient(h, shift)
        mom = mu * mom + grad
        eta = step0 / np.sqrt(it)
        h = h - eta * mom
        h = project_onto_box_sum(h, s)
        scr = compute_c5(h)
        if scr < best_score:
            best_score = scr
            best_h = h.copy()
    return best_h, best_score


def binary_hill_climb(h_int, max_iter=2000, attempts=10, rng=None,
                      start_time=None, budget_s=None):
    """Greedy swap search for binary (0/1) vectors while preserving sum."""
    if rng is None:
        rng = np.random.default_rng()
    h = h_int.astype(int)
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h.astype(float))

    ones = np.flatnonzero(best_h == 1)
    zeros = np.flatnonzero(best_h == 0)

    for _ in range(max_iter):
        if start_time is not None and budget_s is not None:
            if time.time() - start_time > budget_s - 0.5:
                break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(attempts):
            i = rng.integers(len(ones))
            j = rng.integers(len(zeros))
            idx1 = ones[i]
            idx0 = zeros[j]
            cand = best_h.copy()
            cand[idx1] = 0
            cand[idx0] = 1
            scr = compute_c5(cand.astype(float))
            if scr < best_score:
                best_score = scr
                best_h = cand
                # recompute index sets
                ones = np.flatnonzero(best_h == 1)
                zeros = np.flatnonzero(best_h == 0)
                improved = True
                break
        if not improved:
            break
    return best_h.astype(float), best_score


def thue_morse(n):
    """Return Thue–Morse binary (0/1) sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample_to_n(arr, n):
    """Linear interpolation of arr to length n."""
    m = arr.size
    if m == n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, n, endpoint=False)
    return np.interp(x_new, x_old, arr)


# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_score = float('inf')
    n = 256  # default resolution

    # If a parent construction exists, respect its size
    if 'initial_h_values' in globals():
        n_parent = len(initial_h_values)
        if n_parent > 0:
            n = n_parent

    # ---------- generate base candidates ----------
    candidates = []

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != n:
            h0 = upsample_to_n(h0, n)
        h0 = project_onto_box_sum(h0, n / 2.0)
        candidates.append(h0)

    # Thue–Morse (only if n is a power of two)
    if (n & (n - 1)) == 0:  # power of two test
        candidates.append(thue_morse(n))

    # Random binary balanced starts
    for _ in range(5):
        candidates.append(random_balanced_binary(n, rng))

    # Random fractional starts
    for _ in range(3):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))

    # ---------- optimize each base candidate ----------
    for base in candidates:
        if time.time() - start > budget_s - 2.0:
            break
        # Determine if binary
        is_bin = np.all((base == 0) | (base == 1))
        if is_bin:
            # greedy swaps first
            h_opt, sc_opt = binary_hill_climb(base.astype(int),
                                              max_iter=2000,
                                              attempts=10,
                                              rng=rng,
                                              start_time=start,
                                              budget_s=budget_s)
            # then momentum refinement
            h_opt, sc_opt = momentum_optimize(h_opt,
                                              n_iter=3000,
                                              step0=0.5,
                                              mu=0.9,
                                              start_time=start,
                                              budget_s=budget_s)
        else:
            h_opt, sc_opt = momentum_optimize(base,
                                              n_iter=5000,
                                              step0=0.5,
                                              mu=0.9,
                                              start_time=start,
                                              budget_s=budget_s)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # ---------- random restarts ----------
    while time.time() - start < budget_s - 2.0:
        if rng.random() < 0.5:
            # binary start
            h_start = random_balanced_binary(n, rng).astype(int)
            h_opt, sc_opt = binary_hill_climb(h_start,
                                              max_iter=2000,
                                              attempts=10,
                                              rng=rng,
                                              start_time=start,
                                              budget_s=budget_s)
            h_opt, sc_opt = momentum_optimize(h_opt,
                                              n_iter=2000,
                                              step0=0.5,
                                              mu=0.9,
                                              start_time=start,
                                              budget_s=budget_s)
        else:
            # fractional start
            v = rng.random(n)
            h_start = project_onto_box_sum(v, n / 2.0)
            h_opt, sc_opt = momentum_optimize(h_start,
                                              n_iter=3000,
                                              step0=0.5,
                                              mu=0.9,
                                              start_time=start,
                                              budget_s=budget_s)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # ---------- final safeguard ----------
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, n
