def _worst_shift_and_gradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    shift = max_idx - (n-1)
    # gradient array
    grad = np.zeros(n)
    # compute term1 for each i if i - shift in bounds
    if shift >= 0:
        # i - shift >=0 => i >= shift; for i>=shift:
        term1 = 1.0 - h[:n-shift]
        grad[shift:] += term1
        # term2: i+shift < n => i < n-shift
        term2 = h[shift:]

    return max_val, shift, grad

def _random_feasible(n, rng):
    h = np.zeros(n, dtype=float)
    half = n // 2
    indices = rng.permutation(n)[:half]
    h[indices] = 1.0
    if n % 2:
        # assign one entry 0.5 to meet sum required
        remaining = list(set(range(n)) - set(indices))
        idx = rng.choice(remaining)
        h[idx] = 0.5
    return h

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _c5_and_grad(h: np.ndarray):
    """Return (C5, shift_of_worst, gradient) for the current h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (n - 1)          # integer shift that attains the maximum
    # gradient: dC5/dh_i = (1 - h_{i-shift}) - h_{i+shift}
    grad = np.zeros(n, dtype=float)
    if shift >= 0:
        grad[shift:] += 1.0 - h[: n - shift]
        grad[: n - shift] -= h[shift:]
    else:
        s = -shift
        grad[: n - s] += 1.0 - h[s:]
        grad[s:] -= h[: n - s]
    dx = 2.0 / n
    return max_val * dx, shift, grad

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust one entry (recursively if needed) so that sum(h) == target."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]

            np.clip(h, 0.0, 1.0, out=h)
