def gradient_move(h, step, target_sum, rng):
    n = h.shape[0]
    dx = 2.0 / n
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    # find index of max correlation
    k_idx = np.argmax(corr_full)
    k_shift = k_idx - (n - 1)  # lag (could be negative)
    # compute gradient
    g = np.zeros_like(h)
    idx = np.arange(n)
    plus = idx + k_shift
    valid_plus = (plus >= 0) & (plus < n)
    g[valid_plus] += (1.0 - h[plus[valid_plus]])
    minus = idx - k_shift
    valid_minus = (minus >= 0) & (minus < n)
    g[valid_minus] -= h[minus[valid_minus]]
    # sum(g) maybe not zero; enforce zero-sum
    g -= g.mean()
    # compute max feasible alpha
    pos = g > 0
    neg = g < 0
    # For positive g, decreasing h (since we subtract α*g) we need α <= h_i / g_i
    # Actually h_new = h - α*g. So if g[i] > 0, then we subtract positive => h decreases.
    # Must stay >=0, so α <= h[i] / g[i].
    # If g[i] < 0, subtract negative => h increases; must stay <=1, α <= (1 - h[i]) / -g[i].
    max_alpha = np.inf
    if np.any(pos):
        max_alpha = min(max_alpha, np.min(h[pos] / g[pos]))
    if np.any(neg):
        max_alpha = min(max_alpha, np.min((1.0 - h[neg]) / (-g[neg])))
    # adjust α
    alpha = min(step, max_alpha * 0.999)  # multiply slight less to stay within bounds
    if alpha <= 0:
        # fallback to tiny random step
        candidate = h.copy()
    else:
        candidate = h - alpha * g
        # Clip
        np.clip(candidate, 0.0, 1.0, out=candidate)
        # adjust sum to target
        candidate = adjust_sum_to_target(candidate, target_sum, rng)
    cand_c5 = compute_c5(candidate)
    return candidate, cand_c5

def pairwise_move(h, step, target_sum, rng):
    n = h.shape[0]
    i, j = rng.randint(n, size=2)
    if i == j:
        return h.copy(), compute_c5(h)
    hi, hj = h[i], h[j]
    # feasible delta: new hi = hi + delta, new hj = hj - delta
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if low >= high:
        # cannot move
        return h.copy(), compute_c5(h)
    # random delta in [-step, step] intersect [low, high]
    delta = (rng.rand() * 2.0 - 1.0) * step
    delta = np.clip(delta, low, high)
    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    cand_c5 = compute_c5(cand)
    # ensure sum unchanged (by construction) and bounds okay
    return cand, cand_c5
```

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Scale/shift h in-place so that sum(h) equals target while staying in [0,1].
    Small residuals are spread uniformly and finally clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # try to fix the whole deficit/surplus using a single index repeatedly
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of old_h (defined on [0,2)) to length new_n."""
    old_n = old_h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)

def gradient_move(h: np.ndarray, step: float, target_sum: float, rng: np.random.RandomState):
    """Move opposite to the gradient of the worst shift, respecting bounds."""
    n = h.shape[0]
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    k_idx = int(np.argmax(corr_full))
    k_shift = k_idx - (n - 1)

    g = np.zeros_like(h)
    idx = np.arange(n)

    plus = idx + k_shift
    valid_plus = (plus >= 0) & (plus < n)
    g[valid_plus] += (1.0 - h[plus[valid_plus]])

    minus = idx - k_shift
    valid_minus = (minus >= 0) & (minus < n)
    g[valid_minus] -= h[minus[valid_minus]]

    # project to zero‑mean to keep total mass unchanged
    g -= g.mean()

    # maximum feasible step respecting box constraints
    pos = g > 0
    neg = g < 0
    max_alpha = np.inf
    if np.any(pos):
        max_alpha = min(max_alpha, np.min(h[pos] / g[pos]))
    if np.any(neg):
        max_alpha = min(max_alpha, np.min((1.0 - h[neg]) / (-g[neg])))
    # safety factor
    max_alpha = max_alpha * 0.999 if max_alpha != np.inf else step

    alpha = min(step, max_alpha)
    if alpha <= 0.0:
        candidate = h.copy()
    else:
        candidate = h - alpha * g
        np.clip(candidate, 0.0, 1.0, out=candidate)
        candidate = adjust_sum_to_target(candidate, target_sum, rng)

    cand_c5 = compute_c5(candidate)
    return candidate, cand_c5

def targeted_pairwise_move(h: np.ndarray, step: float, target_sum: float, rng: np.random.RandomState):
    """Exchange mass between a positive‑gradient and a negative‑gradient index."""
    n = h.shape[0]
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    k_idx = int(np.argmax(corr_full))
    k_shift = k_idx - (n - 1)

    g = np.zeros_like(h)
    idx = np.arange(n)

    plus = idx + k_shift
    vp = (plus >= 0) & (plus < n)
    g[vp] += (1.0 - h[plus[vp]])

    minus = idx - k_shift
    vm = (minus >= 0) & (minus < n)
    g[minus[vm]] -= h[minus[vm]]

    g -= g.mean()

    pos_idx = np.where(g > 0)[0]
    neg_idx = np.where(g < 0)[0]

    if pos_idx.size == 0 or neg_idx.size == 0:
        i, j = rng.randint(n, size=2)
    else:
        i = rng.choice(pos_idx)
        j = rng.choice(neg_idx)

    hi, hj = h[i], h[j]
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if low >= high:
        # fall back to a random feasible pair
        i, j = rng.randint(n, size=2)
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            return h.copy(), compute_c5(h)

    delta = (rng.rand() * 2.0 - 1.0) * step
    delta = np.clip(delta, low, high)

    cand = h.copy()
    cand[i] = hi + delta
    cand[j] = hj - delta
    cand_c5 = compute_c5(cand)
    return cand, cand_c5

def smoothing_move(h: np.ndarray, strength: float, target_sum: float, rng: np.random.RandomState):
    """Mild smoothing by a three‑point stencil."""
    kernel = np.array([0.25, 0.5, 0.25])
    smooth = np.convolve(h, kernel, mode="same")
    cand = (1.0 - strength) * h + strength * smooth
    np.clip(cand, 0.0, 1.0, out=cand)
    cand = adjust_sum_to_target(cand, target_sum, rng)
    cand_c5 = compute_c5(cand)
    return cand, cand_c5

def random_shake(h: np.ndarray, magnitude: float, target_sum: float, rng: np.random.RandomState):
    """Add Gaussian noise and re‑project."""
    cand = h + rng.randn(h.size) * magnitude
    np.clip(cand, 0.0, 1.0, out=cand)
    cand = adjust_sum_to_target(cand, target_sum, rng)
    return cand

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # small safety margin

    # ----- discretisation size -----
    n_points = 200
    target_sum = n_points / 2.0

    # ----- initialise h -----
    try:
        init = np.array(initial_h_values, dtype=float)  # may raise NameError
        h = upsample_h(init, n_points)
        np.clip(h, 0.0, 1.0, out=h)
        h = adjust_sum_to_target(h, target_sum, rng)
    except Exception:
        h = rng.rand(n_points)
        h = h / h.sum() * target_sum
        h = np.clip(h, 0.0, 1.0, out=h)
        h = adjust_sum_to_target(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    # annealing / schedule parameters
    step_initial = 0.5
    step_final = 0.001
    T_initial = 0.5
    T_final = 1e-5

    no_improve = 0
    shake_counter = 0

    while time.time() - start < max_time:
        progress = (time.time() - start) / max_time
        # exponential schedules
        step = step_initial * (step_final / step_initial) ** progress
        T = T_initial * (T_final / T_initial) ** progress

        r = rng.rand()
        if r < 0.55:

        elif r < 0.90:

        else:

        if cand_c5 < cur_c5:

        else:

            cur_c5 = cand_c5

    best_h = adjust_sum_to_target(best_h, target_sum, rng)
