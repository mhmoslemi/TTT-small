def gradient_of_overlap(h, k):
    n = h.shape[0]
    if k >= 0:
        forward = np.concatenate([1 - h[k:], np.zeros(k, dtype=h.dtype)])
        backward = np.concatenate([np.zeros(k, dtype=h.dtype), h[:n - k]])
    else:
        # k negative => shift backwards
        k_abs = -k
        # For i, forward term: (1 - h[i + k]) = (1 - h[i - k_abs]).
        # Here i + k = i - k_abs >=0 => i >= k_abs: forward = 1 - h[i - k_abs]

        # backward = -h[i - k] where i - k = i + k_abs < n => i < n - k_abs
        # So backward = np.concatenate([h[k_abs:], np.zeros(k_abs)])
        backward = np.concatenate([h[k_abs:], np.zeros(k_abs, dtype=h.dtype)])
    return forward - backward

def full_correlation_fft(h):
    n = h.shape[0]
    # compute linear correlation of h and 1-h
    a = h.astype(np.float64)
    b = (1.0 - h).astype(np.float64)
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(np.concatenate([a, np.zeros(size - n, dtype=np.float64)]))
    B = np.fft.rfft(np.concatenate([b, np.zeros(size - n, dtype=np.float64)]))
    corr = np.fft.irfft(A * np.conj(B), n=size)
    return corr[:2 * n - 1]
```

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()
    init_vec = globals().get('initial_h_values', None)
    if init_vec is not None:
        init_vec = np.array(init_vec, dtype=np.float64).ravel()
        n0 = init_vec.size
    else:
        n0 = 300
        init_vec = np.random.rand(n0)

    # define resolutions
    resolutions = [n0]
    # allow up to 3 levels if time permits
    max_level = 3

def _slsqp_refine(h0, maxiter=200):

        h_opt = _project_to_constraints(res.x, target)

def subgradient_optimize(h0, time_limit, target_sum):
    start = time.time()
    h = _project_to_constraints(h0, target_sum)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    step = 0.5
    decay = 0.999
    min_step = 1e-5
    while time.time() - start < time_limit:
        corr = _full_correlation_fft(h)
        idx = np.argmax(corr)
        lag = idx - (h.size - 1)
        grad = _gradient_overlap(h, lag)
        h_new = h - step * grad
        h_new = _project_to_constraints(h_new, target_sum)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            if best_c5 < 0.3808:
                break
        h = h_new
        step *= decay
        if step < min_step:
            step = min_step
    return best_h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    """Clip to [0,1] and enforce sum(v)=target by water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = float(np.sum(v))
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # fill the largest capacities first
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = float(np.sum(v))
    if abs(cur - target) > 1e-8 and cur > 0.0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _full_correlation_fft(h):
    """Linear correlation of h with (1-h) using FFT (full mode)."""
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    a = np.fft.rfft(np.concatenate([h, np.zeros(size - n, dtype=h.dtype)]))
    b = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(size - n, dtype=h.dtype)]))
    corr = np.fft.irfft(a * np.conj(b), n=size)
    return corr[: 2 * n - 1]

def _compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx (discrete version)."""
    n = h.size
    corr = _full_correlation_fft(h)
    return np.max(corr) * (2.0 / n)

def _gradient_overlap(h, lag):
    """Gradient of overlap O_k = sum_i h_i(1-h_{i+lag}) w.r.t. h."""
    n = h.shape[0]
    forward = np.zeros(n, dtype=h.dtype)
    backward = np.zeros(n, dtype=h.dtype)
    if lag >= 0:
        length = n - lag
        if length > 0:
            forward[:length] = 1.0 - h[lag:]
            backward[lag:] = h[:length]
    else:
        d = -lag
        length = n - d
        if length > 0:
            forward[d:] = 1.0 - h[:length]
            backward[:length] = h[d:]
    return forward - backward

def _subgradient_optimize(h0, time_limit, target):
    """Projected sub‑gradient descent on the max overlap."""
    start = time.time()
    h = _project_to_constraints(h0, target)
    best_h = h.copy()
    best_val = _compute_c5(h)
    step = 0.5
    decay = 0.999
    min_step = 1e-5
    while time.time() - start < time_limit:
        corr = _full_correlation_fft(h)
        idx = int(np.argmax(corr))
        lag = idx - (h.size - 1)
        grad = _gradient_overlap(h, lag)
        h_new = h - step * grad
        h_new = _project_to_constraints(h_new, target)
        val = _compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            if best_val < 0.3808:
                break
        h = h_new
        step *= decay
        if step < min_step:
            step = min_step
    return best_h, best_val

def _slsqp_refine(h0, maxiter=200):
    """Local refinement with SLSQP."""
    n = h0.size
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda x, tgt=target: np.sum(x) - tgt}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _compute_c5,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return h0, _compute_c5(h0)
    if not res.success:
        h_opt = _project_to_constraints(res.x, target)
    else:
        h_opt = _project_to_constraints(res.x, target)
    return h_opt, _compute_c5(h_opt)

    start_time = time.time()
