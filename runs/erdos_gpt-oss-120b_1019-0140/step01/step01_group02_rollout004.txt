def anneal(h, rng, time_remaining):
    start = time.time()
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    cur_c5 = best_c5
    n = len(h)
    target_sum = n/2.0
    # schedule: we will compute fraction elapsed relative to time_remaining
    while time.time() - start < time_remaining:
        # update parameters
        elapsed = time.time() - start
        frac = elapsed / time_remaining
        # step size decays from max_step to min_step
        step = max_step * (1-frac) + min_step * frac
        temp = max_temp * (1-frac) + min_temp * frac

        # choose indices: could select i with probability based on contribution to max overlap
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        hi = cur[i]; hj = cur[j]
        # Determine feasible delta interval to stay within [0,1] and keep sum
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        # restrict magnitude
        low = max(delta_min, -step)
        high = min(delta_max, step)
        if low >= high:
            continue
        delta = rng.random() * (high - low) + low
        cur[i] = hi + delta
        cur[j] = hj - delta
        new_c5 = compute_c5(cur)
        delta_obj = new_c5 - cur_c5
        if delta_obj < 0 or rng.random() < math.exp(-delta_obj / temp):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            cur[i] = hi
            cur[j] = hj
    # final repair
    best = repair_sum(best, target_sum)
    best_c5 = compute_c5(best)
    return best, best_c5
```

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h) matches the official evaluator
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that sum(h) == target while staying in [0,1]."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # choose an index that can move in the needed direction
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute uniformly (rare)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h


def _upscale(h_small: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of a small step function to a larger grid."""
    x_small = np.linspace(0.0, 2.0, len(h_small), endpoint=False)
    x_target = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_target, x_small, h_small)


def _anneal(start_h: np.ndarray,
            rng: np.random.Generator,
            slice_s: float,
            max_step: float,
            min_step: float,
            max_temp: float,
            min_temp: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing segment on a fixed time budget."""
    cur = start_h.copy()
    n = cur.shape[0]
    target = n / 2.0
    cur_c5 = _compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5

    start = time.time()
    while True:
        now = time.time()
        if now - start >= slice_s:
            break
        frac = (now - start) / slice_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur[i], cur[j]

        # feasible delta keeping both entries in [0,1]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        low = max(delta_min, -step)
        high = min(delta_max, step)
        if low >= high:
            continue

        delta = rng.random() * (high - low) + low
        cur[i] = hi + delta
        cur[j] = hj - delta

        new_c5 = _compute_c5(cur)
        delta_obj = new_c5 - cur_c5

        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / temp):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            cur[i] = hi
            cur[j] = hj

    best = _repair_sum(best, target)
    best_c5 = _compute_c5(best)
    return best, best_c5
