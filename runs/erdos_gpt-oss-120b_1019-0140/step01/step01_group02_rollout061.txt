def _gradient_descent_step(h, lr, alpha):
    raw_s = np.correlate(h, 1-h, mode='full')
    max_raw = np.max(raw_s)
    # compute log-sum-exp subtract max for stability
    exp_vals = np.exp(alpha * (raw_s - max_raw))
    w = exp_vals / exp_vals.sum()
    # w sum =1
    w_rev = w[::-1]
    w_sym = w + w_rev
    # compute term_sym via loop
    term = np.zeros_like(h)
    shifts = np.arange(-n+1, n)
    for k, wk in zip(shifts, w_sym):
        if wk == 0:
            continue
        if k >= 0:
            term[:n - k] += wk * h[k:]
        else:
            term[-k:] += wk * h[:n + k]
    grad = 1 - term
    # update
    h_new = h - lr * grad
    # clip and repair sum
    np.clip(h_new, 0.0, 1.0, out=h_new)
    h_new = _repair_sum(h_new, target_sum)
    return h_new

import time, math, numpy as np

def _compute_c5(h):
    n=len(h); dx=2.0/n
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def _repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # adjust one entry
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if idx.size==0:
        # fallback distribute
        h += diff / h.size
        np.clip(h, 0, 1, out=h)
        return h
    i=idx[0]
    new_val = np.clip(h[i] + diff, 0.0, 1.0)
    h[i] = new_val
    # recursion
    if abs((h.sum() - target)) > 1e-12:
        return _repair_sum(h, target)
    return h
```

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry (or distribute) so that sum(h) == target."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # choose a free entry that can move in the needed direction
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: uniform adjustment
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    # recurse if clipping prevented full correction
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h
