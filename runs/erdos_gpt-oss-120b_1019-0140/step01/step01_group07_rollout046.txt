def _project(v, target):
    v = np.clip(v,0,1)
    cur = v.sum()
    if abs(cur - target) < 1e-9:
        return v
    diff = target - cur
    # If diff >0, increase values proportionally where v <1
    # else decrease where v>0
    if diff > 0:
        free = v < 1
        # compute total capacity to add
        cap = np.sum(1.0 - v[free])
        if cap > 0:
            add = diff / cap
            v[free] += add * (1.0 - v[free])
    else:
        free = v > 0
        cap = np.sum(v[free])
        if cap > 0:
            sub = diff / cap  # diff negative
            v[free] += sub * v[free]  # subtract proportionally
    # Clip again and adjust residual
    v = np.clip(v,0,1)
    cur2 = v.sum()
    if abs(cur2-target)>1e-8:
        # fallback: scaling
        if cur2>0:
            v = v * (target/cur2)
            v = np.clip(v,0,1)
    return v

import time
import math
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Maximum overlap integral C5 for discrete step function h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust sum(v) to equal target (water-filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # fill largest caps first
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _random_binary(n, target_sum):

    v = np.zeros(n, dtype=np.float64)

    v[ones] = 1.0
    return v
