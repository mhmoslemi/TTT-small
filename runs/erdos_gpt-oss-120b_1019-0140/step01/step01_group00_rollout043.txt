def gradient_softmax(h, v, beta):
    n = len(h)
    # v: raw overlap vector
    # compute softmax weights
    v_max = v.max()
    exp_beta = np.exp(beta * (v - v_max))
    sum_exp = exp_beta.sum()
    weights = exp_beta / sum_exp
    # Compute gradient
    g = np.zeros(n)
    offset = n - 1
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            tail = slice(0, n - k)
            head = slice(k, n)
        else:
            k2 = -k
            tail = slice(k2, n)
            head = slice(0, n - k2)
        # tail contribution
        g[tail] += w * (1.0 - h[head])
        # head contribution
        g[head] -= w * h[tail]
    return g

def project_sum_and_bounds(h, target):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust sum to target via simple redistribution
    cur = h.sum()
    diff = target - cur
    # add diff uniformly
    h += diff / len(h)
    # clip again
    h = np.clip(h, 0.0, 1.0)
    # final fix residual
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # loop adjusting entries
        for i in range(len(h)):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
    return h

import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # adjust entrywise
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def optimize_h(h0, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n

    max_beta = 20.0

        c5 = v.max() * dx

            best_c5 = c5
            best_h = h.copy()

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap value multiplied by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_sum_and_bounds(x, target):
    """Clip to [0,1] and enforce sum(x) == target (simple repair)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            for i in range(len(x)):
                if abs(diff) < 1e-12:
                    break
                if diff > 0 and x[i] < 1.0:
                    inc = min(diff, 1.0 - x[i])
                    x[i] += inc
                    diff -= inc
                elif diff < 0 and x[i] > 0.0:
                    dec = min(-diff, x[i])
                    x[i] -= dec
                    diff += dec
    return x

def optimize_h(h0, time_budget, seed):
    """Gradient descent on a smooth approximation of the max‑overlap."""
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 2.0          # smoothing parameter for soft‑max
    max_beta = 20.0
    alpha0 = 0.2        # base step size
    no_improve = 0

    while time.time() - start < time_budget:
        # overlap vector (raw, without dx)
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = v.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase smoothing slowly
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

        # soft‑max weights (stable)
        vmax = v.max()
        w = np.exp(beta * (v - vmax))
        w /= w.sum()

        # gradient of the smooth objective
        g = np.zeros(n)
        offset = n - 1
        for idx, weight in enumerate(w):
            if weight < 1e-16:
                continue
            k = idx - offset
            if k >= 0:
                tail = slice(0, n - k)
                head = slice(k, n)
            else:
                k2 = -k
                tail = slice(k2, n)
                head = slice(0, n - k2)
            g[tail] += weight * (1.0 - h[head])
            g[head] -= weight * h[tail]

        # enforce zero‑sum direction (preserve total mass)
        g -= np.mean(g)

        # step‑size schedule (linear decay)
        elapsed = time.time() - start
        alpha = alpha0 * max(0.1, 1.0 - elapsed / time_budget)

        h_new = h - alpha * g
        h_new = project_sum_and_bounds(h_new, target)

        # accept if we improve (or tiny random acceptance)
        c5_new = compute_c5(h_new)
        if c5_new < c5 or rng.random() < 0.001:
            h = h_new

        # random shake when stuck
        if no_improve > 3000:
            h = project_sum_and_bounds(h + rng.normal(scale=0.02, size=n), target)
            no_improve = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    h0 = project_sum_and_bounds(h0, n / 2.0)
    remaining = max(0.0, budget_s - (time.time() - start))
    best_h, best_c5 = optimize_h(h0, remaining, seed + 1)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
