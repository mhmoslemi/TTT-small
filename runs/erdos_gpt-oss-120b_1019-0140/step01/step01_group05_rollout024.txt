def _local_search(h, time_limit, rng, step=0.02):
   cur = h.copy()
   cur_score = _c5_value_fast(cur)
   N = cur.size
   start = time.time()
   while time.time() - start < time_limit:
       i, j = rng.integers(N, size=2)
       if i == j: continue
       up = min(1 - cur[i], cur[j])
       down = min(cur[i], 1 - cur[j])
       if up == 0 and down == 0: continue
       # small step: epsilon = rng.uniform(-step, step) truncated by bounds
       eps = (rng.random() - 0.5) * 2 * step
       eps = max(-down, min(up, eps))
       if eps == 0: continue
       new = cur.copy()
       new[i] += eps
       new[j] -= eps
       new_score = _c5_value_fast(new)
       if new_score < cur_score:
           cur = new
           cur_score = new_score
   return cur, cur_score

def _repair_vector(v):
    # Ensure dtype float64
    v = np.asarray(v, dtype=np.float64)
    np.clip(v, 0.0, 1.0, out=v)
    N = v.size
    target = N / 2.0
    s = v.sum()
    if s == target:
        return v
    # adjust by scaling and clipping iteratively
    for _ in range(10):
        if s == 0:
            # fill equally
            v[:] = target / N
            np.clip(v, 0.0, 1.0, out=v)
            s = v.sum()
            continue
        factor = target / s
        v *= factor
        np.clip(v, 0.0, 1.0, out=v)
        s = v.sum()
        if abs(s - target) < 1e-12:
            break
    # final correction distribute residual
    residual = target - s
    if abs(residual) > 1e-12:
        # distribute to positions not at bounds
        free = np.where((v > 0) & (v < 1))[0]
        if free.size > 0:
            delta = residual / free.size
            v[free] += delta
            # clip again
            np.clip(v, 0.0, 1.0, out=v)
        else:
            # if no free, adjust all entries proportionally
            v += residual / N
            np.clip(v, 0.0, 1.0, out=v)
    # recalc sum may be a bit off but final clipping ensures sum within tolerance
    return v

def _c5_value(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    v = np.asarray(v, dtype=np.float64)
    np.clip(v, 0.0, 1.0, out=v)
    N = v.size
    target = N / 2.0
    s = v.sum()
    # Simple scaling + clipping loop
    for _ in range(12):
        if abs(s - target) <= 1e-12:
            break
        if s == 0.0:
            v[:] = target / N
            np.clip(v, 0.0, 1.0, out=v)
            s = v.sum()
            continue
        factor = target / s
        v *= factor
        np.clip(v, 0.0, 1.0, out=v)
        s = v.sum()
    # Small residual correction on free entries
    residual = target - s
    if abs(residual) > 1e-12:
        free = np.where((v > 0) & (v < 1))[0]
        if free.size > 0:
            delta = residual / free.size
            v[free] += delta
            np.clip(v, 0.0, 1.0, out=v)
        else:
            v += residual / N
            np.clip(v, 0.0, 1.0, out=v)
    return v


def _c5_value_fast(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function using FFT."""
    N = h.size
    dx = 2.0 / N
    g = 1.0 - h
    g_rev = g[::-1]
    size = 2 * N - 1
    # Pad to the required size
    pad_len = size - N
    H = np.fft.rfft(np.concatenate([h, np.zeros(pad_len)]))
    G = np.fft.rfft(np.concatenate([g_rev, np.zeros(pad_len)]))
    conv = np.fft.irfft(H * G, n=size)
    max_corr = np.max(conv)
    return float(max_corr * dx)

    return _c5_value_fast(h)

                                max_eps_end: float = 0.02,

    h_cur = h_start.copy()
    cur_score = _c5_value_fast(h_cur)
    best_h = h_cur.copy()

        frac = max(0.0, 1.0 - elapsed / time_limit)      # 1 → 0
        T = T0 * frac
        max_eps = max_eps_start * frac + max_eps_end * (1.0 - frac)

        i = rng.integers(N)
        j = rng.integers(N - 1)
