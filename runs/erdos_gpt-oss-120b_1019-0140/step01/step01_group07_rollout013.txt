def _worst_lag_and_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    # compute gradient
    g = np.ones_like(h)
    if lag >= 0:
        # forward shift: indexes 0..n-lag-1 get h[i+lag]
        g[:n-lag] -= h[lag:]
        # backward shift: indexes lag..n-1 get h[i-lag]
        g[lag:] -= h[:n-lag]
    else:
        d = -lag
        # forward: shift by -d, i.e., forward shift left
        g[d:] -= h[:n-d]
        # backward: shift right by d
        g[:n-d] -= h[d:]
    return lag, g
```

def _subgradient_optimize(h0, target_sum, max_iter, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_score = _compute_c5(h)
    init_alpha = 0.8
    for t in range(1, max_iter+1):
        if time.time() - start_time > time_limit:
            break
        lag, g = _worst_lag_and_gradient(h)
        step = init_alpha / np.sqrt(t)
        h = h - step * g
        # project
        h = _project_to_constraints(h, target_sum)
        score = _compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # occasionally inject noise
        if t % 200 == 0:
            # random jitter
            perturb = np.random.randn(h.size) * 0.01
            h = _project_to_constraints(h + perturb, target_sum)
    return best_h, best_score
```

import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Compute max_k ∫ h(x)(1-h(x+k))dx via discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip to [0,1] and enforce sum(v)=target by water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _worst_lag_and_gradient(h):
    """Return lag giving max correlation and the gradient of that correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    g = np.ones_like(h)
    if lag >= 0:
        # forward contribution
        g[:n - lag] -= h[lag:]
        # backward contribution
        g[lag:] -= h[:n - lag]
    else:
        d = -lag
        g[d:] -= h[:n - d]
        g[:n - d] -= h[d:]
    return lag, g

def _subgradient_optimize(h0, target_sum, max_iter, max_time):
    """Projected subgradient descent on max correlation."""
    h = h0.copy()
    best_h = h.copy()
    best_score = _compute_c5(h)
    alpha0 = 0.8
    start = time.time()
    for t in range(1, max_iter + 1):
        if time.time() - start > max_time:
            break
        lag, g = _worst_lag_and_gradient(h)
        step = alpha0 / np.sqrt(t)
        h = h - step * g
        h = _project_to_constraints(h, target_sum)
        score = _compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        if t % 200 == 0:
            # occasional jitter to escape plateaus
            pert = np.random.randn(h.size) * 0.01
            h = _project_to_constraints(h + pert, target_sum)
    return best_h, best_score

def _refine_with_slsqp(h, target_sum, time_left):
    """Perform a short SLSQP refinement if time permits."""
    if time_left < 5.0:
        return h
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * h.size
    try:
        res = minimize(_compute_c5, h, method="SLSQP",
                       bounds=bounds, constraints=[cons],
                       options={"maxiter": 200, "ftol": 1e-9, "disp": False})
        if res.success:
            h = _project_to_constraints(res.x, target_sum)
    except Exception:
        pass
    return h
