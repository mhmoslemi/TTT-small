def anneal(initial_h, start_c5, start_time, budget):
   h = initial_h.copy()
   cur_c5 = start_c5
   best_h = h.copy()
   best_c5 = cur_c5

   n = len(h)
   rng = np.random.default_rng()
   # parameters
   T0 = 0.001 # initial temperature (maybe 0.001? but delta values are maybe ~0.0001)
   T = T0
   decay = 0.9995 # per iteration

   while time.time() - start_time < budget:
      # propose swap
      # pick a random 1 and 0
      ones = np.flatnonzero(h == 1)
      zeros = np.flatnonzero(h == 0)
      i = rng.choice(ones)
      j = rng.choice(zeros)
      # swap
      h[i], h[j] = 0, 1
      new_c5 = compute_c5(h)
      delta = new_c5 - cur_c5
      accept = False
      if delta < 0 or rng.random() < np.exp(-delta / T):
         accept = True
         cur_c5 = new_c5
         if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
      if not accept:
         # revert swap
         h[i], h[j] = 1, 0
      # update temperature
      T *= decay
      # perhaps restart temperature periodically
      if T < 1e-9:
         T = T0
   return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_binary(n, rng):
    """Return binary vector of length n with exactly n/2 ones."""
    vec = np.zeros(n, dtype=int)
    idx = rng.choice(n, n // 2, replace=False)
    vec[idx] = 1
    return vec

def tournament_select(pop, scores, k, rng):
    """Select index from population using tournament of size k."""
    idxs = rng.integers(0, len(pop), size=k)
    best = idxs[0]
    best_score = scores[best]
    for i in idxs[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return best

def crossover(parent1, parent2, rng):
    """Uniform crossover producing child and repair to exact sum."""
    mask = rng.random(parent1.shape) < 0.5
    child = np.where(mask, parent1, parent2).astype(int)
    # repair to n/2 ones
    n = len(child)
    target = n // 2
    cur = child.sum()
    if cur > target:
        # too many ones: turn random ones to zero
        excess = cur - target
        ones_idx = np.flatnonzero(child == 1)
        to_zero = rng.choice(ones_idx, excess, replace=False)
        child[to_zero] = 0
    elif cur < target:
        # too few ones: turn random zeros to one
        deficit = target - cur
        zeros_idx = np.flatnonzero(child == 0)
        to_one = rng.choice(zeros_idx, deficit, replace=False)

    ones = np.flatnonzero(vec == 1)
    zeros = np.flatnonzero(vec == 0)

    i = rng.choice(ones)
    j = rng.choice(zeros)
    vec[i], vec[j] = 0, 1

    """Simple hill climbing with swaps, strictly decreasing c5."""
    h = start_h.copy()

        j = rng.choice(zeros)

            best_val = new_val
            best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1, 0
    return best_h, best_val

import time
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def random_binary(n, rng):
    v = np.zeros(n, dtype=int)
    idx = rng.choice(n, n // 2, replace=False)
    v[idx] = 1
    return v

def tournament_select(pop, scores, k, rng):
    idxs = rng.integers(0, len(pop), size=k)
    best = idxs[0]
    best_score = scores[best]
    for i in idxs[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return best

def crossover(p1, p2, rng):
    mask = rng.random(p1.shape) < 0.5
    child = np.where(mask, p1, p2).astype(int)
    n = len(child)
    target = n // 2
    cur = child.sum()
    if cur > target:
        excess = cur - target
        ones = np.flatnonzero(child == 1)
        child[rng.choice(ones, excess, replace=False)] = 0
    elif cur < target:
        deficit = target - cur
        zeros = np.flatnonzero(child == 0)
        child[rng.choice(zeros, deficit, replace=False)] = 1
    return child

def mutate_swap(v, rng):
    ones = np.flatnonzero(v == 1)
    zeros = np.flatnonzero(v == 0)
    if len(ones) == 0 or len(zeros) == 0:
        return
    i = rng.choice(ones)
    j = rng.choice(zeros)
    v[i], v[j] = 0, 1

    best_val = start_c5
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        val = compute_c5(h)
        if val + 1e-12 < best_val:
            best_val = val
            best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best_h, best_val

def anneal(start_h, start_c5, start_time, budget):
    h = start_h.copy()
    cur = start_c5
    best_h = h.copy()
    best = cur
    rng = np.random.default_rng()
    T = 1e-3
    decay = 0.9999
    while time.time() - start_time < budget:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        new = compute_c5(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
        T *= decay
        if T < 1e-9:
            T = 1e-3
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # problem size
    if 'initial_h_values' in globals():
        hv = np.array(initial_h_values, dtype=float)
        n = len(hv)
        binary = np.zeros(n, dtype=int)
        binary[np.argsort(-hv)[: n // 2]] = 1
        init_vec = binary
    else:
        n = 80
        init_vec = random_binary(n, rng)

    pop_size = 30
    tournament_k = 5
    mut_prob = 0.3

    pop = np.empty((pop_size, n), dtype=int)
    scores = np.empty(pop_size, dtype=float)
    pop[0] = init_vec
    scores[0] = compute_c5(pop[0])
    for i in range(1, pop_size):
        pop[i] = random_binary(n, rng)
        scores[i] = compute_c5(pop[i])

    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_c5 = scores[best_idx]

    # genetic phase
    while time.time() - start_time < budget_s * 0.85:
        children = np.empty_like(pop)
        child_scores = np.empty(pop_size, dtype=float)
        for i in range(pop_size):
            p1 = tournament_select(pop, scores, tournament_k, rng)
            p2 = tournament_select(pop, scores, tournament_k, rng)
            child = crossover(pop[p1], pop[p2], rng)
            if rng.random() < mut_prob:
                mutate_swap(child, rng)
            children[i] = child
            child_scores[i] = compute_c5(child)
        combined = np.concatenate((pop, children), axis=0)
        combined_scores = np.concatenate((scores, child_scores), axis=0)
        keep = np.argpartition(combined_scores, pop_size)[:pop_size]
        pop = combined[keep]
        scores = combined_scores[keep]
        cur_best = int(np.argmin(scores))
        if scores[cur_best] + 1e-12 < best_c5:
            best_c5 = scores[cur_best]
            best_h = pop[cur_best].copy()
        if best_c5 < 0.3805:
            break

    # deterministic hill climbing
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 0.5
    if remaining > 0.5:
        refined_h, refined_c5 = hill_climb_strict(best_h, best_c5,
                                                 time.time(), remaining)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # simulated annealing refinement
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 0.1
    if remaining > 0.2:
        refined_h, refined_c5 = anneal(best_h, best_c5,
                                       time.time(), remaining)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    best_h = best_h.astype(float)
    cur_sum = best_h.sum()
    target = n / 2.0
    if abs(cur_sum - target) > 1e-9:
        best_h += (target - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
