    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    # parameters
    max_iters = 200000  # or until time runs out
    step = 0.1  # max delta transfer (observing bounds)
    for iter in range(max_iters):
        if time.time() - start_time > time_budget:
            break
        # compute correlation and find worst lag
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        # find candidate indices where we can move mass from i to i+lag (if lag>=0)
        if lag >= 0:
            i_candidates = np.where((best_h[:n-lag] > 1e-8) & (best_h[lag:] < 1 - 1e-8))[0]
        else:
            # lag negative, shift left - we can move mass from i to i+lag (i+lag < i)
            lag_abs = -lag
            i_candidates = np.where((best_h[lag_abs:] > 1e-8) & (best_h[:n-lag_abs] < 1 - 1e-8))[0] + lag_abs
            # Actually when lag<0, we want i such that i+lag is index earlier; choose i where i+lag within range.
            # We'll compute i indices for which i and i+lag within bounds.

            i = np.random.choice(i_candidates)
            j = i + lag
            # ensure j in [0,n)
            # Determine delta
            delta_max = min(best_h[i], 1 - best_h[j], step)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=20):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # cannot adjust interior, distribute onto extremal points with room
            # use all indices where not at bound in direction of diff
            if diff > 0:
                free = np.where(h < 1.0)[0]
            else:
                free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction on a single element
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h) if cur < target_sum else np.argmax(h)
        # adjust within bounds
        adj = target_sum - cur
        h[i] = np.clip(h[i] + adj, 0.0, 1.0)
    return h

def local_search(h_init, end_time, rng):
    n = len(h_init)
    h = h_init.copy()
    best_c = compute_c5(h)
    step_max = 0.2
    while time.time() < end_time:
        # compute correlation and worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        l = np.argmax(corr)
        lag = l - (n - 1)
        if lag >= 0:
            idx_range = n - lag
            if idx_range > 0:
                i_cand = np.where((h[:idx_range] > 1e-12) & (h[lag:] < 1 - 1e-12))[0]
                if i_cand.size:
                    i = int(rng.choice(i_cand))
                    j = i + lag
                    delta_max = min(h[i], 1 - h[j], step_max)
                    if delta_max > 0:
                        delta = delta_max * rng.random()
                        h_new = h.copy()
                        h_new[i] -= delta
                        h_new[j] += delta
                        c_new = compute_c5(h_new)
                        if c_new < best_c:
                            h = h_new
                            best_c = c_new
                            continue
        else:
            d = -lag
            idx_range = n - d
            if idx_range > 0:
                i_cand = np.where((h[d:] > 1e-12) & (h[:idx_range] < 1 - 1e-12))[0] + d
                if i_cand.size:
                    i = int(rng.choice(i_cand))
                    j = i - d
                    delta_max = min(h[i], 1 - h[j], step_max)
                    if delta_max > 0:
                        delta = delta_max * rng.random()
                        h_new = h.copy()
                        h_new[i] -= delta
                        h_new[j] += delta
                        c_new = compute_c5(h_new)
                        if c_new < best_c:
                            h = h_new
                            best_c = c_new
                            continue
        # fallback random transfer
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] > 1e-12 and h[j] < 1 - 1e-12:
            delta_max = min(h[i], 1 - h[j], step_max)
            if delta_max > 0:
                delta = delta_max * rng.random()
                h_new = h.copy()

                h_new[j] += delta

import time
import numpy as np

def compute_c5(h):
    """Compute the Erdős overlap constant for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=20):
    """Project vector h onto [0,1]^n and the hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # No interior points; use any point that can move in the needed direction
            free = np.where((diff > 0) & (h < 1.0) | (diff < 0) & (h > 0.0))[0]
            if free.size == 0:
                break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # Tiny final correction
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h) if cur < target_sum else np.argmax(h)
        adj = target_sum - cur
        h[i] = np.clip(h[i] + adj, 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def local_search(h_init, end_time, rng):
    """
    Simple hill‑climbing: move mass between points to reduce the maximal overlap.
    Always preserves the integral constraint.
    """
    eps = 1e-12
    step_max = 0.5
    h = h_init.copy()
    best_c = compute_c5(h)
    n = h.shape[0]

    while time.time() < end_time:
        corr = np.correlate(h, 1.0 - h, mode="full")
        lag_idx = int(np.argmax(corr))
        lag = lag_idx - (n - 1)

        # Targeted move on the worst lag
        if lag >= 0:
            length = n - lag
            if length > 0:
                hi = h[:length]
                hj = h[lag:]
                mask = (hi > eps) & (hj < 1.0 - eps)
                if np.any(mask):
                    contrib = hi * (1.0 - hj)
                    contrib = np.where(mask, contrib, -1.0)
                    i_local = int(np.argmax(contrib))
                    i = i_local

                    delta_max = min(h[i], 1.0 - h[j], step_max)
                    if delta_max > 0:
                        delta = delta_max * rng.random()

                        h_new[i] -= delta
                        h_new[j] += delta
                        c_new = compute_c5(h_new)
                        if c_new < best_c:

                            best_c = c_new
                            continue
        else:

            length = n - d
            if length > 0:
                hi = h[d:]           # i >= d
                hj = h[:length]      # j = i - d
                mask = (hi > eps) & (hj < 1.0 - eps)
                if np.any(mask):
                    contrib = hi * (1.0 - hj)
                    contrib = np.where(mask, contrib, -1.0)
                    i_local = int(np.argmax(contrib))
                    i = i_local + d
                    j = i - d
                    delta_max = min(h[i], 1.0 - h[j], step_max)
                    if delta_max > 0:
                        delta = delta_max * rng.random()
                        h_new = h.copy()
                        h_new[i] -= delta
                        h_new[j] += delta
                        c_new = compute_c5(h_new)
                        if c_new < best_c:
                            h = h_new
                            best_c = c_new
                            continue

        # Fallback random transfer
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] > eps and h[j] < 1.0 - eps:
            delta_max = min(h[i], 1.0 - h[j], step_max)
            if delta_max > 0:
                delta = delta_max * rng.random()
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    h = h_new
                    best_c = c_new

        # Slowly shrink the step size
        if rng.random() < 0.01:
            step_max *= 0.99

    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Resolutions to explore (smaller sizes are faster)
    candidate_ns = [95, 190, 380, 760]

    for n in candidate_ns:
        if time.time() - start > budget_s * 0.97:
            break
        # Number of restarts (more for smaller n)
        restarts = 12 if n <= 190 else 6
        for _ in range(restarts):
            if time.time() - start > budget_s * 0.97:
                break

            # Initialise a feasible h
            if base_h is not None and base_h.shape[0] == n:
                h0 = base_h.copy()
            elif base_h is not None:
                h0 = resample_h(base_h, n)
            else:
                h0 = rng.random(n)
                h0 = project_to_feasible(h0, n / 2.0)

            remaining = budget_s - (time.time() - start)
            if remaining <= 5:
                break
            # Allocate up to 30 s for this restart
            time_for_restart = min(30.0, remaining * 0.3)
            end_time = time.time() + time_for_restart

            h_opt, c5_val = local_search(h0, end_time, rng)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

            # Early exit condition (continue a bit to possibly improve further)
            if best_c5 <= 0.3808:
                pass

    # Final refinement with whatever time remains
    remaining = budget_s - (time.time() - start)
    if remaining > 5 and best_h is not None:
        end_time = start + budget_s - 0.5
        best_h, best_c5 = local_search(best_h, end_time, rng)

    # Fallback if nothing was found
    if best_h is None:
        best_n = 95
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Ensure feasibility of the final answer
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
