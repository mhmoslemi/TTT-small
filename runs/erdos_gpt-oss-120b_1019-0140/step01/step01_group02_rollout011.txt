    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

We can implement a simple function that computes diff = target - sum, then add diff to first element that can accommodate.

```
def repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Find index that can adjust
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute uniformly
        h += diff / h.size
        np.clip(h, 0, 1, out=h)
        return h
    i = idx[0]
    new_val = np.clip(h[i] + diff, 0.0, 1.0)
    actual = new_val - h[i]
    h[i] = new_val
    # If not all diff applied due to clipping, recur
    if abs(actual - diff) > 1e-12:
        return repair_sum(h, target)
    return h
```

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that the sum of h equals the target."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = np.clip(h[i] + diff, 0.0, 1.0)
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h


def _grad_given_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of C5 w.r.t. h for a fixed shift (worst shift)."""
    n = h.shape[0]
    if shift >= 0:
        # term h[i+shift]
        plus = np.empty(n, dtype=h.dtype)
        plus[: n - shift] = h[shift:]
        plus[n - shift :] = 0.0
        # term h[i-shift]
        minus = np.empty(n, dtype=h.dtype)
        minus[shift:] = h[: n - shift]
        minus[:shift] = 0.0
    else:
        s = -shift
        plus = np.empty(n, dtype=h.dtype)
        plus[s:] = h[: n - s]
        plus[:s] = 0.0
        minus = np.empty(n, dtype=h.dtype)
        minus[: n - s] = h[s:]
        minus[n - s :] = 0.0
    return 1.0 - plus - minus

    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """

    start_time = time.time()

    max_temp = 0.08          # early‑stage temperature
    min_temp = 1e-6

    used_init = False

                cur_h[j] -= delta
                stall_counter += 1

    # ensure we have a feasible best solution

    else:

        best_c5 = _compute_c5(best_h)
