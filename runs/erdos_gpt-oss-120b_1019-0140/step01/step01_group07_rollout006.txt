def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    N = 2 * n
    rev_one = (1.0 - h)[::-1]
    # FFT computation
    fh = np.fft.fft(h, N)

import numpy as np, time
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-10:
        return v
    # water-filling
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        if free.size > 0:
            caps = 1.0 - v[free]
            # allocate proportionally maybe uniform
            # We can distribute evenly until caps exhausted
            while need > 1e-12 and free.size > 0:
                inc = need / free.size
                inc = np.minimum(inc, caps)
                v[free] += inc
                need -= inc.sum()
                # remove saturated
                mask = v[free] >= 1.0 - 1e-12
                free = free[~mask]
                caps = caps[~mask]
                caps = 1.0 - v[free]
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        if free.size > 0:
            caps = v[free]
            while excess > 1e-12 and free.size > 0:
                dec = excess / free.size
                dec = np.minimum(dec, caps)
                v[free] -= dec
                excess -= dec.sum()
                mask = v[free] <= 1e-12
                free = free[~mask]
                caps = v[free]
    # final clip
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(h):
    n = h.shape[0]

    fr = np.fft.fft(rev_one, N)
    conv = np.fft.ifft(fh * fr).real

def _project_to_constraints(v, target):
    """Clip to [0,1] and enforce sum(v)=target by water-filling."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-9:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        while need > 1e-12 and free.size:
            inc = need / free.size
            caps = 1.0 - v[free]
            incs = np.minimum(inc, caps)
            v[free] += incs
            need -= incs.sum()
            free = free[v[free] < 1.0 - 1e-12]
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        while excess > 1e-12 and free.size:
            dec = excess / free.size
            caps = v[free]
            decs = np.minimum(dec, caps)
            v[free] -= decs
            excess -= decs.sum()
            free = free[v[free] > 1e-12]

    cur = v.sum()
    if abs(cur - target) > 1e-8 and cur > 0:
        v *= target / cur
        v = np.clip(v, 0.0, 1.0)

def _compute_c5(h):
    """Discrete evaluation of max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    N = 2 * n
    rev_one = (1.0 - h)[::-1]
    fh = np.fft.fft(h, N)
    fr = np.fft.fft(rev_one, N)
    conv = np.fft.ifft(fh * fr).real
    max_corr = np.max(conv[:2 * n - 1])
    return max_corr * dx

    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)

    return best_h.astype(np.float64), float(final_score), int(best_h.size)
