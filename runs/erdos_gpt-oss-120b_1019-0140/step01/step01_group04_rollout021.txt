def compute_c5(v):
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx
```

4. Define shift_array(v, shift): if shift>=0 -> left, else right.

```
def shift_array(v, shift):
    n = len(v)
    if shift > 0:
        if shift >= n:
            return np.zeros_like(v)
        return np.concatenate((v[shift:], np.zeros(shift)))
    elif shift < 0:
        s = -shift
        if s >= n:
            return np.zeros_like(v)
        return np.concatenate((np.zeros(s), v[:n-s]))
    else:
        return v.copy()
```

import time
import numpy as np

def _repair(v, n):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    s = v.sum()
    diff = s - target
    if abs(diff) > 1e-12:
        for _ in range(12):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                mask = v > 0.0
            else:
                mask = v < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            adjust = diff / cnt
            v[mask] -= adjust
            v = np.clip(v, 0.0, 1.0)
            s = v.sum()
            diff = s - target
    s = v.sum()
    if abs(s - target) > 1e-9:
        if s > 0:
            v *= target / s
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            v += (target - s) / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _random_feasible(n):
    return _repair(np.random.rand(n), n)

def _shift(v, k):
    """Shift left by k (>0) or right by -k (<0); pad with zeros."""
    n = len(v)
    if k > 0:
        if k >= n:
            return np.zeros_like(v)
        return np.concatenate((v[k:], np.zeros(k)))
    elif k < 0:
        k = -k
        if k >= n:
            return np.zeros_like(v)
        return np.concatenate((np.zeros(k), v[:n - k]))
    else:
        return v.copy()

def _full_corr(v):
    """Full linear correlation of v and (1-v) using FFT."""
    n = v.shape[0]
    m = 2 * n - 1
    L = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(v, n=L)
    fb = np.fft.rfft(1.0 - v, n=L)
    corr = np.fft.irfft(fa * np.conj(fb), n=L)[:m]
    return corr

def _c5(v):
    """Return C5 bound for vector v."""
    n = v.shape[0]
    return np.max(_full_corr(v)) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # initial parent (if provided)
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 200
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0 // 2] = 1.0

    # target resolution (up‑sample if possible)
    target_n = min(760, 800)
    if target_n != n0:
        src_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        tgt_x = np.linspace(0.0, 2.0, target_n, endpoint=False)
        parent = np.interp(tgt_x, src_x, parent, left=parent[0], right=parent[-1])
        n = target_n
    else:
        n = n0

    v = _repair(parent, n)
    curr_score = _c5(v)
    best_v = v.copy()
    best_score = curr_score

    step = 0.2
    min_step = 1e-4
    decay = 0.9995
    stagnation = 0
    max_stagn = 500

    while True:
        if time.time() - start > budget_s - 0.5:
            break

        corr = _full_corr(v)
        k_idx = int(np.argmax(corr))
        k = k_idx - (n - 1)

        # gradient descent on the worst shift
        fwd = _shift(v, k)
        bwd = _shift(v, -k)
        grad = 1.0 - (fwd + bwd)
        v_new = _repair(v - step * grad, n)
        new_score = _c5(v_new)

        improved = False
        if new_score < curr_score:
            v, curr_score = v_new, new_score
            improved = True
        else:
            # random two‑coordinate exchange as fallback
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.1  # ±0.05
            cand = v.copy()
            cand[i] += delta
            cand[j] -= delta
            cand = _repair(cand, n)
            cand_score = _c5(cand)
            if cand_score < curr_score:
                v, curr_score = cand, cand_score
                improved = True

        if improved:
            if curr_score < best_score:
                best_score = curr_score
                best_v = v.copy()
            stagnation = 0
        else:
            stagnation += 1

        step = max(min_step, step * decay)

        if stagnation >= max_stagn:
            # random restart
            v = _random_feasible(n)
            curr_score = _c5(v)
            stagnation = 0
            step = 0.2

    best_v = _repair(best_v, n)
    final_score = _c5(best_v)
    return best_v.astype(np.float64), float(final_score), int(n)
