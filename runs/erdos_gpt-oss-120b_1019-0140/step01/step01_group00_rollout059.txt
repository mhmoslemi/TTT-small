def random_feasible(n, rng):
    target = n / 2.0
    # start with uniform random values
    x = rng.random(n)
    # while loop:
    for _ in range(10):
        # ensure sum = target
        sumx = np.sum(x)
        if sumx == 0:
            # rare case: set uniform
            x = np.full(n, target / n)
            break
        x = x * (target / sumx)
        # clip
        over = x > 1.0
        if not np.any(over):
            break
        # for overshoot, set to 1 and reduce remainder proportionally among non-over indices
        excess = np.sum(x[over] - 1.0)
        x[over] = 1.0
        # adjust remaining
        remain_idx = ~over
        sum_remain = np.sum(x[remain_idx])
        if sum_remain > 0:
            x[remain_idx] -= excess * (x[remain_idx] / sum_remain)
            # ensure non-negatives
            x[remain_idx] = np.maximum(x[remain_idx], 0.0)
        # continue until no overshoot
    # final adjust: small residual diff
    diff = target - np.sum(x)
    # distribute diff across non-boundary indices (0,1) proportionally
    # We'll add diff/n to all entries then clip again
    # but for simplicity, do one final scaling
    x = x + diff / n
    x = np.clip(x, 0.0, 1.0)
    # adjust final to exact sum via distributing small amounts
    current = np.sum(x)
    diff = target - current
    # small adjust
    if abs(diff) > 1e-12:
        # if diff>0, add to some index not at 1; else subtract from some non-zero
        for i in range(n):
            if diff == 0:
                break
            if diff > 0 and x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
    return x

Define function make_random_continuous(n, rng):
    # generate random uniform [0,1] and then adjust to sum exactly target
    x = rng.random(n)
    # scaling and projection loop
    target = n / 2.0
    # Basic scaling to target sum
    x = x * (target / np.sum(x))
    # Clip and adjust if exceed 1
    for _ in range(5):
        over = x > 1.0
        if not over.any():
            break
        excess = np.sum(x[over] - 1.0)
        x[over] = 1.0
        # distribute excess among non-over elements proportionally
        under = ~over
        sum_under = np.sum(x[under])
        if sum_under > 0:
            x[under] -= excess * (x[under] / sum_under)
        # after subtraction, may produce negative values; clip and loop
        x = np.clip(x, 0.0, 1.0)
    # final adjustment for any sum diff
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        # add diff equally across non-boundary entries
        free = (x > 0.0) & (x < 1.0)
        if np.any(free):
            count_free = np.sum(free)
            x[free] += diff / count_free
            # clip again
            x = np.clip(x, 0.0, 1.0)
        else:
            # if none free, just adjust first entry
            x[0] = min(max(x[0] + diff, 0.0), 1.0)
    # Ensure sum exact
    # If still off due to floating, adjust small
    cur_sum = np.sum(x)
    if abs(cur_sum - target) > 1e-9:
        # tiny scaling of all entries not at bounds
        # find indices where 0<x<1
        free = (x > 0.0) & (x < 1.0)
        if np.any(free):
            scale = (target) / cur_sum
            x[free] *= scale
            # keep within [0,1]
            x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # determine n
    if 'initial_h_values' in globals():
        h0_initial = np.array(initial_h_values, dtype=float)
        n = len(h0_initial)
    else:
        n = 80
        h0_initial = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # Helper constraints
    class SumConstraint:
        def __call__(self, x):
            return np.sum(x[:n]) - target_sum

    class OverlapConstraint:
        def __init__(self, k):
            self.k = k
            self.dx = dx
        def __call__(self, x):
            h = x[:n]
            k = self.k
            if k >= 0:
                overl = np.dot(h[:n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[:n - kk])
            return x[-1] - overl * self.dx

    # Prebuild constraints list
    overlap_constraints = [OverlapConstraint(k) for k in range(-n+1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': SumConstraint()}] + \
                 [{'type': 'ineq', 'fun': oc} for oc in overlap_constraints]

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    best_h = None
    best_c5 = np.inf

    # Initial candidate from parent if available, make binary and evaluate
    if h0_initial is not None:
        # Ensure within bounds and sum exactly target
        # Use binary version by selecting top n/2 entries
        binary_h = np.zeros(n, dtype=float)
        top_idx = np.argsort(-h0_initial)[: n // 2]
        binary_h[top_idx] = 1.0
        c5 = np.max(np.correlate(binary_h, 1 - binary_h, mode='full')) * dx
        best_h = binary_h
        best_c5 = c5

    # Continuous SLSQP phase
    # Allocate up to 70% of budget
    while time.time() - start < budget_s * 0.7:
        # random continuous start
        h_start = make_random_continuous(n, rng)
        t_start = np.max(np.correlate(h_start, 1 - h_start, mode='full')) * dx
        x0 = np.concatenate([h_start, [t_start]])

        # minimize
        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 800, 'ftol': 1e-9, 'disp': False}
            )
        except Exception:
            break

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        # enforce sum exactly
        sum_h = np.sum(h_opt)
        if abs(sum_h - target_sum) > 1e-9:
            # adjust uniformly
            h_opt += (target_sum - sum_h) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)

        c5_opt = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
        if c5_opt < best_c5 - 1e-12:
            best_h = h_opt
            best_c5 = c5_opt
            # early exit if good enough
            if best_c5 < 0.3805:
                break

    # Binary improvement phase
    if best_h is not None:
        # Convert to binary (if not already)
        if best_h.dtype != int and not np.all(np.logical_or(best_h == 0, best_h == 1)):
            # take top n/2 entries as 1
            binary_h = np.zeros(n, dtype=int)
            top_idx = np.argsort(-best_h)[: n // 2]
            binary_h[top_idx] = 1
        else:
            binary_h = (best_h > 0.5).astype(int)  # fallback
        c5_bin = np.max(np.correlate(binary_h, 1 - binary_h, mode='full')) * dx
        # Hill climbing with many iterations
        # Use remaining time
        remain_time = budget_s - (time.time() - start)
        if remain_time > 1.0:
            binary_h, c5_bin = hill_climb_binary_best(binary_h, c5_bin, start, budget_s, rng)
            if c5_bin < best_c5 - 1e-12:
                best_h = binary_h.astype(float)
                best_c5 = c5_bin

    # Final adjust to ensure sum exactly target
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        current_sum = np.sum(best_h)
        if abs(current_sum - target_sum) > 1e-9:
            # adjust uniformly across non-boundary entries
            diff = target_sum - current_sum
            # find indices not at bound
            free = np.where((best_h > 1e-12) & (best_h < 1 - 1e-12))[0]
            if len(free) > 0:
                best_h[free] += diff / len(free)
                best_h = np.clip(best_h, 0.0, 1.0)
            else:
                # fallback uniform adjust
                best_h += diff / n
                best_h = np.clip(best_h, 0.0, 1.0)
        # final recompute c5 to be safe
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    else:
        # If nothing found, fall back to uniform half ones
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return best_h, float(best_c5), n

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Core utilities ----------
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def enforce_sum_and_bounds(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    # simple uniform correction
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # fine‑tune a few passes
    for _ in range(3):
        cur = np.sum(h)
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) == 0:
            h += diff / len(h)
        else:
            h[free] += diff / len(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible(n, rng):
    target = n / 2.0
    # start from uniform random numbers
    x = rng.random(n)
    x *= target / np.sum(x)
    # project to [0,1] while preserving sum
    for _ in range(5):
        over = x > 1.0
        if not np.any(over):
            break
        excess = np.sum(x[over] - 1.0)
        x[over] = 1.0
        under = ~over
        sum_under = np.sum(x[under])
        if sum_under > 0:
            x[under] -= excess * (x[under] / sum_under)
        x = np.clip(x, 0.0, 1.0)
    return enforce_sum_and_bounds(x, target)

# ---------- Constraint objects ----------
class SumConstraint:
    def __init__(self, n, target):
        self.n = n
        self.target = target
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ---------- Local continuous hill climbing ----------
def continuous_hill_climb(h, start_time, budget_s, rng):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    step = 0.2
    while time.time() - start_time < budget_s - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_inc = min(step, 1.0 - best[i])
        max_dec = min(step, best[j])
        delta = min(max_inc, max_dec)
        if delta <= 1e-12:
            continue
        delta *= rng.random()
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        val = compute_c5(cand)
        if val + 1e-12 < best_val:
            best, best_val = cand, val
        else:
            # tiny simulated‑annealing acceptance
            elapsed = time.time() - start_time
            T = (budget_s - elapsed) / budget_s * 1e-3
            if T > 0 and rng.random() < np.exp(-(val - best_val) / T):
                best, best_val = cand, val
        # occasional step reduction
        if rng.random() < 0.001:
            step = max(step * 0.9, 0.01)
    return best, best_val

# ---------- Binary hill climbing ----------
def binary_hill_climb(h_bin, start_c5, start_time, budget_s, rng):
    best = h_bin.copy()
    best_val = start_c5
    n = len(best)
    while time.time() - start_time < budget_s - 0.2:
        ones = np.flatnonzero(best == 1)
        zeros = np.flatnonzero(best == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        val = np.max(np.correlate(cand, 1 - cand, mode='full')) * (2.0 / n)
        if val + 1e-12 < best_val:
            best, best_val = cand, val
    return best, best_val

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- problem size -----
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        init_h = np.asarray(initial_h_values, dtype=float)
    else:
        n = 80
        init_h = None

    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- initial feasible point -----
    if init_h is not None:
        h0 = enforce_sum_and_bounds(init_h, target_sum)
    else:
        h0 = random_feasible(n, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----- build constraints for SLSQP -----
    sum_con = SumConstraint(n, target_sum)
    overlap_cons = [OverlapConstraint(k, n, dx) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # ----- Phase 1: many SLSQP restarts -----
    phase1_limit = budget_s * 0.5
    while time.time() - start_time < phase1_limit:
        h_start = random_feasible(n, rng)
        t_start = compute_c5(h_start)
        x0 = np.concatenate([h_start, [t_start]])

        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 800, 'ftol': 1e-9, 'disp': False}
            )
        except Exception:
            break

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = enforce_sum_and_bounds(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)

        if c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3805:
                break

    # ----- Phase 2: continuous hill climbing -----
    if time.time() - start_time < budget_s - 1.0:
        best_h, best_c5 = continuous_hill_climb(best_h, start_time, budget_s, rng)

    # ----- Phase 3: binary refinement -----
    if time.time() - start_time < budget_s - 0.5:
        # binarise by taking the largest n/2 entries
        binary_h = np.zeros(n, dtype=int)
        top_idx = np.argsort(-best_h)[: n // 2]
        binary_h[top_idx] = 1
        bin_c5 = np.max(np.correlate(binary_h, 1 - binary_h, mode='full')) * dx
        binary_h, bin_c5 = binary_hill_climb(binary_h, bin_c5, start_time, budget_s, rng)
        if bin_c5 + 1e-12 < best_c5:
            best_h = binary_h.astype(float)
            best_c5 = bin_c5

    # ----- final safety checks -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_and_bounds(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), n
