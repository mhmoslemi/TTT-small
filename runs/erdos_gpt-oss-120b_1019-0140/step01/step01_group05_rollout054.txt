The problem wants to find a step function h: [0,2] → [0,1] with integral 1 (over domain length 2) and minimize maximum overlap integral with its complement shifted: C5 = max_k ∫ h(x)*(1 - h(x+k)) dx over shift k.

We discretize h as an array of n points (samples) over [0,2] uniformly spaced with dx = 2/n. Integral constraint: sum(h) * dx = 1, i.e., sum(h) = n/2. So each vector must have sum exactly n/2. Bounds: each element must be in [0,1].

import numpy as np, time, math

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    # Compute sum
    s = v.sum()
    if s == 0.0:
        # Set all to equal value:
        v[:] = target / v.size
        v = np.clip(v, 0.0, 1.0)
        return v
    # scale
    factor = target / s
    v = v * factor
    v = np.clip(v, 0.0, 1.0)
    # adjust again if sum off
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v = v * factor2
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start, time_limit, rng, T0=0.1, max_eps=0.3):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)   # epsilon positive bound, increase hi, decrease hj
        down = min(hi, 1.0 - hj) # epsilon negative bound, decrease hi, increase hj
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if max_eps is not None and abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        # ensure bounds
        if not (0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0):
            continue
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import numpy as np
import time
import math

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        v = np.clip(v, 0.0, 1.0)
        return v
    factor = target / s
    v = v * factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v = v * factor2
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _simulated_annealing(h_start, time_limit, rng,
                         T0=0.1, T_end=1e-5, eps0=0.3, eps_end=0.01):
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        if time_limit > 0:
            frac = elapsed / time_limit
        else:
            frac = 0.0
        # exponential schedule
        T = T0 * ((T_end / T0) ** frac)
        max_eps = eps0 * ((eps_end / eps0) ** frac)
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        new_i = hi + eps
        new_j = hj - eps
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue
        # Apply move
        h_cur[i] = new_i
        h_cur[j] = new_j
        new_score = _c5_value(h_cur)
        delta = new_score - cur_score
        accept = delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            # revert
            h_cur[i] = hi
            h_cur[j] = hj
    return best_h, best_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        v = np.clip(v, 0.0, 1.0)
        return v
    factor = target / s
    v = v * factor
    v = np.clip(v, 0.0, 1.0)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v = v * factor2
        v = np.clip(v, 0.0, 1.0)
    return v


def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _simulated_annealing(
    h_start: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
    T0: float = 0.1,
    T_end: float = 1e-5,
    eps0: float = 0.3,
    eps_end: float = 0.01,
) -> tuple[np.ndarray, float]:
    """SA that preserves sum and box constraints while minimizing C5."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit if time_limit > 0 else 0.0
        T = T0 * ((T_end / T0) ** frac)
        max_eps = eps0 * ((eps_end / eps0) ** frac)

        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)
        down = min(hi, 1.0 - hj)
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue

        new_i = hi + eps
        new_j = hj - eps
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue

        # propose move
        h_cur[i] = new_i
        h_cur[j] = new_j
        new_score = _c5_value(h_cur)
        delta = new_score - cur_score
        accept = delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            # revert
            h_cur[i] = hi
            h_cur[j] = hj
    return best_h, best_score
