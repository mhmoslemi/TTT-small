def solve_binary_erdos(n):
    w = n // 2
    h = cp.Variable(n, boolean=True)
    m = cp.Variable(integer=True)  # maybe m integer too
    constraints = [cp.sum(h) == w]

    # product variables dictionary
    # for each pair (i,j) with i<j we need a y var.
    # We'll use a list of lists for y rows: dictionary mapping (i,j) to var.
    y_vars = {}
    for i in range(n):
        for j in range(i+1, n):
            y_vars[(i,j)] = cp.Variable(boolean=True)
            # constraints linking
            constraints.append(y_vars[(i,j)] <= h[i])
            constraints.append(y_vars[(i,j)] <= h[j])
            constraints.append(y_vars[(i,j)] >= h[i] + h[j] - 1)

    for k in range(1, n):
        overlap = cp.sum([ y_vars[(i,i+k)] for i in range(n-k) ])
        constraints.append(overlap >= m)

    # objective maximize m
    objective = cp.Maximize(m)

    prob = cp.Problem(objective, constraints)
    # solve with time limit

        candidate = current_h.copy()
        candidate[i] = 0
        candidate[j] = 1
        cand_c5 = compute_c5(candidate)
        # acceptance
        now = time.time()
        progress = (now - start) / time_left
        T = T0 * (1 - progress) + 1e-4 * progress
        if cand_c5 < current_c5 or rng.rand() < np.exp(-(cand_c5 - current_c5) / T):
            current_h = candidate
            current_c5 = cand_c5

                best_h = candidate.copy()

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def adjust_binary_to_target(arr, target):
    # arr is binary floats maybe with rounding errors.
    # Ensure exactly target ones.
    # Convert to int, compute diff.
    ints = np.rint(arr).astype(int)
    diff = int(target - ints.sum())
    if diff > 0:
        zeros_idx = np.where(ints == 0)[0]
        if len(zeros_idx) >= diff:
            ints[zeros_idx[:diff]] = 1
    elif diff < 0:
        ones_idx = np.where(ints == 1)[0]
        if len(ones_idx) >= -diff:
            ints[ones_idx[: -diff]] = 0
    return ints.astype(float)

def solve_binary_milp(n, target, time_limit):
    # Build problem
    h = cp.Variable(n, boolean=True)
    m = cp.Variable(integer=True)
    constraints = [cp.sum(h) == target]
    y_vars = {}
    for k in range(1, n):
        # create y for each pair
        for i in range(n - k):
            y = cp.Variable(boolean=True)
            y_vars[(i, k)] = y
            j = i + k
            constraints += [y <= h[i],
                            y <= h[j],
                            y >= h[i] + h[j] - 1]
        # overlap >= m
        overlap = cp.sum([y_vars[(i, k)] for i in range(n - k)])
        constraints.append(overlap >= m)
    # objective maximize m
    prob = cp.Problem(cp.Maximize(m), constraints)
    # solve with limit
    try:
        prob.solve(solver=cp.CBC, warm_start=True, max_seconds=time_limit, verbose=False)
    except TypeError:
        # fallback without time limit
        prob.solve(solver=cp.CBC, warm_start=True, verbose=False)
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
        h_val = np.array(h.value).flatten()
        h_val = adjust_binary_to_target(h_val, target)
        return h_val, int(np.rint(m.value))
    else:
        return None, None

    return (best_h, float(final_c5), n_points)

            g -= g.mean()

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Shift entries of h (in‑place) so that its sum equals target while staying in [0,1]."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    max_attempts = 10000
    attempts = 0
    while abs(diff) > 1e-12 and attempts < max_attempts:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        excess = target - h.sum()
        if abs(excess) > 1e-12:
            idx = rng.randint(h.size)
            h[idx] = np.clip(h[idx] + excess, 0.0, 1.0)
    return h
