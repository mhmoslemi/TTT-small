import time
import math
import numpy as np

def _thue_morse(n: int) -> np.ndarray:
    """Balanced binary Thue–Morse sequence of length n (n must be a power of two)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        # parity of popcount(i)
        seq[i] = (bin(i).count("1") & 1)
    return seq

def _corr_max_and_lag(h: np.ndarray):
    """Return (max_correlation_value, lag_of_max, full_correlation_array)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (n - 1), corr

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose an even power‑of‑two size – 256 gives a good trade‑off.
    n = 256
    if n % 2:
        n += 1

    # Initial construction: Thue–Morse balanced binary pattern.
    h = _thue_morse(n)

    # Initial evaluation.
    cur_max, cur_lag, _ = _corr_max_and_lag(h)
    dx = 2.0 / n
    cur_c5 = cur_max * dx
    best_h = h.copy()
    best_c5 = cur_c5

    # Simulated‑annealing parameters.
    init_temp = 0.005
    final_temp = 1e-6

    # Main optimisation loop.
    while True:
        now = time.time()
        if now - start_time >= budget_s:
            break

        # Linear temperature schedule.
        frac = (now - start_time) / budget_s
        temp = init_temp * (1.0 - frac) + final_temp * frac

        lag = cur_lag
        if lag >= 0:
            i_start = 0
            i_end = n - lag
            i_idx = np.arange(i_start, i_end)
            hi_vals = h[i_idx]
            hj_vals = h[i_idx + lag]
        else:
            i_start = -lag
            i_end = n
            i_idx = np.arange(i_start, i_end)
            hi_vals = h[i_idx]
            hj_vals = h[i_idx + lag]

        # Identify mismatched (1 vs 0) and complementary good (0 vs 1) positions.
        mis_mask = (hi_vals == 1) & (hj_vals == 0)
        good_mask = (hi_vals == 0) & (hj_vals == 1)
        mis_i = i_idx[mis_mask]
        good_i = i_idx[good_mask]

        if mis_i.size > 0 and good_i.size > 0:
            i_sel = int(rng.choice(mis_i))
            j_sel = int(rng.choice(good_i))
        else:
            # Fallback: random swap of a 1 and a 0.
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i_sel = int(rng.choice(ones))
            j_sel = int(rng.choice(zeros))

        # Store original values for possible revert.
        hi = h[i_sel]
        hj = h[j_sel]

        # Execute the swap (binary values guarantee hi!=hj).
        h[i_sel] = 0.0
        h[j_sel] = 1.0

        # Evaluate the new configuration.
        new_max, new_lag, _ = _corr_max_and_lag(h)
        new_c5 = new_max * dx
        delta = new_c5 - cur_c5

        # Metropolis acceptance.
        if delta < 0.0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            cur_c5 = new_c5
            cur_max = new_max
            cur_lag = new_lag
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert the swap.
            h[i_sel] = hi
            h[j_sel] = hj

    # Final sanity check and exact C5 computation for the best found vector.
    final_max, _, _ = _corr_max_and_lag(best_h)
    best_c5 = final_max * dx

    return best_h, best_c5, n
