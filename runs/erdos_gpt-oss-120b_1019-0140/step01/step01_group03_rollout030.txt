def optimize_adam(h0, max_iter, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, rng=None):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    for t in range(1, max_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        # Update moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score
```

import time
import numpy as np
import math

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    # ...
    ...

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def optimize_adam(h0, max_iter, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    b1t = 1.0
    b2t = 1.0
    for t in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        b1t *= beta1
        b2t *= beta2
        m_hat = m / (1.0 - b1t)
        v_hat = v / (1.0 - b2t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h - step, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    # Ensure half ones for powers of two
    if n % 2 == 0:
        # For powers of two sum is n/2 exactly
        return seq
    else:
        # fallback to binary balanced random
        return None

def resample_h(h, new_n):
    old_n = h.size
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def binary_hillclimb(h0, max_iter, rng):
    h = h0.copy()
    n = h.size
    best_score = compute_c5(h)
    best_h = h.copy()
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    for _ in range(max_iter):
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            # update index sets
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score

    sizes = [256, 180, 128, 64]
    # Helper to maybe add candidate after time check

    # Base candidates
    candidates = []

            h = project_onto_box_sum(h, n / 2.0)

    else:

            h = project_onto_box_sum(h, n / 2.0)
            candidates.append(h)

    # Thue-Morse for powers of two
    for n in (128, 256):

    for n in sizes:

            v = rng.random(n)
            candidates.append(project_onto_box_sum(v, n / 2.0))

        n = base_h.size

        if score_opt < best_score:

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Discrete C5 for vector h on [0,2] with uniform spacing."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient(h, shift):
    """Subgradient of the active max‑correlation term."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[: n - shift] = h[shift:]
        term2[shift:] = h[: n - shift]
    else:
        s = -shift
        term1[s:] = h[: n - s]
        term2[: n - s] = h[s:]
    return 1.0 - term1 - term2


def optimize_adam(h0, max_iter, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """Projected Adam on the subgradient of max‑correlation."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    b1t = 1.0
    b2t = 1.0
    for t in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        b1t *= beta1
        b2t *= beta2
        m_hat = m / (1 - b1t)
        v_hat = v / (1 - b2t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h - step, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def thue_morse(n):
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq


def resample_h(h, new_n):
    old_n = h.size
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def binary_hillclimb(h0, max_iter, rng):
    """Greedy swap hill‑climb on a balanced 0/1 vector."""
    h = h0.copy()
    n = h.size
    best_score = compute_c5(h)
    best_h = h.copy()
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    for _ in range(max_iter):
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_score = float('inf')
    sizes = [256, 180, 128, 64]

    def time_left(margin=0.5):
        return time.time() - start < budget_s - margin

    # ----- base candidates -------------------------------------------------
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        for n in sizes:

            h = project_onto_box_sum(h, n / 2.0)

    else:
        for n in sizes:
            h = np.full(n, 0.5, dtype=float)
            h = project_onto_box_sum(h, n / 2.0)
            candidates.append(h)

    # Thue–Morse patterns for powers of two
    for n in (128, 256):
        tm = thue_morse(n)
        if tm is not None:
            candidates.append(tm)

    # Random binary and continuous starts
    for n in sizes:

    # ----- Adam optimisation on each candidate ----------------------------
    iter_map = {256: 5000, 180: 4000, 128: 3000, 64: 2000}
    for base_h in candidates:

        n = base_h.size

        h_opt, sc_opt = optimize_adam(base_h, max_iter=iters, lr=0.05)

            best_score = sc_opt
            best_h = h_opt.copy()

    # ----- Turn best continuous solution into binary and hill‑climb -------
    if best_h is not None and time_left(2.0):
        n = best_h.size
        half = n // 2
        idx = np.argpartition(best_h, -half)[-half:]
        binary = np.zeros(n, dtype=float)

        hill_h, hill_sc = binary_hillclimb(binary, max_iter=20000, rng=rng)
        if hill_sc < best_score:
            best_score = hill_sc
            best_h = hill_h.copy()

    # ----- Random binary restarts with hill‑climb -------------------------
    while time_left(1.0):
        n = rng.choice(sizes)
        start_bin = random_binary_balanced(n, rng)
        hill_h, hill_sc = binary_hillclimb(start_bin, max_iter=5000, rng=rng)
        if hill_sc < best_score:
            best_score = hill_sc
            best_h = hill_h.copy()

    # ----- Final fine‑tuning ------------------------------------------------
    if best_h is not None and time_left(1.0):
        n = best_h.size
        iters = iter_map.get(n, 3000)
        best_h, best_score = optimize_adam(best_h, max_iter=iters, lr=0.03)

    if best_h is None:

        best_h = np.full(n, 0.5, dtype=float)

        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
