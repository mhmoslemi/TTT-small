def repair(v):
    v = np.clip(v, 0, 1)
    total = v.sum()
    target = v.size / 2
    v = v * (target / total)
    v = np.clip(v, 0, 1)
    # adjust sum again
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        # add diff equally distributed to all non-saturated entries
        mask = (v > 0) & (v < 1)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0, 1)
    return v

def hill_climb(h, time_limit):
    N = len(h)
    cur_score = evaluate_c5(h)
    best_h = h.copy()
    best_score = cur_score
    t_start = time.time()
    step = 0.2
    while time.time() - t_start < time_limit:
        # compute worst lag (k_star) for min product
        corr_hh = np.correlate(h, h, mode='full')
        prod = corr_hh[len(h)-1:]  # lags >=0
        k_star = np.argmin(prod)
        # compute gradient
        g = np.zeros_like(h)
        k = k_star
        if k == 0:
            g = 2.0 * h
        else:
            # h[i+k] adds to g[i]; h[i] adds to g[i+k]
            g[:N-k] += h[k:]
            g[k:] += h[:N-k]
        # zero mean
        g -= g.mean()
        # normalize gradient magnitude
        norm = np.linalg.norm(g)
        if norm == 0:
            # random perturb
            i, j = rng.integers(0, N, size=2)
            eps = rng.random() * 0.1
            h_new = h.copy()
            delta = min(eps, 1-h[i])
            h_new[i] += delta
            h_new[j] -= delta
            h_new = _repair_vector(h_new)
        else:
            # step size
            alpha = step / norm
            h_new = h + alpha * g
            h_new = _repair_vector(h_new)
        new_score = evaluate_c5(h_new)
        if new_score < cur_score or rng.random() < 0.001:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        # anneal step size
        step *= 0.999
        if step < 1e-4:
            step = 1e-4
    return best_h, best_score
```

def _repair_vector(v):
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0:
        v.fill(target / v.size)
        return v
    v = v * (target / s)
    v = np.clip(v, 0.0, 1.0)
    # final adjustment
    diff = target - v.sum()

        # distribute diff uniformly among interior points (not at bounds)
        mask = (v > 0.0) & (v < 1.0)

        else:
            # fallback distribute equally
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
    return v

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to len(v)/2 exactly."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v.fill(target / v.size)
        return v
    v = v * (target / s)
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
        else:
            v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """SA that preserves sum and box constraints, with a modest temperature."""
    h = h_start.copy()
    cur_score = _c5_value(h)
    best_h = h.copy()
    best_score = cur_score
    N = h.size
    start = time.time()
    T0 = 0.02                     # initial temperature
    max_eps = 0.1                 # max mass moved per step
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / time_limit)   # linear cooling
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = h[i], h[j]
        up = min(1.0 - hi, hj)       # max we can add to i (take from j)
        down = min(hi, 1.0 - hj)     # max we can remove from i (give to j)
        if up + down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        h_new = h.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
    return best_h, best_score

def _compute_gradient(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of sum_i h_i h_{i+k} w.r.t. h (for a fixed lag k)."""
    N = h.size
    if k == 0:
        return 2.0 * h
    g = np.zeros_like(h)
    g[:N - k] += h[k:]
    g[k:] += h[:N - k]
    return g

def _hill_climb(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Projected gradient ascent on the worst lag (min product) with occasional random swaps."""
    h = h_start.copy()
    cur_score = _c5_value(h)
    best_h = h.copy()
    best_score = cur_score
    N = h.size
    start = time.time()
    step = 0.5                 # controls magnitude of updates
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # worst lag (minimum product)
        hh = np.correlate(h, h, mode='full')
        prod = hh[N - 1:]               # lags >= 0
        k_star = int(np.argmin(prod))
        # gradient for that lag
        g = _compute_gradient(h, k_star)
        # project to zero-sum direction
        g -= g.mean()
        norm = np.linalg.norm(g)
        if norm < 1e-12:
            # random perturb when gradient vanishes
            i, j = rng.integers(0, N, size=2)
            while i == j:
                j = rng.integers(0, N)
            hi, hj = h[i], h[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up + down == 0.0:
                continue
            eps = rng.random() * (up + down) - down
            h_new = h.copy()
            h_new[i] = hi + eps
            h_new[j] = hj - eps
        else:
            alpha = step / (norm + 1e-12)
            h_new = h + alpha * g
            h_new = _repair_vector(h_new)
        new_score = _c5_value(h_new)
        if new_score < cur_score or rng.random() < 0.001:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        # decay step size slowly
        step *= 0.9995
        if step < 1e-4:
            step = 1e-4
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # initial vector
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 200
        h0 = rng.random(n0)
    h0 = _repair_vector(h0)
    start_total = time.time()
    best_h = h0.copy()
    best_score = _c5_value(best_h)

    # Phase 1 – multiple short SA runs at base resolution
    phase1_time = budget_s * 0.35
    n_restarts = 3
    time_per_restart = phase1_time / n_restarts
    for _ in range(n_restarts):
        if time.time() - start_total >= budget_s:
            break
        h_start = best_h + rng.normal(scale=0.05, size=best_h.shape)
        h_start = _repair_vector(h_start)
        h_candidate, s_candidate = _simulated_annealing(h_start, time_per_restart, rng)
        if s_candidate < best_score:
            best_score = s_candidate
            best_h = h_candidate.copy()

    # Phase 2 – gradient refinement at current resolution
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.1:
        t_gc = remaining * 0.5
        h_refined, s_refined = _hill_climb(best_h, t_gc, rng)
        if s_refined < best_score:
            best_score = s_refined
            best_h = h_refined.copy()
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed

    # Phase 3 – progressive resolution upgrades with SA + GC
    cur_h = best_h
    cur_n = cur_h.size
    while remaining > budget_s * 0.12 and cur_n < 800:
        n_target = min(800, cur_n * 2)
        if n_target == cur_n:
            break
        x_old = np.linspace(0.0, 2.0, cur_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_up = np.interp(x_new, x_old, cur_h)
        h_up = _repair_vector(h_up)

        # allocate time split between SA and GC
        t_sa = remaining * 0.4
        h_up_sa, s_up_sa = _simulated_annealing(h_up, t_sa, rng)
        if s_up_sa < best_score:
            best_score = s_up_sa
            best_h = h_up_sa.copy()
            cur_h = best_h
            cur_n = cur_h.size
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining <= budget_s * 0.12:
            break

        t_gc = remaining * 0.4
        h_up_gc, s_up_gc = _hill_climb(cur_h, t_gc, rng)
        if s_up_gc < best_score:
            best_score = s_up_gc
            best_h = h_up_gc.copy()
            cur_h = best_h
            cur_n = cur_h.size

        elapsed = time.time() - start_total
        remaining = budget_s - elapsed

    # Final repair and exact evaluation
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
