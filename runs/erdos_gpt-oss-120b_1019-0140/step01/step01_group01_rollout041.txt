Procedure:

```
current_h = best_h.copy()
current_c5 = best_c5

# schedule parameters
T0 = 0.01
T_end = 1e-6
step0 = 0.3
step_end = 1e-4

while time before budget:
    elapsed = time.time() - start
    progress = elapsed / max_time
    T = T0 * (1 - progress) + T_end * progress
    step = step0 * (1 - progress) + step_end * progress

    # propose move: select two indices i and j (distinct)
    i, j = rng.randint(n, size=2)
    if i == j:
        continue
    hi, hj = current_h[i], current_h[j]
    # compute feasible delta: we can move delta from i to j (i decrease, j increase) if hi - delta >=0 and hj + delta <= 1
    max_decrease = hi
    max_increase = 1.0 - hj
    # feasible delta range = [-max_decrease, max_increase] (negative delta means moving mass from j to i)
    # We intend to move by random amount within [-step, step] clipped to feasible.
    delta = (rng.rand() * 2 - 1) * step
    # Clip delta
    delta = max(-max_decrease, min(delta, max_increase))
    if abs(delta) < 1e-12:
        continue
    candidate = current_h.copy()
    candidate[i] = hi - delta
    candidate[j] = hj + delta
    # candidate sum unchanged
    # Clip just in case, then adjust sum maybe minimal
    np.clip(candidate, 0.0, 1.0, out=candidate)
    cand_c5 = compute_c5(candidate)

    # Acceptance
    if cand_c5 < current_c5:
        accept = True
    else:
        prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
        accept = rng.rand() < prob

    if accept:
        current_h = candidate
        current_c5 = cand_c5
        if cand_c5 < best_c5:
            best_h = candidate.copy()
            best_c5 = cand_c5

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def adjust_sum_to_target(h, target, rng):
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # adjust using random indices
    attempts = 0
    max_attempts = 1000
    n = h.size
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(n)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # Distribute leftover uniformly and clip
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def random_binary(n, weight, rng):
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=weight, replace=False)
    arr[idx] = 1.0
    return arr

def solve_binary_milp(n, rng):
    # Returns binary vector (0/1) as numpy array, and t value (float) if solved.
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = []
    constraints.append(cp.sum(h) == n/2)

    # auxiliary Z for products h_i * h_j
    Z = cp.Variable((n, n), boolean=True)

    # product constraints
    for i in range(n):
        for j in range(n):
            constraints.append(Z[i, j] <= h[i])
            constraints.append(Z[i, j] <= h[j])
            constraints.append(Z[i, j] >= h[i] + h[j] - 1)

    # overlap constraints for each lag
    for k in range(-(n-1), n):
        if k >= 0:
            i_vals = np.arange(n - k)
            j_vals = i_vals + k
        else:
            i_vals = np.arange(-k, n)
            j_vals = i_vals + k
        if i_vals.size == 0:
            continue
        # sum Z[i,j] over these pairs >= t
        # Build list for each pair
        pair_expr = [Z[i, j] for i, j in zip(i_vals, j_vals)]
        constraints.append(cp.sum(cp.hstack(pair_expr)) >= t)

    # objective
    obj = cp.Maximize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CBC, warm_start=True)
    except Exception:
        # If solve fails, return None
        return None, None

    if prob.status not in [cp.constants.OPTIMAL, cp.constants.OPTIMAL_INACCURATE]:
        return None, None
    h_val = np.rint(h.value).astype(float)  # round to nearest integer
    t_val = t.value
    return h_val, t_val

    milp_time_limit = max_time * 0.5
    if time.time() - start < milp_time_limit:
        # call solve but with safety: if it takes too long, we interrupt after time limit.
        # simple: run solve, then check elapsed; if exceed we break (but solve is blocking)
        # We'll just attempt, trusting it's fast.
        h_bin, t_val = solve_binary_milp(n, rng)
        if h_bin is not None:
            # Ensure exact sum
            h_bin = adjust_sum_to_target(h_bin, target_sum, rng)
            c5 = compute_c5(h_bin)
            best_h = h_bin
            best_c5 = c5

    # If MILP failed, start from random binary
    if best_h is None:
        best_h = random_binary(n, int(target_sum), rng)
        best_h = adjust_sum_to_target(best_h, target_sum, rng)
        best_c5 = compute_c5(best_h)

    # Local stochastic optimization (continuous swaps)
    current_h = best_h.copy()
    current_c5 = best_c5

    # annealing schedule parameters
    T0 = 1e-2
    T_end = 1e-6
    step0 = 0.3
    step_end = 1e-4

    # stagnation counter for occasional restart
    no_imp = 0
    restart_threshold = 5000

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        progress = (now - start) / max_time
        T = T0 * (1.0 - progress) + T_end * progress
        step = step0 * (1.0 - progress) + step_end * progress

        # propose swapping mass between two positions
        i, j = rng.randint(n, size=2)

        hi, hj = current_h[i], current_h[j]
        # feasible delta (move from i to j)

            current_c5 = cand_c5
            no_imp = 0
            if cand_c5 < best_c5:

                best_h = candidate.copy()
        else:
            no_imp += 1

        if no_imp >= restart_threshold:
            # restart from current best with random perturbation
            # make a random swap of 10 pairs to diversify
            trial = best_h.copy()
            for _ in range(10):
                i, j = rng.randint(n, size=2)
                hi, hj = trial[i], trial[j]
                max_decrease = hi
                max_increase = 1.0 - hj

                    delta = max_increase

            trial = adjust_sum_to_target(trial, target_sum, rng)

            current_c5 = compute_c5(current_h)
            no_imp = 0

    # final adjust
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n)

    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx via discrete correlation."""

            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        cand_c5 = compute_c5(candidate)

        # ----- acceptance -----
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0.0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            no_improve = 0
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
        else:
            no_improve += 1

        # ----- occasional restart to escape stagnation -----
        if no_improve >= restart_limit:
            # perturb the best solution slightly and continue from there
            current_h = best_h.copy()
            for _ in range(10):
                i, j = rng.randint(n, size=2)
                hi, hj = current_h[i], current_h[j]
                low = max(-hi, hj - 1.0)
                high = min(1.0 - hi, hj)
                if low >= high:
                    continue
                delta = (rng.rand() * 2.0 - 1.0) * step0
                delta = np.clip(delta, low, high)
                current_h[i] = hi + delta
                current_h[j] = hj - delta
            current_h = adjust_sum_to_target(current_h, target, rng)
            current_c5 = compute_c5(current_h)
            no_improve = 0

    # final sanity: enforce exact mass
    best_h = adjust_sum_to_target(best_h, target, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n)
