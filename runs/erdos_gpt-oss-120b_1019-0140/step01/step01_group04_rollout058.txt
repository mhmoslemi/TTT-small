            j = i_local + k

            i_local = int(np.argmax(contrib))

import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)

    resolutions = [n0]
    while resolutions[-1] * 2 <= max_n:
        resolutions.append(resolutions[-1] * 2)

    total_budget = max(0.0, budget_s - 2.0)  # reserve a couple seconds for final steps
    best_v = parent.copy()
    best_score = _compute_c5(best_v)

    for idx, n in enumerate(resolutions):
        elapsed = time.time() - start
        if elapsed >= budget_s - 1.0:
            break
        remaining = budget_s - elapsed - 1.0
        remaining_res = len(resolutions) - idx
        time_for_res = remaining / remaining_res

        # Prepare initial guess for this resolution.
        if len(best_v) != n:
            v0 = _upsample_vector(best_v, n)
        else:
            v0 = best_v.copy()

        # Local search focused on the worst overlap.
        v_opt, sc_opt = _targeted_local_search(v0, n, time_limit=time_for_res * 0.85)

        if sc_opt < best_score:
            best_score, best_v = sc_opt, v_opt.copy()

        # Early exit if we have already reached the target bound.
        if best_score <= 0.3808:
            break

    # Optional brief SLSQP refinement if time permits.
    remaining = budget_s - (time.time() - start) - 0.5
    if remaining > 5.0:
        n = len(best_v)
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
        try:
            res = minimize(
                lambda x: _compute_c5(x),
                best_v,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 200, "ftol": 1e-6, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n)
                sc = _compute_c5(v)
                if sc < best_score:
                    best_v, best_score = v, sc
        except Exception:
            pass

    # Final safety repair and evaluation.
    best_v = _repair_vector(best_v, len(best_v))
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(len(best_v))
