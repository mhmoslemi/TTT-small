def compute_c5(h):
    n = h.shape[0]
    # zero pad to length 2n
    N = 1 << (2*n-1).bit_length()  # next power of two >= 2n
    # Actually compute linear correlation via FFT.
    # Compute convolution of h and reversed(1-h)
    a = np.fft.rfft(h, N)
    b = np.fft.rfft((1 - h)[::-1], N) # reverse
    conv = np.fft.irfft(a * b, N)
    # conv length N, but linear conv result in first 2n-1 entries.
    corr = conv[:2*n-1]
    return np.max(corr) * (2.0 / n)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Linear correlation based C5 using FFT (fast)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n-1
    a = np.fft.rfft(h, N)
    b = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(a * b, N)
    corr = conv[: 2 * n - 1]
    return np.max(corr.real) * (2.0 / n)


def project_to_feasible(x, target, tol=1e-8, max_iter=10):
    """Clip to [0,1] and adjust interior entries to meet the sum constraint."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(x)
        diff = target - cur
        if abs(diff) <= tol:
            break
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    cur = np.sum(x)
    if abs(cur - target) > 1e-6:
        if cur < target:
            i = np.argmax(1.0 - x)
            x[i] = min(1.0, x[i] + (target - cur))
        else:
            i = np.argmin(x)
            x[i] = max(0.0, x[i] - (cur - target))
    return x


def binary_projection(v):
    """Return a {0,1} vector with exactly half the entries equal to 1."""
    n = v.shape[0]
    k = n // 2
    idx = np.argpartition(-v, k)[:k]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def resample_to_n(base_h, n):
    """Linear interpolation of base_h to length n."""
    base_h = np.asarray(base_h, dtype=float)
    m = base_h.shape[0]
    if n == m:
        return base_h.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, base_h)


def hill_climb_binary_sa(h_start, time_end, rng, max_iter=10_000_000, t_start=0.001):
    """Simulated‑annealing hill climbing on binary vectors (fixed Hamming weight)."""
    h = h_start.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n = h.shape[0]
    it = 0
    while time.time() < time_end and it < max_iter:
        # linear temperature schedule
        t = t_start * (1.0 - it / max_iter)
        # choose a random 1 and a random 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12))
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
        it += 1
    return best_h, best_c5


def local_optimize_slsqp(h0, time_end, rng):
    """Short SLSQP refinement (if time permits)."""
    n = h0.shape[0]
    target = n / 2.0

    def obj(x):
        return compute_c5(x)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bnds = [(0.0, 1.0)] * n
    try:
        # limit maxiter to keep time low
        res = opt.minimize(
            obj,
            h0,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        if res.success and time.time() < time_end:
            x = np.clip(res.x, 0.0, 1.0)
            x = project_to_feasible(x, target)
            return x, compute_c5(x)
    except Exception:
        pass
    # fallback
    return h0, compute_c5(h0)
