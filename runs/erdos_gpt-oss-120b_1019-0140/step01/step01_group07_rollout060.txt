    """Project vector v onto the box [0,1] and the affine hyperplane sum=v_target."""
    v = np.asarray(v, dtype=np.float64)
    lo, hi = -1.0, 2.0
    for _ in range(60):

                if child_score < best_score:
                    best_score = child_score
                    best_h = child.copy()
                    if best_score < 0.3808:
                        break
        # --------------------------------------------------------------
        # Occasional local refinement with SLSQP (gradient‑free)
        if gen % 25 == 0 and best_score > 0.3808:
            try:
                from scipy.optimize import minimize
                global _eq_target
                _eq_target = target_sum
                bounds = [(0.0, 1.0)] * n
                cons = {"type": "eq", "fun": _sum_eq_constraint}
                res = minimize(_compute_c5, best_h, method="SLSQP",
                               bounds=bounds, constraints=[cons],
                               options={"maxiter": 250, "ftol": 1e-9, "disp": False})
                if res.success:
                    cand = _project_to_capped_simplex(res.x, target_sum)
                    cand_score = _compute_c5(cand)
                    if cand_score < best_score:
                        best_score = cand_score
                        best_h = cand.copy()
            except Exception:
                pass

    # ------------------------------------------------------------------
    # Final evaluation with provided evaluator if available
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, int(n)
