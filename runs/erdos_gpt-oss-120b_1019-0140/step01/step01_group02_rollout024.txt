import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that sum(h) == target (within floating error)."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # distribute evenly as fallback
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _random_binary_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a binary (or half‑binary) vector with sum exactly n/2."""
    target = n / 2.0
    h = np.zeros(n, dtype=float)
    whole = int(math.floor(target))
    if whole > 0:
        idx = rng.choice(n, size=whole, replace=False)
        h[idx] = 1.0
    if target - whole > 1e-12:
        # need a 0.5 entry
        remaining = np.where(h == 0)[0]
        if remaining.size:
            i = rng.choice(remaining)
            h[i] = 0.5
    return h

def _tournament_select(pop, fitness, tsize, rng):
    """Return a population member with smallest fitness among tsize random choices."""
    idxs = rng.choice(len(pop), size=tsize, replace=False)
    best = idxs[0]
    best_fit = fitness[best]
    for i in idxs[1:]:
        f = fitness[i]
        if f < best_fit:
            best = i
            best_fit = f
    return pop[best]

def _crossover(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = p1.shape[0]
    point = rng.integers(1, n)  # split point in [1, n-1]
    child = np.empty(n, dtype=float)
    child[:point] = p1[:point]
    child[point:] = p2[point:]
    return child

def _mutate(h: np.ndarray, step: float, rng: np.random.Generator) -> np.ndarray:
    n = h.shape[0]
    if rng.random() < 0.5:
        # delta move between two indices (preserves sum)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            return h
        hi, hj = h[i], h[j]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        if delta_min >= delta_max:
            return h
        low = max(delta_min, -step)
        high = min(delta_max, step)
        if low >= high:
            return h
        delta = rng.random() * (high - low) + low
        h[i] = hi + delta
        h[j] = hj - delta
    else:
        # simple swap mutation (does not change sum)
        i, j = rng.integers(0, n, size=2)
        h[i], h[j] = h[j], h[i]
    return h

    start = time.time()

    # ---- problem size ----------------------------------------------------
