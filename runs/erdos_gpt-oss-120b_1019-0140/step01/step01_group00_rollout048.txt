import time, numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_weight(arr, target_ones, rng):
    # arr is binary (0/1) but may have different weight
    cur = int(arr.sum())
    diff = cur - target_ones
    if diff > 0:
        # too many ones, turn diff ones to zero
        one_idx = np.flatnonzero(arr)
        turn_off = rng.choice(one_idx, diff, replace=False)
        arr[turn_off] = 0
    elif diff < 0:
        zero_idx = np.flatnonzero(arr == 0)
        turn_on = rng.choice(zero_idx, -diff, replace=False)
        arr[turn_on] = 1
    return arr

def mutate_swap(arr, rng):
    # swap a random 1 with a random 0
    ones = np.flatnonzero(arr)
    zeros = np.flatnonzero(arr == 0)
    if len(ones)==0 or len(zeros)==0:
        return arr
    i = rng.choice(ones)
    j = rng.choice(zeros)
    arr[i], arr[j] = arr[j], arr[i]
    return arr

def tournament_select(pop, fitness, tournament_size, rng):
    idx = rng.integers(len(pop), size=tournament_size)
    best_idx = idx[np.argmin(fitness[idx])]
    return pop[best_idx]

def crossover(parent1, parent2, rng):
    child = np.where(rng.random(parent1.shape[0]) < 0.5, parent1, parent2).astype(np.int8)
    return child

    def random_binary():

        idx = rng.choice(n, target_ones, replace=False)
        vec[idx] = 1
        return vec

    # Initialize population
    pop_size = 60
    pop = []
    fitness = []

    # If initial exists, add binary version
    if 'initial_h_values' in globals():

        arr = init.copy()
        # For numeric values perhaps we treat >0.5 as 1 else 0

        top_idx = np.argsort(-arr)[:target_ones]

        fitness.append(compute_c5(binary_init))
    # Fill rest

        vec = random_binary()
        pop.append(vec)

    pop = list(pop) # ensure list

    best_idx = np.argmin(fitness)
    best = pop[best_idx].copy()
    best_f = float(fitness[best_idx])

    # GA parameters
    tournament_size = 5
    mutation_rate = 0.3
    # main loop
    while time.time() - start_time < budget_s * 0.9:

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap for binary vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_weight(arr, target_ones, rng):
    """Adjust binary array to have exactly target_ones ones."""
    cur = int(arr.sum())
    diff = cur - target_ones
    if diff > 0:
        ones = np.flatnonzero(arr)
        turn_off = rng.choice(ones, diff, replace=False)
        arr[turn_off] = 0
    elif diff < 0:
        zeros = np.flatnonzero(arr == 0)
        turn_on = rng.choice(zeros, -diff, replace=False)
        arr[turn_on] = 1
    return arr

def mutate_swap(arr, rng):
    """Swap a random 1 with a random 0 (preserves weight)."""
    ones = np.flatnonzero(arr)
    zeros = np.flatnonzero(arr == 0)
    if len(ones) == 0 or len(zeros) == 0:
        return arr
    i = rng.choice(ones)
    j = rng.choice(zeros)
    arr[i], arr[j] = arr[j], arr[i]
    return arr

def tournament_select(pop, fitness, size, rng):
    """Choose the best individual among a random subset."""
    idx = rng.integers(len(pop), size=size)
    best = idx[np.argmin(fitness[idx])]
    return pop[best]

def crossover(p1, p2, rng):
    """Uniform crossover then repair to exact weight."""
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2).astype(np.int8)
    return child

def greedy_improve(arr, target_ones, rng, deadline):
    """Exhaustive first‑improvement swap search until no improvement or time out."""
    best = arr.copy()
    best_f = compute_c5(best)
    n = len(arr)
    while time.time() < deadline:
        improved = False
        ones = np.flatnonzero(best)
        zeros = np.flatnonzero(best == 0)
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            for j in zeros:
                # perform swap
                best[i], best[j] = best[j], best[i]
                f = compute_c5(best)
                if f + 1e-12 < best_f:
                    best_f = f
                    improved = True
                    break
                # revert
                best[i], best[j] = best[j], best[i]
            if improved:
                break
        if not improved:
            break
    return best, best_f
