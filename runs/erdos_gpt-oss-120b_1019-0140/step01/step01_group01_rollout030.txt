import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap C5 for step function h (discrete approximation)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_box_sum(v: np.ndarray, target: float, max_iter: int = 60) -> np.ndarray:
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum x_i = target}.
    Finds lambda such that x = clip(v + lambda, 0, 1) has the required sum.
    """
    # bounds guaranteeing sum 0 at lo, sum n at hi
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def gradient_max_shift(h: np.ndarray) -> (np.ndarray, int):
    """
    Compute subgradient of the maximal overlap term.
    Returns gradient vector and the integer shift (lag) that attains the maximum.
    """
    n = h.shape[0]
    # correlation without dx (scale irrelevant for gradient direction)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    g = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(n - shift)
        g[i] += 1.0 - h[i + shift]
        g[i + shift] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        g[i] += 1.0 - h[i - s]
        g[i - s] -= h[i]
    return g, shift

def random_pairwise_move(h: np.ndarray, step: float, rng: np.random.RandomState) -> np.ndarray:
    """Perform a feasible exchange between two coordinates."""
    n = h.shape[0]
    i, j = rng.randint(n, size=2)
    if i == j:
        return None
    hi, hj = h[i], h[j]
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if low >= high:
        return None
    delta = rng.uniform(-step, step)
    delta = max(min(delta, high), low)
    new = h.copy()
    new[i] = hi + delta
    new[j] = hj - delta
    return new
