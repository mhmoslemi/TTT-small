import numpy as np
import time
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if np.isclose(cur, target, atol=1e-12):
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        # sort caps descending
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if not np.isclose(cur, target, atol=1e-8) and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

    b = 1.0 - a
    m = 2 * n - 1

    fb = np.fft.rfft(b, m)
    corr = np.fft.irfft(fa * np.conj(fb), m)

import numpy as np

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if np.isclose(cur, target, atol=1e-12):
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if not np.isclose(cur, target, atol=1e-8) and cur > 0:

        v = np.clip(v, 0.0, 1.0)
