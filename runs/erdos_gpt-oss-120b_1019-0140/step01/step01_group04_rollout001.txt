def compute_c5(v):
    n = v.shape[0]
    w = 1.0 - v
    # reverse w
    wr = w[::-1]
    nfft = 1 << ((2*n-1).bit_length())
    # compute convolution via FFT
    fv = np.fft.rfft(v, n=nfft)
    fw = np.fft.rfft(wr, n=nfft)
    conv = np.fft.irfft(fv * fw, n=nfft)
    # extract only first 2*n-1 elements (full correlation)
    corr = conv[:2*n-1]
    # Each entry corresponds to sum_{i} v[i]*w[i+k], where k from -(n-1) to n-1.
    # scaling factor dx = 2.0/n
    dx = 2.0 / n
    return np.max(corr) * dx

def _c5_fft(v):
    n = v.shape[0]
    w = 1.0 - v
    w_rev = w[::-1]
    nfft = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(v, n=nfft)
    fw = np.fft.rfft(w_rev, n=nfft)
    conv = np.fft.irfft(fv * fw, n=nfft)
    corr = conv[:2*n-1]
    dx = 2.0 / n
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    n_src = len(v)
    src_x = np.linspace(0, 2, n_src, endpoint=False)
    tgt_x = np.linspace(0, 2, n_target, endpoint=False)
    # using interpolation: for each target x, linearly interpolate between source values.
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    # Ensure sum constraint: after interpolation, sum may ≈ n_target/2 but adjust via repair
    return _repair_vector(new_v, n_target)

import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project vector onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n * 0.5
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_fft(v):
    """Compute C5 using FFT‑based full correlation."""
    n = v.shape[0]
    w = 1.0 - v
    w_rev = w[::-1]
    nfft = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(v, n=nfft)
    fw = np.fft.rfft(w_rev, n=nfft)
    conv = np.fft.irfft(fv * fw, n=nfft)
    corr = conv[: 2 * n - 1]
    dx = 2.0 / n
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _anneal(v_init, time_limit, start_temp=0.08, end_temp=1e-6):
    """Simulated annealing with pairwise exchanges."""
    start = time.time()
    cur = v_init.copy()
    cur_score = _c5_fft(cur)
    best = cur.copy()
    best_score = cur_score
    n = len(cur)
    step_max = 0.2
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        T = start_temp * (1.0 - frac) + end_temp * frac
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        vi, vj = cur[i], cur[j]
        max_up_i = 1.0 - vi
        max_down_i = vi
        max_up_j = 1.0 - vj
        max_down_j = vj
        low = -min(max_down_i, max_up_j)
        high = min(max_up_i, max_down_j)
        if low >= high:
            continue
        # restrict magnitude for stability
        low = max(low, -step_max)
        high = min(high, step_max)
        if low >= high:
            continue
        delta = np.random.uniform(low, high)
        cand = cur.copy()
        cand[i] = vi + delta
        cand[j] = vj - delta
        cand_score = _c5_fft(cand)
        diff = cand_score - cur_score
        if diff < 0.0:
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
        else:
            if T > 0.0 and np.exp(-diff / (T + 1e-12)) > np.random.rand():
                cur, cur_score = cand, cand_score
    return best, best_score

def _local_search(v_init, time_limit):
    """Simple hill‑climbing: exchange small amounts between two coordinates."""
    start = time.time()
    best = v_init.copy()
    best_score = _c5_fft(best)
    cur = best.copy()
    cur_score = best_score
    n = len(v_init)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.randint(0, n, size=2)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _c5_fft(cand)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
    return best, best_score

    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    np.random.seed(seed)
    start = time.time()

    # Acquire parent construction if present.
