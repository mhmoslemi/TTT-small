def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_weight(h, target):

    current_sum = h.sum()
    diff = target - current_sum

    if diff == 0: return h

    # diff positive: need to turn some zeros to ones; diff negative: turn some ones to zeros.
    zeros = np.where(h == 0)[0] if diff > 0 else np.where(h == 1)[0]
    # randomly pick diff entries to flip
    if zeros.size > 0:
        idx = rng.choice(zeros, size=int(abs(diff)), replace=False)
        h[idx] = 1 if diff > 0 else 0
    else:
        # fallback adjust by fractional
        pass
    return h

def _repair_sum(h, target):
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # adjust one entry
    if diff > 0:

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _hk_shifted(h: np.ndarray, k: int) -> np.ndarray:
    n = h.shape[0]
    if k >= 0:
        hk = np.empty_like(h)
        hk[:n - k] = h[k:]
        hk[n - k:] = 0.0
    else:
        kp = -k
        hk = np.empty_like(h)
        hk[kp:] = h[:n - kp]
        hk[:kp] = 0.0
    return hk
