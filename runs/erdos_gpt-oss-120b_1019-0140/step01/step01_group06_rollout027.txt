def simulated_annealing(h0, target_sum, max_time, start_temp=0.01, end_temp=1e-5):
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = h0.copy()
    cur_val = best_val
    start = time.time()
    iteration = 0
    while time.time() - start < max_time:
        # update temp
        t = start_temp * ((max_time - (time.time() - start)) / max_time) # linear decreasing
        # propose new
        noise_scale = 0.05  # step size scaling with temp?
        noise = np.random.normal(scale=noise_scale, size=cur_h.shape)
        cand = cur_h + noise
        cand = project_to_feasible(cand, target_sum)

        # acceptance
        if cand_val < cur_val or np.random.rand() < np.exp((cur_val - cand_val) / max(t, 1e-12)):

            cur_val = cand_val

        iteration += 1

    # allowable delta range satisfying bounds:
    max_pos = min(step, 1 - h[i])

def propose_swap(h, step):
    n = len(h)
    i, j = rng.choice(n, 2, replace=False)
    pos_max = min(step, 1 - h[i], h[j]) # delta>0
    neg_max = min(step, h[i], 1 - h[j]) # allowable magnitude for delta<0
    if pos_max <= 0 and neg_max <= 0:
        return h.copy()
    delta = rng.uniform(-neg_max, pos_max)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # clip just in case small numeric errors
    np.clip(new_h, 0, 1, out=new_h)
    return new_h
```

def propose_swap(h, rng, step):
    n = h.shape[0]
    for _ in range(10):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # compute max allowed delta positive and negative
        # positive delta: increase h[i], decrease h[j] => 
        # delta <= step, delta <= 1 - h[i], delta <= h[j]
        pos_max = step
        if h[i] >= 1.0 or h[j] <= 0.0:
            # may not increase i or decrease j
            pass
        pos_max = min(pos_max, 1.0 - h[i], h[j])
        # negative delta: decrease h[i], increase h[j]
        neg_max = step
        # delta negative bound magnitude:
        neg_max = min(neg_max, h[i], 1.0 - h[j])
        # Determine feasible delta interval [-neg_max, pos_max]
        if pos_max <= 0 and neg_max <= 0:
            continue
        delta = rng.uniform(-neg_max, pos_max)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # Clip maybe rarely needed
        np.clip(new_h, 0.0, 1.0, out=new_h)
        return new_h
    # fallback: return h unchanged
    return h.copy()

def optimize_sa(h0, target_sum, time_budget):

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    """Iteratively project onto box [0,1]^n and the hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h)  # most room to increase
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def propose_swap(h, rng, step):
    """Generate a new feasible vector by moving mass between two entries."""
    n = h.shape[0]
    for _ in range(15):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # delta >0 : increase h[i], decrease h[j]
        pos_max = min(step, 1.0 - h[i], h[j])
        # delta <0 : decrease h[i], increase h[j]
        neg_max = min(step, h[i], 1.0 - h[j])
        if pos_max <= 0 and neg_max <= 0:
            continue
        delta = rng.uniform(-neg_max, pos_max)
        if abs(delta) < 1e-12:
            continue
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        np.clip(new_h, 0.0, 1.0, out=new_h)  # safety
        return new_h
    # fallback – random perturbation with projection
    cand = h + rng.normal(scale=step, size=n)
    return project_to_feasible(cand, n / 2.0)

def sum_constraint(x, nd):
    """Equality constraint: sum(x) = nd/2."""
    return np.sum(x) - nd / 2.0

def local_refine(h, n, max_iter=600):
    """Run a short SLSQP refinement from h."""
    bnds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_constraint, "args": (n,)}
    try:
        res = opt.minimize(
            fun=compute_c5,
            x0=h,
            method="SLSQP",
            bounds=bnds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return h
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt

def optimize_sa(h0, target_sum, time_budget, rng):
    """Simple simulated annealing on the vector h0."""
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    # annealing schedule parameters
    T_start = 0.02
    T_end = 1e-6
    step_start = 0.2
    step_end = 1e-4
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T_start * (1 - frac) + T_end * frac
        step = step_start * (1 - frac) + step_end * frac
        # occasional big random restart
        if rng.random() < 0.003:
            cand_h = rng.random(n)
            cand_h = project_to_feasible(cand_h, target_sum)
        else:
            cand_h = propose_swap(cur_h, rng, step)
        cand_val = compute_c5(cand_h)
        # Metropolis acceptance
        if cand_val < cur_val or rng.random() < np.exp((cur_val - cand_val) / max(T, 1e-12)):
            cur_h = cand_h
            cur_val = cand_val
            if cand_val < best_val:
                best_h = cand_h.copy()
                best_val = cand_val
    return best_h, best_val

    parent = globals().get("initial_h_values")
    # Candidate sample sizes (small to medium)

    best_val_global = float("inf")
    best_n_global = None
    base_h = parent  # for upsampling between stages

        if time.time() - start_time > budget_s * 0.95:

        else:

            init_h = project_to_feasible(init_h, n / 2.0)
        # Simulated annealing (majority of allocated time)
        sa_time = alloc * 0.85
        best_h_n, best_val_n = optimize_sa(
            init_h, n / 2.0, sa_time, rng

        remaining = alloc - (time.time() - phase_start)

            best_h_global = best_h_n.copy()

        base_h = best_h_n

    if best_h_global is not None and (budget_s - (time.time() - start_time)) > 1.0:

        best_val_global = compute_c5(best_h_global)

    best_h_global = project_to_feasible(best_h_global, best_n_global / 2.0)
    best_val_global = compute_c5(best_h_global)
