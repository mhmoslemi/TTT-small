import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=50):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = (h > 0.0) & (h < 1.0)
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)
    # correct tiny residual
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # distribute remaining diff to a free index
        # find index with biggest room
        room_up = 1.0 - h
        room_down = h
        # measure ability to add or subtract
        # choose whichever direction needed
        if cur < target_sum:
            # need to increase sum
            i = np.argmax(room_up)
            h[i] = min(1.0, h[i] + (target_sum - cur))
        else:
            i = np.argmax(room_down)
            h[i] = max(0.0, h[i] - (cur - target_sum))
    return h

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def local_optimize(h0, rng, start_time, time_limit, n, max_iters=3000):
    # ensure feasibility
    h = project_to_feasible(h0, n / 2.0)
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 0.2
    min_step = 5e-5
    for it in range(max_iters):
        if time.time() - start_time >= time_limit:
            break
        # decide move type
        if rng.random() < 0.8:
            # pairwise swap
            i, j = rng.integers(n, size=2)
            if i == j:
                continue

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=50):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = (h > 0.0) & (h < 1.0)
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        need = target_sum - cur
        if need > 0:
            i = np.argmax(1.0 - h)
            h[i] = min(1.0, h[i] + need)
        else:
            i = np.argmax(h)
            h[i] = max(0.0, h[i] + need)
    return h

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def binary_initial_from_continuous(base_h, n):
    cont = resample_h(base_h, n) if base_h.shape[0] != n else base_h.copy()
    idx = np.argsort(-cont)
    half = n // 2
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx[:half]] = 1.0
    return bin_h

def binary_local_optimize(h0, rng, start_time, time_limit, n, max_iters=30000):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    for it in range(max_iters):
        if time.time() >= time_limit:
            break
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # swap
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i); zeros.add(i)
        zeros.remove(j); ones.add(j)
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            if best_val <= 0.3808:
                break
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones.add(i); zeros.remove(i)
            zeros.add(j); ones.remove(j)
    return best_h, best_val

def local_optimize(h0, rng, start_time, time_limit, n, max_iters=2000):
    h = project_to_feasible(h0, n / 2.0)
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 0.2
    min_step = 5e-5
    for it in range(max_iters):
        if time.time() >= time_limit:
            break
        if rng.random() < 0.8:
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            max_inc = min(step, 1.0 - h[i], h[j])
            max_dec = min(step, h[i], 1.0 - h[j])
            if max_inc <= 0 and max_dec <= 0:
                continue
            lo, hi = -max_dec, max_inc
            if lo >= hi:
                continue
            delta = rng.uniform(lo, hi)
            if delta == 0.0:
                continue
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            v = rng.normal(size=n)
            v -= np.mean(v)
            norm = np.linalg.norm(v)
            if norm == 0.0:
                continue
            v = v * (step / norm)
            new_h = h + v
            new_h = project_to_feasible(new_h, n / 2.0)
        val = compute_c5(new_h)
        if val < best_val - 1e-12:
            best_val = val
            best_h = new_h.copy()
            h = new_h
            if best_val <= 0.3808:
                break
        if (it + 1) % 200 == 0:
            step = max(step * 0.9, min_step)
    return best_h, best_val
