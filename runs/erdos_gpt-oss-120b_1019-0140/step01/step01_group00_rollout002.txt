    rng = np.random.default_rng(seed)
    # Get n and initial vector.
    if 'initial_h_values' in globals():
        init_cont = np.array(initial_h_values, dtype=float)
        n = len(init_cont)
        # Convert to binary
        half = n // 2
        idx = np.argpartition(-init_cont, half-1)[:half]
        cur_best = np.zeros(n, dtype=int)
        cur_best[idx] = 1
    else:
        n = 80
        half = n // 2
        cur_best = np.zeros(n, dtype=int)
        idx = rng.choice(n, half, replace=False)
        cur_best[idx] = 1
    best_vec = cur_best.copy()
    best_c5 = compute_c5(best_vec)
    # SA parameters
    T0 = 0.02
    # Main loop
    end_time = start_time + budget_s - 0.2
    cur = cur_best.copy()
    cur_c5 = best_c5
    last_improve_iter = 0
    iter_count = 0
    # We may keep a counter for stagnation.
    while time.time() < end_time:
        # Temperature based on remaining fraction
        elapsed = time.time() - start_time
        frac = max(0.0, (budget_s - elapsed) / budget_s)
        T = T0 * frac
        # propose swap
        ones = np.flatnonzero(cur == 1)
        zeros = np.flatnonzero(cur == 0)
        # if any missing, break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand)
        # acceptance
        if cand_c5 < cur_c5:
            accept = True
        else:
            # accept with probability exp(-(delta)/T)
            # if T <= 0, never accept uphill
            if T > 0:
                accept = rng.random() < np.exp(-(cand_c5 - cur_c5) / T)
            else:
                accept = False

import time
import numpy as np

def compute_c5(vec):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for binary/float vector."""
    n = vec.shape[0]
    dx = 2.0 / n
    v = vec.astype(float)
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def local_optimize_bin(start_vec, deadline):
    """Exhaustive pairwise swap improvement until local optimum or time out."""
    n = start_vec.shape[0]
    best_vec = start_vec.copy()
    best_val = compute_c5(best_vec)
    # continue while improvement possible and time permits
    while time.time() < deadline - 0.01:
        improved = False
        ones = np.flatnonzero(best_vec == 1)
        zeros = np.flatnonzero(best_vec == 0)
        for i in ones:
            for j in zeros:
                cand = best_vec.copy()
                cand[i] = 0
                cand[j] = 1
                cand_val = compute_c5(cand)
                if cand_val + 1e-12 < best_val:
                    best_vec = cand
                    best_val = cand_val
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_vec, best_val
