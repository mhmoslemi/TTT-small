def _simulated_annealing(v, time_limit, start_temp=0.01, end_temp=1e-6):
    start = time.time()
    n = len(v)
    best = v.copy()
    best_score = _compute_c5(best)
    cur = v.copy()
    cur_score = best_score
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        t = start_temp * (end_temp / start_temp) ** (elapsed / time_limit)
        # propose: pick i,j random distinct
        i, j = np.random.choice(n, 2, replace=False)
        # propose delta in [-step, step]
        step = t # scale step with temperature?
        delta = (np.random.rand() - 0.5) * 2 * step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(t, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur
                best_score = cur_score
    return best, best_score

def _repair_vector(v, n, max_iter=100):
    # Ensure on [0,1]
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    # Compute current sum
    s = v.sum()
    # If sum close, done
    if abs(s - target) < 1e-12:
        return v
    # Compute residual diff
    diff = s - target
    # Determine free indices (those not at bounds that would move opposite direction)
    # For diff>0 need to reduce sum, can reduce values that >0.
    # For diff<0 need to increase sum, can increase values that <1.
    # We'll iterate adjusting equally among free indices.
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        diff = s - target
    # final correction if needed:
    if abs(v.sum() - target) > 1e-9:
        # scale
        v = np.clip(v * (target / v.sum()), 0.0, 1.0)
        # final small fix
        diff = target - v.sum()
        if abs(diff) > 1e-9:
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v
```

    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)
    # constraint: sum - target = 0
    target = n / 2.0

    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _compute_c5(x)

    # primary run
    try:
        res = minimize(obj, init_v, method="SLSQP", bounds=bounds,
                       constraints=cons, options={"maxiter": 2000, "ftol": 1e-12, "disp": False})
        if res.success:
            v = _repair_vector(res.x, n)
            sc = _compute_c5(v)
            if sc < best_score:
                best_score, best_v = sc, v
    except Exception:
        pass

    # random restarts
    while time.time() - start < time_limit:
        rnd = _random_feasible(n)
        try:
            res = minimize(obj, rnd, method="SLSQP", bounds=bounds,

        except Exception:
            pass
        # break if near time limit

def _simulated_annealing(v, time_limit, start_temp=0.02, end_temp=1e-6):
    start = time.time()
    n = len(v)
    cur = v.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        t = start_temp * (end_temp / start_temp) ** (elapsed / time_limit)
        # propose
        i, j = np.random.choice(n, 2, replace=False)
        # delta magnitude scaling with temperature
        delta = (np.random.rand() - 0.5) * 2.0 * t
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        # acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(t, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur
                best_score = cur_score
    return best, best_score

def _gradient_descent_refine(v, time_limit):
    start = time.time()
    n = len(v)
    best = v.copy()
    best_score = _compute_c5(best)
    step = 0.1
    while time.time() - start < time_limit:
        # compute gradient for worst shift
        corr = np.correlate(best, 1.0 - best, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # gradient
        g = np.zeros_like(best)
        if shift >= 0:
            g[: n - shift] = 1.0 - best[shift:]
        else:
            g[-shift:] = 1.0 - best[: n + shift]
        norm = np.linalg.norm(g)
        if norm == 0:
            break
        cand = best - (step * g / norm)
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
            step = min(step * 1.2, 1.0)
        else:
            step *= 0.5
        if step < 1e-6:
            break
    return best, best_score
```

import time
import numpy as np
from scipy.optimize import minimize

def _repair_vector(v, n, max_iter=20):
    """Project v onto the box [0,1]^n while enforcing sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        free = mask.sum()
        if free == 0:
            break
        adjust = diff / free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final safety fix
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return max_{k} ∫ h(x)(1-h(x+k))dx for discrete vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v onto a finer grid of size n_target, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random vector in [0,1]^n with the required sum."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _optimize_at_resolution(n, init_v, time_limit):
    """Run SLSQP from init_v (plus random restarts) within the allotted time."""
    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _compute_c5(x)

    # primary run
    try:
        res = minimize(
            obj,
            init_v,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 2000, "ftol": 1e-12, "disp": False},
        )
        if res.success:
            v = _repair_vector(res.x, n)
            sc = _compute_c5(v)
            if sc < best_score:
                best_score, best_v = sc, v
    except Exception:
        pass

    # random restarts
    while time.time() - start < time_limit:
        rnd = _random_feasible(n)
        try:
            res = minimize(
                obj,
                rnd,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 1500, "ftol": 1e-12, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n)
                sc = _compute_c5(v)
                if sc < best_score:
                    best_score, best_v = sc, v
        except Exception:
            pass
        if time.time() - start > time_limit * 0.9:
            break
    return best_v, best_score

def _simulated_annealing(v, time_limit, start_temp=0.02, end_temp=1e-6):
    """Simple SA that swaps mass between two coordinates."""
    start = time.time()
    n = len(v)
    cur = v.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        t = start_temp * (end_temp / start_temp) ** (elapsed / time_limit)
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * t
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(t, 1e-12)):
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best, best_score = cur, cur_score
    return best, best_score

def _gradient_descent_refine(v, time_limit):
    """Subgradient descent focusing on the worst shift."""
    start = time.time()
    n = len(v)
    best = v.copy()
    best_score = _compute_c5(best)
    step = 0.1
    while time.time() - start < time_limit:
        corr = np.correlate(best, 1.0 - best, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        g = np.zeros_like(best)
        if shift >= 0:
            g[: n - shift] = 1.0 - best[shift:]
        else:
            g[-shift:] = 1.0 - best[: n + shift]
        norm = np.linalg.norm(g)
        if norm == 0:
            break
        cand = best - (step * g / norm)
        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
            step = min(step * 1.2, 1.0)
        else:
            step *= 0.5
        if step < 1e-7:
            break
    return best, best_score
