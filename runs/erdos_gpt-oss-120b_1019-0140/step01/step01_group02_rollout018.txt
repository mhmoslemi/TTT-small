import time, math, numpy as np

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    eps = 1e-12
    while abs(diff) > eps:
        if diff > 0:
            # need increase
            idx = np.where(h < 1.0 - eps)[0]
            if idx.size == 0:
                break
            i = idx[0]
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            idx = np.where(h > eps)[0]
            if idx.size == 0:
                break
            i = idx[0]
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h
```

def select(pop, fitness, rng, tournament_size):
    idx = rng.choice(len(pop), size=tournament_size, replace=False)

    best_f = fitness[best]

def local_search(h, rng, time_limit, start):
    n = h.shape[0]
    target_sum = n/2.0
    cur = h.copy()
    cur_c5 = _compute_c5(cur)
    max_step = 0.05
    min_step = 0.001
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        frac = (now - start) / time_limit
        step = max_step * (1-frac) + min_step * frac
        # pick two indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur[i], cur[j]
        # delta bounds
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        low = max(delta_min, -step)
        high = min(delta_max, step)
        if low >= high:
            continue
        delta = rng.random() * (high - low) + low
        cur[i] = hi + delta
        cur[j] = hj - delta
        new_c5 = _compute_c5(cur)
        if new_c5 < cur_c5 or rng.random() < 0.1:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                # update outer best
                best_c5 = new_c5
                best_h = cur.copy()
        else:
            cur[i] = hi
            cur[j] = hj

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Compute the discretized C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float, tol: float = 1e-12, max_iter: int = 1000) -> np.ndarray:
    """
    Adjust entries of h (in‑place) so that sum(h) == target while keeping
    0 ≤ h[i] ≤ 1. Uses a simple greedy redistribution.
    """
    diff = target - h.sum()
    it = 0
    while abs(diff) > tol and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - tol)[0]
            if idx.size == 0:
                break
            i = idx[0]
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            idx = np.where(h > tol)[0]
            if idx.size == 0:
                break
            i = idx[0]
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
        it += 1
    np.clip(h, 0.0, 1.0, out=h)
    return h


def _crossover(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """One‑point crossover producing a new offspring."""
    n = p1.shape[0]
    pt = rng.integers(1, n)  # avoid trivial cuts
    child = np.empty_like(p1)
    child[:pt] = p1[:pt]
    child[pt:] = p2[pt:]
    return child


def _mutate(ind: np.ndarray, rng: np.random.Generator, step: float, p_mut: float = 0.2) -> np.ndarray:
    """Uniform mutation on a subset of genes."""
    mask = rng.random(ind.shape[0]) < p_mut
    n_mut = mask.sum()
    if n_mut:
        deltas = (rng.random(n_mut) * 2.0 - 1.0) * step
        ind[mask] += deltas
        np.clip(ind, 0.0, 1.0, out=ind)
    return ind


def _tournament(pop, fitness, rng, size=4):
    """Select one individual via tournament."""
    idx = rng.choice(len(pop), size=size, replace=False)
    best = idx[0]
    best_f = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_f:
            best = i
            best_f = fitness[i]
    return pop[best]


def _local_search(h: np.ndarray, rng: np.random.Generator, time_limit: float) -> tuple:
    """
    Simulated‑annealing style refinement on a single solution.
    Returns (best_h, best_c5) found within time_limit seconds.
    """
    n = h.shape[0]
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _compute_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    start = time.time()
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        frac = elapsed / time_limit
        # schedules
        temp = 0.01 * (1.0 - frac) + 1e-6 * frac
        max_step = 0.1 * (1.0 - frac) + 0.001 * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur[i], cur[j]

        # feasible delta interval respecting box constraints
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        low = max(delta_min, -max_step)
        high = min(delta_max, max_step)
        if low >= high:
            continue
        delta = rng.random() * (high - low) + low

        cur[i] = hi + delta
        cur[j] = hj - delta
        _repair_sum(cur, target)

        new_c5 = _compute_c5(cur)
        d_obj = new_c5 - cur_c5
        accept = False
        if d_obj < 0:
            accept = True
        else:
            if rng.random() < math.exp(-d_obj / temp):
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur.copy()
        else:
            cur[i] = hi
            cur[j] = hj
            # sum unchanged; no need to repair here.
    return best_h, best_c5
