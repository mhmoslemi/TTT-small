import numpy as np
import time
import math

def _repair_vector(v):
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_triangle(n):
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    return _repair_vector(h)

def _interpolate(h_old, n_new):
    n_old = h_old.size
    if n_old == n_new:
        return h_old.copy()
    xs_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(xs_new, xs_old, h_old)
    return h_new

    best_h = h_cur.copy()

        if elapsed >= time_limit:
            break
        frac = max(0.0, 1.0 - elapsed / time_limit)

        hi = h_cur[i]

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust sum to size/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (values in [0,1], sum = n/2)."""
    v = rng.random(n)
    return _repair_vector(v)

def _interpolate(h_old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of h_old to a new resolution."""
    n_old = h_old.size
    if n_new == n_old:
        return h_old.copy()
    xs_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xs_new, xs_old, h_old)

def _sa_optimize(h_start: np.ndarray, time_limit: float,
                 rng: np.random.Generator, max_eps: float = 0.2) -> tuple:
    """Simple simulated annealing on the C5 objective."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.05          # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * (frac ** 0.5)          # sqrt cooling
        # pick two distinct indices
        i = rng.integers(0, N)
        j = rng.integers(0, N - 1)
        if j >= i:
            j += 1
        hi, hj = h_cur[i], h_cur[j]
        up = min(1.0 - hi, hj)          # max we can add to hi / remove from hj
        down = min(hi, 1.0 - hj)        # max we can subtract from hi / add to hj
        if up == 0.0 and down == 0.0:
            continue
        eps = rng.random() * (up + down) - down   # uniform in [-down, up]
        # temper move size
        eps = eps * (0.5 + 0.5 * T / T0)
        if abs(eps) > max_eps:
            eps = math.copysign(max_eps, eps)
        if eps == 0.0:
            continue
        h_new = h_cur.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # --- initial construction -------------------------------------------------
    try:

    best_h = h0
    best_score = _c5_value(best_h)

    # --- phase 1: annealing on the original resolution -----------------------
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5.0:
        phase = time_left * 0.4
        h_opt, s_opt = _sa_optimize(best_h, phase, rng)
        if s_opt < best_score:
            best_h, best_score = h_opt, s_opt

    # --- phase 2: multiple random restarts ----------------------------------
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 5.0:
            break
        phase = min(30.0, time_left * 0.3)
        n = best_h.size
        rnd_start = _make_random(n, rng)
        h_new, s_new = _sa_optimize(rnd_start, phase, rng)
        if s_new < best_score:
            best_h, best_score = h_new, s_new

    # --- phase 3: increase resolution if time permits ------------------------
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 10.0 or best_h.size >= 1000:
            break
        n_curr = best_h.size
        n_new = min(1000, n_curr * 2)
        up_h = _interpolate(best_h, n_new)
        up_h = _repair_vector(up_h)
        phase = min(30.0, time_left * 0.5)
        h_up_opt, s_up_opt = _sa_optimize(up_h, phase, rng)
        if s_up_opt < best_score:
            best_h, best_score = h_up_opt, s_up_opt
        else:
            best_h = up_h  # keep finer grid for possible further refinement

    # --- final repair and evaluation -----------------------------------------
    best_h = _repair_vector(best_h)
    final_score = _c5_value(best_h)
    return best_h, float(final_score), best_h.size
