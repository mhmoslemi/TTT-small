import time
import numpy as np
import scipy.optimize as opt

# -------------------------------------------------------------
# Feasibility utilities
# -------------------------------------------------------------
def project_to_feasible(h, target_sum, tol=1e-8, max_iter=15):
    """Project h onto [0,1]^n ∩ {sum = target_sum} by iterative redistribution."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # no interior points; distribute to the closest interior
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h)  # safest place to add/subtract
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    """Linear interpolation of base_h (length m) to length n_target, then project."""
    base_h = np.asarray(base_h, dtype=float)
    m = base_h.shape[0]
    if n_target == m:
        h_new = base_h.copy()
    else:
        x_src = np.linspace(0.0, 2.0, m, endpoint=False)
        x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_tgt, x_src, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def compute_c5(h):
    """Return the overlap constant C5 for sampled function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def compute_c5_and_grad(h):
    """
    Return (C5, grad) where grad is the subgradient of C5 w.r.t. h.
    Uses the lag L attaining the maximal correlation.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_corr = corr[max_idx]
    L = max_idx - (n - 1)          # lag in [-n+1, n-1]
    dx = 2.0 / n

    # a_i = 1 - h_{i+L} if i+L in range else 1
    a = np.ones(n)
    if L >= 0:
        a[: n - L] = 1.0 - h[L:]
    else:
        k = -L
        a[k:] = 1.0 - h[: n - k]

    # b_i = h_{i-L} if i-L in range else 0
    b = np.zeros(n)
    if L >= 0:
        b[L:] = h[: n - L]
    else:
        k = -L
        end = n - k
        if end > 0:
            b[:end] = h[k:]

    grad = (a - b) * dx
    val = max_corr * dx
    return val, grad

def refine_h(h, max_iters=12, step=0.03, time_limit=None, start_time=None, budget=None):
    """Iteratively move opposite to subgradient while staying feasible."""
    n = h.shape[0]
    val = compute_c5(h)
    for _ in range(max_iters):
        if time_limit is not None and start_time is not None and budget is not None:
            if time.time() - start_time > budget * 0.96:
                break
        _, grad = compute_c5_and_grad(h)
        cand = h - step * grad
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_to_feasible(cand, n / 2.0)
        cand_val = compute_c5(cand)
        if cand_val < val:
            h, val = cand, cand_val
            step = min(step * 1.2, 0.12)
        else:
            step *= 0.5
            if step < 1e-5:
                break
    return h, val

                best_c5, best_h, best_n = c5_val, h_opt.copy(), n

            # early exit if we already beat the target bound
            if best_c5 <= 0.3808:
                # continue a few more iterations for possible further gains
                pass

    # fallback if nothing found (unlikely)
    if best_h is None:
        if base_h is not None:
            best_h = project_to_feasible(base_h, base_h.shape[0] / 2.0)
            best_n = best_h.shape[0]
        else:
            best_n = 190
            best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # final safety projection
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
