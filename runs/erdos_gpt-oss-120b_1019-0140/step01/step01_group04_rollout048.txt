def simulated_annealing(v, time_limit):

    """Project v onto the feasible set: 0<=v<=1 and sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)

            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()

            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the (unscaled) overlap bound for vector v."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * (2.0 / n)

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.linspace(0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _optimize_at_resolution(n, init_v, time_limit):
    """Run a short SLSQP optimization from init_v within time_limit."""
    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)

    def obj(x):
        return _compute_c5(x)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bounds = [(0.0, 1.0)] * n

    # primary run
    try:
        res = minimize(
            obj,
            init_v,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 300, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            v = _repair_vector(res.x, n)
            s = _compute_c5(v)
            if s < best_score:
                best_score, best_v = s, v
    except Exception:
        pass

    # random restarts until time runs out
    while time.time() - start < time_limit * 0.9:
        rnd = _random_feasible(n)
        try:
            res = minimize(
                obj,
                rnd,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 200, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n)
                s = _compute_c5(v)
                if s < best_score:
                    best_score, best_v = s, v
        except Exception:
            pass
        if time.time() - start > time_limit * 0.95:
            break
    return best_v, best_score

def _local_search(v, n, time_limit):
    """Simple hill‑climbing: exchange small amounts between two coordinates."""
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        s = _compute_c5(cand)
        if s < best_score:
            best_score, best_v = s, cand
    return best_v, best_score

def _subgradient_descent(v, time_limit):
    """Projected sub‑gradient descent on the maximal overlap."""
    start = time.time()
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(best_v)

    while time.time() - start < time_limit:
        corr = np.correlate(v, 1.0 - v, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)      # shift causing the current max
        # construct shifted versions with zero padding
        if k >= 0:
            v_shift_plus = np.concatenate([v[k:], np.zeros(k)])
            v_shift_minus = np.concatenate([np.zeros(k), v[:n - k]])
        else:
            kp = -k
            v_shift_plus = np.concatenate([np.zeros(kp), v[:n - kp]])
            v_shift_minus = np.concatenate([v[kp:], np.zeros(kp)])
        grad = 1.0 - v_shift_plus - v_shift_minus  # sub‑gradient of the max term
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = 0.08 * (1.0 - frac) + 0.001   # decreasing step
        cand = v - step * grad
        cand = _repair_vector(cand, n)
        s = _compute_c5(cand)
        if s < best_score:
            best_score, best_v = s, cand.copy()
        v = cand
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function for the Erdős C5 problem."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Acquire an initial feasible construction.
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 200
        parent = None

    if parent is None:
        # sinusoidal start, already respects bounds
        idx = np.arange(n0)
        parent = 0.5 + 0.5 * np.sin(2.0 * np.pi * idx / n0)
    parent = _repair_vector(parent, n0)

    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    # Define a list of resolutions (doubling while reasonable)
    max_n = 800
    resolutions = [n0]
    while resolutions[-1] * 2 <= max_n:
        resolutions.append(resolutions[-1] * 2)

    # ------------------------------------------------------------------
    # Main optimisation loop over resolutions.
    for r_idx, n in enumerate(resolutions):
        elapsed = time.time() - start
        remaining = budget_s - elapsed - 0.5  # safety margin
        if remaining <= 0:
            break
        remaining_res = len(resolutions) - r_idx
        time_for_res = remaining / remaining_res

        # Prepare initial guess for this resolution
        if len(best_v) != n:
            init_v = _upsample_vector(best_v, n)
        else:
            init_v = best_v.copy()

        # ---------- SLSQP phase ----------
        slsqp_time = min(15.0, time_for_res * 0.15)
        if slsqp_time > 0.1:
            v_opt, sc_opt = _optimize_at_resolution(n, init_v, slsqp_time)
            if sc_opt < best_score:
                best_v, best_score = v_opt, sc_opt
                best_n = n

        # ---------- Sub‑gradient descent ----------
        remaining_after_slsqp = time_for_res - slsqp_time - 0.5
        subg_time = max(0.0, remaining_after_slsqp * 0.7)
        if subg_time > 0.2:
            v_sg, sc_sg = _subgradient_descent(best_v if len(best_v) == n else init_v, subg_time)
            if sc_sg < best_score:
                best_v, best_score = v_sg, sc_sg
                best_n = n

        # ---------- Local search refinement ----------
        remaining_after_subg = time_for_res - slsqp_time - subg_time
        local_time = max(0.0, remaining_after_subg - 0.1)
        if local_time > 0.2:
            v_ls, sc_ls = _local_search(best_v, n, local_time)
            if sc_ls < best_score:
                best_v, best_score = v_ls, sc_ls
                best_n = n

    # Final repair and evaluation
    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(best_n)
