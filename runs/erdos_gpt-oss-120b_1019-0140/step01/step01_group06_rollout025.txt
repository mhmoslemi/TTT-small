import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # scaling factor dx = 2/n
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=20):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        # indices interior
        interior = (h > 0) & (h < 1)
        count = np.sum(interior)
        if count == 0:
            # no interior to adjust; adjust smallest margin
            # find index with max capacity to move
            capacity_up = 1 - h
            capacity_down = h
            # Choose direction of diff
            if diff > 0:
                # increase some index with largest capacity_up
                i = np.argmax(capacity_up)
                inc = min(diff, capacity_up[i])
                h[i] += inc
            else:
                i = np.argmax(capacity_down) # largest down capacity
                dec = min(-diff, capacity_down[i])
                h[i] -= dec
            # then clip
            h = np.clip(h, 0.0, 1.0)
            break
        delta = diff / count
        h[interior] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction if still off within tolerance?
    if abs(np.sum(h) - target_sum) > 1e-6:
        # adjust by distributing proportionally to capacity
        diff = target_sum - np.sum(h)
        # add diff to the element with largest capacity in direction
        if diff > 0:
            capacities = 1 - h
            i = np.argmax(capacities)
            h[i] = min(1.0, h[i] + diff)
        else:
            capacities = h
            i = np.argmax(capacities)
            h[i] = max(0.0, h[i] + diff) # diff negative
    return h

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = len(base_h)
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target/2.0)
    return h_new

            population.append(h0)
        # Evaluate

        # Sort population by score

        population = [population[i] for i in idx]
        scores = [scores[i] for i in idx]

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Compute C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-9, max_iter=20):
    """Clip to [0,1] and adjust to satisfy sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        interior = (h > 0.0) & (h < 1.0)
        cnt = np.count_nonzero(interior)
        if cnt == 0:
            # No interior variables: adjust a single entry with most slack.
            if diff > 0:
                slack = 1.0 - h
                i = int(np.argmax(slack))
                inc = min(diff, slack[i])
                h[i] += inc
            else:
                slack = h
                i = int(np.argmax(slack))
                dec = min(-diff, slack[i])
                h[i] -= dec
            h = np.clip(h, 0.0, 1.0)
            break
        delta = diff / cnt
        h[interior] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction (if needed)
    cur = np.sum(h)
    diff = target_sum - cur
    if abs(diff) > 1e-8:
        if diff > 0:
            slack = 1.0 - h
            i = int(np.argmax(slack))
            h[i] = min(1.0, h[i] + diff)
        else:
            slack = h
            i = int(np.argmax(slack))
            h[i] = max(0.0, h[i] + diff)
    return h


def resample_h(base_h, n_target):
    """Linear interpolation of base_h to length n_target, then project."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    return project_to_feasible(h_new, n_target / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h minimizing the Erdős C₅ bound."""
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Candidate resolutions (must keep n modest for speed)
    resolutions = [380, 760, 950]

    # Parent construction, if supplied
    parent_h = globals().get("initial_h_values")
    if parent_h is not None:
        parent_h = np.asarray(parent_h, dtype=float)

    for n in resolutions:
        if time.time() - start > budget_s * 0.97:
            break
        target_sum = n / 2.0

        # Population initialization
        pop_size = 20
        population = []

        if parent_h is not None:
            population.append(resample_h(parent_h, n))

        while len(population) < pop_size:
            rand = rng.random(n)
            population.append(project_to_feasible(rand, target_sum))

        # Evaluate initial population
        scores = [compute_c5(h) for h in population]
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_c5:
            best_c5 = scores[0]
            best_h = population[0].copy()
            best_n = n

        # Evolutionary search
        sigma = 0.12  # initial mutation scale
        gen = 0
        while time.time() - start < budget_s * 0.94:
            gen += 1
            # Generate offspring from top half
            offspring = []
            for h in population[:pop_size // 2]:
                for _ in range(2):
                    child = h + rng.normal(scale=sigma, size=n)
                    child = project_to_feasible(child, target_sum)
                    offspring.append(child)
            # Occasionally inject fresh random individuals
            if rng.random() < 0.07:
                for _ in range(2):
                    fresh = rng.random(n)
                    offspring.append(project_to_feasible(fresh, target_sum))

            # Combine and select best pop_size
            candidates = population + offspring
            cand_scores = [compute_c5(h) for h in candidates]
            idx = np.argsort(cand_scores)[:pop_size]
            population = [candidates[i] for i in idx]
            scores = [cand_scores[i] for i in idx]

            # Update global best
            if scores[0] < best_c5:
                best_c5 = scores[0]
                best_h = population[0].copy()
                best_n = n

            # Decay mutation scale slowly
            sigma = max(0.001, sigma * 0.998)

        # Local polishing with SLSQP (if time permits)
        if time.time() - start < budget_s * 0.92:
            h0 = best_h.copy()
            cons = {"type": "eq", "fun": lambda h, nd=n: np.sum(h) - nd / 2.0}
            bnds = [(0.0, 1.0)] * n
            try:
                res = opt.minimize(
                    fun=lambda h: compute_c5(h),
                    x0=h0,
                    method="SLSQP",
                    bounds=bnds,
                    constraints=cons,
                    options={"maxiter": 600, "ftol": 1e-9, "disp": False},
                )
                if res.success:
                    h_opt = np.clip(res.x, 0.0, 1.0)
                    h_opt = project_to_feasible(h_opt, target_sum)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        best_n = n
            except Exception:
                pass

    # Ensure final feasibility and correct bound
    if best_h is None:
        # fallback to a simple balanced step
        n_fb = 380
        best_h = np.full(n_fb, 0.5)
        best_n = n_fb
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
