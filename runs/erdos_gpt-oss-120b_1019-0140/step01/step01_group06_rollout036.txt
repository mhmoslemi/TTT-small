def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # free indices where 0 < h < 1
        mask = (h > 0.0) & (h < 1.0)
        free = np.where(mask)[0]
        if free.size == 0:
            # If no free variables, clamp by adjusting extremes
            # Spread diff proportionally to those at bounds with room.
            # We'll distribute to all indices equally but clipping ensures bounds.
            free = np.arange(len(h))
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final small fix
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust the element with max capacity
        # capacity to increase if cur < target, else capacity to decrease
        if cur < target_sum:
            # increase the smallest h
            i = np.argmin(h)
            h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
        else:
            # reduce the largest
            i = np.argmax(h)
            h[i] = np.clip(h[i] - (cur - target_sum), 0.0, 1.0)
    return h

import time
import numpy as np
import scipy.optimize as opt

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=200):
    """Clip to [0,1] and adjust sum to target_sum by distributing the deficit."""
    h = np.clip(h, 0.0, 1.0)
    n = h.shape[0]
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            free = np.arange(n)
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-8:
        idx = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[idx] = np.clip(h[idx] + (target_sum - cur), 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def compute_c5(h):
    """Exact evaluation using np.correlate (matches the harness)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def corr_gradient(h, lag):
    """Gradient of the correlation term for a single lag."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        L = lag
        if L < n:
            g[:n - L] += 1.0 - h[L:]
            g[L:] -= h[:n - L]
    else:
        m = -lag
        if m < n:
            g[m:] += 1.0 - h[:n - m]
            g[:n - m] -= h[m:]
    return g

def compute_c5_and_grad(h):
    """Return C5 and its gradient using FFT for speed."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)
    corr = np.real(np.fft.ifft(H * np.conj(G)))[:2 * n - 1]
    max_corr = np.max(corr)
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    g = corr_gradient(h, lag)
    grad = (2.0 / n) * g
    c5 = max_corr * (2.0 / n)
    return c5, grad

def projected_gradient_descent(h0, n, start_time, budget_s,
                              max_iter=500, step0=0.2,
                              decay=0.85, growth=1.1):
    """Simple PGD on the max‑correlation objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step0
    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.99:
            break
        c5, grad = compute_c5_and_grad(h)
        h_new = h - step * grad
        h_new = project_to_feasible(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:

            h = h_new
            step = min(step * growth, 1.0)
        else:

        if step < 1e-8:
