import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Compute max_k ∫ h(x)*(1-h(x+k)) dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Scale/shift entries of h so that sum(h) == target while keeping 0 <= h_i <= 1.
    Small residual errors are spread uniformly and clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Attempt to fix the whole deficit/surplus by touching random positions
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _gradient_of_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Sub‑gradient of the worst‑shift term f_k(h)=∑ h_i(1-h_{i+shift})
    with respect to h (sum of components is zero).
    """
    n = h.shape[0]
    g = np.ones(n, dtype=float)

    if shift >= 0:
        # term −h_{i+shift}
        if shift < n:
            g[: n - shift] -= h[shift:]
        # term −h_{i-shift}
        if shift == 0:
            g -= h            # subtract twice when shift==0
        else:
            g[shift:] -= h[: n - shift]
    else:
        l = -shift
        # term −h_{i-shift}  (i >= l)
        g[l:] -= h[: n - l]
        # term −h_{i+shift}  (i < n-l)
        g[: n - l] -= h[l:]

    # make the step leave the total mass unchanged
    g -= g.mean()
    return g

                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = np.clip(delta, low, high)
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        # enforce box constraints and exact mass
        np.clip(candidate, 0.0, 1.0, out=candidate)
        candidate = adjust_sum_to_target(candidate, target_sum, rng)

        cand_c5 = compute_c5(candidate)
        delta_val = cand_c5 - current_c5

        if delta_val < 0.0 or rng.rand() < np.exp(-delta_val / max(T, 1e-12)):
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                last_improve = now

        # if we have been idle for a while, perturb to escape stagnation
        if now - last_improve > 10.0:
            noise = (rng.rand(n_points) - 0.5) * step * 10.0
            pert = best_h + noise
            np.clip(pert, 0.0, 1.0, out=pert)
            pert = adjust_sum_to_target(pert, target_sum, rng)
            current_h = pert
            current_c5 = compute_c5(current_h)
            last_improve = now

    # -------- final polishing with random feasible swaps --------
    # small fixed budget (≈1 % of total time) for greedy improvement
    polishing_end = time.time() + min(5.0, max_time * 0.01)
    while time.time() < polishing_end:
        i, j = rng.randint(n_points, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = rng.rand() * (high - low) + low
        cand = best_h.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand
            # keep the improvement time stamp to avoid premature restart
            last_improve = time.time()

    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
