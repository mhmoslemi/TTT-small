def local_optimize(h0, n, rng, start_time, budget_s):

def simulated_annealing(start_h, n, rng, time_limit):
    start = time.time()
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    # schedule parameters
    max_iter = 20000  # limit
    i = 0
    # initial temperature
    T0 = 0.2
    Tf = 0.001
    while True:
        if time.time() - start > time_limit:
            break
        # compute progress ratio
        progress = i / max_iter
        T = T0 * ((Tf / T0) ** progress)  # exponential cooling
        # propose
        # choose sigma decreasing with temperature: sigma = 0.1 * (T/T0 + 0.1)
        sigma = 0.1 * (T / T0 + 0.1)  # ensures sigma >= 0.01
        # add noise
        new_h = cur_h + rng.normal(scale=sigma, size=n)
        # optionally clip
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_feasible(new_h, n/2)
        new_val = compute_c5(new_h)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
        i += 1
        # limit iterations
        if i >= max_iter:
            break
    return best_h, best_val

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    # Clip to bounds
    h = np.clip(h, 0.0, 1.0)
    # Compute current sum
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # Find indices not at bounds
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust a random index which can accommodate change
        # find index where adding/subtracting diff does not violate bounds
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff)
                break
            elif diff < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff)
                break
        # after adjusting, clip again
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for sampled h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    """Project onto [0,1]^n ∩ {sum = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # small correction on any index that can absorb the remaining diff
        diff = target_sum - cur
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff)
                break
            if diff < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff)
                break
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Linear interpolation of base_h to length n_target with feasibility projection."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def local_optimize(h0, n, rng, start_time, budget_s):
    """Few‑iteration SLSQP refinement, respecting the overall time budget."""
    if time.time() - start_time > budget_s * 0.96:
        h_feas = project_to_feasible(h0, n / 2.0)
        return h_feas, compute_c5(h_feas)

    def obj(h):
        return compute_c5(h)

    cons = {"type": "eq", "fun": lambda h: np.sum(h) - n / 2.0}
    bounds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(
            obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 250, "ftol": 1e-8, "disp": False},
        )
        h_opt = np.clip(res.x, 0.0, 1.0)
    except Exception:
        h_opt = h0
    h_opt = project_to_feasible(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)


def simulated_annealing(start_h, n, rng, time_limit):
    """Simple SA on the feasible set."""
    start = time.time()
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    max_iter = 20000
    T0, Tf = 0.20, 0.001
    i = 0
    while True:
        if time.time() - start > time_limit or i >= max_iter:
            break
        progress = i / max_iter
        T = T0 * ((Tf / T0) ** progress)  # exponential cooling
        sigma = 0.10 * (T / T0 + 0.1)      # keep sigma >= 0.01
        new_h = cur_h + rng.normal(scale=sigma, size=n)
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_feasible(new_h, n / 2.0)
        new_val = compute_c5(new_h)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
        i += 1
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h minimizing the Erdős C5 overlap."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)

    # Candidate sample sizes (even sizes make binary patterns easier)
    candidate_ns = [95, 190, 380, 760]

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Phase 1: many quick SLSQP restarts
    while time.time() - start_time < budget_s * 0.75:
        n = int(rng.choice(candidate_ns))

        # Choose an initialization strategy
        if base_h is not None and rng.random() < 0.4:
            h0 = resample_h(base_h, n)
        elif rng.random() < 0.5 and n % 2 == 0:
            # Random binary pattern with exactly n/2 ones
            idx = rng.choice(n, size=n // 2, replace=False)
            h0 = np.zeros(n)
            h0[idx] = 1.0
        else:
            h0 = rng.random(n)
            h0 = project_to_feasible(h0, n / 2.0)

        # Local refinement
        h_opt, c5_val = local_optimize(h0, n, rng, start_time, budget_s)

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = n

        # Very early stop if we already beat the target comfortably
        if best_c5 <= 0.3806:
            break

    # Phase 2: refine the current best with simulated annealing (if time permits)
    remaining = budget_s - (time.time() - start_time)
    if best_h is not None and remaining > 6:
        # allocate most of the leftover time to SA
        sa_time = remaining * 0.6
        h_sa, c5_sa = simulated_annealing(best_h, best_n, rng, sa_time)
        # one final SLSQP pass
        h_ref, c5_ref = local_optimize(h_sa, best_n, rng, start_time, budget_s)
        if c5_ref < best_c5:
            best_c5, best_h = c5_ref, h_ref.copy()

    # Safety fallback
    if best_h is None:
        if base_h is not None:
            best_h = base_h.copy()
            best_n = best_h.shape[0]
        else:
            best_n = 95
            best_h = np.full(best_n, 0.5)
        best_h = project_to_feasible(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    # Final projection and evaluation to be sure
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
