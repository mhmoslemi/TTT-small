def enforce_sum_and_bounds(h, target):
    # Clip to [0,1]
    np.clip(h,0,1,out=h)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return
    # Determine indices that are not at bounds
    if diff > 0:
        # Need to increase some entries
        free = np.where(h < 1 - 1e-12)[0]
        if free.size == 0:
            return
        inc = diff / free.size
        # maybe need loop to respect bounds; do iterative distribution
        while diff > 1e-12 and free.size > 0:
            inc = diff / free.size
            # cap inc to max allowed per entry
            max_inc = 1 - h[free]
            # entries where inc <= max_inc, else limited
            limited = max_inc < inc
            # update limited entries to 1
            if np.any(limited):
                # amount used
                used = np.sum(max_inc[limited])
                h[free[limited]] = 1.0
                diff -= used
                # remove those indices from free
                free = free[~limited]
                continue
            # else inc fits
            h[free] += inc
            diff = 0
    else:
        # need to decrease
        free = np.where(h > 1e-12)[0]
        while diff < -1e-12 and free.size > 0:
            dec = -diff / free.size
            max_dec = h[free]
            limited = max_dec < dec
            if np.any(limited):
                used = np.sum(max_dec[limited])
                h[free[limited]] = 0.0
                diff += used
                free = free[~limited]
                continue
            h[free] -= dec
            diff = 0

def _repair_sum(h, target):
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Find indices that can adjust
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]

        np.clip(h, 0.0, 1.0, out=h)

    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:

    return h

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry (recursively) so that sum(h) == target."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # distribute uniformly (fallback)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _make_binary_candidate(n: int, target: float, rng) -> np.ndarray:
    """Binary (or half‑binary) vector with given total sum."""
    int_part = int(math.floor(target))
    frac = target - int_part
    vec = np.zeros(n, dtype=float)
    if int_part > 0:
        vec[:int_part] = 1.0
    if frac > 1e-12 and int_part < n:
        vec[int_part] = frac
    rng.shuffle(vec)
    return vec

def _gradient_step(h: np.ndarray, target: float, step: float, beta: float) -> np.ndarray:
    """One gradient‑ascent step on the soft‑min of autocorrelation."""
    n = h.shape[0]
    # autocorrelation (zero‑padded)
    ac = np.correlate(h, h, mode="full")
    min_ac = np.min(ac)
    # soft‑min weights (larger weight for smaller autocorrelation)
    w = np.exp(-beta * (ac - min_ac))
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    grad = np.zeros_like(h)
    # accumulate gradient contributions from each shift
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            end = n - shift
            if end > 0:
                grad[:end] += weight * h[shift:]
                grad[shift:] += weight * h[:end]
        else:
            s = -shift
            end = n - s
            if end > 0:
                grad[s:] += weight * h[:end]
                grad[:end] += weight * h[s:]
    new_h = h + step * grad
    np.clip(new_h, 0.0, 1.0, out=new_h)
    _repair_sum(new_h, target)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] minimizing the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise problem size and feasible starting point -----
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = int(init.shape[0])
        target = n / 2.0
        init = np.clip(init, 0.0, 1.0)
        init = _repair_sum(init, target)
    except NameError:
        n = 120
        target = n / 2.0
        init = _make_binary_candidate(n, target, rng)

    # ----- build initial population for differential evolution -----
    popsize = max(12, min(30, n // 4))
    pop = np.empty((popsize, n), dtype=float)
    pop[0] = init.copy()
    pop[1] = _make_binary_candidate(n, target, rng)
    for i in range(2, popsize):
        vec = rng.random(n)
        vec = _repair_sum(vec, target)
        pop[i] = vec
    fitness = np.empty(popsize, dtype=float)
    for i in range(popsize):
        fitness[i] = _compute_c5(pop[i])
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]

    # ----- differential evolution phase (≈60% of budget) -----
    de_time = budget_s * 0.6
    F = 0.8
    CR = 0.9
    while time.time() - start < de_time:
        for i in range(popsize):
            if time.time() - start >= de_time:
                break
            # mutation
            idxs = rng.choice([j for j in range(popsize) if j != i], size=3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            mutant = a + F * (b - c)
            np.clip(mutant, 0.0, 1.0, out=mutant)
            mutant = _repair_sum(mutant, target)
            # crossover
            cross = pop[i].copy()
            jrand = rng.integers(0, n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            cross[mask] = mutant[mask]
            cross = _repair_sum(cross, target)
            # selection
            trial_c5 = _compute_c5(cross)
            if trial_c5 < fitness[i]:
                pop[i] = cross
                fitness[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = cross.copy()

    # ----- gradient‑ascent refinement on the best vector (remaining time) -----
    # schedule: step decays linearly, beta grows to sharpen the min
    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        elapsed = now - start
        frac = elapsed / budget_s
        step = 0.12 * (1.0 - frac) + 1e-5 * frac
        beta = 0.5 + 19.5 * frac  # from 0.5 to 20
        cand = _gradient_step(best_h, target, step, beta)
        cand_c5 = _compute_c5(cand)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
        else:
            # occasional random perturbation to escape plateaus
            if rng.random() < 0.01:
                pert = best_h + rng.normal(scale=step, size=n)
                np.clip(pert, 0.0, 1.0, out=pert)
                pert = _repair_sum(pert, target)
                pert_c5 = _compute_c5(pert)
                if pert_c5 < best_c5:
                    best_c5 = pert_c5
                    best_h = pert.copy()

    # final feasibility enforcement
    best_h = _repair_sum(best_h, target)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
