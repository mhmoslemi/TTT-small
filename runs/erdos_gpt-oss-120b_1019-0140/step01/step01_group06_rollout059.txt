def compute_c5(h):
    n=len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum):
    # same as earlier
    h = np.clip(h,0.0,1.0)
    for _ in range(10):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = np.where((h>0.0) & (h<1.0))[0]
        if free.size == 0:
            # distribute to all entries if cannot adjust? break
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h,0.0,1.0)
    # final correction
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust one entry
        i = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def gradient_descent_max_corr(h, max_iter=2000, alpha=10.0, lr=0.5, seed=None, budget_s=1100, start_time=None):

    rng = np.random.default_rng(seed)

def local_refine_slsqp(h, max_iter=500, seed=None):

                        x0=h,

    h_opt = np.clip(res.x,0.0,1.0)
    h_opt = project_to_feasible(h_opt, target_sum)
    return h_opt, compute_c5(h_opt)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-10, max_iter=20):
    """Project h onto [0,1]^n ∩ {sum = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-8:
        i = np.argmin(h) if cur < target_sum else np.argmax(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def subgradient_descent(h0, max_iter=1500, base_lr=1.0, seed=0,
                        budget_s=1100, start_time=None):
    """Subgradient descent on the max‑overlap objective."""
    rng = np.random.default_rng(seed)
    n = h0.shape[0]
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if start_time is not None and time.time() - start_time > budget_s * 0.96:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)  # integer lag
        # compute subgradient for this lag
        g = np.zeros(n, dtype=float)
        if lag >= 0:
            # first term (1 - h_{j+lag})
            g[0:n - lag] = 1.0 - h[lag:n]
            # second term -h_{j-lag}
            g[lag:n] -= h[0:n - lag]
        else:
            l = -lag
            g[l:n] = 1.0 - h[0:n - l]
            g[0:n - l] -= h[l:n]
        grad = dx * g
        # diminishing step size
        step = base_lr / (1.0 + it * 0.01)
        h_new = h - step * grad
        # occasional jitter to escape flat regions
        if it % 500 == 0 and it > 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_to_feasible(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def local_refine_slsqp(h0, max_iter=800):
    """Refine a feasible h using SLSQP."""
    n = h0.shape[0]
    target_sum = n / 2.0
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(fun=lambda x: compute_c5(x),
                           x0=h0,
                           method="SLSQP",
                           bounds=bnds,
                           constraints=cons,
                           options={"maxiter": max_iter,
                                    "ftol": 1e-9,
                                    "disp": False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, target_sum)
    return h_opt, compute_c5(h_opt)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Initial construction (if supplied)
    parent = globals().get("initial_h_values")
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Resolution cascade (start coarse, then finer)
    for n in (380, 760):
        if time.time() - start_time > budget_s * 0.96:
            break

        # Choose start point: upsample best so far if available, else parent, else random
        if best_h is not None:
            h0 = resample_h(best_h, n)
        elif parent is not None:
            h0 = resample_h(parent, n)
        else:
            h0 = rng.random(n)
            h0 = project_to_feasible(h0, n / 2.0)

        # Perform a few independent restarts
        for restart in range(5):
            if time.time() - start_time > budget_s * 0.96:
                break
            if restart == 0:
                h_start = h0.copy()
            else:
                # perturb to diversify
                h_start = h0 + rng.normal(scale=0.04, size=n)
                h_start = project_to_feasible(h_start, n / 2.0)

            # Subgradient descent phase
            gd_h, gd_c5 = subgradient_descent(
                h_start,
                max_iter=1500,
                base_lr=1.2,
                seed=seed + restart,
                budget_s=budget_s,
                start_time=start_time,
            )

            # Local SLSQP refinement
            refined_h, refined_c5 = local_refine_slsqp(gd_h, max_iter=800)

            # Update global best
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
                best_n = n

            # Early stop if exceptionally good
            if best_c5 <= 0.3805:
                break

    # Fallback in the unlikely case no solution was found
    if best_h is None:
        n = 380
        best_n = n
        best_h = np.full(n, 0.5)
        best_h = project_to_feasible(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)

    # Final safety projection
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
