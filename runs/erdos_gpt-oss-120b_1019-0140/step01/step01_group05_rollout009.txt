def _simulated_annealing_custom(h_start, time_limit, rng):
    h_cur = h_start.copy()
    N = h_cur.size
    dx = 2.0/N
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    start = time.time()
    # temperature schedule linear bound
    T0 = 0.02 # high initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac
        # choose move type
        if rng.random() < 0.5:
            # random pair move
            i, j = rng.integers(0, N, size=2)
            while i == j:
                j = rng.integers(0, N)
            hi, hj = h_cur[i], h_cur[j]
            up = min(1.0 - hi, hj)   # positive epsilon bound
            down = min(hi, 1.0 - hj)  # negative epsilon magnitude bound
            if up == 0.0 and down == 0.0:
                continue
            eps = rng.random() * (up + down) - down
            # cap by max_eps
            max_eps = 0.25
            if abs(eps) > max_eps:
                eps = math.copysign(max_eps, eps)
            if eps == 0.0:
                continue
            h_new = h_cur.copy()
            h_new[i] = hi + eps
            h_new[j] = hj - eps
        else:
            # target worst shift move
            # compute correlation and get worst shift
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            idx_max = int(np.argmax(corr))
            k = idx_max - (N - 1)
            # overlapping region
            i_start = max(0, -k)
            i_end = min(N, N - k)
            if i_end - i_start < 1:
                # fallback to random move
                continue
            i = rng.integers(i_start, i_end)
            j = i + k
            hi, hj = h_cur[i], h_cur[j]
            up_target = min(1.0 - hi, 1.0 - hj)
            if up_target <= 0.0:
                continue
            # donor index
            # choose donor p not i or j with h>0
            attempts = 0
            p = None
            while attempts < 5:
                p_candidate = rng.integers(0, N)
                if p_candidate in (i, j):
                    attempts += 1
                    continue
                if h_cur[p_candidate] > 0.0:
                    p = p_candidate
                    break
                attempts += 1
            if p is None:
                continue
            # donor capacity for giving 2*eps
            donor_cap = h_cur[p] / 2.0
            max_eps_target = min(up_target, donor_cap, 0.25)
            if max_eps_target <= 0.0:
                continue
            eps = rng.random() * max_eps_target
            # apply move
            h_new = h_cur.copy()
            h_new[i] = hi + eps
            h_new[j] = hj + eps
            h_new[p] = h_cur[p] - 2.0 * eps
        # ensure bounds
        if np.any(h_new < 0.0) or np.any(h_new > 1.0):
            # optional repair quickly (clip but may change sum)
            h_new = _repair_vector(h_new)
        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import numpy as np
import time
import math

# ----- utility functions -----------------------------------------------------

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) == len(v)/2 (integral = 1)."""
    np.clip(v, 0.0, 1.0, out=v)
    N = v.size
    target = N / 2.0
    s = v.sum()
    if s == 0.0:
        v.fill(target / N)
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v


def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _shape_descending(N: int) -> np.ndarray:
    x = np.linspace(0.0, 2.0, N, endpoint=False)
    return np.maximum(0.0, 1.0 - x / 2.0)


def _shape_ascending(N: int) -> np.ndarray:
    x = np.linspace(0.0, 2.0, N, endpoint=False)
    return np.maximum(0.0, x / 2.0)


def _shape_tent(N: int) -> np.ndarray:
    x = np.linspace(0.0, 2.0, N, endpoint=False)
    return np.where(x <= 1.0, x, 2.0 - x)


def _shape_sine(N: int) -> np.ndarray:
    x = np.linspace(0.0, 2.0, N, endpoint=False)
    return np.sin(np.pi * x / 2.0)


def _get_initial_vector(N: int, rng: np.random.Generator) -> np.ndarray:
    """Return a start vector of length N, using the global
    `initial_h_values` if present, otherwise a small set of heuristics."""
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        if h0.size != N:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        return _repair_vector(h0.copy())
    except NameError:
        # generate a few candidate shapes, keep the best after quick eval
        cand = [
            _shape_descending(N),
            _shape_ascending(N),
            _shape_tent(N),
            _shape_sine(N),
            rng.random(N),               # pure random
        ]
        best = None
        best_score = float('inf')
        for v in cand:
            v = _repair_vector(v.copy())
            sc = _c5_value(v)
            if sc < best_score:
                best_score = sc
                best = v
        return best


def _random_initial(N: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random start respecting the constraints."""
    v = rng.random(N)
    return _repair_vector(v)


# ----- simulated annealing with directed worst‑shift moves ------------------

TARGET_BOUND = 0.3808  # early‑stop target


def _simulated_annealing(h_start: np.ndarray,
                         time_limit: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Annealing that mixes random pair adjustments with a move that
    explicitly tries to increase the inner product for the worst shift."""
    h_cur = h_start.copy()
    N = h_cur.size
    dx = 2.0 / N
    cur_corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    cur_score = float(np.max(cur_corr) * dx)
    best_h = h_cur.copy()
    best_score = cur_score

    start = time.time()
    T0 = 0.02                 # initial temperature
    max_eps = 0.25            # absolute bound on epsilon
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        # linear cooling
        frac = max(0.0, 1.0 - (now - start) / time_limit)
        T = T0 * frac

        # decide move type
        if rng.random() < 0.5:
            # ----- random pair move (preserve sum) -----
            i, j = rng.integers(0, N, size=2)
            while i == j:
                j = rng.integers(0, N)
            hi, hj = h_cur[i], h_cur[j]
            up = min(1.0 - hi, hj)   # amount we can increase hi / decrease hj
            down = min(hi, 1.0 - hj) # amount we can decrease hi / increase hj
            if up == 0.0 and down == 0.0:
                continue
            eps = rng.random() * (up + down) - down   # uniform in [-down, up]
            if abs(eps) > max_eps:
                eps = math.copysign(max_eps, eps)
            if eps == 0.0:
                continue
            h_new = h_cur.copy()
            h_new[i] = hi + eps
            h_new[j] = hj - eps
        else:
            # ----- directed move targeting the worst shift -----
            # correlation already computed above
            idx_max = int(np.argmax(cur_corr))
            k = idx_max - (N - 1)               # shift value
            if k == 0:
                # fall back to random move
                continue
            i_start = max(0, -k)
            i_end = min(N, N - k)               # exclusive
            if i_end - i_start < 1:
                continue
            i = rng.integers(i_start, i_end)
            j = i + k
            hi, hj = h_cur[i], h_cur[j]
            up_target = min(1.0 - hi, 1.0 - hj)
            if up_target <= 0.0:
                continue
            # donor index not overlapping with i or j, with some mass
            donor = None
            for _ in range(5):
                p = rng.integers(0, N)
                if p == i or p == j:
                    continue
                if h_cur[p] > 0.0:
                    donor = p
                    break
            if donor is None:
                continue
            donor_cap = h_cur[donor] / 2.0
            max_eps_target = min(up_target, donor_cap, max_eps)
            if max_eps_target <= 0.0:
                continue
            eps = rng.random() * max_eps_target
            h_new = h_cur.copy()
            h_new[i] = hi + eps
            h_new[j] = hj + eps
            h_new[donor] = h_cur[donor] - 2.0 * eps

        # compute new score
        new_corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_score = float(np.max(new_corr) * dx)

        # Metropolis acceptance
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h_cur = h_new
            cur_corr = new_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= TARGET_BOUND:
                    # sufficient improvement – finish early
                    return best_h, best_score
    return best_h, best_score


# ----- main entry point -----------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # Choose a base resolution (not too large, not too small)
    base_N = 200
    try:
        init_len = len(initial_h_values)
        if init_len > base_N:
            base_N = init_len
    except NameError:
        pass
    base_N = min(base_N, 800)  # enforce the 1k limit comfortably

    # Initial vector (try to use the parent if available)
    h0 = _get_initial_vector(base_N, rng)

    # Phase 1 – SA at the base resolution
    time_used = time.time() - start_total
    phase1_time = min(budget_s * 0.6, budget_s - time_used)
    best_h, best_score = _simulated_annealing(h0, phase1_time, rng)

    # Phase 2 – upscale if time permits
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.2:
        up_N = min(base_N * 2, 800)
        if up_N != best_h.size:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, up_N, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
        else:
            h_up = best_h.copy()
        h_up = _repair_vector(h_up)
        phase2_time = remaining * 0.9
        h2, s2 = _simulated_annealing(h_up, phase2_time, rng)
        if s2 < best_score:
            best_h, best_score = h2, s2

    # Phase 3 – occasional random restarts with remaining budget
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    while remaining > 15.0:
        # a fresh random start of the current best resolution
        h_rand = _random_initial(best_h.size, rng)
        chunk = min(30.0, remaining * 0.5)
        cand_h, cand_s = _simulated_annealing(h_rand, chunk, rng)
        if cand_s < best_score:
            best_h, best_score = cand_h, cand_s
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed

    # Final repair and exact evaluation
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), best_h.size
