def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def random_binary_vector(n, rng):
    # target exactly n//2 ones
    half = n // 2
    # generate random permutation and set first half to 1
    indices = rng.permutation(n)[:half]
    vec = np.zeros(n, dtype=int)
    vec[indices] = 1
    return vec

def to_binary_from_continuous(v):
    n = len(v)
    half = n // 2
    idx = np.argpartition(v, -half)[-half:]
    bin_vec = np.zeros(n, dtype=int)
    bin_vec[idx] = 1
    return bin_vec

def make_child(p1, p2, rng):
    n = len(p1)
    half = n // 2
    # positions where both have 1
    common_one = np.where((p1 == 1) & (p2 == 1))[0]
    child = np.zeros(n, dtype=int)
    child[common_one] = 1
    need = half - len(common_one)
    if need > 0:

        diff_one = np.where((p1 == 1) ^ (p2 == 1))[0]
        # randomly choose need of them
        chosen = rng.choice(diff_one, size=need, replace=False)
        child[chosen] = 1
    # else if need <=0, child already has half ones, but could have more? But need can't be negative theoretically because common_one <= half.
    return child

def mutate_swap(v, rng):
    ones = np.flatnonzero(v)
    zeros = np.flatnonzero(1 - v)
    if len(ones)==0 or len(zeros)==0:

def tournament_select(fitnesses, population, k, rng):
    # select k random individuals and return index of best (lowest fitness)
    idx = rng.choice(len(population), size=k, replace=False)
    best_idx = idx[0]
    best_fit = fitnesses[best_idx]
    for i in idx[1:]:
        if fitnesses[i] < best_fit:
            best_fit = fitnesses[i]
            best_idx = i
    return best_idx

def ga_search(initial_vec, n, budget_s, start_time, rng):
    # GA parameters
    POP_SIZE = 40
    TOURNAMENT_SIZE = 5
    MUTATION_PROB = 0.2
    # initialize population
    population = []
    fitnesses = []
    # include seeded initial vector if provided
    if initial_vec is not None:
        population.append(initial_vec.copy())
        fitnesses.append(compute_c5(initial_vec))
    # fill rest random
    while len(population) < POP_SIZE:
        vec = random_binary_vector(n, rng)
        # ensure not duplicate maybe but not required
        population.append(vec)
        fitnesses.append(compute_c5(vec))
    # sort initially? maybe not needed
    best_idx = min(range(len(population)), key=lambda i: fitnesses[i])
    best_vec = population[best_idx].copy()
    best_fit = fitnesses[best_idx]
    # generation loop
    while time.time() - start_time < budget_s * 0.7:
        # generate children
        children = []
        children_fitness = []
        for _ in range(POP_SIZE):
            # select parents
            p1_idx = tournament_select(fitnesses, population, TOURNAMENT_SIZE, rng)
            p2_idx = tournament_select(fitnesses, population, TOURNAMENT_SIZE, rng)
            # ensure different? not necessary.
            child = make_child(population[p1_idx], population[p2_idx], rng)
            # mutation
            if rng.random() < MUTATION_PROB:
                child = mutate_swap(child, rng)
            # evaluate
            child_fit = compute_c5(child)
            children.append(child)
            children_fitness.append(child_fit)
            # update global best
            if child_fit < best_fit:
                best_fit = child_fit
                best_vec = child.copy()
        # combine and select top POP_SIZE
        combined = population + children
        combined_fitness = fitnesses + children_fitness
        # get indices of smallest POP_SIZE fitnesses
        # use np.argsort
        sorted_idx = np.argsort(combined_fitness)[:POP_SIZE]
        # update population and fitnesses
        population = [combined[i] for i in sorted_idx]
        fitnesses = [combined_fitness[i] for i in sorted_idx]
    return best_vec, best_fit

                best_vec = cur_vec.copy()

            # revert
            cur_vec[i], cur_vec[j] = 1, 0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            # shift forward by k positions
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx
```

def slsqp_refine(start_h, start_c5, start_time, time_limit):
    n = len(start_h)
    sum_cons = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_cons}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    best_h = start_h.copy()
    best_val = start_c5
    # maybe multiple restarts
    while time.time() - start_time < time_limit:
        # random perturbation of start_h
        perturb = rng.normal(scale=0.05, size=n)  # small noise
        h0 = start_h + perturb
        # clip
        h0 = np.clip(h0, 0.0, 1.0)
        # enforce sum via projection (maybe adjust a bit)
        h0 = enforce_sum(h0)
        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])
        res = minimize(fun=lambda x: x[-1],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})

        h_opt = np.clip(res.x[:n], 0.0, 1.0)

        h_opt = enforce_sum(h_opt)
        val = compute_c5(h_opt)
        if val < best_val:
            best_h = h_opt
            best_val = val
        # early exit if very good
        if best_val < 0.3805:  # but we can adjust threshold
            break
    return best_h, best_val

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ---------- basic utilities ----------
def compute_c5(h):
    """max_k ∫ h(x)(1‑h(x+k)) dx for discrete sample h."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def to_binary_from_continuous(v):
    """Return a binary vector of same length with the largest half entries set to 1."""
    n = len(v)
    half = n // 2
    idx = np.argpartition(v, -half)[-half:]
    b = np.zeros(n, dtype=int)
    b[idx] = 1
    return b

def random_binary_vector(n, rng):
    half = n // 2
    vec = np.zeros(n, dtype=int)
    vec[rng.choice(n, size=half, replace=False)] = 1
    return vec

def enforce_sum(vec):
    """Shift vec into the feasible hyperplane sum(vec)=len(vec)/2, respecting [0,1]."""
    n = len(vec)
    target = n / 2.0
    cur = np.sum(vec)
    if abs(cur - target) > 1e-12:
        vec = vec + (target - cur) / n
        vec = np.clip(vec, 0.0, 1.0)
        cur = np.sum(vec)
        diff = target - cur
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0 and vec[i] < 1.0:
                inc = min(diff, 1.0 - vec[i])
                vec[i] += inc
                diff -= inc
            elif diff < 0 and vec[i] > 0.0:
                dec = min(-diff, vec[i])
                vec[i] -= dec
                diff += dec
    return vec

# ---------- constraints for SLSQP ----------
class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[: self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= dx * Σ h_i (1‑h_{i+k}) for given shift k."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            ov = np.dot(h[: self.n - k], 1.0 - h[k :])
        else:
            kk = -k
            ov = np.dot(h[kk :], 1.0 - h[: self.n - kk])
        return x[-1] - ov * self.dx

# ---------- objective for DE ----------
class DEObjective:
    def __init__(self, n, target_sum, penalty):
        self.n = n
        self.target_sum = target_sum
        self.penalty = penalty
        self.dx = 2.0 / n
    def __call__(self, h):
        sum_pen = (np.sum(h) - self.target_sum) ** 2
        max_ov = np.max(np.correlate(h, 1.0 - h, mode="full")) * self.dx
        return max_ov + self.penalty * sum_pen

# ---------- main entry point ----------

    rng = np.random.default_rng(seed)

    # ---- determine candidate problem sizes ----
    base_n = 80

    candidates = [base_n]
    # add a larger size if time permits

    if budget_s - (time.time() - start_time) > 600:
        candidates.append(base_n + 80)          # e.g. 160

    best_h = None

        elapsed = time.time() - start_time

        de_budget = remaining * 0.6

        de_obj = DEObjective(n, target_sum, penalty=1e3)
        bounds = [(0.0, 1.0)] * n

            if c5_opt < best_c5:

    # ---- final safety check ----
    if best_h is None:

        best_h = random_binary_vector(n, rng).astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
