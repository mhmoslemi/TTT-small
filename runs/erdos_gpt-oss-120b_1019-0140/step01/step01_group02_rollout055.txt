import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the max overlap integral C5 for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: spread uniformly and clip
        h[:] = np.clip(h + diff / h.size, 0.0, 1.0)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that lowers the Erdős overlap constant C5.
    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Desired discretisation size (choose a moderate size)
    n_target = 300  # < 1k as requested
    target_sum = n_target / 2.0

    # ----- initialise a feasible candidate ---------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)  # use parent construction if present
        n0 = init.shape[0]
        # interpolate to desired size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        current = np.interp(x_new, x0, init)
    except NameError:
        # No parent construction – start from a random balanced vector
        current = rng.uniform(0.0, 1.0, size=n_target)

    # clip and enforce exact sum
    np.clip(current, 0.0, 1.0, out=current)
    current = _repair_sum(current, target_sum)

    # ----- bookkeeping ------------------------------------------------------
    current_c5 = _compute_c5(current)
    best_h = current.copy()
    best_c5 = current_c5

    # parameters for annealing / moves
    max_step = 0.5
    min_step = 0.001
    max_temp = 0.01
    min_temp = 1e-6
    p_swap = 0.15          # probability of performing a full swap move
    no_improve_limit = 60000
    no_improve_counter = 0

    # ----- main optimisation loop -----------------------------------------
    while True:
        now = time.time()
        if now - start_time >= budget_s:
            break
        frac = (now - start_time) / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # Decide move type
        if rng.random() < p_swap:
            # full swap move
            i, j = rng.integers(0, n_target, size=2)
            while i == j:
                i, j = rng.integers(0, n_target, size=2)
            # execute swap
            current[i], current[j] = current[j], current[i]
            new_c5 = _compute_c5(current)
            delta_obj = new_c5 - current_c5
            if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / temp):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current.copy()
                    no_improve_counter = 0
                else:
                    no_improve_counter += 1
            else:
                # revert swap
                current[i], current[j] = current[j], current[i]
                no_improve_counter += 1
        else:
            # small delta move on a pair
            i, j = rng.integers(0, n_target, size=2)
            while i == j:
                i, j = rng.integers(0, n_target, size=2)
            hi = current[i]
            hj = current[j]
            # feasible delta interval respecting bounds
            delta_min = max(-hi, -(1.0 - hj))
            delta_max = min(1.0 - hi, hj)
            low = max(delta_min, -step)
            high = min(delta_max, step)
            if low >= high:
                no_improve_counter += 1
                continue
            delta = rng.random() * (high - low) + low
            # apply move
            current[i] = hi + delta
            current[j] = hj - delta
            new_c5 = _compute_c5(current)
            delta_obj = new_c5 - current_c5
            if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / temp):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current.copy()
                    no_improve_counter = 0
                else:
                    no_improve_counter += 1
            else:
                # revert
                current[i] = hi
                current[j] = hj
                no_improve_counter += 1

        # occasional perturbation when stalled
        if no_improve_counter >= no_improve_limit:
            # re‑seed current around the best known solution
            current = best_h.copy()
            # perform a few random swaps to diversify
            for _ in range(30):

            current = _repair_sum(current, target_sum)
            current_c5 = _compute_c5(current)
            no_improve_counter = 0

    # Final repair to guarantee exact sum (tiny drift possible)
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n_target
