import numpy as np, time

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free_idx = np.where(v < 1.0)[0]
        caps = 1.0 - v[free_idx]
        # allocate proportionally to capacity
        # for simplicity, fill in order of largest caps
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free_idx[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free_idx = np.where(v > 0.0)[0]
        caps = v[free_idx]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free_idx[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _anneal_optimize(h_start, n, target_sum, time_budget, start_time):
    # h_start may be different size; interpolate
    if h_start.size != n:
        x_src = np.linspace(0, 2, h_start.size, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_src, h_start)
    else:
        h = h_start.copy()
    h = _project_to_constraints(h, target_sum)
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    max_iters = 20000
    # parameters
    temp_start = 0.02
    temp_end = 0.0001
    for it in range(max_iters):
        if time.time() - start_time >= time_budget * 0.99:
            break
        # linear temperature decay
        t = temp_start * ((temp_end / temp_start) ** (it / max_iters))
        sigma = 0.05 * (t / temp_start) + 0.005  # keep minimal noise
        proposal = h + sigma * np.random.randn(n)
        proposal = _project_to_constraints(proposal, target_sum)
        proposal_score = _compute_c5(proposal)
        delta = proposal_score - cur_score
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(t, 1e-12)):
            h = proposal
            cur_score = proposal_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                if best_score < 0.3808:
                    break
    return best_h, best_score

import numpy as np
import time

def _compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h) (equivalent to np.correlate with mode='full')
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(v, target):
    """Clip v to [0,1] and adjust to satisfy sum(v)=target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) < 1e-12:
        return v
    if cur < target:
        need = target - cur
        free = np.where(v < 1.0)[0]
        caps = 1.0 - v[free]
        order = np.argsort(caps)[::-1]  # largest capacity first
        for idx in order:
            i = free[idx]
            add = min(need, caps[idx])
            v[i] += add
            need -= add
            if need <= 1e-12:
                break
    else:
        excess = cur - target
        free = np.where(v > 0.0)[0]
        caps = v[free]
        order = np.argsort(caps)[::-1]
        for idx in order:
            i = free[idx]
            sub = min(excess, caps[idx])
            v[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    if abs(cur - target) > 1e-8 and cur > 0:
        v = v * (target / cur)
        v = np.clip(v, 0.0, 1.0)
    return v

def _anneal_optimize(h_start, n, target_sum, time_budget, global_start):
    """Simulated‑annealing style refinement for a given resolution."""
    # interpolate if size mismatch
    if h_start.size != n:
        x_src = np.linspace(0, 2, h_start.size, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_src, h_start)
    else:
        h = h_start.copy()
    h = _project_to_constraints(h, target_sum)
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    max_iters = 20000
    temp_start = 0.02
    temp_end = 0.0001

    for it in range(max_iters):
        if time.time() - global_start >= time_budget * 0.99:
            break
        # exponential temperature decay
        t = temp_start * ((temp_end / temp_start) ** (it / max_iters))
        sigma = 0.05 * (t / temp_start) + 0.005
        proposal = h + sigma * np.random.randn(n)
        proposal = _project_to_constraints(proposal, target_sum)
        prop_score = _compute_c5(proposal)
        delta = prop_score - cur_score
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(t, 1e-12)):
            h = proposal
            cur_score = prop_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                if best_score < 0.3808:
                    break
    return best_h, best_score
