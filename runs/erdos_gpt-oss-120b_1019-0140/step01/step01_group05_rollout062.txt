def _c5_value(h):
    N = h.shape[0]
    dx = 2.0 / N
    # Use FFT convolution.
    # Compute correlation via convolution with reversed (1-h).
    b = 1.0 - h
    b_rev = b[::-1]
    size = 1 << ((2*N-1).bit_length())
    # Use np.fft.rfft for real inputs.
    H = np.fft.rfft(h, n=size)
    B = np.fft.rfft(b_rev, n=size)
    conv = np.fft.irfft(H * B, n=size)[:2*N-1]
    max_corr = conv.max()
    return float(max_corr * dx)

Note: Using rfft for speed; size is power of 2.

def _simulated_annealing_general(h_start, time_limit, rng):
    h = h_start.copy()
    cur_score = _c5_value(h)
    best_h = h.copy()
    best_score = cur_score
    N = h.size
    start = time.time()
    T0 = 0.5  # initial temperature (scale)
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # temperature linearly decreasing
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)
        # pick i,j distinct
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1
        hi = h[i]; hj = h[j]
        if hi == 1.0 and hj == 0.0:
            # cannot increase hi nor decrease hj, but we can swap entire mass
            eps = -1.0
        elif hi == 0.0 and hj == 1.0:
            eps = 1.0
        else:
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up == 0.0 and down == 0.0:
                continue
            # sometimes full transfer
            if rng.random() < 0.1:
                # full move
                eps = up if rng.random() < 0.5 else -down
            else:
                eps = rng.random() * (up + down) - down
        if eps == 0.0:
            continue
        # apply
        h[i] = hi + eps
        h[j] = hj - eps
        # clip just in case due to floating error
        # but we ensured eps within bounds, but floating may exceed.
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        new_score = _c5_value(h)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
    return best_h, best_score

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and scale to exact sum = len(v)/2."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for discretised step function."""
    N = h.size
    dx = 2.0 / N
    b = 1.0 - h
    b_rev = b[::-1]
    size = 1 << ((2 * N - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    B = np.fft.rfft(b_rev, n=size)
    conv = np.fft.irfft(H * B, n=size)[:2 * N - 1]
    max_corr = np.max(conv)
    return float(max_corr * dx)

def _simulated_annealing(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """SA with full-mass swaps, respecting box and sum constraints."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.5                     # initial temperature
    while True:
        if time.time() - start >= time_limit:
            break
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)

        # pick two distinct indices
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        hi, hj = h_cur[i], h_cur[j]

        up = min(1.0 - hi, hj)   # amount we can move from j to i
        down = min(hi, 1.0 - hj) # amount we can move from i to j
        if up == 0.0 and down == 0.0:
            continue

        # 20% of the time take a random fraction, otherwise a full transfer
        if rng.random() < 0.2:
            eps = rng.random() * (up + down) - down
        else:
            eps = up if rng.random() < 0.5 else -down
        if eps == 0.0:
            continue

        # apply move
        h_cur[i] = hi + eps
        h_cur[j] = hj - eps
        # numerical safety
        h_cur[i] = max(0.0, min(1.0, h_cur[i]))
        h_cur[j] = max(0.0, min(1.0, h_cur[j]))

        new_score = _c5_value(h_cur)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            # revert
            h_cur[i] = hi
            h_cur[j] = hj
    return best_h, best_score

def _interpolate(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current grid to a grid of size new_n."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
