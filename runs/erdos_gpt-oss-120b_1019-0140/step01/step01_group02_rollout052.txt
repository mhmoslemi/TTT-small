    max_step = 0.5
    min_step = 1e-4
    max_temp = 5e-3
    min_temp = 1e-7

    # ----- optimisation loop ---------------------------------------------
    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        elapsed = now - start
        frac = elapsed / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # correlation and worst shift
        corr = np.correlate(current, 1.0 - current, mode="full")
        c5 = float(np.max(corr) * (2.0 / n))
        k = int(np.argmax(corr) - (n - 1))
        current_c5 = c5

        # gradient for the worst shift
        forward = _shift(current, k)
        backward = _shift(current, -k)
        g = 1.0 - forward - backward

        # choose donor (positive gradient) and receiver (negative gradient)
        i_pos = None
        i_neg = None
        pos_order = np.argsort(-g)  # descending
        for idx in pos_order:
            if g[idx] > 1e-12 and current[idx] > 1e-12:
                i_pos = int(idx)
                break
        neg_order = np.argsort(g)   # ascending
        for idx in neg_order:
            if g[idx] < -1e-12 and current[idx] < 1.0 - 1e-12:
                i_neg = int(idx)
                break

        moved = False
        if i_pos is not None and i_neg is not None:
            hi = float(current[i_pos])
            hj = float(current[i_neg])
            delta = min(step, hi, 1.0 - hj)
            if delta > 0:
                current[i_pos] = hi - delta
                current[i_neg] = hj + delta
                new_c5 = _compute_c5(current)
                if new_c5 <= current_c5 or rng.random() < math.exp(-(new_c5 - current_c5) / temp):
                    current_c5 = new_c5
                    moved = True
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = current.copy()
                else:
                    current[i_pos] = hi
                    current[i_neg] = hj

        if not moved:
            # fallback random transfer preserving the sum
            i, j = rng.integers(0, n, size=2)
            if i != j:
                hi = float(current[i])
                hj = float(current[j])
                delta_min = max(-hi, -(1.0 - hj))
                delta_max = min(1.0 - hi, hj)
                low = max(delta_min, -step)
                high = min(delta_max, step)
                if low < high:
                    delta = rng.random() * (high - low) + low
                    current[i] = hi + delta
                    current[j] = hj - delta
                    new_c5 = _compute_c5(current)
                    if new_c5 <= current_c5 or rng.random() < math.exp(-(new_c5 - current_c5) / temp):
                        current_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = current.copy()
                    else:
                        current[i] = hi
                        current[j] = hj

    # ----- final repair and return -----------------------------------------
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, n
