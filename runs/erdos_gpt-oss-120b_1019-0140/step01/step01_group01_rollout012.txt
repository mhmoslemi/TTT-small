def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.2

    # determine n_points
    try:
        init = np.array(initial_h_values, dtype=float)
        n_points = init.size
    except Exception:
        n_points = 200
        init = None

    if n_points % 2 != 0:
        # ensure even for binary weight
        n_points = n_points + 1

    target_ones = n_points // 2

    # function to generate random binary with exact weight
    def random_binary():
        arr = np.zeros(n_points, dtype=np.uint8)
        arr[:target_ones] = 1
        rng.shuffle(arr)
        return arr

    # initialize population
    popsize = 30
    pop = np.empty((popsize, n_points), dtype=np.uint8)
    for i in range(popsize):
        pop[i] = random_binary()
    # if initial exists, incorporate its binary form by adjusting
    if init is not None:
        # convert to binary by threshold >=0.5
        init_bin = (init >= 0.5).astype(np.uint8)
        ones = init_bin.sum()
        # adjust to target_ones
        if ones > target_ones:
            # turn some 1s to 0
            idx_one = np.flatnonzero(init_bin)
            rng.shuffle(idx_one)
            init_bin[idx_one[:ones-target_ones]] = 0
        elif ones < target_ones:
            idx_zero = np.flatnonzero(init_bin == 0)
            rng.shuffle(idx_zero)
            init_bin[idx_zero[:target_ones-ones]] = 1
        # place into population
        pop[0] = init_bin

    # precompute fitness
    def fitness(arr):
        return compute_c5(arr.astype(float))

    fitness_vals = np.empty(popsize)
    for i in range(popsize):
        fitness_vals[i] = fitness(pop[i])

    best_idx = np.argmin(fitness_vals)
    best_h = pop[best_idx].copy()
    best_c5 = float(fitness_vals[best_idx])

    # tournament selection
    tournament_k = 3

    # GA loop
    while True:
        now = time.time()
        if now - start >= max_time:
            break

        # select parents
        # tournament for parent A
        idxs = rng.randint(popsize, size=tournament_k)
        parent_a = pop[idxs[np.argmin(fitness_vals[idxs])]]
        # tournament for parent B
        idxs = rng.randint(popsize, size=tournament_k)
        parent_b = pop[idxs[np.argmin(fitness_vals[idxs])]]

        # create offspring via crossover or copy
        if rng.rand() < 0.5:
            # two-point crossover
            cut1, cut2 = sorted(rng.choice(n_points, size=2, replace=False))
            child = np.empty(n_points, dtype=np.uint8)
            child[:cut1] = parent_a[:cut1]
            child[cut1:cut2] = parent_b[cut1:cut2]
            child[cut2:] = parent_a[cut2:]
            # repair to ensure correct weight
            diff = child.sum() - target_ones
            if diff > 0:
                # too many ones: turn diff ones to zero
                idx_one = np.flatnonzero(child)
                if idx_one.size > 0:
                    rng.shuffle(idx_one)
                    child[idx_one[:diff]] = 0
            elif diff < 0:
                # need more ones
                idx_zero = np.flatnonzero(child == 0)
                if idx_zero.size > 0:
                    rng.shuffle(idx_zero)
                    child[idx_zero[: -diff]] = 1
        else:
            # copy parent_a
            child = parent_a.copy()

        # mutation: swap bits
        # determine number of swaps based on remaining time progress (more swap early)
        progress = (now - start) / max_time
        # early: many swaps up to ~5; later: 1
        max_swaps = 5
        num_swaps = max(1, int(max_swaps * (1 - progress)))

            child[i] = child[j]
            child[j] = tmp

            fitness_vals[worst_idx] = child_c5
            if child_c5 < best_c5:
                best_c5 = child_c5
                best_h = child.copy()

    # Return best_h as float array normalized (0/1)
    best_h_float = best_h.astype(float)
    final_c5 = compute_c5(best_h_float)
    return (best_h_float, float(final_c5), n_points)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); np.correlate implements sum_i a[i]*v[i+lag]
    corr = np.correlate(h.astype(float), 1.0 - h, mode="full")
    return float(corr.max() * dx)


def random_binary(n: int, ones: int, rng: np.random.RandomState) -> np.ndarray:
    """Uniform random binary vector of length n with exactly `ones` ones."""
    arr = np.zeros(n, dtype=np.uint8)
    arr[:ones] = 1
    rng.shuffle(arr)
    return arr


def repair_binary(arr: np.ndarray, target_ones: int, rng: np.random.RandomState) -> np.ndarray:
    """Adjust `arr` (binary) to have exactly `target_ones` ones."""
    diff = int(arr.sum() - target_ones)
    if diff > 0:
        # too many ones -> turn some to zero
        idx_one = np.flatnonzero(arr)
        if idx_one.size:
            rng.shuffle(idx_one)
            arr[idx_one[:diff]] = 0
    elif diff < 0:
        # too few ones -> turn some zeros to one
        idx_zero = np.flatnonzero(arr == 0)
        if idx_zero.size:
            rng.shuffle(idx_zero)
            arr[idx_zero[: -diff]] = 1
    return arr


def two_point_crossover(a: np.ndarray, b: np.ndarray, n: int,
                        target_ones: int, rng: np.random.RandomState) -> np.ndarray:
    """Create child by two‑point crossover of parents a and b, then repair weight."""
    cut1, cut2 = sorted(rng.choice(n, size=2, replace=False))
    child = np.empty(n, dtype=np.uint8)
    child[:cut1] = a[:cut1]
    child[cut1:cut2] = b[cut1:cut2]
    child[cut2:] = a[cut2:]
    return repair_binary(child, target_ones, rng)


def mutate_swap(arr: np.ndarray, swaps: int, rng: np.random.RandomState) -> None:
    """Swap `swaps` pairs of entries in place (preserves number of ones)."""
    n = arr.size
    for _ in range(swaps):
        i, j = rng.randint(n, size=2)
        if i != j:
            arr[i], arr[j] = arr[j], arr[i]


def local_improve(h: np.ndarray, max_iters: int, rng: np.random.RandomState) -> np.ndarray:
    """
    Greedy improvement: for the most offending shift k, swap a 'bad' one
    with a 'good' zero to reduce overlap.
    """
    n = h.size
    for _ in range(max_iters):
        # find lag with maximal overlap
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        k_idx = int(np.argmax(corr_full))
        k = k_idx - (n - 1)

        idx = np.arange(n)
        if k >= 0:
            valid = idx + k < n
            shifted = np.empty(n, dtype=np.uint8)
            shifted[valid] = h[idx[valid] + k]
            shifted[~valid] = 0
        else:
            valid = idx + k >= 0
            shifted = np.empty(n, dtype=np.uint8)
            shifted[valid] = h[idx[valid] + k]
            shifted[~valid] = 0

        bad = (h == 1) & (shifted == 0)   # contributes 1 to the overlap
        good = (h == 0) & (shifted == 1)  # would reduce the overlap if turned on

        bad_idx = np.flatnonzero(bad)
        good_idx = np.flatnonzero(good)

        if bad_idx.size == 0 or good_idx.size == 0:
            break

        i = rng.choice(bad_idx)
        j = rng.choice(good_idx)
        h[i] = 0
        h[j] = 1
    return h

        progress = (now - start) / max_time
        swaps = max(1, int(5 * (1 - progress)))
        mutate_swap(child, swaps, rng)

        # evaluate child
        child_c5 = compute_c5(child.astype(float))

        # replace worst individual if child is better
        worst_idx = int(np.argmax(fitness_vals))
        if child_c5 < fitness_vals[worst_idx]:
            population[worst_idx] = child
            fitness_vals[worst_idx] = child_c5
            if child_c5 < best_c5:
                best_c5 = child_c5
                best_h = child.astype(float).copy()

    # ----- greedy local refinement ---------------------------------------
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        trial_h = best_h.copy()
        trial_h = local_improve(trial_h, max_iters=30, rng=rng)
        trial_c5 = compute_c5(trial_h)
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = trial_h.copy()
        else:
            break

    # final check (binary guarantees exact mass)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
