def _repair(v, n):
    v = np.clip(v, 0.0, 1.0)
    target = n * 0.5
    # adjust sum iteratively
    for _ in range(10):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # identify indices that can be moved in direction of diff
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = np.count_nonzero(mask)
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final small scaling if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        # adjust uniformly small
        v += (target - s) / n
        v = np.clip(v, 0.0, 1.0)
    return v
```

import time
import numpy as np
from scipy.signal import fftconvolve

def _repair(v, n):
    v = np.clip(v, 0.0, 1.0)
    target = n * 0.5
    # iterative correction
    for _ in range(15):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = np.count_nonzero(mask)
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final fine-tuning
    s = v.sum()
    if abs(s - target) > 1e-9:
        v += (target - s) / n
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5(v):
    n = v.shape[0]
    dx = 2.0 / n
    # correlation via FFT convolution
    corr = fftconvolve(v, (1.0 - v)[::-1], mode='full')
    return float(np.max(corr) * dx)

def _random_feasible(n):
    v = np.random.rand(n)
    return _repair(v, n)

def _anneal(v0, n, time_budget):

    best = cur.copy()
    best_score = cur_score
    # annealing parameters
    step = 0.2
    T = 0.025
    cooling = 0.999
    min_T = 1e-6
    while time.time() - start < time_budget:
        i, j = np.random.randint(n, size=2)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair(cand, n)
        cand_score = _c5(cand)
        dE = cand_score - cur_score
        if dE < 0 or np.random.rand() < np.exp(-dE / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
        T = max(min_T, T * cooling)
        # slowly shrink step size with temperature
        step = max(0.005, step * (0.9995 + 0.0005 * (T / 0.025)))
