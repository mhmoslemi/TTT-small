def _binary_repair(v):
    n = len(v)
    k = n // 2
    # if n odd, handle fractional? We'll enforce n even.
    idx = np.argpartition(v, -k)[-k:]

    h[idx] = 1.0
    return h

def _random_binary(n):
    k = n // 2
    h = np.zeros(n, dtype=np.float64)
    indices = np.random.choice(n, k, replace=False)
    h[indices] = 1.0

def _anneal_binary(start_vec, time_limit, delta_buffer):
    n = len(start_vec)
    A = start_vec.copy()  # binary values (0.0/1.0)
    B = 1.0 - A
    B_rev = B[::-1].copy()
    # compute correlation:
    C = np.correlate(A, B, mode='full')
    cur_max = C.max()
    # create lists of indices:
    ones = [int(i) for i in range(n) if A[i] > 0.5]
    zeros = [int(i) for i in range(n) if A[i] < 0.5]
    start_time = time.time()

    # start_temp = 0.1 * (cur_max - (n/2 - n/4)???). But simply set start_temp = 0.01? Since values are small.
    start_temp = 0.01

    best_h = A.copy()
    best_score = cur_max

    while time.time() - start_time < time_limit:

        elapsed = time.time() - start_time
        t_frac = elapsed / time_limit

def _anneal_binary(start_h, time_limit):
    n = len(start_h)
    A = start_h.copy()
    B = 1.0 - A
    C = np.correlate(A, B, mode='full')
    cur_max = C.max()
    ones = [int(i) for i in range(n) if A[i] > 0.5]
    zeros = [int(i) for i in range(n) if A[i] < 0.5]
    start = time.time()
    no_improve = 0
    best_h = A.copy()
    best_max = cur_max
    delta = np.empty(2*n - 1, dtype=np.float64)
    while time.time() - start < time_limit:
        if not ones or not zeros:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        # compute delta
        delta.fill(0.0)
        # term1
        start_i = n - 1 - i
        start_j = n - 1 - j
        delta[start_i:start_i + n] -= B
        delta[start_j:start_j + n] += B
        # term2
        A_rev = A[::-1]  # recompute each iteration; could move outside but after each swap A changes, so need update.
        delta[i:i + n] += A_rev
        delta[j:j + n] -= A_rev
        # term3
        center = n - 1
        delta[center] -= 2.0
        d1 = center + j - i
        d2 = center + i - j
        delta[d1] += 1.0
        delta[d2] += 1.0
        cand_max = np.max(C + delta)
        if cand_max + 1e-12 < cur_max:
            # accept
            C += delta
            cur_max = cand_max
            # swap A, B values
            A[i] = 0.0
            A[j] = 1.0
            B[i] = 1.0
            B[j] = 0.0
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            no_improve = 0
            # update best
            if cur_max < best_max:
                best_max = cur_max
                best_h = A.copy()
        else:
            no_improve += 1
            if no_improve > 5000:
                # restart randomly
                # generate new random binary vector and recompute everything
                A = _random_binary(n)
                B = 1.0 - A
                C = np.correlate(A, B, mode='full')
                cur_max = C.max()
                ones = [int(idx) for idx in range(n) if A[idx] > 0.5]
                zeros = [int(idx) for idx in range(n) if A[idx] < 0.5]
                no_improve = 0
                # optionally keep best seen.
    # return best_h and best_max
    return best_h, best_max

def _upsample_to_n(v, n_target):
    src_n = len(v)
    src_x = np.linspace(0, 2, src_n, endpoint=False)
    tgt_x = np.linspace(0, 2, n_target, endpoint=False)
    up = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return up

Now binary conversion:

def _binary_repair(v):
    n = len(v)
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

Now _generate_thue_morse:

def _generate_thue_morse(n):
    # parity of ones in binary representation
    # Use vectorized operation via bitwise count.
    # Could use np.arange and for each compute popcount using builtins.
    arr = np.arange(n, dtype=np.uint64)
    parity = np.frompyfunc(lambda x: bin(x).count('1') % 2, 1, 1)(arr).astype(np.int8)
    # parity 0 for even popcount, 1 for odd.
    # Use whichever yields half ones: we can just return parity as float.
    return parity.astype(np.float64)

def _random_binary(n):
    k = n // 2
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    h[idx] = 1.0
    return h

    import time, numpy as np
    np.random.seed(seed)
    start_time = time.time()
    # Determine target length
    # Try to extract initial_h_values.
    try:
        parent_arr = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent_arr)
    except Exception:
        parent_arr = None
        n0 = None

    # Choose target n: attempt up to 760, ensure even.
    # If n0 available and <= 200, maybe target = max(760, n0*4?) but we fix 760.
    target_n = 760
    if target_n % 2 == 1:
        target_n += 1

    seeds = []
    if parent_arr is not None:
        # Upsample to target_n via linear interp then binary
        src_n = len(parent_arr)
        src_x = np.linspace(0, 2, src_n, endpoint=False)
        tgt_x = np.linspace(0, 2, target_n, endpoint=False)

    for _ in range(2):

    # Define time allocation
    remaining_budget = max(0.0, budget_s - 5.0)  # keep 5 sec margin
    per_seed_budget = remaining_budget / len(seeds) if seeds else 0.0

    best_h = None
    best_score = np.inf
    for seed_h in seeds:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1.0:
            break
        time_for_this = min(per_seed_budget, budget_s - elapsed - 0.5)
        if time_for_this <= 0:

        h_opt, score_opt = _anneal_binary(seed_h, time_for_this)
        # score_opt is unscaled max correlation
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
    # After seeds, if time remains, refine best_h further
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed - 0.2)

    final_c5 = float(best_score * (2.0 / target_n))

import time
import numpy as np

def _binary_repair(v):
    """Return a binary vector with exactly floor(len(v)/2) ones, choosing the highest entries."""
    n = v.shape[0]
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def _random_binary(n):
    """Random binary vector with exactly n/2 ones (n assumed even)."""
    k = n // 2
    h = np.zeros(n, dtype=np.float64)
    h[np.random.choice(n, k, replace=False)] = 1.0
    return h

def _generate_thue_morse(n):
    """Parity of popcount of indices (0/1); length n."""
    arr = np.arange(n, dtype=np.uint64)
    # vectorized parity via Python function (fast enough for n≈800)
    parity = np.frompyfunc(lambda x: bin(x).count("1") % 2, 1, 1)(arr).astype(np.int8)
    return parity.astype(np.float64)

def _anneal_binary(start_h, time_limit):
    """Hill‑climbing / simulated‑annealing on a binary vector.
    Returns (best_vector, best_unscaled_score) where score is max correlation
    (not yet multiplied by dx)."""
    n = start_h.shape[0]
    A = start_h.copy()                     # 0/1 values
    B = 1.0 - A                             # complement
    C = np.correlate(A, B, mode="full")    # current correlation, length 2n‑1
    cur_max = C.max()
    best_h = A.copy()
    best_score = cur_max

    # indices of ones and zeros
    ones = [int(i) for i in range(n) if A[i] > 0.5]
    zeros = [int(i) for i in range(n) if A[i] < 0.5]

    delta = np.empty(2 * n - 1, dtype=np.float64)
    start = time.time()
    no_improve = 0
    # simple annealing schedule (mostly greedy)
    while time.time() - start < time_limit:
        if not ones or not zeros:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        # compute delta C for swapping i (1) with j (0)
        delta.fill(0.0)

        # term 1: effect of changing A entries on correlation (uses B)
        start_i = n - 1 - i
        start_j = n - 1 - j
        delta[start_i:start_i + n] -= B
        delta[start_j:start_j + n] += B

        # term 2: effect of changing B entries on correlation (uses reversed A)
        A_rev = A[::-1]
        delta[i:i + n] += A_rev
        delta[j:j + n] -= A_rev

        # term 3: cross term from both changes
        center = n - 1
        delta[center] -= 2.0
        d1 = center + j - i
        d2 = center + i - j
        delta[d1] += 1.0
        delta[d2] += 1.0

        cand_max = np.max(C + delta)

        if cand_max + 1e-12 < cur_max:
            # accept improvement
            C += delta
            cur_max = cand_max
            # update vectors
            A[i] = 0.0
            A[j] = 1.0
            B[i] = 1.0
            B[j] = 0.0
            # update index lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            no_improve = 0
            # record best
            if cur_max < best_score:
                best_score = cur_max
                best_h = A.copy()
        else:
            no_improve += 1
            if no_improve > 5000:
                # restart from a fresh random binary vector
                A = _random_binary(n)
                B = 1.0 - A
                C = np.correlate(A, B, mode="full")
                cur_max = C.max()
                ones = [int(k) for k in range(n) if A[k] > 0.5]
                zeros = [int(k) for k in range(n) if A[k] < 0.5]
                no_improve = 0
    return best_h, best_score

def _compute_c5(h):
    """Scaled C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function minimizing C5."""
    np.random.seed(seed)
    start_time = time.time()

    # Determine target resolution (even, ≤800)
    TARGET_N = 800
    if TARGET_N % 2:
        TARGET_N += 1

    # Build a list of seed vectors
    seeds = []

    # 1) parent (if supplied)
    try:
        parent = np.array(initial_h_values, dtype=np.float64)  # pre‑imported when available
        src_n = parent.shape[0]
        src_x = np.linspace(0.0, 2.0, src_n, endpoint=False)
        tgt_x = np.linspace(0.0, 2.0, TARGET_N, endpoint=False)
        up = np.interp(tgt_x, src_x, parent, left=parent[0], right=parent[-1])
        seeds.append(_binary_repair(up))
    except Exception:
        pass

    # 2) Thue‑Morse pattern
    thue = _generate_thue_morse(TARGET_N)
    seeds.append(_binary_repair(thue))

    # 3) a couple of random binary starts
    seeds.append(_random_binary(TARGET_N))
    seeds.append(_random_binary(TARGET_N))

    # Allocate time across seeds (reserve a few seconds for final steps)
    remaining_budget = max(0.0, budget_s - 5.0)
    per_seed = remaining_budget / len(seeds) if seeds else 0.0

    best_h = None
    best_score = np.inf  # unscaled max correlation
    for seed_vec in seeds:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1.0:
            break
        time_for_seed = min(per_seed, budget_s - elapsed - 0.5)
        if time_for_seed <= 0:
            break
        h_opt, sc_opt = _anneal_binary(seed_vec, time_for_seed)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # Use any remaining time for a final refinement of the best found vector
    elapsed = time.time() - start_time
    leftover = max(0.0, budget_s - elapsed - 0.2)
    if leftover > 0.2 and best_h is not None:
        h_opt, sc_opt = _anneal_binary(best_h, leftover)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # Fallback in the extremely unlikely case that no seed succeeded
    if best_h is None:
        best_h = _random_binary(TARGET_N)
        best_score = np.max(np.correlate(best_h, 1.0 - best_h, mode="full"))

    final_c5 = float(best_score * (2.0 / TARGET_N))
    return best_h.astype(np.float64), final_c5, int(TARGET_N)
