def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_obj(h):
    corr = compute_corr(h)
    return float(np.max(corr) * dx)  # dx = 2.0 / n

def softmax_weights(corr, alpha):
    # shift for numerical stability
    c = alpha * corr
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    sum_exp = np.sum(exp_c)
    return exp_c / sum_exp

def gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    p = softmax_weights(corr, alpha)  # shape 2n-1
    # Compute convs
    p_rev = p[::-1]
    conv1 = np.convolve(p_rev, h, mode='full')
    conv2 = np.convolve(p, h, mode='full')
    start = n - 1
    end = start + n
    s1 = conv1[start:end]
    s2 = conv2[start:end]
    grad = alpha * (1.0 - (s1 + s2))
    return grad

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value (max overlap integral) for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Choose an index that can move in the needed direction
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute evenly and clip
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _gradient(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of log‑sum‑exp(alpha * corr) w.r.t. h (unnormalized)."""
    n = h.shape[0]
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights over all shifts
    c = alpha * corr
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    p = exp_c / np.sum(exp_c)               # shape 2n‑1, sums to 1
    # convolution terms for the two symmetric parts of the derivative
    p_rev = p[::-1]
    conv1 = np.convolve(p_rev, h, mode='full')
    conv2 = np.convolve(p, h, mode='full')
    start = n - 1
    end = start + n
    s1 = conv1[start:end]
    s2 = conv2[start:end]
    grad = alpha * (1.0 - (s1 + s2))
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that lowers the
    Erdős overlap constant C5.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise a feasible candidate ---------------------------------
    try:
        init0 = np.asarray(initial_h_values, dtype=float)  # may be length 47
        n0 = init0.shape[0]
        # use at least 200 points for a richer search
        n = max(200, n0)
        old_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n, endpoint=False)
        init = np.interp(new_x, old_x, init0)
    except NameError:
        n = 200
        init = rng.random(n)

    target_sum = n / 2.0
    init = np.clip(init, 0.0, 1.0)
    init = _repair_sum(init, target_sum)

    current = init.copy()
    best_h = current.copy()
    best_c5 = _compute_c5(best_h)
    current_c5 = best_c5

    no_improve = 0
    # ----- optimisation loop -------------------------------------------------
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s:
            break
        frac = elapsed / budget_s

        # schedules
        alpha = 5.0 + 95.0 * frac                    # 5 → 100
        raw_step = 0.2 * (1.0 - frac) + 0.001 * frac  # 0.2 → 0.001
        step = raw_step / max(alpha, 1.0)            # scale down with alpha
        temp = 0.02 * (1.0 - frac) + 1e-6 * frac    # 0.02 → 1e‑6

        # gradient step
        grad = _gradient(current, alpha)
        proposal = current - step * grad
        np.clip(proposal, 0.0, 1.0, out=proposal)
        proposal = _repair_sum(proposal, target_sum)

        new_c5 = _compute_c5(proposal)
        delta = new_c5 - current_c5

        # Metropolis acceptance
        if delta < 0.0 or rng.random() < math.exp(-delta / temp):
            current = proposal
            current_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current.copy()
        else:

            current_c5 = _compute_c5(current)
            no_improve = 0

    # final repair and evaluation
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, n
