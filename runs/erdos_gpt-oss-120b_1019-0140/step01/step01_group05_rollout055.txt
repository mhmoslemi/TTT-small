def _repair_vector(v):
    N = v.size
    np.clip(v,0.0,1.0, out=v)
    target = N/2.0
    s = v.sum()
    # adjust sum using scaling factor and clipping iteratively
    if s == 0:
        v[:] = target / N
        np.clip(v,0.0,1.0, out=v)
        return v
    v *= target / s
    np.clip(v,0.0,1.0, out=v)
    # after clipping, sum may be off; adjust again
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        v *= target / s2
        np.clip(v,0.0,1.0, out=v)
    return v
```

    N = h.size

    corr = np.correlate(h, 1.0 - h, mode='full')

def _greedy_optimize(h_start, time_limit, rng):
    h = h_start.copy()
    N = h.size
    best_h = h.copy()
    best_score = _c5_value(h)
    start = time.time()
    # parameters
    min_eps = 1e-6
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        max_corr = corr[max_idx]   # raw correlation sum (no dx)
        # Determine shift
        shift = max_idx - (N - 1)  # integer
        # Determine overlapping index range
        if shift >= 0:
            i_range = np.arange(N - shift)
            j_range = i_range + shift
        else:
            shift_abs = -shift
            i_range = np.arange(shift_abs, N)
            j_range = i_range + shift  # shift negative
        # partner values
        partners = h[j_range]
        # source candidates (i where h[i] > min_eps)
        src_mask = h[i_range] > min_eps
        if not np.any(src_mask):
            # cannot move mass from any source (all zeros)
            break
        src_indices = i_range[src_mask]
        src_partners = partners[src_mask]
        # target candidates (i where h[i] < 1 - min_eps)
        tgt_mask = h[i_range] < (1.0 - min_eps)
        if not np.any(tgt_mask):
            # can't add mass
            break
        tgt_indices = i_range[tgt_mask]
        tgt_partners = partners[tgt_mask]
        # Identify src with smallest partner value
        src_idx = src_indices[np.argmin(src_partners)]
        src_partner_val = h[src_idx + shift] if shift >= 0 else h[src_idx + shift]  # same j_range logic
        # Identify tgt with largest partner value
        tgt_idx = tgt_indices[np.argmax(tgt_partners)]
        tgt_partner_val = h[tgt_idx + shift] if shift >=0 else h[tgt_idx + shift]
        # Determine max possible transfer
        max_eps_src = h[src_idx]
        max_eps_tgt = 1.0 - h[tgt_idx]
        max_eps_move = min(max_eps_src, max_eps_tgt)
        # If no move possible (should be >0)
        if max_eps_move <= 0:
            # fallback to random move
            # Perform random epsilon move between random pair preserving sum
            i, j = rng.integers(0, N, size=2)
            while i == j:

            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)

            h[i] = hi + eps
            h[j] = hj - eps

            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()

                h[j] = hj
            continue

        # Apply move

        # Ensure within [0,1] due to numeric issues
        # (but step ensures boundaries)

def _greedy_optimize(h_start, time_limit, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_score = _c5_value(h)
    start = time.time()
    N = h.size
    min_eps = 1e-8
    # We'll run for time_limit seconds, with early exit if too many failed attempts
    fail_streak = 0
    max_fail_streak = 2000
    while True:
        if time.time() - start >= time_limit:
            break
        # compute correlation array
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        # deduce shift
        shift = max_idx - (N - 1)
        # define overlapping indices
        if shift >= 0:
            i_range = np.arange(N - shift)
            partner_idx = i_range + shift
        else:
            shift_abs = -shift
            i_range = np.arange(shift_abs, N)
            partner_idx = i_range + shift
        partners = h[partner_idx]
        # candidate source indices: i_range where h[i] > min_eps
        src_mask = h[i_range] > min_eps
        if not np.any(src_mask):
            break
        src_indices = i_range[src_mask]
        src_partners = partners[src_mask]
        # candidate target indices: i_range where h[i] < 1 - min_eps
        tgt_mask = h[i_range] < (1.0 - min_eps)
        if not np.any(tgt_mask):
            break
        tgt_indices = i_range[tgt_mask]
        tgt_partners = partners[tgt_mask]
        # if no improvement possible (i.e., max tgt partner <= min src partner)
        # then try random move
        if np.max(tgt_partners) <= np.min(src_partners):
            # random move
            i, j = rng.integers(0, N, size=2)
            while i == j:
                j = rng.integers(0, N)
            hi, hj = h[i], h[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up + down == 0:
                continue
            eps = rng.random() * (up + down) - down
            if eps == 0.0:
                continue
            h[i] = hi + eps
            h[j] = hj - eps
            new_score = _c5_value(h)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                fail_streak = 0
            else:
                # revert
                h[i] = hi
                h[j] = hj
                fail_streak += 1
                if fail_streak >= max_fail_streak:
                    break
            continue
        # choose source with minimal partner
        src_idx = src_indices[np.argmin(src_partners)]
        # choose target with maximal partner
        tgt_idx = tgt_indices[np.argmax(tgt_partners)]
        # Determine epsilon max
        max_move = min(h[src_idx], 1.0 - h[tgt_idx])
        if max_move <= min_eps:
            # tiny possible move, skip
            fail_streak += 1
            if fail_streak >= max_fail_streak:
                break
            continue
        # Determine step size proportionally to remaining time and random factor
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        # step = max_move * (0.5 + 0.5 * rng.random()) * frac
        step = max_move * (0.1 + 0.9 * rng.random()) * frac
        if step < min_eps:
            step = min_eps
        # Apply move
        h[src_idx] -= step
        h[tgt_idx] += step
        # compute new score
        new_score = _c5_value(h)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = h.copy()
            fail_streak = 0
        else:
            # revert move
            h[src_idx] += step
            h[tgt_idx] -= step
            fail_streak += 1
            if fail_streak >= max_fail_streak:
                break
    return best_h, best_score

import numpy as np
import time

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip v to [0,1] and enforce sum(v) = len(v)/2 exactly."""
    n = v.size
    np.clip(v, 0.0, 1.0, out=v)
    target = n / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / n
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for discretised step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _greedy_optimize(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Local search targeting the worst lag, with random escapes."""
    h = h_start.copy()
    best_h = h.copy()
    best_score = _c5_value(h)
    n = h.size
    start = time.time()
    min_eps = 1e-12
    fail_streak = 0
    max_fail = 2000

    while True:
        if time.time() - start >= time_limit:
            break
        # correlation and worst lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)          # integer shift
        # overlapping indices for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            partner_idx = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            partner_idx = i_range + shift
        partners = h[partner_idx]

        src_mask = h[i_range] > min_eps
        tgt_mask = h[i_range] < (1.0 - min_eps)

        if not np.any(src_mask) or not np.any(tgt_mask):
            break

        src_indices = i_range[src_mask]
        src_parts = partners[src_mask]
        tgt_indices = i_range[tgt_mask]
        tgt_parts = partners[tgt_mask]

        # If no improvement possible for this lag, do a random epsilon move
        if np.max(tgt_parts) <= np.min(src_parts):
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            hi, hj = h[i], h[j]
            up = min(1.0 - hi, hj)
            down = min(hi, 1.0 - hj)
            if up + down == 0.0:
                continue
            eps = rng.random() * (up + down) - down
            if eps == 0.0:
                continue
            h[i] = hi + eps
            h[j] = hj - eps
            new_score = _c5_value(h)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                fail_streak = 0
            else:
                h[i] = hi
                h[j] = hj
                fail_streak += 1
                if fail_streak >= max_fail:
                    break
            continue

        # deterministic improvement: move from low‑partner to high‑partner
        src_idx = src_indices[np.argmin(src_parts)]
        tgt_idx = tgt_indices[np.argmax(tgt_parts)]
        max_move = min(h[src_idx], 1.0 - h[tgt_idx])
        if max_move <= min_eps:
            fail_streak += 1
            if fail_streak >= max_fail:
                break
            continue

        # step size shrinks as time runs out
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        step = max_move * (0.1 + 0.9 * rng.random()) * frac
        if step < min_eps:
            step = min_eps

        # apply move
        h[src_idx] -= step
        h[tgt_idx] += step
        new_score = _c5_value(h)
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_h = h.copy()
            fail_streak = 0
        else:
            # revert
            h[src_idx] += step
            h[tgt_idx] -= step
            fail_streak += 1
            if fail_streak >= max_fail:
                break

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # initial vector (use provided construction if any)
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()

        lvl_start = time.time()

        while True:

            # perturb current best slightly

            h_pert = h_opt + noise
            h_pert = _repair_vector(h_pert)
            run_time = remaining_lvl * 0.5
            h_new, score_new = _greedy_optimize(h_pert, run_time, rng)
            if score_new < score_opt:
                h_opt, score_opt = h_new, score_new
            # continue loop; remaining time shrinks automatically
        # update global best
        if score_opt < best_score_overall:

    return best_h_overall, float(best_score_overall), best_h_overall.size
