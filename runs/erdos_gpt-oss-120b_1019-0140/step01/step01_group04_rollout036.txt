import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        nfree = mask.sum()
        if nfree == 0:
            break
        adj = diff / nfree
        v[mask] -= adj
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the max overlap integral for a sample vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation to a new length, followed by repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _grad_worst_shift(v):
    """Gradient of the worst (max) correlation term w.r.t v."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode="full")
    k = int(np.argmax(corr))
    shift = k - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        idx = np.arange(n - shift)
        grad[idx] += 1.0 - v[idx + shift]
        grad[idx + shift] -= v[idx]
    else:
        s = -shift
        idx = np.arange(n - s)
        grad[idx + s] += 1.0 - v[idx]
        grad[idx] -= v[idx + s]
    return grad

def _gradient_descent(v0, time_limit):
    """Reduce the worst correlation by gradient steps."""
    start = time.time()
    n = len(v0)
    cur = v0.copy()
    cur_score = _compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    while time.time() - start < time_limit:
        grad = _grad_worst_shift(cur)
        # project to zero‑sum direction to respect the integral constraint
        grad -= grad.mean()
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = 0.1 * (1.0 - frac) + 1e-5

        cand_score = _compute_c5(cand)
        if cand_score < cur_score:

            cur_score = cand_score
        if cand_score < best_score:

            best_score = cand_score

    """Simple SA that swaps mass between two coordinates."""
