def _repair_sum(h, target):
    # iterative repair as before
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Identify indices able to move in needed direction
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]

        return h
    i = idx[0]

    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:

    return h

    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr_raw))
    lag = idx - (n - 1)
    i_low = max(0, -lag)
    i_high = min(n, n - lag)
    if i_high <= i_low:
        # no overlap, fallback
        return _random_swap(h, step, target_sum, rng)

    j = i + lag

import time, math, numpy as np

def _compute_c5(h):
    """Return (C5, full correlation array) for vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def _repair_sum(h, target):
    """Adjust a single entry (recursively) to enforce sum(h)==target."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

def _random_swap(h, step, target, rng):
    """Propose a feasible two‑index mass exchange."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None, None
    hi, hj = h[i], h[j]
    delta_min = max(-hi, -(1.0 - hj))
    delta_max = min(1.0 - hi, hj)
    low = max(delta_min, -step)
    high = min(delta_max, step)
    if low >= high:
        return None, None
    delta = rng.random() * (high - low) + low
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    new_h = _repair_sum(new_h, target)
    new_c5, _ = _compute_c5(new_h)
    return new_h, new_c5

def _targeted_move(h, step, target, rng):
    """Move mass between a pair linked by the worst shift."""
    n = h.shape[0]
    # compute worst shift
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr_raw))
    lag = idx - (n - 1)
    i_low = max(0, -lag)
    i_high = min(n, n - lag)
    if i_high <= i_low:
        return _random_swap(h, step, target, rng)
    i = rng.integers(i_low, i_high)
    j = i + lag
    # direction i -> j
    max_delta_fwd = min(h[i], step, 1.0 - h[j])
    # direction j -> i
    max_delta_rev = min(h[j], step, 1.0 - h[i])
    if max_delta_fwd <= 1e-12 and max_delta_rev <= 1e-12:
        return _random_swap(h, step, target, rng)
    if max_delta_rev > max_delta_fwd:
        delta = rng.random() * max_delta_rev
        new_h = h.copy()
        new_h[i] = h[i] + delta
        new_h[j] = h[j] - delta
    else:
        delta = rng.random() * max_delta_fwd
        new_h = h.copy()
        new_h[i] = h[i] - delta
        new_h[j] = h[j] + delta
    np.clip(new_h, 0.0, 1.0, out=new_h)
    new_h = _repair_sum(new_h, target)
    new_c5, _ = _compute_c5(new_h)
    return new_h, new_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initialise -----
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.shape[0]
        n = 200  # higher resolution for optimisation
        if n != n0:
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, init)
        else:
            h = init.copy()
    except NameError:
        n = 200
        h = np.zeros(n, dtype=float)
        half = n // 2
        h[:half] = 1.0
        if n % 2:
            h[half] = 0.5
        rng.shuffle(h)

    target_sum = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    h = _repair_sum(h, target_sum)

    current = h
    cur_c5, _ = _compute_c5(current)
    best_h = current.copy()
    best_c5 = cur_c5

    max_step = 0.6
    min_step = 0.001
    max_temp = 0.02
    min_temp = 1e-6
    p_target = 0.65  # probability to use targeted move

    # ----- main optimisation loop -----
    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        frac = (now - start) / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        if rng.random() < p_target:
            new_h, new_c5 = _targeted_move(current, step, target_sum, rng)
        else:
            new_h, new_c5 = _random_swap(current, step, target_sum, rng)

        if new_h is None:
            continue

        # acceptance (Metropolis)
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
            current = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = current.copy()

    # final repair and evaluation
    best_h = _repair_sum(best_h, target_sum)
    best_c5, _ = _compute_c5(best_h)
    return best_h, best_c5, n
