def adam_optimize(h0, max_iter, lr=0.05, alpha_start=0.5, alpha_end=30.0, alpha_ramp=0.5):
    # alpha ramp fraction: ramp up to alpha_end during first ramp fraction of iterations
    m = np.zeros_like(h0)
    v = np.zeros_like(h0)
    h = h0.copy()
    n = h.size
    s = n/2.0
    for t in range(1, max_iter+1):
        # compute alpha
        if t <= int(alpha_ramp * max_iter):
            alpha = alpha_start + (alpha_end - alpha_start) * (t-1) / (int(alpha_ramp*max_iter)-1)
        else:
            alpha = alpha_end
        # compute gradient smooth
        grad = compute_grad_smooth(h, alpha)
        # Adam moments
        m = 0.9*m + 0.1*grad
        v = 0.999*v + 0.001*(grad**2)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
    final_score = compute_c5(h)
    return h, final_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    # extreme cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # bracket for lambda: f(lambda) = sum clip(v - lambda, lo, hi) is decreasing
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure proper bracketing
    while f(lam_low) < s:
        lam_low -= 1.0
    while f(lam_high) > s:
        lam_high += 1.0

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx on the discrete grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of C_shift = Σ h_i (1‑h_{i+shift}) with respect to h."""
    n = h.size
    grad = np.ones_like(h)
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:]    -= h[:n - shift]
    else:
        s = -shift
        grad[s:]       -= h[:n - s]
        grad[:n - s]   -= h[s:]
    return grad

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1 as float)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):

    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

    """Linear interpolation of arr to length new_n, then project to constraints."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    new_arr = np.interp(x_new, x_old, arr)
    return project_onto_box_sum(new_arr, new_n / 2.0)

def adam_max_shift(h0, n_iter=4000, lr=0.07, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer using gradient of the currently worst shift."""
    n = h0.size
    s = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

    return best_h, final_score, best_h.size
