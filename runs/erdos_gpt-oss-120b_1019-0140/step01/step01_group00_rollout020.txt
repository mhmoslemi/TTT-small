def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def hill_climb_random(start_h, start_c5, start_time, budget_s):
    n = len(start_h)
    dx = 2.0 / n
    best_h = start_h.copy()
    best_val = start_c5
    rng = np.random.default_rng()
    attempts_without_improvement = 0
    max_no_improve = 1000   # adapt.
    while time.time() - start_time < budget_s - 0.1:
        # random swap of one 1 and one 0
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        val = compute_c5_vals(cand)
        if val + 1e-12 < best_val:
            best_h = cand
            best_val = val
            attempts_without_improvement = 0
        else:
            attempts_without_improvement += 1
            if attempts_without_improvement > max_no_improve:
                break
    return best_h, best_val
```

import time
import numpy as np
import cvxpy as cp

def _compute_c5(vec, dx):
    """Return C5 value for binary/float vector."""
    return np.max(np.correlate(vec, 1 - vec, mode="full")) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and possible initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        init_vec = np.ravel(initial_h_values).astype(float)
        init_vec = np.rint(init_vec).astype(int)          # enforce binary
    else:
        n = 80
        init_vec = None

    # ensure even n (required for half‐ones)
    if n % 2:
        n -= 1

    half = n // 2
    dx = 2.0 / n
    target_bound = 0.3808

    best_h = None
    best_c5 = float("inf")

    # ------------------------------------------------------------------
    # 1) Try a mixed‑integer linear programme (MILP) to find optimum
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    milp_time = max(5.0, remaining * 0.75)   # leave time for fallback

    try:
        # binary variables for the step function
        x = cp.Variable(n, boolean=True)

        # auxiliary binary variables for products x_i * x_j (upper triangle)
        y = {}
        cons = [cp.sum(x) == half]

        for i in range(n):
            for j in range(i + 1, n):
                y_ij = cp.Variable(boolean=True)
                y[(i, j)] = y_ij
                cons += [
                    y_ij <= x[i],
                    y_ij <= x[j],
                    y_ij >= x[i] + x[j] - 1,
                ]

        # auxiliary upper bound variable
        t = cp.Variable()
        cons.append(t >= 0)

        # overlap constraints for every non‑zero shift
        for d in range(1, n):
            # sum of x over the overlapping region
            sum_x_left = cp.sum(x[: n - d])
            sum_x_right = cp.sum(x[d:])

            # sum of product variables for this shift
            sum_y = 0
            for i in range(n - d):
                sum_y += y[(i, i + d)]

            cons += [
                sum_x_left - sum_y <= t,
                sum_x_right - sum_y <= t,
            ]

        prob = cp.Problem(cp.Minimize(t), cons)

        # try GLPK_MI first, fall back to ECOS_BB if unavailable
        try:
            prob.solve(solver=cp.GLPK_MI, glpk={"msg_lev": "GLP_MSG_OFF", "tmlim": milp_time})
        except Exception:
            prob.solve(solver=cp.ECOS_BB, max_iters=20000, feastol=1e-6)

        if prob.status in ("optimal", "optimal_incomplete") and x.value is not None:
            h_int = np.rint(x.value).astype(int)

                    h_int[flip] = 0

    except Exception:

        pass

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    elapsed = time.time() - start_time

    if best_h is None or best_c5 > target_bound:
        # heuristic helpers

            return _compute_c5(v, dx)

        # start from a good seed if available

            cur_val = best_c5
