import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Adjust h in-place so that sum(h) == target while staying in [0,1]."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def subgradient_max_overlap(h: np.ndarray) -> np.ndarray:
    """Subgradient of the max overlap functional (without dx scaling)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    max_idxs = np.flatnonzero(corr == max_val)

    g = np.zeros(n)
    for idx in max_idxs:
        k = idx - (n - 1)
        # bias term (derivative of sum_i h_i)
        g += 1.0
        if k >= 0:
            if k != 0:
                g[: n - k] -= h[k:]
                g[k:] -= h[: n - k]
            else:
                g -= 2.0 * h
        else:
            kk = -k
            g[kk:] -= h[: n - kk]
            g[: n - kk] -= h[kk:]
    # project to hyperplane sum(g)=0
    g -= g.mean()
    return g

    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # initialise h and problem size

    except Exception:
        n_points = 200
        target_sum = n_points / 2.0
        h = rng.rand(n_points)
        h = h / h.sum() * target_sum

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # annealing parameters
