import time

def _repair_vector(v, n, max_iter=20):
    # Clip and adjust sum to n/2.
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # Determine free indices that can be adjusted without hitting bounds
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # Final correction if still off:
    s = v.sum()
    if abs(s - target) > 1e-10:
        # Scale entire vector
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            # distribute residual uniformly
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    n = len(v)
    dx = 2.0 / n
    # correlation full mode
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def _upscale_vector(v, n_target):
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):

import time
import numpy as np

def _repair_vector(v, n, max_iter=20):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-10:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * (2.0 / n)

def _upscale_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _gradient_for_shift(v, shift):
    """Subgradient of the correlation at a given integer shift."""
    n = len(v)
    g = np.zeros_like(v)
    if shift >= 0:
        # term1: (1 - v[i+shift]) for i+shift < n
        g[: n - shift] = 1.0 - v[shift:]
        # term2: v[i-shift] for i-shift >= 0
        g[shift:] -= v[: n - shift]
    else:
        s = -shift
        g[s:] = 1.0 - v[: n - s]
        g[: n - s] -= v[s:]
    return g

def _projected_gradient_descent(v_init, time_limit):
    """Subgradient descent with projection onto the feasible box+hyperplane."""
    start = time.time()
    if time_limit <= 0:
        return v_init.copy(), _compute_c5(v_init)
    n = len(v_init)
    best_v = v_init.copy()
    best_score = _compute_c5(best_v)
    v = v_init.copy()
    step = 0.5
    no_improve = 0
    while time.time() - start < time_limit:
        corr = np.correlate(v, 1.0 - v, mode='full')
        max_val = corr.max()
        max_idxs = np.flatnonzero(corr == max_val)
        grad = np.zeros_like(v)
        for idx in max_idxs:
            shift = idx - (n - 1)
            grad += _gradient_for_shift(v, shift)
        grad /= len(max_idxs)

        v_new = v - step * grad
        v_new = _repair_vector(v_new, n)
        new_score = _compute_c5(v_new)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_v = v_new.copy()
            v = v_new
            step = min(step * 1.02, 1.0)
            no_improve = 0
        else:
            # Simulated‑annealing style acceptance
            elapsed = time.time() - start
            remaining = time_limit - elapsed
            temp = max(0.01 * (remaining / time_limit), 1e-4)
            if np.random.rand() < np.exp(-(new_score - best_score) / temp):
                v = v_new
            else:

                no_improve += 1
                if no_improve >= 200:

        # occasional jitter
        if np.random.rand() < 0.001:
            v = _repair_vector(v + (np.random.rand(n) - 0.5) * 0.1, n)
    return best_v, best_score

def _local_swap_search(v_init, time_limit):
    """Hill‑climbing by exchanging small amounts between two points."""
    start = time.time()
    n = len(v_init)
    v = v_init.copy()
    best_v = v.copy()
    best_score = _compute_c5(v)
    step = 0.05
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)

        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score = score
            best_v = cand.copy()
            v = cand
    return best_v, best_score

        v_opt, sc_opt = _projected_gradient_descent(init_guess, slice_time * 0.8)

        if sc_opt < best_score:
            best_score, best_v, best_n = sc_opt, v_opt, n

        # Follow‑up local‑search refinement
        extra = slice_time * 0.2
        if extra > 0.1:
            v_ls, sc_ls = _local_swap_search(best_v, extra)
            if sc_ls < best_score:
                best_score, best_v = sc_ls, v_ls

        remaining_res -= 1

    # ------------------------------------------------------------------ #
    # Final short random‑restart phase if time permits
    while time.time() - start < budget_s - 0.4:
        remaining = budget_s - (time.time() - start) - 0.4
        if remaining <= 0.2:
            break
        # Allocate a small chunk (≤5 s) for each attempt
        attempt_time = min(remaining * 0.6, 5.0)
        rnd = _random_feasible(best_n)
        v_opt, sc_opt = _projected_gradient_descent(rnd, attempt_time)
        if sc_opt < best_score:
            best_score, best_v = sc_opt, v_opt
        # Tiny local refinement
        extra = min(0.2, remaining * 0.2)
        if extra > 0.05:
            v_ls, sc_ls = _local_swap_search(best_v, extra)
            if sc_ls < best_score:
                best_score, best_v = sc_ls, v_ls

    # ------------------------------------------------------------------ #
    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(best_n)
