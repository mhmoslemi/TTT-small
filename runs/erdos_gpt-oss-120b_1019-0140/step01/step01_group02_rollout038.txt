import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """
    Adjust a single entry to enforce the exact sum constraint (sum(h) == target).
    The function modifies h in‑place.
    """
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0.0:
        # need to increase some entry that is not already at 1
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        # need to decrease some entry that is not already at 0
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute evenly (should rarely happen)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h

    Search for a step function h on [0,2] (discretized) that lowers the

        c5_bound: the achieved C5 value for h_values
        n_points: number of discretisation points (int)
    """

    n = 200                     # moderate size (<1k) for richer search space
