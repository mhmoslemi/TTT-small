def crossover(p1, p2, rng):
    # p1, p2 are 1D numpy arrays of bool or uint8
    child = p1.copy()
    mask = rng.random(p1.shape) < 0.5
    child[mask] = p2[mask]
    # Now adjust to target sum
    target = len(p1)//2
    cur = child.sum()
    if cur > target:
        # need to reduce ones: select excess ones randomly and set to 0
        excess = cur - target
        ones_idx = np.flatnonzero(child)
        remove_idx = rng.choice(ones_idx, excess, replace=False)
        child[remove_idx] = 0
    elif cur < target:
        deficit = target - cur
        zeros_idx = np.flatnonzero(child==0)
        add_idx = rng.choice(zeros_idx, deficit, replace=False)
        child[add_idx] = 1
    return child
```

import time, numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def random_binary_vector(n, rng):
    vec = np.zeros(n, dtype=np.uint8)
    vec[:n//2] = 1
    rng.shuffle(vec)
    return vec

def tournament_select(pop, scores, rng, tsize=5):
    idx = rng.choice(len(pop), tsize, replace=False)
    best = idx[np.argmin(scores[idx])]
    return pop[best]

def crossover(p1, p2, target_sum, rng):
    child = p1.copy()
    mask = rng.random(p1.shape) < 0.5
    child[mask] = p2[mask]
    cur = int(child.sum())
    if cur > target_sum:
        excess = cur - target_sum
        ones_idx = np.flatnonzero(child)
        rem = rng.choice(ones_idx, excess, replace=False)
        child[rem] = 0
    elif cur < target_sum:
        deficit = target_sum - cur
        zeros_idx = np.flatnonzero(child == 0)
        add = rng.choice(zeros_idx, deficit, replace=False)
        child[add] = 1
    return child

def mutate_swap(ind, rng):
    ones = np.flatnonzero(ind)
    zeros = np.flatnonzero(ind == 0)
    if len(ones) == 0 or len(zeros) == 0:
        return ind
    i = rng.choice(ones)
    j = rng.choice(zeros)
    ind[i], ind[j] = 0, 1
    return ind

def hill_climb(start_h, start_c5, start_time, budget_s, rng):
    h = start_h.copy()
    best_c5 = start_c5
    n = len(h)
    target_sum = n // 2
    # Precompute ones/zeros indices quickly update each iteration? Could recompute each time.
    while time.time() - start_time < budget_s - 0.2:
        # maybe random swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0
        h[j] = 1
        c5 = compute_c5(h)
        if c5 + 1e-12 < best_c5:
            best_c5 = c5
        else:
            # revert
            h[i] = 1
            h[j] = 0
        # early stop if below target
        if best_c5 < 0.3808:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0_full = np.array(initial_h_values, dtype=float)
        n = len(h0_full)
        h0_bin = (h0_full > 0.5).astype(np.uint8)
    else:
        n = 80
        h0_bin = None

    # Optionally increase resolution
    # try to increase n to 120 if original n<120 for better bound
    # but we should keep simple.
    # We'll keep n as is.

    target_sum = n // 2

    pop_size = 60
    # Ensure at least 2 individuals
    pop = []
    scores = []
    # Add initial if binary and sum correct
    if h0_bin is not None and h0_bin.sum() == target_sum:
        pop.append(h0_bin)
        scores.append(compute_c5(h0_bin))
    # Fill rest
    while len(pop) < pop_size:
        vec = random_binary_vector(n, rng)
        pop.append(vec)
        scores.append(compute_c5(vec))
    pop = np.array(pop, dtype=np.uint8)
    scores = np.array(scores, dtype=float)

    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_c5 = scores[best_idx]

    # GA loop
    mutation_prob = 0.3
    tournament_size = 5
    while time.time() - start < budget_s - 1.0:
        # generate offspring
        offspring = []
        offspring_scores = []
        for _ in range(pop_size):
            p1 = tournament_select(pop, scores, rng, tournament_size)
            p2 = tournament_select(pop, scores, rng, tournament_size)
            child = crossover(p1, p2, target_sum, rng)
            if rng.random() < mutation_prob:
                child = mutate_swap(child, rng)
            c5 = compute_c5(child)
            offspring.append(child)
            offspring_scores.append(c5)
        # combine and select next generation
        combined = np.concatenate((pop, np.array(offspring, dtype=np.uint8)), axis=0)
        combined_scores = np.concatenate((scores, np.array(offspring_scores, dtype=float)), axis=0)
        # select best pop_size
        idx_sort = np.argsort(combined_scores)[:pop_size]
        pop = combined[idx_sort]
        scores = combined_scores[idx_sort]
        # update best
        if scores[0] < best_c5 - 1e-12:
            best_c5 = scores[0]
            best_h = pop[0].copy()
        # early break if lower than target
        if best_c5 < 0.3808:
            break

    # local hill climbing from best solution
    if time.time() - start < budget_s - 0.5:
        best_h, best_c5 = hill_climb(best_h, best_c5, start, budget_s, rng)

    # if we still haven't used initial_h_values with continuous values maybe we can produce float array conversion
    h_float = best_h.astype(float)
    return h_float, float(best_c5), n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap for binary (or float) vector h on [0,2]."""
    hf = h.astype(float)
    n = hf.size
    dx = 2.0 / n
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    return float(corr.max() * dx)

def random_binary_vector(n, rng):
    v = np.zeros(n, dtype=np.uint8)
    v[: n // 2] = 1
    rng.shuffle(v)
    return v

def tournament_select(pop, scores, rng, tsize=5):
    idx = rng.choice(pop.shape[0], tsize, replace=False)
    best = idx[np.argmin(scores[idx])]
    return pop[best]

def crossover(p1, p2, target_sum, rng):
    child = p1.copy()
    mask = rng.random(p1.shape) < 0.5
    child[mask] = p2[mask]
    cur = int(child.sum())
    if cur > target_sum:
        excess = cur - target_sum
        ones = np.flatnonzero(child)
        child[rng.choice(ones, excess, replace=False)] = 0
    elif cur < target_sum:
        deficit = target_sum - cur
        zeros = np.flatnonzero(child == 0)
        child[rng.choice(zeros, deficit, replace=False)] = 1
    return child

def mutate_swap(ind, rng):
    ones = np.flatnonzero(ind)
    zeros = np.flatnonzero(ind == 0)
    if ones.size == 0 or zeros.size == 0:
        return ind
    i = rng.choice(ones)
    j = rng.choice(zeros)
    ind[i] = 0
    ind[j] = 1
    return ind

def hill_climb(start_h, start_c5, start_time, budget_s, rng):
    h = start_h.copy()
    best = start_c5
    n = h.size
    target = n // 2
    while time.time() - start_time < budget_s - 0.2:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        val = compute_c5(h)
        if val + 1e-12 < best:
            best = val
        else:
            h[i], h[j] = 1, 0  # revert
        if best < 0.3808:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine dimension and possible initial binary vector
    if 'initial_h_values' in globals():
        h0_full = np.array(initial_h_values, dtype=float)
        n = h0_full.size
        h0_bin = (h0_full > 0.5).astype(np.uint8)
    else:
        n = 80
        h0_bin = None

    target_sum = n // 2

    # Fix initial binary sum if needed
    if h0_bin is not None and h0_bin.sum() != target_sum:
        diff = int(h0_bin.sum() - target_sum)
        if diff > 0:
            ones = np.flatnonzero(h0_bin)
            h0_bin[rng.choice(ones, diff, replace=False)] = 0
        elif diff < 0:
            zeros = np.flatnonzero(h0_bin == 0)
            h0_bin[rng.choice(zeros, -diff, replace=False)] = 1

    pop_size = 60
    # Build initial population
    pop = []
    scores = []
    if h0_bin is not None:
        pop.append(h0_bin)
        scores.append(compute_c5(h0_bin))
    while len(pop) < pop_size:
        vec = random_binary_vector(n, rng)
        pop.append(vec)
        scores.append(compute_c5(vec))
    pop = np.array(pop, dtype=np.uint8)
    scores = np.array(scores, dtype=float)

    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_c5 = float(scores[best_idx])

    mutation_prob = 0.3
    tournament_size = 5

    # Genetic algorithm loop
    while time.time() - start < budget_s - 1.0:
        offspring = []
        off_scores = []
        for _ in range(pop_size):
            p1 = tournament_select(pop, scores, rng, tournament_size)
            p2 = tournament_select(pop, scores, rng, tournament_size)
            child = crossover(p1, p2, target_sum, rng)
            if rng.random() < mutation_prob:
                child = mutate_swap(child, rng)
            val = compute_c5(child)
            offspring.append(child)
            off_scores.append(val)
        combined = np.concatenate((pop, np.array(offspring, dtype=np.uint8)), axis=0)
        combined_scores = np.concatenate((scores, np.array(off_scores, dtype=float)), axis=0)
