        if elapsed >= time_limit:

        # linear cooling schedule for step
        fraction = elapsed / time_limit

import time
import numpy as np

def _compute_c5(v):
    """Return the C5 bound for a vector v (step function values)."""
    n = v.shape[0]
    dx = 2.0 / n
    # full correlation of v with (1-v)
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _repair_vector(v, n, max_iter=15):
    """Project v onto the feasible set: 0<=v<=1 and sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if still slightly off
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _binary_random_feasible(n):
    """Return a binary feasible vector (sum = n/2). For odd n the last entry is 0.5."""
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 1.0
    if n % 2 == 1:
        v[half] = 0.5
    np.random.shuffle(v)
    return v

def _simulated_annealing(v0, time_limit):
    """
    Simple simulated annealing that preserves the sum exactly by exchanging mass
    between two coordinates. Returns the best vector found and its C5 value.
    """
    n = v0.shape[0]
    v = v0.copy()
    cur_score = _compute_c5(v)
    best_v = v.copy()
    best_score = cur_score
    start = time.time()
    max_step_initial = 0.5          # maximal exchange magnitude at start
    T0 = 0.02                       # initial temperature
    min_T = 1e-6
    iter_count = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        fraction = elapsed / time_limit
        # linearly shrink the allowed exchange step
        max_step = max_step_initial * (1.0 - fraction) + 1e-5
        # temperature schedule (linear decay)
        T = max(min_T, T0 * (1.0 - fraction))

        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        vi, vj = v[i], v[j]

        # compute feasible delta range (exchange amount)
        delta_max = min(max_step, 1.0 - vi, vj)      # positive delta
        delta_min = -min(max_step, vi, 1.0 - vj)    # negative delta
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue

        v_new = v.copy()
        v_new[i] = vi + delta
        v_new[j] = vj - delta

        new_score = _compute_c5(v_new)

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / T):
            v = v_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_v = v_new.copy()

        iter_count += 1
        # occasional random restart to avoid long stagnation
        if iter_count % 5000 == 0 and np.random.rand() < 0.05:
            v = _binary_random_feasible(n)
            cur_score = _compute_c5(v)

    return best_v, best_score

def _local_refine(v0, time_limit):
    """
    Fine-grained hill climbing with diminishing step size.
    Only exchanges respecting the box constraints are performed, thus the sum stays exact.
    """
    n = v0.shape[0]
    v = v0.copy()
    best_v = v.copy()
    best_score = _compute_c5(v)
    cur_score = best_score
    start = time.time()
    step_initial = 0.05

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        fraction = elapsed / time_limit
        step = step_initial * (1.0 - fraction) + 1e-6

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        vi, vj = v[i], v[j]

        delta_max = min(step, 1.0 - vi, vj)
        delta_min = -min(step, vi, 1.0 - vj)
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue

        cand = v.copy()
        cand[i] = vi + delta
        cand[j] = vj - delta
        cand_score = _compute_c5(cand)

        if cand_score < cur_score:
            v = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_v = cand.copy()
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # Acquire parent construction if provided.
    try:
        parent = np.array(initial_h_values, dtype=np.float64)  # may be pre‑imported
        n = parent.shape[0]
    except Exception:
        n = 180
        parent = _binary_random_feasible(n)

    parent = _repair_vector(parent, n)
    best_v = parent.copy()
    best_score = _compute_c5(best_v)

    # Reserve a small margin for final safety operations.
    total_budget = max(0.0, budget_s - 0.5)

    # Allocate most time to simulated annealing, remainder to fine refinement.
    sa_budget = total_budget * 0.85
    refine_budget = total_budget - sa_budget

    # Simulated annealing phase
    v_sa, score_sa = _simulated_annealing(best_v, sa_budget)
    if score_sa < best_score:
        best_v, best_score = v_sa, score_sa

    # Local refinement phase (if any time left)
    if refine_budget > 0.1:
        v_ref, score_ref = _local_refine(best_v, refine_budget)
        if score_ref < best_score:
            best_v, best_score = v_ref, score_ref

    # Final repair (should already be feasible) and exact evaluation
    best_v = _repair_vector(best_v, n)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(n)
