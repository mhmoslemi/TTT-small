def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) < tol:
            break
        # indices not at bounds: interior
        interior = (h > 0.0) & (h < 1.0)
        if np.count_nonzero(interior) == 0:
            # no interior, break
            break
        delta = diff / np.count_nonzero(interior)
        h[interior] += delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
    # final correction if still off:
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # adjust a random interior or any index
        diff = target_sum - cur
        # find any index we can adjust within bounds
        # for simplicity, adjust the first element within bounds
        # compute allowable increase/decrease
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
            if abs(diff) < 1e-9:
                break
    return h
```

            h = project_to_feasible(h, target_sum)

    return best_h, best_val

def subgradient(h, k):
    n = h.shape[0]
    idx = np.arange(n)
    idx_plus = idx + k
    term1 = np.ones(n)
    mask1 = (idx_plus >= 0) & (idx_plus < n)

    term2 = np.zeros(n)

    term2[mask2] = h[idx_minus[mask2]]

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        interior = (h > 0.0) & (h < 1.0)
        interior_idx = np.where(interior)[0]
        if interior_idx.size == 0:
            break
        delta = diff / interior_idx.size
        h[interior_idx] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        # distribute remaining diff to first non-bound element
        diff = target_sum - cur
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
            if abs(diff) < 1e-9:
                break
    return h

def resample_h(base_h, n_target):
    base_h = np.asarray(base_h, dtype=float)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    """Project onto [0,1]^n intersect {sum = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        interior = (h > 0.0) & (h < 1.0)
        idx = np.where(interior)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final fix
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        diff = target_sum - cur
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
            if abs(diff) < 1e-9:
                break
    return h


def resample_h(base_h, n_target):
    """Interpolate base_h to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        new_h = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_target = np.linspace(0.0, 2.0, n_target, endpoint=False)
        new_h = np.interp(x_target, x_base, base_h)
    return project_to_feasible(new_h, n_target / 2.0)


def subgradient_shift(h, k):
    """Subgradient of the active lag-k term."""
    n = h.shape[0]
    idx = np.arange(n)
    idx_plus = idx + k
    term1 = np.ones(n)
    mask1 = (idx_plus >= 0) & (idx_plus < n)
    if mask1.any():
        term1[mask1] = 1.0 - h[idx_plus[mask1]]
    idx_minus = idx - k
    term2 = np.zeros(n)
    mask2 = (idx_minus >= 0) & (idx_minus < n)
    if mask2.any():
        term2[mask2] = h[idx_minus[mask2]]
    return term1 - term2


def gradient_descent(h0, target_sum, max_iters, start_time, deadline, rng):
    """Projected gradient descent on the max‑lag objective."""
    h = h0.copy()

    step0 = 0.6
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx] * (2.0 / n)
        if max_val < best_val:
            best_val = max_val
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = subgradient_shift(h, k)
        step = step0 / np.sqrt(it + 1)
        h = h - step * grad
        if rng.random() < 0.01:  # occasional jitter
            h += rng.normal(scale=0.02, size=n)
        h = project_to_feasible(h, target_sum)
    final_val = compute_c5(best_h)
    return best_h, final_val
