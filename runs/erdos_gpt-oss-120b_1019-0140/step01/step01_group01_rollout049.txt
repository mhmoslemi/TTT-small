        # temperature for acceptance: T = T0 * (0.5 ** (iter / 200)) maybe T0=0.01
        T = initial_T * (0.5 ** (iter / 500))

        # generate candidate via gradient move
        # 0.5 probability use gradient, else pairwise
        if rng.rand() < 0.5:
            # gradient move
            # compute correlation to find worst shift
            corr_full = np.correlate(h, 1.0 - h, mode='full')
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (len(h)-1)
            # gradient
            g = np.zeros_like(h)
            idx = np.arange(len(h))
            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < len(h))
            g[valid_plus] += (1.0 - h[plus[valid_plus]])
            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < len(h))
            g[valid_minus] -= h[minus[valid_minus]]
            # projection to sum=0 hyperplane: subtract mean
            g -= g.mean()
            candidate = h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        else:
            # pairwise swap move
            i, j = rng.randint(len(h), size=2)
            if i != j:
                hi, hj = h[i], h[j]
                # compute feasible delta range: delta added to i, sub from j
                low = max(-hi, hj - 1.0)
                high = min(1.0 - hi, hj)
                if low < high:
                    delta = (rng.rand() * 2 - 1) * step  # between -step and step
                    if delta < low:
                        delta = low
                    elif delta > high:
                        delta = high
                    candidate = h.copy()
                    candidate[i] = hi + delta
                    candidate[j] = hj - delta
                else:
                    candidate = h.copy()
            else:
                candidate = h.copy()
        cand_c5 = compute_c5(candidate)
        # acceptance
        if cand_c5 < current_c5:
            h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_local_c5:

def fix_sum(v, target_sum, rng):
    v = np.clip(v, 0, 1, out=v)
    diff = target_sum - v.sum()
    if abs(diff) < 1e-12: return v
    # While diff not zero:
    # Determine eligible indices where we can increase (if diff>0) or decrease (if diff<0)
    if diff > 0:
        # need to increase
        eligible = np.where(v < 1)[0]
    else:
        eligible = np.where(v > 0)[0]
    # If eligible empty, we can't adjust; return v.
    # Distribute diff uniformly among eligible, but cap at bounds.
    while abs(diff) > 1e-12 and eligible.size > 0:
        inc = diff / eligible.size
        # apply inc respecting bounds
        # For each idx, compute max possible increase (or decrease)
        if diff > 0:
            max_inc = 1 - v[eligible]
            actual_inc = np.minimum(inc, max_inc)
        else:
            max_dec = -v[eligible]  # negative
            actual_inc = np.maximum(inc, max_dec)  # inc negative
        v[eligible] += actual_inc
        diff = target_sum - v.sum()
        # Update eligible for remaining diff
        if diff > 0:
            eligible = np.where(v < 1)[0]
        else:
            eligible = np.where(v > 0)[0]
    return v

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """Scale/adjust h in-place so that sum(h) == target while 0<=h<=1."""
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h
