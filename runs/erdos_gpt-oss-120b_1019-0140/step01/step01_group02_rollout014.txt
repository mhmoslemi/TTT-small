def compute_c5_fft(h):
    n = h.shape[0]
    v = 1.0 - h
    N = 2*n - 1
    H = np.fft.rfft(h, n=N)
    V = np.fft.rfft(v[::-1], n=N)
    conv = np.fft.irfft(H * V, n=N)
    max_val = conv.max()
    return float(max_val * (2.0 / n))

def propose_delta_move(h, rng, max_delta):
    i, j = rng.integers(0, n, size=2)
    if i == j: return None
    hi, hj = h[i], h[j]
    delta_min = max(-hi, -(1.0 - hj))
    delta_max = min(1.0 - hi, hj)
    # intersection with ±max_delta
    low = max(delta_min, -max_delta)
    high = min(delta_max, max_delta)
    if low >= high: return None
    delta = rng.random() * (high - low) + low
    return i, j, delta

    n = h.shape[0]

import time
import math
import numpy as np

def _compute_c5_fft(h: np.ndarray) -> float:
    """Fast evaluation of C5 using FFT (max correlation * dx)."""
    n = h.shape[0]
    v = 1.0 - h
    N = 2 * n - 1
    # convolution of h with reversed v
    H = np.fft.rfft(h, n=N)
    V = np.fft.rfft(v[::-1], n=N)
    conv = np.fft.irfft(H * V, n=N)
    max_val = np.max(conv)
    return float(max_val * (2.0 / n))

def _compute_c5_exact(h: np.ndarray) -> float:
    """Exact (np.correlate) evaluation of C5, used for the final answer."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Choose a coordinate that can move in the needed direction
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute uniformly (should rarely happen)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    # If clipping prevented full correction, recurse
    if not math.isclose(h[i] - (h[i] - diff), diff, abs_tol=1e-12):
        _repair_sum(h, target)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h on [0,2] that minimizes the Erdős C₅ bound.

    Returns:
        h_values: 1‑D numpy array of length n_points, values in [0,1]
        c5_bound: the achieved C₅ value for h_values
        n_points: number of discretisation points (int)
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ----------------------------------------------------------------------
    # 1)  Possible pre‑supplied construction (length 47)
    # ----------------------------------------------------------------------
    best_h = None
    best_c5 = float('inf')
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.shape[0]
        target0 = n0 / 2.0
        init = np.clip(init, 0.0, 1.0)
        init = _repair_sum(init, target0)
        c0 = _compute_c5_exact(init)
        best_h = init.copy()
        best_c5 = c0
    except NameError:
        pass  # no initial_h_values supplied

    # ----------------------------------------------------------------------
    # 2)  Main optimisation on a larger, even‑sized grid (n = 200)
    # ----------------------------------------------------------------------
    n = 200
    target = n / 2.0
    # start from a random binary vector with exactly n/2 ones
    cur = np.zeros(n, dtype=float)
    ones_idx = rng.choice(n, size=n // 2, replace=False)
    cur[ones_idx] = 1.0
    cur_c5 = _compute_c5_fft(cur)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = cur.copy()

    # annealing schedule parameters
    max_step_initial = 0.5
    min_step_final = 0.001
    temp_initial = 0.02
    temp_final = 1e-7

    # ----------------------------------------------------------------------
    # 3)  Simulated annealing loop
    # ----------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break
        # progress fraction
        frac = (now - start) / budget_s
        # linearly decay step magnitude and temperature
        step = max_step_initial * (1.0 - frac) + min_step_final * frac
        temp = temp_initial * (1.0 - frac) + temp_final * frac

        # decide move type: 30% swaps, 70% delta transfers
        if rng.random() < 0.3:
            # swap a 1 and a 0 (binary‑preserving)
            idx_one = np.where(cur > 0.5)[0]
            idx_zero = np.where(cur < 0.5)[0]
            if idx_one.size == 0 or idx_zero.size == 0:
                continue
            i = rng.choice(idx_one)
            j = rng.choice(idx_zero)
            cur[i], cur[j] = cur[j], cur[i]
            changed = ('swap', i, j)
        else:
            # delta transfer between two coordinates
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            hi, hj = cur[i], cur[j]
            # feasible interval keeping entries inside [0,1]
            dmin = max(-hi, -(1.0 - hj))
            dmax = min(1.0 - hi, hj)
            low = max(dmin, -step)
            high = min(dmax, step)
            if low >= high:
                continue
            delta = rng.random() * (high - low) + low
            cur[i] = hi + delta
            cur[j] = hj - delta
            changed = ('delta', i, j, hi, hj)

        # evaluate the new configuration
        new_c5 = _compute_c5_fft(cur)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur.copy()
        else:
            # revert the move
            if changed[0] == 'swap':
                i, j = changed[1], changed[2]
                cur[i], cur[j] = cur[j], cur[i]
            else:
                i, j, hi, hj = changed[1], changed[2], changed[3], changed[4]
                cur[i] = hi
                cur[j] = hj

        # occasional early exit if we already beat the target bound
        if best_c5 <= 0.3808:
            # give a small buffer to finish cleanly
            if now - start > 5.0:
                break

    # ----------------------------------------------------------------------
    # 4)  Final repair & exact C5 evaluation
    # ----------------------------------------------------------------------
    if best_h is None:

        best_h = np.zeros(2, dtype=float)
        best_h[0] = 1.0
    # ensure exact sum (tiny drift possible)
    best_h = _repair_sum(best_h, best_h.shape[0] / 2.0)
    best_c5 = _compute_c5_exact(best_h)

    return best_h, best_c5, best_h.shape[0]
