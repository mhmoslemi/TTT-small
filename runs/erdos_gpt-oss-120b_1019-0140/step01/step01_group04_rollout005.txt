def anneal(v, time_limit, max_temp=0.02, min_temp=0.0001):
    start = time.time()
    n = len(v)

    # aim to iterate until time limit
    while time.time() - start < time_limit:
        # compute temperature based on elapsed
        elapsed = time.time() - start
        # linear schedule
        T = max_temp * (1 - elapsed / time_limit)
        T = max(T, min_temp)
        # propose move
        i = np.random.randint(n)
        # find i with v[i] > 0, else repeat few times
        # we can simply select i and then j ensure appropriate.

        while attempts < 10:
            i = np.random.randint(n)
            if v[i] > 1e-12:
                break
            attempts += 1
        if attempts >= 10:
            continue
        attempts = 0
        while attempts < 10:
            j = np.random.randint(n)
            if v[j] < 1 - 1e-12 and j != i:
                break
            attempts += 1
        if attempts >= 10:

        new_v = v.copy()
        new_v[i] -= delta
        new_v[j] += delta
        # compute new_score
        new_score = _compute_c5(new_v)
        # acceptance condition
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / T):

            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_v = new_v.copy()
    return best_v, best_score

import time
import numpy as np

def _repair_vector(v, n, max_iter=15):
    """Project vector v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if still slightly off
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.linspace(0.0, 2.0, n_src, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _anneal(v, time_limit):
    """Simulated annealing with sum‑preserving transfers."""
    n = len(v)
    cur_v = v.copy()
    cur_score = _compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    start = time.time()
    max_temp = 0.02
    min_temp = 0.0001
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = max_temp * (1 - frac) + min_temp * frac
        # pick donor i (v[i] > 0) and receiver j (v[j] < 1)
        i = np.random.randint(n)
        for _ in range(5):
            if cur_v[i] > 1e-12:
                break
            i = np.random.randint(n)
        j = np.random.randint(n)
        for _ in range(5):
            if cur_v[j] < 1.0 - 1e-12 and j != i:
                break
            j = np.random.randint(n)
        cap = min(cur_v[i], 1.0 - cur_v[j])
        if cap <= 0:
            continue
        # step size scales with temperature
        scale = (T / max_temp)
        delta = cap * (0.1 + 0.9 * np.random.rand() * scale)
        new_v = cur_v.copy()
        new_v[i] -= delta
        new_v[j] += delta
        new_score = _compute_c5(new_v)
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            cur_v = new_v
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_v = new_v.copy()
    return best_v, best_score

def _optimize_at_resolution(n, init_v, time_limit):
    """Run annealing from init_v and a few random restarts within time_limit."""
    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)
    # primary anneal
    half = time_limit * 0.5
    v_opt, sc_opt = _anneal(init_v, half)
    if sc_opt < best_score:
        best_v, best_score = v_opt, sc_opt
    # random restarts in remaining time
    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        if remaining < 0.5:
            break
        rnd = _random_feasible(n)
        # allocate a modest chunk (no more than 5 s) for each restart
        chunk = min(5.0, remaining)
        v_rnd, sc_rnd = _anneal(rnd, chunk)
        if sc_rnd < best_score:
            best_v, best_score = v_rnd, sc_rnd
    return best_v, best_score
