import time
import math
import numpy as np

def project_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto the set {x: lo<=x_i<=hi, sum x_i = target_sum}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # bounds for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        clipped = np.clip(v - lam_mid, lo, hi)
        s = clipped.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return clipped

def compute_c5(h):
    """Exact discrete C5 for a vector h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of corr[shift] = sum_i h_i (1 - h_{i+shift}) w.r.t. h."""
    n = h.size
    grad = np.ones(n, dtype=float)
    if shift >= 0:
        if shift != 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            # shift == 0 => subtract twice
            grad -= 2.0 * h
    else:
        s = -shift
        grad[s:] -= h[:n - s]
        grad[:n - s] -= h[s:]
    return grad

def smooth_pg_opt(h0, deadline, beta0=0.5, beta_max=50.0, step0=0.5, max_iter=5000):
    """Projected gradient descent on a smooth approximation of max correlation."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # temperature schedule (exponential increase of beta)
        beta = beta0 * (beta_max / beta0) ** (it / max_iter)
        max_corr = np.max(corr)
        exp_vals = np.exp(beta * (corr - max_corr))
        weights = exp_vals / np.sum(exp_vals)  # sum to 1
        # gradient of smooth max
        grad = np.ones(n, dtype=float)  # constant term (sum of weights = 1)
        n_shift = n - 1
        # iterate over all shifts
        for idx_shift, shift in enumerate(range(-n_shift, n_shift + 1)):
            w = weights[idx_shift]
            if w == 0.0:
                continue
            if shift >= 0:
                if shift != 0:
                    grad[:n - shift] -= w * h[shift:]
                    grad[shift:] -= w * h[:n - shift]
                else:
                    grad -= 2.0 * w * h
            else:
                s = -shift
                grad[s:] -= w * h[:n - s]
                grad[:n - s] -= w * h[s:]
        eta = step0 / math.sqrt(it)
        h = h - eta * grad
        h = project_box_sum(h, n / 2.0)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def projected_gradient_descent(h0, deadline, step0=0.5, max_iter=5000):
    """Classic projected subgradient descent focusing on the active shift."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h = h - eta * grad
        h = project_box_sum(h, n / 2.0)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def thue_morse_seq(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5  # small safety margin

    # Determine problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h_init = np.asarray(initial_h_values, dtype=float)
        h_init = project_box_sum(h_init, n / 2.0)
    else:
        n = 256
        h_init = None

    candidates = []

    # 1) Use parent construction if available
    if h_init is not None:
        candidates.append(h_init)

    # 2) Thue–Morse (if length is a power of two)
    if (n & (n - 1)) == 0:
        tm = thue_morse_seq(n)
        diff = int(tm.sum() - n / 2)
        if diff != 0:
            # adjust to exact half‑ones
            ones = np.where(tm == 1)[0]
            zeros = np.where(tm == 0)[0]
            if diff > 0:
                idx = rng.choice(ones, size=diff, replace=False)
                tm[idx] = 0.0
            else:
                idx = rng.choice(zeros, size=-diff, replace=False)
                tm[idx] = 1.0
        candidates.append(tm)

    # 3) Random balanced binary vectors
    for _ in range(5):
        idx = rng.choice(n, size=n // 2, replace=False)
        hb = np.zeros(n, dtype=float)
        hb[idx] = 1.0
        candidates.append(hb)

    # 4) Random fractional vectors (projected)
    for _ in range(5):
        v = rng.random(n)

    best_h = None
    best_score = float('inf')

    # Phase A: smooth PGD on each candidate (up to ~30 s each)
    for h0 in candidates:

        h_opt, score_opt = smooth_pg_opt(h0, cand_deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    while time.time() < overall_deadline:

        if best_h[i] < 1.0 and best_h[j] > 0.0:

                h_new[i] += delta

                if score < best_score:
                    best_h = h_new
                    best_score = score

        if best_h[j] < 1.0 and best_h[i] > 0.0:
            delta = min(0.01, 1.0 - best_h[j], best_h[i])

    # Final feasibility sanity projection
