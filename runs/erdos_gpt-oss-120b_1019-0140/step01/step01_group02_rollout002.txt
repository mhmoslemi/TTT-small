    best_local = h.copy()

    # schedule
    start_local = time.time()
    while time.time() - start_local < budget:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * (2.0 / n))
        # Optionally compute idx_max to guide deterministic move
        idx_max = np.argmax(corr)
        shift_k = idx_max - (n - 1)  # shift relative to zero
        # Determine step size and temp based on elapsed fraction
        elapsed = time.time() - start_local
        frac = elapsed / budget
        max_step = 0.5
        min_step = 0.001
        step = max_step * (1.0 - frac) + min_step * frac
        max_temp = 0.01
        min_temp = 1e-6
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # Try deterministic improvement for worst shift
        improved = False
        # compute overlapping region indices
        if shift_k >= 0:
            idxs = np.arange(0, n - shift_k)
        else:
            idxs = np.arange(-shift_k, n)
        partner = h[idxs + shift_k]
        hi = h[idxs]
        # candidate increase
        inc_score = partner * (1.0 - hi)
        if inc_score.max() > 1e-12:
            i_local = idxs[np.argmax(inc_score)]
            # candidate decrease
            # find indices where hi>0
            mask = hi > 1e-12
            if np.any(mask):
                dec_scores = partner[mask] * hi[mask]  # product
                # want minimal partner * hi
                j_local = idxs[mask][np.argmin(dec_scores)]
                # compute delta
                delta = min(step, 1.0 - h[i_local], h[j_local])
                if delta > 1e-12:
                    # apply move
                    h[i_local] += delta
                    h[j_local] -= delta
                    new_c5 = _compute_c5(h)
                    if new_c5 < best_local_c5:
                        best_local_c5 = new_c5
                        best_local = h.copy()
                    # accept if improves or with probability
                    if new_c5 < c5 or rng.random() < math.exp(-(new_c5 - c5) / temp):
                        # keep move
                        c5 = new_c5
                        improved = True
                    else:
                        # revert
                        h[i_local] -= delta
                        h[j_local] += delta

        if not improved:
            # random pair move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            hi = h[i]
            hj = h[j]
            delta_min = max(-hi, -(1.0 - hj))

            h[j] = hj - delta
            new_c5 = _compute_c5(h)

            # acceptance

            else:
                h[i] = hi
                h[j] = hj

    # final repair
    best_local = _repair_sum(best_local, target)

def _repair_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Find indices that can be adjusted
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if idx.size == 0:
            # fallback: distribute diff uniformly
            h += diff / h.size
            np.clip(h, 0.0, 1.0, out=h)
            return h
        i = idx[0]
        add = min(diff, 1.0 - h[i])
        h[i] += add
    else:
        idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            h += diff / h.size
            np.clip(h, 0.0, 1.0, out=h)
            return h
        i = idx[0]
        sub = min(-diff, h[i])
        h[i] -= sub
    # Recurse if still diff remains
    if abs(h.sum() - target) > 1e-12:
        return _repair_sum(h, target)
    return h

        partner = h[idxs + shift]
        hi = h[idxs]
        inc_score = partner * (1.0 - hi)
        if inc_score.max() > 1e-12:
            i = idxs[int(np.argmax(inc_score))]
            mask = hi > 1e-12
            if mask.any():
                dec_score = partner[mask] * hi[mask]
                j = idxs[mask][int(np.argmin(dec_score))]
                delta = min(step, 1.0 - h[i], h[j])
                if delta > 1e-12:
                    h[i] += delta
                    h[j] -= delta
                    new_c5 = _compute_c5(h)
                    if new_c5 < best_c5:
                        best_c5, best_h = new_c5, h.copy()
                    if new_c5 < c5 or rng.random() < math.exp(-(new_c5 - c5) / temp):
                        continue
                    h[i] -= delta
                    h[j] += delta

        # random two‑index move
        i, j = rng.integers(0, n, size=2)
        if i != j:
            hi, hj = h[i], h[j]
            dmin = max(-hi, -(1.0 - hj))
            dmax = min(1.0 - hi, hj)
            low, high = max(dmin, -step), min(dmax, step)
            if low < high:
                delta = rng.random() * (high - low) + low
                h[i], h[j] = hi + delta, hj - delta
                new_c5 = _compute_c5(h)
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, h.copy()
                if new_c5 < c5 or rng.random() < math.exp(-(new_c5 - c5) / temp):
                    continue
                h[i], h[j] = hi, hj

        # occasional restart
        if rng.random() < 0.001:
            h = np.zeros(n)
            half = n // 2
            h[:half] = 1.0
            if n % 2:
                h[half] = 0.5
            rng.shuffle(h)
            h = _repair_sum(h, target)

    best_h = _repair_sum(best_h, target)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()
    best_h, best_c5, best_n = None, float('inf'), None

    # Phase 1: use provided construction if any
    try:
        init = np.asarray(initial_h_values, dtype=float)
        init = np.clip(init, 0.0, 1.0)
        n0 = init.shape[0]
        budget0 = budget_s * 0.45
        h0, c5_0 = _optimize_h(init, budget0, rng)
        best_h, best_c5, best_n = h0, c5_0, n0
    except NameError:
        pass

    # Phase 2: larger search
    elapsed = time.time() - start_total
    left = budget_s - elapsed
    if left > 5.0:
        # try a few different sizes
        sizes = [120, 200, 300]
        for n in sizes:
            if time.time() - start_total >= budget_s:
                break
            target = n / 2.0
            h = np.zeros(n)
            half = n // 2
            h[:half] = 1.0
            if n % 2:
                h[half] = 0.5
            rng.shuffle(h)
            h = _repair_sum(h, target)
            sub_budget = min(left * 0.3, 30.0)
            h_opt, c5_opt = _optimize_h(h, sub_budget, rng)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
            left = budget_s - (time.time() - start_total)

    # final safeguard
    if best_h is None:
        # fallback simple half‑ones pattern
        n = 200
        h = np.zeros(n)
        half = n // 2
        h[:half] = 1.0
        if n % 2:
            h[half] = 0.5
        rng.shuffle(h)
        best_h = _repair_sum(h, n / 2.0)
        best_c5 = _compute_c5(best_h)
        best_n = n

    return best_h, best_c5, int(best_n)
