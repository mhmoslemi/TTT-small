def local_search(h, max_steps, init_temp, final_temp, rng):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for t in range(max_steps):
        # linear annealing schedule
        temp = init_temp + (final_temp - init_temp) * t / max_steps
        # propose a move
        i, j = rng.choice(n, size=2, replace=False)
        # delta determined by random normal scaled by temp?
        delta = (rng.random() - 0.5) * 0.1 * temp  # step magnitude scaling
        # attempt to shift delta from i to j while staying within [0,1]
        new_h = best_h.copy()
        # Ensure we keep sum, we modify i by -delta, j by +delta.
        # Ensure bound constraints:
        new_i = new_h[i] - delta
        new_j = new_h[j] + delta
        if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
            continue
        new_h[i] = new_i
        new_h[j] = new_j
        # evaluate

import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # binary search for lambda
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of correlation term for a given integer shift."""
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        # left part of gradient
        grad[:n - shift] = 1.0 - h[shift:]
        # right part
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] = 1.0 - h[:n - s]
        grad[:n - s] -= h[s:]
    return grad

def compute_gradient_max_shift(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    return compute_gradient(h, shift)

def upsample(arr, new_n):
    """Linear interpolation of arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (values 0.0 or 1.0)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                      dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def half_block_pattern(n):
    h = np.zeros(n, dtype=float)
    h[n // 2:] = 1.0
    return h

def alternating_pattern(n):
    h = np.zeros(n, dtype=float)
    h[1::2] = 1.0
    return h

def pgd_momentum(h0, n_iter, step0=0.5, beta=0.9,
                 time_limit=None, start_time=None):
    """Projected gradient descent with momentum on max-correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, n_iter + 1):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        # gradient from currently active shift
        grad = compute_gradient_max_shift(h)
        # momentum update
        v = beta * v + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * v
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # occasional tiny random shake to escape flat regions
        if it % 5000 == 0:
            shake = (np.random.random(n) - 0.5) * 0.01
            h = project_onto_box_sum(h + shake, s)
    return best_h, best_score

    best_score = float('inf')

    # Optimize each base candidate
    for h_start in candidates:
        if time.time() - start > budget_s - 2.0:
            break
        # allocate iterations (more for higher resolution)
        iters = 12000
        remaining = budget_s - (time.time() - start) - 1.0
        h_opt, score_opt = pgd_momentum(h_start, n_iter=iters,
                                        step0=0.6, beta=0.85,
                                        time_limit=remaining,
                                        start_time=start)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
        if best_score <= TARGET_BOUND:
            break

    # Random restarts until budget exhausted
    while time.time() - start < budget_s - 1.5:
        v = rng.random(target_n)
        h0 = project_onto_box_sum(v, target_n / 2.0)
        iters = 8000
        remaining = budget_s - (time.time() - start) - 0.5
        h_opt, score_opt = pgd_momentum(h0, n_iter=iters,
                                        step0=0.6, beta=0.85,
                                        time_limit=remaining,
                                        start_time=start)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
        if best_score <= TARGET_BOUND:
            break

    # Fallback if nothing found
    if best_h is None:
        best_h = np.full(target_n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, target_n / 2.0)
        best_score = compute_c5(best_h)
    else:
        # final feasibility projection
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
