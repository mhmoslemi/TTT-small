import time, math
import numpy as np

# ---------- Helper utilities ----------

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx using discretisation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient_shift(h, shift):
    """Gradient of the active term for a given integer shift."""
    n = h.size
    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2


def projected_gradient_descent(h0, n_iter=2000, step0=0.5, target_sum=None):
    """Classic PGD using the gradient of the currently active shift."""
    h = h0.copy()
    n = h.size
    if target_sum is None:
        target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient_shift(h, shift)
        step = step0 / math.sqrt(it)
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def thue_morse(n):
    """Return the Thue–Morse binary sequence of length n (values 0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


# ---------- Softmax based descent ----------

def softmax_descent(
    h0,
    n_iters,
    step0=0.5,
    beta_start=1.0,
    beta_end=30.0,
    momentum=0.9,
    target_sum=None,
    patience=200,
):
    """Projected gradient descent using a softmax weighted gradient."""
    h = h0.copy()
    n = h.size
    if target_sum is None:
        target_sum = n / 2.0
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0

    for t in range(1, n_iters + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)

        # annealed softmax weights
        beta = beta_start + (beta_end - beta_start) * (t - 1) / n_iters
        w = np.exp(beta * (corr - max_corr))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # weighted gradient via two convolutions
        conv1 = np.convolve(h, w, mode='full')
        term2 = conv1[n - 1 : n - 1 + n]               # h shifted left
        conv2 = np.convolve(h, w[::-1], mode='full')
        term1 = conv2[n - 1 : n - 1 + n]               # h shifted right
        grad = 1.0 - term1 - term2

        step = step0 / math.sqrt(t)
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_onto_box_sum(h_new, target_sum)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break
        h = h_new
    return best_h, best_score


# ---------- Main entry point ----------
