Now SA algorithm:

```
def simulated_annealing(h0, dx, time_limit, seed):
    rng = np.random.default_rng(seed)
    current = h0.copy()
    current_val = compute_c5(current, dx)
    best = current.copy()
    best_val = current_val
    start = time.time()
    # temperature schedule parameters
    T = 0.01
    # set number of iterations based on time left and steps.
    while time.time() - start < time_limit:
        # pick random i where current[i]==1, random j where current[j]==0
        ones = np.flatnonzero(current)
        zeros = np.flatnonzero(1 - current) # as current entries are 0/1
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new = current.copy()
        new[i] = 0
        new[j] = 1
        new_val = compute_c5(new, dx)
        d = new_val - current_val
        if d < 0 or rng.random() < np.exp(-d / T):
            current = new
            current_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val
        # cooling schedule
        T *= 0.99995  # adjust factor for slow cooling
        if T < 1e-7:
            T = 1e-7
    return best, best_val
```

            break

    # ----- enforce exact constraints (clip and sum) -----
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = np.sum(best_h)
    if abs(cur_sum - target_sum) > 1e-9:
        # deterministic binary projection to exact sum
        idx_desc = np.argsort(-best_h)
        binary_fixed = np.zeros(n, dtype=float)
        binary_fixed[idx_desc[:n // 2]] = 1.0
        best_h = binary_fixed

    final_c5 = compute_c5(best_h, dx)
    return best_h, final_c5, n
