def _improve_step(h, rng):

    N = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    max_idx = np.argmax(corr)
    s = max_idx - (N - 1)
    # Compute D_i
    D = np.zeros(N, dtype=np.float64)
    # Use slices
    if s >= 0:
        D[:N-s] += h[s:]
        D[s:] += h[:N-s]
        # Overlap region: for i in [s, N-s-1] D[i] = h[i-s] + h[i+s]; for edges only one term
    else:
        t = -s
        D[t:] += h[:N-t]
        D[:N-t] += h[t:]
    # Choose i_up, i_down
    # Candidate max index among allowed (h[i] < 1)
    allowed_up = np.where(h < 1 - 1e-12)[0]
    if allowed_up.size == 0:
        return h, False
    # Among allowed, find i_up with max D
    i_up = allowed_up[np.argmax(D[allowed_up])]
    # For down: h > 0
    allowed_down = np.where(h > 1e-12)[0]
    if allowed_down.size == 0:
        return h, False
    i_down = allowed_down[np.argmin(D[allowed_down])]
    if i_up == i_down:
        # set aside, try random alternative
        # pick random up/down
        i_up = rng.choice(allowed_up)
        i_down = rng.choice(allowed_down)
        if i_up == i_down:
            return h, False
    # Determine epsilon. To keep steps moderate, limit max change
    max_eps = 0.1  # choose
    eps = min(1.0 - h[i_up], h[i_down], max_eps)
    if eps <= 0:
        return h, False
    h_new = h.copy()
    h_new[i_up] += eps
    h_new[i_down] -= eps
    # Ensure vector repair
    _repair_vector(h_new) # adjust as needed (though sum should stay correct).
    return h_new, True

import numpy as np
import time
import math

def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(v) == len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    target = v.size / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / v.size
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v


def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _guided_anneal(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Guided simulated annealing using the worst shift's gradient information."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    T0 = 0.01  # initial temperature
    max_move_abs = 0.2  # maximal epsilon per move
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = T0 * frac

        # worst shift
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (N - 1)  # shift where overlap is maximal

        # gradient for increasing product sum of that shift
        D = np.zeros(N, dtype=np.float64)
        if s >= 0:
            D[:N - s] += h_cur[s:]
            D[s:] += h_cur[:N - s]
        else:
            t = -s
            D[t:] += h_cur[:N - t]
            D[:N - t] += h_cur[t:]

        # choose source (decrease) and sink (increase) indices
        up_mask = h_cur < 1.0 - 1e-12
        down_mask = h_cur > 1e-12
        up_candidates = np.where(up_mask)[0]
        down_candidates = np.where(down_mask)[0]
        if up_candidates.size == 0 or down_candidates.size == 0:
            break

        i_up = up_candidates[np.argmax(D[up_candidates])]
        i_down = down_candidates[np.argmin(D[down_candidates])]
        if i_up == i_down:
            # fallback to random distinct pair
            i_up = rng.choice(up_candidates)
            i_down = rng.choice(down_candidates)
            if i_up == i_down:
                continue

        cap_up = 1.0 - h_cur[i_up]
        cap_down = h_cur[i_down]
        move_cap = min(cap_up, cap_down, max_move_abs)
        if move_cap <= 0.0:
            continue
        eps = rng.random() * move_cap  # uniform in [0, move_cap]

        h_new = h_cur.copy()
        h_new[i_up] += eps
        h_new[i_down] -= eps

        new_score = _c5_value(h_new)
        delta = new_score - cur_score
        if delta < 0.0 or (T > 0.0 and rng.random() < math.exp(-delta / (T + 1e-15))):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score


def _greedy_optimize(h_start: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Pure greedy hill‑climbing using the worst shift's derivative."""
    h_cur = h_start.copy()
    cur_score = _c5_value(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    N = h_cur.size
    start = time.time()
    max_move_abs = 0.05
    no_improve_steps = 0
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        # worst shift
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (N - 1)

        # gradient
        D = np.zeros(N, dtype=np.float64)
        if s >= 0:
            D[:N - s] += h_cur[s:]
            D[s:] += h_cur[:N - s]
        else:
            t = -s
            D[t:] += h_cur[:N - t]
            D[:N - t] += h_cur[t:]

        up_mask = h_cur < 1.0 - 1e-12
        down_mask = h_cur > 1e-12
        up_candidates = np.where(up_mask)[0]
        down_candidates = np.where(down_mask)[0]
        if up_candidates.size == 0 or down_candidates.size == 0:
            break

        i_up = up_candidates[np.argmax(D[up_candidates])]
        i_down = down_candidates[np.argmin(D[down_candidates])]
        if i_up == i_down:
            i_up = rng.choice(up_candidates)
            i_down = rng.choice(down_candidates)
            if i_up == i_down:
                break

        cap_up = 1.0 - h_cur[i_up]
        cap_down = h_cur[i_down]
        move_cap = min(cap_up, cap_down, max_move_abs)
        if move_cap <= 0.0:
            break
        eps = move_cap  # greedy: use full allowed move

        h_new = h_cur.copy()
        h_new[i_up] += eps
        h_new[i_down] -= eps

        new_score = _c5_value(h_new)
        if new_score < cur_score - 1e-12:
            h_cur = h_new
            cur_score = new_score
            no_improve_steps = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # revert and count stall
            no_improve_steps += 1
            if no_improve_steps >= 200:
                break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # ------------------------------------------------------------------
    # Initialise the base vector (use parent if present)
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        n0 = 200
        h0 = rng.random(n0)
        h0 *= (n0 / 2.0) / h0.sum()
    else:
        n0 = h0.size
    h0 = _repair_vector(h0)

    best_h, best_score = h0, _c5_value(h0)

    # ------------------------------------------------------------------
    # Phase 1 – guided annealing at the original resolution
    phase1_budget = budget_s * 0.55
    best_h, best_score = _guided_anneal(best_h, phase1_budget, rng)

    # ------------------------------------------------------------------
    # Phase 2 – occasional greedy refinements (multiple short restarts)
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    greedy_time_per = 2.0
    while remaining > greedy_time_per * 1.5:
        h_rand = rng.random(n0)
        h_rand *= (n0 / 2.0) / h_rand.sum()
        h_rand = _repair_vector(h_rand)
        h_opt, s_opt = _greedy_optimize(h_rand, greedy_time_per, rng)
        if s_opt < best_score:
            best_h, best_score = h_opt, s_opt
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed

    # ------------------------------------------------------------------
    # Phase 3 – increase resolution if time permits
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.15:
        n_up = min(best_h.size * 2, 800)
        if n_up != best_h.size:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = _repair_vector(h_up)
            # guided anneal on the finer grid
            h_up, s_up = _guided_anneal(h_up, remaining * 0.9, rng)
            if s_up < best_score:
                best_h, best_score = h_up, s_up

    # ------------------------------------------------------------------
    # Final repair and exact evaluation
    best_h = _repair_vector(best_h)
    best_score = _c5_value(best_h)

    return best_h, float(best_score), best_h.size
