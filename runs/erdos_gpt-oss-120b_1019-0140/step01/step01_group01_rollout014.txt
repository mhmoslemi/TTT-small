import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust array h so that its sum equals target while keeping 0 <= h_i <= 1.
    Small residuals are distributed uniformly and clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _gradient_worst_shift(h: np.ndarray, k_shift: int) -> np.ndarray:
    """
    Compute gradient of the worst overlap term wrt h.
    Gradient of f_k = Σ h_i (1 - h_{i+k}) is g_j = 1 - h_{j+k} - h_{j-k},
    where out-of-range indices contribute zero.
    """
    n = h.shape[0]
    g = np.ones_like(h)
    if k_shift >= 0:
        k = k_shift
        if k < n:
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
    else:
        k = -k_shift
        if k < n:
            g[k:] -= h[:n - k]
            g[:n - k] -= h[k:]
    return g

            i, j = rng.randint(n_points, size=2)
            if i != j:
                hi, hj = current_h[i], current_h[j]
                low = max(-hi, hj - 1.0)
                high = min(1.0 - hi, hj)
                if low < high:
                    delta = (rng.rand() * 2.0 - 1.0) * step
                    delta = np.clip(delta, low, high)
                    candidate = current_h.copy()
                    candidate[i] = hi + delta
                    candidate[j] = hj - delta
                else:
                    candidate = current_h.copy()
            else:
                candidate = current_h.copy()

        cand_c5 = compute_c5(candidate)

        if cand_c5 < current_c5:
            accept = True
        else:
            # Metropolis accept
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()

    # final projection to ensure exact mass
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n_points)
