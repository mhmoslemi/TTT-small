import time
import numpy as np

def compute_c5(h):
    """Return C5 for binary vector h (values in [0,1])."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _random_binary(rng, n, target):
    """Generate random binary vector of length n with exactly target ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, target, replace=False)
    h[idx] = 1.0
    return h

def _repair_sum(vec, target):
    """Adjust binary vector to have exactly target ones."""
    cur = int(np.sum(vec))
    diff = target - cur
    if diff > 0:
        zero_idx = np.flatnonzero(vec == 0)
        sel = np.random.choice(zero_idx, diff, replace=False)
        vec[sel] = 1.0
    elif diff < 0:
        one_idx = np.flatnonzero(vec == 1)
        sel = np.random.choice(one_idx, -diff, replace=False)
        vec[sel] = 0.0
    return vec

def run(seed=42, budget_s=1100, **kwargs):

    # --- initialise
    if 'initial_h_values' in globals():

        n = init.shape[0]
        target = n // 2
        order = np.argsort(-init)[:target]
        best_h = np.zeros(n, dtype=float)
        best_h[order] = 1.0
    else:
        n = 80
        target = n // 2
        best_h = _random_binary(rng, n, target)

    best_c5 = compute_c5(best_h)

    # --- population for GA
    pop_size = 60
    tournament = 5
    mutation_prob = 0.2
    max_mut_swaps = 3

    fitness = np.array([compute_c5(h) for h in population])

    fitness = fitness[order]
