import time, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry (or a few) to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Choose an index that can move in the needed direction.
    if diff > 0:
        cand = np.where(h < 1.0 - 1e-12)[0]
    else:
        cand = np.where(h > 1e-12)[0]
    if cand.size == 0:  # fallback: distribute uniformly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = cand[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h
