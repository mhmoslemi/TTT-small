def project_to_sum_and_bounds(v, target, tol=1e-9, max_iter=50):
    # Ensure feasible: target must be between sum of min possible (0) and sum of max possible (n)
    # Actually each w_i ∈ [0,1], so possible sum range [0, n]. target is within [0, n].
    # We'll binary search lambda in [-1, 1] maybe larger. Since v_i can be out of [0,1].
    # For large lambda negative (i.e., we add something), f reduces? Wait we subtract lambda.
    # Let lambda be shift: w_i = clip(v_i - lambda, 0, 1). As lambda -> -inf, v_i - lambda -> +inf, so w_i = 1, sum = n.
    # As lambda -> +inf, w_i = 0, sum = 0.
    # So f decreasing monotonic with lambda.
    # We'll find lambda s.t. sum = target.

    lo = -max(v) - 1.0  # ensure v_i - lo is huge positive => w_i=1, sum=n.
    hi = max(v) + 1.0   # ensure v_i - hi is huge negative => w_i=0, sum=0.

    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) < tol:
            return w
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - ((lo+hi)/2), 0.0, 1.0)

def project_to_sum_and_bounds(v, target, tol=1e-10, max_iter=100):
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - ((low + high) / 2.0), 0.0, 1.0)

def project_to_sum_and_bounds(v, target, tol=1e-12, max_iter=60):

def solve_milp_binary(n, rng):
    import cvxpy as cp
    x = cp.Variable(n, boolean=True)
    # target sum
    target = n // 2
    constraints = [cp.sum(x) == target]
    # list of overlap expressions per shift
    overlap_vars = []
    w_vars = []  # maybe not needed separate list for each shift
    for k in range(1, n):
        num = n - k
        w = cp.Variable(num, boolean=True)
        xi = x[:num]  # x_i for i=0..n-k-1
        xj = x[k:]   # x_{i+k} correspond
        constraints.append(w <= xi)
        constraints.append(w <= xj)
        constraints.append(w >= xi + xj - 1)
        overlap_vars.append(cp.sum(w))
        w_vars.append(w)
    t = cp.Variable()
    constraints.append(t >= 0)
    for ov in overlap_vars:
        constraints.append(t <= ov)
    prob = cp.Problem(cp.Maximize(t), constraints)
    # try solvers
    solvers = [cp.CBC, cp.GLPK_MI, cp.SCIP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False, warm_start=True, max_iters=1000000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
                break
        except Exception:
            continue
    # after solve, get x
    if x.value is None:
        # fallback random solution
        return rng.rand(n)
    xv = np.array(x.value).flatten()
    # round to 0/1
    xv = np.rint(xv)  # round to nearest integer
    # enforce sum = target via adjusting: if sum deviates, adjust randomly
    diff = int(target - xv.sum())
    if diff > 0:
        zeros = np.where(xv == 0)[0]
        if zeros.size > 0:
            idx = rng.choice(zeros, size=min(diff, zeros.size), replace=False)
            xv[idx] = 1
    elif diff < 0:
        ones = np.where(xv == 1)[0]
        if ones.size > 0:
            idx = rng.choice(ones, size=min(-diff, ones.size), replace=False)
            xv[idx] = 0
    return xv.astype(float)

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_to_sum_and_bounds(v: np.ndarray, target: float,
                              tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """
    Project vector v onto the intersection of the hyperplane sum(v)=target
    and the box [0,1]^n. Uses a simple bisection on the additive shift.
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - ((lo + hi) / 2.0), 0.0, 1.0)

def solve_milp_binary(n: int, rng: np.random.RandomState):
    """
    Solve a binary integer program to maximize the minimum overlap
    Σ_i x_i x_{i+k} over all shifts k=1..n-1, with Σ_i x_i = n/2.
    Returns a binary vector (as float) of length n.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    # Boolean vector of length n
    x = cp.Variable(n, boolean=True)
    target = n // 2
    constraints = [cp.sum(x) == target]

    overlap_exprs = []
    for k in range(1, n):
        m = n - k
        w = cp.Variable(m, boolean=True)
        xi = x[:m]          # x_i for i=0..n-k-1
        xj = x[k:]          # x_{i+k}
        constraints += [w <= xi, w <= xj, w >= xi + xj - 1]
        overlap_exprs.append(cp.sum(w))

    t = cp.Variable()
    constraints += [t >= 0]
    for ov in overlap_exprs:
        constraints.append(t <= ov)

    prob = cp.Problem(cp.Maximize(t), constraints)

    # Try a few MILP solvers
    solvers = [cp.CBC, cp.GLPK_MI, cp.SCIP] if hasattr(cp, "CBC") else [cp.GLPK_MI, cp.SCIP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False, warm_start=True)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE):
                break
        except Exception:
            continue

    if x.value is None:
        return None

    xv = np.rint(np.array(x.value, dtype=float).flatten())
    # enforce exact sum (might be off by a small integer due to tolerance)
    diff = int(target - xv.sum())
    if diff > 0:
        zeros = np.where(xv == 0)[0]
        if zeros.size:
            idx = rng.choice(zeros, size=min(diff, zeros.size), replace=False)
            xv[idx] = 1
    elif diff < 0:
        ones = np.where(xv == 1)[0]
        if ones.size:
            idx = rng.choice(ones, size=min(-diff, ones.size), replace=False)
            xv[idx] = 0
    return xv.astype(float)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] with integral 1 to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # Discretization size (must be even)
    n_points = 200
    target_sum = n_points / 2.0

    # ----- build initial h -----
    init_h = None
    if "initial_h_values" in globals():
        try:
            raw = np.array(initial_h_values, dtype=float)
            if raw.size != n_points:
                # linear interpolation to desired size
                x_old = np.linspace(0.0, 2.0, raw.size, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
                init_h = np.interp(x_new, x_old, raw)
            else:
                init_h = raw.copy()
        except Exception:
            init_h = None

    if init_h is None:
        # Try a small exact MILP solution and upsample it if time permits
        base_n = 70 if 70 <= n_points else (n_points - (n_points % 2))
        if (time.time() - start) < (max_time - 30):
            binary_base = solve_milp_binary(base_n, rng)
            if binary_base is not None:
                if binary_base.size != n_points:
                    x_old = np.linspace(0.0, 2.0, binary_base.size, endpoint=False)
                    x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
                    init_h = np.interp(x_new, x_old, binary_base)
                else:
                    init_h = binary_base.copy()
        if init_h is None:
            # fallback random start
            init_h = rng.rand(n_points)

    # Ensure feasibility
    init_h = np.clip(init_h, 0.0, 1.0)
    init_h = project_to_sum_and_bounds(init_h, target_sum)

    current_h = init_h.copy()
    best_h = init_h.copy()
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5

    # Pre‑allocate index array for gradient computation
    idx_arr = np.arange(n_points)

    # Scheduling parameters
    T0, T_end = 1e-3, 1e-7
    step0, step_end = 0.2, 1e-5
    grad_prob = 0.4   # probability of performing a gradient move

    while True:
        now = time.time()
        if now - start >= max_time:
            break

        progress = (now - start) / max_time
        # exponential cooling / step decay
        T = T0 * (T_end / T0) ** progress
        step = step0 * (step_end / step0) ** progress

        if rng.rand() < grad_prob:
            # ----- gradient move targeting the worst k‑shifts -----
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            # take top 3 worst shifts to smooth the direction
            top_m = 3
            if corr_full.size > top_m:
                top_idxs = np.argpartition(corr_full, -top_m)[-top_m:]
            else:
                top_idxs = np.arange(corr_full.size)
            g = np.zeros_like(current_h)
            for idx in top_idxs:
                k = int(idx) - (n_points - 1)
                # contribution from h_i * (1 - h_{i+k})
                plus = idx_arr + k
                valid_plus = (plus >= 0) & (plus < n_points)
                g[idx_arr[valid_plus]] += (1.0 - current_h[plus[valid_plus]])
                # contribution from -(h_{i-k}) part
                minus = idx_arr - k
                valid_minus = (minus >= 0) & (minus < n_points)
                g[idx_arr[valid_minus]] -= current_h[minus[valid_minus]]
            # Gradient descent step (move opposite to gradient of worst Ck)
            cand = current_h - step * g
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_to_sum_and_bounds(cand, target_sum)
        else:
            # ----- pairwise exchange move -----
            i, j = rng.randint(n_points, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = step * (rng.rand() * 2.0 - 1.0)  # in [-step, step]
            delta = np.clip(delta, low, high)
            cand = current_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta

        cand_c5 = compute_c5(cand)

        # ----- acceptance -----
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = math.exp(-(cand_c5 - current_c5) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = cand
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

        # occasional small random perturbation of the best solution
        if rng.rand() < 0.001:
            noise = rng.normal(scale=step, size=n_points)
            pert = best_h + noise
            pert = np.clip(pert, 0.0, 1.0)
            pert = project_to_sum_and_bounds(pert, target_sum)
            pert_c5 = compute_c5(pert)
            if pert_c5 < best_c5:
                best_h = pert
                best_c5 = pert_c5
                current_h = pert
                current_c5 = pert_c5

    # final projection to guarantee feasibility
    best_h = project_to_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
