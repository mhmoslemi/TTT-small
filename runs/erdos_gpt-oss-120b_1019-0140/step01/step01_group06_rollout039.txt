def compute_c5_fft(h):
    n = h.shape[0]
    g = 1.0 - h
    m = 2 * n - 1
    # rev_g = g[::-1]
    # compute rfft of h and rev_g
    fh = np.fft.rfft(h, m)
    frg = np.fft.rfft(g[::-1], m)
    conv = np.fft.irfft(fh * frg, m)
    # max correlation
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)
```

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=50):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = np.sum(h)
        diff = target_sum - cur_sum
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # distribute diff proportionally among all entries, clamped
            # Add diff uniformly to all entries (could exceed bounds), then clip again
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction if still off
    cur_sum = np.sum(h)
    if abs(cur_sum - target_sum) > 1e-6:
        # adjust the entry with most capacity
        if cur_sum < target_sum:
            # need to increase some entry
            cap = 1.0 - h
            i = np.argmax(cap)
            h[i] = min(1.0, h[i] + (target_sum - cur_sum))
        else:
            # need to decrease some entry
            cap = h
            i = np.argmax(cap)
            h[i] = max(0.0, h[i] - (cur_sum - target_sum))
    return h

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Fast computation of C5 using FFT for full linear correlation."""
    n = h.shape[0]
    m = 2 * n - 1
    # correlation of h with (1-h) via convolution with reversed second argument
    g_rev = (1.0 - h)[::-1]
    fh = np.fft.rfft(h, n=m)
    fg = np.fft.rfft(g_rev, n=m)
    corr = np.fft.irfft(fh * fg, n=m)
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-9, max_iter=50):
    """Project h onto [0,1]^n ∩ {sum = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break

        if free.size == 0:
            # spread diff uniformly then re‑clip
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        if cur < target_sum:
            cap = 1.0 - h
            i = np.argmax(cap)
            h[i] = min(1.0, h[i] + (target_sum - cur))
        else:
            cap = h
            i = np.argmax(cap)
            h[i] = max(0.0, h[i] - (cur - target_sum))
    return h


def resample_h(base_h, n_target):
    """Linear interpolation of base_h to n_target points with feasibility projection."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    return project_to_feasible(h_new, n_target / 2.0)


def sum_constraint(h, target):
    """Equality constraint: sum(h) = target."""
    return np.sum(h) - target


def objective(h):
    """Objective for optimizer: C5."""
    return compute_c5(h)


def init_lowfreq(n, rng, freq_frac=0.25):
    """Initialize with a random low‑frequency spectrum."""
    spec = np.zeros(n // 2 + 1, dtype=complex)
    m = max(1, int(freq_frac * (n // 2 + 1)))
    real = rng.normal(size=m)
    imag = rng.normal(size=m)
    spec[:m] = real + 1j * imag
    spec[0] = rng.normal()
    h = np.fft.irfft(spec, n=n)
    h = np.clip(h, 0.0, 1.0)
    return project_to_feasible(h, n / 2.0)


def init_sine_mix(n, rng, num_modes=5):
    """Initialize as a sum of a few sinusoids."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for _ in range(num_modes):
        freq = rng.uniform(0.5, 3.0) * np.pi
        phase = rng.uniform(0.0, 2 * np.pi)
        amp = rng.uniform(0.0, 0.5)
        h += amp * np.sin(freq * x + phase)
    h -= h.min()
    if h.max() > 0:
        h /= h.max()
    return project_to_feasible(h, n / 2.0)


def subgradient_descent(h0, max_iter, step0, target_sum, rng, early_target=None):
    """Projected sub‑gradient descent on the max‑correlation objective."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = 2 * n - 1
    for t in range(max_iter):
        # correlation via FFT
        g_rev = (1.0 - h)[::-1]
        fh = np.fft.rfft(h, n=m)
        fg = np.fft.rfft(g_rev, n=m)
        corr = np.fft.irfft(fh * fg, n=m)
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        c5 = max_corr * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            if early_target is not None and best_c5 <= early_target:
                break
        lag = max_idx - (n - 1)
        a = abs(lag)
        # gradient of the active lag term
        grad = np.ones(n)
        if a > 0:
            grad[: n - a] -= h[a:]
            grad[a:] -= h[: n - a]
        step = step0 / np.sqrt(t + 1.0)
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_to_feasible(h, target_sum)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low C5 bound."""
    start = time.time()
    rng = np.random.default_rng(seed)

    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)

    best_h = None
    best_c5 = np.inf
    best_n = None
    target_bound = 0.3808

    # Small to medium resolutions – larger n gives finer control.
    candidate_ns = [190, 380, 760]

    for n in candidate_ns:
        if time.time() - start > budget_s * 0.94:
            break
        target_sum = n / 2.0
        restart = 0
        # generous cap; loop will stop by time‑budget
        while restart < 2000 and (time.time() - start) < budget_s * 0.94:
            # ----- initialization -------------------------------------------------
            if base_h is not None and n == base_h.shape[0] and restart == 0:
                h0 = resample_h(base_h, n)
            else:
                method = rng.choice(["uniform", "lowfreq", "sine"])
                if method == "uniform":
                    h0 = rng.random(n)
                    h0 = project_to_feasible(h0, target_sum)
                elif method == "lowfreq":
                    h0 = init_lowfreq(n, rng, freq_frac=0.2)
                else:
                    h0 = init_sine_mix(n, rng, num_modes=4)

            # ----- sub‑gradient descent -----------------------------------------
            step0 = 0.8
            max_iter = 500
            h_opt, c5_val = subgradient_descent(
                h0, max_iter, step0, target_sum, rng, early_target=target_bound
            )

            # ----- local refinement with SLSQP ---------------------------------
            cons = {"type": "eq", "fun": sum_constraint, "args": (target_sum,)}
            bnds = [(0.0, 1.0)] * n
            try:
                res = opt.minimize(
                    fun=objective,
                    x0=h_opt,
                    method="SLSQP",
                    bounds=bnds,
                    constraints=cons,
                    options={"maxiter": 300, "ftol": 1e-9, "disp": False},
                )
                h_ref = np.clip(res.x, 0.0, 1.0)
                h_ref = project_to_feasible(h_ref, target_sum)
                c5_ref = compute_c5(h_ref)
            except Exception:
                h_ref = h_opt
                c5_ref = c5_val

            # ----- update global best -------------------------------------------
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n

            if best_c5 <= target_bound:
                # continue a few more attempts to possibly improve further
                pass

            restart += 1

    # ----- fallback -----------------------------------------------------------
    if best_h is None:
        if base_h is not None:
            best_h = base_h.copy()
            best_n = best_h.shape[0]
        else:
            best_n = 190
            best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # ----- final safety projection --------------------------------------------
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
