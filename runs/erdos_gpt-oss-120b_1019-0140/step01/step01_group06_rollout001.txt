def binary_simulated_annealing(h0, max_time, start_time, rng):
    h = h0.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    zeros = list(np.where(h == 0)[0])
    ones = list(np.where(h == 1)[0])
    # Temperature schedule parameters
    T0 = 0.01  # could be larger maybe 0.05
    # total iterations approximated by some large number?
    it = 0
    # We'll bound number of iterations by 200000 or by time
    max_iter = 200000
    while True:
        if time.time() - start_time > max_time * 0.97:
            break
        if it >= max_iter:
            break
        # update temperature
        T = T0 * (1 - it / max_iter)
        # propose swap
        if not zeros or not ones:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        # compute new candidate
        h[i] = 1.0
        h[j] = 0.0
        # compute c5
        new_c5 = compute_c5(h)
        # decide acceptance
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            # accept
            cur_c5 = new_c5
            # update best
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            # update zero/one lists
            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)
            ones.append(i)
        else:
            # revert swap
            h[i] = 0.0
            h[j] = 1.0
        it += 1
    return best_h, best_c5

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Overlap bound C5 for sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    """Project onto [0,1]^n ∩ {sum = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h)
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    """Linear interpolation of base_h to length n_target with feasibility projection."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def binary_simulated_annealing(h0, end_time, rng):
    """Binary simulated annealing preserving sum by swapping 0/1 entries."""
    h = h0.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    max_iter = 500000
    T0 = 0.05  # initial temperature
    for it in range(max_iter):
        if time.time() > end_time:
            break
        T = T0 * (1.0 - it / max_iter)
        # propose swap of two opposite bits
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
                accept = True
        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    end_time = start + budget_s * 0.97
    rng = np.random.default_rng(seed)

    base_h = globals().get("initial_h_values")
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # candidate sample sizes (even numbers)
    candidate_ns = [95, 190, 380]

    for n in candidate_ns:
        if time.time() > end_time:
            break
        # number of SA restarts for this n
        restarts = 3
        for r in range(restarts):
            if time.time() > end_time:
                break
            # initialise binary vector
            if base_h is not None and r == 0:
                h_res = resample_h(base_h, n)
                h_bin = (h_res >= 0.5).astype(float)
                cur_ones = int(h_bin.sum())
                target_ones = n // 2
                if cur_ones != target_ones:
                    diff = target_ones - cur_ones
                    zeros = np.where(h_bin == 0)[0]
                    ones = np.where(h_bin == 1)[0]
                    if diff > 0:
                        flip = rng.choice(zeros, size=diff, replace=False)
                        h_bin[flip] = 1.0
                    elif diff < 0:

            best_h = h_up_sa.copy()

        # Local refinement at higher resolution

    best_c5 = compute_c5(best_h)
