def exchange_move(v, step):
    n = len(v)
    i, j = np.random.choice(n, 2, replace=False)
    # compute max possible increase for i
    max_inc_i = min(step, 1 - v[i])
    max_dec_i = min(step, v[i])
    # generate delta in range [-max_dec_i, max_inc_i]
    delta = (np.random.rand() * (max_inc_i + max_dec_i) - max_dec_i)
    # if delta>0, increase i, decrease j; else opposite
    if delta > 0:
        # need to ensure j can decrease by delta
        max_dec_j = v[j]
        if delta > max_dec_j:
            delta = max_dec_j
        v_new = v.copy()
        v_new[i] += delta
        v_new[j] -= delta
    else:
        # delta negative: decrease i, increase j
        delta_abs = -delta
        max_dec_i = v[i]
        max_inc_j = 1 - v[j]
        if delta_abs > max_dec_i:
            delta_abs = max_dec_i
        if delta_abs > max_inc_j:
            delta_abs = max_inc_j
        v_new = v.copy()
        v_new[i] -= delta_abs
        v_new[j] += delta_abs
    return v_new

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v) = n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _optimize_at_resolution(n, init_v, time_limit):
    """Run SLSQP from init_v (with random restarts) respecting time_limit."""
    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)

    def obj(x):
        return _compute_c5(x)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bounds = [(0.0, 1.0)] * n

    # primary run
    try:
        res = minimize(
            obj,
            init_v,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 500, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            v = _repair_vector(res.x, n)
            sc = _compute_c5(v)
            if sc < best_score:
                best_score, best_v = sc, v
    except Exception:
        pass

    # random restarts
    while time.time() - start < time_limit:
        rnd = _random_feasible(n)
        try:
            res = minimize(
                obj,
                rnd,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n)
                sc = _compute_c5(v)
                if sc < best_score:
                    best_score, best_v = sc, v
        except Exception:
            pass
        if time.time() - start > time_limit * 0.9:
            break
    return best_v, best_score

def _targeted_local_search(v, time_limit):
    """Directed hill‑climbing: move mass from i to its partner i+shift where shift gives maximal overlap."""
    start = time.time()
    cur_v = v.copy()
    cur_score = _compute_c5(cur_v)
    best_v = cur_v.copy()
    best_score = cur_score
    n = len(v)
    step = 0.2
    last_improve = start

    while time.time() - start < time_limit:
        corr = np.correlate(cur_v, 1.0 - cur_v, mode="full")
        s_idx = np.argmax(corr)
        shift = s_idx - (n - 1)

        if shift >= 0:
            valid_i = np.arange(0, n - shift, dtype=int)
            partner = valid_i + shift
        else:
            s_pos = -shift
            valid_i = np.arange(s_pos, n, dtype=int)
            partner = valid_i - s_pos

        if valid_i.size == 0:
            i, j = np.random.choice(n, 2, replace=False)
        else:
            prod = cur_v[valid_i] * (1.0 - cur_v[partner])
            idx = np.argmax(prod)
            i = int(valid_i[idx])
            j = int(partner[idx])

        # attempt to move mass from i to j
        max_dec = min(step, cur_v[i])
        max_inc = min(step, 1.0 - cur_v[j])
        if max_dec > 0 and max_inc > 0:
            delta = min(max_dec, max_inc) * np.random.rand()
            cand = cur_v.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # fallback random exchange
            i2, j2 = np.random.choice(n, 2, replace=False)
            max_dec = min(step, cur_v[i2])
            max_inc = min(step, 1.0 - cur_v[j2])
            if max_dec == 0 or max_inc == 0:
                continue
            delta = min(max_dec, max_inc) * np.random.rand()
            cand = cur_v.copy()
            cand[i2] -= delta
            cand[j2] += delta

        new_score = _compute_c5(cand)
        if new_score < cur_score - 1e-12:
            cur_v, cur_score = cand, new_score
            if new_score < best_score:
                best_v, best_score = cand.copy(), new_score
                last_improve = time.time()
        # adapt step size
        if time.time() - last_improve > 5.0:
            step = min(step * 1.02, 0.2)
            last_improve = time.time()
        else:
            step = max(step * 0.9995, 1e-4)

    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function minimizing the Erdős C5 overlap bound."""
    np.random.seed(seed)
    start = time.time()

    # Acquire parent construction if available.
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 190
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0 // 2] = 1.0
    parent = _repair_vector(parent, n0)

    total_budget = max(0.0, budget_s - 2.0)  # keep a tiny safety margin
    base_budget = total_budget * 0.3
    up_budget = total_budget * 0.3
    search_budget = total_budget - base_budget - up_budget

    # Base resolution optimisation.
    v_base, sc_base = _optimize_at_resolution(n0, parent.copy(), base_budget)
    best_v, best_score, best_n = v_base, sc_base, n0

    # Upsample if we can stay below 800 points.
    up_n = n0 * 2
    if up_n <= 800:
        v_up = _upsample_vector(best_v, up_n)
        v_up_opt, sc_up_opt = _optimize_at_resolution(up_n, v_up, up_budget)
        if sc_up_opt < best_score:
            best_v, best_score, best_n = v_up_opt, sc_up_opt, up_n

    # Targeted local search with remaining time.
    elapsed = time.time() - start
    remaining = budget_s - elapsed - 0.5
    if remaining > 0.2:
        v_search, sc_search = _targeted_local_search(best_v, remaining)
        if sc_search < best_score:
            best_v, best_score = v_search, sc_search

    # Final repair and evaluation.
    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(best_n)
