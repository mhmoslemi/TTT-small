def subgradient(h):
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(conv)
    # shift k = max_idx - (n-1)
    k = max_idx - (n-1)
    # Compute gradient: for each i where overlapping:
    # For i in [0,n-1], j = i + k (if 0 <= j < n)
    # f = sum_i h[i] * (1 - h[i+k]) => derivative w.r.t h[i] = (1 - h[i+k]) (if i+k in range)
    # derivative w.r.t h[j] = -h[i] (if j = i+k)
    grad = np.zeros_like(h)
    for i in range(n):
        j = i + k
        if 0 <= j < n:
            grad[i] += 1 - h[j]
            grad[j] -= h[i]
        else:
            # no contribution (since 1-h zero outside)
            pass
    return grad

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def max_corr_and_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    max_val = corr[idx]
    shift = idx - (n - 1)  # positive shift means h shifted right
    return max_val, shift

def gradient_for_shift(h, shift):
    n = len(h)
    grad = np.zeros_like(h)
    if shift >= 0:
        # i from 0 to n-1-shift have i+shift in range
        i = np.arange(0, n-shift)
        j = i + shift
        # contributions
        grad[i] += 1.0 - h[j]
        grad[j] -= h[i]
    else:
        # shift <0, j = i+shift => i+shift >=0 => i >= -shift
        shift_abs = -shift
        i = np.arange(shift_abs, n)
        j = i + shift  # i + (-shift_abs) = i - shift_abs
        grad[i] += 1.0 - h[j]
        grad[j] -= h[i]
    return grad
```

def local_search(h0, max_iter=2000, lr=0.05, decay=0.999, patience=200):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    no_improve = 0

        max_val, shift = max_corr_and_shift(h)

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the Erdős overlap bound C5 for sampled h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum, tol=1e-9, max_iter=100):
    """Project vector onto box [0,1]^n and hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = (h > 0.0) & (h < 1.0)
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        if cur < target_sum:
            i = np.argmin(h)      # most room to increase
        else:
            i = np.argmax(h)      # most room to decrease
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h

def resample_h(base_h, n_target):
    """Linear interpolation of base_h to a new length with feasibility projection."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new

def max_corr_shift(h):
    """Return (max correlation value, corresponding shift index)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return corr[idx], shift

def gradient_for_shift(h, shift):
    """Subgradient of correlation at the active shift."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        hi = h[:n - shift]
        hj = h[shift:]
        grad[:n - shift] += 1.0 - hj
        grad[shift:] -= hi
    else:
        s = -shift
        hi = h[s:]
        hj = h[:n - s]
        grad[s:] += 1.0 - hj
        grad[:n - s] -= hi
    return grad

def random_feasible(n, rng):
    """Generate a random feasible vector."""
    h = rng.random(n)
    return project_to_feasible(h, n / 2.0)

def local_search(h0, start_time, budget_s, max_iter=2000, lr=0.05,
                 decay=0.999, patience=200):
    """Simple projected sub‑gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    no_improve = 0
    cur_lr = lr
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.96:
            break
        _, shift = max_corr_shift(h)
        grad = gradient_for_shift(h, shift)
        h_new = h - cur_lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new, target)
        new_val = compute_c5(h_new)
        if new_val < cur_val - 1e-12:
            h = h_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
        else:
            no_improve += 1
        cur_lr *= decay
        if no_improve >= patience:
            break
    return best_h, best_val

def refine_slsqp(h, n):
    """One‑shot SLSQP refinement starting from h."""
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(lambda x: compute_c5(x), x0=h,
                           method="SLSQP", bounds=bnds,
                           constraints=cons,
                           options={"maxiter": 600, "ftol": 1e-12, "disp": False})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_to_feasible(h_opt, n / 2.0)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h, compute_c5(h)

    Search for a step function h minimizing the Erdős overlap constant C5.

    """

    candidates = [128, 256, n_base]
