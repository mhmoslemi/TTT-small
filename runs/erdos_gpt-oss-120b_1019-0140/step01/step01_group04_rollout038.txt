def _full_corr(v):
    n = len(v)
    w = 1.0 - v
    v_pad = np.concatenate([v, np.zeros(n-1)])
    w_rev_pad = np.concatenate([w[::-1], np.zeros(n-1)])
    N = 1 << ((2*n - 1 - 1).bit_length())  # Next power of 2
    fft_v = np.fft.fft(v_pad, N)
    fft_w = np.fft.fft(w_rev_pad, N)
    conv = np.real(np.fft.ifft(fft_v * fft_w))[:2*n-1]
    return conv[::-1]  # O array

def _full_corr(v):
    n = v.shape[0]
    w = 1.0 - v
    # pad to length 2n-1
    length = 2*n - 1
    N = 1 << (length - 1).bit_length()
    v_pad = np.concatenate([v, np.zeros(N - n, dtype=v.dtype)])  # Actually need length N

def _local_search(v, n, time_limit):
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(v)
    step = 0.1
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2 * step
        cand = best_v.copy()
        cand[i] = np.clip(cand[i] + delta, 0.0, 1.0)
        cand[j] = np.clip(cand[j] - delta, 0.0, 1.0)
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score = score
            best_v = cand
            step = min(step * 1.1, 0.2)  # increase step when improving
        else:
            step = max(step * 0.9, 0.01)  # decrease step otherwise
    return best_v, best_score

import time
import numpy as np
from numpy.fft import fft, ifft

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _repair_vector(v: np.ndarray, n: int, max_iter: int = 15) -> np.ndarray:
    """Project onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _random_feasible(n: int) -> np.ndarray:
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _full_corr(v: np.ndarray) -> np.ndarray:
    """Full correlation of v with (1-v) as used for C5."""
    n = v.shape[0]
    w = 1.0 - v
    length = 2 * n - 1
    N = _next_pow2(length)
    v_pad = np.concatenate([v, np.zeros(N - n, dtype=v.dtype)])
    w_rev = w[::-1]
    w_pad = np.concatenate([w_rev, np.zeros(N - n, dtype=v.dtype)])
    conv = np.real(ifft(fft(v_pad, N) * fft(w_pad, N)))[:length]
    return conv[::-1]  # O(k) for shifts -n+1 … n-1

def _compute_c5(v: np.ndarray) -> float:
    n = len(v)
    dx = 2.0 / n
    corr = _full_corr(v)
    return np.max(corr) * dx

def _gradient_for_shift(v: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the active overlap term for shift k."""
    n = len(v)
    if k > 0:
        term1 = np.concatenate([v[k:], np.zeros(k, dtype=v.dtype)])
        term2 = np.concatenate([np.zeros(k, dtype=v.dtype), v[: n - k]])
    elif k < 0:
        kk = -k
        term1 = np.concatenate([np.zeros(kk, dtype=v.dtype), v[: n - kk]])
        term2 = np.concatenate([v[kk:], np.zeros(kk, dtype=v.dtype)])
    else:  # k == 0
        term1 = v
        term2 = v
    return np.ones_like(v) - term1 - term2

def _grad_desc(v0: np.ndarray, time_limit: float) -> tuple:
    """Projected gradient descent on the max-overlap objective."""
    start = time.time()
    v = v0.copy()
    n = len(v)
    best_v = v.copy()
    best_score = _compute_c5(v)
    eta = 0.2
    while time.time() - start < time_limit:
        corr = _full_corr(v)
        idx = int(np.argmax(corr))
        k = idx - (n - 1)
        grad = _gradient_for_shift(v, k)
        v_new = v - eta * grad
        v_new = _repair_vector(v_new, n)
        score = _compute_c5(v_new)
        if score < best_score:
            best_score = score
            best_v = v_new.copy()
            eta = min(eta * 1.05, 1.0)
        else:
            eta = max(eta * 0.5, 1e-5)
        v = v_new
    return best_v, best_score

def _local_search(v0: np.ndarray, time_limit: float) -> tuple:
    """Simple hill‑climbing exchange of mass between two coordinates."""
    start = time.time()
    n = len(v0)
    best_v = v0.copy()
    best_score = _compute_c5(best_v)
    step = 0.1
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] = np.clip(cand[i] + delta, 0.0, 1.0)
        cand[j] = np.clip(cand[j] - delta, 0.0, 1.0)
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score = score
            best_v = cand
            step = min(step * 1.1, 0.2)
        else:
            step = max(step * 0.9, 0.01)
    return best_v, best_score

    """Optimize a step function for the Erdős C5 problem."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------ #
    # Initial construction (parent) if supplied
    try:

        parent = _random_feasible(n0)
    parent = _repair_vector(parent, n0)

    best_v = parent.copy()
    best_score = _compute_c5(best_v)
    best_n = n0

    # Resolution schedule (keep total size ≤ 800)
    resolutions = [n0]
    if n0 * 2 <= 800:
        resolutions.append(n0 * 2)
    if n0 * 4 <= 800:
        resolutions.append(n0 * 4)

    total_budget = max(0.0, budget_s - 5.0)  # reserve a few seconds for final steps
    per_res = total_budget / len(resolutions) if resolutions else 0.0

    for n in resolutions:
        if time.time() - start >= budget_s - 2.0:
            break

        # Upsample if needed
        if len(best_v) != n:
            src_x = np.linspace(0.0, 2.0, len(best_v), endpoint=False)
            tgt_x = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(tgt_x, src_x, best_v, left=best_v[0], right=best_v[-1])
        else:
            init = best_v.copy()

        remaining = budget_s - (time.time() - start) - 2.0
        stage_time = min(per_res, remaining)
        if stage_time <= 0.0:
            break

        # Number of independent restarts (roughly one every ~3 s)
        restarts = max(1, int(stage_time // 3.0))
        time_per_restart = stage_time / restarts

        for r in range(restarts):
            if r == 0:
                v0 = init.copy()
            else:
                v0 = _random_feasible(n)

            # Gradient descent (majority of the allocated time)
            v_opt, sc_opt = _grad_desc(v0, time_per_restart * 0.8)
            if sc_opt < best_score:
                best_score, best_v, best_n = sc_opt, v_opt, n

            # Short local‑search refinement
            extra = time_per_restart * 0.2
            if extra > 0.15:
                v_ls, sc_ls = _local_search(best_v, extra)
                if sc_ls < best_score:
                    best_score, best_v = sc_ls, v_ls

    # Final polishing

    if remaining > 0.2:
        best_v, best_score = _local_search(best_v, remaining)

    best_v = _repair_vector(best_v, best_n)
    best_score = _compute_c5(best_v)

    return best_v.astype(np.float64), float(best_score), int(best_n)
