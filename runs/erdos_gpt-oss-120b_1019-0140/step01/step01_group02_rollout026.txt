def _crossover(p1, p2, rng):
    n = p1.shape[0]
    cut = rng.integers(1, n)  # 1 .. n-1
    child = np.empty(n, dtype=float)
    child[:cut] = p1[:cut]
    child[cut:] = p2[cut:]
    np.clip(child, 0.0, 1.0, out=child)

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------
def _compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry to enforce the exact sum constraint."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # choose a variable that can move in the needed direction
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute uniformly (unlikely)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return _repair_sum(h, target)
    i = idx[0]
    new_val = np.clip(h[i] + diff, 0.0, 1.0)
    actual = new_val - h[i]
    h[i] = new_val
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h


def _random_binary_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a binary (or half‑binary) feasible vector with sum n/2."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    if n % 2:
        remain = np.setdiff1d(np.arange(n), idx, assume_unique=True)
        h[rng.choice(remain)] = 0.5
    return h


def _tournament_select(c5_vals, tournament_size, rng):
    """Return index of the best individual among a random tournament."""
    ids = rng.integers(0, len(c5_vals), size=tournament_size)
    best_local = ids[0]
    best_val = c5_vals[best_local]
    for i in ids[1:]:
        v = c5_vals[i]
        if v < best_val:
            best_val = v
            best_local = i
    return best_local


def _crossover(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """One‑point crossover producing a child."""
    n = p1.shape[0]
    cut = rng.integers(1, n)  # 1 … n‑1
    child = np.empty(n, dtype=float)
    child[:cut] = p1[:cut]
    child[cut:] = p2[cut:]
    np.clip(child, 0.0, 1.0, out=child)
    return child


def _mutate(h: np.ndarray, rng: np.random.Generator, max_step: float):
    """Apply a few random pairwise moves that keep the sum unchanged."""
    n = h.shape[0]
    # number of elementary moves grows slowly with n
    moves = max(1, int(0.02 * n))
    for _ in range(moves):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        # feasible delta so both stay in [0,1]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        # intersect with max_step bound
        low = max(delta_min, -max_step)
        high = min(delta_max, max_step)
        if low >= high:
            continue
        delta = rng.random() * (high - low) + low
        h[i] = hi + delta
        h[j] = hj - delta
    # floating‑point drift may appear; repair to exact sum later by caller
    return h

        pop_c5.append(_compute_c5(h))

    # track global best
    best_idx = int(np.argmin(pop_c5))
    best_h = pop_h[best_idx].copy()
    best_c5 = pop_c5[best_idx]

    # ---- main evolutionary loop -----------------------------------
    target_c5 = 0.3808  # stop early if reached
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s:
            break
        # linear schedules
        frac = elapsed / budget_s
        cur_step = step_start * (1.0 - frac) + step_end * frac
        cur_mut_rate = mut_rate_start * (1.0 - frac) + mut_rate_end * frac

        # parent selection
        p1_idx = _tournament_select(pop_c5, tournament_sz, rng)
        p2_idx = _tournament_select(pop_c5, tournament_sz, rng)

        p2 = pop_h[p2_idx]

        if rng.random() < crossover_rate:
            child = _crossover(p1, p2, rng)
        else:
            child = p1.copy()

        # mutation
        if rng.random() < cur_mut_rate:
            _mutate(child, rng, cur_step)

        # repair (tiny drift may appear)
        child = _repair_sum(child, target_sum)

        # evaluation
        child_c5 = _compute_c5(child)

        # replace worst if improvement
        worst_idx = int(np.argmax(pop_c5))
        if child_c5 < pop_c5[worst_idx]:

            pop_c5[worst_idx] = child_c5

        # update global best
        if child_c5 < best_c5:
            best_c5 = child_c5
            best_h = child.copy()
            # early exit if we beat the target bound
            if best_c5 <= target_c5:
                break

        # occasional random injection to maintain diversity
        if rng.random() < 0.02:
            rand_h = _random_binary_feasible(n, rng)
            rand_c5 = _compute_c5(rand_h)
            w_idx = int(np.argmax(pop_c5))
            if rand_c5 < pop_c5[w_idx]:
                pop_h[w_idx] = rand_h
                pop_c5[w_idx] = rand_c5

    # final repair and evaluation
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
