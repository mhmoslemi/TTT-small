    n = len(v)
    g = np.zeros(n)
    # term1: (1 - v[i+shift]) for i where i+shift<n
    if shift >= 0:
        # for i in [0, n-1-shift]:
        i = np.arange(n - shift)
        g[i] += 1 - v[i + shift]
        # term2: -v[i - shift] for i where i-shift >=0 => i ∈ [shift, n-1]
        j = np.arange(shift, n)
        g[j] -= v[j - shift]
    else:
        # shift negative => shift = -s, s>0.
        s = -shift
        # term1: (1 - v[i - s]) for i where i-s>=0 => i ∈ [s, n-1]

        j = np.arange(n - s)
        g[j] -= v[j + s]
    return g

            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for a discrete step function."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _gradient_of_shift(v, shift):
    """Gradient of the correlation for a given integer shift."""
    n = len(v)
    g = np.zeros_like(v)
    if shift >= 0:
        if shift < n:
            g[: n - shift] += 1.0 - v[shift:]
        g[shift:] -= v[: n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - v[: n - s]
            g[: n - s] -= v[s:]
    return g

def _upsample_vector(v, n_target):
    """Linear interpolation to a finer grid followed by repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Random vector projected onto the feasible polytope."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _local_opt(v, n, time_end):
    """Hybrid gradient / random-exchange search until time_end."""
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    cur_v = v.copy()
    cur_score = best_score
    step_max = 0.08
    total_phase = max(time_end - time.time(), 1e-9)  # allocated duration
    start_phase = time.time()
    while time.time() < time_end:
        if np.random.rand() < 0.6:
            # gradient move on current worst shift
            corr = np.correlate(cur_v, 1.0 - cur_v, mode="full")
            idx = np.argmax(corr)
            shift = idx - (n - 1)
            grad = _gradient_of_shift(cur_v, shift)
            maxabs = np.max(np.abs(grad))
            if maxabs < 1e-12:
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 2 * step_max
                cand = cur_v.copy()
                cand[i] += delta
                cand[j] -= delta
            else:
                direction = -grad / maxabs
                elapsed = time.time() - start_phase
                step = step_max * (1.0 - elapsed / total_phase)
                cand = cur_v + step * direction
        else:
            # simple exchange move
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 2 * step_max
            cand = cur_v.copy()
            cand[i] += delta
            cand[j] -= delta

        # occasional jitter around the best found so far
        if np.random.rand() < 0.02:
            cand = best_v + np.random.randn(n) * 0.01

        cand = _repair_vector(cand, n)
        cand_score = _compute_c5(cand)

        if cand_score < cur_score:
            cur_v = cand
            cur_score = cand_score
        if cand_score < best_score:
            best_score = cand_score
            best_v = cand.copy()
    return best_v, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low C5 bound."""
    np.random.seed(seed)
    start = time.time()
    # Acquire parent construction if it exists.
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        n0 = len(parent)
    except Exception:
        n0 = 190
        parent = np.zeros(n0, dtype=np.float64)
        parent[: n0 // 2] = 1.0
    parent = _repair_vector(parent, n0)

    best_v = parent.copy()
    best_score = _compute_c5(best_v)

    # Determine resolution schedule (stay ≤800 points for speed).
    resolutions = [n0]
    if n0 * 2 <= 800:
        resolutions.append(n0 * 2)
    if n0 * 4 <= 800:
        resolutions.append(n0 * 4)

    total_budget = max(0.0, budget_s - 0.5)  # safety margin
    # Give larger weight to finer grids.
    weights = [1.0] * len(resolutions)
    if len(weights) >= 1:
        weights[-1] *= 2.0
    if len(weights) >= 2:
        weights[-2] *= 1.5
    weight_sum = sum(weights)

    for idx, n in enumerate(resolutions):
        elapsed = time.time() - start
        remaining = total_budget - elapsed
        if remaining <= 0.2:
            break
        # Proportional time share.
        time_alloc = remaining * (weights[idx] / weight_sum)
        # Prepare initial guess for this resolution.
        if len(best_v) != n:
            init = _upsample_vector(best_v, n)
        else:
            init = best_v.copy()
        # Optimize locally within allocated window.
        time_end = time.time() + time_alloc
        opt_v, opt_score = _local_opt(init, n, time_end)
        if opt_score < best_score:
            best_score = opt_score
            best_v = opt_v.copy()
        weight_sum -= weights[idx]
        if weight_sum <= 0:
            break

    # Optional final SLSQP refinement if time permits and SciPy is available.
    try:
        from scipy.optimize import minimize
        elapsed = time.time() - start
        remaining = budget_s - elapsed - 0.2
        if remaining > 1.0:
            n = len(best_v)
            cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
            bounds = [(0.0, 1.0)] * n
            res = minimize(
                lambda x: _compute_c5(x),
                best_v,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n)
                sc = _compute_c5(v)
                if sc < best_score:
                    best_score = sc
                    best_v = v.copy()
    except Exception:
        pass

    best_v = _repair_vector(best_v, len(best_v))
    final_score = _compute_c5(best_v)
    return best_v.astype(np.float64), float(final_score), int(len(best_v))
