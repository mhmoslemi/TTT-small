def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5 # safety

    # Determine n_points
    # Try to use initial_h if available; else default to 200
    try:
        init_h = np.array(initial_h_values, dtype=float)
        base_n = init_h.size
        # Choose n_points = max(base_n, 150) maybe; We'll upsample to 200.
        n_points = 200
        # Upsample initial to target n_points
        # Interpolation
        old_x = np.linspace(0, 2, base_n, endpoint=False)
        new_x = np.linspace(0, 2, n_points, endpoint=False)
        h0 = np.interp(new_x, old_x, init_h)
    except Exception:
        n_points = 200
        h0 = None

    target_sum = n_points / 2.0
    if h0 is not None:
        # Clip and adjust sum
        np.clip(h0, 0, 1, out=h0)
        h0 = adjust_sum_to_target(h0, target_sum, rng)
    else:
        h0 = random_h(n_points, target_sum, rng)

    # Initialize GA
    population_size = 30
    elite_size = 5
    population = [h0.copy()]
    for _ in range(population_size - 1):
        individual = random_h(n_points, target_sum, rng)
        population.append(individual)

    # Evaluate:
    scores = [compute_c5(ind) for ind in population]

    # Keep best
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # GA loop
    # Determine generation count based on time: iterate while time left > some threshold
    # We'll use while time.time() - start < max_time * 0.5 for GA (half of budget)
    # Then local refinement for remaining time.

    # Set parameters:
    mutation_rate = 0.2
    mutation_scale = 0.2  # initial amplitude
    # We'll apply linear cooling for mutation_scale during GA.

    while True:
        now = time.time()
        if now - start >= max_time * 0.5:
            break
        progress = (now - start) / (max_time * 0.5)
        # linearly reduce mutation scale
        cur_mut_scale = mutation_scale * (1 - progress) + 0.01 * progress

        # Selection: tournament of size 3
        def tournament():
            idxs = rng.choice(population_size, size=3, replace=False)
            # pick best among them (lowest score)
            best_of = min(idxs, key=lambda i: scores[i])
            return population[best_of]

        parent1 = tournament()
        parent2 = tournament()
        # Crossover: uniform
        mask = rng.rand(n_points) < 0.5
        child = np.where(mask, parent1, parent2).astype(float)
        # Mutation: add Gaussian noise to random subset
        if rng.rand() < 0.8:
            # mutate each component with prob mutation_rate
            mutate_mask = rng.rand(n_points) < mutation_rate
            # Gaussian noise
            noise = rng.randn(n_points) * cur_mut_scale
            child[mutate_mask] += noise[mutate_mask]
        # Clip and adjust sum
        np.clip(child, 0., 1., out=child)
        child = adjust_sum_to_target(child, target_sum, rng)

        child_score = compute_c5(child)
        # If child better than worst, replace worst
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            # Check if new best
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()

    # After GA, local refinement (SA/hill climbing) on best_h
    # We'll reuse similar structure as previous but maybe enhance parameters.

    current_h = best_h.copy()
    current_score = best_score

    # SA parameters
    T0 = 0.02
    T_end = 1e-5
    step0 = 0.3
    step_end = 1e-4

    while True:
        now = time.time()
        if now - start >= max_time:
            break
        # Compute progress relative to remaining time fraction
        total_elapsed = now - start
        remaining = max_time - total_elapsed
        # Use linear schedule based on elapsed fraction of remaining budget (or full)
        progress = total_elapsed / max_time
        T = T0 * (1 - progress) + T_end * progress
        step = step0 * (1 - progress) + step_end * progress

        # Choose move: 70% pairwise exchange, 30% gradient
        if rng.rand() < 0.3:
            # gradient move using worst lag
            corr_full = np.correlate(current_h, 1.0 - current_h, mode='full')
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n_points - 1)

            # Compute gradient exactly: g[t] = 1 - (h[t+k] if valid) - (h[t-k] if valid)
            g = np.ones_like(current_h)
            idx = np.arange(n_points)
            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n_points)
            g[valid_plus] -= current_h[plus[valid_plus]]
            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n_points)
            g[valid_minus] -= current_h[minus[valid_minus]]
            # Ensure sum(g) = 0? Actually sum(g) = n_points - 2 * sum_i h[i] *? Wait sum(g) = n_points - 2 * sum_{i valid} ... but we need to keep sum constraint.
            # Project to zero-mean
            g -= g.mean()
            candidate = current_h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        else:
            # pairwise exchange
            i, j = rng.randint(n_points, size=2)
            if i == j:
                continue
            hi, hj = current_h[i], current_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:

            delta = (rng.rand() * 2.0 - 1.0) * step
            delta = np.clip(delta, low, high)
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        cand_score = compute_c5(candidate)

        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / T) if T > 0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
