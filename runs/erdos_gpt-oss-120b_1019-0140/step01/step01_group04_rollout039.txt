def targeted_move(v, k0, max_delta):
    n = len(v)
    if k0 == 0:
        # no meaningful shift, fallback random move
        return None
    g = np.zeros(n, dtype=v.dtype)
    # slice update
    g[:n - k0] += v[k0:]
    g[k0:] += v[:n - k0]
    # Determine i_max (max g where v <1), i_min (min g where v>0)
    # Use masked array to avoid picking indices where bound prevents movement
    # Actually we can compute masks:
    inc_mask = v < 1 - 1e-12
    dec_mask = v > 1e-12
    # Set g where inc not possible to -inf for max, dec not possible to +inf for min
    # Use sentinel values
    g_max = np.where(inc_mask, g, -np.inf)
    i_max = np.argmax(g_max)
    g_min = np.where(dec_mask, g, np.inf)
    i_min = np.argmin(g_min)
    # Check max allowable delta
    delta_possible = min(max_delta, 1 - v[i_max], v[i_min])
    if delta_possible <= 0:
        return None
    delta = np.random.rand() * delta_possible
    new_v = v.copy()
    new_v[i_max] += delta
    new_v[i_min] -= delta
    return new_v

def _repair_vector(v, n, max_iter=20):
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    # adjust sum via projection
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # Identify free variables: those not at bound opposite to diff direction
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):

    corr = np.correlate(v, v, mode='full')
    overlap = corr[n-1:]  # overlaps for shift 0..n-1
    min_overlap = np.min(overlap)  # shift 0 gives max, not min

def _targeted_move(v, k, max_delta):
    """Move mass from low‑gradient to high‑gradient coordinate for shift k."""
    n = len(v)
    if k == 0 or k >= n:
        return None
    # gradient of overlap_k w.r.t v_i is v[i+k] + v[i-k]
    g = np.zeros(n, dtype=v.dtype)
    # forward contribution
    g[:n - k] += v[k:]
    # backward contribution
    g[k:] += v[:n - k]
    # masks for feasible increase / decrease
    inc_mask = v < 1.0 - 1e-12
    dec_mask = v > 1e-12
    if not inc_mask.any() or not dec_mask.any():
        return None
    # pick i_max and i_min
    g_max = np.where(inc_mask, g, -np.inf)
    i_max = int(np.argmax(g_max))
    g_min = np.where(dec_mask, g, np.inf)
    i_min = int(np.argmin(g_min))
    # feasible amount
    delta = max_delta
    delta = min(delta, 1.0 - v[i_max], v[i_min])

    delta = np.random.rand() * delta
    new_v = v.copy()
    new_v[i_max] += delta
    new_v[i_min] -= delta
    return new_v

import time
import numpy as np
from scipy.optimize import minimize

def _repair_vector(v, n, max_iter=20):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) > 1e-8:
            diff = target - s
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_from_overlap(v):
    """Fast C5 = 1 - min_overlap*dx using autocorrelation."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, v, mode='full')
    overlap = corr[n - 1 :]          # shifts 0 … n‑1
    min_ov = np.min(overlap)        # shift 0 has max value, not min
    return 1.0 - min_ov * dx

def _c5_direct(v):
    """Reference C5 using original definition (max correlation with 1‑v)."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def _targeted_move(v, k, max_delta):
    """Exchange mass from a low‑gradient to a high‑gradient index for shift k."""
    n = len(v)
    if k <= 0 or k >= n:
        return None
    g = np.zeros(n, dtype=v.dtype)
    g[: n - k] += v[k:]                 # forward term
    g[k:] += v[: n - k]                 # backward term
    inc_mask = v < 1.0 - 1e-12
    dec_mask = v > 1e-12
    if not inc_mask.any() or not dec_mask.any():
        return None
    g_max = np.where(inc_mask, g, -np.inf)
    i_max = int(np.argmax(g_max))
    g_min = np.where(dec_mask, g, np.inf)
    i_min = int(np.argmin(g_min))
    delta = min(max_delta, 1.0 - v[i_max], v[i_min])
    if delta <= 0:
        return None
    delta *= np.random.rand()
    nv = v.copy()
    nv[i_max] += delta
    nv[i_min] -= delta
    return nv
